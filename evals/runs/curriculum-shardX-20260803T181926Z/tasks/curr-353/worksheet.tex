\wstitleblock{Dividing Complex Numbers with Conjugates}{Algebra 2}{}

\begin{center}
\textit{Multiplying a complex number by its conjugate, dividing by a pure
imaginary number, dividing by a binomial denominator, and using complex
division to find impedance and current in an AC circuit}
\end{center}

\noindent\small
\textbf{The one move.} A quotient of complex numbers is not in standard form
until the denominator is real. Multiply the numerator and the denominator by
the \emph{conjugate} of the denominator, which changes the sign of its
imaginary part only.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item The conjugate of $a + bi$ is $a - bi$, and
        $(a+bi)(a-bi) = a^2 + b^2$ --- always a real number, never negative.
  \item Multiplying the top and the bottom by the same number does not change
        the value of the quotient; it only changes how it is written.
  \item Finish by dividing \emph{both} parts of the numerator by that real
        denominator and writing the answer as $a + bi$.
  \item In circuit work, voltage $V$, current $I$ and impedance $Z$ are complex
        and obey $V = I \cdot Z$, so a division recovers whichever one is
        missing. Voltage is in volts, current in amps, impedance in ohms.
\end{itemize}
\normalsize

\problem[4cm]{Write the conjugate of $7 + 2i$, then multiply $7 + 2i$ by that
conjugate. Give the product.}

\problem[4.5cm]{Write in standard form: \quad $\dfrac{6 + 4i}{2i}$
\par\smallskip
The denominator is pure imaginary, so its conjugate is $-2i$.}

\problem[5cm]{Write in standard form: \quad $\dfrac{5 + 5i}{2 + i}$}

\problem[5cm]{Write in standard form: \quad $\dfrac{11 + 10i}{4 - i}$}

\problem{An AC circuit carries a current of $3 + 2i$ amps while the voltage
across it is $2 + 23i$ volts. Impedance is defined by $Z = V / I$. Find the
impedance of the circuit in ohms, written in standard form.
\par\vspace*{6cm}\answerline{ohms}}

\problem[5.5cm]{Write in standard form: \quad $\dfrac{11 - 10i}{3 + 2i}$
\par\smallskip
Priya divided the real parts and the imaginary parts separately. Say in one
sentence why that method does not work here, then do it correctly.}

\problem[5.5cm]{Write in standard form: \quad $\dfrac{9 - 6i}{3i}$}

\problem[5.5cm]{Write in standard form: \quad $\dfrac{25}{3 + 4i}$
\par\smallskip
The numerator is already real. State what the denominator times its conjugate
equals before you simplify.}

\problem{A speaker with impedance $2 + i$ ohms is driven by a voltage of
$1 + 8i$ volts. Ohm's law for AC circuits says $V = I \cdot Z$. Find the
current $I$ through the speaker in amps, written in standard form.
\par\vspace*{6cm}\answerline{amps}}
