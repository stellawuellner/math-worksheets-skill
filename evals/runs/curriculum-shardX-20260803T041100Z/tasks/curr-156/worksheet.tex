\wstitleblock{Finding Unit Rates}{Grade 6--7 Ratios and Proportional Relationships}{}

\noindent\emph{Unit rates from two quantities, ratio tables, comparing rates, and applying a unit rate}

\vspace{0.15cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item Every problem comes with a ratio table. Fill in the ``per one'' column
        first --- that column \emph{is} the unit rate.
  \item Write the division you used beside the table, then write the rate with
        its units on the answer line.
  \item A unit rate always answers the question ``how much for exactly one?''
\end{itemize}

\problem{A box of 8 granola bars costs 24 dollars. What is the cost per
bar?
\par
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Bars} & 8 & 1 \\[0.5cm]\hline
\textbf{Cost (dollars)} & 24 & \\[0.7cm]\hline
\end{tabular}
\end{center}
\par\vspace*{3.5cm}\par\vspace*{0pt}\answerline{dollars per bar}}
