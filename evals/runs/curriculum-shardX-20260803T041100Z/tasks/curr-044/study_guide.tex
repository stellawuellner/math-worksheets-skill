\sstitleblock{Gaps and Overlaps When We Measure}

\skillheading{Skill 1 --- Lay the cubes with no gaps and no overlaps}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
Start at one end. Put the cubes \textbf{side by side}: touching, but never on top of each
other, and never with a space between. Then the \textbf{number of cubes is the length}.\par
\begin{center}\goodlay\ \small\textbf{good}\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pencil is covered by cubes laid end to end, with no gaps and no
overlaps. How long is the pencil?
\step{The cubes touch and none sit on top of another, so every cube covers new pencil ---
that means I may simply count the cubes.}
\step{Touch each cube once and count on without skipping or repeating; the last cube
counted is 6.}
$\Rightarrow\quad$ the pencil is $\ans{6}$ cubes long
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A spoon is covered by cubes laid end to end, no gaps and no
overlaps, and the last cube counted is 4. How long is the spoon?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Spot a gap or an overlap mistake}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
\textbf{Gaps} leave bare spaces, so too few cubes reach the end: the count is
\textbf{too small}.\par
\textbf{Overlaps} let cubes pile onto each other, so extra cubes squeeze on: the count is
\textbf{too big}.\par
\begin{center}\gaplay\ \small\textbf{gaps: too small}\qquad \overlay\ \small\textbf{overlaps: too big}\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Ben left gaps and counted 5. A careful measure gives 7. Write
$<$ or $>$: Ben's count \underline{\hspace{1.2cm}} real length.
\step{Ben left gaps, so I expect his number to be too small --- but I still check by
comparing the two counts.}
\step{Ben's count is 5 and the real length is 7; 5 comes first when we count, so it is the
smaller number, and the symbol opens toward the bigger one.}
$\Rightarrow\quad$ Ben's count $\ans{<}$ real length
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Lia let her cubes overlap and counted 9. A careful measure gives 6.
Write $<$ or $>$: Lia's count \underline{\hspace{1.2cm}} real length.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\skillheading{Skill 3 --- Fix the count after a gap or overlap}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
The object never changes. Only the lay-down does. To fix a count, take the
\textbf{careful} length and the \textbf{sloppy} count and find \textbf{how far apart}
they are --- count on from the smaller number to the bigger one, and that is how many
cubes the gaps hid or the overlaps added.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Ben's gaps: he counted 5, but the careful length is 7. How many
cubes did his gaps hide?
\step{``How many hidden'' asks how far apart the two numbers are, so this is a
difference, not a total.}
\step{Count on from 5 up to 7 --- that is two counts, so the difference is 2.}
$\Rightarrow\quad$ the gaps hid $\ans{2}$ cubes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Lia's overlaps: she counted 9, but the careful length is 6. How
many extra cubes did the overlapping add?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A \emph{bigger number} is not a \emph{better measurement}. Always
look at how the cubes were laid before you trust a count --- and never add the two counts
when a question asks ``how many more'' or ``how many hidden''.
\end{watchoutbox}
