\begin{center}
\textit{Adding, subtracting, and decomposing fractions with like denominators}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item The bar pictures show fractions of one whole. Use them to help you
        see what is happening, then write the matching number sentence.
  \item Give every fraction answer in simplest form.
\end{itemize}

$\tfrac{2}{6} + \tfrac{3}{6}$

\end{tikzpicture}

\problem[3cm]{Every fraction is a sum of unit fractions (fractions with a 1 on

\problem{Ella walks $\tfrac{3}{10}$ mi from home to the park, and then
$\tfrac{4}{10}$ mi more from the park to school. How far does she walk in all,
in miles?
\par\vspace*{4cm}\answerline{mi}}

\problem[4cm]{Find the missing fraction:
$\square + \tfrac{4}{10} = \tfrac{7}{10}$}

\problem[3.5cm]{Fill in the missing numerator:
$\tfrac{5}{6} = \tfrac{3}{6} + \tfrac{\square}{6}$}

\problem{Ravi mixes paint: $\tfrac{3}{8}$ L of red, $\tfrac{3}{8}$ L of blue,
and $\tfrac{1}{8}$ L of white. How many liters of paint does he make in all?
\par\vspace*{4.5cm}\answerline{L}}

\problem[4.5cm]{A garden has 12 equal sections. In all, $\tfrac{9}{12}$ of the
garden is planted, and beans fill $\tfrac{5}{12}$ of the garden. What fraction
of the garden is planted with something other than beans? On the lines below
your work, explain why the denominator stays 12 when you subtract.}

\problem[4.5cm]{\textbf{Challenge.} Maya's water bottle starts full. She
drinks $\tfrac{1}{8}$ of it in the morning and $\tfrac{5}{8}$ of it at lunch.
\par\vspace{2pt}
(a) What fraction of the bottle has she drunk? \ansblank
\par\vspace{2pt}
(b) What fraction of the bottle is left? \ansblank}
