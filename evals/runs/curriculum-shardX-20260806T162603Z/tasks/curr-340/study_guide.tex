\sstitleblock{Choosing the Right Circle Theorem}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Central and inscribed angles}

\begin{formulabox}
\textbf{Rules:} a central angle equals its arc. An inscribed angle is \emph{half} its intercepted arc, so two inscribed angles on the same arc are equal. Opposite angles of a quadrilateral inscribed in a circle are supplementary.\par
\textbf{Tell them apart:} the vertex is at the centre (central) or on the circle (inscribed).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Central angle $\angle POQ$ measures $110^\circ$. Find inscribed angle $\angle PRQ$ on the same arc.
\step{Both angles sit on arc $PQ$, and the vertex of the second one is on the circle --- so the inscribed-angle theorem applies and the answer is half, not equal.}
\step{$m\angle PRQ = 110^\circ / 2$}

$\Rightarrow\quad$ $\ans{55^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An inscribed angle measures $28^\circ$. Find the arc it intercepts.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56^\circ}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Chord, secant and tangent lengths}

\begin{formulabox}
\textbf{Two chords crossing inside:} the products of the two pieces are equal, $AE \cdot EB = CE \cdot ED$.\par
\textbf{Tangent and secant from an outside point:} $PT^2 = PA \cdot PB$, where $PA$ is the \emph{outer} part and $PB$ the \emph{whole} secant. Note the square --- it is the step most often skipped.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two chords cross inside a circle: one is cut into $5$ and $x$, the other into $3$ and $10$. Find $x$.
\step{The crossing point is \emph{inside} the circle, so this is the intersecting-chords rule --- multiply each chord's two pieces and set the products equal.}
\step{$5x = 3 \cdot 10 = 30$}

$\Rightarrow\quad x = $ $\ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From an outside point, a tangent of length $6$ and a secant with outer part $4$. Find the whole secant.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Arc length and sector area}

\begin{formulabox}
\textbf{Both are fractions of a whole circle,} and the fraction is the same: $\dfrac{\text{central angle}}{360^\circ}$.\par
arc length $= \dfrac{\theta}{360}\cdot 2\pi r$ \qquad sector area $= \dfrac{\theta}{360}\cdot \pi r^2$\par
Arc length carries one factor of $r$; sector area carries two. That is the only difference.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A sector has radius $6$ and central angle $60^\circ$. Find its arc length and its area.
\step{The angle is a fraction of a full turn, so both answers are that same fraction of the whole circumference and of the whole area --- one calculation set up twice.}
\step{arc $= \frac{60}{360}\cdot 2\pi(6) = 2\pi$ \quad and \quad area $= \frac{60}{360}\cdot \pi(6)^2 = 6\pi$}

$\Rightarrow\quad$ arc $\approx \ans{6.28}$ \quad area $\approx \ans{18.85}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the arc length for radius $15$ and central angle $48^\circ$, to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12.57}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- The circle's equation in a configuration}

\begin{formulabox}
\textbf{Centre-radius form:} $(x-h)^2 + (y-k)^2 = r^2$. If the equation shows loose $x$ and $y$ terms, complete both squares first --- adding each completing term to \emph{both} sides.\par
Once you have the centre and $r$, the configuration facts follow: a tangent point lies one radius from the centre, and a chord of length $c$ sits $\sqrt{r^2 - (c/2)^2}$ from the centre.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Put $x^2 + y^2 - 6x + 8y - 11 = 0$ into centre-radius form.
\step{The loose $-6x$ and $+8y$ terms cannot appear in centre-radius form, which is the signal to complete the square on each variable before reading anything off.}
\step{Half of $-6$ is $-3$ and half of $8$ is $4$: add $9$ and $16$ to both sides, giving $(x-3)^2 + (y+4)^2 = 11 + 9 + 16$}

$\Rightarrow\quad \ans{(x-3)^2 + (y+4)^2 = 36}$, so the centre is $(3, -4)$ and $r = 6$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Put $x^2 + y^2 + 2x - 10y + 10 = 0$ into centre-radius form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+1)^2 + (y-5)^2 = 16}$}
\end{tryitbox}
