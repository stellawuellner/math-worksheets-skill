\sstitleblock{Zeros, Multiplicity, and Graph Behavior}

\skillheading{1. Factor first, then read the zeros}

\begin{formulabox}
\textbf{Factored form is the graph's form.} A zero is a value that makes one
factor zero, so the zeros are invisible until the polynomial is factored.\\[3pt]
\textbf{Order of attack:} GCF $\rightarrow$ difference of squares
$\rightarrow$ trinomial $\rightarrow$ quadratic in $x^2$.\\[3pt]
Forgetting the GCF loses the zero at $x = 0$ --- the most common lost root.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Factor $x^3 - 9x$ and list its zeros.
\step{Every term has an $x$, so pull the GCF before anything else --- that keeps
the zero at the origin.}
\step{$x^3 - 9x = x(x^2 - 9)$, and $x^2 - 9$ is a difference of squares.}
\step{$\ans{x(x-3)(x+3)}$, so the zeros are $x = 0$, $x = 3$, $x = -3$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^3 - 2x^2 - 8x$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x(x-4)(x+2)}$}
\end{tryitbox}

\skillheading{2. Multiplicity: crossing, touching, flattening}

\begin{formulabox}
\textbf{Multiplicity} is the exponent on a factor. It decides the shape of the
curve at that zero:
\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item odd, equal to one --- passes straight through (crosses)
  \item even --- touches the axis and turns back, no sign change
  \item odd, greater than one --- crosses, but flattens as it passes
\end{itemize}
The multiplicities add up to the degree.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Describe the behavior of $y = (x-1)^2(x+4)$ at each zero.
\step{The polynomial is already factored, so read each zero and its exponent
straight off rather than expanding.}
\step{$(x-1)^2$ gives $x = 1$ with multiplicity $2$ (even, so it touches);
$(x+4)$ gives $x = -4$ with multiplicity $1$ (odd, so it crosses).}
\step{Zeros: $\ans{x = 1}$ (touch) and $\ans{x = -4}$ (cross); degree
$2 + 1 = 3$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Give the zeros of $y = (x+3)^2(x-2)$ and say which one
touches.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -3}$ (touch), $\ans{x = 2}$ (cross)}
\end{tryitbox}

\skillheading{3. End behavior from degree and leading coefficient}

\begin{formulabox}
\textbf{Only the leading term matters} once $|x|$ is large.\\[3pt]
\textbf{Even degree:} both ends agree --- both up if the leading coefficient is
positive, both down if it is negative.\\[3pt]
\textbf{Odd degree:} the ends disagree --- up on the right and down on the left
if the leading coefficient is positive, reversed if it is negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the end behavior of $y = -x^4 + 7x$ as
$x \to \infty$.
\step{End behavior is set by the leading term alone, so compare degree and sign
instead of evaluating the whole polynomial.}
\step{Degree $4$ is even, so both ends agree; the leading coefficient is
negative, so both ends fall.}
\step{As $x \to \infty$, $y \to \ans{-\infty}$ (and the left end falls too).}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $y = 5x^3 - 2x$, what happens as $x \to -\infty$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\infty}$}
\end{tryitbox}

\skillheading{4. Building a polynomial from its zeros}

\begin{formulabox}
\textbf{Run the reading backwards.} A zero at $x = r$ contributes the factor
$(x - r)$; how the graph behaves there sets the exponent.\\[3pt]
Crosses $\rightarrow$ exponent $1$. \quad Touches $\rightarrow$ exponent $2$.\\[3pt]
``Least degree'' means use the smallest exponent that gives the stated behavior,
then multiply out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Least-degree polynomial, leading coefficient $1$, that
touches at $x = -1$ and crosses at $x = 4$.
\step{Behavior fixes each exponent: touching needs an even power, crossing needs
an odd one, so take the smallest of each.}
\step{$(x+1)^2(x-4) = (x^2 + 2x + 1)(x - 4)$, and multiplying out gives
$x^3 - 2x^2 - 7x - 4$.}
\step{$\ans{x^3 - 2x^2 - 7x - 4}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Least degree, leading coefficient $1$, touching at
$x = -2$ and crossing at $x = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^3 + 3x^2 - 4}$}
\end{tryitbox}
