\wstitleblock{Wrong Centre, Wrong Vector, Wrong Scale Factor}{High School Geometry}{}

\begin{center}
\small\textit{Find the mistake, fix the number, and name what was misunderstood}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Translation vector from $P$ to $P'$: each component is the image coordinate \emph{minus} the original.
  \item Dilation about the origin: $(x, y) \to (kx,\ ky)$. About a centre $(h, j)$: $(x, y) \to \bigl(h + k(x - h),\ \ j + k(y - j)\bigr)$.
  \item Scale factor $= \dfrac{\text{image length}}{\text{original length}}$, and areas of similar figures scale by $k^2$.
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\problem{A translation maps $A(3, -2)$ to $A'(7, 1)$. Ana writes the rule as $(x, y) \to (x - 4,\ y - 3)$.
\par\vspace{0.15cm}
(a) Find the correct horizontal component of the translation. \quad \ansblank
\par\vspace*{2.2cm}
(b) Find the correct vertical component. \quad \ansblank
\par\vspace*{2.2cm}
(c) Explain what Ana did wrong, and say what her rule would do to $A$.
\par\vspace*{3.0cm}}

\problem{A dilation with centre the origin and scale factor $3$ should map $B(-2, 5)$ to $B'$. Ben reports $B'(1, 8)$, because he added $3$ to each coordinate.
\par\vspace{0.15cm}
(a) Write the correct coordinates of $B'$. \quad $B'\,($ \ansblank $,$ \ansblank $)$
\par\vspace*{2.4cm}
(b) Explain what Ben did wrong, and name the transformation he actually carried out.
\par\vspace*{3.0cm}}

\problem{Two similar triangles have corresponding sides of $9$ cm and $15$ cm. Cara says the scale factor from the smaller triangle to the larger one is $6$, because $15 - 9 = 6$.
\par\vspace{0.15cm}
(a) Find the correct scale factor, to the nearest hundredth. \quad \ansblank
\par\vspace*{2.4cm}
(b) Explain why a difference of two side lengths can never be a scale factor.
\par\vspace*{3.0cm}}

\problem{A dilation with centre $C(1, 4)$ and scale factor $2$ maps $P(5, 6)$ to $P'$. A student ignored the given centre, used the origin instead, and reported $P'(10, 12)$.
\par\vspace{0.15cm}
(a) Write the correct coordinates of $P'$. \quad $P'\,($ \ansblank $,$ \ansblank $)$
\par\vspace*{2.4cm}
(b) Explain what the centre of a dilation controls, and what goes wrong when the origin is used in its place.
\par\vspace*{3.0cm}}

\problem{Two similar rectangles have areas $32$ cm$^2$ and $72$ cm$^2$. A student says the linear scale factor from the smaller to the larger is $2.25$, because $72 \div 32 = 2.25$.
\par\vspace{0.15cm}
(a) Find the correct linear scale factor. \quad \ansblank
\par\vspace*{2.4cm}
(b) Explain why the ratio of the areas is not the linear scale factor.
\par\vspace*{3.0cm}}

\problem{Segment $\overline{MN}$ has endpoints $M(2, 3)$ and $N(6, 3)$. It is dilated by a scale factor of $3$ about the point $M$, so $M$ itself does not move. A student dilated about the origin instead and reported $M'(6, 9)$ and $N'(18, 9)$.
\par\vspace{0.15cm}
(a) A dilation about $M$ leaves the $y$-coordinate at $3$. Find the correct $x$-coordinate of $N'$. \quad \ansblank
\par\vspace*{2.4cm}
(b) Find the length of the correct image segment $\overline{M'N'}$. \quad \ansblank
\par\vspace*{2.4cm}
(c) The student's image segment also has the correct length. Explain why the length came out right while the position did not.
\par\vspace*{3.0cm}}
