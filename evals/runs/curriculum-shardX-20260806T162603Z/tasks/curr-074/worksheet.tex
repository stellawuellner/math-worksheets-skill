\wstitleblock{Division: Group Size or Group Count?}{Grade 3}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item A division answer counts \textbf{one} of two things: how many are \textbf{in each group} (the group size), or \textbf{how many groups} there are (the group count).
  \item Before you write an answer, ask: \emph{what does this number count?}
  \item Some problems show another student's thinking with a mistake in it. Find the mistake and fix it.
\end{itemize}

\problem[3cm]{Amara bakes 12 cookies and shares them equally into 3 gift bags. How many cookies go in each bag?}

\problem[3cm]{A coach has 15 tennis balls and packs 5 balls into each can. How many cans does the coach fill?}

\problem[3cm]{Dev puts 20 grapes equally onto 4 snack plates. How many grapes are on each plate?}

\problem{There are 24 crayons shared equally among 6 boxes. Jo says: ``Each box holds 6 crayons, because the 6 is right there in the problem.''
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] What went wrong in Jo's thinking?\par\vspace{6pt}\noindent\underline{\hspace{12.5cm}}\par\vspace{14pt}\noindent\underline{\hspace{12.5cm}}
  \item[(b)] How many crayons are really in each box?\par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[3cm]{A toy maker has 28 wheels. Each wagon needs 4 wheels. How many wagons can the toy maker build?}

\problem{Rin has 18 stickers and shares them equally between 2 friends.\par\vspace{4pt}How many stickers does each friend get? \ansblank\par\vspace{8pt}Now say what your answer counts: the number of friends, or the number of stickers each friend gets?\par\vspace{6pt}\noindent\underline{\hspace{12.5cm}}\par\vspace*{2.2cm}}

\problem{Ben has 30 marbles and puts 5 marbles in each pouch. He writes $30 \div 5 = 6$ and says: ``So each pouch holds 6 marbles.''
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] Name the mistake in what Ben said. (His division is correct!)\par\vspace{6pt}\noindent\underline{\hspace{12cm}}\par\vspace{14pt}\noindent\underline{\hspace{12cm}}
  \item[(b)] What does the 6 really count --- how many pouches does Ben fill?\par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem{\textbf{Challenge.} Write two different division stories for $24 \div 4$.
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] A story where the answer tells how many are \textbf{in each group}.\par\vspace{6pt}\noindent\underline{\hspace{12cm}}\par\vspace{14pt}\noindent\underline{\hspace{12cm}}
  \item[(b)] A story where the answer tells \textbf{how many groups} there are.\par\vspace{6pt}\noindent\underline{\hspace{12cm}}\par\vspace{14pt}\noindent\underline{\hspace{12cm}}
  \item[(c)] What is $24 \div 4$? \ansblank\par\vspace*{1.6cm}
\end{itemize}}
