\sstitleblock{Distributions, Models, and Qualified Inference}

\skillheading{Skill 1 --- Summarising a distribution}

\begin{formulabox}
\textbf{Two summaries, two jobs.} The \textbf{mean} uses every value and moves when an
extreme value does; the \textbf{median} uses position and does not.\par
\smallskip
\textbf{Median} of an even-sized ordered list $=$ the mean of the two middle values.
\ \textbf{Quartiles} (school convention): split the ordered list into halves and take the
median of each; $\text{IQR} = Q_3 - Q_1$.\par
\smallskip
\textbf{Reading shape from the pair:} mean above median $\Rightarrow$ right-skewed; mean
below median $\Rightarrow$ left-skewed; roughly equal $\Rightarrow$ roughly symmetric.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the median of $4,\ 7,\ 9,\ 12,\ 15,\ 20$.
\step{The list is even-sized, so no single value is in the middle --- the median is defined
as the mean of the two central values, and the list is already ordered.}
\step{The two central values are $9$ and $12$, so the median is $(9 + 12) \div 2$.}
$\Rightarrow\quad \text{median} = \ans{10.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the median of $3,\ 5,\ 8,\ 11,\ 14,\ 20$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9.5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Comparing two models with residuals}

\begin{formulabox}
\textbf{Residual} $=$ observed $-$ predicted, always in that order.\par
\smallskip
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Sign} says direction: positive $\Rightarrow$ the model under-predicted
        there; negative $\Rightarrow$ it over-predicted.
  \item \textbf{Size} says accuracy. Compare two models by comparing $|$residual$|$, or by
        summing squared residuals over all points.
  \item \textbf{Pattern} says whether the model has the right \emph{shape}. Signs scattered
        at random $\Rightarrow$ right shape. All one sign, or a run of signs that flips
        across the range $\Rightarrow$ wrong shape, however small the misses are.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A model predicts $y = 5x + 3$. At $x = 4$ the observed value is
$22$. Find the residual and read it.
\step{Comparing a model to data means residuals, so evaluate the model at that input and
subtract it from the observed value in that order.}
\step{Predicted $= 5(4) + 3$, so the residual is $22 - (5(4) + 3) = -1$, and the negative
sign says the observation fell below the model.}
$\Rightarrow\quad \text{residual} = \ans{-1}$, the model over-predicted
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A model predicts $y = 6x + 4$. At $x = 3$ the observed value is
$19$. Find the residual.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Qualifying an inference}

\begin{formulabox}
\textbf{Before stating a conclusion, answer all four:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Range.} Is the input inside the observed data? Inside is interpolation;
        outside is an assumption.
  \item \textbf{Plausibility.} Does the situation itself limit the output --- a percentage
        above 100, a culture that must run out of food?
  \item \textbf{Bias.} Do the residuals lean one way? A model can be accurate and still
        systematically low.
  \item \textbf{Sample.} How many independent runs or subjects? One run gives no
        variability, so no margin of error can be attached.
\end{itemize}
\textbf{Probability from counts:} $P = \dfrac{\text{favourable}}{\text{total}}$, in lowest
terms. Count carefully: a residual of exactly zero is neither positive nor negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Of $5$ residuals, $3$ are positive. One of the five points is chosen
at random. Find the probability it has a positive residual.
\step{All points are equally likely, so the probability is the count of favourable points
over the total count --- no model arithmetic is needed.}
\step{Three of the five qualify, so $P = 3 \div 5$, already in lowest terms.}
$\Rightarrow\quad P = \ans{\tfrac{3}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $8$ residuals, $2$ are positive. One of the eight points is
chosen at random. Find the probability it has a positive residual.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{4}}$}
\end{tryitbox}
