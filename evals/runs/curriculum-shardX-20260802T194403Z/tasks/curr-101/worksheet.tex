\wstitleblock{Reading and Comparing Multi-Digit Whole Numbers}{Grades 4--5 Number and Operations in Base Ten}{}

\noindent\textbf{Directions:} Read each number out loud before you answer. The place-value
charts are there to help you line digits up --- use them. Show your thinking in the space
under each problem, then write your final answer on the answer line.\par\vspace{0.15cm}
\noindent{\small\textit{This sheet practices: Digit values, expanded and word form,
comparing and ordering, rounding to estimate.}}

\problem{The place-value chart below shows one whole number.
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
\textbf{Ten thousands} & \textbf{Thousands} & \textbf{Hundreds} & \textbf{Tens} & \textbf{Ones}\\\hline
\rule{0pt}{0.9cm}{\Large 4} & {\Large 7} & {\Large 3} & {\Large 8} & {\Large 2}\\\hline
\end{tabular}
\end{center}
\begin{enumerate}[label=(\alph*), itemsep=2.6cm, leftmargin=1.2cm]
  \item Write the number in standard form. \ansline
  \item What is the \textbf{value} of the digit 7 in that number? (Not just ``7'' --- what is
        the 7 worth?) \ansline
\end{enumerate}\vspace*{0.2cm}}

\problem[4cm]{The number $44{,}000$ is written with two 4s. The 4 on the left is worth
$40{,}000$. The 4 beside it is worth $4{,}000$.\par\vspace{0.1cm}
How many times as much is the left 4 worth as the 4 next to it?}

\problem[4cm]{A number has been broken into the value of each of its digits. Put it back
together and write it in standard form.\par\vspace{0.15cm}
\hspace*{0.8cm}$500{,}000 \; + \; 30{,}000 \; + \; 8{,}000 \; + \; 60 \; + \; 2$}

\problem[3.5cm]{Compare the two numbers below. Write $<$ or $>$ in the blank. The chart
lines the digits up for you --- start at the left and find the first place where they differ.
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
\textbf{Ten thousands} & \textbf{Thousands} & \textbf{Hundreds} & \textbf{Tens} & \textbf{Ones}\\\hline
\rule{0pt}{0.7cm}4 & 6 & 3 & 1 & 5\\\hline
\rule{0pt}{0.7cm}4 & 6 & 3 & 5 & 1\\\hline
\end{tabular}
\end{center}
\hspace*{1.5cm}$46{,}315$ \quad \underline{\hspace{1.4cm}} \quad $46{,}351$}

\problem[4cm]{Write this number in standard form:\par\vspace{0.15cm}
\hspace*{0.8cm}\textbf{two hundred nine thousand, forty}}

\problem[4cm]{Order these three numbers from \textbf{least to greatest}. Write them on the
answer line with $<$ signs between them.\par\vspace{0.2cm}
\hspace*{0.8cm}$80{,}912$ \qquad $8{,}921$ \qquad $80{,}921$}

\problem[4cm]{Round $47{,}382$ to the nearest thousand. The table shows the two thousands
this number sits between, with the halfway point in the middle.
\begin{center}\small
\begin{tabular}{|c|c|c|}\hline
\textbf{Thousand below} & \textbf{Halfway point} & \textbf{Thousand above}\\\hline
\rule{0pt}{0.7cm}$47{,}000$ & $47{,}500$ & $48{,}000$\\\hline
\end{tabular}
\end{center}}

\problem[4cm]{Mia looks at the number $350{,}842$ and says, ``The value of the digit 5 is
$500{,}000$.'' Mia has the wrong place. Write the \textbf{correct} value of the digit 5.}

\problem[4.5cm]{A town library lent $18{,}472$ books in June and $21{,}650$ books in July.
Estimate the total number of books lent in the two months by first rounding each number to
the nearest thousand, then adding.}

\problem{Two towns report their populations: Ashford has $34{,}650$ people and Belmont has
$34{,}509$ people.
\begin{enumerate}[label=(\alph*), itemsep=2.7cm, leftmargin=1.2cm]
  \item Round Ashford's population to the nearest thousand. \ansline
  \item Rounding hides the difference: both towns round to the same thousand. Which town
        actually has more people? Write the two populations with $>$ between them. \ansline
\end{enumerate}\vspace*{0.2cm}}
