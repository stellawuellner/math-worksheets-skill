\wstitleblock{The Derivative as a Limit and as a Rate}{AP Calculus AB/BC}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item An \emph{average} rate over an interval is a slope between two points: $\dfrac{f(b)-f(a)}{b-a}$. An \emph{instantaneous} rate at a point is what that slope becomes as the second point slides in: $f'(a) = \displaystyle\lim_{h \to 0}\frac{f(a+h)-f(a)}{h}$.
  \item Where a problem says ``by the limit definition'', write the difference quotient, simplify until the $h$ in the denominator cancels, and only then let $h \to 0$.
  \item Keep the units of a rate attached: output units per input unit.
\end{itemize}

\problem{The position of a cart on a track is
$s(t) = t^2 + 3t$ metres at time $t$ seconds. The table shows four of its
positions.
\par\vspace{0.2cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$t$ (s) & $1$ & $2$ & $3$ & $4$ \\ \hline
$s(t)$ (m) & $4$ & $10$ & $18$ & $28$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.15cm}
Find the cart's \textbf{average} velocity from $t = 1$ to $t = 3$ seconds.
\par\vspace*{4.2cm}\answerline{m/s}}

\problem[4.6cm]{Let $f(x) = x^2$. Use the limit definition
\[
  f'(3) = \lim_{h \to 0} \frac{f(3+h) - f(3)}{h}
\]
to find the slope of the tangent line to $y = x^2$ at $x = 3$. Expand, cancel
the $h$, then let $h \to 0$.}

\problem{Same cart as problem 1: $s(t) = t^2 + 3t$ metres. Now find the
\textbf{instantaneous} velocity at $t = 3$ seconds by evaluating
\[
  \lim_{h \to 0} \frac{s(3+h) - s(3)}{h} .
\]
Compare your answer with problem 1 --- one of them is a slope between two
points on the curve, the other is the slope of the curve itself.
\par\vspace*{4.6cm}\answerline{m/s}}

\problem[5.6cm]{Let $f(x) = \dfrac{1}{x}$. Use the limit definition to find
$f'(2)$. (Combine the two fractions in the numerator over a common denominator
before you divide by $h$.)}

\problem[5cm]{A tank is draining. Its volume is $V(t) = 60 - 2t^2$ litres at
time $t$ minutes, for $0 \le t \le 5$.
\par\vspace{0.15cm}
\textbf{(a)} Use the limit definition to find $V'(5)$: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(b)} Explain what that number says about the tank at $t = 5$ minutes.
Give its units and say what the sign means.}

\problem[5.6cm]{Let $f(x) = \sqrt{x}$. Use the limit definition to find $f'(9)$.
(Multiply the difference quotient by the conjugate of its numerator.) Give the
exact value.}

\problem[5cm]{Let $g(x) = x^3 - 2x$.
\par\vspace{0.15cm}
\textbf{(a)} Find the derivative function $g'(x)$: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(b)} Use it to find the slope of the tangent line to $y = g(x)$ at
$x = 2$: \quad \ansblank}

\problem[5.4cm]{\textbf{Challenge.} A ball is thrown straight up. Its height is
$h(t) = 40t - 5t^2$ metres at time $t$ seconds.
\par\vspace{0.15cm}
\textbf{(a)} Find the velocity function $h'(t)$: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(b)} Find the velocity at $t = 2$ seconds, in metres per second: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(c)} Compute the velocity at $t = 5$ seconds and explain, in one
sentence, what it says about the ball at that instant.}
