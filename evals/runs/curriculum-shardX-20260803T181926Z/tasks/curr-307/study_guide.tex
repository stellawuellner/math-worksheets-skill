\begin{formulabox}
\textbf{With $\ell \parallel m$, every angle pair is one of two kinds.}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item \textbf{Congruent (equal):} corresponding ($a$ and $e$), alternate
interior ($c$ and $e$), alternate exterior ($a$ and $g$), and vertical angles
($a$ and $c$).
\item \textbf{Supplementary (sum a straight angle):} any linear pair ($a$ and
$b$), and same-side interior angles ($d$ and $e$).
\end{itemize}
Deciding which kind you have is the whole problem. Everything after it is
arithmetic.
\end{formulabox}

\skillheading{1. Find an angle from one you already know}

\begin{formulabox}
Name the pair first, then act: congruent means copy the measure across;
supplementary means subtract from a straight angle. When no single pair links
the two angles, go through a third angle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $m\angle a = 130^\circ$. Find $m\angle e$ and $m\angle d$.
\step{$\angle e$ sits at line $m$ where $\angle a$ sits at line $\ell$, so
they correspond and the parallel lines make them congruent.}
\step{So $m\angle e$ copies across. For $\angle d$, note that $\angle a$ and
$\angle d$ are a linear pair, so $180 - 130 = 50$.}

$m\angle e = \ans{130}^\circ$ and $m\angle d = \ans{50}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $m\angle c = 68^\circ$. Find $m\angle g$ and $m\angle f$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m\angle g = \ans{68}^\circ$, $m\angle f
= \ans{112}^\circ$}
\end{tryitbox}

\skillheading{2. Solve for x when the angles are equal}

\begin{formulabox}
Corresponding, alternate interior, alternate exterior and vertical pairs are
congruent, so set the two expressions equal and solve. Then substitute back:
the question almost always wants the angle, not $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $m\angle b = (2x + 25)^\circ$ and $m\angle f = (4x - 5)^\circ$.
Find $x$ and $m\angle b$.
\step{$\angle b$ and $\angle f$ are corresponding angles on parallel lines, so
they are congruent --- set the expressions equal rather than adding them.}
\step{$2x + 25 = 4x - 5$ gives $25 + 5 = 4x - 2x$, so $x = 15$; then
$2(15) + 25 = 55$.}

$x = \ans{15}$ and $m\angle b = \ans{55}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $m\angle d = (6x - 7)^\circ$ and $m\angle f = (3x + 20)^\circ$
are alternate interior angles. Find $x$ and $m\angle d$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{9}$, $m\angle d = \ans{47}^\circ$}
\end{tryitbox}

\skillheading{3. Solve for x when the angles add to 180}

\begin{formulabox}
A linear pair and a same-side interior pair are supplementary, so add the two
expressions and set the sum to a straight angle. Same finish as before:
substitute $x$ back to get the measure.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $m\angle d = (5x + 10)^\circ$ and $m\angle e = (3x + 2)^\circ$.
Find $x$ and $m\angle d$.
\step{$\angle d$ and $\angle e$ are same-side interior angles, the one
parallel-line pair that is supplementary rather than congruent --- so add,
do not equate.}
\step{$(5x + 10) + (3x + 2) = 180$, and solving gives $x = 21$; then
$5(21) + 10 = 115$.}

$x = \ans{21}$ and $m\angle d = \ans{115}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $m\angle c = (7x + 4)^\circ$ and $m\angle f = (5x + 8)^\circ$
are same-side interior angles. Find $x$ and $m\angle c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{14}$, $m\angle c = \ans{102}^\circ$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} two habits cost the most marks. The first is treating
every parallel-line pair as congruent --- same-side interior pairs add to a
straight angle instead. The second is answering with $x$ when the question
asked for an angle. A quick test: add your two angle measures and see whether
the result matches the pair you claimed.
\end{watchoutbox}
