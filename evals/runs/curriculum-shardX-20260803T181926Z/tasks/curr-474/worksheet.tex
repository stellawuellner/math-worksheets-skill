\wstitleblock{Definite Integrals and the Fundamental Theorem --- Diagnosing Bounds, Sign and Variable-of-Integration Errors}{AP Calculus AB/BC}{}

\noindent\small Three slips account for most lost points on FTC free-response
work. Every problem below contains one of them.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Bound order and sign:} $\int_a^b f = -\int_b^a f$. Bounds that
        run backwards, a variable sitting at the \emph{lower} bound, and a
        substitution that changes the variable but not the bounds all show up as
        a sign or a scale that is simply wrong.
  \item \textbf{Evaluate, then subtract as a whole:} $\int_a^b f = F(b) - F(a)$.
        The parentheses around $F(a)$ are not decoration --- the minus reaches
        every term inside them.
  \item \textbf{The variable of integration is a dummy:} it is consumed by the
        integral. It cannot survive into the answer of a definite integral, and
        it is not something you substitute a bound into.
  \item In each problem: name the error in words first, then show corrected work.
\end{itemize}
\normalsize

\problem[5cm]{\textbf{Find and fix.} Evaluate $\displaystyle\int_{5}^{2} 2x\,dx$.
\par\smallskip
Jae writes ``the antiderivative is $x^2$, so the answer is $5^2 - 2^2 = 21$.''
\par\smallskip
Say in one sentence what Jae did to the bounds, then give the correct value.}

\problem[6cm]{\textbf{Find and fix.} Evaluate
$\displaystyle\int_{1}^{3} \bigl(3x^2 - 4\bigr)\,dx$.
\par\smallskip
Ravi finds the antiderivative correctly, then writes
$\;27 - 12 - 1 - 4 = 10$.
\par\smallskip
\textbf{(a)} Write the antiderivative $F(x)$ Ravi should be using.
\par
\textbf{(b)} Name the punctuation Ravi lost, then evaluate the integral
correctly.}

\problem[5.5cm]{\textbf{Find and fix.} Evaluate
$\displaystyle\int_{0}^{2} \bigl(3t^2 + 2\bigr)\,dt$.
\par\smallskip
Noor writes ``$t$ runs up to $2$, so put $2$ in for $t$'': $3(2)^2 + 2 = 14$.
\par\smallskip
Explain what Noor skipped between the integrand and the bounds, then evaluate the
integral correctly.}

\problem[6cm]{\textbf{Find and fix.} A continuous function $f$ satisfies
$\displaystyle\int_{1}^{6} f(x)\,dx = 10$ and
$\displaystyle\int_{1}^{4} f(x)\,dx = 3$.
\par\smallskip
Asked for $\displaystyle\int_{6}^{4} f(x)\,dx$, a student subtracts and answers
$7$.
\par\smallskip
\textbf{(a)} Say which integral the value $7$ actually is.
\par
\textbf{(b)} Give the correct value of $\displaystyle\int_{6}^{4} f(x)\,dx$.}

\problem[6cm]{\textbf{Find and fix.} Evaluate
$\displaystyle\int_{0}^{\pi/3} \sin x\,dx$.
\par\smallskip
Lena writes ``the antiderivative of $\sin x$ is $\cos x$'' and reports
$\cos(\pi/3) - \cos(0)$.
\par\smallskip
State the correct antiderivative, say what Lena's version does to the answer, and
evaluate the integral correctly.}

\problem[6cm]{\textbf{Find and fix.} Let
$\displaystyle G(x) = \int_{x}^{5} \bigl(t^2 + 1\bigr)\,dt$.
\par\smallskip
Kwame writes $G'(x) = x^2 + 1$ and reports $G'(2) = 5$.
\par\smallskip
\textbf{(a)} Rewrite $G(x)$ with the variable at the upper bound, and say what
that does to the sign.
\par
\textbf{(b)} Compute $G'(2)$ correctly.}

\problem[7cm]{\textbf{Find and fix.} Evaluate
$\displaystyle\int_{0}^{2} x\bigl(x^2 + 1\bigr)^{3}\,dx$ by substitution.
\par\smallskip
Ines sets $u = x^2 + 1$, gets $\dfrac{u^4}{8}$, and then evaluates it at the
original bounds, reporting $2$.
\par\smallskip
\textbf{(a)} Convert the bounds to $u$ and state the new interval.
\par
\textbf{(b)} Evaluate the integral correctly.}

\problem[7.5cm]{\textbf{Explain.}
\par\smallskip
\textbf{(a)} Using $\int_a^b f = F(b) - F(a)$, explain why swapping the bounds
negates a definite integral. Say what that means for the signed area.
\par
\textbf{(b)} Explain why the letter used for the variable of integration can
never appear in the value of a definite integral, and why it \emph{can} appear in
the derivative of an accumulation function.
\par
\textbf{(c)} Describe what goes wrong when a substitution changes the variable
but not the bounds, and give a habit that prevents it.
