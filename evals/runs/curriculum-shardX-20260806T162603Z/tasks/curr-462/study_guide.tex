% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Three Inverse-Trig Rules}

\begin{formulabox}
\textbf{Rules} (chain rule already built in, with $u$ a function of $x$):
\[
  \frac{d}{dx}\arcsin u = \frac{u'}{\sqrt{1-u^2}}, \quad
  \frac{d}{dx}\arccos u = \frac{-u'}{\sqrt{1-u^2}}, \quad
  \frac{d}{dx}\arctan u = \frac{u'}{1+u^2}
\]
Arcsine and arccosine differ only by a sign. Write down $u$ and $u'$ before you
start, and remember to square $u$, not $x$, in the denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = \arcsin(5x)$
\step{Name the inside function first: $u = 5x$, so $u' = 5$ --- then the rule is a fill-in-the-blanks job.}
\step{$y' = \dfrac{5}{\sqrt{1-(5x)^2}} = \dfrac{5}{\sqrt{1-25x^2}}$.}

$\Rightarrow\quad y' = $ $\ans{\dfrac{5}{\sqrt{1-25x^2}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = \arctan(4x)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{4}{1+16x^2}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The Inverse-Function Rule}

\begin{formulabox}
\textbf{Rule:} if $f(a) = b$, then
\[
  \left(f^{-1}\right)'(b) = \frac{1}{f'(a)} .
\]
Order matters: find the input $a$ that produces the output $b$, differentiate
$f$, evaluate $f'$ \emph{at $a$}, then take the reciprocal. Evaluating at $b$ is
the standard error. You never need a formula for $f^{-1}$ --- which is what makes
the rule worth having.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^3 + x + 6$ and $f(2) = 16$. Find
$\left(f^{-1}\right)'(16)$.
\step{The output is $16$ and the input that gives it is $2$, so the reciprocal must be taken of $f'$ evaluated at $2$, not at $16$.}
\step{$f'(x) = 3x^2 + 1$, so $f'(2) = 12 + 1 = 13$, and the inverse's slope is $\frac{1}{13}$.}

$\Rightarrow\quad f'(x) = $ $\ans{3x^2+1}$ \quad and \quad $\left(f^{-1}\right)'(16) = $ $\ans{\tfrac{1}{13}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^3 + 2x + 5$ and $f(1) = 8$. Find $f'(x)$ and
$\left(f^{-1}\right)'(8)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^2+2}$ \ and \ $\ans{\tfrac{1}{5}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Inverse Trig Inside a Product}

\begin{formulabox}
\textbf{Rule:} an inverse-trig function is just another factor, so the product,
quotient and chain rules apply as usual. Expect the answer to keep an
$\arcsin$, $\arccos$ or $\arctan$ term --- an answer with the inverse-trig
function still in it is normal, not a sign that something went wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = x\arcsin x$
\step{Two non-constant factors are multiplied, so this is the product rule with $u = x$ and $v = \arcsin x$.}
\step{$y' = (1)\arcsin x + x\cdot\dfrac{1}{\sqrt{1-x^2}} = \arcsin x + \dfrac{x}{\sqrt{1-x^2}}$.}

$\Rightarrow\quad y' = $ $\ans{\arcsin x + \dfrac{x}{\sqrt{1-x^2}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = x\arccos x$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\arccos x - \tfrac{x}{\sqrt{1-x^2}}}$}
\end{tryitbox}
