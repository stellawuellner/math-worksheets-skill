\sstitleblock{Indexes, Ratios and Convergence}

\skillheading{Skill 1 --- Counting terms and reading the index}

\begin{formulabox}
\textbf{The index is data, not decoration.} A rule written for $n \ge 1$ puts the first term
at $n = 1$, so the $k$th term is the value at $n = k$. Substituting $k - 1$ gives the term
\emph{before} the one you wanted.\par
\textbf{In a sum, read the limits.} $\sum_{k=1}^{4}$ has four terms and no $k = 0$ term;
$\sum_{k=0}^{4}$ has five. Count the terms before you add: from $k = p$ to $k = q$ there are
$q - p + 1$ of them.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $a_n = 3n + 5$ with $n \ge 1$, find the fourth term.
\step{The rule starts at $n = 1$, so ``fourth term'' means the value at $n = 4$ --- no shifting is needed or allowed.}
\step{$a_4 = 3(4) + 5 = 12 + 5$}

$\Rightarrow\quad \ans{17}$ \quad (the value at $n = 3$ would be $14$, the \emph{third} term)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $a_n = 7n - 2$ with $n \ge 1$, find the sixth term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40}$}
\end{tryitbox}

\skillheading{Skill 2 --- Common ratio and geometric sums}

\begin{formulabox}
\textbf{Find the ratio by dividing, never subtracting:} $r = a_{n+1} / a_n$, and it must come
out the same for every consecutive pair.\par
\textbf{Finite sum:} $S_n = a_1(1 - r^n)/(1 - r)$. \quad
\textbf{Infinite sum:} $S = a_1/(1 - r)$, \emph{but only when} $-1 < r < 1$. Outside that
range the terms do not shrink, the series has no sum, and the formula returns a meaningless
number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $8 + 4 + 2 + 1 + \cdots$, find the ratio, the sum of the first four terms, and the infinite sum.
\step{Dividing consecutive terms gives the ratio, and its size decides whether the infinite sum formula may be used at all.}
\step{$r = 4/8$, and that ratio lies between $-1$ and $1$, so both formulas apply.}
\step{$S_4 = 8\bigl(1 - (1/2)^4\bigr)/(1 - 1/2) = 8(15/16)(2)$, and $S = 8/(1 - 1/2) = 8(2)$.}

$\Rightarrow\quad r = \ans{\tfrac{1}{2}}, \qquad S_4 = \ans{15}, \qquad S = \ans{16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $9 + 3 + 1 + \cdots$, find the common ratio and the infinite sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{\tfrac{1}{3}}$, $S = \ans{\tfrac{27}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} for $3 + 6 + 12 + \cdots$ the ratio is $2$, so there is no infinite sum. Writing $3/(1-2) = -3$ is using a formula outside the range where it means anything --- and a sum of positive terms can never be negative.
\end{watchoutbox}

\skillheading{Skill 3 --- Binomial expansions term by term}

\begin{formulabox}
\textbf{The term rule:} in $(A + B)^n$ written in descending powers of $A$, the term at index
$k$ is $\binom{n}{k} A^{n-k} B^{k}$, and $k$ starts at $0$. So the \emph{third} term uses
$k = 2$, not $k = 3$.\par
\textbf{Square the whole first piece.} In $(2x + 3)^5$ the $A$ is $2x$, so $A^2$ is $4x^2$ ---
the coefficient is squared along with the letter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $(x + 3)^3$.
\step{Row $3$ of Pascal's triangle supplies the coefficients, and the powers of $3$ rise as the powers of $x$ fall, so the whole expansion can be written straight down.}
\step{Coefficients $1, 3, 3, 1$ with $3^0, 3^1, 3^2, 3^3$: \ $x^3 + 3x^2(3) + 3x(9) + 27$.}

$\Rightarrow\quad \ans{x^3 + 9x^2 + 27x + 27}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $(x + 1)^4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^4 + 4x^3 + 6x^2 + 4x + 1}$}
\end{tryitbox}
