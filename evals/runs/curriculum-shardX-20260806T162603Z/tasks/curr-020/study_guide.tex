% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Find the Missing Partner to Ten}

\begin{formulabox}
\textbf{Rule:} Every number to 9 has a \textbf{partner} that fills it up to
ten. Partner pairs: 1 and 9, \ 2 and 8, \ 3 and 7, \ 4 and 6, \ 5 and 5.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $3\;+\;?\;=\;10$
\step{We know one partner and want the other, so we count up from 3 to 10.}
\step{4, 5, 6, 7, 8, 9, 10 --- that is 7 counts, and $3 + 7 = 10$. $\checkmark$}

$\Rightarrow\quad$ the missing partner is $\ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $8\;+\;?\;=\;10$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Decompose Ten}

\begin{formulabox}
\textbf{Rule:} Taking a number out of ten leaves its partner:\par
if \ part $+$ part $=10$, \ then \ $10 -$ part $=$ other part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $10 - 4$
\step{Ten breaks into partner pairs, so instead of counting back we ask which partner goes with 4.}
\step{$4 + 6 = 10$, so taking 4 away leaves 6. $\checkmark$}

$\Rightarrow\quad 10 - 4 = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $10 - 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Make Ten to Add}

\begin{formulabox}
\textbf{Rule:} To add two numbers, fill the bigger one up to ten first: move
just enough from the smaller number, then add what is left onto 10.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 + 5$
\step{Adding onto 10 is easy, so we fill 8 up to ten using part of the 5.}
\step{8 needs 2: split 5 into $2+3$. Then $8+2=10$ and $10+3=13$. $\checkmark$}

$\Rightarrow\quad 8 + 5 = \ans{13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $9 + 6$ \ (fill 9 up to ten first)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When you move dots to make ten, do not forget the
leftovers! In $9+4$, moving 1 to make ten leaves 3 more to add --- the answer
is 13, not 10.
\end{watchoutbox}
