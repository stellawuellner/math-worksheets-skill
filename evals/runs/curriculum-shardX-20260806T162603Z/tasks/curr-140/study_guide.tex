% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Round First, Then Estimate}

\begin{formulabox}
\textbf{Rule:} round each number to the nearest whole number, then compute
with the rounded numbers. Use the estimate to check whether an exact answer
is reasonable. If both numbers were rounded the same direction, you also know
whether the true answer is bigger or smaller than the estimate.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $5.8 \times 3.2$.
\step{An estimate wants easy numbers --- round each factor to the nearest
whole number before multiplying.}
\step{$5.8 \to 6$ and $3.2 \to 3$, so the product is about $6 \times 3$.}

$\Rightarrow\quad 5.8 \times 3.2 \approx \ans{18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $11.7 \div 3.9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Add and Subtract: Line Up the Point}

\begin{formulabox}
\textbf{Rule:} line up the \emph{decimal points} (never the last digits),
fill empty places with zeros, then add or subtract as usual. The point in
the answer drops straight down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $4.6 + 3.75$.
\step{Line up the decimal points and give both numbers the same number of
decimal places before adding.}
\step{$4.60 + 3.75$: hundredths $0+5$, tenths $6+7$ regroups, wholes
$4+3+1$.}

$\Rightarrow\quad 4.6 + 3.75 = \ans{8.35}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $7 - 1.25$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.75}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiply, Then Place the Point}

\begin{formulabox}
\textbf{Rule:} multiply as whole numbers, then give the product as many
decimal places as \emph{both factors have together} (tenths $\times$ tenths
$=$ hundredths).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $0.6 \times 0.8$.
\step{Multiply the digits as whole numbers, then count decimal places in
both factors to place the point.}
\step{$6 \times 8 = 48$; one place $+$ one place $=$ two places.}

$\Rightarrow\quad 0.6 \times 0.8 = \ans{0.48}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $1.5 \times 0.4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.6}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Divide with a Whole-Number Divisor}

\begin{formulabox}
\textbf{Rule:} shift the decimal point in \emph{both} numbers until the
divisor is whole. Both numbers grow tenfold together, so the quotient stays
the same.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $4.8 \div 0.6$.
\step{Make the divisor a whole number by shifting both numbers one place to
the right.}
\step{$4.8 \div 0.6 = 48 \div 6$}

$\Rightarrow\quad 4.8 \div 0.6 = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $6.3 \div 0.7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}
