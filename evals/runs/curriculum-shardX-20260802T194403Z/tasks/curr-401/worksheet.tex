\wstitleblock{Transformations with Multiple Parameters}{Precalculus \quad Functions and Composition}{}

\begin{center}\small\textit{Transformations from a table, parameters inside an equation, and the order of composition}\end{center}

\noindent\small\textbf{Directions:} Every problem here has \emph{more than one} parameter
acting at once. Name each parameter before you compute, and keep the inside-the-function
moves separate from the outside-the-function moves.\normalsize

\vspace{0.2cm}
\noindent\fbox{\begin{minipage}{0.965\linewidth}
\small\textbf{The parameter map --- read every problem against this}\par\vspace{2pt}
\[ g(x) \;=\; \underbrace{a}_{\substack{\text{vertical stretch}\\ \text{reflect if } a<0}}
   \cdot f\Bigl(\underbrace{b}_{\substack{\text{horizontal}\\ \text{compress by } 1/b}}
   (x - \underbrace{h}_{\substack{\text{shift right}\\ h \text{ units}}})\Bigr)
   + \underbrace{k}_{\substack{\text{shift up}\\ k \text{ units}}} \]
\textbf{Outside} the function ($a$, $k$) acts on the \emph{output} and does what it says.
\textbf{Inside} ($b$, $h$) acts on the \emph{input} and does the opposite of what it looks
like. The inside must be \emph{factored} into the form $b(x-h)$ before $h$ can be read.
\end{minipage}}

\problem[5cm]{Let $f(x) = x^2$ and $g(x) = 3f(x - 2) - 4$. Fill in the chain, then state
$g(5)$.\par\vspace{3pt}

\problem[5cm]{Let $f(x) = x^2$ and $h(x) = -f(x + 2) + 5$.\\[2pt]
Which two parameters are non-trivial here, and what does each one do? Then find $h(1)$.}

\problem[5.5cm]{Let $g(x) = 2\sqrt{x - 5} + 3$.\\[2pt]
Solve $g(x) = 11$. Undo the outside parameters first, then the inside one, and say which
step each parameter is undone in.}

\problem[6cm]{Let $y = 3\sin\bigl(2\left(x - \tfrac{\pi}{4}\right)\bigr) + 2$.\\[2pt]
Identify the amplitude, the period, the phase shift and the midline. Then find $y$ when
$x = \dfrac{\pi}{2}$.}

\problem[5cm]{Two transformation maps: $s(x) = x + 5$ (a shift) and $d(x) = 3x$ (a stretch).\\[2pt]
Write a simplified formula for $(d \circ s)(x) = d(s(x))$, and state in words which move
happens first.}

\problem[5.5cm]{Using the same $s(x) = x + 5$ and $d(x) = 3x$:\\[2pt]
For what value of $x$ does $(d \circ s)(x) = 27$? Then explain in one sentence why
$(s \circ d)(x)$ would \emph{not} give the same answer.}

\problem[6cm]{Rewrite $y = 4\sin\left(3x - \dfrac{\pi}{2}\right)$ in the form
$A\sin\bigl(B(x - C)\bigr)$.\\[2pt]
State the phase shift $C$, and explain why it is not $\dfrac{\pi}{2}$.}

\problem[6cm]{Let $f(x) = \sqrt{x}$ (graphed above) and $g(x) = -2f\bigl(4(x - 1)\bigr) + 7$.\\[2pt]
List all four parameters and what each one does, then compute $g(5)$.}

\problem[6cm]{Let $g(x) = 3\sqrt{x + 6} - 9$.\\[2pt]
Find the $x$-intercept of $g$. (Set the output to zero and undo the parameters in the
reverse of the order that builds them.)}
