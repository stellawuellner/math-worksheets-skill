\sstitleblock{Ratios, Inverses and Calculator Mode}

\skillheading{Skill 1 --- Choosing the right ratio}

\begin{formulabox}
\textbf{Name the two sides first, then the ratio follows.} With the angle $A$
fixed, label the sides \textbf{opp}, \textbf{adj} and \textbf{hyp}:
\[ \sin A = \frac{\text{opp}}{\text{hyp}}, \qquad
   \cos A = \frac{\text{adj}}{\text{hyp}}, \qquad
   \tan A = \frac{\text{opp}}{\text{adj}}. \]
Tangent never involves the hypotenuse, so it can neither find one nor use one.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has hypotenuse $18$ and one acute angle
$41^\circ$. Find the side adjacent to that angle.
\step{Want the adjacent side, know the hypotenuse --- adjacent over hypotenuse
is cosine, and the unknown is on top so I multiply.}
\step{$\text{adj} = 18\cos 41^\circ = 13.5847\ldots$}

$\text{adj} \approx \ans{13.58}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has hypotenuse $25$ and one acute angle
$63^\circ$. Find the side opposite that angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{22.28}$}
\end{tryitbox}

\skillheading{Skill 2 --- Choosing the right inverse}

\begin{formulabox}
\textbf{Rule.} Two sides are known and the angle is not: form their ratio, then
undo it with the matching inverse.
\begin{itemize}[leftmargin=0.6cm, itemsep=1pt, topsep=2pt]
  \item opp and adj $\rightarrow \tan^{-1}$
  \item opp and hyp $\rightarrow \sin^{-1}$
  \item adj and hyp $\rightarrow \cos^{-1}$
\end{itemize}
Dividing alone gives the \emph{ratio}, not the angle. The inverse function is
what converts one into the other.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has a side of $9$ opposite $\angle A$ and a
side of $14$ adjacent to it. Find $m\angle A$.
\step{The two known sides are opposite and adjacent, which is the tangent
pairing --- so the angle is the inverse tangent of their quotient.}
\step{$A = \tan^{-1}(9/14) = 32.7352\ldots^\circ$}

$m\angle A \approx \ans{32.74}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has a side of $5$ adjacent to $\angle A$ and a
hypotenuse of $12$. Find $m\angle A$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{65.38}$ degrees}
\end{tryitbox}

\skillheading{Skill 3 --- Degree mode and a size check}

\begin{formulabox}
\textbf{Two habits that catch a mode error every time.}
\begin{itemize}[leftmargin=0.6cm, itemsep=1pt, topsep=2pt]
  \item Confirm the calculator shows \textsf{DEG} before the first line of work.
        In radian mode an angle such as $52$ is read as $52$ radians, and the
        result can even come out negative --- impossible for a side length.
  \item Estimate before you accept. A leg is always shorter than the
        hypotenuse; a $45^\circ$ angle splits a right triangle evenly, so a
        smaller angle means a shorter opposite side and a longer adjacent one.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has a side of $30$ adjacent to a $25^\circ$
angle. Find the opposite side, and check the size of the answer.
\step{Opposite over adjacent is tangent, and the unknown is on top --- multiply,
with the calculator in degree mode.}
\step{$\text{opp} = 30\tan 25^\circ = 13.9892\ldots$, and since $25^\circ$ is
well under $45^\circ$ the opposite side should indeed be well under $30$.}

$\text{opp} \approx \ans{13.99}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has a side of $16$ adjacent to a $40^\circ$
angle. Find the opposite side.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13.43}$}
\end{tryitbox}
