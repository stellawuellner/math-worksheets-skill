\begin{center}
\textit{Simplifying to find holes, locating vertical asymptotes, finding
intercepts, and reading end behaviour for a horizontal asymptote}
\end{center}

\noindent\textbf{Before you start.} Four questions answer almost everything
about a rational graph, and the order matters.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Factor first.} A factor that cancels leaves a \emph{hole}; a
factor still in the denominator after cancelling is a \emph{vertical
asymptote}.
\item \textbf{Intercepts.} The $x$-intercepts are the zeros of the
\emph{simplified} numerator; the $y$-intercept is the value at $x = 0$.
\item \textbf{End behaviour.} Compare the degrees top and bottom to decide the
horizontal asymptote.
\item Show the work that produced each feature --- a bare list of answers earns
nothing here.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Factor the numerator and the denominator completely, then write
$f$ as a single factored fraction.
\[ f(x) = \frac{x^2 - 9}{x^2 + 2x - 15} \]}

\problem[5cm]{Let $g(x) = \dfrac{x^2 - x - 6}{x^2 - 4}$.

(a) Write $g$ in lowest terms. \quad (b) The graph of $g$ has a hole. Give its
coordinates.}

\problem[4.5cm]{Find every vertical asymptote of
$h(x) = \dfrac{x + 1}{x^2 - 3x - 10}$. Show why nothing cancels first.}

\problem[5.5cm]{For $p(x) = \dfrac{x^2 - 2x - 8}{x^2 + 4x + 3}$, find

(a) every $x$-intercept, \quad (b) the $y$-intercept.}

\problem[4.5cm]{Find the horizontal asymptote of
$q(x) = \dfrac{3x^2 - 5}{x^2 + 4}$ by describing what happens to $q(x)$ as $x$
grows without bound.}

\problem[5.5cm]{Dana was asked for the intercepts of
$r(x) = \dfrac{2x - 6}{x^2 - x - 12}$. She wrote: ``$r(0) = -0.5$, so the
$y$-intercept is $(0, -0.5)$.'' Find and fix her mistake, then give the
$x$-intercept as well.}

\problem[5.5cm]{$s(x) = \dfrac{x + 4}{x^2 - 16}$ is undefined at two values of
$x$. Find both, decide which one is a vertical asymptote and which one is a
hole, and give the coordinates of the hole.}

\problem[4.5cm]{Find the horizontal asymptote of
$t(x) = \dfrac{2x + 7}{x^2 + 3x}$, and say in one sentence what the graph does
far out to the right.}

\problem[5.5cm]{Both $x = 2$ and $x = -1$ make the numerator of
$u(x) = \dfrac{x^2 - x - 2}{x^2 - 4}$ equal to zero, but only one of them is an
$x$-intercept. Say which, and give the coordinates of the feature at the other
value.}

\problem[5.5cm]{Jonah says: ``$v(x) = \dfrac{x^2 - 4}{x - 2}$ has a vertical
asymptote at $x = 2$, because the denominator is zero there.'' Explain why he
is wrong, and describe exactly what the graph does at $x = 2$.}

\problem[6.5cm]{Let $w(x) = \dfrac{2x^2 + 3x}{x^2 - 9}$.

(a) Find the horizontal asymptote. \quad (b) A graph may cross its horizontal
asymptote. Does this one? If it does, find the $x$-value where it happens.}

an open circle and the asymptotes with dashed lines.
