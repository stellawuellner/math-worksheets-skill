\wstitleblock{Counting a Set and Telling How Many}{Kindergarten--Grade 1 \quad Counting and Cardinality}{}

\begin{center}
\textit{Counting one object at a time, counting two groups, and telling which
set has more}
\end{center}

\noindent\textbf{How to count a set.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Touch one object. Say one number. Touch the next object. Say the next
number.
\item Touch \emph{every} object, and touch each one \emph{only once}.
\item The \textbf{last number you say} tells how many are in the whole set.
\end{itemize}

\noindent\small A five-frame or ten-frame helps: a full row is $5$, and a full
ten-frame is $10$, so you can start from there and count on.\normalsize

\problem[3cm]{Touch each star and count. How many stars are in the set?
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\star$ & $\star$ & $\star$ & $\star$ & $\star$ \\[0.3cm]\hline
$\star$ & $\star$ & & & \\[0.3cm]\hline
\end{tabular}
\end{center}
Say the last number out loud. That is how many stars there are.}

\problem[3cm]{These counters are not in a straight line. Touch each one and
cross it off as you count, so you do not count any counter twice.
\begin{center}\Large
\begin{tabular}{cccccc}
$\bullet$ & & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.5cm]
$\bullet$ & $\bullet$ & & $\bullet$ & & $\bullet$ \\
\end{tabular}
\end{center}
How many counters are in the set?}

\problem[3cm]{The ten-frame is completely full, so it shows $10$. Start at $10$
and count on for the loose dots.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.3cm]\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.3cm]\hline
\end{tabular}\qquad $\bullet$ \quad $\bullet$
\end{center}
How many dots are there altogether?}

\problem[3cm]{Ducks are in two places. Count the ducks in the pond, then keep
counting the ducks on the grass.
\begin{center}\Large
Pond: $\circ$ \; $\circ$ \; $\circ$ \; $\circ$ \qquad
Grass: $\circ$ \; $\circ$ \; $\circ$ \; $\circ$ \; $\circ$
\end{center}
How many ducks are there in all?}

\problem[3cm]{The buttons are in rows that are not the same length. Count all
of the rows.
\begin{center}\Large
\begin{tabular}{l}
$\blacksquare$ \; $\blacksquare$ \; $\blacksquare$ \\[0.3cm]
$\blacksquare$ \; $\blacksquare$ \; $\blacksquare$ \\[0.3cm]
$\blacksquare$ \; $\blacksquare$ \\
\end{tabular}
\end{center}
How many buttons are in the set?}

\problem[3.5cm]{Rosa and Milo each found some shells.
\begin{center}\Large
\begin{tabular}{l}
Rosa: $\diamond$ \; $\diamond$ \; $\diamond$ \; $\diamond$ \; $\diamond$ \quad
$\diamond$ \; $\diamond$ \; $\diamond$ \\[0.35cm]
Milo: $\diamond$ \; $\diamond$ \; $\diamond$ \; $\diamond$ \quad
$\diamond$ \; $\diamond$ \\
\end{tabular}
\end{center}
Count Rosa's shells. Count Milo's shells. Then write $>$ or $<$ in the blank so
the sentence is true.

\smallskip
\hspace{1cm}\Large Rosa's shells \underline{\hspace{1.2cm}} Milo's shells}

\problem[3.5cm]{Three baskets hold apples. Count one basket, then keep counting
