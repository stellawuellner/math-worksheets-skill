\problem{$\dfrac{x}{3} + 5 = 9$

\textbf{Step 1.} Undo the addition: subtract $5$ from both sides, leaving
$\dfrac{x}{3} = 4$.

\textbf{Step 2.} Undo the division: multiply both sides by $3$, so $x = 12$.

\textbf{Check.} $\dfrac{12}{3} + 5 = 4 + 5 = 9$ $\checkmark$ \quad
$\ans{x = 12}$}

\problem{$\dfrac{2}{5}x - 3 = 7$

\textbf{Step 1.} Add $3$ to both sides: $\dfrac{2}{5}x = 10$.

\textbf{Step 2.} Multiply both sides by the reciprocal $\dfrac{5}{2}$:
$x = 10 \cdot \dfrac{5}{2} = 25$.

\textbf{Check.} $\dfrac{2}{5}(25) - 3 = 10 - 3 = 7$ $\checkmark$ \quad
$\ans{x = 25}$}

\problem{$\dfrac{x}{4} + \dfrac{x}{6} = 10$

\textbf{Step 1.} The denominators are $4$ and $6$, so the LCD is $12$.
Multiply \emph{every} term by $12$:
\[ 12 \cdot \frac{x}{4} + 12 \cdot \frac{x}{6} = 12 \cdot 10
   \quad\Longrightarrow\quad 3x + 2x = 120. \]

\textbf{Step 2.} Combine like terms: $5x = 120$.

\textbf{Step 3.} Divide both sides by $5$: $x = 24$.

\textbf{Check.} $\dfrac{24}{4} + \dfrac{24}{6} = 6 + 4 = 10$ $\checkmark$ \quad
$\ans{x = 24}$}

\problem{$\dfrac{3}{4}(x - 8) = 6$

\textbf{Step 1.} The fraction multiplies the whole binomial, so undo it in one
move: multiply both sides by $\dfrac{4}{3}$, giving $x - 8 = 8$.

\textbf{Step 2.} Add $8$ to both sides: $x = 16$.

\textbf{Check.} $\dfrac{3}{4}(16 - 8) = \dfrac{3}{4}(8) = 6$ $\checkmark$ \quad
$\ans{x = 16}$}

\problem{$\dfrac{2}{3}x + 1 = \dfrac{x}{6} + 4$

\textbf{Step 1.} Denominators $3$ and $6$ give LCD $6$. Multiply every term on
\emph{both} sides by $6$:
\[ 4x + 6 = x + 24. \]

\textbf{Step 2.} Collect the variable on one side: subtract $x$ from both
sides, leaving $3x + 6 = 24$.

\textbf{Step 3.} Subtract $6$, then divide by $3$: $3x = 18$, so $x = 6$.

\textbf{Check.} Left: $\dfrac{2}{3}(6) + 1 = 5$. Right: $\dfrac{6}{6} + 4 = 5$
$\checkmark$ \quad $\ans{x = 6}$}

\problem{$\dfrac{x}{2} - 3 > 4$

\textbf{Step 1.} Add $3$ to both sides: $\dfrac{x}{2} > 7$.

\textbf{Step 2.} Multiply both sides by $2$. Two is \emph{positive}, so the
symbol does not turn: $x > 14$.

\textbf{Check.} $x = 20$ gives $10 - 3 = 7 > 4$ $\checkmark$; $x = 14$ gives
exactly $4$, which is not greater, so $14$ is correctly excluded. \quad
$\ans{x > 14}$}

\problem{$\dfrac{x + 2}{3} + \dfrac{x - 1}{2} = 6$

\textbf{Step 1.} LCD $= 6$. Multiplying a fraction such as $\dfrac{x+2}{3}$ by
$6$ multiplies the \emph{whole} numerator, so keep the parentheses:
\[ 2(x + 2) + 3(x - 1) = 36. \]

\textbf{Step 2.} Distribute: $2x + 4 + 3x - 3 = 36$.

\textbf{Step 3.} Combine like terms: $5x + 1 = 36$, so $5x = 35$ and $x = 7$.

\textbf{Check.} $\dfrac{9}{3} + \dfrac{6}{2} = 3 + 3 = 6$ $\checkmark$ \quad
$\ans{x = 7}$}

\problem{$\dfrac{3}{4}x - \dfrac{2}{3} = \dfrac{x}{6} + \dfrac{5}{3}$

\textbf{Step 1.} Denominators $4$, $3$ and $6$ give LCD $12$. Multiply every
term by $12$:
\[ 9x - 8 = 2x + 20. \]

\textbf{Step 2.} Subtract $2x$ from both sides: $7x - 8 = 20$.

\textbf{Step 3.} Add $8$, then divide by $7$: $7x = 28$, so $x = 4$.

\textbf{Check.} Left: $3 - \dfrac{2}{3} = \dfrac{7}{3}$. Right:
$\dfrac{4}{6} + \dfrac{5}{3} = \dfrac{2}{3} + \dfrac{5}{3} = \dfrac{7}{3}$
$\checkmark$ \quad $\ans{x = 4}$}

\problem{$-\dfrac{2}{3}x + 1 \ge 5$

\textbf{Step 1.} Subtract $1$ from both sides: $-\dfrac{2}{3}x \ge 4$.

\textbf{Step 2.} Multiply both sides by $-\dfrac{3}{2}$. The multiplier is
\emph{negative}, so the inequality symbol reverses: $x \le -6$.

\textbf{Check.} $x = -9$ gives $6 + 1 = 7 \ge 5$ $\checkmark$, while $x = 0$
gives $1 \ge 5$, which is false --- so the solution really is the interval
below $-6$. \quad $\ans{x \le -6}$}

\problem{Clear the fractions from $\dfrac{5}{6}x - \dfrac{1}{2} = \dfrac{1}{3}x + 2$.

\textbf{Step 1.} The denominators are $6$, $2$ and $3$, so the LCD is $6$.

\textbf{Step 2.} Multiply \emph{each of the four terms} by $6$:
\[ 6 \cdot \frac{5}{6}x - 6 \cdot \frac{1}{2}
   = 6 \cdot \frac{1}{3}x + 6 \cdot 2. \]

\textbf{Step 3.} Each product is an integer term, which gives the equivalent
integer equation below. It is equivalent because multiplying both sides of an
equation by the same nonzero number never changes its solution set --- both
equations are true exactly when $x = 5$.

$\ans{5x - 3 = 2x + 12}$}

\problem{$\dfrac{1}{2}\left(\dfrac{x}{3} - 4\right) + \dfrac{5}{6}x = 3$

\textbf{Step 1.} Distribute the $\dfrac{1}{2}$ across the inner parentheses:
$\dfrac{x}{6} - 2 + \dfrac{5}{6}x = 3$.

\textbf{Step 2.} The two $x$-terms already share the denominator $6$:
$\dfrac{x}{6} + \dfrac{5x}{6} = \dfrac{6x}{6} = x$, so the equation becomes
$x - 2 = 3$.

\textbf{Step 3.} Add $2$ to both sides: $x = 5$.

\textbf{Check.} $\dfrac{1}{2}\left(\dfrac{5}{3} - 4\right) + \dfrac{5}{6}(5)
= \dfrac{1}{2}\left(-\dfrac{7}{3}\right) + \dfrac{25}{6}
= -\dfrac{7}{6} + \dfrac{25}{6} = 3$ $\checkmark$ \quad $\ans{x = 5}$}

\problem{$\dfrac{3(2x - 1)}{4} - \dfrac{2(x + 1)}{3} = \dfrac{5}{12}x + 4$

\textbf{Step 1.} Denominators $4$, $3$ and $12$ give LCD $12$. Multiply every
term by $12$, keeping each numerator inside its parentheses:
\[ 9(2x - 1) - 8(x + 1) = 5x + 48. \]

\textbf{Step 2.} Distribute on the left: $18x - 9 - 8x - 8 = 5x + 48$.

\textbf{Step 3.} Combine like terms: $10x - 17 = 5x + 48$.

\textbf{Step 4.} Subtract $5x$ and add $17$: $5x = 65$, so $x = 13$.

\textbf{Check.} Left: $\dfrac{3(25)}{4} - \dfrac{2(14)}{3}
= \dfrac{75}{4} - \dfrac{28}{3} = \dfrac{225 - 112}{12} = \dfrac{113}{12}$.
Right: $\dfrac{65}{12} + 4 = \dfrac{113}{12}$ $\checkmark$ \quad
$\ans{x = 13}$}
