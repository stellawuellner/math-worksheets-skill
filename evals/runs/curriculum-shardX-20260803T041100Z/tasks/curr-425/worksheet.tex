\wstitleblock{Identities, Multiple Angles, and Exact Solutions}{Precalculus}{}

\noindent\textbf{Simplify with an identity, convert a multiple angle, and list
every exact solution on the interval.}

\smallskip
\noindent\small These problems mix three jobs that look alike on the page but
need different first moves. Decide which one you are being asked for before you
write anything:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Simplify / verify.} No equals sign to solve --- rewrite
        everything in sines and cosines, or reach for the Pythagorean
        identity, and reduce.
  \item \textbf{Multiple angle.} A $2x$ (or $2\theta$) appears. Replace it with
        a double-angle form \emph{chosen} so that only the function you already
        know survives.
  \item \textbf{Exact solution set.} There is an interval. Reduce to one
        trigonometric function, factor, then read \emph{every} unit-circle
        angle in the interval --- exact values, not decimals.
\end{itemize}
Show the identity you used at each step. Leave all answers exact.\normalsize

\noindent\rule{\linewidth}{0.4pt}

\problem[5.5cm]{Simplify to a single trigonometric function:
\[ \frac{1 - \cos^2 x}{\sin x} \]}

\problem[6cm]{Simplify to a single trigonometric function:
\[ \sec x - \sec x \sin^2 x \]}

\problem[6cm]{Rewrite $\cos 2x$ so that it contains $\sin x$ and no other
trigonometric function.}

\problem[6.5cm]{Find every exact solution of
\[ 2\sin x - 1 = 0 \]
on the interval $[0, 2\pi)$.}

\problem[6.5cm]{Suppose $\sin\theta = \dfrac{3}{5}$ and $\theta$ lies in
quadrant I, so that $\cos\theta = \dfrac{4}{5}$.
\par\smallskip
Find the exact value of $\sin 2\theta$.}

\problem[6cm]{Simplify to a single term:
\[ \tan x \cos x + \sin x \]}

\problem[7cm]{Find every exact solution of
\[ 2\cos^2 x - \cos x - 1 = 0 \]
on the interval $[0, 2\pi)$. \emph{Hint:} it factors as a quadratic in
$\cos x$.}

\problem[6.5cm]{Simplify to a single trigonometric function:
\[ \frac{\sin 2x}{1 + \cos 2x} \]}

\problem[7cm]{Combine and simplify. Your answer should contain no fractions
inside fractions:
\[ \frac{1}{1 - \sin x} + \frac{1}{1 + \sin x} \]}

\problem[7cm]{Find every exact solution of
\[ \sin 2x = \sin x \]
on the interval $[0, 2\pi)$. \emph{Hint:} expand the double angle first, then
bring everything to one side and factor --- do not divide by $\sin x$.}

\problem[6.5cm]{Suppose $\cos\theta = -\dfrac{5}{13}$ and $\theta$ lies in
quadrant II.
\par\smallskip
Find the exact value of $\cos 2\theta$.}

\problem[8cm]{\textbf{Challenge.} Find every exact solution of
\[ 2\cos^2 x + \sin x = 1 \]
on the interval $[0, 2\pi)$.
\par\smallskip
This one needs all three moves: an identity to reach a single function, a
factorisation, and a complete sweep of the interval.}
