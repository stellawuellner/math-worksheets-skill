\wstitleblock{Integrating Rational Functions with Partial Fractions}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Factor the denominator first, then write one unknown per factor: a
        constant over each linear factor, an extra term over a repeated factor,
        and a linear numerator over an irreducible quadratic.
  \item The applied problems state their units. Carry them into your answer.
  \item Give exact values ($\ln$, $\arctan$, fractions) unless a decimal is
        requested, and add $+\,C$ to every indefinite answer.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Write $\dfrac{1}{(x - 1)(x + 3)}$ as a sum of two fractions, one
over each linear factor.}

\problem[5.5cm]{Evaluate $\displaystyle\int \frac{5}{(x - 2)(x + 3)}\,dx$.}

\problem{A reaction produces a compound at the rate
$r(x) = \dfrac{6}{(x + 1)(x + 4)}$ grams per minute, where $x$ is the number of
minutes since the reaction started.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the exact mass produced during the first $2$ minutes.
    \par\vspace*{3.6cm}\answerline{g}
  \item[(b)] Give that mass to three decimal places.
    \par\vspace*{2.0cm}\ansline
\end{itemize}}

\problem[6cm]{Evaluate $\displaystyle\int \frac{x + 4}{(x + 2)^2}\,dx$. (The
repeated factor needs two terms.)}

\problem[6.5cm]{A population $P$, measured in thousands, satisfies a logistic
model whose separated form requires
$\displaystyle\int \frac{1}{P(10 - P)}\,dP$. Evaluate that integral, treating
$P$ as the variable.}

\problem[6.5cm]{Evaluate
$\displaystyle\int \frac{x^2 + 2x + 4}{x\left(x^2 + 4\right)}\,dx$. (The
quadratic factor does not factor further, so its numerator is linear.)}

\problem{A filter removes a pollutant from a lake at the rate
$r(t) = \dfrac{8}{(t + 1)(t + 3)}$ kilograms per year, where $t$ is measured in
years from installation.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the exact mass removed during the first $3$ years.
    \par\vspace*{3.6cm}\answerline{kg}
  \item[(b)] Give that mass to three decimal places.
    \par\vspace*{2.0cm}\ansline
\end{itemize}}

\problem{A second filter removes a pollutant at the rate
$r(x) = \dfrac{2}{x(x + 2)}$ kilograms per year, for $x \ge 1$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the exact mass removed between $x = 1$ and $x = 3$.
    \par\vspace*{3.2cm}\ansline
  \item[(b)] For $b > 1$ the mass removed between $x = 1$ and $x = b$ equals
    $\ln\!\left(\dfrac{b}{b + 2}\right) + \ln 3$. Find its limit as
    $b \rightarrow \infty$.
    \par\vspace*{2.4cm}\ansline
  \item[(c)] Explain what that limit says about the filter's lifetime total.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}
