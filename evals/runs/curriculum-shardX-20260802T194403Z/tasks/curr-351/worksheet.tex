\begin{center}\small
\textit{powers of i, plotting on the complex plane, multiplying by i as a quarter turn, and non-real solutions}
\end{center}

\noindent\small\textbf{Directions.} Every problem below has two halves: a
\emph{picture or pattern} and the \emph{notation} that records it. Do both. On
the complex plane, the number $a + bi$ is the point $(a, b)$ --- the real part
runs along the horizontal (Re) axis and the imaginary part along the vertical
(Im) axis. Write every answer in $a + bi$ form.\normalsize

\problem[3cm]{\textbf{The cycle.} Fill in the three empty cells, then answer the
question underneath.

\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
$i^{1}$ & $i^{2}$ & $i^{3}$ & $i^{4}$ & $i^{5}$ \\\hline
\rule{0pt}{3ex}$i$ & $-1$ & \rule{1.4cm}{0pt} & \rule{1.4cm}{0pt} & \rule{1.4cm}{0pt} \\[4pt]\hline
\end{tabular}
\end{center}

\noindent What is $i^{5}$, and why does the table start repeating there?}

\problem[3.5cm]{\textbf{Using the cycle.} Simplify $i^{27}$. Show the division
you used to decide where the exponent $27$ lands in the four-step cycle.}

\problem[3cm]{\textbf{Addition is tip-to-tail.} Plot $3 + 2i$ and $-1 + 4i$ as
arrows from the origin on the plane below. Slide the second arrow so its tail
sits at the tip of the first; the tip now lands on the sum. Write that sum in
$a + bi$ form and check it by adding the real and imaginary parts separately.

\begin{center}
\begin{tikzpicture}[scale=0.4]
  \draw[gray!30, very thin] (-6,-6) grid (6,6);
  \draw[->, thick] (-6.5,0) -- (6.5,0) node[right] {Re};

\end{tikzpicture}
\end{center}}

\problem[3cm]{\textbf{Multiplying by $i$ is a quarter turn.} Plot $3 + 4i$.
Then compute $i(3 + 4i)$ using $i^{2} = -1$, write the result in $a + bi$ form,
and plot it too. Describe in one sentence what multiplying by $i$ did to the
arrow.

\problem[4.5cm]{\textbf{No real solutions, two complex ones.} Solve
$x^{2} + 9 = 0$. Then say where the two solutions sit on the complex plane and
what the picture has to do with the fact that the parabola $y = x^{2} + 9$ never
crosses the $x$-axis.}

\problem[4.5cm]{\textbf{Two powers at once.} Simplify $3i^{15} - 2i^{22}$ to
$a + bi$ form. Reduce each power with the cycle first, then combine.}

\problem[4.5cm]{\textbf{Four quarter turns.} Start at $1 + 2i$ and multiply by
$i$ four times in a row --- that is, compute $i^{4}(1 + 2i)$. Write where you
land, and explain why the picture guarantees that answer before you do any
arithmetic.}

\problem[5cm]{\textbf{All the zeros.} Find all four zeros of $x^{4} - 16$.
(Hint: factor as a difference of squares twice.) Say which zeros are real and
which are not, and describe the pattern the four points make around the origin.}

\problem[5.5cm]{\textbf{Conjugates.} Solve $x^{2} - 4x + 13 = 0$ by completing
the square or with the quadratic formula. Plot both solutions mentally: they are
mirror images of each other across the real axis. Explain why that mirror
symmetry has to happen for any quadratic with real coefficients.}
