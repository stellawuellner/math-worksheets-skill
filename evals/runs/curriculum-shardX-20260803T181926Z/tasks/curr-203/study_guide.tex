\sstitleblock{Approximating Irrational Roots on a Number Line}

\begin{formulabox}
\textbf{Perfect squares to know:} $1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121,
144, 169$.\\[3pt]
\textbf{The one rule everything uses:} for positive numbers, squaring keeps the
order. So $p < q$ exactly when $p^2 < q^2$ --- which means you can compare
roots without ever computing one, just by squaring the number you are comparing
it to.
\end{formulabox}

\skillheading{1. Trap the root between two whole numbers}

\begin{formulabox}
Find the perfect square just below the radicand and the one just above it. If
$p^2 < N < q^2$ with $q = p + 1$, then $p < \sqrt{N} < q$. Every later estimate
lives inside that gap.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Between which two consecutive whole numbers is $\sqrt{30}$?
\step{$30$ is not a perfect square, so the root is irrational --- trap it
between the perfect squares on either side.}
\step{The perfect squares around $30$ are $25$ and $36$, whose roots are $5$
and $6$, and $25 < 30 < 36$.}

$\Rightarrow\quad \ans{5 < \sqrt{30} < 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Between which two consecutive whole numbers is $\sqrt{90}$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9 < \sqrt{90} < 10}$}
\end{tryitbox}

\skillheading{2. Squeeze it to the nearest tenth}

\begin{formulabox}
Inside the bracket, test tenths by \emph{squaring} them --- never by guessing.
Find the two tenths whose squares straddle the radicand, then take whichever
square is closer to it. Write the answer with $\approx$, never $=$: the decimal
never ends.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Estimate $\sqrt{52}$ to the nearest tenth.
\step{The bracket is $7$ to $8$, so the answer is $7$-point-something --- test
tenths by squaring them.}
\step{$7.2^2 = 51.84$ is just under $52$, and $7.3^2 = 53.29$ is well over, so
$52$ is much nearer the lower tenth.}

$\Rightarrow\quad \sqrt{52} \approx \ans{7.2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Estimate $\sqrt{68}$ to the nearest tenth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8.2}$}
\end{tryitbox}

\skillheading{3. Plot it on the number line}

\begin{formulabox}
Label the line with the tenths inside your bracket, then put the dot at the
tenth your squaring chose --- nudged left if the radicand was below that
tenth's square, right if it was above. The dot's position, not the label, is
what shows you understand the size of the number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Plot $\sqrt{23}$ on a line marked in tenths.
\step{First find which tenth it belongs to, then read the position off the
line.}
\step{$4.7^2 = 22.09$ and $4.8^2 = 23.04$, so $23$ sits just below the upper
tenth's square and the dot goes a hair left of $4.8$.}

$\Rightarrow\quad \sqrt{23} \approx \ans{4.8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Which tenth does $\sqrt{33}$ plot at?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.7}$}
\end{tryitbox}

\skillheading{4. Order roots together with rational numbers}

\begin{formulabox}
Do not estimate the root and compare decimals --- square the \emph{rational}
number instead and compare it with the radicand. It is exact, and it is faster:
$\sqrt{N}$ versus $d$ becomes $N$ versus $d^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Which is larger, $\sqrt{11}$ or $3.4$?
\step{One side is irrational, so square the rational side and compare
radicands --- that decides it exactly.}
\step{$3.4^2 = 11.56$, and $11 < 11.56$, so the root is the smaller one.}

$\Rightarrow\quad \sqrt{11} \ans{<} 3.4$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $<$ or $>$ between $\sqrt{27}$ and $5.1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\sqrt{27} \ans{>} 5.1$}
\end{tryitbox}
