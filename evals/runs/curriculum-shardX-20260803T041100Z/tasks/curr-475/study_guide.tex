\sstitleblock{Riemann Sums, Accumulation, and the Fundamental Theorem}

\begin{formulabox}
\textbf{One chain.} A Riemann sum \emph{estimates} accumulated change from
finitely many samples; its limit \emph{is} the definite integral; the
Fundamental Theorem then evaluates it exactly --- and read backwards,
differentiates it.
\textbf{FTC 2:} $\int_a^b f(x)\,dx = F(b) - F(a)$. \quad
\textbf{FTC 1:} $\frac{d}{dx}\int_a^x f(t)\,dt = f(x)$.
\end{formulabox}

\skillheading{1. Riemann sums as estimates}

\begin{formulabox}
With $n$ equal subintervals, $\Delta x = \dfrac{b-a}{n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \textbf{Left} sum: skip the last sample. \textbf{Right} sum: skip the
first. Each uses $n$ values.
\item \textbf{Trapezoid:} $\dfrac{\Delta x}{2}\left[y_0 + 2y_1 + \cdots +
2y_{n-1} + y_n\right]$ --- interior values count twice.
\item If $f$ is \textbf{increasing}, left underestimates and right
overestimates; if decreasing, the reverse.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rate is sampled every 2 units as $3, 7, 10, 12$. Find the
left Riemann sum over the three subintervals.
\step{A left sum uses the value at the start of each subinterval, so the last
reading is not used at all --- identify which three values that leaves.}
\step{The left values are 3, 7 and 10 with $\Delta x = 2$, so the sum is
$2(3+7+10) = 2(20)$.}
\[ \text{left sum} = \ans{40} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rate is sampled every 3 units as $2, 5, 6, 9$. Find the left
Riemann sum over the three subintervals.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{39}$}
\end{tryitbox}

\skillheading{2. Antiderivatives: reversing the derivative}

\begin{formulabox}
$\displaystyle\int x^n dx = \frac{x^{n+1}}{n+1} + C$ \; ($n \neq -1$); \;
$\int \cos x\,dx = \sin x + C$; \; $\int \sin x\,dx = -\cos x + C$.\\[2pt]
Antidifferentiate term by term, and check by differentiating --- that check is
exact and costs one line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $\displaystyle\int (3x^2 - 4x)\,dx$.
\step{The integrand is a sum of powers, so reverse the power rule on each term
separately rather than looking for a substitution.}
\step{$3x^2$ raises to $\dfrac{3x^3}{3} = x^3$ and $-4x$ raises to
$-\dfrac{4x^2}{2} = -2x^2$.}
\[ \ans{x^3 - 2x^2 + C} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $\displaystyle\int (8x^3 + 2)\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x^4 + 2x + C}$}
\end{tryitbox}

\skillheading{3. Evaluating definite integrals (FTC 2)}

\begin{formulabox}
Find any antiderivative $F$, then compute $F(b) - F(a)$. The $+C$ cancels, so
one antiderivative is enough.\\[2pt]
Order matters: $F(a) - F(b)$ gives the negative of the answer. A negative result
is not an error --- the definite integral measures \textbf{signed} area, and
regions below the axis count negatively.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $\displaystyle\int_1^4 (2x+1)\,dx$.
\step{The integrand has an elementary antiderivative, so FTC 2 applies and the
work is one antiderivative plus one subtraction.}
\step{$F(x) = x^2 + x$, so $F(4) = 20$ and $F(1) = 2$, and the integral is
$20 - 2$.}
\[ \ans{18} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\displaystyle\int_0^2 3x^2\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{4. Accumulation functions (FTC 1)}

\begin{formulabox}
$F(x) = \int_a^x f(t)\,dt$ is a \emph{function of its upper limit}: net
accumulation from $a$ to $x$, with $F(a) = 0$. By FTC 1, $F'(x) = f(x)$, so $F$
increases where $f$ is positive.\\[2pt]
For a quantity with a known rate, $A(t) = A(0) + \int_0^t r(u)\,du$ --- the
integral gives the \textbf{change}, never the amount.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $F(x) = \displaystyle\int_0^x 2t\,dt$. Find $F(5)$.
\step{The upper limit is the input, so evaluating $F$ at 5 means integrating
from 0 to 5 --- the variable $t$ is internal and never appears in the answer.}
\step{An antiderivative of $2t$ is $t^2$, so $F(5) = 25 - 0$. (By FTC 1,
$F'(x) = 2x$, which is positive for $x>0$, so $F$ is increasing.)}
\[ F(5) = \ans{25} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $G(x) = \displaystyle\int_0^x 3t^2\,dt$. Find $G(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}
