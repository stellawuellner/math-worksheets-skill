\sstitleblock{Dilations and Similar Figures}

\begin{formulabox}
\textbf{A dilation multiplies every length by the scale factor $k$ and leaves every angle alone.} That is why the image is \emph{similar} to the original --- same shape, different size. A dilation is the one transformation on this sheet that is not rigid.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Coordinates of a dilated point}

\begin{formulabox}
\textbf{About the origin:} $(x, y) \to (kx,\ ky)$.\par
\textbf{About a centre $(h, j)$:} $(x, y) \to \bigl(h + k(x - h),\ \ j + k(y - j)\bigr)$ --- measure the offset from the centre, scale the offset, add it back.\par
$k > 1$ enlarges, $0 < k < 1$ shrinks, and the centre is the one point that never moves.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Dilate $(-4, 5)$ about the origin by a factor of $3$.
\step{The centre is the origin, so each coordinate can be scaled on its own --- no offsets are needed.}
\step{$x' = 3(-4)$ \quad and \quad $y' = 3(5)$}

$\Rightarrow\quad ($ $\ans{-12}$ $,$ $\ans{15}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Dilate $(7, 4)$ about $(1, 1)$ by a factor of $3$ and give the $x$-coordinate of the image.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Finding the scale factor from two lengths}

\begin{formulabox}
\textbf{Rule:} $k = \dfrac{\text{image length}}{\text{original length}}$ --- a division of matching sides, never a subtraction.\par
Convert both lengths to the same unit first. A model in centimetres against a real object in metres gives a scale factor a hundred times too small if the conversion is skipped.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A $6$ cm side becomes $21$ cm after an enlargement. Find the scale factor.
\step{A scale factor answers ``how many times as long'', and that is a ratio, so divide the new length by the old one rather than finding the gap between them.}
\step{$k = 21/6$}

$\Rightarrow\quad k = $ $\ans{3.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An $18$ cm side becomes $45$ cm. Find the scale factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Similar figures in real situations}

\begin{formulabox}
\textbf{Set up a proportion of matching parts:} $\dfrac{\text{height}_1}{\text{shadow}_1} = \dfrac{\text{height}_2}{\text{shadow}_2}$, and cross-multiply.\par
Keep both fractions built the same way round --- height over shadow on both sides, or shadow over height on both sides. Mixing them is the commonest error in this topic.\par
Scale drawings work the same way: real length $=$ drawn length $\times$ the scale.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A $2$ m post casts a $3$ m shadow; a tree casts a $9$ m shadow. Find the tree's height.
\step{The post with its shadow and the tree with its shadow form similar right triangles, so matching sides are in the same ratio --- set up height over shadow on both sides.}
\step{$\dfrac{h}{9} = \dfrac{2}{3}$, so $h = \dfrac{2 \times 9}{3}$}

$\Rightarrow\quad h = $ $\ans{6}$ m
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A $1.8$ m post casts a $4$ m shadow; a tree casts a $20$ m shadow. Find the tree's height in metres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a scale factor $k$ multiplies lengths by $k$, but it multiplies \emph{areas} by $k^2$. Doubling every side of a rectangle quadruples its area --- so never apply a linear scale factor straight to an area.
\end{watchoutbox}
