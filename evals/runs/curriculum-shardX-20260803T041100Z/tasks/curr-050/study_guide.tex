\sstitleblock{Putting Small Shapes Together to Make Bigger Shapes}

\skillheading{Skill 1 --- Counting the pieces that cover a shape}

\begin{formulabox}
\textbf{Rule:} when the pieces are laid in \textbf{equal groups} (rows, or one small
shape repeated inside a bigger one), do not count one piece at a time.\par
\smallskip
\textbf{Count what is in one group, then count the groups}, and skip-count. If a build
has layers (triangles inside rhombuses inside a hexagon), start with the smallest piece
and work up one layer at a time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rectangle is covered with square tiles: 3 rows, and each row
holds 4 tiles. How many tiles?
\step{Every row holds the same number of tiles, so these are equal groups --- count one
row, then count the rows.}
\step{One row is 4 tiles and there are 3 rows: $4 + 4 + 4 = 12$, or $3 \times 4 = 12$.}
$\Rightarrow\quad \ans{12}$ tiles
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A rectangle is covered with square tiles: 2 rows, and each row
holds 5 tiles. How many tiles?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Naming the shape the pieces make}

\begin{formulabox}
\textbf{Rule:} pieces must touch along \textbf{whole sides}, with no gaps and nothing
on top of anything.\par
\smallskip
To name the new shape, \textbf{trace the outline} and ignore the lines inside the build.
Count the straight sides and the corners of that outline: three straight sides names a
triangle; four straight sides with square corners names a rectangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two identical right triangles are joined so that their longest
sides touch exactly. Name the new shape.
\step{The pieces touch along a whole side, so the line between them is inside the build
--- trace only the outline.}
\step{The outline has four straight sides and four square corners, and one pair of sides
is longer than the other pair.}
$\Rightarrow\quad \ans{\text{rectangle}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two identical red trapezoids are joined so that their longest
sides touch exactly. Trace the outline and name the new shape.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{hexagon}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two ways to cover a shape}

\begin{formulabox}
\textbf{Rule:} when \emph{the same} big shape is covered two different ways, compare the
two \textbf{counts of pieces} --- never how big one piece looks.\par
\smallskip
Write the two counts and put $>$ or $<$ between them. The bigger count is the covering
that used more pieces. \textbf{Smaller pieces always means more of them.}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One rectangle is covered by 4 square tiles. The same rectangle is
covered by 8 triangle tiles. Which covering uses more pieces?
\step{Both coverings fill the same rectangle, so the only thing to compare is how many
pieces each one took.}
\step{Four pieces against eight pieces, and $4$ is less than $8$.}
$\Rightarrow\quad \ans{4 < 8}$, so the triangle covering uses more pieces
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One hexagon is covered by 2 red trapezoids. The same hexagon is
covered by 6 green triangles. Which covering uses more pieces?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2 < 6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Bigger pieces'' does not mean ``more pieces''. The bigger the
piece, the \emph{fewer} of them it takes to cover the same shape. Also check that the
pieces really touch along whole sides --- a build with a gap in it has not covered
anything.
\end{watchoutbox}
