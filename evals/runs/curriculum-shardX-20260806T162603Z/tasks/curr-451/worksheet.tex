\wstitleblock{Evaluating Limits by Factoring and Rationalizing}{AP Calculus AB/BC}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Every limit below gives the indeterminate form $\frac{0}{0}$ on direct substitution.
  \item Rewrite the expression first --- factor and cancel, or multiply by the conjugate --- and only then substitute.
  \item Give exact values (fractions, not decimals). Write the limit notation on every line of work, not just at the end.
\end{itemize}

\problem[4.4cm]{The table lists values of
$f(x) = \dfrac{x^2 - 16}{x - 4}$ near $x = 4$.
\par\vspace{0.2cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $3.9$ & $3.99$ & $4.01$ & $4.1$ \\ \hline
$f(x)$ & $7.9$ & $7.99$ & $8.01$ & $8.1$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.15cm}
Predict $\displaystyle\lim_{x \to 4} f(x)$ from the table, then confirm the
prediction by factoring the numerator and cancelling.}

\problem[5cm]{Evaluate \quad $\displaystyle\lim_{x \to -2} \frac{x^2 + 5x + 6}{x + 2}$}

\problem[5.5cm]{Evaluate \quad $\displaystyle\lim_{x \to 0} \frac{\sqrt{x + 9} - 3}{x}$}

\problem[5.5cm]{Evaluate \quad $\displaystyle\lim_{x \to 3} \frac{x^2 - x - 6}{x^2 - 9}$}

\problem[5.5cm]{Evaluate \quad $\displaystyle\lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4}$}

\problem[6cm]{The quotient below is the average rate of change of $y = x^3$ from
$x = 2$ to $x = 2 + h$, so its limit is the instantaneous rate of change at
$x = 2$. Expand the cube, cancel the $h$, and evaluate
\[
  \lim_{h \to 0} \frac{(2 + h)^3 - 8}{h}
\]}

\problem[6.2cm]{The table lists values of $g(x) = \dfrac{x^3 - 8}{x^2 - 4}$
near $x = 2$, rounded to four decimal places.
\par\vspace{0.2cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $1.9$ & $1.99$ & $2.01$ & $2.1$ \\ \hline
$g(x)$ & $2.9256$ & $2.9925$ & $3.0075$ & $3.0756$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.15cm}
\textbf{(a)} Factor both the numerator and the denominator, cancel, and give the
exact value of $\displaystyle\lim_{x \to 2} g(x)$: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(b)} Explain why the four table entries on their own do not prove the
value you found in part (a).}

\problem[6cm]{\textbf{Challenge.} The radical is in the denominator this time.
Evaluate
\[
  \lim_{x \to 1} \frac{x - 1}{\sqrt{x + 3} - 2}
\]}
