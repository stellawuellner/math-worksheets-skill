\wstitleblock{Unit Rates with Whole Numbers}{Grade 6}{}

\begin{center}
\textit{Finding unit rates: divide down to one unit, use the rate, and compare rates}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item A \textbf{unit rate} tells how much for \textbf{one}: one hour, one
        pound, one ticket. In notation: rate $=$ total $\div$ number of units.
  \item Each model (equal groups, tables) shows the same idea — connect the
        model to the division you write.
\end{itemize}

\problem[3.2cm]{A carton of 12 eggs is used to make 4 pancake batches, each
batch using the same number of eggs. Sort the eggs into the groups below, then
write the unit rate: eggs per batch.\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \foreach \i in {0,...,11} \fill[black!60] ({mod(\i,6)*0.7},{-floor(\i/6)*0.7}) circle (0.22);
  \foreach \i in {0,...,3} \draw[thick, rounded corners] ({5.2+\i*2},-0.9) rectangle ({6.8+\i*2},0.4);
\end{tikzpicture}
\end{center}}
