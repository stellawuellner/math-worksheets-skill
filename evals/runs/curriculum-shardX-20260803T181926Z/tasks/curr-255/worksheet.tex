\wstitleblock{Choosing the Form That Shows the Structure}{Algebra 1 \quad Expressions and polynomial operations}{}

\begin{center}
\textit{Expanding to reveal the terms, factoring to reveal the structure, and
choosing the form that answers the question}
\end{center}

\noindent\textbf{Before you start.} An expression can be written in more than
one form, and the forms are equally true but not equally useful.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Expanded form} (a sum of terms) shows the degree, the leading
coefficient, and the constant term.
\item \textbf{Factored form} (a product) shows the zeros and any common factor.
\item After the first few problems the two jobs are mixed, so read each
question and decide which form it is asking for before you write anything.
\end{itemize}

\medskip

\problem[5cm]{Write $(x+5)(x+3)$ in expanded form.}

\problem[5cm]{Write $2x(3x-7)$ in expanded form.}

\problem[5cm]{Factor $x^2-9x+20$ completely.}

\problem[5cm]{Factor $x^2-49$ completely.}

\problem[5cm]{Without multiplying the two binomials together, find the value of
$(x-6)(x+2)$ when $x=6$. Write one sentence saying which form told you the
answer.}

\problem[6cm]{Write $(2x-3)^2$ in expanded form. Show the middle term ---
squaring a binomial is not the same as squaring each piece.}

\problem[6cm]{Factor $3x^2+11x-4$ completely.}

\problem[6cm]{Expand and simplify $(x+3)^2-(x-3)^2$. The simplified form is a
single term; write it.}

\problem[6cm]{Write $(x+2)(x^2-2x+4)$ in expanded form. Collect like terms
carefully --- the answer is much shorter than the work.}

\problem[6cm]{Factor $2x^3-18x$ completely. Take out the greatest common factor
before you look for a pattern.}

\problem[6cm]{Let $m=100$. Find the value of $(m-3)(m+3)$ using whichever form
makes the arithmetic easiest, and write down the form you chose.}

\problem[8cm]{\textbf{Synthesis.} Consider the expression $x^4-13x^2+36$.

\smallskip
(a) Factor it completely into four linear factors. (Start by treating $x^2$ as
a single quantity.)

\smallskip
(b) Use your factored form --- not the original --- to find the value of the
expression when $x=5$.}
