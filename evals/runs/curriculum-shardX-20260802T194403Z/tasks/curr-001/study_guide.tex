\sstitleblock{Counting Forward Within 100}

\skillheading{Skill 1 --- Counting forward by ones}
\begin{formulabox}\textbf{Rule:}\\[4pt]
To count forward, start at the given number and say the next numbers in order.
Each step adds \textbf{one}. Do not start over at 1.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Start at 41 and count forward 3.
\step{Put 41 in your head. It is the starting number, not the first count.}
\step{Count on three times: 42, 43, 44. The last number said is $\ans{44}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Start at 57 and count forward 4.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{61}$}\end{tryitbox}

\skillheading{Skill 2 --- Counting forward by tens}
\begin{formulabox}\textbf{Rule:}\\[4pt]
Counting by tens adds \textbf{ten} each step. The ones digit stays the same and
the tens digit goes up by one: 20, 30, 40, \ldots
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Start at 20 and count forward 2 tens.
\step{One ten past 20 is 30.}
\step{One more ten past 30 is 40, so the answer is $\ans{40}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Start at 25 and count forward 3 tens.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{55}$}\end{tryitbox}

\skillheading{Skill 3 --- Comparing what you counted}
\begin{formulabox}\textbf{Rule:}\\[4pt]
Finish \textbf{both} counts first, then compare the two ending numbers.
$<$ means \emph{is less than}; $>$ means \emph{is greater than}.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Compare $30+5$ and $32+2$.
\step{Count each one: 30 forward 5 lands on 35; 32 forward 2 lands on 34.}
\step{35 is greater than 34, so the symbol is $\ans{>}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Compare $60+3$ and $61+3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Do not count the starting number as
your first count. Starting at 41 and counting 3 gives 44, not 43.
\end{watchoutbox}
