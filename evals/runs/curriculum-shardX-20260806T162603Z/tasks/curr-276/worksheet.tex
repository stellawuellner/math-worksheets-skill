\wstitleblock{Telling Exponential Change from Linear Change}{Algebra 1}{}

\begin{center}
\small Constant differences against constant ratios, evaluating both models, and comparing them over the long run
\end{center}
\vspace{0.15cm}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item One test settles which model a table follows: over equal steps in the input, a \textbf{linear} pattern has equal \emph{differences} between outputs, and an \textbf{exponential} pattern has equal \emph{ratios}.
  \item Linear rules look like $a + bx$; exponential rules look like $a \cdot r^{\,x}$, where $r$ is the number you multiply by at each step.
  \item Round money and population answers to two decimal places where a decimal appears.
  \item Where a problem asks you to compare two models, use the numbers you computed as the evidence.
\end{itemize}
