\wstitleblock{Integration by Parts, Chosen Strategically}{AP Calculus AB/BC}{}

\noindent\small
Single and repeated integration by parts, parts for logarithms and inverse
functions, and parts inside definite and improper integrals.
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Parts is $\displaystyle\int u\,dv = uv - \int v\,du$. It earns its keep
        only when $\int v\,du$ is \emph{easier} than what you started with, so
        state your $u$ and $dv$ before you compute anything.
  \item LIATE is a tie-breaker, not a law: choose $u$ from the first category
        present --- Logarithmic, Inverse trig, Algebraic, Trigonometric,
        Exponential --- and $dv$ from the last.
  \item Antiderivatives need $+ C$; definite integrals do not.
  \item For an improper integral, replace the infinite endpoint by a letter,
        integrate, and take the limit. Write the limit; do not substitute
        infinity directly.
  \item Leave exact answers exact: $\pi$, $e$, and fractions, not decimals.
\end{itemize}
\normalsize
\vspace{0.2cm}

\problem[7cm]{Evaluate $\displaystyle\int x e^{x}\,dx$. State your choice of $u$
and $dv$ before integrating.}

\problem[7cm]{Evaluate $\displaystyle\int x \cos x\,dx$.}

\problem[7cm]{Evaluate $\displaystyle\int \ln x\,dx$.
\par\smallskip
\emph{Strategy hint: there is only one factor, so the choice of $dv$ is forced.}}

\problem[8cm]{Evaluate $\displaystyle\int x^{2} e^{x}\,dx$. One application of
parts will not finish it --- say what the remaining integral is, then handle it.}

\problem[8cm]{Evaluate $\displaystyle\int_{0}^{1} x e^{2x}\,dx$. Give the exact
value.}

\problem[8cm]{Evaluate $\displaystyle\int x \ln x\,dx$. Explain in one line why
$u = \ln x$ rather than $u = x$.}

\problem[8cm]{Evaluate $\displaystyle\int x \sec^{2} x\,dx$. Choose the factor
whose antiderivative you already know as $dv$.}

\problem[9cm]{Evaluate $\displaystyle\int e^{x} \sin x\,dx$. Parts twice returns
you to the integral you started with --- treat that integral as an unknown and
solve for it.}

\problem[9cm]{Evaluate $\displaystyle\int_{0}^{\pi} x^{2} \cos x\,dx$. Give the
exact value.}

\problem{Evaluate $\displaystyle\int_{1}^{\infty} \frac{\ln x}{x^{2}}\,dx$, or
show that it diverges. Replace the infinite limit by a finite $t$, integrate by
parts, and then let $t \to \infty$.}
\begin{enumerate}[label=(\alph*), itemsep=4.5cm, leftmargin=1.3cm]
  \item Parts on $\displaystyle\int_{1}^{t} \frac{\ln x}{x^{2}}\,dx$ produces
        the boundary term $-\dfrac{\ln t + 1}{t}$. Find
        $\displaystyle\lim_{t \to \infty} \left(-\frac{\ln t + 1}{t}\right)$,
        and say which growth rate wins. \ansline
  \item Use part (a) to state the exact value of the improper integral.
        \ansline
\end{enumerate}
