\wstitleblock{Derivatives: Power, Product, and Quotient Rules}{AP Calculus AB/BC \quad Differentiation}{}

\begin{center}
\textit{Power rule with rewritten radicals and negative exponents, product
rule, quotient rule, and one derivative from the limit definition}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Name the rule you are using before you use it, then show the substitution
into that rule.
\item Rewrite radicals and reciprocals as powers ($\sqrt{x} = x^{1/2}$,
$\frac{1}{x^n} = x^{-n}$) \emph{before} differentiating.
\item Simplify each answer, but do not expand a squared denominator.
\item Product rule: $(uv)' = u'v + uv'$. \quad Quotient rule:
$\left(\dfrac{u}{v}\right)' = \dfrac{u'v - uv'}{v^2}$ --- the numerator order
matters.
\end{itemize}

\problem[5cm]{Find $f'(x)$. \qquad $f(x) = 4x^5 - 3x^2 + 7$}

\problem[6cm]{Find $g'(x)$. Rewrite each term as a power of $x$ first.
\qquad $g(x) = 6\sqrt{x} + \dfrac{5}{x^2}$}

\problem[8cm]{Use the \textbf{limit definition} of the derivative to find
$f'(x)$ for $f(x) = x^3$. Expand the numerator, cancel the $h$, then take the
limit. State whether your answer agrees with the power rule.
\[ f'(x) = \lim_{h \to 0} \frac{(x + h)^3 - x^3}{h} \]}

\problem[6cm]{Find $y'$ using the product rule. \qquad
$y = (x^2 + 3)(2x - 5)$}

\problem[6cm]{Find $f'(x)$ using the quotient rule, and simplify the numerator
completely. \qquad $f(x) = \dfrac{3x - 1}{x + 4}$}

\problem[6cm]{Find $h'(x)$. \qquad $h(x) = x^2\cos x$}

\problem[6.5cm]{Find $f'(x)$ and simplify. \qquad
$f(x) = \dfrac{x^2 + 1}{x^2 - 1}$}

\problem[8cm]{Let $f(x) = (2x + 1)\sqrt{x}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] Find $f'(x)$ using the product rule, and write it with a single
radical in each term.
\item[(b)] Find the slope of the tangent line to $y = f(x)$ at $x = 4$.
\end{itemize}}

\problem[8cm]{Let $f(x) = \dfrac{x^2}{2x - 3}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] Find $f'(x)$ using the quotient rule. Leave the denominator squared.
\item[(b)] Evaluate $f'(1)$.
\end{itemize}}

\problem[9cm]{Find $f'(x)$. You will need the quotient rule on the outside and
the product rule inside the numerator; you do not have to expand.
\[ f(x) = \frac{x\sin x}{x + 1} \]}
