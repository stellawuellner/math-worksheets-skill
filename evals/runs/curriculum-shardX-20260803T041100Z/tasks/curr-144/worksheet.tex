\wstitleblock{Square or Cubic? Finding the Unit Mistake}{Grade 4--5}{}

\begin{center}\small\textit{Reading the unit off the formula, finding and fixing a
squared-for-cubed mistake, and converting square and cubic units}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Every answer needs a \textbf{number} \emph{and} the \textbf{right kind of unit}.
        Count how many lengths you multiplied: two lengths give \emph{square} units,
        three lengths give \emph{cubic} units.
  \item On a \textbf{find the mistake} problem, name the mistake in words first, then
        show correct work and give the correct answer.
  \item Write units in the answer blank exactly as shown.
\end{itemize}
\normalsize

\unitpanel

\noindent\small\textit{How units are named. No measurements are shown here --- use the
numbers given in each problem.}\normalsize

\problem{A sticker is 8 centimeters long and 5 centimeters wide. Find the area of
the sticker.\par\vspace*{4cm}
\answerline{cm$^2$}}

\problem{A gift box is 8 centimeters long, 5 centimeters wide, and 3 centimeters
tall. Find the volume of the box.\par\vspace*{4.5cm}
\answerline{cm$^3$}}

\problem{\textbf{Find the mistake.} Dev found the volume of a solid cube whose edges
are 6 centimeters long. He wrote:
\[ V = 6 \times 6 = 36 \text{ cubic centimeters} \]
(a) Name Dev's mistake in words. \quad (b) Show correct work and give the correct volume.\par\vspace*{7cm}
\answerline{cm$^3$}}

\problem{A number cube has edges 4 centimeters long. Find its surface area, which is
the total area of all six faces.\par\vspace*{5cm}
\answerline{cm$^2$}}

\problem{One centimeter is the same length as 10 millimeters. How many square
millimeters exactly cover one square centimeter? Draw the square and its rows of tiny
squares to prove your answer.\par\vspace*{6cm}
\answerline{mm$^2$}}

\problem{\textbf{Find the mistake.} Ana wrote: \emph{``One cubic centimeter is 10
cubic millimeters, because one centimeter is 10 millimeters.''} Her partner said,
\emph{``No, it is 100 cubic millimeters.''} \textbf{Both} are wrong.

\smallskip
(a) Explain what each of them did with the factor. \quad (b) Give the correct number of
cubic millimeters in one cubic centimeter.\par\vspace*{7cm}
\answerline{mm$^3$}}

\problem{\textbf{Find the mistake.} A rectangular pool is 12 meters long, 5 meters
wide, and 2 meters deep. Kai needs the area of the pool \emph{floor} so he can order a
cover. He wrote:
\[ 12 \times 5 \times 2 = 120 \text{ square meters} \]
(a) Name the two things wrong with Kai's work --- what he multiplied, and what he called
the answer. \quad (b) Give the correct floor area.\par\vspace*{7cm}
\answerline{m$^2$}}

\problem[8cm]{A classmate says: \emph{``Area and volume are the same thing. You just
multiply the sides.''}

\smallskip
Draw a flat rectangle covered with unit squares, and a box filled with unit cubes. Then
write two or three sentences explaining why multiplying two lengths gives square units
but multiplying three lengths gives cubic units.
\noansline}
