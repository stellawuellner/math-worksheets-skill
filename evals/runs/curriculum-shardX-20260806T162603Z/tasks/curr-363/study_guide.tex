\sstitleblock{Zeros, Multiplicity, and Graph Behavior}

\skillheading{Skill 1 --- Reading zeros from factored form}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
If $P(x) = a(x - r_1)^{m_1}(x - r_2)^{m_2}\cdots$, then the zeros are
$x = r_1,\ r_2,\ \ldots$, and the exponent on a factor is that zero's
\textbf{multiplicity}. Set each factor equal to zero and solve; a factor
$(x + k)$ gives the zero $x = -k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the zeros of $f(x) = (x-5)(x+2)^2$.
\step{The polynomial is already a product, and a product is zero exactly when one
of its factors is zero, so no multiplying out is needed.}
\step{$x - 5 = 0$ gives $x = 5$; \quad $x + 2 = 0$ gives $x = -2$, and the
exponent $2$ says that zero has multiplicity two.}

Zeros: $\ans{5}$ and $\ans{-2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the zeros of $g(x) = (x-1)(x+6)^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$ and $\ans{-6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Factor first, then read the zeros}

\begin{formulabox}
\textbf{Method:} pull out the greatest common factor, factor what is left, then
set each factor to zero. A common factor of $x$ always produces the zero $x = 0$,
which is also the $x$-intercept at the origin.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $h(x) = x^3 - 6x^2 + 8x$ and list its zeros.
\step{Every term contains an $x$, so take out the greatest common factor before
trying anything else --- that turns a cubic into a quadratic you already know.}
\step{$x^3 - 6x^2 + 8x = x(x^2 - 6x + 8)$, and two numbers with product $8$ and
sum $-6$ are $-2$ and $-4$.}

$h(x) = \ans{x(x-2)(x-4)}$, \quad so the zeros are $\ans{0}$, $\ans{2}$, $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $p(x) = x^3 - x^2 - 12x$ and list its zeros.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x(x-4)(x+3)}$, zeros $\ans{0}$, $\ans{4}$, $\ans{-3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Multiplicity and what the graph does}

\begin{formulabox}
\textbf{At a zero of multiplicity $m$:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item $m$ odd --- the curve \textbf{crosses} the $x$-axis there;
  \item $m$ even --- the curve \textbf{touches} the axis and turns back;
  \item $m \ge 3$ --- it also flattens out as it goes through or turns.
\end{itemize}
\textbf{Far left and far right} are decided by the leading term alone: odd degree
sends the two ends opposite ways, even degree sends them the same way, and a
negative leading coefficient flips both.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = (x+1)^2(x-3)$, say what happens at each zero,
and find $f(0)$.
\step{The exponent on each factor is its multiplicity, and multiplicity is the
only thing that decides touching versus crossing.}
\step{$x = -1$ comes from a squared factor, so the curve touches and turns there;
$x = 3$ comes from a single factor, so the curve crosses. The $y$-intercept is
$f(0) = (0+1)^2(0-3)$.}

$f(0) = \ans{-3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = (x-2)^2(x+4)$, which zero is only touched, and
what is $g(0)$?\\[2pt]
\rotatebox{180}{\footnotesize check: touched at $x = 2$; \ $g(0) = \ans{16}$}
\end{tryitbox}

\skillheading{Skill 4 --- Building a polynomial from its zeros}

\begin{formulabox}
\textbf{Rule:} a zero $x = r$ of multiplicity $m$ contributes the factor
$(x - r)^m$. Multiply the factors together, then multiply by a constant $a$ if the
graph must also pass through a stated point. The degree is the sum of the
multiplicities.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write a polynomial with zeros $x = 2$ (multiplicity one)
and $x = -3$ (multiplicity two), leading coefficient $1$, then expand it.
\step{Each zero becomes a factor and its multiplicity becomes that factor's
exponent, so the factored form can be written straight down.}
\step{$(x-2)(x+3)^2 = (x-2)(x^2 + 6x + 9)$, and multiplying out term by term
gives $x^3 + 6x^2 + 9x - 2x^2 - 12x - 18$.}

$\ans{x^3 + 4x^2 - 3x - 18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write and expand a polynomial with zeros $x = -1$
(multiplicity one) and $x = 5$ (multiplicity two), leading coefficient $1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^3 - 9x^2 + 15x + 25}$}
\end{tryitbox}
