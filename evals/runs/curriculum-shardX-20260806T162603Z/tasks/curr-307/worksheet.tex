\wstitleblock{Solving for Angle Measures on Parallel Lines}{High-school geometry}{}

\begin{center}\small\textit{congruent angle pairs, supplementary angle pairs, and two-step chains}\end{center}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Double arrowheads mark the parallel lines. Name the angle pair first, then write the equation the name gives you: congruent pairs are \emph{equal}; supplementary pairs \emph{add to} $180$.
  \item Congruent on parallel lines: corresponding, alternate interior, alternate exterior. Supplementary: same-side interior, and any linear pair.
  \item Show the equation before you solve it, and check your value by substituting back.
\end{itemize}
