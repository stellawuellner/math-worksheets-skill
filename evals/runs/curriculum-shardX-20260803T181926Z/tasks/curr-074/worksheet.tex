\wstitleblock{Sharing Into Equal Groups --- How Many Groups? How Many in Each?}{Grades 2--3}{}

\noindent\small Every division story has three numbers: the \textbf{total}, the
number of \textbf{groups}, and how many are \textbf{in each group}. You are
given two of them and you find the third.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Read the question twice. Is it asking \emph{how many groups} or
        \emph{how many in each group}?
  \item Never write back a number the story already gave you --- that is the
        number you were told, not the one you were asked for.
  \item Check every answer by multiplying: groups $\times$ in each group should
        give the total.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{Ken has $18$ stickers. He puts them into $3$ bags with the same
number in every bag.\par
How many stickers are \textbf{in each bag}?}

\problem[5cm]{Mia has $20$ stickers. She puts $5$ stickers in each bag.\par
How many \textbf{bags} does she fill?}

\problem[5cm]{\textbf{Find and fix.} There are $24$ crackers. A plate gets $4$
crackers. Ravi says, ``There are $4$ plates.''\par
Ravi wrote back a number the story already told him. Which one was it? Then
write how many \textbf{plates} there really are.}

\problem[5cm]{\textbf{Find and fix.} Leo plants $35$ seeds in $5$ pots. The same
number of seeds goes in each pot. Leo says, ``There are $5$ seeds in each
pot.''\par
Leo used the number of pots for the wrong job. Write how many \textbf{seeds are
in each pot}.}

\problem[6cm]{Ana has $48$ apples. She puts $8$ apples in each basket. She
starts by writing this:\par
\vspace{0.15cm}
\hspace{1cm}$\underline{\hspace{1.2cm}} \times 8 = 48$\par
\vspace{0.15cm}
Fill in the missing number. That number tells how many \textbf{baskets} Ana
fills.}

\problem[6cm]{\textbf{Find and fix.} A teacher has $40$ markers. She gives $8$
markers to each group of children. Rio says, ``There are $8$ groups.''\par
First check Rio's answer by multiplying: $8$ groups with $8$ markers in each
group. Is that total bigger or smaller than $40$?\par
Then write how many \textbf{groups} really get markers.}

\problem[7cm]{Jen and Ben each have $36$ shells.\par
Jen makes rows with $4$ shells in each row.\par
Ben makes $4$ rows and shares his shells so every row has the same number.\par
How many shells are in \textbf{one of Ben's rows}?}

\problem[8cm]{\textbf{Explain, then solve.}\par
\textbf{(a)} A friend keeps mixing up the two jobs a number can have in a
division story. Write, in your own words, how you can tell whether a number
tells \emph{how many groups} or \emph{how many in each group}. Use the words
``in each'' somewhere in your answer.\par
\textbf{(b)} There are $30$ pencils. Each cup holds $6$ pencils. How many
\textbf{cups} are filled? Write that number on the answer line.}
