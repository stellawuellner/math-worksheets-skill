\wstitleblock{Building Equivalent Ratio Tables}{Grade 6}{}

\begin{center}
\textit{Building equivalent ratio tables: scaling up, unit-rate columns, and missing values}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Every row (or column) of a ratio table shows the \textbf{same} ratio.
        To build a new column, multiply or divide \textbf{both} quantities by
        the same number.
  \item Worked pattern to follow: if the ratio is $2$ pencils for $6$ crayons,
        then multiplying both by $3$ gives the new column $6$ pencils for $18$
        crayons.
  \item Show the multiplier you used in the work space under each table.
\end{itemize}
