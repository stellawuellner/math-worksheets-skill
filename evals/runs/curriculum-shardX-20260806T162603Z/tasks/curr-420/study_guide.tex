\sstitleblock{Modeling Periodic Behavior from Graphs}

\skillheading{Skill 1 --- Amplitude and midline from a graph}

\begin{formulabox}
\textbf{From the two extremes} (highest value $h$, lowest value $l$):\par
amplitude $A = \dfrac{h - l}{2}$ \quad (half the spread) \qquad midline $y = D$ with $D = \dfrac{h + l}{2}$ \quad (the average).\par
Amplitude is measured \emph{from the midline}, not from the horizontal axis.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A graph rises to $7$ and falls to $-3$. Find the amplitude and the midline.
\step{The two extremes are all that is needed --- half their difference is the amplitude, their average is the midline.}
\step{$A = \dfrac{7 - (-3)}{2} = \dfrac{10}{2}$ \quad and \quad $D = \dfrac{7 + (-3)}{2} = \dfrac{4}{2}$}
$\Rightarrow\quad A = \ans{5}$, \quad midline $y = \ans{2}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph rises to $11$ and falls to $5$. Find the amplitude and the midline value.\\[2pt]
\rotatebox{180}{\footnotesize check: $A = \ans{3}$, midline $y = \ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Period and the value of B}

\begin{formulabox}
\textbf{Period} is the horizontal length of one complete cycle: peak to next peak, or crossing to the matching crossing.\par
For $y = A\sin(Bx) + D$: \quad period $= \dfrac{2\pi}{B}$, \quad so \quad $B = \dfrac{2\pi}{\text{period}}$.\par
Period longer than $2\pi$ gives $B < 1$; period shorter gives $B > 1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A graph repeats every $6$ units. Find $B$.
\step{The picture gives the period directly, and $B$ is what turns a period into the standard $2\pi$ cycle.}
\step{$B = \dfrac{2\pi}{6}$}
$\Rightarrow\quad B = \ans{\tfrac{\pi}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph repeats every $10$ units. Find $B$.\\[2pt]
\rotatebox{180}{\footnotesize check: $B = \ans{\tfrac{\pi}{5}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Writing the model from a graph}

\begin{formulabox}
\textbf{Three readings, three parameters.} Highest $h$, lowest $l$, period $p$:\par
$A = \dfrac{h-l}{2}$, \quad $D = \dfrac{h+l}{2}$, \quad $B = \dfrac{2\pi}{p}$.\par
Then pick the shape: a peak at $x = 0$ takes $\cos$; a midline crossing on the way up takes $\sin$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A graph peaks at $12$, dips to $4$, repeats every $4$ units, and peaks at $x = 0$. Write the model.
\step{A peak at $x = 0$ is cosine's own starting shape, so no shift is needed and only $A$, $D$ and $B$ remain.}
\step{$A = \dfrac{12-4}{2}$, \quad $D = \dfrac{12+4}{2}$, \quad $B = \dfrac{2\pi}{4}$}
$\Rightarrow\quad A = \ans{4}$, $D = \ans{8}$, $B = \ans{\tfrac{\pi}{2}}$, so $y = 4\cos\!\left(\tfrac{\pi}{2}x\right) + 8$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph peaks at $30$, dips to $10$, repeats every $12$ units, and peaks at $x = 0$. Find $A$, $D$ and $B$.\\[2pt]
\rotatebox{180}{\footnotesize check: $A = \ans{10}$, $D = \ans{20}$, $B = \ans{\tfrac{\pi}{6}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Solving with the model}

\begin{formulabox}
\textbf{Method:} isolate the sine or cosine, then use \emph{both} angles in one turn that have that value --- for sine, $\theta$ and $\pi - \theta$; for cosine, $\theta$ and $-\theta$. Undo the inside last.\par
A full period of the input always yields two solutions unless the target is an extreme value.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4\sin\!\left(\dfrac{\pi t}{6}\right) + 12 = 14$ for $0 \le t < 12$.
\step{Strip the outside first so the equation reads sine equals a number, then take both angles with that sine.}
\step{$\sin\!\left(\dfrac{\pi t}{6}\right) = \dfrac{1}{2} \Rightarrow \dfrac{\pi t}{6} = \dfrac{\pi}{6}$ or $\dfrac{5\pi}{6}$}
$\Rightarrow\quad t = \ans{1}$ or $t = \ans{5}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $6\sin\!\left(\dfrac{\pi t}{4}\right) + 10 = 13$ for $0 \le t < 8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $t = \ans{\tfrac{2}{3}}$ or $t = \ans{\tfrac{10}{3}}$}
\end{tryitbox}
