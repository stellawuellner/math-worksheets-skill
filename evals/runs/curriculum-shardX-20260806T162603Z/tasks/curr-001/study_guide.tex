\sstitleblock{Counting Forward from Any Number}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Counting On by Ones}

\begin{formulabox}
\textbf{Rule:} You do not have to start at 1. Say your starting number, then
count on: each next number is \textbf{one more} than the number before it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Start at 24. Write the next two numbers.
\step{We are counting forward, so each new number is one more than the last one --- start saying the count at 24.}
\step{One more than 24 is $\ans{25}$; one more than 25 is $\ans{26}$. So the count goes 24, 25, 26.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Start at 42. Write the next two numbers.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{43}$, $\ans{44}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Crossing into a New Tens Group}

\begin{formulabox}
\textbf{Rule:} When a number ends in \textbf{9}, the next number starts a new
tens group: after 29 comes 30, after 59 comes 60, after 99 comes 100.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Start at 29. Write the next two numbers.
\step{29 ends in 9, so the next number begins a new tens group --- the thirties.}
\step{One more than 29 is $\ans{30}$, and one more than 30 is $\ans{31}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Start at 49. Write the next two numbers.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{50}$, $\ans{51}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} After 39 many counters say ``thirty-ten.'' There is no
thirty-ten --- the next number is \textbf{40}, the start of the forties.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Reading a Hundred-Chart Row}

\begin{formulabox}
\textbf{Rule:} In a hundred-chart row you read left to right, and every box is
\textbf{one more} than the box on its left.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A chart row shows 61, 62, 63, \_\_, 65. What goes in
the empty box?
\step{The empty box sits just right of 63, and moving one box right adds one --- so count on from 63.}
\step{One more than 63 is $\ans{64}$, and that fits: 64 comes right before 65.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A chart row shows 31, 32, \_\_, 34. What goes in the
empty box?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{33}$}
\end{tryitbox}
