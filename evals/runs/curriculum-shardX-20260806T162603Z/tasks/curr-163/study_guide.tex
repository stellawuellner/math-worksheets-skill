% =================== SKILL 1 ===================
\skillheading{Adding a Percent On: Tax and Tip}

\begin{formulabox}
\textbf{Rule:} tax and tip are percents \emph{of the price before them}, and
they are added on. Two routes give the same number:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item find the extra, then add: extra $= \frac{p}{100} \times$ price
  \item multiply once: total $= $ price $\times \left(1 + \frac{p}{100}\right)$
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A bill is 45 dollars and the tax rate is 8 percent. Find
the tax and the total.
\step{The tax is a percent of the bill and is added on, so find the extra first
and then add it.}
\step{Tax $= \frac{8}{100} \times 45 = $ $\ans{3.6}$ dollars, so the total is
$45 + 3.6 = $ $\ans{48.6}$ dollars --- the same as $45 \times 1.08$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A 25 dollar bill gets a 16 percent tip. What is the total
paid, in dollars?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{29}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Taking a Percent Off: Discount and Sale Price}

\begin{formulabox}
\textbf{Rule:} a discount is a percent \emph{of the original price}, and it is
taken off. So $d$ percent off leaves $100 - d$ percent behind:
sale price $= $ original $\times \left(1 - \frac{d}{100}\right)$. Thirty percent
off means you still pay seventy percent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A coat is marked 80 dollars with 35 percent off. Find
the discount and the sale price.
\step{The percent is of the marked price, so take 35 percent of 80 and subtract
it --- or multiply by the 65 percent that is left.}
\step{Discount $= \frac{35}{100} \times 80 = $ $\ans{28}$ dollars, so the sale
price is $80 - 28 = $ $\ans{52}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A 90 dollar tent is 20 percent off. What is the sale
price, in dollars?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{72}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Markup, and Working Back to the Original}

\begin{formulabox}
\textbf{Rule:} a markup is a percent of what the shop \emph{paid}, so
selling price $= $ cost $\times \left(1 + \frac{m}{100}\right)$. Going backwards
from the selling price to the cost means \textbf{dividing} by that same factor,
never subtracting the same percent from the selling price.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A kettle sells for 78 dollars after a 30 percent markup.
What did the shop pay, and how large was the markup?
\step{The selling price is the cost multiplied by $1.30$, so undo the
multiplying: divide the selling price by $1.30$.}
\step{Cost $= 78 \div 1.30 = $ $\ans{60}$ dollars, so the markup itself is
$78 - 60 = $ $\ans{18}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A jacket sells for 96 dollars after a 20 percent markup.
What did the shop pay, in dollars?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{80}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a discount percent is measured against the
\emph{original} price, not the sale price, and undoing a markup is a division.
Taking 20 percent off 84 dollars gives $67.20$, not the 70 dollars the shop
actually paid.
\end{watchoutbox}
