\sstitleblock{Place-Value Synthesis Review}

\skillheading{Skill 1 --- Expanded Form}

\begin{formulabox}
\textbf{Rule:} expanded form writes a number as a sum of place values, largest
place first. Going back to standard form, put a \textbf{zero} in any place with
no addend.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $90{,}000 + 1{,}000 + 50 + 3$ in standard form.
\step{Each addend fills one place; the hundreds place has no addend, so it gets
a zero.}
\step{9 ten-thousands, 1 thousand, 0 hundreds, 5 tens, 3 ones:
$91{,}053$}
$\Rightarrow\quad \ans{91{,}053}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $20{,}000 + 700 + 6$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20{,}706}$}
\end{tryitbox}

\skillheading{Skill 2 --- Comparing Numbers}

\begin{formulabox}
\textbf{Rule:} more digits wins; with the same number of digits, compare place
by place from the \textbf{left} and let the first difference decide. Trailing
digits never outrank a higher place.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $54{,}631$ and $54{,}613$.
\step{Same digit count; scan from the left: ten-thousands, thousands, and
hundreds all tie.}
\step{Tens differ: $3 > 1$, and everything after the tens no longer matters.}

$\Rightarrow\quad \ans{54{,}631 > 54{,}613}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compare $7{,}208$ and $7{,}280$ with $<$, $>$, or
$=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7{,}208 < 7{,}280}$}
\end{tryitbox}

\skillheading{Skill 3 --- Rounding Decisions}

\begin{formulabox}
\textbf{Rule:} underline the digit in the named place; the digit one place
right decides (0--4 down, 5--9 up); everything right of the named place becomes
zero. A round-up of a 9 carries left.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $6{,}285$ to the nearest hundred.
\step{Underline the hundreds digit 2; the decider is the tens digit, 8.}
\step{$8 \geq 5$, so 2 hundreds become 3 hundreds: $6{,}285 \rightarrow
6{,}300$}
$\Rightarrow\quad \ans{6{,}300}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $6{,}285$ to the nearest \emph{thousand}.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6{,}000}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} in a mixed review, half the battle is picking the tool.
``Standard form'' means build the number; a symbol blank means compare
place-by-place; ``nearest \ldots'' means round. Name the tool before you
compute.
\end{watchoutbox}
