\sstitleblock{Fitting and Interpreting Exponential Models}

\skillheading{Skill 1 --- Fitting a model to two data points}

\begin{formulabox}
\textbf{Model:} $y = a\,b^{\,x}$, where $a$ is the value at $x = 0$ and $b$ is the factor per unit of $x$.\par
Given $(0, a)$ and $(n, f)$: \quad $a\,b^{\,n} = f$, so $b = \left(\dfrac{f}{a}\right)^{1/n}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A colony holds $50$ bacteria at $x = 0$ h and $800$ at $x = 4$ h. Find $b$.
\step{The value at $x = 0$ is the coefficient, so $a = 50$; the second point then fixes $b$, and taking a root undoes the power.}
\step{$50\,b^{4} = 800 \Rightarrow b^{4} = 16 \Rightarrow b = 16^{1/4}$}
$\Rightarrow\quad b = \ans{2}$ per hour, so the model is $y = 50 \cdot 2^{\,x}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tank holds $12$ litres of algae at $x = 0$ h and $972$ litres at $x = 4$ h. Find $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Reading the rate from a model}

\begin{formulabox}
\textbf{Rule:} in $y = a\,b^{\,x}$ the base $b$ is the multiplier per unit of $x$.\par
$b > 1$: growth of $(b - 1) \times 100\%$ per unit. \quad $b < 1$: decay of $(1 - b) \times 100\%$ per unit.\par
The percent is always taken from the \emph{current} amount, so the changes multiply, never add.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $A(t) = 700(1.035)^t$. State the annual percent change.
\step{The base is above $1$, so this is growth, and the rate is how far the base sits above $1$.}
\step{$1.035 - 1 = 0.035$}
$\Rightarrow\quad \ans{3.5}\%$ growth per year.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B(t) = 900(0.88)^t$. State the annual percent change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$ percent decrease per year}
\end{tryitbox}

\skillheading{Skill 3 --- Solving for time with logarithms}

\begin{formulabox}
\textbf{Method:} to solve $a\,b^{\,t} = y$, divide by $a$, then take $\ln$ of both sides:
$t = \dfrac{\ln(y/a)}{\ln b}$. \quad If the exponent is $t/h$, multiply the result by $h$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $30(1.15)^t = 90$ for $t$, to two decimal places.
\step{The unknown sits in the exponent, and a logarithm is the tool that brings an exponent down.}
\step{$(1.15)^t = 3 \Rightarrow t\ln(1.15) = \ln 3 \Rightarrow t = \dfrac{\ln 3}{\ln 1.15}$}
$\Rightarrow\quad t \approx \ans{7.86}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $50(1.2)^t = 200$ for $t$, to two decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7.60}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\dfrac{\ln 90}{\ln 30}$ is not $\ln\!\left(\dfrac{90}{30}\right)$. Divide the numbers \emph{inside} the logarithm first, then take the logarithm.
\end{watchoutbox}

\skillheading{Skill 4 --- Rewriting a model with a new base}

\begin{formulabox}
\textbf{Rules:} $b^{\,t} = e^{\,t\ln b}$, so $a\,b^{\,t} = a\,e^{kt}$ with $k = \ln b$.\par
And $(b^{\,m})^{t} = b^{\,mt}$, so a model with exponent $mt$ has yearly factor $b^{\,m}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $400(1.05)^t$ as $400e^{kt}$. Give $k$ to four decimal places.
\step{The two forms match for every $t$ exactly when the bases match, so I need the $k$ with $e^{k} = 1.05$ --- that is what $\ln$ returns.}
\step{$k = \ln(1.05) = 0.0487902\ldots$}
$\Rightarrow\quad k \approx \ans{0.0488}$, so $400(1.05)^t = 400e^{0.0488t}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $250(1.12)^t$ as $250e^{kt}$. Give $k$ to four decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.1133}$}
\end{tryitbox}
