\sstitleblock{Division Three Ways}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- From a story to an equation}

\begin{formulabox}
\textbf{Rule:}\quad Find the \emph{total} first --- it always comes before the $\div$ sign.\par
Then divide by the number the story gives: the groups, or the number in each group.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 14 carrots are shared equally by 2 rabbits. Write the equation and solve.
\step{The total is 14 and there are 2 equal groups, so the equation is $14 \div 2$.}
\step{$14 \div 2 = 7$, because $2 \times 7 = 14$.}

$\Rightarrow\quad\ans{7}$ carrots each
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 25 grapes are shared equally onto 5 plates. Write the equation and solve.\\[2pt]
\rotatebox{180}{\footnotesize check: $25 \div 5 = \ans{5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- From a drawing to an equation}

\begin{formulabox}
\textbf{Rule:}\quad Count all the dots (the total), then count the groups.\par
total $\div$ groups $=$ dots in each group.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A drawing shows 12 dots in 3 equal rings. Write the equation it shows.
\step{The whole drawing holds 12 dots split into 3 equal rings, so it shows $12 \div 3$.}
\step{Each ring holds $12 \div 3 = 4$ dots --- count one ring to check.}

$\Rightarrow\quad\ans{4}$ dots in each ring
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A drawing shows 18 dots in 2 equal groups. What equation does it show, and how many dots are in each group?\\[2pt]
\rotatebox{180}{\footnotesize check: $18 \div 2 = \ans{9}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From an equation to a story}

\begin{formulabox}
\textbf{Rule:}\quad For total $\div$ number: pick objects for the \emph{total}, then decide what the second number is --- how many groups, or how many in each group.\par
End your story with a question the quotient answers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write a story for $15 \div 3$ and solve it.
\step{Pick 15 objects and make the 3 a number of groups: ``15 shells are shared equally into 3 buckets --- how many in each?''}
\step{$15 \div 3 = 5$, because $3 \times 5 = 15$.}

$\Rightarrow\quad\ans{5}$ shells in each bucket
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write your own story for $20 \div 4$, then solve it.\\[2pt]
\rotatebox{180}{\footnotesize check: $20 \div 4 = \ans{5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The total always comes \emph{first} in a division equation. A drawing of 24 dots in 4 groups shows $24 \div 4$, never $4 \div 24$.
\end{watchoutbox}
