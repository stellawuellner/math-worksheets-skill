% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Volume of a Box}

\begin{formulabox}
\textbf{Formula:} \quad $V = \ell \times w \times h$ \quad (cubic units,
e.g.\ cm$^3$).\par
Think in layers: the base holds $\ell \times w$ unit cubes, and there are
$h$ layers. Missing a dimension? Divide the volume by what you know.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the volume of a $6 \times 3 \times 2$ cm box.
\step{A box's volume counts unit cubes, so multiply length, width, and
height, and answer in cubic centimeters.}
\step{Base layer: $6 \times 3 = 18$ cubes; $2$ layers make $36$.}

$\Rightarrow\quad V = \ans{36\ \text{cm}^3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the volume of a $4 \times 4 \times 2$ cm box.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32\ \text{cm}^3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Converting Units}

\begin{formulabox}
\textbf{Key facts:} \quad
m $\to$ cm: $\times\,100$ \quad\textbullet\quad km $\to$ m: $\times\,1000$
\quad\textbullet\quad kg $\to$ g: $\times\,1000$ \quad\textbullet\quad
L $\to$ mL: $\times\,1000$ \quad\textbullet\quad hours $\to$ minutes:
$\times\,60$.\par
Going from a \emph{bigger} unit to a \emph{smaller} one multiplies; going
back divides.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert $3.5$ m to centimeters.
\step{Meters are bigger than centimeters, so the count of units grows ---
multiply by the number of centimeters in one meter.}
\step{$3.5 \times 100 = 350$.}

$\Rightarrow\quad 3.5\ \text{m} = \ans{350\ \text{cm}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $4$ kg to grams.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4000\ \text{g}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Finding a Missing Angle}

\begin{formulabox}
\textbf{Key facts:} \quad a right angle is $90^\circ$ \quad\textbullet\quad
a straight line is $180^\circ$ \quad\textbullet\quad a full turn is
$360^\circ$.\par
Angles \emph{add}: when a big angle is split, the parts sum to the whole,
so the missing part is whole $-$ known part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two angles make a straight line; one is $140^\circ$.
Find the other.
\step{The two parts must add to the straight line's whole, so subtract the
known part from that whole.}
\step{$180 - 140 = 40$.}

$\Rightarrow\quad \ans{40^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A right angle is split; one part is $25^\circ$. Find
the other part.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{65^\circ}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} before computing, make every measurement use the
\emph{same unit} --- you cannot subtract meters from kilometers. Convert
first, then compute.
\end{watchoutbox}
