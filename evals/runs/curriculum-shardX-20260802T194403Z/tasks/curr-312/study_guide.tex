\sstitleblock{Completing Two-Column Triangle-Congruence Proofs}

\noindent\small\textbf{How every proof here is built:} list what you are given, add the free
facts the picture supplies (a shared side or angle, vertical angles, parallel-line angles,
midpoints and bisectors), count what kind of pairs you now have --- S or A, and in what order ---
then name the test on the last line. Never assume from appearance.\normalsize

\skillheading{1. Side-based proofs: SSS and SAS (proof-sss-sas)}
\begin{formulabox}\textbf{The tests:}\quad
\textbf{SSS} --- three pairs of sides. \quad
\textbf{SAS} --- two pairs of sides and the angle \emph{between} them.\\[3pt]
\textbf{SSA is not a test.} If the angle is not included, the triangles need not be congruent.\\[3pt]
\textbf{Free facts to look for:} a shared side ($\overline{AD} \cong \overline{AD}$, Reflexive);
a midpoint (gives two congruent segments); vertical angles at an intersection.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $E$ is the midpoint of $\overline{AC}$ and of
$\overline{BD}$. Prove $\triangle AEB \cong \triangle CED$.
\step{Two midpoints give two pairs of sides, and the angles at $E$ are vertical --- sides around an angle points to SAS.}
\step{Write the pairs in the order S--A--S so the angle sits between the two sides:}
\par\vspace{2pt}{\renewcommand{\arraystretch}{1.0}\noindent\footnotesize
\begin{tabular}{|p{0.42\linewidth}|p{0.44\linewidth}|}\hline\pfhead
1.\ $\overline{AE} \cong \overline{EC}$ & 1.\ Definition of midpoint \\\hline
2.\ $\angle AEB \cong \angle CED$ & 2.\ Vertical angles are congruent \\\hline
3.\ $\overline{BE} \cong \overline{ED}$ & 3.\ Definition of midpoint \\\hline
4.\ $\triangle AEB \cong \triangle CED$ & 4.\ SAS \\\hline
\end{tabular}\normalsize}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\overline{AB} \cong \overline{AC}$ and $D$ is the midpoint
of $\overline{BC}$. Which test proves $\triangle ABD \cong \triangle ACD$, and which pair does
the picture hand you for free?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{SSS}}$ --- the free pair is
$\overline{AD} \cong \overline{AD}$ (Reflexive)}\end{tryitbox}

\skillheading{2. Angle-based proofs: ASA, AAS and HL (proof-asa-aas-hl)}
\begin{formulabox}\textbf{The tests:}\quad
\textbf{ASA} --- two angles and the side \emph{between} them. \quad
\textbf{AAS} --- two angles and a side \emph{not} between them.\quad
\textbf{HL} --- right triangles only: hypotenuse plus one leg.\\[3pt]
ASA and AAS use the same three marks; what decides the name is \emph{where the side sits}.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\overline{AB} \parallel \overline{DC}$, $E$ is the
midpoint of $\overline{AC}$, and $B$, $E$, $D$ are collinear. Prove
$\triangle ABE \cong \triangle CDE$.
\step{Parallel lines give congruent alternate interior angles, the midpoint gives the side between them --- angle--side--angle.}
\step{Check the position of the side before naming the test: $\overline{AE}$ lies between the two marked angles, so it is ASA, not AAS.}
\par\vspace{2pt}{\renewcommand{\arraystretch}{1.0}\noindent\footnotesize
\begin{tabular}{|p{0.42\linewidth}|p{0.44\linewidth}|}\hline\pfhead
1.\ $\angle BAE \cong \angle DCE$ & 1.\ Alternate interior angles \\\hline
2.\ $\overline{AE} \cong \overline{CE}$ & 2.\ Definition of midpoint \\\hline
3.\ $\angle AEB \cong \angle CED$ & 3.\ Vertical angles are congruent \\\hline
4.\ $\triangle ABE \cong \triangle CDE$ & 4.\ ASA \\\hline
\end{tabular}\normalsize}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Two right triangles share the hypotenuse
$\overline{BD}$, and one pair of legs is congruent. Which test applies, and why is the
similar-looking SSA not a test?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{HL}}$ --- the right angle}\\
\rotatebox{180}{\footnotesize removes the SSA swing (Pythagoras fixes side 3)}\end{tryitbox}

\skillheading{3. After the proof: CPCTC equations (cpctc-algebra)}
\begin{formulabox}\textbf{CPCTC:} once two triangles are proved congruent, \emph{Corresponding
Parts of Congruent Triangles are Congruent}.\\[3pt]
So matching sides are equal in length and matching angles are equal in measure. Set the two
expressions equal and solve. CPCTC is a \emph{conclusion} --- it can never be the reason two
triangles are congruent.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\triangle ABC \cong \triangle DEF$ with $AB = 4x + 1$
and $DE = 2x + 11$. Find $x$.
\step{$\overline{AB}$ and $\overline{DE}$ correspond (read the vertex order), so CPCTC makes the two expressions equal.}
\step{$4x + 1 = 2x + 11 \ \Rightarrow\ 2x = 10$}
$\Rightarrow\ \ans{x = 5}$ \quad (check: both sides give $21$)
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\triangle ABC \cong \triangle DEF$ with $BC = 6x - 5$
and $EF = 4x + 7$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Match vertices in the order the congruence statement
names them --- $\triangle WXZ \cong \triangle YZX$ says $W \leftrightarrow Y$, not
$W \leftrightarrow Z$. And a shared side or angle is worth a full line of the proof: leaving out
the Reflexive step is the most common way a correct-looking proof loses credit.
\end{watchoutbox}
