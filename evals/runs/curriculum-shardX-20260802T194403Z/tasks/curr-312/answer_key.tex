\noindent\small Proofs are open reasoning and are marked for human review: accept any logically
complete proof, including a different but valid order of statements or an equivalent reason
(``Vertical Angles Theorem'' for ``vertical angles are congruent''). A proof is wrong only if a
statement is unsupported, a reason is circular, or the wrong congruence test is named.\normalsize

\problem{Given $\overline{AB} \cong \overline{AC}$ and $D$ the midpoint of $\overline{BC}$;
prove $\triangle ABD \cong \triangle ACD$. (Two reasons were missing.)

\textbf{Model proof:}
{\proofstart
1.\ $\overline{AB} \cong \overline{AC}$ & 1.\ Given \\\hline
2.\ $D$ is the midpoint of $\overline{BC}$ & 2.\ Given \\\hline
3.\ $\overline{BD} \cong \overline{DC}$ & 3.\ Definition of midpoint \\\hline
4.\ $\overline{AD} \cong \overline{AD}$ & 4.\ \textbf{Reflexive Property of Congruence} \\\hline
5.\ $\triangle ABD \cong \triangle ACD$ & 5.\ \textbf{SSS} \\\hline
\proofend}
\textbf{Why SSS:} three pairs of sides were matched and no angle was used. The shared side
$\overline{AD}$ is the pair students forget --- every proof with a common side needs the
reflexive line.}

\problem{$E$ is the midpoint of $\overline{AC}$ and of $\overline{BD}$; prove
$\triangle AEB \cong \triangle CED$. (All reasons were missing.)

\textbf{Model proof:}
{\proofstart
1.\ $\overline{AE} \cong \overline{EC}$ & 1.\ Definition of midpoint (given: $E$ is the midpoint of $\overline{AC}$) \\\hline
2.\ $\angle AEB \cong \angle CED$ & 2.\ Vertical angles are congruent \\\hline
3.\ $\overline{BE} \cong \overline{ED}$ & 3.\ Definition of midpoint (given: $E$ is the midpoint of $\overline{BD}$) \\\hline
4.\ $\triangle AEB \cong \triangle CED$ & 4.\ SAS \\\hline
\proofend}
\textbf{Why SAS and not SSA:} the congruent angle at $E$ sits \emph{between} the two pairs of
congruent sides, which is what SAS requires.}

\problem{$\overline{PQ} \cong \overline{PR}$ and $\overrightarrow{PS}$ bisects $\angle QPR$;
prove $\triangle PQS \cong \triangle PRS$.

\textbf{Model proof:}
{\proofstart
1.\ $\overline{PQ} \cong \overline{PR}$ & 1.\ Given \\\hline
2.\ $\overrightarrow{PS}$ bisects $\angle QPR$ & 2.\ Given \\\hline
3.\ $\angle QPS \cong \angle RPS$ & 3.\ Definition of angle bisector \\\hline
4.\ $\overline{PS} \cong \overline{PS}$ & 4.\ Reflexive Property of Congruence \\\hline
5.\ $\triangle PQS \cong \triangle PRS$ & 5.\ SAS \\\hline
\proofend}
\textbf{Note:} the included angle is the one the bisector creates, so the order side--angle--side
is genuine. Do not assume $S$ is a midpoint --- that is a conclusion (by CPCTC), not a given.}

\problem{$\triangle ABC \cong \triangle DEF$ with $AB = 3x - 4$ and $DE = 2x + 5$. Find $x$.

\textbf{Solution:} $\overline{AB}$ and $\overline{DE}$ are corresponding sides of congruent
triangles, so CPCTC gives $AB = DE$:
\begin{align*}
3x - 4 &= 2x + 5 \\
x - 4 &= 5
\end{align*}
$\ans{x = 9}$ \quad Check: $3(9) - 4 = 23$ and $2(9) + 5 = 23$ $\checkmark$}

\problem{$\overline{AB} \parallel \overline{DC}$, $E$ the midpoint of $\overline{AC}$, with
$B$, $E$, $D$ collinear; prove $\triangle ABE \cong \triangle CDE$.

\textbf{Model proof:}
{\proofstart
1.\ $\overline{AB} \parallel \overline{DC}$ & 1.\ Given \\\hline
2.\ $\angle BAE \cong \angle DCE$ & 2.\ Alternate interior angles (parallel lines cut by transversal $\overline{AC}$) \\\hline
3.\ $\overline{AE} \cong \overline{CE}$ & 3.\ Definition of midpoint \\\hline
4.\ $\angle AEB \cong \angle CED$ & 4.\ Vertical angles are congruent \\\hline
5.\ $\triangle ABE \cong \triangle CDE$ & 5.\ ASA \\\hline
\proofend}
\textbf{Why ASA:} the congruent side $\overline{AE} \cong \overline{CE}$ lies between the two
pairs of congruent angles. (Naming AAS here is not accepted --- read the order off the figure.)}

\problem{$\angle W \cong \angle Y$ and $\angle WXZ \cong \angle YZX$; prove
$\triangle WXZ \cong \triangle YZX$. (Statements were missing.)

\textbf{Model proof:}
{\proofstart
1.\ $\angle W \cong \angle Y$ & 1.\ Given \\\hline
2.\ $\angle WXZ \cong \angle YZX$ & 2.\ Given \\\hline
3.\ $\overline{XZ} \cong \overline{ZX}$ & 3.\ Reflexive Property of Congruence \\\hline
4.\ $\triangle WXZ \cong \triangle YZX$ & 4.\ AAS \\\hline
\proofend}
\textbf{Why AAS, not ASA:} the shared side $\overline{XZ}$ is \emph{not} between the two marked
angles --- $\angle W$ is opposite it --- so the pattern is angle--angle--side. Note the vertex
order in the conclusion: $W \leftrightarrow Y$, $X \leftrightarrow Z$, $Z \leftrightarrow X$.}

\problem{$\overrightarrow{LN}$ bisects $\angle KLM$ and $\angle K \cong \angle M$; prove
$\triangle KLN \cong \triangle MLN$.

\textbf{Model proof:}
{\proofstart
1.\ $\overrightarrow{LN}$ bisects $\angle KLM$ & 1.\ Given \\\hline
2.\ $\angle KLN \cong \angle MLN$ & 2.\ Definition of angle bisector \\\hline
3.\ $\angle K \cong \angle M$ & 3.\ Given \\\hline
4.\ $\overline{LN} \cong \overline{LN}$ & 4.\ Reflexive Property of Congruence \\\hline
5.\ $\triangle KLN \cong \triangle MLN$ & 5.\ AAS \\\hline
\proofend}
\textbf{Why AAS:} $\overline{LN}$ is not between $\angle KLN$ and $\angle K$; it is opposite
$\angle K$. Accept ASA only if the student instead pairs $\angle KLN$ with $\angle KNL$, which
requires proving those angles congruent first.}

\problem{$\triangle ABC \cong \triangle DEF$ with $m\angle A = 5x + 12$ and
$m\angle D = 7x - 8$. Find $x$.

\textbf{Solution:} $\angle A$ and $\angle D$ correspond, so CPCTC gives $m\angle A = m\angle D$:
\begin{align*}
5x + 12 &= 7x - 8 \\
20 &= 2x
\end{align*}
$\ans{x = 10}$ \quad Check: $5(10) + 12 = 62$ and $7(10) - 8 = 62$, so $m\angle A = 62^\circ$
in both triangles $\checkmark$}

\problem{$\angle A$ and $\angle C$ are right angles and $\overline{AB} \cong \overline{CB}$;
prove $\triangle ABD \cong \triangle CBD$, then explain the SSA/HL distinction.

\textbf{Model proof:}
{\proofstart
1.\ $\angle A$ and $\angle C$ are right angles & 1.\ Given \\\hline
2.\ $\triangle ABD$ and $\triangle CBD$ are right triangles & 2.\ Definition of right triangle \\\hline
3.\ $\overline{AB} \cong \overline{CB}$ (legs) & 3.\ Given \\\hline
4.\ $\overline{BD} \cong \overline{BD}$ (hypotenuses) & 4.\ Reflexive Property of Congruence \\\hline
5.\ $\triangle ABD \cong \triangle CBD$ & 5.\ HL \\\hline
\proofend}
\textbf{Model explanation (open --- review by hand):} SSA fails in general because two different
triangles can share the same two sides and non-included angle --- the third side can swing to
two positions (the ambiguous case). When the known angle is a right angle that ambiguity
disappears: the third side is fixed by the Pythagorean theorem, so hypotenuse--leg is enough.
HL is SSA with the one angle that makes it safe.}

\problem{Overlapping triangles: $\overline{PS} \cong \overline{PT}$ and
$\overline{PQ} \cong \overline{PR}$; prove $\triangle PQT \cong \triangle PRS$ and apply CPCTC.

\textbf{Model proof:}
{\proofstart
1.\ $\overline{PQ} \cong \overline{PR}$ & 1.\ Given \\\hline
2.\ $\angle QPT \cong \angle RPS$ & 2.\ Reflexive Property (both name $\angle P$) \\\hline
3.\ $\overline{PT} \cong \overline{PS}$ & 3.\ Given \\\hline
4.\ $\triangle PQT \cong \triangle PRS$ & 4.\ SAS \\\hline
5.\ $\overline{QT} \cong \overline{RS}$ & 5.\ CPCTC \\\hline
\proofend}
\textbf{The move that unlocks it:} the two triangles overlap and share the angle at $P$, so
that shared angle is the included angle for both. Redrawing $\triangle PQT$ and
$\triangle PRS$ separately makes the correspondence $Q \leftrightarrow R$,
$T \leftrightarrow S$ visible. CPCTC is the last line, never a reason for congruence itself.}
