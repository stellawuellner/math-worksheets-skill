\sstitleblock{Adding, Subtracting, and Multiplying Complex Numbers}

\noindent\small Treat $i$ as a variable while you distribute and collect, then
--- and only then --- use the one fact that makes complex numbers different:
$i^2 = -1$. Every answer belongs in standard form $a + bi$.
\normalsize

\skillheading{1. Adding and subtracting}
\begin{formulabox}
\textbf{Rule:} combine like parts.
\[ (a+bi) \pm (c+di) = (a \pm c) + (b \pm d)i \]
Subtraction distributes the minus sign over \emph{both} parts of the second
number. Nothing here needs $i^2$; addition and subtraction never create one.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $(6 + 2i) + (1 - 9i)$
\step{No products appear, so this is pure collecting --- real with real,
imaginary with imaginary.}
\step{Real: $6 + 1 = 7$; \ imaginary: $2i - 9i = -7i$.}
\par
$\ans{7-7i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $(2 - 5i) - (7 + 3i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5-8i}$}
\end{tryitbox}

\skillheading{2. Multiplying complex numbers}
\begin{formulabox}
\textbf{Rule:} distribute all four products, then replace $i^2$ with $-1$ and
re-collect.
\[ (a+bi)(c+di) = (ac - bd) + (ad + bc)i \]
\textbf{Conjugates:} $(a+bi)(a-bi) = a^2 + b^2$, always real --- the outer and
inner products cancel and the $-b^2i^2$ becomes $+b^2$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $(3 + 4i)(2 - i)$
\step{Two binomials, so FOIL first and postpone every simplification until all
four products are on the page.}
\step{$6 - 3i + 8i - 4i^2 = 6 + 5i + 4$, using $-4i^2 = +4$.}
\par
$\ans{10+5i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $(1 + 6i)(2 + 3i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-16+15i}$}
\end{tryitbox}

\skillheading{3. Powers of $i$}
\begin{formulabox}
\textbf{Rule:} the powers cycle with period $4$.
\[ i^1 = i, \quad i^2 = -1, \quad i^3 = -i, \quad i^4 = 1 \]
For $i^n$, divide $n$ by $4$ and keep only the remainder $r$; then
$i^n = i^r$, with $i^0 = 1$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Simplify $i^{27}$.
\step{The exponent is large but the powers repeat every four, so only the
remainder on division by $4$ matters.}
\step{$27 = 4 \cdot 6 + 3$, remainder $3$, so $i^{27} = i^3$.}
\par
$\ans{-i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Simplify $i^{30}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1}$}
\end{tryitbox}

\skillheading{4. Complex solutions of a quadratic}
\begin{formulabox}
\textbf{Rule:} when $\Delta = b^2 - 4ac < 0$, write
$\sqrt{\Delta} = i\sqrt{|\Delta|}$ and finish the quadratic formula as usual.
\[ x = \frac{-b \pm i\sqrt{|\Delta|}}{2a} \]
The two solutions are always conjugates, so checking one checks both.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Solve $x^2 - 4x + 5 = 0$.
\step{$\Delta = 16 - 20 = -4$ is negative, so there are no real solutions and
the square root must be written with $i$.}
\step{$x = \dfrac{4 \pm i\sqrt{4}}{2} = \dfrac{4 \pm 2i}{2}$, then divide both
parts by $2$.}
\par
$\ans{x=2+i \ \text{or} \ x=2-i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Solve $x^2 + 2x + 10 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x=-1+3i \ \text{or} \ x=-1-3i}$}
\end{tryitbox}
