\sstitleblock{Reading and Comparing Multi-Digit Numbers}

\skillheading{Skill 1 --- The Value of a Digit}

\begin{formulabox}
\textbf{Rule:} value of a digit $=$ digit $\times$ its place. Each place is
worth \textbf{10 times} the place to its right:
ten-thousands $\cdot$ thousands $\cdot$ hundreds $\cdot$ tens $\cdot$ ones.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the value of the 7 in $74{,}206$.
\step{Locate the place first: counting from the ones, the 7 sits in the
ten-thousands place, so multiply the digit by $10{,}000$.}
\step{$7 \times 10{,}000 = 70{,}000$}
$\Rightarrow\quad \ans{70{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the value of the 3 in $5{,}382$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{300}$}
\end{tryitbox}

\skillheading{Skill 2 --- Number Forms}

\begin{formulabox}
\textbf{Three ways to write one number:} standard form ($24{,}601$), expanded
form (a sum of place values), and word form. An empty place gets a \textbf{zero}
in standard form.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $20{,}000 + 4{,}000 + 600 + 1$ in standard form.
\step{Each addend names one place, so fill the places from largest to smallest
and put a zero in any place with no addend.}
\step{2 ten-thousands, 4 thousands, 6 hundreds, 0 tens, 1 one:
$20{,}000 + 4{,}000 + 600 + 1 = 24{,}601$}
$\Rightarrow\quad \ans{24{,}601}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $50{,}000 + 800 + 20 + 7$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{50{,}827}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing Numbers}

\begin{formulabox}
\textbf{Rule:} \textbf{(1)} More digits wins. \textbf{(2)} Same number of
digits: line up the places and compare from the \textbf{left}; the first place
that differs decides. Symbols: $<$ less than, $>$ greater than, $=$ equal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $36{,}254$ and $36{,}245$.
\step{Both have five digits, so line up the places and scan from the left until
a place differs.}
\step{Ten-thousands 3 = 3, thousands 6 = 6, hundreds 2 = 2; tens differ:
$5 > 4$.}
$\Rightarrow\quad \ans{36{,}254 > 36{,}245}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compare $8{,}109$ and $8{,}190$ with $<$, $>$, or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8{,}109 < 8{,}190}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Never compare just the first digits. $9{,}876$ starts
with a 9, but it is \emph{smaller} than $12{,}345$ --- the five-digit number
wins because it reaches a higher place.
\end{watchoutbox}
