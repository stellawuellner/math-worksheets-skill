\sstitleblock{Two Forms of the Same Line}

\begin{formulabox}
\textbf{Standard form:} $Ax + By = C$ \quad --- both variables on the left, a
number on the right. Good for budgets and totals, and for reading intercepts.\par
\textbf{Slope-intercept form:} $y = mx + b$ \quad --- solved for $y$. Good for
reading the rate of change and for graphing.
\end{formulabox}

\skillheading{Skill 1 --- Standard form to slope-intercept form}

\begin{formulabox}
\textbf{Rule:} move the $x$-term to the right, then divide \emph{every} term
(both of them) by the coefficient of $y$. Dividing only one term is the usual
mistake.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $2x + 5y = 20$ in slope-intercept form.
\step{The target form is solved for $y$, so isolate $y$ --- subtract the
$x$-term first, and divide only at the end.}
\step{$5y = 20 - 2x$, then divide both terms by 5: $y = 4 - \frac{2}{5}x$.}

Written in $mx + b$ order: \quad $\ans{y = -\tfrac{2}{5}x + 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $3x + 4y = 24$ in slope-intercept form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -\tfrac{3}{4}x + 6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} dividing by the $y$-coefficient must hit the constant
too. From $5y = 20 - 2x$, both the 20 and the $2x$ are divided by 5.
\end{watchoutbox}

\skillheading{Skill 2 --- Slope-intercept form to standard form}

\begin{formulabox}
\textbf{Rule:} substitute the $mx + b$ expression into $Ax + By$ (choosing
$A$ and $B$ so the fraction clears). The $x$-terms cancel and what is left is
the constant $C$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $y = -\frac{4}{3}x + 6$ as $4x + 3y = C$.
\step{The denominator is 3, so multiplying $y$ by 3 clears the fraction ---
that is why $B = 3$ was chosen.}
\step{$4x + 3\left(-\frac{4}{3}x + 6\right) = 4x - 4x + 18$.}

The $x$-terms cancel, leaving the constant. \quad $\ans{4x + 3y = 18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $y = \frac{3}{2}x - 5$ as $3x - 2y = C$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x - 2y = 10}$}
\end{tryitbox}

\skillheading{Skill 3 --- Intercepts straight from standard form}

\begin{formulabox}
\textbf{Rule:} the $x$-intercept is where $y = 0$, the $y$-intercept where
$x = 0$. In $Ax + By = C$ each substitution kills one term and leaves a
one-step equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find both intercepts of $3x + 5y = 15$.
\step{Intercepts are the points where the line meets an axis, so set the
other coordinate to zero and solve what is left.}
\step{With $y = 0$: $3x = 15$. With $x = 0$: $5y = 15$.}

$x$-intercept $\ans{x = 5}$, \quad $y$-intercept $\ans{y = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find both intercepts of $2x + 7y = 14$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$ and $\ans{y = 2}$}
\end{tryitbox}
