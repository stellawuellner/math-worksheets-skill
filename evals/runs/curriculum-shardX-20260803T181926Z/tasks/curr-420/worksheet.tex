\begin{center}
\textit{Recovering amplitude, midline and period from a graph, building a
sinusoidal model from a situation, evaluating that model, and solving a
periodic equation on one cycle}
\end{center}

\noindent\textbf{Before you start.} For $y = A\sin\big(B(t - C)\big) + D$:
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item $|A| = \dfrac{\text{max} - \text{min}}{2}$ \quad and \quad
$D = \dfrac{\text{max} + \text{min}}{2}$ \quad (amplitude and midline).
\item The period is $\dfrac{2\pi}{B}$, so $B = \dfrac{2\pi}{\text{period}}$.
Measure a period between two \emph{successive maxima}. A maximum and the next
\emph{minimum} are only \textbf{half} a period apart.
\item Work in radians. Every model on this sheet uses radian measure, so a
calculator left in degree mode will return a plausible but wrong number.
\item A periodic equation almost never has one solution on a full cycle. Find
them all, then keep the ones inside the interval you were given.
\end{itemize}

\problem[5cm]{Curve G is the graph of $h(t) = 5\sin\!\left(\dfrac{\pi t}{4}\right) + 9$.
Find $h(2)$ exactly, then check your value against the graph.}

\problem[4.5cm]{Use \textbf{Figure 1}. Curve G rises to a highest value of $14$
and falls to a lowest value of $4$. Find its amplitude.}

\problem[4.5cm]{Use \textbf{Figure 1}. Find the midline of Curve G --- the
value of $y$ about which the curve oscillates.}

\problem{Use \textbf{Figure 1}. Curve G reaches successive maxima at $t = 2$
and at $t = 10$.

\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=6pt]
  \item What is the period of Curve G? \ansline
  \item Find the value of $B$ for which Curve G can be written as
        $y = A\sin(Bt) + D$. \ansline
\end{enumerate}}

\problem{A Ferris wheel has diameter $40$ metres. Its lowest point is $3$
metres above the ground, and it completes one full turn every $12$ minutes. A
rider boards at the lowest point at time $t = 0$.

\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=6pt]
  \item What is the amplitude of the rider's height model, in metres? \ansline
  \item What value of $B$ belongs in $h(t) = -A\cos(Bt) + D$? \ansline
\end{enumerate}}

\problem[7cm]{The Ferris wheel of Problem 5 is modelled by
$h(t) = -20\cos\!\left(\dfrac{\pi t}{6}\right) + 23$, with $h$ in metres and
$t$ in minutes. Find the rider's height four minutes after boarding.}

\problem[9cm]{The rider is level with the centre of that wheel when
$h(t) = 23$, which happens exactly when
$\cos\!\left(\dfrac{\pi t}{6}\right) = 0$. Find \emph{all} times $t$ in one
full turn, $0 \le t < 12$ minutes, at which this happens.}

\problem{In a harbour the water is deepest at $12$ feet, at $t = 3$ hours, and
shallowest at $4$ feet, at $t = 9$ hours. The depth is modelled by
$d(t) = A\cos\big(B(t - 3)\big) + D$.

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item Find $A$. \ansline
  \item Find $D$. \ansline
  \item Find $B$. (Careful: a high tide and the \emph{next} low tide are half a
        period apart.) \ansline
\end{enumerate}}

\problem{A different sinusoid reaches a maximum of $18$ at $x = 1$ and its next
minimum of $2$ at $x = 7$.

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item Find its amplitude. \ansline
  \item Find its midline. \ansline
  \item Find its period. \ansline
\end{enumerate}}

\problem[7cm]{Using the harbour model from Problem 8, which is
$d(t) = 4\cos\!\left(\dfrac{\pi (t - 3)}{6}\right) + 8$, find the depth of the
water at $t = 5$ hours.}

\problem[9.5cm]{Find \emph{all} times $t$ with $0 \le t < 12$ hours at which
the harbour depth $d(t) = 4\cos\!\left(\dfrac{\pi (t - 3)}{6}\right) + 8$
equals $10$ feet. There is more than one; explain briefly how the graph tells
you how many to expect.}

\problem{\textbf{Synthesis.} Two students model the same Ferris wheel from
Problem 5. Ana writes
$h(t) = -20\cos\!\left(\dfrac{\pi t}{6}\right) + 23$; Ben writes
$h(t) = 20\sin\!\left(\dfrac{\pi (t - 3)}{6}\right) + 23$.

\begin{enumerate}[label=(\alph*), itemsep=3.2cm, leftmargin=1.5cm, topsep=6pt]
  \item Evaluate \emph{both} models at $t = 5$, exactly. \ansline
  \item Explain why both students can be right, and say which parameters of a
        sinusoid a graph pins down uniquely and which it does not. \noansline
\end{enumerate}}
