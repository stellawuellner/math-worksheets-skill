\sstitleblock{Two-Step Equations \& Checking Solutions}

\skillheading{Skill 1 --- Solve $ax + b = c$}

\begin{formulabox}
\textbf{Rule:} undo in reverse order of operations --- undo the added or subtracted
number \emph{first}, then undo the multiplication. Same move on both sides, every
time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3x + 4 = 19$
\step{Undo the operations in reverse order: the $+\,4$ goes first, then the times 3.}
\step{Subtract 4 from both sides, then divide by 3: \ $x = (19 - 4)/3 = 5$.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2x + 7 = 19$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Solve $\dfrac{x}{a} + b = c$}

\begin{formulabox}
\textbf{Rule:} same order --- undo the added number first, then undo the division
by multiplying both sides by $a$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{2} - 3 = 5$
\step{The subtraction is the outer operation, so add 3 to both sides first, then undo the division by 2.}
\step{$x = (5 + 3) \cdot 2 = 16$.}

$\Rightarrow\quad \ans{x = 16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{5} + 1 = 4$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 15}$}
\end{tryitbox}

\skillheading{Skill 3 --- A negative coefficient}

\begin{formulabox}
\textbf{Rule:} treat $-ax$ like any coefficient: isolate it, then divide both
sides by the \emph{negative} number, sign included.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $9 - 2x = 1$
\step{Move the plain number away first, so the term with $x$ stands alone before you divide by its negative coefficient.}
\step{Subtract 9 from both sides: $-2x = 1 - 9$. Divide by $-2$: \ $x = (1 - 9)/(-2) = 4$.}

$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $11 - 3x = 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} divide by the coefficient \emph{with its sign}. Dividing
$-2x = -8$ by 2 instead of $-2$ leaves the sign error in your answer.
\end{watchoutbox}

\skillheading{Skill 4 --- Check by substitution}

\begin{formulabox}
\textbf{Rule:} a value is a solution only if substituting it makes the equation
\emph{true}: left side value $=$ right side value. Checking catches sign and
order-of-steps errors immediately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $x = 5$ a solution of $3x - 2 = 13$?
\step{Substitute the candidate value into the left side and compare the result with the right side of the equation.}
\step{$3 \cdot 5 - 2 = \ans{13}$ --- it equals the right side, so $x = 5$ \emph{is} the solution.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $x = 3$ a solution of $5x + 1 = 20$? Evaluate $5x + 1$
at $x = 3$ and compare.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$ --- not 20, so no}
\end{tryitbox}
