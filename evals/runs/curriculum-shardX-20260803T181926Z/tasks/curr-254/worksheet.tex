\wstitleblock{Sign and Like-Term Errors in Polynomial Operations}{Algebra 1 --- Expressions and polynomial operations}{}

\noindent\textbf{The three errors this sheet hunts for:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{The stranded minus.} A subtraction sign in front of a parenthesis multiplies \emph{every} term inside it, not only the first one.
  \item \textbf{The false like term.} Terms combine only when their variable parts are identical: $x^2$ and $x$ never merge, however tempting the coefficients look.
  \item \textbf{The missing middle term.} Squaring a binomial is a product, so the two cross terms survive; $(a-b)^2$ is not $a^2 + b^2$.
\end{itemize}

\noindent On every find-and-fix item, quote the exact step that goes wrong,
write the correct simplification, and then confirm it by substituting the value
the problem names. A substitution check is the fastest way to prove a sign is
wrong without re-reading the whole line.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3.5cm]{Combine like terms: $7x^2 + 3x - 5x^2 + 8x - 2$.}

\problem[4cm]{Subtract, showing the line in which the subtraction sign has been
distributed: $(5x^2 - 3x + 4) - (2x^2 + 7x - 6)$.}

\problem[4cm]{Multiply and simplify: $-3x(2x^2 - 4x + 5)$.}

\problem[6cm]{\textbf{Find and fix the mistake.} A student writes
\[ (4x^2 - 6x + 1) - (x^2 - 2x - 5) = 3x^2 - 8x - 4 \]
Name the step that goes wrong, write the correct simplification, then evaluate
your corrected answer at $x = 2$ to confirm it.}

\problem[6cm]{\textbf{Find and fix the mistake.} A student writes
\[ -2x(3x - 7) + 5(x - 4) = -6x^2 - 9x - 20 \]
Exactly one term is mis-signed. Name it, expand the left side correctly, then
evaluate your corrected answer at $x = 1$ to confirm it.}

\problem[5cm]{Factor $x^2 - 5x - 24$. Then check the factorisation by evaluating
your factored form at $x = 1$ and comparing it with the value of $x^2 - 5x - 24$
at $x = 1$.}

\problem[6cm]{\textbf{Find and fix the mistake.} A student claims that
$(2x - 3)^2$ and $4x^2 + 9$ are the same expression. Decide whether they are,
expand $(2x - 3)^2$ correctly, and support your decision by evaluating your
expansion at $x = 1$.}

\problem[5.5cm]{\textbf{Explain.} A classmate says: ``When you subtract
polynomials you only have to flip the sign of the first term inside the
parentheses, and anyway $3x^2 + 5x$ is just $8x^2$.'' Write a short paragraph
correcting both claims. State the subtraction rule properly, give one
substitution that proves $3x^2 + 5x$ and $8x^2$ are different, and say what has
to be true of two terms before they may be combined. \noansline}
