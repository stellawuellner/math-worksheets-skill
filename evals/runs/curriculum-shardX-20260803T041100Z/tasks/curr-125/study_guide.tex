\sstitleblock{Ordering Fractions and Mixed Numbers}

\noindent\small Fractions can only be ordered once they are in \emph{one}
form. Rewrite first, compare second.\normalsize

\skillheading{1. Equivalent forms: rewriting without changing value}

\begin{formulabox}
\textbf{Rule:} multiplying (or dividing) the numerator \emph{and} the
denominator by the same non-zero number never changes the value.\\[3pt]
\hspace*{0.4cm}$\dfrac{a}{b} = \dfrac{a \times k}{b \times k}$
\qquad find $k$ from the denominators, then apply it to the numerator.\\[3pt]
\textbf{Mixed number $\rightarrow$ fraction:} each whole is one full
denominator, so multiply the whole part by the denominator and add the
numerator.\\[3pt]
\textbf{Fraction $\rightarrow$ mixed number:} divide; the quotient is the whole
part and the remainder stays over the same denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Rewrite $\tfrac{2}{3}$ with denominator $15$.
\step{The denominator is fixed for me, so find the scale factor from the denominators first and then apply the same factor upward.}
\step{$3$ scaled to $15$ is a factor of $5$, so the numerator scales the same way: $2 \times 5 = 10$.}
$\tfrac{2}{3} = \ans{\,\dfrac{10}{15}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Rewrite $\tfrac{3}{5}$ with denominator $20$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\,\dfrac{12}{20}}$}
\end{tryitbox}

\skillheading{2. Comparing a pair: benchmark, then common denominator}

\begin{formulabox}
\textbf{Try these in order --- stop at the first one that decides it:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=1pt]
  \item \textbf{Benchmark.} Is each number below a half, above a half, or above
        one? Half of a fraction's denominator, written as the numerator, is
        exactly one half.
  \item \textbf{Gap to one.} If both sit just below one, the one with the
        \emph{smaller} missing piece is larger.
  \item \textbf{Matching tops or bottoms.} Same numerator: bigger denominator
        means smaller pieces, so a smaller number. Same denominator: compare
        the numerators.
  \item \textbf{Common denominator.} Always works. Multiply the two
        denominators if no smaller common one is obvious.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Which is greater, $\tfrac{4}{9}$ or $\tfrac{5}{8}$?
\step{The two numbers look close, so test the half benchmark before doing any arithmetic --- it is the cheapest test that can settle it.}
\step{Half of $9$ would be a numerator between $4$ and $5$, so $\tfrac{4}{9}$ is below a half; half of $8$ is $4$, so $\tfrac{5}{8}$ is above a half.}
$\ans{\,\tfrac{4}{9} < \tfrac{5}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Which is greater, $\tfrac{7}{12}$ or $\tfrac{5}{9}$?
(Both are just above a half, so go to a common denominator.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\,\tfrac{7}{12} > \tfrac{5}{9}}$}
\end{tryitbox}

\skillheading{3. Ordering a whole set across representations}

\begin{formulabox}
\textbf{Method:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=1pt]
  \item \textbf{Sort by size class first.} Separate the numbers below one from
        those above one --- often half the ordering is done for free.
  \item \textbf{Convert to one form.} Mixed numbers are usually easiest: turn
        every fraction greater than one into a mixed number.
  \item \textbf{Compare whole parts, then fraction parts.} Only when the whole
        parts tie do you need a common denominator for the fraction parts.
  \item \textbf{Write the answer in the original forms}, joined by $<$ or $>$,
        so the reader can see which given number went where.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Order $\tfrac{5}{6}$, $1\tfrac{1}{4}$, $\tfrac{7}{8}$
from least to greatest.
\step{Size class settles part of it immediately: only one of these is above one, so it must be the largest and the other two need a real comparison.}
\step{$\tfrac{5}{6}$ and $\tfrac{7}{8}$ are both below one, and over twenty-fourths they are $\tfrac{20}{24}$ and $\tfrac{21}{24}$.}
$\ans{\,\tfrac{5}{6} < \tfrac{7}{8} < 1\,\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Order $\tfrac{3}{4}$, $\tfrac{2}{3}$, $\tfrac{5}{6}$
from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\,\tfrac{2}{3} < \tfrac{3}{4} < \tfrac{5}{6}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a bigger denominator does \emph{not} mean a bigger
number --- it means smaller pieces. And never compare numerators across
different denominators; rewrite first.
\end{watchoutbox}
