\sstitleblock{Signed Rational Numbers: Subtraction \& Sign Errors}

\skillheading{Skill 1 --- Subtracting signed numbers}

\begin{formulabox}
\textbf{Rule:} \quad $a - b = a + (-b)$ \quad --- subtracting means \emph{adding the opposite}.\par
Same signs: add the absolute values and keep the sign. Different signs: subtract the
absolute values and keep the sign of the number farther from zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $-6 - 9$
\step{We are subtracting a positive number from a negative one, so rewrite the subtraction as adding the opposite and use the addition sign rules.}
\step{$-6 - 9 = -6 + (-9)$. Same signs: add $6 + 9 = 15$ and keep the negative sign.}

$\Rightarrow\quad -6 - 9 = \ans{-15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $-4 - 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-13}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the order matters --- $a - b$ and $b - a$ are opposites.
Never swap the two numbers just because one is negative.
\end{watchoutbox}

\skillheading{Skill 2 --- Subtracting a negative}

\begin{formulabox}
\textbf{Rule:} \quad $a - (-b) = a + b$ \quad --- two minus signs in a row make a plus.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $-3 - (-10)$
\step{The number being subtracted is negative, so change the double negative into addition before computing anything.}
\step{$-3 - (-10) = -3 + 10$. Different signs: subtract $10 - 3 = 7$ and keep the plus sign, because $10$ is farther from zero.}

$\Rightarrow\quad -3 - (-10) = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $-8 - (-3)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$}
\end{tryitbox}

\skillheading{Skill 3 --- A minus in front of parentheses}

\begin{formulabox}
\textbf{Rule:} \quad $a - (b - c) = a - b + c$ \quad --- the minus applies to
\emph{every} term inside the parentheses, so each inside term flips its sign.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $4 - (6 - 13)$
\step{A minus sign sits in front of a parenthesis, so distribute it to every term inside, flipping each sign, then combine left to right.}
\step{$4 - (6 - 13) = 4 - 6 + 13 = 11$, combining the terms from left to right.}

$\Rightarrow\quad 4 - (6 - 13) = \ans{11}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $2 - (5 - 9)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} distributing the minus to only the \emph{first} term inside is
the most common sign error on this sheet. Flip every sign inside, or none.
\end{watchoutbox}
