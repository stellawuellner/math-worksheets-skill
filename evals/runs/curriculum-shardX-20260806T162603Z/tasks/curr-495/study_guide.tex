% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Motion given as a vector}

\begin{formulabox}
\textbf{From the velocity $\langle x'(t),\, y'(t)\rangle$:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item speed $= \sqrt{\big(x'\big)^2 + \big(y'\big)^2}$ --- a length, so the components go under one root
  \item position $= $ starting point $+$ the integral of each component separately
  \item distance travelled $= \int_a^b \text{speed}\,dt$; displacement compares endpoints only
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A particle has $x'(t) = t$ and $y'(t) = 2$. Find its speed at $t = 3$.
\step{Speed is the length of the velocity vector, so evaluate both components at that instant and take the magnitude.}
\step{At $t = 3$ the velocity is $\langle 3, 2\rangle$, so the speed is $\sqrt{9 + 4} = \sqrt{13}$.}

$\Rightarrow\quad \ans{3.606}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A particle has $x'(t) = t$ and $y'(t) = 5$. Find its speed at $t = 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.385}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Where two polar curves meet}

\begin{formulabox}
\textbf{Two steps, always both.} (1) Set $r_1(\theta) = r_2(\theta)$ and solve on $[0, 2\pi)$. (2) Check the pole separately --- each curve may reach $r = 0$ at a different $\theta$, so the algebra in step 1 cannot see it. For the area between two curves on an interval where one is outside the other, $A = \frac{1}{2}\int \big(r_{\text{out}}^2 - r_{\text{in}}^2\big)\,d\theta$ between consecutive intersection angles.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find every $\theta$ in $[0, 2\pi)$ where $r = 2$ meets $r = 4\sin\theta$.
\step{Both curves are given as $r$ in terms of $\theta$, so equate the two expressions and solve the resulting trigonometric equation over one full turn.}
\step{$4\sin\theta = 2$ gives $\sin\theta = \tfrac{1}{2}$, whose solutions on $[0, 2\pi)$ are the two first-and-second-quadrant angles.}

$\Rightarrow\quad \ans{\theta = \frac{\pi}{6},\ \frac{5\pi}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find every $\theta$ in $[0, 2\pi)$ where $r = 3$ meets $r = 6\cos\theta$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = \frac{\pi}{3},\ \frac{5\pi}{3}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Accumulating a rate}

\begin{formulabox}
\textbf{Rule:} the total accumulated between $t = a$ and $t = b$ is $\displaystyle\int_a^b r(t)\,dt$ --- integrate the rate, do not divide by anything. Check the units: litres per minute times minutes gives litres. If the question instead asks \emph{when} the rate reaches some level, that is an equation to solve, not an integral.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Water enters at $r(t) = 4 + 6t$ litres per minute. How much enters in the first $3$ minutes?
\step{"How much enters" asks for a total, so integrate the rate across the given time interval.}
\step{$\displaystyle\int_0^3 (4 + 6t)\,dt = \big[4t + 3t^2\big]_0^3 = 12 + 27$.}

$\Rightarrow\quad \ans{39}$ litres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Water enters at $r(t) = 5 + 4t$ litres per minute. How much enters in the first $4$ minutes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{52}$ litres}
\end{tryitbox}
