\wstitleblock{Radicals Do Not Split Across a Sum}{Algebra 2}{}

\begin{center}
\textit{Testing a suspect rule with numbers, splitting radicals over products and quotients only, and finding and fixing the split-over-a-sum error}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=3pt]
  \item A radical comes apart over a \emph{product} and a \emph{quotient}: $\sqrt{ab} = \sqrt{a}\,\sqrt{b}$ and $\sqrt{a/b} = \sqrt{a}/\sqrt{b}$ for $a, b \ge 0$.
  \item There is \textbf{no} matching rule for a sum. Whenever you are tempted to write $\sqrt{a+b}$ as $\sqrt{a}+\sqrt{b}$, test it with two numbers first.
  \item The same warning applies in exponent notation: $(a+b)^{1/2}$ is not $a^{1/2}+b^{1/2}$.
  \item Four problems show work a student handed in. Compute the correct values first, then say what went wrong.
\end{itemize}
