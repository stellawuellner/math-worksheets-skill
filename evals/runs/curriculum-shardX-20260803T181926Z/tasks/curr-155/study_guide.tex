\sstitleblock{Choosing a Ratio Representation}

\skillheading{1. The ratio table: multiply, never add}

\begin{formulabox}
\textbf{Rule:} equivalent ratios come from multiplying \emph{both} parts by
the same number. $a : b$ and $ka : kb$ are the same ratio; $a + k : b + k$ is
not.\\[3pt]
\textbf{Use a table when} the second number is a friendly multiple of the
first (double it, triple it, halve it). Write the multiplier above each arrow
so you can see the jump.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A paint mix is $4$ parts blue to $3$ parts white. How much
white goes with $20$ parts blue?
\step{The new blue amount is a whole-number multiple of the old one, so a
ratio table is the quickest tool --- find the multiplier, then apply it.}
\step{$20 \div 4 = 5$, so the multiplier is $5$; apply it to the white part:
$3 \times 5$.}

$\Rightarrow\quad \ans{15}$ parts white
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A mix is $7$ parts sand to $4$ parts cement. How much cement
goes with $21$ parts sand?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{2. The unit rate: the cost of exactly one}

\begin{formulabox}
\textbf{Rule:} divide to get ``per one''. Total cost $\div$ number of items
$=$ cost per item; distance $\div$ time $=$ distance per unit of time.\\[3pt]
\textbf{Use a unit rate when} the new amount is \emph{not} a friendly multiple,
or when two offers have different pack sizes. Once you know the per-one value,
every other quantity is one multiplication.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $4$ smoothies cost $9.60$ dollars. What does one cost?
\step{The question asks for the price of a single item, which is the
definition of a unit rate --- divide the total by the count.}
\step{$9.60 \div 4 = 2.40$ dollars.}

$\Rightarrow\quad \ans{2.40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $8$ smoothies cost $21.60$ dollars. What does one cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.70}$}
\end{tryitbox}

\skillheading{3. The proportion: one unknown, one known ratio}

\begin{formulabox}
\textbf{Rule:} set the two ratios equal and solve:
$\dfrac{x}{\text{new}} = \dfrac{\text{old part}}{\text{old whole}}$, so
$x = \text{new} \cdot \dfrac{\text{old part}}{\text{old whole}}$.\\[3pt]
\textbf{Keep the units in the same places} on both sides --- cups over cups,
kilometres over kilometres. A flipped side is the most common error here.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A recipe for $6$ people uses $5$ cups of stock. How much
stock for $18$ people?
\step{There is one unknown and one known ratio, which is exactly what a
proportion is for.}
\step{$\dfrac{x}{18} = \dfrac{5}{6}$, so $x = 18 \cdot \dfrac{5}{6} =
3 \cdot 5$.}

$\Rightarrow\quad \ans{x = 15}$ cups
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A recipe for $8$ people uses $7$ cups of stock. How much for
$24$ people?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 21}$}
\end{tryitbox}

\skillheading{4. Comparing two rates fairly}

\begin{formulabox}
\textbf{Rule:} two rates can only be compared once they are in the \emph{same}
form. Convert both to ``per one'' of the same thing, then compare.\\[3pt]
\textbf{Say which direction wins before you compare:} for price per item the
\emph{smaller} number is better; for pages per minute the \emph{larger} number
is better. Equal ratios are a real possible answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $3$ pens for $4.50$ dollars, or $5$ pens for $7.00$ dollars
--- which is the better buy?
\step{The pack sizes differ, so the totals say nothing on their own; convert
both to price per pen and remember that here the smaller rate wins.}
\step{$4.50 \div 3 = 1.50$ and $7.00 \div 5 = 1.40$, so the five-pack is
cheaper per pen.}

$\Rightarrow\quad \ans{1.50 > 1.40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $3$ pens for $4.20$ dollars, or $5$ pens for $7.50$ dollars?
Compare the price per pen.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1.40 < 1.50}$ --- the three-pack wins}
\end{tryitbox}
