\sstitleblock{Comparing Three-Digit Numbers}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Comparing with symbols}

\begin{formulabox}
\textbf{Rule:} Compare place by place, starting with the biggest place.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Hundreds first. More hundreds wins --- stop there.
  \item Tie? Compare tens. Still tied? Compare ones.
  \item The open mouth of $<$ or $>$ always faces the bigger number.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$, $>$, or $=$: compare 463 and 437.
\step{Compare hundreds first: both have four hundreds, so move to the tens.}
\step{463 has six tens and 437 has only three tens, so 463 is greater.}
$\Rightarrow\quad \ans{463 > 437}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$: compare 528 and 582.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{528 < 582}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Big tens or ones can never beat a hundreds
difference. Always settle the hundreds \emph{before} looking at anything
else.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Ordering several numbers}

\begin{formulabox}
\textbf{Rule:} To put numbers in order, compare them two at a time.
Find the least, write it first, then order what is left the same way.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Order least to greatest: 341, 314, 431.
\step{Sort by hundreds first: 431 has four hundreds, so it is greatest; the other two tie at three hundreds.}
\step{Break the tie with tens: 314 has one ten, 341 has four tens, so 314 comes first.}
$\Rightarrow\quad \ans{314,\ 341,\ 431}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Order least to greatest: 268, 286, 226.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{226,\ 268,\ 286}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Comparing in a story}

\begin{formulabox}
\textbf{Rule:} In a word problem, first pull out the two quantities and their
units, write the comparison with a symbol, then answer the question in words
(\emph{who} or \emph{which one}).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A young oak tree is 175 cm tall and a maple is 157 cm
tall. Which tree is taller?
\step{Pull out the two heights and compare hundreds first: they tie at one hundred, so the tens decide.}
\step{175 cm has seven tens and 157 cm has five tens, so the oak is taller.}
$\Rightarrow\quad \ans{175 > 157}$, so the oak is taller.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Ana scored 342 points and Ben scored 324 points.
Compare with a symbol. Who scored more?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{342 > 324}$ --- Ana}
\end{tryitbox}
