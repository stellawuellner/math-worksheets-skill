\sstitleblock{Turnaround and Break-Apart Fact Strategies}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Turnaround Facts}

\begin{formulabox}
\textbf{Rule --- the order of the factors does not change the product.} \ If you know one fact, you know its turnaround for free. Picture an array of counters: turning the array on its side swaps the rows and columns but does not add or remove a single counter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad You know $6 \times 3 = 18$. What is $3 \times 6$?

\step{The same two factors are being multiplied, just written in the other order, so no new counting is needed.}
\step{Turn the fact around: $3 \times 6$ has the same product as $6 \times 3$.}

$\Rightarrow\quad \ans{18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad You know $5 \times 4 = 20$. What is $4 \times 5$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Breaking a Factor Apart}

\begin{formulabox}
\textbf{Rule --- split one factor, multiply each piece, then put the pieces together.} \ Add the two part-products when you split by adding, and take one away when you split by subtracting:
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item Split by adding: $6 \times 8 = (6 \times 5) + (6 \times 3)$
  \item Split by subtracting: $9 \times 7 = (10 \times 7) - 7$
\end{itemize}
Split off a five or a ten --- those are the pieces you can already count.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $6 \times 8$ by breaking the $8$ into $5 + 3$.

\step{Eights are hard to count by, but fives and threes are easy, so split the harder factor into two friendly pieces.}
\step{$6 \times 5 = 30$ and $6 \times 3 = 18$, and the two part-products go back together by adding.}

$\Rightarrow\quad \ans{48}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Use $9 \times 7 = (10 \times 7) - 7$ to find $9 \times 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{63}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} when you go down from a ten, take away a whole \emph{group}, not one. For $9 \times 7$ you remove one group of seven, so the step is $70 - 7$.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Dividing with a Missing-Factor Fact}

\begin{formulabox}
\textbf{Rule --- every division fact is a multiplication fact with a hole in it.} \ To work out a division, ask what fills the hole in the matching multiplication fact, then use the times table you already know.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is $18 \div 3$?

\step{Dividing by three asks how many threes make eighteen, which is the same as filling the hole in a multiplication fact.}
\step{Count the threes: $3, 6, 9, 12, 15, 18$ --- that is six of them, so the missing factor is six.}

$\Rightarrow\quad \ans{n = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What is $28 \div 4$? Think: $4 \times \underline{\hspace{0.9cm}} = 28$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 7}$}
\end{tryitbox}
