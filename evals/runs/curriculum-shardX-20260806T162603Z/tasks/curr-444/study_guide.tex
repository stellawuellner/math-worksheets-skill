\sstitleblock{Index-Shift and Convergence-Condition Errors}

\skillheading{Skill 1 --- Counting the terms a sum really has}

\begin{formulabox}
\textbf{Rule:} a sum from $p$ to $q$ has $q - p + 1$ terms, not $q$ of them. Standard formulas
such as $1 + 2 + \cdots + n = \dfrac{n(n+1)}{2}$ start at $1$; to start higher, compute the
whole thing and subtract the terms you do not want.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\sum_{k=3}^{7} k$.
\step{The closed formula starts at $1$, so build the sum to $7$ and remove the part below $3$.}
\step{$\dfrac{7 \cdot 8}{2} - (1 + 2) = 28 - 3$, and the count checks out: $7 - 3 + 1 = 5$ terms.}

$\Rightarrow\quad \displaystyle\sum_{k=3}^{7} k = \ans{25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\sum_{k=4}^{8} k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$}
\end{tryitbox}

\skillheading{Skill 2 --- Shifting an index moves both limits}

\begin{formulabox}
\textbf{Rule:} substituting $m = n - c$ replaces $n$ by $m + c$ in the term \emph{and} lowers
both limits by $c$. Check the shift by comparing the first term before and after --- they must
be the same number, and the term count must not change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\sum_{n=1}^{3} 2^{\,n}$, then check it against the shifted form $\displaystyle\sum_{m=0}^{2} 2^{\,m+1}$.
\step{Both forms should list the same three terms, so evaluate the original and use the count as the check on the shift.}
\step{$2 + 4 + 8$, and the shifted form gives $2^{1} + 2^{2} + 2^{3}$ --- the same three terms.}

$\Rightarrow\quad \displaystyle\sum_{n=1}^{3} 2^{\,n} = \ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\sum_{n=2}^{4} 2^{\,n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{Skill 3 --- When a geometric series has a sum}

\begin{formulabox}
\textbf{Rule:} $\displaystyle\sum a r^{\,n}$ converges only when $|r| < 1$, and then the sum is
$\dfrac{a}{1-r}$, where $a$ is the series' \emph{first term as written}. If $|r| \ge 1$ there
is no finite sum, and applying the formula anyway can return a nonsense value such as a
negative number for a series of positive terms.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\sum_{n=0}^{\infty} \left(\tfrac{1}{3}\right)^{n}$.
\step{Check the ratio first: $r = 1/3$ has absolute value below $1$, so the sum formula is legal here.}
\step{The first term is the $n = 0$ term, $1$, so the sum is $\dfrac{1}{1 - 1/3} = \dfrac{1}{2/3}$.}

$\Rightarrow\quad$ the sum is $\ans{\tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\sum_{n=1}^{\infty} 4\left(\tfrac{1}{2}\right)^{n-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 4 --- Limits of rational sequences}

\begin{formulabox}
\textbf{Rule:} for a ratio of polynomials in $n$, divide top and bottom by the highest power of
$n$. Constants then vanish but \emph{coefficients do not}: the limit is the ratio of the
leading coefficients, not $1$. A sequence whose terms alternate in sign has no limit at all,
however well the sizes settle down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\lim_{n \to \infty} \frac{5n - 1}{2n + 3}$.
\step{Top and bottom both grow like $n$, so divide through by $n$ and let the constant pieces die.}
\step{$\dfrac{5 - 1/n}{2 + 3/n} \to \dfrac{5 - 0}{2 + 0}$.}

$\Rightarrow\quad$ the limit is $\ans{\tfrac{5}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\lim_{n \to \infty} \frac{2n + 7}{n - 4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}
