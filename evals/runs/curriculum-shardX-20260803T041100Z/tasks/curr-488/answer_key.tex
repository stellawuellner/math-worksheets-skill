\problem{\textbf{Square cross sections over the region under $y = x$, $0 \le x \le 4$.}

\textbf{Step 1.} At position $x$ the base runs from the $x$-axis up to $y = x$,
so the slice is a square of side $s = x$ and area $A(x) = x^2$.

\textbf{Step 2.} $V = \displaystyle\int_0^4 x^2\,dx
= \left[\frac{x^3}{3}\right]_0^4 = \frac{64}{3}$.

\textbf{Step 3.} Sanity check: the solid sits inside the $4 \times 4 \times 4$
box, and $\frac{64}{3} \approx 21.3$ is well under $64$ $\checkmark$

\textit{Common wrong answer:} $8$ --- that is the area of the triangular base,
not a volume. \ans{\dfrac{64}{3}\ \text{cubic units}}}

\problem{\textbf{Semicircular cross sections over the region under $y = \sqrt{x}$, $0 \le x \le 9$.}

\textbf{Step 1.} The base at position $x$ has height $\sqrt{x}$, and that
segment is the \emph{diameter} of the semicircle, so the radius is
$\tfrac{1}{2}\sqrt{x}$ and
$A(x) = \tfrac{1}{2}\pi r^2 = \tfrac{\pi}{8}\left(\sqrt{x}\right)^2
= \tfrac{\pi x}{8}$.

\textbf{Step 2.} $V = \displaystyle\int_0^9 \frac{\pi x}{8}\,dx
= \frac{\pi}{8}\left[\frac{x^2}{2}\right]_0^9 = \frac{\pi}{8}\cdot\frac{81}{2}$.

\textbf{Step 3.} That is $\dfrac{81\pi}{16} \approx 15.9$.

\textit{Common wrong answer:} about $63.6$, from taking $\sqrt{x}$ as the
radius rather than the diameter --- which quadruples the area of every slice.
\ans{\dfrac{81\pi}{16}\ \text{cubic units}}}

\problem{\textbf{Shells: $y = x^2$ on $[0,3]$ about the $y$-axis.}

\textbf{Step 1.} A vertical strip at $x$ has radius $x$ from the axis of
revolution and height $x^2$, so the shell contributes
$2\pi x \cdot x^2\,dx$.

\textbf{Step 2.} $V = \displaystyle\int_0^3 2\pi x^3\,dx
= 2\pi\left[\frac{x^4}{4}\right]_0^3 = 2\pi \cdot \frac{81}{4}$.

\textbf{Step 3.} $V = \dfrac{81\pi}{2} \approx 127.2$.

\textit{Common wrong answer:} $20.25$ --- the $2\pi$ circumference factor was
dropped. \ans{\dfrac{81\pi}{2}\ \text{cubic units}}}

\problem{\textbf{Square cross sections over the half-disc under $y = \sqrt{16-x^2}$.}

\textbf{Step 1.} The base is the upper half of the circle of radius $4$, so $x$
runs from $-4$ to $4$ and the slice at $x$ is a square of side
$s = \sqrt{16 - x^2}$.

\textbf{Step 2.} $A(x) = s^2 = 16 - x^2$ --- squaring removes the radical,
which is why square cross sections on a circular base are pleasant.

\textbf{Step 3.} $V = \displaystyle\int_{-4}^{4}\left(16 - x^2\right)dx
= \left[16x - \frac{x^3}{3}\right]_{-4}^{4}
= \frac{256}{3} \approx 85.3$.

\textit{Common wrong answer:} $8\pi \approx 25.1$, from integrating the side
length instead of its square. \ans{\dfrac{256}{3}\ \text{cubic units}}}

\problem{\textbf{Shells: $y = \sqrt{x}$ on $[0,4]$ about the $y$-axis.}

\textbf{Step 1.} Radius $x$, height $\sqrt{x}$, so the integrand is
$2\pi x\sqrt{x} = 2\pi x^{3/2}$.

\textbf{Step 2.} $V = \displaystyle\int_0^4 2\pi x^{3/2}\,dx
= 2\pi\left[\frac{2}{5}x^{5/2}\right]_0^4 = 2\pi \cdot \frac{2}{5}\cdot 32$.

\textbf{Step 3.} $V = \dfrac{128\pi}{5} \approx 80.4$.

\textit{Common wrong answer:} $8\pi$ --- discs about the $x$-axis answer a
different question entirely. \ans{\dfrac{128\pi}{5}\ \text{cubic units}}}

\problem{\textbf{Equilateral-triangle cross sections over the region under $y = 4-x^2$.}

\textbf{Step 1.} The parabola meets the $x$-axis at $x = \pm 2$, and the slice
at $x$ is an equilateral triangle of side $s = 4 - x^2$, so
$A(x) = \frac{\sqrt{3}}{4}\left(4 - x^2\right)^2$.

\textbf{Step 2.} $\displaystyle\int_{-2}^{2}\left(4-x^2\right)^2 dx
= \int_{-2}^{2}\left(16 - 8x^2 + x^4\right)dx = \frac{512}{15}$.

\textbf{Step 3.} Multiply by the constant:
$V = \frac{\sqrt{3}}{4}\cdot\frac{512}{15} = \frac{128\sqrt{3}}{15}$.

\textit{Common wrong answer:} $34.13$, the square-cross-section volume --- the
$\frac{\sqrt{3}}{4}$ factor was never applied.
\ans{\dfrac{128\sqrt{3}}{15} \approx 14.78\ \text{cubic units}}}

\problem{\textbf{Shells between $y = 2x$ and $y = x^2$ about the $y$-axis.}

\textbf{Step 1.} The curves meet where $x^2 = 2x$, that is $x^2 - 2x = 0$, so
$x(x-2) = 0$. \ans{x = 0 \ \text{and} \ x = 2}

\textbf{Step 2.} On $[0,2]$ the line is above the parabola, so the shell height
is the gap $2x - x^2$, and the radius is $x$:
$V = \displaystyle\int_0^2 2\pi x\left(2x - x^2\right)dx$.

\textbf{Step 3.} $= 2\pi\left[\frac{2x^3}{3} - \frac{x^4}{4}\right]_0^2
= 2\pi\left(\frac{16}{3} - 4\right) = 2\pi\cdot\frac{4}{3}$, so
$V = \dfrac{8\pi}{3} \approx 8.4$.

\textit{Common wrong answer:} about $33.5$, from using $2x$ alone as the shell
height and forgetting that the parabola cuts the region off from below.
\ans{\dfrac{8\pi}{3}\ \text{cubic units}}}

\problem{\textbf{Choosing the method: $y = x^2$, the $y$-axis and $y = 4$, about the $y$-axis.}

\textbf{Step 1.} Revolving about the $y$-axis with the region described by
$y$ as a function of $x$ is the shell case: shells integrate in $x$, so no
rewriting of $y = x^2$ as $x = \sqrt{y}$ is needed, and the region's left
boundary (the $y$-axis) is the axis of revolution, so there is no inner radius
to subtract.

\textbf{Step 2.} A strip at $x$ runs from the parabola up to $y = 4$, so the
height is $4 - x^2$; the region ends where $x^2 = 4$, at $x = 2$. Hence
$V = \displaystyle\int_0^2 2\pi x\left(4 - x^2\right)dx$.

\textbf{Step 3.} $= 2\pi\left[2x^2 - \frac{x^4}{4}\right]_0^2
= 2\pi\left(8 - 4\right) = 8\pi \approx 25.1$.

\textit{Manual review:} the written justification is an open response. Accept
any argument that shells integrate in $x$ so the given function needs no
inversion, or that washers in $y$ would require $x = \sqrt{y}$ first --- both
are correct reasons, and a student who sets up the washer integral in $y$
correctly and gets $8\pi$ has also answered the question.
\ans{8\pi\ \text{cubic units}}}
