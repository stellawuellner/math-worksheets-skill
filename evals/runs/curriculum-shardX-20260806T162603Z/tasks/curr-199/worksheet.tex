\wstitleblock{Biased Samples and Mismatched Denominators}{Grade 7}{}

\begin{center}
\textit{Find and fix the mistake: wrong totals under a probability, samples that
were never random, and counts that changed after a draw}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Each problem shows somebody's answer. Some are wrong. Your job is
        \textbf{both} halves: give the corrected value, \emph{and} name the
        mistake in words.
  \item Two questions catch almost every error on this page:\par
        \emph{Does the bottom of the fraction count every outcome the question
        chooses from?}\par
        \emph{Did every member of the group have the same chance of being in
        the sample?}
  \item Give probabilities as fractions in lowest terms.
\end{itemize}

\problem[7cm]{A bag holds 5 red counters and 7 blue counters, and one is drawn
at random. Sam writes:\par\vspace{3pt}
\textit{``There are 5 red and 7 blue, so the probability of red is
$\frac{5}{7}$.''}\par\vspace{3pt}
\begin{itemize}
  \item[(a)] the corrected probability of red: \ansblank
  \item[(b)] Name Sam's mistake, in the work space, in one or two sentences.
\end{itemize}}

\problem[7cm]{A teacher wants to know how many minutes the students at the
school spend on homework each night. She asks the 10 students who stayed late in
the library. Their answers, in minutes, were:\par\vspace{4pt}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|c|c|}
\hline
80 & 90 & 100 & 85 & 95 & 110 & 90 & 100 & 105 & 95 \\ \hline
\end{tabular}
\end{center}\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the mean of this sample, in minutes: \ansblank
  \item[(b)] Explain, in the work space, why this mean is a misleading estimate
             for the whole school.
\end{itemize}}

\problem[7cm]{A box holds 4 green and 6 yellow sweets. Ana eats one green sweet,
then draws one of the rest at random. She writes:\par\vspace{3pt}
\textit{``The probability of green is still $\frac{4}{10}$.''}\par\vspace{3pt}
\begin{itemize}
  \item[(a)] the corrected probability of green: \ansblank
  \item[(b)] Name Ana's mistake, in the work space.
\end{itemize}}

\problem[7cm]{A spinner has 12 equal sections: 4 red, 3 blue and 5 green. Ben is
asked for the probability that it does \textbf{not} land on red, and
writes:\par\vspace{3pt}
\textit{``Not red is $\frac{4}{12}$.''}\par\vspace{3pt}
\begin{itemize}
  \item[(a)] the corrected probability of not red: \ansblank
  \item[(b)] Name Ben's mistake, in the work space.
\end{itemize}}

\problem[8cm]{A town has 20000 residents. A student surveys 40 people as they
leave a cinema, and 30 of them say they go to the cinema every
week.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the number of residents this survey estimates go weekly:
             \ansblank
  \item[(b)] Explain, in the work space, why that estimate cannot be trusted.
\end{itemize}}

\problem[7.5cm]{A pack holds 20 cards numbered 1 to 20. The card numbered 7 is
taken out and thrown away. Priya then draws one of the remaining cards at random
and writes:\par\vspace{3pt}
\textit{``The probability of an odd number is $\frac{10}{20}$.''}\par\vspace{3pt}
\begin{itemize}
  \item[(a)] the corrected probability of an odd number: \ansblank
  \item[(b)] Name Priya's mistake, in the work space.
\end{itemize}}

\problem[8cm]{In a class of 30 students, 18 play an instrument, and 12 of the
class both play an instrument and sing. A student is chosen at random from the
class. Leo writes:\par\vspace{3pt}
\textit{``The probability that the student both plays and sings is
$\frac{12}{18}$.''}\par\vspace{3pt}
\begin{itemize}
  \item[(a)] the corrected probability: \ansblank
  \item[(b)] Name Leo's mistake, in the work space.
\end{itemize}}

\problem[9cm]{A club has 250 members. The organiser emails every member asking
whether they will come on a trip. 40 members reply, and 34 of those say yes. The
organiser then announces the expected turnout.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the number the organiser calculated from the replies: \ansblank
  \item[(b)] Explain, in the work space, what is wrong with using the 40
             repliers as the sample.
  \item[(c)] Describe, in the work space, a better way for the club to estimate
             the turnout.
\end{itemize}}
