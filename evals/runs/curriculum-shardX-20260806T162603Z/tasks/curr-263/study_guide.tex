\sstitleblock{Solving Linear Systems by Elimination}

\skillheading{Skill 1 --- Scaling one equation to match a pair}

\begin{formulabox}
\textbf{Rule:} Multiplying \emph{both sides} of an equation by a nonzero number gives an equivalent equation --- same solutions. So you may rescale one equation until a pair of like terms are opposites, then add.\par
\textbf{When one multiplication is enough:} the coefficient of the chosen variable in one equation divides the one in the other. Multiply by the quotient, and match the signs so the pair cancels.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2x + 5y = 19$ and $3x - y = 3$.
\step{The $y$-coefficients are $5$ and $-1$, and $1$ divides $5$ --- so one multiplication is enough: scale the second equation, not both.}
\step{Multiply the second equation by $5$: $15x - 5y = 15$. Add it to the first: $17x = 34$, so $x = 2$.}
\step{Substitute back into $3x - y = 3$: $6 - y = 3$, so $y = 3$. Check the first equation: $2(2) + 5(3) = 19$ $\checkmark$}

$x = \ans{2}$ \quad and \quad $y = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $4x + y = 14$ and $3x - 2y = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{3}$, $y = \ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Scaling both equations with an LCM}

\begin{formulabox}
\textbf{Rule:} When neither coefficient of the chosen variable divides the other, scale \emph{both} equations to their least common multiple. Pick the variable whose LCM is smallest --- that keeps the arithmetic light.\par
\textbf{Signs:} if the scaled terms end up with the \emph{same} sign, subtract; if they end up \emph{opposite}, add.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3x + 2y = 16$ and $5x + 3y = 26$.
\step{Neither $3$ nor $5$ divides the other, so no single multiplication works --- go to the LCM of the $x$-coefficients, which is $15$.}
\step{Multiply the first equation by $5$ and the second by $3$: $15x + 10y = 80$ and $15x + 9y = 78$. Same sign, so subtract: $y = 2$.}
\step{Substitute into $3x + 2y = 16$: $3x + 4 = 16$, so $x = 4$. Check: $5(4) + 3(2) = 26$ $\checkmark$}

$x = \ans{4}$ \quad and \quad $y = \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2x + 3y = 13$ and $5x + 4y = 22$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{2}$, $y = \ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Multiply \emph{every} term, including the constant on the right. Scaling $3x + 2y = 16$ by $5$ and writing $15x + 10y = 16$ is the single most common elimination error.
\end{watchoutbox}

\skillheading{Skill 3 --- Writing a system from a word problem}

\begin{formulabox}
\textbf{Model:} name each unknown with its unit (``$a$ = price of one adult ticket, in dollars''), then write one equation per \emph{purchase or trip} described. Two descriptions give two equations.\par
\textbf{Read the answer back into the story} --- a negative price or a fractional person means an equation was mis-copied.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two adult and three child tickets cost \$34; four adult and one child ticket cost \$38. Find both prices in dollars.
\step{Two separate purchases of the same two items is a two-equation model --- let $a$ and $c$ be the adult and child prices in dollars.}
\step{$2a + 3c = 34$ and $4a + c = 38$. The $c$-coefficients are $3$ and $1$, so scale the second equation by $3$: $12a + 3c = 114$; subtract the first: $10a = 80$, so $a = 8$.}
\step{Then $4(8) + c = 38$ gives $c = 6$. Check the first purchase: $2(8) + 3(6) = 34$ $\checkmark$}

$a = \ans{8}$ dollars \quad and \quad $c = \ans{6}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Three notebooks and two pens cost \$16; five notebooks and four pens cost \$28. Find both prices in dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $n = \ans{4}$, $p = \ans{2}$}
\end{tryitbox}
