\wstitleblock{Circle Equations in Context}{High School Geometry}{}

\begin{center}
\small\textit{Writing, reading and completing the square on circle equations from real site plans}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Every map in this set has its axes in the units named by the problem, and every centre-radius form is $(x-h)^2 + (y-k)^2 = r^2$.
  \item Leave answers exact unless a problem asks you to round.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.8cm]{A circular splash pad is being marked out on a park map whose axes are in metres. Its centre is at $(4, 7)$, and a drinking fountain at $(4, 14)$ sits exactly on its edge. Write the equation of the splash pad's boundary in centre-radius form.}

\problem{A cell tower's coverage area is modelled by $(x + 5)^2 + (y - 2)^2 = 81$, where $x$ and $y$ are kilometres east and north of the town hall.
\par\vspace{0.15cm}
(a) Find the radius of the coverage area.
\par\vspace*{2.3cm}\answerline{km}
\par\vspace{0.25cm}
(b) A farm stands at $(4, 2)$. Evaluate the left-hand side of the equation at the farm's position. \quad \ansblank
\par\vspace*{2.4cm}}

\problem{A circular garden bed is planned so that the two endpoints of one diameter fall at $(1, 3)$ and $(9, 9)$ on the site plan, with axes in metres.
\par\vspace{0.15cm}
(a) Find the centre of the bed. \quad \ansblank
\par\vspace*{2.2cm}
(b) Find the radius of the bed.
\par\vspace*{2.2cm}\answerline{m}
\par\vspace{0.25cm}
(c) Write the equation of the bed's boundary in centre-radius form. \quad \ansblank
\par\vspace*{2.2cm}}

\problem{A drone operator's circular no-fly zone satisfies $x^2 + y^2 - 6x + 10y + 18 = 0$, with $x$ and $y$ measured in kilometres from the control tower.
\par\vspace{0.15cm}
(a) Rewrite the equation in centre-radius form. \quad \ansblank
\par\vspace*{3.0cm}
(b) Find the radius of the no-fly zone.
\par\vspace*{2.2cm}\answerline{km}}

\problem{The edge of a lighthouse's visible range is the circle $(x - 12)^2 + (y + 5)^2 = 169$, with $x$ and $y$ in kilometres east and north of the harbour office. A ship is at $(0, 7)$.
\par\vspace{0.15cm}
(a) Evaluate the left-hand side of the equation at the ship's position. \quad \ansblank
\par\vspace*{2.2cm}
(b) The lighthouse stands at the centre. Find the distance from the lighthouse to the ship, to the nearest hundredth.
\par\vspace*{2.2cm}\answerline{km}
\par\vspace{0.25cm}
(c) A point is on the range boundary exactly when the left-hand side equals $169$. Write $<$, $=$ or $>$ to compare your answer to part (a) with $169$, then state whether the ship is inside, on, or outside the range. \quad \ansblank
\par\vspace*{2.2cm}}

\problem{A circular running track on a park map (axes in metres) has its centre at $(6, 2)$ and is tangent to a straight fence that runs along the line $y = -4$.
\par\vspace{0.15cm}
(a) Find the radius of the track.
\par\vspace*{2.3cm}\answerline{m}
\par\vspace{0.25cm}
(b) Write the equation of the track in centre-radius form. \quad \ansblank
\par\vspace*{2.4cm}}

\problem{A city's flood-warning zone is the circle $x^2 + y^2 + 14x - 8y - 16 = 0$, with $x$ and $y$ in kilometres from the city centre, which sits at the origin.
\par\vspace{0.15cm}
(a) Rewrite the equation in centre-radius form. \quad \ansblank
\par\vspace*{2.8cm}
(b) Find the radius of the zone.
\par\vspace*{2.2cm}\answerline{km}
\par\vspace{0.25cm}
(c) Find the distance from the origin to the centre of the zone, to the nearest hundredth, then explain whether the city centre lies inside the zone. \quad \ansblank
\par\vspace*{3.4cm}}

\problem{Two water mains meet the boundary of a circular reservoir at $A(2, 9)$ and $B(10, 3)$, and $\overline{AB}$ is a diameter. The axes are in metres. The reservoir is to be rebuilt with the same centre but a radius one and a half times as large.
\par\vspace{0.15cm}
(a) Find the centre of the reservoir. \quad \ansblank
\par\vspace*{2.2cm}
(b) Find the new radius.
\par\vspace*{2.2cm}\answerline{m}
\par\vspace{0.25cm}
(c) Write the equation of the rebuilt boundary in centre-radius form. \quad \ansblank
\par\vspace*{2.2cm}}
