\sstitleblock{Normal-Model $z$-Scores and Percentiles}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Standardising: from a Value to $z$}

\begin{formulabox}
\textbf{Formula:} $z = \dfrac{x - \mu}{\sigma}$ \ --- how many standard deviations the value sits above ($z>0$) or below ($z<0$) its own mean.\par
\textbf{Order matters:} it is value minus mean, never mean minus value. Getting it backwards flips the sign and turns a below-average result into an above-average one.\par
\textbf{Why bother:} $z$ strips the units off, so two results measured on completely different scales can be compared.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A test has $\mu = 70$ and $\sigma = 5$. Standardise a score of $78$.
\step{The question asks how unusual one value is inside its own distribution, which is exactly what standardising measures, so use the $z$ formula.}
\step{$z = \dfrac{78 - 70}{5} = \dfrac{8}{5}$.}

$\Rightarrow\quad z = \ans{1.6}$, so the score is $1.6$ standard deviations above the mean.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same test ($\mu = 70$, $\sigma = 5$). Standardise a score of $62$.\\[2pt]
\rotatebox{180}{\footnotesize check: $z = \ans{-1.6}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- From $z$ to a Percentile}

\begin{formulabox}
\textbf{Every entry below is the area to the LEFT of that $z$} --- the proportion of the distribution at or below it. Multiply by $100$ and you have the percentile. For an area to the RIGHT, subtract the entry from $1$.
\par\vspace{2pt}
\footnotesize
\begin{tabular}{|l|c|c|c|c|c|c|}
\hline
$z$ & $-2.00$ & $-1.50$ & $-1.28$ & $-1.00$ & $-0.50$ & $0.00$ \\ \hline
area left & $0.0228$ & $0.0668$ & $0.1003$ & $0.1587$ & $0.3085$ & $0.5000$ \\ \hline
$z$ & $0.50$ & $0.67$ & $1.00$ & $1.28$ & $1.50$ & $2.00$ \\ \hline
area left & $0.6915$ & $0.7486$ & $0.8413$ & $0.8997$ & $0.9332$ & $0.9772$ \\ \hline
\end{tabular}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What proportion lies below $z = 0.50$, and what proportion lies above it?
\step{The table only ever reports left-tail area, so read the left answer straight off and get the right-hand one by complement.}
\step{Left: the entry at $0.50$. Right: $1$ minus that entry.}

below $= \ans{0.6915}$, \quad above $= \ans{0.3085}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What proportion lies below $z = -0.50$, and what proportion lies above it?\\[2pt]
\rotatebox{180}{\footnotesize check: below $\ans{0.3085}$, above $\ans{0.6915}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From a Percentile Back to a Value}

\begin{formulabox}
\textbf{Reverse formula:} $x = \mu + z\sigma$. \ Multiply by $\sigma$ FIRST --- adding $z$ straight onto $\mu$ is the most common mistake in this direction.\par
\textbf{Method:} (1) turn the percentile into a left-tail area; (2) find the $z$ whose table entry is closest to it; (3) substitute into $x = \mu + z\sigma$.\par
\textbf{Top $p\%$?} That is a right-tail statement. Convert it first: area to the left $= 1 - p/100$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\mu = 70$, $\sigma = 5$, which value sits at the $75$th percentile?
\step{A percentile names an area, not a value, so find the $z$ with that area to its left and then undo the standardising.}
\step{Closest table area to $0.75$ is $\ans{0.7486}$ at $z = 0.67$; then $x = 70 + 0.67(5) = 70 + 3.35$.}

$\Rightarrow\quad x = \ans{73.35}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\mu = 70$, $\sigma = 5$, which value sits at the $10$th percentile?\\[2pt]
\rotatebox{180}{\footnotesize check: area $\ans{0.1003}$ at $z = -1.28$, so $x = \ans{63.6}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Comparing Results Across Distributions}

\begin{formulabox}
\textbf{Never compare raw values from different distributions.} Standardise each one against its OWN mean and its OWN standard deviation, then compare the two $z$-scores: the larger $z$ is the relatively stronger result.\par
\textbf{Say it in context:} ``$z = 1.5$ means that result is $1.5$ standard deviations above the mean of its own group.''
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $78$ on a test with $\mu = 70$, $\sigma = 5$, or $88$ on one with $\mu = 80$, $\sigma = 8$?
\step{The two scales are different, so the raw numbers cannot be compared; standardise each result inside its own distribution instead.}
\step{First: $z = 1.6$ from Skill 1. Second: $z = \dfrac{88 - 80}{8}$.}

second $z = \ans{1.0}$, and $\ans{1.6 > 1.0}$, so the first result is stronger.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $66$ on a test with $\mu = 60$, $\sigma = 4$ stronger than a result with $z = 1.2$?\\[2pt]
\rotatebox{180}{\footnotesize check: $z = \ans{1.5}$ and $\ans{1.5 > 1.2}$, so yes}
\end{tryitbox}
