\sstitleblock{Adding and Subtracting with Like Denominators}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Adding with like denominators}

\begin{formulabox}
\textbf{Rule:} when the denominators match, the pieces are the same size ---
add the numerators (the counts of pieces) and \emph{keep the denominator}.
Simplify the answer if you can.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $\tfrac{2}{5} + \tfrac{1}{5}$.
\step{Both fractions count fifths --- same-size pieces --- so I can add the counts and keep the denominator.}
\step{$2 + 1 = 3$ pieces, each one fifth: $\tfrac{2}{5} + \tfrac{1}{5} = \ans{\tfrac{3}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $\tfrac{3}{7} + \tfrac{2}{7}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{7}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Subtracting with like denominators}

\begin{formulabox}
\textbf{Rule:} subtract the numerators and keep the denominator.
Taking pieces away never changes the size of the pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $\tfrac{7}{10} - \tfrac{3}{10}$.
\step{Both fractions count tenths, so subtract the counts and keep the denominator.}
\step{$7 - 3 = 4$ tenths, and $\tfrac{4}{10}$ simplifies (divide top and bottom by 2) to $\ans{\tfrac{2}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $\tfrac{5}{9} - \tfrac{2}{9}$. Simplify your
answer.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Breaking a fraction into parts (decomposing)}

\begin{formulabox}
\textbf{Idea:} every fraction is a sum of smaller fractions with the same
denominator. You can always break it into unit fractions:
$\tfrac{3}{4} = \tfrac{1}{4} + \tfrac{1}{4} + \tfrac{1}{4}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fill in: $\tfrac{4}{6} = \tfrac{3}{6} +
\tfrac{\square}{6}$
\step{The two parts must add back to 4 sixths, so ask: 3 plus what makes 4?}
\step{$4 - 3 = 1$, so the missing numerator is $\ans{1}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Fill in: $\tfrac{7}{8} = \tfrac{5}{8} +
\tfrac{\square}{8}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never add the denominators.
$\tfrac{2}{5} + \tfrac{1}{5}$ is \emph{not} $\tfrac{3}{10}$ --- the
denominator names the size of the pieces, and the pieces stay fifths.
\end{watchoutbox}
