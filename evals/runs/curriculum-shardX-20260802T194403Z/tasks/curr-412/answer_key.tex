\problem{\textbf{$\log_2 x = 5$.}
\emph{Domain:} $x > 0$.
\emph{Rewrite:} $\log_b A = c \Leftrightarrow A = b^c$, so $x = 2^5$.
\[ x = 32 \]
\emph{Check:} $32 > 0$, and $\log_2 32 = 5$ $\checkmark$. \ans{x = 32}}

\problem{\textbf{$\log_3(x - 4) = 2$.}
\emph{Domain:} the argument must be positive, $x - 4 > 0$, so $x > 4$.
\emph{Rewrite:} $x - 4 = 3^2 = 9$.
\emph{Solve:} $x = 13$.
\emph{Check:} $13 > 4$ $\checkmark$, and $\log_3(13 - 4) = \log_3 9 = 2$
$\checkmark$. \ans{x = 13}}

\problem{\textbf{$\log_5(2x + 1) = \log_5(x + 9)$.}
\emph{Domain:} $2x + 1 > 0$ and $x + 9 > 0$, so $x > -\tfrac{1}{2}$ (the tighter
of the two).
\emph{Solve:} logarithms with the same base are equal only when their arguments
are equal, so
\[ 2x + 1 = x + 9 \quad\Longrightarrow\quad x = 8 \]
\emph{Check:} $8 > -\tfrac{1}{2}$ $\checkmark$; both sides become $\log_5 17$
$\checkmark$. \ans{x = 8}}

\problem{\textbf{Rewrite $\log_2(8x)$.}
Product rule: $\log_b(MN) = \log_b M + \log_b N$, so
\[ \log_2(8x) = \log_2 8 + \log_2 x \]
and $\log_2 8 = 3$ because $2^3 = 8$. \ans{\log_2 (8x) = 3 + \log_2 x}
\emph{Valid for $x > 0$}: the rewriting splits one logarithm into two, and the
new $\log_2 x$ needs its own positive argument. This is the move that turns
$\log_2(8x) = k$ into a one-logarithm equation.}

\problem{\textbf{$\log_2 x + \log_2(x - 2) = 3$.}
\emph{Domain:} $x > 0$ \emph{and} $x - 2 > 0$, so $x > 2$.
\emph{Combine:} product rule gives $\log_2\big(x(x-2)\big) = 3$.
\emph{Rewrite and solve:}
\begin{align*}
x(x - 2) &= 2^3 = 8 \\
x^2 - 2x - 8 &= 0 \\
(x - 4)(x + 2) &= 0
\end{align*}
Candidates $x = 4$ and $x = -2$. Only $4$ lies in $x > 2$.
\emph{Check:} $\log_2 4 + \log_2 2 = 2 + 1 = 3$ $\checkmark$. \ans{x = 4}}

\problem{\textbf{Why $x = -2$ fails.}
Evaluate the argument of the second logarithm at the candidate:
\[ x - 2 = (-2) - 2 = -4 \]
\ans{-4}
The candidate would require $\log_2(-4)$, and a logarithm is undefined for a
negative argument, so $x = -2$ is extraneous --- it solves the quadratic that
came \emph{after} the logarithms were cleared, not the original equation. The
domain of the original equation is $x > 2$ (both $x > 0$ and $x - 2 > 0$ must
hold).
\emph{Note for the grader: the one-sentence explanation is judged by you; the
evaluated argument $-4$ is machine-verified.}}

\problem{\textbf{$\ln(x - 1) = 2$.}
\emph{Domain:} $x - 1 > 0$, so $x > 1$.
\emph{Rewrite} with base $e$: $x - 1 = e^2$.
\[ x = e^2 + 1 \]
\emph{Check:} $e^2 \approx 7.389$, so $x \approx 8.389 > 1$ $\checkmark$.
\ans{x = e^2 + 1}}

\problem{\textbf{$\log_2(x^2 + 4x) = \log_2 12$.}
\emph{Domain:} $x^2 + 4x > 0$, i.e. $x(x + 4) > 0$, which holds for $x < -4$ or
$x > 0$.
\emph{Solve:} equal logarithms, equal arguments:
\begin{align*}
x^2 + 4x &= 12 \\
x^2 + 4x - 12 &= 0 \\
(x - 2)(x + 6) &= 0
\end{align*}
Candidates $x = 2$ and $x = -6$.
\emph{Check both:} at $x = 2$, the argument is $4 + 8 = 12 > 0$ $\checkmark$; at
$x = -6$, the argument is $36 - 24 = 12 > 0$ $\checkmark$ (and $-6 < -4$, inside
the domain). \ans{x = 2 \text{ or } x = -6}
A negative candidate is \emph{not} automatically extraneous --- what matters is
the sign of the argument, not the sign of $x$.}

\problem{\textbf{$\ln(2x - 1) = 3$, to the nearest hundredth.}
\emph{Domain:} $2x - 1 > 0$, so $x > \tfrac{1}{2}$.
\emph{Rewrite:} $2x - 1 = e^3$, so
\[ x = \frac{e^3 + 1}{2} \approx \frac{20.0855 + 1}{2} = 10.5428\ldots \]
\emph{Check:} $10.54 > \tfrac{1}{2}$ $\checkmark$. \ans{x \approx 10.54}
Keep the exact form until the last line: rounding $e^3$ early moves the answer.}

\problem{\textbf{$\log(x + 3) - \log(x - 1) = 1$ (base $10$).}
\emph{Domain:} $x + 3 > 0$ and $x - 1 > 0$, so $x > 1$.
\emph{Combine} with the quotient rule, then rewrite:
\begin{align*}
\log\!\left(\frac{x+3}{x-1}\right) &= 1 \\
\frac{x+3}{x-1} &= 10^1 = 10 \\
x + 3 &= 10x - 10 \\
13 &= 9x
\end{align*}
\ans{x = \frac{13}{9}}
\emph{Check:} $\tfrac{13}{9} \approx 1.44 > 1$ $\checkmark$, and the arguments
are $\tfrac{40}{9}$ and $\tfrac{4}{9}$, whose quotient is $10$ $\checkmark$.}

\problem{\textbf{$\log_2(x + 5) = 4.7$, to the nearest hundredth.}
\emph{Domain:} $x + 5 > 0$, so $x > -5$.
\emph{Rewrite:} $x + 5 = 2^{4.7}$, so
\[ x = 2^{4.7} - 5 \approx 25.9921 - 5 = 20.9921 \]
\emph{Check:} $20.99 > -5$ $\checkmark$. \ans{x \approx 20.99}
A non-integer exponent is fine --- exponential form does not require $b^c$ to be
a whole number.}

\problem{\textbf{Synthesis: $2\log_5 x - \log_5(x + 6) = 0$.}
\emph{Domain:} $x > 0$ and $x + 6 > 0$; together, $x > 0$.
\emph{Combine} with the power rule then the quotient rule:
\begin{align*}
\log_5 x^2 - \log_5(x + 6) &= 0 \\
\log_5\!\left(\frac{x^2}{x+6}\right) &= 0 \\
\frac{x^2}{x + 6} &= 5^0 = 1 \\
x^2 - x - 6 &= 0 \\
(x - 3)(x + 2) &= 0
\end{align*}
Candidates $x = 3$ and $x = -2$; the domain $x > 0$ eliminates $-2$, which would
require $\log_5(-2)$.
\emph{Check:} $2\log_5 3 - \log_5 9 = \log_5 9 - \log_5 9 = 0$ $\checkmark$.
\ans{x = 3}}
