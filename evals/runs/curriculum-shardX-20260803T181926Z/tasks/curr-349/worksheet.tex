\wstitleblock{Transformations --- Diagnosing a Wrong Centre, Vector or Scale Factor}{High School Geometry}{}

\noindent\small Three pieces of data decide a transformation, and almost every
wrong image comes from one of them being wrong rather than from bad arithmetic.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Vector.} A translation vector is \emph{image minus pre-image},
        in that order. Reverse the order and you get the translation that undoes
        it.
  \item \textbf{Scale factor.} A scale factor is a \emph{ratio} of distances
        from the centre, never a difference.
  \item \textbf{Centre.} A dilation moves each point along the ray from
        \emph{its own centre}. Multiplying the coordinates only works when that
        centre is the origin.
  \item Every claim here can be tested. Where a length is asked for, use
        \[ d=\sqrt{(x_2-x_1)^2+(y_2-y_1)^2} \]
        and give two decimal places when the answer is not exact.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[6cm]{\textbf{Warm-up.} A translation maps $A(-3,\,2)$ to
$A'(4,\,-1)$. Write the translation vector
$\langle a,\ b \rangle$.}

\problem[6.5cm]{\textbf{Find and fix.} Lee says the translation that maps
$P(2,\,5)$ to $P'(-3,\,1)$ is $\langle 5,\ 4 \rangle$.
\par\smallskip
Lee subtracted in the wrong order. Explain what that does to a vector, then
write the correct translation vector.}

\problem[8cm]{\textbf{Find and fix.} Kim translates $A(1,\,2)$ and $B(5,\,5)$ by
the vector $\langle 3,\ -1 \rangle$ and records the images as $A'(4,\,1)$ and
$B'(8,\,5)$.
\par\smallskip
\textbf{(a)} Find $AB$.
\quad \textbf{(b)} Find $A'B'$ using Kim's coordinates, to two decimal places.
\quad \textbf{(c)} A translation is a rigid motion, so what do your two answers
prove about Kim's work? Give the corrected $y$-coordinate of $B'$.}

\problem[7.5cm]{\textbf{Find and fix.} A dilation centred at the origin maps
$P(3,\,4)$ to $P'(9,\,12)$. Jo writes: ``the scale factor is $6$, because
$9-3=6$.''
\par\smallskip
\textbf{(a)} Find the distance from the origin to $P$.
\quad \textbf{(b)} Find the distance from the origin to $P'$.
\quad \textbf{(c)} Use those two distances to find the real scale factor, and
say in one sentence why a difference can never be a scale factor.}

\problem[8cm]{\textbf{Find and fix.} A dilation has centre $C(2,\,1)$ and scale
factor $2$. It maps $A(5,\,3)$ to $A'$. Ravi doubles the coordinates of $A$ and
writes $A'(10,\,6)$.
\par\smallskip
\textbf{(a)} Ravi used the wrong centre. Which centre did his method actually
use?
\quad \textbf{(b)} Find the correct $x$-coordinate and $y$-coordinate of $A'$,
using $A' = C + k\,(A - C)$.
\quad \textbf{(c)} Find $CA'$ to two decimal places.}

\problem{\textbf{Show what you know.} Two dilations both use scale factor $2$,
but one is centred at $C_1$ and the other at $C_2$, two different points.
\par\smallskip
\textbf{(a)} Sketch a triangle and both of its images on the grid below. Label
the two centres and both images.
\par
\textbf{(b)} Describe how the two images compare in \emph{size} and in
\emph{position}, and say why.
\par
\textbf{(c)} Suppose you are handed a point and its image under an unknown
dilation. Explain how you would locate the centre, and why your method works.
