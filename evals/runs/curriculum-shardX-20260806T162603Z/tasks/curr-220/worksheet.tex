\wstitleblock{Comparing Rates of Change}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Rates of change from tables, from graphs, from equations, and side by
side}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item A rate of change is always (change in output) $\div$ (change in
        input), whatever form the relationship is given in.
  \item From a table, subtract two rows. From a graph, use two marked points.
        From an equation $y = mx + b$, the rate is the number multiplying the
        input.
  \item When a problem asks you to compare two rates, write the comparison
        with $>$, $<$, or $=$, keeping the two rates in the order the question
        names them.
\end{itemize}

\problem{A gym's total cost after $x$ months is shown in the table.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Months} & 0 & 2 & 4 & 6 \\ \hline
\textbf{Total cost (\$)} & 30 & 54 & 78 & 102 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] By how much does the total cost change from month 0 to month 6?
        \par\vspace*{2cm}\ansline
  \item[(b)] Find the rate of change, in dollars per month.
        \par\vspace*{2cm}\ansline
\end{itemize}}
