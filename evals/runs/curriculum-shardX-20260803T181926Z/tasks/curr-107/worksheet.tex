\wstitleblock{Multiplying by a One-Digit Number}{Grades 4--5 \quad Number and Operations in Base Ten}{}

\begin{center}
\textit{Multiplying three- and four-digit numbers by one digit, estimating a
product by rounding, and using multiplication to answer story questions}
\end{center}

\noindent\textbf{How to work these problems.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Stack the numbers so the ones digits line up. Multiply the one-digit
factor by each digit of the other number, starting at the ones place.
\item When a partial product reaches ten or more, write its ones digit and
\emph{regroup} the tens digit into the next column. Multiply first, then add
the regrouped digit.
\item Every place gets multiplied --- including a zero. A zero column is not a
column you skip.
\item Show the regrouped digits. A grader can find a slip in one column much
faster than in one wrong final answer.
\end{itemize}
