\sstitleblock{Evaluating Absolute-Value and Piecewise Functions}

\skillheading{Skill 1 --- Evaluating an absolute-value rule}

\begin{formulabox}
\textbf{What the bars do:} $|n|$ is the distance from $n$ to zero, so it is never
negative: $|5| = 5$ and $|-5| = 5$.\par
\smallskip
\textbf{Order of work, every time:} (1) finish everything \emph{inside} the bars,
(2) take the absolute value, (3) do whatever is left outside --- multiply, then add or
subtract.\par
\smallskip
\textbf{Picture:} the graph of $y = a|x - h| + k$ is a V with its point at $(h, k)$. The
inside hits zero exactly at $x = h$, which is why the V turns there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = |x - 8| + 1$, find $f(3)$.
\step{The input goes inside the bars, so the bars are a grouping symbol here --- subtract
first and take the absolute value before touching the $+1$.}
\step{$f(3) = |3 - 8| + 1 = |-5| + 1 = 5 + 1$}

$\Rightarrow\quad f(3) = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = 3|x + 1| - 4$, find $g(-6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $g(-6) = \ans{11}$}
\end{tryitbox}

\skillheading{Skill 2 --- Choosing the branch of a piecewise rule}

\begin{formulabox}
\textbf{A piecewise rule is a set of formulas plus a set of conditions.} The conditions
come first.
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Ask: which condition does my input make \emph{true}? Exactly one will.
  \item Only then substitute --- into that branch alone.
  \item At a boundary, the branch whose sign is $\leq$ or $\geq$ is the one that owns
        the number. Strict $<$ and $>$ do not.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $p(x) = 2x - 5$ when $x < 3$ and $p(x) = x^2 + 1$ when
$x \geq 3$, find $p(3)$.
\step{Test the input against the conditions before substituting: $3 < 3$ is false, and
$3 \geq 3$ is true, so the second branch owns this input.}
\step{$p(3) = 3^2 + 1 = 9 + 1$}

$\Rightarrow\quad p(3) = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $q(x) = 4 - x$ when $x < 1$ and $q(x) = 2x^2$ when
$x \geq 1$, find $q(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $q(1) = \ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reading a piecewise model in context}

\begin{formulabox}
\textbf{Real prices and fees change rule partway.} A flat charge up to some point, then a
per-unit charge after it, is a two-branch model: the second branch usually reads
\emph{base} $+$ \emph{rate} $\times$ (\emph{amount past the cutoff}).\par
\smallskip
Watch the ``past the cutoff'' part: it is $w - 2$, not $w$, when the extra charges start
after $2$ units. Substituting $w$ there charges the customer twice for the base.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Shipping costs \$5 for a parcel up to 2\,kg, and \$5 plus \$2 for
each kilogram over 2\,kg after that. Find the cost of a 6\,kg parcel.
\step{Six is past the two-kilogram cutoff, so the second branch applies, and only the
weight \emph{above} the cutoff gets charged the per-kilogram rate.}
\step{cost $= 5 + 2(6 - 2) = 5 + 2(4) = 5 + 8$}

$\Rightarrow\quad \ans{13}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A taxi charges \$7 for the first 3\,km and \$3 for each kilometre
after that. Find the fare for a 9\,km trip.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25}$ dollars}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the two errors that cost the most marks are the same error twice.
\emph{Substituting before reading} --- picking a branch by where it sits on the page
instead of by its condition --- and \emph{dropping the bars}, so a negative inside stays
negative. Both come from starting the arithmetic too early. Read the condition, or close
the bars, and only then compute.
\end{watchoutbox}
