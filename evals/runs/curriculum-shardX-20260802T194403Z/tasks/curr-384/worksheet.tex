\wstitleblock{Composition Order and Inverse Domains}{Algebra 2 --- Error Analysis}{}

\noindent\small\textbf{Directions:} These problems are about two errors that look
harmless on paper: composing in the wrong order, and accepting an inverse without
checking its domain. Show the substitution you actually performed. On the
find-and-fix problems (4 and 6), name the mistake in one sentence \emph{and}
give the corrected value.
\normalsize

\problem[5cm]{Let $f(x) = 2x - 5$ and $g(x) = x^2 + 1$.\\[2pt]
Find $(f \circ g)(3)$. Write down $g(3)$ first, then feed it to $f$.}

\problem[5.5cm]{Using the same $f(x) = 2x - 5$ and $g(x) = x^2 + 1$, find
$(g \circ f)(3)$.\\[2pt]
Then compare with your answer to problem 1 and state in one sentence what the
comparison shows about the order of composition.}

\problem[5cm]{Let $f(x) = 3x + 7$. The inverse $f^{-1}$ undoes $f$, so
$f^{-1}(22)$ is the input that $f$ sends to $22$.\\[2pt]
Solve $3x + 7 = 22$ to find $f^{-1}(22)$.}

\problem[6.5cm]{\textbf{Find and fix.} With $f(x) = 3x + 1$ and $g(x) = x - 6$,
Nia was asked for $(f \circ g)(4)$. She wrote:\\[3pt]
\hspace*{0.8cm}$(f \circ g)(4) = f(4) - 6 = 13 - 6 = 7$\\[3pt]
Explain in one sentence which composition Nia actually computed, then compute
$(f \circ g)(4)$ correctly.}

\problem[6cm]{Let $f(x) = 3x + 7$, whose inverse is $f^{-1}(x) = \dfrac{x - 7}{3}$.\\[2pt]
Show that $f\bigl(f^{-1}(x)\bigr)$ simplifies to $x$ for every real $x$. Write
each line of the simplification.}

\problem[7cm]{\textbf{Find and fix.} Let $h(x) = (x - 3)^2$ with the domain
restricted to $x \ge 3$. Sam claims the inverse is
$h^{-1}(x) = 3 - \sqrt{x}$, and reports $h^{-1}(16) = -1$.\\[3pt]
Test Sam's claim by checking whether $-1$ lies in the domain $x \ge 3$ of $h$.
Then state the correct inverse rule and compute $h^{-1}(16)$.}

\problem[6cm]{Let $g(x) = x^2$ with \emph{no} restriction on its domain, and
suppose someone writes ``$g^{-1}(x) = \sqrt{x}$''.\\[2pt]
Solve $x^2 = 9$ and list every real solution. Then explain in one sentence what
your list proves about $g$ having an inverse on all of $\mathbb{R}$.}

\problem[7cm]{Let $f(x) = \sqrt{x - 1}$ and $g(x) = x^2 + 1$. Marcus checks
$f(g(x)) = \sqrt{(x^2 + 1) - 1} = \sqrt{x^2} = x$ and concludes that $f$ and $g$
are inverses for every real number $x$.\\[3pt]
Explain why his conclusion holds only for $x \ge 0$, state which composition
check he skipped, and give the domain of $f$ and the domain of $g$ that make the
pair genuine inverses.\noansline}
