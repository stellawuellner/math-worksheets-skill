\sstitleblock{Putting Three Amounts in Order}

\skillheading{Skill 1 --- Ordering three amounts}

\begin{formulabox}
\textbf{Rule:} \emph{Least} is the smallest amount, \emph{greatest} is the biggest.\par
To order three numbers from least to greatest, count up out loud and write each number down as you reach it.\par
\textbf{Check:} the amounts must all be counted in the \emph{same unit} --- cups with cups, minutes with minutes --- before the numbers can be compared.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Three cups hold $7$, $2$ and $5$ marbles. Put the counts in order from least to greatest.
\step{Least to greatest means smallest first, and counting up reaches the smallest number first, so I walk up the counting order and pick off each number as it comes.}
\step{Counting up from one, I reach $2$ first, then $5$, then $7$. So the cup with $2$ marbles is least and the cup with $7$ is greatest.}

$\ans{2,\ 5,\ 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Three baskets hold $9$, $4$ and $6$ eggs. Put the counts in order from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4,\ 6,\ 9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Write the \emph{amounts} in order, not the names or the days. Leo's biggest day still goes last on the line.
\end{watchoutbox}

\skillheading{Skill 2 --- Counting a picture, then ordering}

\begin{formulabox}
\textbf{Rule:} When the amounts are shown as objects, count each group first and write its number down. Touch each object once as you say the counting word, and the last word you say is how many.\par
Then order the numbers you wrote, exactly as in Skill 1.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Put these three groups in order from least to greatest.\par
Group A: $\bullet\ \bullet\ \bullet\ \bullet$ \qquad Group B: $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$ \qquad Group C: $\bullet\ \bullet\ \bullet$
\step{The picture gives no numerals, so before I can order anything I have to turn each group into a number by counting it.}
\step{Touching one dot per counting word: Group A has $4$, Group B has $6$, Group C has $3$. Counting up, $3$ comes first, then $4$, then $6$.}

$\ans{3,\ 4,\ 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Count and order from least to greatest.\par
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet$ \qquad $\bullet\ \bullet$ \qquad $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2,\ 5,\ 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two amounts}

\begin{formulabox}
\textbf{Rule:} $<$ is \emph{is less than}, $>$ is \emph{is greater than}, $=$ is \emph{is equal to}.\par
The wide open end of the symbol always faces the bigger amount; the point faces the smaller one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One tray holds $8$ muffins and another holds $5$. Write $<$, $>$, or $=$ between the counts.
\step{Only two amounts here, so instead of a whole order I just decide which one is bigger and turn the symbol to face it.}
\step{Counting up I say $5$ first and $8$ later, so $8$ is the bigger count. It is written on the left, so the wide end opens to the left.}

$\ans{8 > 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One box holds $3$ pencils and another holds $7$. Write $<$, $>$, or $=$ between the counts.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3 < 7}$}
\end{tryitbox}
