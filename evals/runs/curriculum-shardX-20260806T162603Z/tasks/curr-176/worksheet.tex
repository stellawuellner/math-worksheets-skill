\wstitleblock{One-Step Equations with Rational Coefficients}{Grade 6--7 Mathematics}{}
\begin{center}
\textit{one-step equations: undo adding or subtracting, divide both sides, multiply by the reciprocal, or undo a division}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Think of every equation as a \emph{balance}: whatever you do to one side,
  do to the other, so the pans stay level.
  \item Solve each equation for $x$, showing the undo step you chose, and check
  your solution in the original equation.
\end{itemize}

\problem[3cm]{Solve. \emph{Balance model:} the left pan holds $x$ and 7 unit
weights, the right pan holds 12 unit weights --- remove 7 from \emph{both} pans.
\[ x + 7 = 12 \]}

\problem{Solve. \emph{Tape model:} the whole strip is $x$, cut into 5 equal parts,
and each part is worth 4.
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
4 & 4 & 4 & 4 & 4 \\
\hline
\end{tabular}
\end{center}
\[ \frac{x}{5} = 4 \]
\par\vspace*{3cm}\ansline}

\problem[3.2cm]{Solve: \quad $x - \dfrac{3}{4} = 2$}

\problem[3.2cm]{Solve, writing the answer as a fraction: \quad $6x = 21$}

\problem[3.2cm]{Solve: \quad $0.4x = 10$}

\problem{Solve. \emph{Bar model:} the bar $x$ is cut into 3 equal parts, and 2 of
those parts together total 8.
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\rule{1.6cm}{0pt} & \rule{1.6cm}{0pt} & \rule{1.6cm}{0pt} \\
\hline
\end{tabular}
\end{center}
\[ \frac{2}{3}x = 8 \]
\par\vspace*{3.2cm}\ansline}

\problem[3.4cm]{Solve: \quad $\dfrac{x}{2.5} = 6$}

\problem[3.8cm]{Solve, writing the answer as a fraction in simplest form:
\quad $\dfrac{3}{4}x = \dfrac{9}{10}$}

\problem[3.8cm]{Solve: \quad $x + 2.6 = 1.4$}

\problem{\textbf{Challenge.} Two-fifths of a number $x$ is 18.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Write an equation that says this in symbols.\par\vspace*{1.5cm}\ansline
  \item[(b)] Solve your equation to find the number.\par\vspace*{2cm}\ansline
\end{itemize}}
