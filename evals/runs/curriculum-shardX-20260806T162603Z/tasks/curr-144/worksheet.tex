\wstitleblock{Squared or Cubed? --- Measurement Unit Detective}{Grade 5}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item Every answer needs a \emph{number} and the \emph{right unit}. Flat
        space takes square units; filled space takes cubic units.
  \item Some problems show another student's work with a mistake in it. Find
        the mistake, then fix it.
\end{itemize}

\vspace{0.2cm}

\problem{A sticker is a rectangle $7$ cm long and $4$ cm wide. What is its
area?
\par\vspace*{4cm}\answerline{cm$^2$}}

\problem{A small box is packed full of centimeter cubes (each cube is one
centimeter on every side). Each layer holds $6$ cubes, and the box holds $4$
layers. What is the volume of the box?
\par\vspace*{4cm}\answerline{cm$^3$}}

\problem{Pick the best unit for each measurement. Write cm, m$^2$, mL, or
cm$^3$.
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item[(a)] the size of a classroom floor \quad \ansblank
        \par\vspace*{0.9cm}
  \item[(b)] the amount of juice in a glass \quad \ansblank
        \par\vspace*{0.9cm}
  \item[(c)] the length of a pencil \quad \ansblank
        \par\vspace*{0.9cm}
\end{itemize}}

\problem{A cereal box is $5$ cm long, $3$ cm wide, and $2$ cm tall. What is
its volume?
\par\vspace*{4.5cm}\answerline{cm$^3$}}

\problem{Tomas measures a jewelry box: $6$ cm long, $2$ cm wide, $3$ cm
tall. He writes: \quad ``Volume $= 6 \times 2 \times 3 = 36$ cm$^2$.''
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] Name the error in Tomas's answer.\par\vspace*{1.8cm}\ansline
  \item[(b)] Write the corrected answer, with its unit.
        \par\vspace*{1.6cm}\answerline{cm$^3$}
\end{itemize}}

\problem{An aquarium is $40$ cm long, $20$ cm wide, and $30$ cm deep. How
much water can it hold, in cubic centimeters?
\par\vspace*{4.5cm}\answerline{cm$^3$}}

\problem{Lena's poster board has an area of $2$ m$^2$. She remembers that
each meter is $100$ centimeters, so she writes: \quad
``$2$ m$^2 = 2 \times 100 = 200$ cm$^2$.''
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] Name the error in Lena's conversion.\par\vspace*{2cm}\ansline
  \item[(b)] Write the corrected number of square centimeters.
        \par\vspace*{2.2cm}\answerline{cm$^2$}
\end{itemize}}

\problem{\textbf{Challenge.} A cube has edges $2$ cm long. A second cube has
edges twice as long: $4$ cm.
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item[(a)] Volume of the small cube:\par\vspace*{1.6cm}\answerline{cm$^3$}
  \item[(b)] Volume of the large cube:\par\vspace*{1.6cm}\answerline{cm$^3$}
  \item[(c)] The large volume is how many times the small volume?
        \par\vspace*{1.4cm}\ansline
  \item[(d)] Roy says: ``The edge doubled, so the volume doubled too.''
        Name the error in Roy's thinking.\par\vspace*{1.8cm}\ansline
\end{itemize}}
