\wstitleblock{Quotients and Remainders with Models}{Grade 4 --- Whole-Number Division}{}

\begin{center}
\small Reading a quotient off a model, deciding what the remainder means, and estimating first
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions:}
\begin{itemize}
  \item Every division here can be written as \ (total) $=$ (groups) $\times$ (size of a group) $+$
        (remainder), and the remainder is always smaller than the group size.
  \item Where a model is drawn, mark it up: circle groups, cross off tiles, label what is left over.
  \item Read each question carefully --- sometimes the answer is the quotient, sometimes the
        remainder, and sometimes one more than the quotient.
\end{itemize}

\vspace{0.15cm}

\problem[3.4cm]{The counters below are shared into equal groups of $5$. Circle the groups.
\par\vspace{0.15cm}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {1,...,23}{\fill[blue!65] (0.42*\i,0) circle (2.8pt);}
\end{tikzpicture}
\end{center}
\begin{itemize}
  \item[(a)] How many full groups of $5$ are there? \quad \ansblank
  \item[(b)] How many counters are left over? \quad \ansblank
\end{itemize}}

\problem[3.4cm]{A tile pattern is laid out in rows of $6$ tiles. There are $34$ tiles in the pattern.
\par\vspace{0.15cm}
\begin{center}
\begin{tikzpicture}
  \foreach \r in {0,...,4}{\foreach \c in {1,...,6}{
    \draw[fill=orange!35] (0.5*\c,-0.5*\r) rectangle ++(0.4,0.4);}}
  \foreach \c in {1,...,4}{\draw[fill=orange!35] (0.5*\c,-2.5) rectangle ++(0.4,0.4);}
\end{tikzpicture}
\end{center}
\begin{itemize}
  \item[(a)] How many complete rows of $6$ are there? \quad \ansblank
  \item[(b)] How many tiles are left over, not in a complete row? \quad \ansblank
\end{itemize}}

\problem[3.4cm]{A class of $38$ students is travelling to a museum. Each van seats $8$ students.
\begin{itemize}
  \item[(a)] How many vans are needed so that every student has a seat? \answerline{vans}
  \item[(b)] Explain why the leftover students change the answer you would get from dividing alone.
\end{itemize}}

\problem[4.4cm]{A baker packs $45$ cookies into boxes that hold $6$ cookies each. Only full boxes are
sold. How many full boxes can the baker sell?}

\problem[3.2cm]{A gardener shares $356$ bulbs equally among $8$ beds.
\begin{itemize}
  \item[(a)] Round $356$ to the nearest hundred. \quad \ansblank
  \item[(b)] Use that to estimate how many bulbs go in each bed. \quad \ansblank
  \item[(c)] Now divide exactly: each bed gets \ansblank\ bulbs, with \ansblank\ bulbs left over.
\end{itemize}}

\problem[3.4cm]{The bar below stands for $96$ counters, split into $4$ equal parts. How many counters
are in each part?
\par\vspace{0.15cm}
\begin{center}
\begin{tikzpicture}
  \draw[thick] (0,0) rectangle (10,1);
  \foreach \x in {2.5,5,7.5}{\draw[thick] (\x,0) -- (\x,1);}
\end{tikzpicture}
\end{center}}

\problem[3.4cm]{The model below shows $60$ counters put into $7$ equal groups, with $4$ counters
left over. How many counters are in each group?
\par\vspace{0.15cm}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {0,...,6}{\draw[thick] (1.3*\i,0) circle (0.42);}
  \foreach \i in {1,...,4}{\fill[blue!65] (9.6+0.42*\i,0) circle (2.8pt);}
\end{tikzpicture}
\end{center}}

\problem[4.4cm]{A collector puts $52$ marbles into bags of $9$. How many marbles are left over after
every full bag is filled?}

\problem[3.2cm]{A printer produces $1{,}235$ leaflets, packed equally into $6$ cartons.
\begin{itemize}
  \item[(a)] Round $1{,}235$ to the nearest hundred. \quad \ansblank
  \item[(b)] Use that to estimate how many leaflets go in each carton. \quad \ansblank
  \item[(c)] Now divide exactly: each carton holds \ansblank\ leaflets, with \ansblank\ left over.
\end{itemize}}

\problem[3.6cm]{$83$ pencils are shared among $6$ friends.
\begin{itemize}
  \item[(a)] How many pencils does each friend get? \quad \ansblank
  \item[(b)] How many pencils are left over? \quad \ansblank
  \item[(c)] Now suppose the same $83$ pencils are packed into boxes that hold $6$ pencils. How many
        boxes are needed to hold every pencil? \quad \ansblank
  \item[(d)] The same division gave two different answers in (a) and (c). Explain why.
\end{itemize}}
