\wstitleblock{Reading and Making Line Plots}{Grade 2--3}{}

\begin{center}\small\textit{Reading X's off a whole-number line plot, drawing the missing
X's to finish one, and comparing or totalling what the X's show}\end{center}

\noindent\small\textbf{How to use this sheet.}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A \emph{line plot} is a number line with one \textsf{X} for every thing that was
        counted or measured. The numbers go along the bottom in order, evenly spaced.
  \item One \textsf{X} always stands for one thing. To find out how many things had a
        certain number, do not read the number --- count the \textsf{X}'s stacked above it.
  \item A number with no \textsf{X}'s above it still belongs on the line. It tells you that
        nothing landed there.
  \item Touch each \textsf{X} with your pencil as you count it, and say the count out loud.
        Counting a stack twice is the mistake this sheet keeps coming back to.
\end{itemize}
\normalsize

\problem[0.3cm]{\textbf{Count one stack.} Room 4 made a line plot of how many books each
child read last week.

\par\smallskip
\noindent\begin{minipage}[t]{\figcol}\vspace{0pt}
\begin{tikzpicture}[x=0.72cm, y=0.32cm]
  \draw[thick] (1.4,0) -- (6.9,0);
  \foreach \v in {2,...,6} {\draw (\v,0.4) -- (\v,-0.4);
     \node[below, font=\small] at (\v,-0.4) {\v};}
  \foreach \v/\n in {2/4, 3/6, 4/3, 5/2, 6/1} {%
     \foreach \i in {1,...,\n} {\node[font=\small] at (\v,\i) {\textsf{X}};}}
  \node[font=\small] at (4.15,-3.1) {Books read last week};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[t]{\qcol}\vspace{0pt}
\small Each \textsf{X} is one child.\normalsize
\par\smallskip
Put your pencil on the $4$ at the bottom. Now count straight up.
\par\smallskip
How many children read exactly $4$ books?~\ansblank
\par\vspace*{2.5cm}
\end{minipage}}
