\sstitleblock{Diagnosing Reversed and Non-Equivalent Ratios}

% =================== SKILL 1 ===================
\skillheading{Catching a Reversed Ratio}

\begin{formulabox}
\textbf{Rule:} The quantity named \textbf{first} goes first. ``The ratio of A to
B'' means A's number, then B's number. In a rate, the word \textbf{per} points at
the bottom: miles \emph{per} hour means miles on top, hours on the bottom.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A basket holds 11 lemons and 6 limes. Write the ratio of
lemons to limes.
\step{Lemons are named first, so the lemon count goes on top of the fraction.}
\step{11 lemons to 6 limes gives $\ans{\tfrac{11}{6}}$ --- flipping it to
$\tfrac{6}{11}$ would answer a different question.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A drawer holds 9 spoons and 4 forks. Write the ratio of
spoons to forks as a fraction.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{9}{4}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A reversed ratio often \emph{looks} fine because it uses
the same two numbers. Read the question again and check which quantity was named
first.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Testing Whether Two Ratios Are Equivalent}

\begin{formulabox}
\textbf{Rule:} Simplify both ratios to simplest form. Equivalent ratios simplify
to the \textbf{same} ratio. Multiplying both parts by the same number keeps a
ratio equivalent; \textbf{adding} the same number to both parts does not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Are $8{:}12$ and $10{:}15$ equivalent? Write $=$ or
$\neq$.
\step{Simplify both: dividing 8 and 12 by 4 gives two-thirds, and dividing 10
and 15 by 5 also gives two-thirds.}
\step{Same simplest form, so $\ans{\tfrac{8}{12} = \tfrac{10}{15}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Are $6{:}9$ and $8{:}12$ equivalent? Write $=$ or
$\neq$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{6}{9} = \tfrac{8}{12}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Fixing a Broken Ratio Table}

\begin{formulabox}
\textbf{Rule:} Test every column the same way: divide one row by the other (or
simplify the column's ratio) and check it matches the table's ratio. To repair a
bad entry, scale the \textbf{base column} by the right multiplier.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A $3{:}5$ table pairs a top entry of 12 with a bottom
entry of 21. Fix the bottom entry.
\step{Find the multiplier from the base column: $12 \div 3 = 4$, so every entry
in that column is the base times 4.}
\step{Bottom entry: $5 \times 4 = 20$, so the corrected entry is $\ans{20}$
(the 21 came from a skip-counting slip).}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A $4{:}3$ table pairs a top entry of 16 with a smudged
bottom entry. What should the bottom entry be?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Reversed ratio:] the right numbers in the wrong order
  \item[Equivalent ratios:] ratios that simplify to the same form
  \item[Multiplier:] what both parts are scaled by
  \item[Rate:] a ratio of two different kinds of units
\end{description}
\end{multicols}
