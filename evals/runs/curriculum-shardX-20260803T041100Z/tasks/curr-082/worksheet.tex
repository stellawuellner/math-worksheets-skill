\begin{center}
\textit{Naming the fraction at a marked point, placing a fraction on a
partitioned line, and comparing two fractions on the line}
\end{center}

\noindent\textbf{How to read a fraction on a number line.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Count the \emph{equal parts} between $0$ and $1$ --- not the tick marks.
That count is the bottom number (the denominator).
\item Count how many of those parts you travel from $0$ to reach the point.
That count is the top number (the numerator).
\item Always start counting at $0$, and always count whole parts.
\end{itemize}

\problem[4cm]{The line below goes from $0$ to $1$. It is cut into $2$ equal
parts. Point $A$ sits on the first tick mark after $0$. Write the fraction that
names point $A$.
\begin{center}
\begin{tikzpicture}[x=6cm,y=1cm]
  \draw[thick] (0,0) -- (2,0);
  \foreach \i in {0,1,2} \draw[thick] (\i,0.16) -- (\i,-0.16);
  \node[below=4pt] at (0,0) {$0$};
  \node[below=4pt] at (2,0) {$1$};
  \fill (1,0) circle (2.4pt);
  \node[above=5pt] at (1,0) {$A$};
\end{tikzpicture}
\end{center}}

\problem[4cm]{The top line is cut into thirds and the bottom line is cut into
fourths. Mark $\frac{2}{3}$ on the top line and $\frac{2}{4}$ on the bottom
line. Which one lands farther from $0$? Write $>$ or $<$ in the blank so the
sentence is true.
\begin{center}
\begin{tikzpicture}[x=4cm,y=1cm]
  \draw[thick] (0,0) -- (3,0);
  \foreach \i in {0,1,2,3} \draw[thick] (\i,0.16) -- (\i,-0.16);
  \node[below=4pt] at (0,0) {$0$};
  \node[below=4pt] at (3,0) {$1$};
\end{tikzpicture}\\[0.35cm]
\begin{tikzpicture}[x=3cm,y=1cm]
  \draw[thick] (0,0) -- (4,0);
  \foreach \i in {0,1,2,3,4} \draw[thick] (\i,0.16) -- (\i,-0.16);
  \node[below=4pt] at (0,0) {$0$};
  \node[below=4pt] at (4,0) {$1$};
\end{tikzpicture}
\end{center}
\hspace{1cm}$\frac{2}{3}$ \underline{\hspace{1.2cm}} $\frac{2}{4}$}
