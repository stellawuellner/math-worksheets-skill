\sstitleblock{Convergence Tests for Infinite Series}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Ratio Test}

\begin{formulabox}
\textbf{When to reach for it:} the terms contain $n!$, $r^{n}$, or a power $(x-c)^{n}$ --- anything that simplifies when you divide consecutive terms.\par
\textbf{Test:} let $L=\lim\limits_{n\to\infty}\left|a_{n+1}/a_{n}\right|$. Then $L<1$ converges absolutely; $L>1$ (or $L=\infty$) diverges; $L=1$ tells you nothing --- use another test.\par
\textbf{Habit:} write $a_{n+1}$ and multiply by the RECIPROCAL of $a_{n}$; cancel before taking the limit.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum_{n=1}^{\infty}\frac{n^{2}}{3^{n}}$.
\step{The term has $3^{n}$ in it, so consecutive terms differ by a constant factor --- that is exactly what the ratio test exploits.}
\step{$\left|\dfrac{a_{n+1}}{a_{n}}\right|=\dfrac{(n+1)^{2}}{3^{\,n+1}}\cdot\dfrac{3^{n}}{n^{2}}=\dfrac{1}{3}\left(\dfrac{n+1}{n}\right)^{2}\to\dfrac{1}{3}$.}

$L = \ans{\tfrac{1}{3}} < 1$, so the series converges absolutely.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $L$ for $\displaystyle\sum_{n=1}^{\infty}\frac{n}{4^{n}}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $L = \ans{\tfrac{1}{4}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The Integral Test}

\begin{formulabox}
\textbf{When to reach for it:} $a_{n}=f(n)$ for a function you can actually antidifferentiate --- typically $\tfrac{1}{n^{p}}$ or something with $\ln n$.\par
\textbf{Hypotheses (state them):} on $[k,\infty)$, $f$ must be positive, continuous, and decreasing.\par
\textbf{Test:} $\sum f(n)$ and $\int_{k}^{\infty}f(x)\,dx$ converge together. The integral's value is NOT the sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{3/2}}$.
\step{The term is $f(n)$ for $f(x)=x^{-3/2}$, which is easy to antidifferentiate, so the integral test is cheaper than any comparison.}
\step{$f$ is positive, continuous, decreasing on $[1,\infty)$, and $\int_{1}^{\infty}x^{-3/2}dx=\lim\limits_{b\to\infty}\left[-2x^{-1/2}\right]_{1}^{b}$.}

The integral equals $\ans{2}$, a finite number, so the series converges.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_{1}^{\infty}\frac{dx}{x^{3}}$ and say what it proves about $\sum 1/n^{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2}}$, so the series converges}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Direct Comparison}

\begin{formulabox}
\textbf{When to reach for it:} the term is ``almost'' a geometric series or a $p$-series, spoiled by one extra piece.\par
\textbf{Test (non-negative terms only):} if $0\le a_{n}\le b_{n}$ and $\sum b_{n}$ converges, then $\sum a_{n}$ converges. If $a_{n}\ge c_{n}\ge 0$ and $\sum c_{n}$ diverges, then $\sum a_{n}$ diverges.\par
\textbf{Benchmarks:} $\sum r^{n}$ converges iff $|r|<1$; \ $\sum 1/n^{p}$ converges iff $p>1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum_{n=1}^{\infty}\frac{1}{2^{n}+n}$.
\step{Adding $n$ to the denominator only makes each term smaller, so compare upward against the geometric series it came from.}
\step{$0<\dfrac{1}{2^{n}+n}\le\dfrac{1}{2^{n}}$, and $\displaystyle\sum_{n=1}^{\infty}\dfrac{1}{2^{n}}$ is geometric with $r=\tfrac{1}{2}$, summing to $\dfrac{1/2}{1-1/2}$.}

The benchmark sum is $\ans{1}$, finite, so the original series converges.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which benchmark bounds $\displaystyle\sum_{n=1}^{\infty}\frac{1}{3^{n}+1}$, and what is its sum?\\[2pt]
\rotatebox{180}{\footnotesize check: $\sum 1/3^{n} = \ans{\tfrac{1}{2}}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Alternating Series and the Error Bound}

\begin{formulabox}
\textbf{Test:} $\sum(-1)^{n+1}b_{n}$ with $b_{n}>0$ converges when $b_{n}$ is decreasing AND $b_{n}\to 0$. Both conditions must be checked.\par
\textbf{Error bound:} $|S-S_{N}|\le b_{N+1}$ --- the FIRST term you left out, never the last one you kept.\par
\textbf{Conditional vs absolute:} if $\sum b_{n}$ also converges the convergence is absolute; if $\sum b_{n}$ diverges it is only conditional.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n}$, give $S_{5}$ and bound the error.
\step{The signs alternate and $1/n$ decreases to zero, so the alternating series test applies and its remainder bound is available.}
\step{$S_{5}=1-\tfrac{1}{2}+\tfrac{1}{3}-\tfrac{1}{4}+\tfrac{1}{5}$, and the bound is the next term $b_{6}=\tfrac{1}{6}$.}

$S_{5} = \ans{\tfrac{47}{60}}$ \quad and \quad $|S-S_{5}| \le \ans{\tfrac{1}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{2}}$, give $S_{4}$ and the error bound.\\[2pt]
\rotatebox{180}{\footnotesize check: $S_{4} = \ans{\tfrac{115}{144}}$, $|S-S_{4}| \le \ans{\tfrac{1}{25}}$}
\end{tryitbox}
