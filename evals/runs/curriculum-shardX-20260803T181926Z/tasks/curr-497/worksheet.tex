\wstitleblock{Convergence Tests for Infinite Series}{AP Calculus BC}{}

\begin{center}
\textit{Choosing among the direct comparison, integral, ratio, and alternating series tests}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Name the test you are using before you use it, and check its hypotheses.
  \item Give exact values (fractions, logarithms) rather than decimals.
  \item A convergence conclusion with no supporting limit or integral earns no credit --- show the quantity the test depends on.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{Consider $\displaystyle\sum_{n=0}^{\infty}\frac{1}{3^{n}}$.
\par\vspace{0.1cm}
(a) Find its exact sum.
\par\vspace{0.1cm}
(b) Use a direct comparison to explain why $\displaystyle\sum_{n=0}^{\infty}\frac{1}{3^{n}+n}$ also converges.}

\problem[6cm]{Apply the ratio test to $\displaystyle\sum_{n=1}^{\infty}\frac{n}{2^{n}}$. Simplify $\left|\dfrac{a_{n+1}}{a_{n}}\right|$, evaluate $L=\displaystyle\lim_{n\to\infty}\left|\dfrac{a_{n+1}}{a_{n}}\right|$, and state the conclusion.}

\problem[6.5cm]{Check that $f(x)=\dfrac{1}{x^{2}}$ is positive, continuous, and decreasing on $[1,\infty)$, then use the integral test on $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}}$. Evaluate $\displaystyle\int_{1}^{\infty}\frac{dx}{x^{2}}$ exactly and state what it proves.}

\problem[6.5cm]{Let $S=\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{3}}$.
\par\vspace{0.1cm}
(a) Compute the fourth partial sum $S_{4}$ as an exact fraction.
\par\vspace{0.1cm}
(b) Give the alternating-series error bound for $|S-S_{4}|$ as an exact fraction.}

\problem[6cm]{Use the ratio test on $\displaystyle\sum_{n=0}^{\infty}\frac{2^{n}}{n!}$. Simplify $\dfrac{a_{n+1}}{a_{n}}$ completely before taking the limit, then state the conclusion.}

\problem[7cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}+3n}$.
\par\vspace{0.1cm}
(a) Use a direct comparison with a $p$-series to show the series converges.
\par\vspace{0.1cm}
(b) Using $\dfrac{1}{n(n+3)}=\dfrac{1}{3}\left(\dfrac{1}{n}-\dfrac{1}{n+3}\right)$, telescope the partial sums and find the exact sum.}

\problem[7cm]{Use the integral test to decide whether $\displaystyle\sum_{n=2}^{\infty}\frac{1}{n(\ln n)^{2}}$ converges. Evaluate the improper integral exactly (a substitution $u=\ln x$ helps) and state the conclusion.}

\problem[7cm]{For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{2}}$, find the smallest number of terms $N$ for which the alternating-series bound guarantees $|S-S_{N}|<0.01$, and give the bound $|a_{N+1}|$ as an exact fraction.}

\problem[7cm]{Use the ratio test to find the radius of convergence of the power series $\displaystyle\sum_{n=1}^{\infty}\frac{n\,(x-2)^{n}}{5^{n}}$, then state the open interval on which the ratio test guarantees convergence.}

\problem[8cm]{Show that $\displaystyle\sum_{n=1}^{\infty}(-1)^{n}\,\frac{n}{n^{2}+1}$ converges but is not absolutely convergent. Name the test used for each half of the argument, and verify the hypotheses of the test you use for convergence.}
