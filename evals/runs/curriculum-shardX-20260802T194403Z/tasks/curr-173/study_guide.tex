\sstitleblock{Expressions: Distribute, Combine, Factor}

\skillheading{1. Distributing over a sum (distribute-over-sum)}

\begin{formulabox}
\textbf{Rule:} \quad $a(b + c) = ab + ac$ \quad and \quad $a(b - c) = ab - ac$\\[3pt]
\textbf{Why:} $a(b+c)$ is the area of a rectangle of height $a$ whose base is cut
into $b$ and $c$. Cutting the base cuts the area into $ab$ and $ac$.
The outside factor multiplies \textbf{every} term inside --- and each term keeps
the sign in front of it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $6(x + 5)$.
\step{Two terms sit inside one set of parentheses, so this is a distribution: multiply the outside factor by each term.}
\step{$6 \cdot x = 6x$ and $6 \cdot 5 = 30$, so $6(x+5) = \ans{6x + 30}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $3(4x + 7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12x + 21}$}
\end{tryitbox}

\skillheading{2. Combining like terms (combine-like-terms)}

\begin{formulabox}
\textbf{Like terms} have exactly the same variable part: $5x$ and $3x$ are like;
$5x$ and $5$ are not. \quad \textbf{Rule:} add or subtract the coefficients and
keep the variable part: $5x + 3x = 8x$.\par
Every term travels with the sign in front of it, so $9 - 4$ is a subtraction of
like number terms, not a stray minus.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $5x + 9 + 3x - 4$.
\step{There are no parentheses, only terms of two different kinds, so this is pure sorting: group like with like.}
\step{$(5x + 3x) + (9 - 4) = \ans{8x + 5}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $7x + 12 - 2x - 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x + 7}$}
\end{tryitbox}

\skillheading{3. Distribute, then combine (distribute-then-combine)}

\begin{formulabox}
\textbf{Order matters:} you can only combine terms once nothing is trapped
inside parentheses.\par
\textbf{Step 1} distribute every product. \quad \textbf{Step 2} collect like terms.\par
A term \emph{outside} the parentheses (like the $5x$ in $2(x+6) + 5x$) is never
multiplied by the outside factor --- it only joins in at step 2.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $2(x + 6) + 5x$.
\step{Part of this expression is locked inside parentheses, so distribute first and only then look for like terms.}
\step{$2(x+6) = 2x + 12$, then $2x + 12 + 5x = (2x + 5x) + 12 = \ans{7x + 12}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $4(3x - 1) - 2x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10x - 4}$}
\end{tryitbox}

\skillheading{4. Factoring out a common factor}

\begin{formulabox}
\textbf{Rule (distribution backwards):} \quad $ab + ac = a(b + c)$\\[3pt]
Find the \textbf{greatest} common factor of the terms, write it outside, and
divide each term by it to get what stays inside. Check by distributing: you must
land back on the expression you started with.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $8x + 20$.
\step{Both terms share a whole-number factor, so run the area model backwards: the shared factor is the height.}
\step{GCF of $8$ and $20$ is $4$; $8x \div 4 = 2x$ and $20 \div 4 = 5$, so $8x + 20 = \ans{4(2x + 5)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $9x - 24$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3(3x - 8)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $3(2x - 5) \neq 6x - 5$. The outside factor must reach
the second term too: $6x - 15$. Test any doubtful step by substituting a number
--- equivalent expressions agree for every value.
\end{watchoutbox}
