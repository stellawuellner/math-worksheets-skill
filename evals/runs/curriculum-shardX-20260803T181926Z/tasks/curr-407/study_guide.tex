\sstitleblock{Polynomial Zeros and Turning-Point Constraints}

\skillheading{1. Factor, then read the zeros}

\begin{formulabox}
\textbf{Rule.} The real zeros of a polynomial are the $x$-values where every
factor can be made zero: factor completely, set each linear factor to $0$, and
solve.\\[3pt]
\textbf{Order of attack:} common factor first, then a recognisable pattern
(difference of squares, quadratic in $x^{2}$), then the Rational Root Theorem
--- any rational zero of a monic polynomial divides the constant term.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the real zeros of $f(x) = x^{2} - x - 12$.
\step{Monic quadratic with no common factor, so look for a factor pair of the
constant term rather than reaching for the quadratic formula.}
\step{$-4$ and $3$ multiply to $-12$ and add to $-1$, so
$f(x) = (x - 4)(x + 3)$.}

$\Rightarrow\quad \ans{x = 4,\ -3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the real zeros of $x^{2} + 2x - 15$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3,\ -5}$}
\end{tryitbox}

\skillheading{2. Multiplicity: cross or touch}

\begin{formulabox}
\textbf{Rule.} The exponent on a linear factor is that zero's
\textbf{multiplicity}.\\[3pt]
\textbf{Odd} multiplicity: the factor changes sign, so the graph
\emph{crosses} the axis. \textbf{Even} multiplicity: no sign change, so the
graph \emph{touches} and turns back --- and that touch point is itself a
turning point.\\[3pt]
Multiplicities add up to the degree, so they also tell you how many crossings
to expect.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Describe the zeros of $f(x) = (x + 1)^{2}(x - 4)$.
\step{The form is already factored, so nothing needs solving --- read each
zero and its exponent straight off.}
\step{$x = -1$ comes from a squared factor (multiplicity $2$, even, so the
graph touches); $x = 4$ comes from a first-power factor (multiplicity $1$,
odd, so the graph crosses).}

$\Rightarrow\quad \ans{x = -1,\ 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the zeros of $x^{2}(x - 5)$ and say which one is
a touch.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 0,\ 5}$ (touch at $0$)}
\end{tryitbox}

\skillheading{3. What the degree allows}

\begin{formulabox}
\textbf{Bounds.} A polynomial of degree $n$ has \textbf{at most $n$} real
zeros and \textbf{at most $n - 1$} turning points.\\[3pt]
\textbf{Read them both ways.} Forwards: degree $6$ allows up to $5$ turns.
Backwards: a graph showing $m$ turning points needs degree at least
$m + 1$.\\[3pt]
\textbf{And a lower bound:} the graph must turn between two consecutive
distinct real zeros, so $k$ distinct zeros force at least $k - 1$ turns.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is the greatest number of turning points a
degree-$6$ polynomial can have?
\step{This is the forwards reading of the bound, so subtract one from the
degree --- no factoring is involved.}
\step{$n - 1$ with $n = 6$.}

$\Rightarrow\quad \ans{5}$ turning points.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph has $3$ turning points. What is the smallest
degree it could have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{4. End behaviour of a rational function}

\begin{formulabox}
\textbf{Rule.} Compare degrees of numerator and denominator.\\[3pt]
Numerator degree \emph{smaller}: the limit is $0$, so $y = 0$. \ \ Degrees
\emph{equal}: the limit is the ratio of leading coefficients. \ \ Numerator
degree \emph{larger}: no horizontal asymptote.\\[3pt]
\textbf{The move that proves it:} divide every term by the highest power of
$x$; each leftover fraction goes to $0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find
$\displaystyle\lim_{x \to \infty} \frac{2x^{2} - x}{x^{2} + 3}$.
\step{Top and bottom have the same degree, so expect a nonzero horizontal
asymptote --- divide through by $x^{2}$ to see it.}
\step{The expression becomes
$\dfrac{2 - 1/x}{1 + 3/x^{2}}$, and both small fractions go to $0$.}

$\Rightarrow\quad \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find
$\displaystyle\lim_{x \to \infty} \dfrac{5x^{3} + 1}{2x^{3} - x}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{5}{2}}$}
\end{tryitbox}
