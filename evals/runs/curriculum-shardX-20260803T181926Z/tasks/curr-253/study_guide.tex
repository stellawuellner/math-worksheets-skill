\sstitleblock{Rewriting Expressions with Distributive Structure}

\skillheading{1. Distribute a factor over a sum}

\begin{formulabox}
\textbf{Rule:} $a(b + c) = ab + ac$ \quad and \quad $a(b - c) = ab - ac$.\\[3pt]
\textbf{Model:} a rectangle $a$ tall and $(b + c)$ wide splits into an
$a$-by-$b$ piece and an $a$-by-$c$ piece. One rectangle, two pieces, same area
--- that is the whole rule.\\[3pt]
The multiplier reaches \emph{every} term inside the parentheses, and each term
keeps its own sign.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Expand $7(3x + 4)$.
\step{One factor outside, a sum inside --- distribute, so the $7$ multiplies
both terms.}
\step{$7 \cdot 3x = 21x$ and $7 \cdot 4 = 28$, then write them as a sum.}

$\Rightarrow\quad 7(3x + 4) = \ans{21x + 28}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Expand $5(2x - 9)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10x - 45}$}
\end{tryitbox}

\skillheading{2. Expand a product of two binomials}

\begin{formulabox}
\textbf{Rule:} $(x + p)(x + q) = x^2 + qx + px + pq$ --- distribute twice, so
every term of the first factor meets every term of the second (four products,
then combine the two like terms).\\[3pt]
\textbf{Model:} a rectangle $(x + p)$ by $(x + q)$ cuts into four smaller
rectangles: the $x$-by-$x$ square, two strips, and the $p$-by-$q$ corner.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Expand $(x + 4)(x + 6)$.
\step{Both factors are sums, so the same rule is applied twice --- there is no
single number outside to distribute.}
\step{The four products are $x \cdot x = x^2$, $6x$, $4x$ and $24$; the middle
two are like terms, and $6x + 4x = 10x$.}

$\Rightarrow\quad (x + 4)(x + 6) = \ans{x^2 + 10x + 24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Expand $(x + 2)(x + 7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 9x + 14}$}
\end{tryitbox}

\skillheading{3. Factor out the greatest common factor}

\begin{formulabox}
\textbf{Rule:} $ab + ac = a(b + c)$ --- factoring is distribution read
right-to-left.\\[3pt]
\textbf{How to find the GCF:} take the largest number that divides every
coefficient, times the highest power of the variable that appears in
\emph{every} term. Then divide each term by it and write the quotients inside
the parentheses. Distribute back to check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $8x + 20$.
\step{Both terms are being added and share a numerical factor, so pull out the
greatest common factor rather than expanding anything.}
\step{The GCF of $8$ and $20$ is $4$; there is no $x$ in the second term, so no
$x$ comes out. Divide: $8x \div 4 = 2x$ and $20 \div 4 = 5$.}

$\Rightarrow\quad 8x + 20 = \ans{4(2x + 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $9x^2 + 15x$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x(3x + 5)}$}
\end{tryitbox}

\skillheading{4. Check whether two forms are equivalent}

\begin{formulabox}
\textbf{Two ways to decide.} \textbf{(1) Rewrite:} expand and combine like
terms until both sides are in the form $ax + b$; equal forms means equivalent
expressions. \textbf{(2) Test a value:} substitute the same number for $x$ in
both. Equivalent expressions agree at every value, so one disagreement proves
they are different --- but agreement at one value is not a proof.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $3(2x + 5) + 4x$ in the form $ax + b$.
\step{Parentheses first, loose terms after --- distribute, then collect like
terms.}
\step{$3(2x + 5) = 6x + 15$; adding $4x$ gives $6x + 4x = 10x$ and the constant
$15$ is unchanged.}

$\Rightarrow\quad 3(2x + 5) + 4x = \ans{10x + 15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $2(4x - 3) + 5$ in the form $ax + b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8x - 1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} distributing to the first term only is the most common
error --- $5(2x - 3)$ is $10x - 15$, never $10x - 3$. And ``factored'' means
\emph{completely} factored: $5(3x^2 + 2x)$ still hides a shared $x$.
\end{watchoutbox}
