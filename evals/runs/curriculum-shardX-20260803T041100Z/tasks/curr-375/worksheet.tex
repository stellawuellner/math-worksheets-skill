\wstitleblock{Radicals, Rational Exponents, Domains, and Approximate Solutions}{Algebra 2 \quad Radical Expressions and Equations}{}

\begin{center}
\textit{Rational exponents, combining like radicals, solving radical equations
with domain and extraneous-root checks, and approximating irrational solutions}
\end{center}

\noindent\textbf{The rules this sheet runs on.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item $a^{m/n} = \sqrt[n]{a^m}$ --- the denominator is the root, the numerator is
the power.
\item Radicals combine like terms only when the radicands match after
simplifying: $2\sqrt{3} + 5\sqrt{3} = 7\sqrt{3}$, but $\sqrt{2} + \sqrt{3}$ does
not combine.
\item Squaring both sides of an equation can \textbf{create} solutions that the
original equation does not have. Every candidate must be substituted back.
\item $\sqrt{\phantom{x}}\,$ returns the \textbf{non-negative} root, so an
equation whose other side is negative has no solution there.
\end{itemize}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Decide first what kind of task it is: rewrite, combine, solve, or
approximate. The four kinds alternate.
\item On every equation, state the domain restriction and check each candidate.
\item Round approximations to two decimal places unless told otherwise.
\end{itemize}

\problem[6cm]{Evaluate without a calculator: \quad $49^{1/2}$

\smallskip
Say in one sentence what the exponent $\tfrac{1}{2}$ instructs you to do.}

\problem[6cm]{Write $\sqrt{50}$ in simplest radical form.

\smallskip
Find the largest perfect square that divides 50, then take its root outside the
radical.}

\problem[7cm]{Solve for $x$: \quad $\sqrt{x+7} = 4$

\smallskip
\textbf{(a)} State the domain: for what values of $x$ is the left side defined?

\smallskip
\textbf{(b)} Solve, then substitute your answer back into the original equation
to confirm it works.}

\problem[6cm]{Approximate $\sqrt{20}$ to two decimal places.

\smallskip
Simplify the radical first, then approximate. Show the simplified exact form as
well as the decimal.}

\problem[6cm]{Evaluate without a calculator: \quad $27^{2/3}$

\smallskip
Take the root indicated by the denominator before applying the power in the
numerator --- the numbers stay small that way.}

\problem[7cm]{Combine into a single radical term:
\[
3\sqrt{12} + \sqrt{27}
\]
Neither radical is in simplest form yet, so simplify both before deciding
whether they are like terms.}

\problem[8cm]{Solve for $x$: \quad $\sqrt{2x+3} = x$

\smallskip
\textbf{(a)} Square both sides and solve the resulting quadratic. You should get
two candidates.

\smallskip
\textbf{(b)} Test both candidates in the \emph{original} equation. Put the
surviving solution on the answer line, and explain in one sentence why the other
one fails.}

\problem[7cm]{Approximate $\sqrt{2} + \sqrt{8}$ to two decimal places.

\smallskip
Simplify and combine the two radicals into a single term first. Then say why
$\sqrt{2} + \sqrt{8}$ is \emph{not} $\sqrt{10}$.}

\problem[6cm]{Evaluate without a calculator: \quad $16^{-1/2}$

\smallskip
The exponent is both negative and fractional. Handle one at a time, and state
whether the result is positive or negative before you compute it.}

\problem[8cm]{Solve for $x$: \quad $\sqrt{x+5} = x - 1$

\smallskip
\textbf{(a)} Before squaring, state what must be true of $x - 1$ for a solution
to exist at all.

\smallskip
\textbf{(b)} Square, solve, and test both candidates in the original equation.
Put the surviving solution on the answer line.}

\problem[7cm]{Write as a single power of $x$:
\[
x^{1/2} \cdot x^{1/3}
\]
The bases match, so the exponents add. Give the answer as $x$ to a single
rational exponent.}

\problem[8.5cm]{\textbf{Challenge.} Two equations that look almost the same:
\[
\text{(i)}\quad \sqrt{2x+7} = x + 2
\qquad\qquad
\text{(ii)}\quad \sqrt{2x+7} = x + 1
\]

\smallskip
\textbf{(a)} Solve equation (i). Square, solve the quadratic, test both
candidates, and put the surviving solution on the answer line.

\smallskip
\textbf{(b)} Now solve equation (ii). Its exact solutions come out as
$\pm\sqrt{6}$. One is extraneous --- say which and why, then give the surviving
solution rounded to two decimal places.

\smallskip
\textbf{(c)} In one sentence, explain why squaring is what makes the
extraneous-root check necessary in both parts.}
