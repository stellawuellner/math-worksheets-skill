\sstitleblock{Handling Ambiguous SSA Cases Completely}

\begin{formulabox}
\textbf{SSA means you know $A$, its opposite side $a$, and the adjacent side
$b$.} The side $a$ swings from the far vertex like a hinge, so it may meet the
base line twice, once, or never. Test before you solve, using the height
$h = b\sin A$:
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item $a < h$: \textbf{no} triangle --- the side cannot reach.
\item $a = h$: \textbf{one}, with a right angle at $B$.
\item $h < a < b$: \textbf{two} --- the ambiguous case.
\item $a \ge b$: \textbf{one} --- $B$ must be acute, since the larger side faces
the larger angle.
\end{itemize}
The law of sines returns only $\arcsin$, which is always acute. The supplement
$180^\circ - B$ has the same sine, and it is a second triangle exactly when
$A + (180^\circ - B) < 180^\circ$.
\end{formulabox}

\skillheading{1. Test the case first}

\begin{formulabox}
Compute $h = b\sin A$, then compare $h$, $a$ and $b$ before touching the law of
sines. One multiplication decides how many answers the problem has --- and
therefore how many you must report. The gap $a - h$ is worth writing down too:
it is how much shorter the measured side could have been before the second
triangle disappears.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A = 25^\circ$, $b = 18$, $a = 10$. How many triangles?
\step{The count depends only on how the swinging side $a$ compares with the
height, so compute $h$ before solving anything.}
\step{$h = 18\sin 25^\circ$, and since this is less than $a = 10$, which is in
turn less than $b = 18$, the measurement gives two triangles.}
\[ \ans{h = 7.61} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $A = 35^\circ$, $b = 20$, $a = 8$. Find $h$ and say how many
triangles exist.

\rotatebox{180}{\footnotesize check: $\ans{h = 11.47}$, and $a < h$, so none}
\end{tryitbox}

\skillheading{2. Two triangles: finish both}

\begin{formulabox}
Find the acute $B_1$ from $\sin B = \frac{b\sin A}{a}$, then take
$B_2 = 180^\circ - B_1$ and confirm $A + B_2 < 180^\circ$. Each $B$ has its own
third angle $C = 180^\circ - A - B$ and its own side
$c = \frac{a\sin C}{\sin A}$. Report both triangles.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A = 32^\circ$, $b = 15$, $a = 10$. Find both values of $B$.
\step{Here $h = 7.95 < a < b$, so two triangles exist and the acute answer alone
would be half the solution.}
\step{$\sin B = \frac{15\sin 32^\circ}{10} = 0.7949$ gives the acute value
below, and its supplement $127.36^\circ$ passes the test since
$32^\circ + 127.36^\circ < 180^\circ$.}
\[ \ans{B_1 = 52.64^\circ} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $A = 41^\circ$, $b = 22$, $a = 16$. Find the acute $B$, then
its supplement.

\rotatebox{180}{\footnotesize check: $\ans{B_1 = 64.43^\circ}$, so
$B_2 = 115.57^\circ$}
\end{tryitbox}

\skillheading{3. One triangle: show the other candidate fails}

\begin{formulabox}
When $a \ge b$ the supplement always fails, because $A + (180^\circ - B)$ comes
out above $180^\circ$. Say so explicitly --- ``$B_2$ rejected, $A + B_2 >
180^\circ$'' is the line that shows you tested rather than forgot.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A = 55^\circ$, $b = 9$, $a = 14$. Find $B$ and justify that
it is the only one.
\step{Since $a = 14$ is longer than $b = 9$, the angle opposite $a$ is the
larger one, so $B$ cannot be obtuse and one triangle is guaranteed.}
\step{$\sin B = \frac{9\sin 55^\circ}{14} = 0.5266$, and the supplement
$148.22^\circ$ would give $A + B_2 = 203.22^\circ > 180^\circ$, so it is
rejected.}
\[ \ans{B = 31.78^\circ} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $A = 63^\circ$, $b = 12$, $a = 20$. Find $B$ and say in one
line why there is no second triangle.

\rotatebox{180}{\footnotesize check: $\ans{B = 32.32^\circ}$, and $a > b$
rejects the supplement}
\end{tryitbox}

\begin{watchoutbox}
\textbf{Four ways this goes wrong.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item Reporting the calculator's $\arcsin$ and stopping. It only ever returns
the acute angle, so the second triangle never appears unless you look for it.
\item Using $a\sin A$ for the height. It is the \emph{fixed} side $b$ that
swings down to the base line, so $h = b\sin A$.
\item Reaching for the cosine. The height is opposite $A$ in the right triangle
that defines it, so it is a sine.
\item Rounding $B_1$ before subtracting it from $180^\circ$. Keep full precision
until the last step, or $B_2$, $C$ and $c$ all inherit the error.
\end{itemize}
\end{watchoutbox}
