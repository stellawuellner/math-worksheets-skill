\sstitleblock{Writing and Interpreting Circle Equations}

\skillheading{Skill 1 --- Read the center and radius off an equation}
\begin{formulabox}\textbf{Standard form:}\quad $(x-h)^2 + (y-k)^2 = r^2$, center $(h, k)$, radius $r$.\\[4pt]
The form \emph{subtracts} the center, so $x + 1$ means $h = -1$. The right-hand side is
$r^2$, so take a square root to get $r$. A point is on the circle exactly when its
coordinates make the left side equal $r^2$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Give the center and radius of $(x+1)^2 + (y-4)^2 = 64$.
\step{Rewrite each bracket as a subtraction to read the center, then square-root the right side for the radius.}
\step{$x+1 = x-(-1)$ and $y-4$ give center $(-1, 4)$; the radius is $\sqrt{64} = \ans{8}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Find the radius of $(x-5)^2 + (y+2)^2 = 45$, to two
decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6.71}$}\end{tryitbox}

\skillheading{Skill 2 --- Write the equation from a center and radius}
\begin{formulabox}\textbf{Recipe:} substitute $h$, $k$ and $r$ into $(x-h)^2 + (y-k)^2 = r^2$.\\[4pt]
Square the radius --- the right-hand side is $r^2$, never $r$. If the radius is not given,
it is the distance from the center to a known point of the circle.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Write the equation with center $(2, -3)$ and radius $4$.
\step{The center and radius are both given, so substitute directly and square the radius.}
\step{$h = 2$, $k = -3$, $r^2 = 4^2 = 16$, giving $\ans{(x-2)^2 + (y+3)^2 = 16}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write the equation with center $(-5, 1)$ and radius $7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+5)^2 + (y-1)^2 = 49}$}\end{tryitbox}

\skillheading{Skill 3 --- Complete the square to reach standard form}
\begin{formulabox}\textbf{From $x^2 + y^2 + Dx + Ey + F = 0$:}\\[4pt]
group the $x$-terms and the $y$-terms, move $F$ to the right, then add
$\left(\tfrac{D}{2}\right)^2$ and $\left(\tfrac{E}{2}\right)^2$ to \textbf{both} sides.
Whatever you add on the left must be added on the right, or the radius comes out wrong.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Write $x^2 + y^2 - 4x + 2y - 4 = 0$ in standard form.
\step{The equation is in general form, so group each variable's terms and complete both squares.}
\step{Half of $-4$ squared is $4$ and half of $2$ squared is $1$: $(x^2-4x+4)+(y^2+2y+1) = 4+4+1$, giving $\ans{(x-2)^2 + (y+1)^2 = 9}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $x^2 + y^2 + 6x - 8y + 9 = 0$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+3)^2 + (y-4)^2 = 16}$}\end{tryitbox}

\skillheading{Skill 4 --- Use the equation to answer a question}
\begin{formulabox}\textbf{Substitute what you know.}\\[4pt]
To find where a circle meets a horizontal line $y = c$, put $c$ in for $y$ and solve the
remaining equation for $x$; a positive leftover gives two crossing points, zero gives one
(tangent), and a negative leftover means the line misses the circle.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Where does $(x-1)^2 + (y-2)^2 = 25$ meet the line $y = 6$?
\step{The line fixes $y$, so substitute $y = 6$ and the circle equation becomes a single equation in $x$.}
\step{$(x-1)^2 + 16 = 25$, so $(x-1)^2 = 9$ and $x - 1 = \pm 3$: $\ans{x = -2 \text{ or } x = 4}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Where does $(x+2)^2 + (y-1)^2 = 100$ meet the line $y = 7$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -10 \text{ or } x = 6}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} The number on the right is $r^2$. A circle
written $=\ 20$ has radius $\sqrt{20} \approx 4.47$, and an equation whose right side comes
out negative has no graph at all.
\end{watchoutbox}
