\sstitleblock{Powers of Ten and Place-Value Relationships}

\begin{formulabox}
\textbf{One rule runs the whole system.} Each place is worth $10$ times the
place to its right and $\tfrac{1}{10}$ of the place to its left.\\[2pt]
\begin{tabular}{@{}cccccccc@{}}
hundred th. & ten th. & thousands & hundreds & tens & ones & tenths & hundredths \\
$10^5$ & $10^4$ & $10^3$ & $10^2$ & $10^1$ & $10^0$ & $\frac{1}{10}$ & $\frac{1}{100}$ \\
\end{tabular}\\[2pt]
An exponent on $10$ counts \emph{places}, and for whole numbers it also counts
\emph{zeros}: $10^4 = 10000$.
\end{formulabox}

\skillheading{1. Comparing two digit values}

\begin{formulabox}
Compare \textbf{values}, never digits. Write what each digit is worth, then
divide the larger value by the smaller. Places one apart give $10$; two apart
give $100$; three apart give $1000$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} In $770$, how many times as much is the first $7$ worth as the
second $7$?
\step{The two digits are the same, so only their places can differ --- write
each one's value before comparing anything.}
\step{The first $7$ is worth $700$ and the second is worth $70$, and
$700 \div 70$ gives the comparison.}
\[ \ans{10} \quad\text{times as much (one place apart)} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} In $90900$, how many times as much is the $9$ in the
ten-thousands place worth as the $9$ in the hundreds place?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{100}$}
\end{tryitbox}

\skillheading{2. Powers of ten in exponent form}

\begin{formulabox}
$10^n$ means $10$ used as a factor $n$ times:
$10^1 = 10$, $10^2 = 100$, $10^3 = 1000$, and $10^0 = 1$.\\[2pt]
A number that ends in zeros can be written as a digit times a power of ten:
$700 = 7 \times 10^2$. Expanded form uses one term per non-empty place.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write the value of $10^5$.
\step{The exponent is a count of factors of ten, so it is also a count of zeros
for a whole-number power --- no long multiplication is needed.}
\step{Five factors of ten means a $1$ followed by five zeros.}
\[ 10^5 = \ans{100000} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $8 \times 10^3$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8000}$}
\end{tryitbox}

\skillheading{3. Shifting digits by 10, 100 and 1000}

\begin{formulabox}
Multiplying by $10^n$ shifts every digit $n$ places \textbf{left}; dividing by
$10^n$ shifts every digit $n$ places \textbf{right}. Empty whole-number places
take $0$ as a placeholder.\\[2pt]
Say ``the digits move'', not ``add a zero'' --- the zero rule fails the moment a
decimal appears.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $2.4 \times 100$.
\step{$100 = 10^2$, so this is a two-place shift to the left rather than a
multiplication to work out.}
\step{The $2$ moves from ones to hundreds and the $4$ moves from tenths to tens,
leaving the ones place empty, so it takes a $0$.}
\[ 2.4 \times 100 = \ans{240} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $65 \div 10$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6.5}$}
\end{tryitbox}

\skillheading{4. Rounding to a place to estimate}

\begin{formulabox}
Rounding to a place \emph{is} rounding to a power of ten: nearest ten $=10^1$,
nearest hundred $=10^2$, nearest thousand $=10^3$.\\[2pt]
Find the place you were asked for, look at the \textbf{one digit to its right},
and round up when that digit is $5$ or more. Every place to the right becomes
$0$. Then compute with the rounded numbers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Estimate $382 + 419$ to the nearest hundred.
\step{An estimate is exact arithmetic on rounded numbers, so round each addend
first and only then add.}
\step{$382$: the tens digit is $8$, so round up to $400$. $419$: the tens digit
is $1$, so round down to $400$. Now $400 + 400$.}
\[ \ans{800} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Estimate $5170 - 2840$ to the nearest thousand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2000}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} count how many places apart two digits are before you
answer ``ten times'' --- two places apart is $100$ times. And round by the place
you were \emph{asked} for: to round to the nearest hundred, read the tens digit,
not the ones digit.
\end{watchoutbox}
