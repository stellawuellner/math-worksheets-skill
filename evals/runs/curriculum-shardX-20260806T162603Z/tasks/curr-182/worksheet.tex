\wstitleblock{Solving and Graphing One-Step Inequalities}{Grade 7}{}

\begin{center}
\textit{One-step inequalities: undo an addition, undo a multiplication, and turn
the sign around when the divisor is negative}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item \textbf{Worked pattern.} To solve $x - 4 > 3$, add 4 to both sides:
        $x > 7$. Graph it with an \emph{open} circle at 7 (because $>$ does not
        include 7) and shade to the right.
  \item Solve each inequality in one step. Write the solution in the form
        ``letter, sign, number'' on the answer line, then graph it.
  \item \textbf{Open or filled?} $<$ and $>$ get an open circle; $\le$ and
        $\ge$ get a filled one.
  \item \textbf{Turn the sign around} only when you multiply or divide both
        sides by a \emph{negative} number --- never because the answer happens
        to be negative.
\end{itemize}
