\sstitleblock{Function Notation: When $f(x)$ Is Not Multiplication}

\skillheading{Skill 1 --- Evaluating Function Notation}

\begin{formulabox}
\textbf{What the notation says:} $f(\text{input}) = \text{output}$.\par
$f$ names a \emph{rule}, not a number, so the parentheses hold an input slot --- never a factor.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item $f(3)$: replace every $x$ in the rule with $3$, then simplify.
  \item Nothing may be distributed across $f$: $f$ has no value to distribute.
  \item If an $x$ survives after you were asked for $f(3)$, you never substituted.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 2x + 7$. Find $f(5)$.
\step{The $5$ sits in the input slot, so the only legal move is substitution --- replace $x$ by $5$.}
\step{$f(5) = 2(5) + 7 = 10 + 7$, so $f(5) = \ans{17}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $g(x) = x^2 - 3$. Find $g(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 2 --- Fix a Notation-as-Multiplication Error}

\begin{formulabox}
\textbf{Three signatures of the multiplication misreading:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item The work still shows an $x$ after a numeric input was given.
  \item The whole rule got multiplied by the input, so the answer is input-times-too-big.
  \item The rule was ``distributed'' over a sum in the input slot: $f(a+b)$ became $f(a)+f(b)$.
\end{itemize}
\textbf{Also keep straight:} $2f(3)$ doubles the \emph{output}; $f(2 \cdot 3)$ doubles the \emph{input}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $f(x) = 3x - 2$ a student writes $f(4) = 4(3x - 2) = 40$. Fix it.
\step{An $x$ is still present and the input was used as a multiplier, so this is the times-the-rule misreading, not an arithmetic slip.}
\step{Substitute instead: $f(4) = 3(4) - 2 = 12 - 2$, so $f(4) = \ans{10}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $g(x) = x + 6$ a student writes $g(2) = 2(x + 6) = 16$. Fix it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Solving for the Input}

\begin{formulabox}
\textbf{Two different questions:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item ``Find $f(4)$'' gives the input, asks for the output: substitute.
  \item ``Find $x$ with $f(x) = 13$'' gives the output, asks for the input: set the rule equal to the output and solve.
\end{itemize}
You can never divide by $f$ to undo it --- $f$ is a name, not a factor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 3x + 1$. Find every $x$ with $f(x) = 13$.
\step{The output is known and the input is missing, so replace the output by the rule and solve the equation.}
\step{$3x + 1 = 13 \Rightarrow 3x = 12 \Rightarrow \ans{x = 4}$; check $f(4) = 13$ $\checkmark$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = 2x - 5$. Find every $x$ with $f(x) = 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}
