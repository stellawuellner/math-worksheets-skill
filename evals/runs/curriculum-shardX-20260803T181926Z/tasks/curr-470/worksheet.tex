\wstitleblock{Derivative Applications and Modeling}{AP Calculus AB/BC \quad Applications of Derivatives}{}

\begin{center}
\textit{Curve analysis with the first and second derivative, related rates from
one changing quantity, and optimization with a domain check}
\end{center}

\noindent\textbf{Before you start.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item The problems are deliberately mixed. Decide first \emph{which} question
is being asked: a rate that is changing in time (related rates), a largest or
smallest value (optimization), or the shape of a graph (curve analysis).
\item In a related-rates problem, differentiate the relationship with respect
to $t$ \emph{before} substituting any numbers. Substituting first freezes a
variable that is not constant.
\item In an optimization problem, an answer is not finished until you have
checked the endpoints of the physical domain and rejected the roots that lie
outside it.
\item Show the justification, not just the number. On an AP free response the
sign chart or the second-derivative value is where the points are.
\end{itemize}

\bigskip

\problem[6cm]{Let $f(x) = x^3 - 6x^2 + 9x + 2$. \quad Find $f'(x)$.}

\problem[8cm]{Using the same $f(x) = x^3 - 6x^2 + 9x + 2$, find every critical
number of $f$, and state on which intervals $f$ is increasing.}

\problem[8cm]{A spherical weather balloon is inflated so that its radius grows
at a constant $2$ centimetres per second. Its volume is
$V = \dfrac{4}{3}\pi r^3$.

\par\smallskip
Find $\dfrac{dV}{dt}$ at the instant the radius is $5$ centimetres. Give an
exact answer in terms of $\pi$ (cubic centimetres per second).}

\problem{A rectangular garden is fenced on three sides; the fourth side is an
existing wall. There are $60$ metres of fence available. If $x$ is the length
of each of the two sides perpendicular to the wall, the enclosed area is
$A(x) = x(60 - 2x)$.

\begin{enumerate}[label=(\alph*), itemsep=3.6cm, leftmargin=1.5cm, topsep=6pt]
  \item Find the value of $x$ that maximizes $A$. \ansline
  \item Find the maximum area, in square metres. \ansline
\end{enumerate}}

\problem[8cm]{Let $f(x) = x^4 - 4x^3$. Find every $x$ at which $f''(x) = 0$,
then decide which of them are inflection points of $f$ and justify your
decision.}

\problem{A $13$ foot ladder leans against a vertical wall. The foot of the
ladder is pulled away from the wall at a constant $4$ feet per second.

\par\smallskip
How fast is the top of the ladder sliding down the wall at the instant the foot
is $5$ feet from the wall? A negative answer is expected --- say what the sign
means.
\par\vspace*{8.5cm}\answerline{ft/s}}

\problem{An open-topped box is made from a square sheet of cardboard $12$
inches on a side by cutting a square of side $x$ from each corner and folding
up the flaps. Its volume is $V(x) = x(12 - 2x)^2$, and physically
$0 < x < 6$.

\begin{enumerate}[label=(\alph*), itemsep=3.8cm, leftmargin=1.5cm, topsep=6pt]
  \item Solve $V'(x) = 0$. Report both roots, then say which one the domain
        rejects and why. \ansline
  \item Find the maximum volume, in cubic inches. \ansline
\end{enumerate}}

\problem{Let $f(x) = x^3 - 3x^2 - 9x + 5$ on the closed interval $[0, 4]$.

\begin{enumerate}[label=(\alph*), itemsep=3.4cm, leftmargin=1.5cm, topsep=6pt]
  \item Find every critical number of $f$, before the interval is imposed.
        \ansline
  \item Find the absolute maximum value and the absolute minimum value of $f$
        on $[0, 4]$, and say where each occurs. \ansline
\end{enumerate}}

\problem{Water is poured into an inverted right circular cone at a constant
$9$ cubic feet per minute. The cone's radius is always half its water depth, so
$r = h/2$ and $V = \dfrac{\pi h^3}{12}$.

\par\smallskip
Find $\dfrac{dh}{dt}$ at the instant the depth is $6$ feet. Give an exact
answer in terms of $\pi$.
\par\vspace*{9cm}\answerline{ft/min}}

\problem{\textbf{Synthesis.} A rectangle has its base on the $x$-axis and its
two upper corners on the parabola $y = 9 - x^2$. If the upper right corner is
$(x, 9 - x^2)$ with $0 \le x \le 3$, the rectangle's area is
$A(x) = 2x(9 - x^2)$.

\begin{enumerate}[label=(\alph*), itemsep=3.6cm, leftmargin=1.5cm, topsep=6pt]
  \item Solve $A'(x) = 0$ and report both roots. \ansline
  \item Justify that the root you kept gives the \emph{absolute} maximum area
        on $[0, 3]$ --- not merely a critical point --- and explain why a table
        of a few values of $A$ could never settle that question. \noansline
\end{enumerate}}
