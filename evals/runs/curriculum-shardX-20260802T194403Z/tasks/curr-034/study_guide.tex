\sstitleblock{Teen Numbers --- Fixing Swapped Tens and Ones}

\skillheading{Skill 1 --- Reading 1 ten and some ones}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
A teen number is \textbf{1 ten} and some \textbf{ones}.
Build it and write it in this order: \textbf{tens first, ones second}.\par
\smallskip
1 ten $=$ one full ten-frame $=$ one ten-stick. Then count the loose ones.
Write the tens digit in the \textbf{left} box and the ones digit in the
\textbf{right} box: \fbox{tens}\,\fbox{ones}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Build the number that is 1 ten and 4 ones.
\step{The first digit counts full ten-frames, so it is 1; the second digit counts
the loose dots, so it is 4.}
\step{Tens first, ones second: 1 ten is $10$, and $10$ with 4 more is $\ans{14}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Build the number that is 1 ten and 9 ones.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$}
\end{tryitbox}

\skillheading{Skill 2 --- Fixing a swapped number}

\begin{formulabox}
\textbf{How to check for a swap:}\\[4pt]
\textbf{1.} Count the \textbf{tens} in the picture. \quad
\textbf{2.} Count the \textbf{ones}. \quad
\textbf{3.} Ask: does the \emph{first} digit match the number of tens?\par
\smallskip
If the first digit is bigger than the number of full ten-frames, the digits were
\textbf{swapped}. Put them back the right way round: tens digit first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A child builds 1 ten and 2 ones, then writes a number whose
first digit is 2.
\step{The first digit should count the \emph{tens}, and there is only 1 ten ---
so the two digits were swapped.}
\step{Swap them back: 1 ten first, then 2 ones, giving $10$ with 2 more, which is
$\ans{12}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A child builds 1 ten and 5 ones but writes the 5 first.
Write the correct number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing a swapped pair}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
The same two digits swapped are \textbf{not} the same number.
To compare, look at the \textbf{tens} digit first --- more tens always wins.\par
\smallskip
$<$ means \emph{is less than}. \qquad $>$ means \emph{is greater than}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Compare the number made from 1 ten and 4 ones with the
number made from 4 tens and 1 one.
\step{Compare the tens first: the first number has 1 ten and the second has 4 tens.}
\step{4 tens is more than 1 ten, so the first number is the smaller one:
$\ans{<}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Compare the number made from 6 tens and 1 one with the number
made from 1 ten and 6 ones.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Teen numbers are \emph{said} ones-first (``six-teen'') but
\emph{written} tens-first. Hearing the ones word first is exactly why the digits
get swapped --- build the ten before you write anything down.
\end{watchoutbox}
