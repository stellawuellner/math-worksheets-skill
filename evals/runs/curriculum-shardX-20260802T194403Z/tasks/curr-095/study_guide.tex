\sstitleblock{Composite Rectilinear Figures: Area and Perimeter}

\noindent\small A \textbf{composite rectilinear figure} is a shape made only of rectangles, with
every corner a square corner. You never need a new formula for one --- you only need to decide
\emph{where to cut it}, and whether the question is asking about the \emph{inside} (area) or the
\emph{edge} (perimeter).\normalsize

\skillheading{1. Cut it into rectangles and add (composite-area-decompose)}

\begin{formulabox}
\textbf{Rule:} Area of a rectangle $= \text{length} \times \text{width}$, counted in
\emph{square} units.\\[3pt]
Draw one straight line across the shape so it becomes two rectangles. Find each area, then
\textbf{add}. The cutting line is never part of the answer --- it is only there to help you see
the rectangles.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An L-shape: the bottom part is $6$ across and $2$ tall; sitting on its left end
is a part $3$ across and $3$ tall (so the whole left side is $5$).
\step{The shape is not a rectangle, so cut straight across at the top of the bottom part --- now
there are two rectangles and I can use length $\times$ width on each.}
\step{Bottom rectangle: $6 \times 2 = 12$. \quad Top rectangle: $3 \times 3 = 9$.}
$\Rightarrow\quad 12 + 9 = \ans{21}$ square centimetres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} An L-shape has a bottom part $7$ across and $2$ tall, with a part $4$ across and
$4$ tall sitting on its left end. Find the area in square centimetres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$}
\end{tryitbox}

\skillheading{2. Fill in the corner, then take it away (composite-area-subtract)}

\begin{formulabox}
\textbf{Rule:} If the shape looks like a rectangle with one corner bitten out:\\[3pt]
$\text{area} = (\text{whole rectangle}) - (\text{missing corner})$\\[3pt]
Use this when the missing corner is easy to see. It gives the same answer as cutting and adding,
so pick whichever is less work --- and use the other one to check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle $8$ across and $5$ tall has a corner piece $2$ across and $3$ tall
removed.
\step{The bite is a whole rectangle by itself, so it is quicker to find the big area and subtract
it than to cut the leftover shape up.}
\step{Whole: $8 \times 5 = 40$. \quad Missing corner: $2 \times 3 = 6$.}
$\Rightarrow\quad 40 - 6 = \ans{34}$ square centimetres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle $9$ across and $6$ tall has a corner piece $3$ across and $2$ tall
removed. Find the area in square centimetres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{48}$}
\end{tryitbox}

\skillheading{3. Walk all the way around (composite-perimeter)}

\begin{formulabox}
\textbf{Rule:} Perimeter is the distance \emph{around the edge}, so \textbf{add every side}, in
plain units --- never multiply.\\[3pt]
Start at one corner, put your finger on each side in order, and tick it off as you add it. An
L-shape has \textbf{six} sides, not four: the two short sides inside the notch count too.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Going around an L-shape the sides measure $7$, $3$, $4$, $2$, $3$ and $5$.
\step{The question asks how far it is around the edge, so this is perimeter --- add the sides,
and count how many sides there should be first.}
\step{Six sides, so six numbers: $7 + 3 + 4 + 2 + 3 + 5$.}
$\Rightarrow\quad \ans{24}$ centimetres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Going around an L-shape the sides measure $8$, $2$, $5$, $4$, $3$ and $6$.
Find the perimeter in centimetres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{4. Sides that line up must match (missing-side)}

\begin{formulabox}
\textbf{Rule:} In a rectilinear figure, the short pieces along one direction add up to the long
side facing them:\\[3pt]
$\text{short} + \text{short} = \text{long}$, \quad so \quad
$\text{missing} = \text{long} - \text{known short}$.\\[3pt]
Horizontal pieces match the horizontal side; vertical pieces match the vertical side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} The bottom of a shape is $11$ long. Along the top, one piece is $4$ and the
other is unknown.
\step{The two top pieces together stretch exactly as far as the bottom, so this is a
subtraction, not a guess.}
\step{$4 + ? = 11$, so $? = 11 - 4$.}
$\Rightarrow\quad \ans{7}$ centimetres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The left side of a shape is $10$ tall. On the right, the upper piece is $6$ tall
and the lower piece is unknown. How tall is the lower piece, in centimetres?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} \emph{Area} is covered space and is counted in \textbf{square} units
(cm$^2$, ft$^2$); \emph{perimeter} is distance around and is counted in plain units (cm, ft).
Underline the word ``area'' or ``perimeter'' in the question before you pick up your pencil, and
never leave out the two hidden sides inside a notch.
\end{watchoutbox}
