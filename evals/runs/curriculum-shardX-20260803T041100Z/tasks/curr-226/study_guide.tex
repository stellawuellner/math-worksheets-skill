\sstitleblock{Intersections as Solutions to Systems of Two Linear Equations}

\skillheading{Skill 1 --- Test Whether a Point Solves a System}

\begin{formulabox}
\textbf{Definition:} $(x, y)$ is a \textbf{solution of the system} when it makes
\emph{every} equation true.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item One equation true, the other false $\Rightarrow$ the point is on one line only --- not a solution.
  \item Both true $\Rightarrow$ the point is on both graphs, so it is where they cross.
\end{itemize}
Substitute into each equation \textbf{separately}; never mix the two.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $(2, 5)$ the solution of $x + y = 7$ and $y - x = 3$?
\step{A solution has to satisfy both equations, so test the same pair twice rather than solving anything.}
\step{$x + y = 2 + 5 = 7$ $\checkmark$ \quad and \quad $y - x = 5 - 2 = 3$ $\checkmark$. Both hold, so the lines cross at $\ans{(2,\ 5)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Is $(1, 4)$ the solution of $y = 3x + 1$ and $x + y = 5$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(1,\ 4)}$ satisfies both}
\end{tryitbox}

\skillheading{Skill 2 --- Read an Intersection from a Graph or Table}

\begin{formulabox}
\textbf{On a graph:} find the one point the two lines share and read its coordinates.\\[2pt]
\textbf{In a table:} run down the two output columns and find the row where the
$y$-values are \textbf{equal}. That $x$ and shared $y$ are the intersection.\par
If the outputs are approaching each other, the crossing is nearby; if the gap is
constant, the lines are parallel and never meet.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Where do $y = x + 1$ and $y = -x + 5$ meet?
\step{The crossing is the input that makes both rules give the same output, so scan the table for equal outputs.}
\step{At $x = 2$ both columns read $3$; at $x = 1$ they read $2$ and $4$, and at $x = 3$ they read $4$ and $2$ --- the columns swap order, so they meet at $\ans{(2,\ 3)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Tables for $y = x - 1$ and $y = -2x + 8$ agree at one row. Which point?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\ 2)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Solve to Find the Intersection}

\begin{formulabox}
\textbf{Why algebra works:} at the crossing point both equations describe the same
$y$ for the same $x$, so you may set the two expressions for $y$ equal.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item Both solved for $y$: set them equal, solve for $x$, then back-substitute for $y$.
  \item Not solved for $y$: isolate one variable first, then substitute.
  \item Always write the answer as an ordered pair and check it in \emph{both} originals.
  \item A false statement such as $1 = -4$ means parallel lines: \textbf{no solution}.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $y = 4x - 3$ and $y = x + 3$.
\step{Both are solved for $y$, and at the intersection the two $y$-values are the same number, so the expressions must be equal there.}
\step{$4x - 3 = x + 3 \Rightarrow 3x = 6 \Rightarrow x = 2$, then $y = 2 + 3 = 5$ (and $4(2) - 3 = 5$ $\checkmark$), so $\ans{(2,\ 5)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $y = 2x - 4$ and $y = -x + 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\ 2)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A point that satisfies only one equation is \emph{not} a
solution --- it just lies on one of the two lines. And after solving for $x$, do not
stop: the solution of a system is a \textbf{point}, so report both coordinates.
\end{watchoutbox}
