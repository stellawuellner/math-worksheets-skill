\problem{Rewrite $y = x^2 + 6x + 5$ in vertex form.}

\textbf{Solution:} Half of $6$ is $3$, so the completing constant is $3^2 = 9$. Add it and
subtract it again so nothing changes.
\begin{align*}
  y &= (x^2 + 6x + 9) - 9 + 5 \\
    &= (x + 3)^2 - 4
\end{align*}
\ans{y = (x+3)^2 - 4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Rewrite $x^2 + y^2 - 4x - 12 = 0$ in standard form.}

\textbf{Solution:} Only $x$ needs completing: half of $-4$ is $-2$, giving $4$.
\begin{align*}
  (x^2 - 4x + 4) + y^2 &= 12 + 4 \\
  (x - 2)^2 + y^2 &= 16
\end{align*}
Centre $(2, 0)$, radius $4$. \ans{(x-2)^2 + y^2 = 16}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Rewrite $y = x^2 - 10x + 18$ in vertex form.}

\textbf{Solution:} Half of $-10$ is $-5$, so the completing constant is $25$.
\begin{align*}
  y &= (x^2 - 10x + 25) - 25 + 18 \\
    &= (x - 5)^2 - 7
\end{align*}
\ans{y = (x-5)^2 - 7}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Rewrite $x^2 + y^2 + 8x - 6y + 9 = 0$ in standard form.}

\textbf{Solution:} Group the $x$ terms and the $y$ terms, then complete each square. Half of
$8$ is $4$ (giving $16$); half of $-6$ is $-3$ (giving $9$).
\begin{align*}
  (x^2 + 8x + 16) + (y^2 - 6y + 9) &= -9 + 16 + 9 \\
  (x + 4)^2 + (y - 3)^2 &= 16
\end{align*}
Centre $(-4, 3)$, radius $4$. \ans{(x+4)^2 + (y-3)^2 = 16}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$x^2 + y^2 - 6x + 4y - 12 = 0$. (a) Standard form. (b) Radius.}

\textbf{Solution (a):} Half of $-6$ is $-3$ (giving $9$); half of $4$ is $2$ (giving $4$).
\begin{align*}
  (x^2 - 6x + 9) + (y^2 + 4y + 4) &= 12 + 9 + 4 \\
  (x - 3)^2 + (y + 2)^2 &= 25
\end{align*}
\ans{(x-3)^2 + (y+2)^2 = 25}

\textbf{Solution (b):} The right-hand side is $r^2$, so the radius is $\sqrt{25}$. \ans{5}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Rewrite $y = 2x^2 - 12x + 13$ in vertex form.}

\textbf{Solution:} Factor the $2$ out of the terms containing $x$ \emph{before} completing
the square --- the completing constant is computed inside the bracket and then carried back
out multiplied by $2$.
\begin{align*}
  y &= 2(x^2 - 6x) + 13 \\
    &= 2\bigl((x - 3)^2 - 9\bigr) + 13 \\
    &= 2(x - 3)^2 - 18 + 13
\end{align*}
\ans{y = 2(x-3)^2 - 5}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$y = x^2 + 5x + 3$. (a) Vertex form, exact fractions. (b) Vertex $x$-coordinate.}

\textbf{Solution (a):} Half of $5$ is $\tfrac{5}{2}$, so the completing constant is
$\tfrac{25}{4}$.
\begin{align*}
  y &= \left(x^2 + 5x + \frac{25}{4}\right) - \frac{25}{4} + 3 \\
    &= \left(x + \frac{5}{2}\right)^2 - \frac{25}{4} + \frac{12}{4}
\end{align*}
\ans{y = \left(x + \tfrac{5}{2}\right)^2 - \tfrac{13}{4}}

\textbf{Solution (b):} The bracket is zero when $x = -\tfrac{5}{2}$, which is also
$-b/(2a)$ with $a = 1$ and $b = 5$. \ans{x = -\tfrac{5}{2}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Rewrite $2x^2 + 2y^2 - 12x + 20y + 18 = 0$ in standard form.}

\textbf{Solution:} Divide the whole equation by $2$ first --- standard form needs the squared
terms with coefficient $1$.
\begin{align*}
  x^2 + y^2 - 6x + 10y + 9 &= 0 \\
  (x^2 - 6x + 9) + (y^2 + 10y + 25) &= -9 + 9 + 25 \\
  (x - 3)^2 + (y + 5)^2 &= 25
\end{align*}
Centre $(3, -5)$, radius $5$. \ans{(x-3)^2 + (y+5)^2 = 25}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Circle with centre $(-4, 7)$, radius $6$: standard form, then general form.}

\textbf{Solution:} Standard form reads straight off: $h = -4$, $k = 7$, $r^2 = 36$, so
$(x + 4)^2 + (y - 7)^2 = 36$. Expanding,
\begin{align*}
  x^2 + 8x + 16 + y^2 - 14y + 49 - 36 &= 0 \\
  x^2 + y^2 + 8x - 14y + 29 &= 0
\end{align*}
\ans{x^2 + y^2 + 8x - 14y + 29 = 0}

\problem{Parabola with vertex $(2,-9)$ through $(5,0)$. (a) Find $a$. (b) Expand to $y = x^2 + bx + c$.}

\textbf{Solution (a):} Substitute $x = 5$, $y = 0$ into $y = a(x-2)^2 - 9$:
\begin{align*}
  0 &= a(5 - 2)^2 - 9 \\
  0 &= 9a - 9
\end{align*}
\ans{a = 1}

\textbf{Solution (b):} With $a = 1$ the equation is $y = (x-2)^2 - 9$, so
\begin{align*}
  y &= x^2 - 4x + 4 - 9
\end{align*}
\ans{y = x^2 - 4x - 5}
A check: the roots $x = -1$ and $x = 5$ are symmetric about $x = 2$, the vertex.

\problem{$y = 3x^2 + 12x + 7$. (a) Vertex form. (b) Vertex $y$-coordinate.}

\textbf{Solution (a):} Factor the $3$ out of the $x$ terms first.
\begin{align*}
  y &= 3(x^2 + 4x) + 7 \\
    &= 3\bigl((x + 2)^2 - 4\bigr) + 7 \\
    &= 3(x + 2)^2 - 12 + 7
\end{align*}
\ans{y = 3(x+2)^2 - 5}

\textbf{Solution (b):} The vertex sits at $x = -2$, where the bracket vanishes:
$y = 3(-2)^2 + 12(-2) + 7 = 12 - 24 + 7$. \ans{-5}

\problem{$x^2 + y^2 - 2x + 8y + 8 = 0$. (a) Standard form. (b) Radius. (c) Why it cannot be a parabola.}

\textbf{Solution (a):} Half of $-2$ is $-1$ (giving $1$); half of $8$ is $4$ (giving $16$).
\begin{align*}
  (x^2 - 2x + 1) + (y^2 + 8y + 16) &= -8 + 1 + 16 \\
  (x - 1)^2 + (y + 4)^2 &= 9
\end{align*}
\ans{(x-1)^2 + (y+4)^2 = 9}

\textbf{Solution (b):} The radius is $\sqrt{9}$. \ans{3}

\vspace{0.15cm}
\textbf{Solution (c) --- instructor-judged.} A parabola's equation carries a squared term in
one variable only; this one has $x^2$ \emph{and} $y^2$, with the same coefficient, so no
rearrangement could ever produce a parabola --- the equal coefficients also say it is a
circle rather than an ellipse. Full credit refers to both squared terms being present and
matching. Answering that the picture is round, with no reference to the equation, does not
use the standard-form reasoning the question asks for.
