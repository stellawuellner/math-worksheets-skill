\sstitleblock{Composite Solids, Surface Area, and Density Design}

\skillheading{1. Volume of a composite solid (composite-volume)}

\begin{formulabox}
\textbf{Volumes add.} Cut the solid at the joins, find each piece, add.
$V_{\text{cyl}} = \pi r^2 h$, \quad $V_{\text{cone}} = \tfrac{1}{3}\pi r^2 h$, \quad
$V_{\text{sphere}} = \tfrac{4}{3}\pi r^3$ (hemisphere: $\tfrac{2}{3}\pi r^3$), \quad
$V_{\text{prism}} = Bh$.\\[3pt]
A \emph{hole} is a piece you \textbf{subtract}: $V = V_{\text{solid}} - V_{\text{hole}}$.
Keep $\pi$ symbolic until the last line so nothing is lost to rounding.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tank is a cylinder ($r = 2$, $h = 6$ cm) with a hemisphere lid of
the same radius. Find its volume.
\step{Two solids joined at a shared circle, and volume asks how much space is inside, so the
two volumes simply add --- the shared circle has no volume.}
\step{$\pi(2)^2(6) = 24\pi$ and $\tfrac{2}{3}\pi(2)^3 = \tfrac{16}{3}\pi$, so
$V = \tfrac{88}{3}\pi = \ans{92.15\ \text{cm}^3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Cylinder $r = 3$, $h = 5$ cm with a hemisphere lid. Find the volume.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{197.92\ \text{cm}^3}$}
\end{tryitbox}

\skillheading{2. Surface area of a composite solid}

\begin{formulabox}
\textbf{Surfaces do NOT add.} Count only the faces you could paint: at every join, both
pieces lose the circle (or face) they share.\\[3pt]
Cylinder side $= 2\pi r h$, \quad cone slant side $= \pi r \ell$, \quad circle $= \pi r^2$,
\quad sphere $= 4\pi r^2$ (hemisphere curved part: $2\pi r^2$).\\[3pt]
A drilled hole \emph{removes} two circles and \emph{adds} the wall $2\pi r h$ inside.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Same tank: cylinder $r = 2$, $h = 6$ cm, hemisphere lid. Find the
outside surface area.
\step{Walk the outside of the solid: the flat base, the cylinder wall, the dome --- the
circle where the dome meets the cylinder is inside, so it is not counted.}
\step{$\pi(2)^2 + 2\pi(2)(6) + 2\pi(2)^2 = 4\pi + 24\pi + 8\pi = 36\pi
= \ans{113.10\ \text{cm}^2}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Cylinder $r = 3$, $h = 5$ cm with a hemisphere lid: outside surface
area (base $+$ wall $+$ dome).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{179.07\ \text{cm}^2}$}
\end{tryitbox}

\skillheading{3. Density: turning volume into mass (density-mass)}

\begin{formulabox}
\[ \text{density} = \frac{\text{mass}}{\text{volume}} \qquad\Longrightarrow\qquad
   m = \rho V \qquad\text{and}\qquad V = \frac{m}{\rho} \]
Units decide the arithmetic: a density in g/cm$^3$ needs a volume in cm$^3$; a density in
kg/m$^3$ needs m$^3$. Find the volume first, then multiply --- never multiply density by a
length.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A solid aluminium cylinder, $r = 2$ cm, $h = 8$ cm, density
$2.7$ g/cm$^3$. Find its mass.
\step{Mass is asked and density is given, so the missing piece is volume --- compute $V$,
then multiply by the density.}
\step{$V = \pi(2)^2(8) = 32\pi$ cm$^3$, so $m = 2.7 \times 32\pi = 86.4\pi
= \ans{271.43\ \text{g}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A solid copper sphere, $r = 2$ cm, density $8.96$ g/cm$^3$. Find its
mass.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{300.25\ \text{g}}$}
\end{tryitbox}

\skillheading{4. Design backwards: solve for a dimension}

\begin{formulabox}
A design problem fixes the \emph{answer} (a volume, a mass) and asks for a \emph{dimension}.
Write the formula as an equation, then solve:
\[ \pi r^2 h = V \Rightarrow h = \frac{V}{\pi r^2}, \qquad
   \tfrac{4}{3}\pi r^3 = V \Rightarrow r = \sqrt[3]{\frac{3V}{4\pi}}, \qquad
   V = \frac{m}{\rho}\ \text{first if a mass is given.} \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A cylindrical can of radius $5$ cm must hold $400$ cm$^3$. How tall
must it be?
\step{The volume is given and a dimension is unknown, so set the volume formula equal to the
target and solve for $h$ instead of computing forward.}
\step{$\pi(5)^2 h = 400$, so $h = \dfrac{400}{25\pi} = \ans{5.09\ \text{cm}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A cylindrical can of radius $4$ cm must hold $300$ cm$^3$. Find its
height.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.97\ \text{cm}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not add whole surface areas at a join --- that double-counts the
hidden circle. And if a design fixes the volume, changing the radius cannot change the mass:
$m = \rho V$, so a wider solid is simply a shorter one.
\end{watchoutbox}
