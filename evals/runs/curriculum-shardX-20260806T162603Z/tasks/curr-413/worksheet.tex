\wstitleblock{Fitting and Interpreting Exponential Models}{Precalculus}{}

\begin{center}\small\textit{Fitting exponential models to data, reading growth and decay rates, solving for time with logarithms, and rewriting a model with a different base}\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Every model below states its variable and its units. Answer in the unit the question names.
  \item Round only at the last step, and only to the precision the question asks for.
  \item Write the equation you solved, not just the number you got.
\end{itemize}
\vspace{0.15cm}

\problem{A savings account is modelled by $A(t) = 1200(1.045)^t$, where $A$ is the balance in dollars and $t$ is the time in years.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] The balance at $t = 0$ years, in dollars. \ansline
  \vspace*{1.6cm}
  \item[(b)] The annual percent rate of increase. \ansline
  \vspace*{1.6cm}
\end{itemize}}

\problem{A lab culture holds $200$ cells when the count begins and $3200$ cells $4$ hours later. The technician models the culture as $N(t) = a\,b^{\,t}$ cells, with $t$ in hours.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find the hourly growth factor $b$. \ansline
  \vspace*{2.4cm}
  \item[(b)] Use the model to predict the cell count at $t = 7$ hours. \ansline
  \vspace*{2.4cm}
\end{itemize}}

\problem{A dose of medication in the bloodstream is modelled by $M(t) = 250(0.86)^t$, where $M$ is in milligrams and $t$ is the time in hours after the dose.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] What percent of the drug present is lost each hour? \ansline
  \vspace*{2.2cm}
  \item[(b)] How much remains after $6$ hours, to the nearest $0.1$ mg? \answerline{mg}
  \vspace*{2.6cm}
\end{itemize}}

\problem{A town's population is modelled by $P(t) = 18000(1.023)^t$ people, where $t$ is the number of years after the last census. Find, to the nearest hundredth of a year, how long it takes the population to reach $25000$ people.
\par\vspace*{5.5cm}\answerline{years}}

\problem{A car is worth $24000$ dollars when new and $15360$ dollars two years later. Its value is modelled by $V(t) = a\,b^{\,t}$ dollars, with $t$ in years.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find the yearly decay factor $b$. \ansline
  \vspace*{2.4cm}
  \item[(b)] Predict the value after $5$ years, to the nearest dollar. \ansline
  \vspace*{2.4cm}
\end{itemize}}

\problem{An investment is modelled by $A(t) = 5000(1.08)^t$ dollars, with $t$ in years. The same investment can be written in continuous form as $A(t) = 5000e^{kt}$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find $k$, to four decimal places. \ansline
  \vspace*{2.4cm}
  \item[(b)] Use the continuous form to find the balance after $12$ years, to the nearest dollar. \ansline
  \vspace*{2.4cm}
\end{itemize}}

\problem{A radioactive tracer decays according to $N(t) = 40(0.5)^{\,t/6}$, where $N$ is in milligrams and $t$ is the time in hours. How long until $7$ mg of the tracer remains? Give the time to the nearest $0.1$ hour.
\par\vspace*{6cm}\answerline{hours}}

\problem{A city's annual water use is modelled by $W(t) = c\,(0.91)^t$ million litres, where $c$ is the use in the survey year and $t$ is the number of years after it.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] By what percent does annual water use fall each year? \ansline
  \vspace*{1.7cm}
  \item[(b)] A councillor says use will have fallen by $90$ percent after $10$ years. Find the actual percent decrease over $10$ years, to the nearest $0.1$ percent. \ansline
  \vspace*{2.2cm}
  \item[(c)] Explain in one or two sentences why the councillor's reasoning fails. \ansline
  \vspace*{2.2cm}
\end{itemize}}

\problem{Two analysts model the same fund. Analyst A writes $F(t) = 300(1.06)^{2t}$ dollars; Analyst B writes $F(t) = 300(1.1236)^t$ dollars. In both, $t$ is the time in years.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Rewrite Analyst A's model with a single yearly growth factor. \ansline
  \vspace*{2.0cm}
  \item[(b)] State Analyst B's annual percent growth rate. \ansline
  \vspace*{1.7cm}
  \item[(c)] Justify in one or two sentences why the two models agree for every $t$. \ansline
  \vspace*{2.2cm}
\end{itemize}}

\problem{A cup of coffee cools according to $T(t) = 21 + 62(0.94)^t$, where $T$ is the temperature in degrees Celsius and $t$ is the time in minutes after pouring. The room stays at $21$ degrees Celsius.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find the temperature at the moment the coffee is poured. \answerline{degrees}
  \vspace*{2.2cm}
  \item[(b)] Find how long the coffee takes to cool to $45$ degrees Celsius, to the nearest $0.1$ minute. \answerline{minutes}
  \vspace*{4.4cm}
\end{itemize}}
