\noindent\small Every solution names the inner function $u$ (or the rule being combined) before it
differentiates. On the AP exam the chain-rule factor $u'$ is the mark most often lost, so it is
written explicitly in each step.\normalsize

\problem{Differentiate $y = \arctan(3x)$.}
\textbf{Solution:} $u = 3x$, so $u' = 3$, and
$\dfrac{d}{dx}\arctan(u) = \dfrac{u'}{1 + u^{2}}$:
\[ y' = \frac{3}{1 + (3x)^{2}} \]
\ans{y' = \dfrac{3}{\,1 + 9x^{2}}}

\problem{Differentiate $y = \arcsin(x^{2})$.}
\textbf{Solution:} $u = x^{2}$, so $u' = 2x$, and
$\dfrac{d}{dx}\arcsin(u) = \dfrac{u'}{\sqrt{1 - u^{2}}}$:
\[ y' = \frac{2x}{\sqrt{1 - (x^{2})^{2}}} \]
\ans{y' = \dfrac{2x}{\sqrt{1 - x^{4}}}}
\emph{Common error:} writing $\sqrt{1 - x^{2}}$ in the denominator --- the square is applied to
$u$, not to $x$.

\problem{$f(x) = x^{3} + 2x + 1$; find $b$ with $f(b) = 4$, then $\left(f^{-1}\right)'(4)$.}
\textbf{(a) Solution:} Solve $x^{3} + 2x + 1 = 4$, i.e.\ $x^{3} + 2x - 3 = 0$. Factoring gives
$(x - 1)(x^{2} + x + 3) = 0$, and the quadratic has no real root, so the only real solution is
\ans{b = 1}

\textbf{(b) Solution:} $f'(x) = 3x^{2} + 2$, so $f'(1) = 5$. The inverse-function rule
$\left(f^{-1}\right)'(a) = \dfrac{1}{f'(b)}$ with $b = 1$ gives
\ans{\left(f^{-1}\right)'(4) = \tfrac{1}{5}}
\emph{Note:} the reciprocal is of $f'$ evaluated at the \emph{input} $b$, never at the output $a$;
$1/f'(4) = 1/50$ is the classic wrong answer.

\problem{Differentiate $y = x\arctan(x)$.}
\textbf{Solution:} Product rule with $f = x$, $g = \arctan x$:
\[ y' = (1)\arctan(x) + x \cdot \frac{1}{1 + x^{2}} \]
\ans{y' = \arctan(x) + \dfrac{x}{1 + x^{2}}}

\problem{Differentiate $y = \arccos(2x)$.}
\textbf{Solution:} $u = 2x$, $u' = 2$, and
$\dfrac{d}{dx}\arccos(u) = \dfrac{-u'}{\sqrt{1 - u^{2}}}$ --- the $\arccos$ derivative is the
negative of the $\arcsin$ one:
\[ y' = \frac{-2}{\sqrt{1 - (2x)^{2}}} \]
\ans{y' = \dfrac{-2}{\sqrt{1 - 4x^{2}}}}

\problem{Second derivative of $y = \arctan(x)$.}
\textbf{Solution:} $y' = \dfrac{1}{1 + x^{2}} = (1 + x^{2})^{-1}$. Differentiate again with the
chain rule (inner $1 + x^{2}$, derivative $2x$):
\[ y'' = -1(1 + x^{2})^{-2}(2x) \]
\ans{y'' = \dfrac{-2x}{(1 + x^{2})^{2}}}
\emph{Reading:} $y''$ is positive for $x < 0$ and negative for $x > 0$, matching the S-shape of
$\arctan$ with its inflection at the origin.

\problem{$g(x) = 2x + \sin(x)$; compute $\left(g^{-1}\right)'(0)$.}
\textbf{Solution:} First the input: $g(0) = 0 + \sin 0 = 0$, so $b = 0$. Then
$g'(x) = 2 + \cos(x)$, giving $g'(0) = 2 + 1 = 3$, and
\ans{\left(g^{-1}\right)'(0) = \tfrac{1}{3}}
Since $g'(x) = 2 + \cos x \geq 1 > 0$, $g$ is strictly increasing everywhere, which is what
guarantees the inverse exists.

\problem{Differentiate $y = e^{x}\arctan(x)$.}
\textbf{Solution:} Product rule; the derivative of $e^{x}$ is itself.
\[ y' = e^{x}\arctan(x) + e^{x}\cdot\frac{1}{1 + x^{2}} \]
\ans{y' = e^{x}\!\left(\arctan(x) + \dfrac{1}{1 + x^{2}}\right)}

\problem{Differentiate $y = \arctan(\sqrt{x})$.}
\textbf{Solution:} $u = \sqrt{x} = x^{1/2}$, so $u' = \dfrac{1}{2\sqrt{x}}$ and
$u^{2} = x$:
\[ y' = \frac{1}{1 + x}\cdot\frac{1}{2\sqrt{x}} \]
\ans{y' = \dfrac{1}{\,2\sqrt{x}\,(1 + x)}}
\emph{Why the denominator collapses:} $(\sqrt{x})^{2} = x$ exactly, so no radical survives in the
first factor.

\problem{Challenge: second derivative of $y = \arctan(2x)$.}
\textbf{Solution:} First derivative by the chain rule ($u = 2x$, $u' = 2$):
$y' = \dfrac{2}{1 + 4x^{2}} = 2(1 + 4x^{2})^{-1}$. Differentiate again, chain rule on
$1 + 4x^{2}$ (derivative $8x$):
\begin{align*}
y'' &= 2 \cdot (-1)(1 + 4x^{2})^{-2}(8x)
\end{align*}
\ans{y'' = \dfrac{\,-16x}{(1 + 4x^{2})^{2}}}
\emph{Check the structure:} $y''$ vanishes only at $x = 0$, the inflection point of
$\arctan(2x)$, and the outer constant $2$ times the inner $8x$ is where the $16$ comes from.
