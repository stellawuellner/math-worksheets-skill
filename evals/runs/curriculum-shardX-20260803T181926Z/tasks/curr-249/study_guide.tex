\sstitleblock{Scatter Plots, Association, and Causal Claims}

\skillheading{Skill 1 --- Reading a scatter plot}

\begin{formulabox}
\textbf{How to read a scatter plot:} each dot is one case, plotted as
(horizontal quantity, vertical quantity).
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Positive association:} as the horizontal value rises, the
        vertical value tends to rise. \textbf{Negative:} it tends to fall.
  \item To say how big the change is, subtract the two \textbf{vertical}
        values --- change $=$ end value $-$ start value.
  \item Subtracting the horizontal values answers a different question. Ask
        yourself which axis the question is about \emph{before} you subtract.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A plot has the points $(2, 5)$, $(4, 8)$, $(6, 10)$ and
$(8, 15)$. Describe the association and give the change in the vertical
quantity across the plot.

\step{The question is about the vertical quantity, so read the vertical values
at the two ends and subtract them --- the horizontal values only say where to
look.}

\step{At the left end the height is $5$; at the right end it is $15$. Change
$= 15 - 5$.}

The vertical quantity rises by $\ans{10}$, so the association is positive.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A plot has the points $(1, 20)$, $(3, 16)$, $(5, 11)$ and
$(7, 9)$. What is the change in the vertical quantity from the left end to the
right end?

\rotatebox{180}{\footnotesize check: $\ans{-11}$}
\end{tryitbox}

\skillheading{Skill 2 --- Does the gap prove cause?}

\begin{formulabox}
\textbf{A gap between two groups is a starting point, not a verdict.} Compute
each group's mean from \emph{that group's} data only, then ask three questions.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Lurking variable:} what else differs between these groups?
  \item \textbf{Reverse cause:} could the outcome have produced the grouping?
  \item \textbf{Who decided?} If people \textbf{chose} their own group, the study
        is observational and supports association only. Only \textbf{random
        assignment} makes the groups alike beforehand, and that is what lets a
        later difference be credited to the treatment.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Students who joined a study club scored $90$, $72$, $84$;
students who did not scored $70$, $68$, $75$. Find each group's mean, then judge
the claim ``the club raises scores''.

\step{Compare like with like: average each group using only its own scores,
because a pooled average would answer a different question.}

\step{Add each group's three scores and divide by three.}

Club mean $\ans{82}$; no-club mean $\ans{71}$. The club group is ahead --- but
the students joined the club themselves, so the gap is association, not proof of
cause.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad One group scored $88$, $92$, $84$; the other scored $80$,
$74$, $77$. Find both means.

\rotatebox{180}{\footnotesize check: $\ans{88}$ and $\ans{77}$}
\end{tryitbox}

\skillheading{Skill 3 --- Fixing a cause headline}

\begin{formulabox}
\textbf{Swap the causal verb for a "goes with" verb, and say who the data
covers.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Out: \emph{causes}, \emph{makes}, \emph{leads to}, \emph{raises}.
  \item In: \emph{is associated with}, \emph{tends to go with}, \emph{students
        who \dots\ also \dots}
  \item Scope it to the data actually collected ("among these six students"),
        never to everyone.
  \item Then say what evidence \emph{would} settle it: assign the treatment at
        random and compare.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A plot of screen hours against sleep hours holds
$(0, 9)$, $(2, 8)$, $(4, 7)$ and $(6, 6)$. Fix the headline ``screens steal
sleep''.

\step{First measure what the plot really reports --- the change in the vertical
quantity, sleep, across the range of screen time.}

\step{From $0$ hours of screen time to $6$ hours, sleep goes from $9$ down to
$6$: change $= 6 - 9$.}

Sleep changes by $\ans{-3}$ hours, so the honest headline is ``in this group,
more screen time went with less sleep''.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A plot holds $(0, 60)$, $(1, 54)$, $(2, 47)$ and
$(3, 39)$. What is the change in the vertical quantity across the plot?

\rotatebox{180}{\footnotesize check: $\ans{-21}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A strong pattern is not strong evidence of cause. The
tighter the dots sit around a line, the more tempting the cause story gets ---
but tightness measures how well one quantity \emph{predicts} the other, and
prediction is not causation.
\end{watchoutbox}
