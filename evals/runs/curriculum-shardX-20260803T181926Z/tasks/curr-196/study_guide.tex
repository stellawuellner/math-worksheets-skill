\sstitleblock{Statistical Questions, Centre and Spread}

\skillheading{Skill 1 --- Is it a statistical question?}

\begin{formulabox}
\textbf{A question is statistical when you expect the answers to vary.} It is about the
\emph{answers}, not the topic.\par
\smallskip
Concrete test: collect the answers and find the range. Range $= 0$ means every answer was
the same, so nothing varied and the question was not statistical. A question about
\emph{one} person or thing almost never varies; a question about a \emph{group} usually
does.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is ``How many minutes did the students in my group read last
night?'' statistical? Answers collected: $10, 25, 25, 40$.
\step{Whether a question is statistical is decided by variability in its answers, so the
thing to compute is the range of what was collected.}
\step{Largest $40$ minus smallest $10$.}

$\Rightarrow\quad$ range $= \ans{30}$, not zero, so \textbf{yes}, it is statistical
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is ``How many minutes did Mr Diaz read last night?'' statistical?
Answers collected from four people: $25, 25, 25, 25$.\\[2pt]
\rotatebox{180}{\footnotesize check: range $= \ans{0}$, so no --- one person, one answer}
\end{tryitbox}

\skillheading{Skill 2 --- Centre: mean and median}

\begin{formulabox}
\textbf{Mean} $=$ (add all the values) $\div$ (how many values). \quad
\textbf{Median} $=$ the middle value once the data is \emph{sorted}; with an even number of
values, halfway between the middle two.\par
\smallskip
Sort first, always. And when one value is far from the rest, the mean chases it while the
median barely moves --- that is why both are worth quoting.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the mean and the median of $4, 5, 7, 8, 16$.
\step{Both are centres, but they answer different questions, so compute each with its own
rule rather than expecting them to agree.}
\step{Mean: $(4+5+7+8+16) \div 5 = 40 \div 5$. \quad Median: already sorted, five values,
so the middle one is the third.}

$\Rightarrow\quad$ mean $= \ans{8}$, \quad median $= \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the mean and the median of $3, 6, 9, 10, 22$.\\[2pt]
\rotatebox{180}{\footnotesize check: mean $= \ans{10}$, median $= \ans{9}$}
\end{tryitbox}

\skillheading{Skill 3 --- Spread: range and IQR}

\begin{formulabox}
\textbf{Range} $=$ largest $-$ smallest. It uses only the two extremes, so one unusual
value controls it.\par
\smallskip
\textbf{Interquartile range} describes the middle half instead. Sort the data, split it
into a lower half and an upper half, take the median of each: those are $Q_1$ and $Q_3$,
and $\text{IQR} = Q_3 - Q_1$. With an odd number of values, leave the overall median out of
both halves.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Range and IQR of $2, 4, 4, 5, 7, 8, 9, 9, 10, 15$.
\begin{center}
\begin{tikzpicture}[x=0.52cm, y=0.30cm]
  \draw[thick] (1,0) -- (16,0);
  \foreach \t in {1,...,16} {\draw (\t,0) -- (\t,-0.5);}
  \node[below, font=\tiny] at (2,-0.6) {2};
  \node[below, font=\tiny] at (6,-0.6) {6};
  \node[below, font=\tiny] at (10,-0.6) {10};
  \node[below, font=\tiny] at (14,-0.6) {14};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (2,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (4,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (4,1.8) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (5,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (7,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (8,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (9,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (9,1.8) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (10,0.9) {};
  \node[circle, fill=blue!70, inner sep=1.3pt] at (15,0.9) {};
\end{tikzpicture}
\end{center}
\step{Two spreads are wanted and they measure different things --- the range reads the two
ends of the plot, the IQR reads only the middle half --- so take them separately.}
\step{Range: $15 - 2$. \quad Halves of five: $2, 4, 4, 5, 7$ gives $Q_1 = 4$, and
$8, 9, 9, 10, 15$ gives $Q_3 = 9$, so IQR $= 9 - 4$.}

$\Rightarrow\quad$ range $= \ans{13}$, \quad IQR $= \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Range and IQR of $5, 6, 6, 7, 9, 10, 12, 13, 14, 20$.\\[2pt]
\rotatebox{180}{\footnotesize check: range $= \ans{15}$, IQR $= \ans{7}$}
\end{tryitbox}

\skillheading{Skill 4 --- Chance from a data set}

\begin{formulabox}
\textbf{Probability} $=$ (how many values fit the description) $\div$ (how many values
there are altogether), written as a fraction in lowest terms.\par
\smallskip
Read the words exactly: ``more than $9$'' excludes a value of $9$, ``at least $9$'' includes
it. Count the dots on the plot rather than the labels on the axis.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One of the ten values above is chosen at random. Find
$P(\text{value is at least } 9)$.
\step{Every value is equally likely to be chosen, so the probability is just a count of the
values that fit, over the ten values there are.}
\step{At least $9$ means $9, 9, 10, 15$ --- four of the ten --- and $4/10$ reduces.}

$\Rightarrow\quad P = \ans{\tfrac{2}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From the same ten values, find $P(\text{value is at most } 4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{10}}$}
\end{tryitbox}
