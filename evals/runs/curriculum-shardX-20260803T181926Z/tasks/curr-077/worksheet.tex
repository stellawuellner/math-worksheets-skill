\wstitleblock{Fact Strategies: Turnarounds and Break-Aparts}{Grade 3 Math}{}

\begin{center}
\small\itshape Turnaround facts, breaking a factor apart, and finding a missing factor
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item A \textit{turnaround fact} swaps the two factors. The product stays the same.
  \item \textit{Breaking apart} splits one factor into two friendlier pieces, multiplies each piece, and then puts the two answers together.
  \item Show the fact you used, not just the answer.
\end{itemize}

\vspace{0.2cm}

\problem[3.5cm]{Ana already knows that $4 \times 3 = 12$.
Use the turnaround fact to find $3 \times 4$.}

\problem[3.5cm]{Ben already knows that $8 \times 2 = 16$.
Use the turnaround fact to find $2 \times 8$.}

\problem[4.5cm]{Break apart to multiply. Finish the strategy, then write the product.
\[ 6 \times 7 \;=\; 6 \times (5 + 2) \;=\; ? \]}

\problem[4.5cm]{Break apart to multiply. Finish the strategy, then write the product.
\[ 7 \times 8 \;=\; (7 \times 4) + (7 \times 4) \;=\; ? \]}

\problem[4cm]{Find both products. What do you notice about them?
\[ 9 \times 5 \qquad\text{and}\qquad 5 \times 9 \]}

\problem[3.5cm]{Use a missing-factor fact to divide. Think: $6 \times \underline{\hspace{0.9cm}} = 24$.
\[ 24 \div 6 \;=\; ? \]}

\problem[4.5cm]{Nines are easier from tens. Finish the strategy, then write the product.
\[ 9 \times 6 \;=\; (10 \times 6) - (1 \times 6) \;=\; ? \]}

\problem[5cm]{Break $8 \times 9$ apart into two easier facts of your own choosing.
Write the two facts you used, then write the product.}

\problem[5cm]{Malik says that $7 \times 9$ gives the same answer as $7 \times 10$ with one group of $7$ taken away.
Use Malik's strategy to find $7 \times 9$.}

\problem[5cm]{There are $35$ stickers shared equally among $5$ friends.
Write the missing-factor fact that matches this story, then find how many stickers each friend gets.}
