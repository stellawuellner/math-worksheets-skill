\sstitleblock{Comparing Unlike Fractions: Benchmarks and Common Denominators}

\noindent\small Two fractions can only be compared when they count the
\emph{same size pieces of the same whole}. That is the one idea behind all four
skills below: a benchmark compares each fraction to a piece you already know,
and a common denominator makes the pieces literally the same size.\normalsize

\skillheading{1. Compare with a benchmark (benchmark-comparison)}
\begin{formulabox}
\textbf{Rule:} A fraction is \textbf{less than} $\frac{1}{2}$ when the numerator
is less than half the denominator, \textbf{equal} when it is exactly half, and
\textbf{greater} when it is more.\\[3pt]
So: half of the denominator is the number to beat. If one fraction lands below
$\frac{1}{2}$ and the other above it, you are done --- no common denominator
needed.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Compare $\frac{3}{7}$ and $\frac{5}{8}$.
\step{The two fractions look close, so test each one against $\frac{1}{2}$
before doing any arithmetic with denominators.}
\step{Half of $7$ is $3.5$, and $3 < 3.5$, so $\frac{3}{7} < \frac{1}{2}$.
Half of $8$ is $4$, and $5 > 4$, so $\frac{5}{8} > \frac{1}{2}$.}
Below the benchmark is smaller than above it:
$\ans{\frac{3}{7} < \frac{5}{8}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare $\frac{2}{9}$ and $\frac{4}{7}$ using the
benchmark $\frac{1}{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{2}{9} < \frac{4}{7}}$}
\end{tryitbox}

\skillheading{2. Compare with a common denominator (common-denominator)}
\begin{formulabox}
\textbf{Rule:} Rewrite both fractions with the same denominator --- any common
multiple works, the least one keeps the numbers small --- then compare the
numerators.\\[3pt]
$\dfrac{a}{b} = \dfrac{a \times k}{b \times k}$\qquad Same denominator means
same size pieces, so more pieces is more.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Compare $\frac{3}{4}$ and $\frac{7}{10}$.
\step{Both fractions are above $\frac{1}{2}$, so the benchmark cannot separate
them --- rebuild them on a common denominator.}
\step{$20$ is a multiple of both $4$ and $10$:
$\frac{3}{4} = \frac{15}{20}$ and $\frac{7}{10} = \frac{14}{20}$, and
$15 > 14$.}
$\ans{\frac{3}{4} > \frac{7}{10}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare $\frac{5}{6}$ and $\frac{8}{9}$ on a common
denominator.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{6} < \frac{8}{9}}$}
\end{tryitbox}

\skillheading{3. Build an equivalent fraction (equivalent-fractions)}
\begin{formulabox}
\textbf{Rule:} Multiply the numerator \emph{and} the denominator by the same
number. That is multiplying by $\frac{k}{k} = 1$, so the amount does not
change --- only the size and count of the pieces do.\\[3pt]
To reach a target denominator, ask ``\emph{times what?}'' and use that same
factor on top.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Write $\frac{2}{3}$ with denominator $12$.
\step{The target denominator is given, so find the factor that turns $3$ into
$12$ and apply it to both parts.}
\step{$3 \times 4 = 12$, so multiply the numerator by $4$ as well:
$\frac{2 \times 4}{3 \times 4}$.}
$\ans{\frac{8}{12}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Write $\frac{3}{5}$ with denominator $30$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{18}{30}}$}
\end{tryitbox}

\skillheading{4. Order three or more fractions (ordering-sets)}
\begin{formulabox}
\textbf{Rule:} Put \emph{every} fraction on one common denominator, order the
numerators, then write the answer back in the original fractions --- the
question asked about those, not about the rewritten ones.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Order $\frac{2}{5}$, $\frac{1}{3}$, $\frac{1}{2}$ from
least to greatest.
\step{Three fractions cannot be compared in pairs reliably, so convert all
three to one denominator at once.}
\step{$30$ works for $5$, $3$, and $2$:
$\frac{2}{5} = \frac{12}{30}$, $\frac{1}{3} = \frac{10}{30}$,
$\frac{1}{2} = \frac{15}{30}$; the numerators order $10 < 12 < 15$.}
$\ans{\frac{1}{3} < \frac{2}{5} < \frac{1}{2}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Order $\frac{3}{4}$, $\frac{5}{6}$, $\frac{2}{3}$ from
greatest to least.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{6} > \frac{3}{4} > \frac{2}{3}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A bigger denominator does \emph{not} mean a bigger
fraction --- it means smaller pieces. With the same numerator,
$\frac{1}{10} < \frac{1}{8}$. And never compare by numerators alone:
$\frac{5}{12}$ is less than $\frac{2}{3}$ even though $5 > 2$.
\end{watchoutbox}
