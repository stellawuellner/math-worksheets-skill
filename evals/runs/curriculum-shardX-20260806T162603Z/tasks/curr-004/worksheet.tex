\wstitleblock{Find It and Fix It: Counting Detective}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:} Be a counting detective!
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Whisper-count each row to yourself, pointing at each number.
  \item Fill every empty box with the number that belongs there.
  \item When a count has a mistake in it, find it and fix it.
\end{itemize}

\problem[2.6cm]{Count forward along the track. One box is empty --- write the
number that belongs in it.\par\vspace{0.3cm}
\begin{center}{\renewcommand{\arraystretch}{1.8}%
\begin{tabular}{|c|c|c|c|c|c|}\hline
{\Large 8} & {\Large 9} & {\Large 10} & \ansblank & {\Large 12} & {\Large 13} \\ \hline
\end{tabular}}\end{center}}

\problem[2.6cm]{Two boxes are empty on this track. Write the missing
numbers.\par\vspace{0.3cm}
\begin{center}{\renewcommand{\arraystretch}{1.8}%
\begin{tabular}{|c|c|c|c|c|c|}\hline
{\Large 45} & {\Large 46} & \ansblank & {\Large 48} & \ansblank & {\Large 50} \\ \hline
\end{tabular}}\end{center}}

\problem[3.2cm]{\textbf{Find and fix.} Mia counted out loud:
\begin{center}{\Large 18,\quad 19,\quad 21,\quad 22}\end{center}
She left one number out. Write the missing number: \ansblank\par\vspace{0.2cm}
Then say what Mia should remember when she counts. You can tell a grown-up or
write it below.}

\problem[3.2cm]{\textbf{Find and fix.} Leo counted out loud:
\begin{center}{\Large 38,\quad 39,\quad 30}\end{center}
One number is wrong. Draw an X on it, and write the number that belongs there:
\ansblank\par\vspace{0.2cm}
Then tell what went wrong in Leo's count.}

\problem[2.8cm]{Three number cards fell on the floor and got mixed up:
\begin{center}\fbox{\huge\ 54\ }\quad\fbox{\huge\ 52\ }\quad\fbox{\huge\ 53\ }\end{center}
\vspace{0.2cm}\par Write the numbers in counting order, from least to
greatest:\quad \ansblank\quad \ansblank\quad \ansblank}

\problem[3.5cm]{\textbf{Challenge.} This hundred-chart row has two empty boxes
--- and one number sneaked into the wrong spot!\par\vspace{0.3cm}
\begin{center}{\renewcommand{\arraystretch}{1.8}%
\begin{tabular}{|c|c|c|c|c|c|c|c|c|c|}\hline
91 & 92 & \ansblank & 94 & 95 & \textbf{69} & 97 & \ansblank & 99 & 100 \\ \hline
\end{tabular}}\end{center}
\vspace{0.2cm}\par Write the number that should be where \textbf{69} is:
\ansblank\par\vspace{0.2cm}
Then tell why 69 cannot come between 95 and 97.}
