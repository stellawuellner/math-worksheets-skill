\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Solve each inequality and show every step in the work space.

  \item Two problems show another student's work that contains a mistake.
        Find the mistake, then fix it.
\end{itemize}
