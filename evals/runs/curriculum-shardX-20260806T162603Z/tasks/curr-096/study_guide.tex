\sstitleblock{Reading Scaled Picture and Bar Graphs}

% =================== SKILL 1: read-the-scale ===================
\skillheading{Picture Graphs: Read the Scale First}

\begin{formulabox}
\textbf{Rule:} In a scaled picture graph, one symbol stands for \emph{more
than one} thing. The key tells you how many. \par
amount $=$ number of symbols $\times$ what one symbol stands for
\end{formulabox}

\begin{examplebox}
\textbf{Example:} In a picture graph, each sun symbol stands for 3 sunny
days. A month's row shows 6 sun symbols. How many sunny days is that?
\step{The key says one symbol is worth 3 days, so multiply the symbols by the
scale instead of counting symbols as ones.}
\step{$6 \times 3 = 18$.}

That row shows $\ans{18}$ sunny days.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Each flower symbol stands for 4 flowers. A row shows
5 flower symbols. How many flowers is that?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$ flowers}
\end{tryitbox}

% =================== SKILL 2: read-bar-graph ===================
\skillheading{Bar Graphs: Count by the Scale Steps}

\begin{formulabox}
\textbf{Rule:} Find where the bar \emph{ends}, then read that spot on the
number line. When the number line skips (by twos, fives, tens), count the
gridline steps and multiply by the step size.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} On a bar graph the number line counts by twos. The bar for
``red'' ends 3 gridline steps above zero. How many red things is that?
\step{Each gridline step is worth 2, so multiply the steps the bar climbs by
the step size.}
\step{$3 \times 2 = 6$.}

The red bar shows $\ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The number line counts by fives, and the ``blue'' bar ends
4 gridline steps above zero. How many blue things is that?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

% =================== SKILL 3: compare-with-graph ===================
\skillheading{Compare Two Bars or Rows}

\begin{formulabox}
\textbf{Rule:} ``How many \emph{more}'' or ``how many \emph{fewer}'' means
read \emph{both} amounts from the graph, then \textbf{subtract}:
bigger $-$ smaller.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A bar graph shows 14 blue ribbons and 9 green ribbons.
How many more blue ribbons than green ribbons are there?
\step{Comparing two amounts from the graph means subtracting the smaller bar
from the bigger one.}
\step{$14 - 9 = 5$.}

There are $\ans{5}$ more blue ribbons.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A graph shows 16 apples and 7 pears. How many more apples
than pears are there?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$ more apples}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not count symbols or gridlines as \emph{ones}
without checking the key. One symbol can stand for 5 books; one gridline step
can stand for 2 pets. Reading the scale first is the whole game.
\end{watchoutbox}
