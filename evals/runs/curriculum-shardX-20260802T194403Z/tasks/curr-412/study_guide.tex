\sstitleblock{Solving Logarithmic Equations with Domain Checks}

\begin{formulabox}
\textbf{The domain rule (use it first, every time):} $\log_b A$ exists only when
$A > 0$ (and $b > 0$, $b \neq 1$). Write the restriction on $x$ \emph{before}
solving; clearing logarithms can invent candidates the original equation never
had.\par
\textbf{Conversion:} $\log_b A = c \iff A = b^c$. \quad
\textbf{Properties (valid where every argument is positive):}
$\log_b M + \log_b N = \log_b(MN)$; \
$\log_b M - \log_b N = \log_b\!\frac{M}{N}$; \
$p\log_b M = \log_b M^p$.
\end{formulabox}

\skillheading{1. One log $=$ a number (log-to-exponential)}

\begin{formulabox}
Isolate the single logarithm, then convert: $\log_b(\text{argument}) = c$ becomes
$\text{argument} = b^c$. Solve the resulting equation and confirm the answer
keeps the argument positive.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_3(x + 2) = 4$.
\step{One logarithm, alone, equal to a number --- that is the conversion case, so switch to exponential form.}
\step{Domain $x + 2 > 0 \Rightarrow x > -2$; \ $x + 2 = 3^4 = 81$, so $x = \ans{79}$, and $79 > -2$ $\checkmark$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_5 x = 3$ (state the domain too).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 125}$}
\end{tryitbox}

\skillheading{2. Log $=$ log, same base (equate-like-logs)}

\begin{formulabox}
$\log_b M = \log_b N \iff M = N$, \textbf{provided} $M > 0$ and $N > 0$. Drop the
logarithms, solve, then test each candidate in \emph{both} original arguments.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_7(3x - 2) = \log_7(x + 8)$.
\step{Same base on both sides and nothing else in the way, so the logarithms cancel and the arguments must match.}
\step{Domain: $3x - 2 > 0$ and $x + 8 > 0 \Rightarrow x > \tfrac{2}{3}$; \ $3x - 2 = x + 8$ gives $x = \ans{5}$, and both arguments equal $13 > 0$ $\checkmark$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_2(4x + 1) = \log_2(x + 10)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\skillheading{3. Two logs on one side (combine-logs-then-solve)}

\begin{formulabox}
Use the product, quotient, or power rule to make it \emph{one} logarithm, then
convert. This is where extraneous candidates appear: the quadratic you get comes
from the equation \emph{after} the logarithms are gone, so every root must be
tested against the domain.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_3 x + \log_3(x - 8) = 2$.
\step{Two logarithms of the same base added on one side, so the product rule collapses them into one before converting.}
\step{Domain $x > 8$; \ $x(x-8) = 3^2 = 9 \Rightarrow x^2 - 8x - 9 = 0 \Rightarrow (x-9)(x+1) = 0$; \ $x = -1$ fails $x > 8$, so $x = \ans{9}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_2(x + 1) + \log_2(x - 1) = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\skillheading{4. Decimal answers (approximate-log-solutions)}

\begin{formulabox}
Convert to exponential form, keep the exact expression ($e^2 - 3$, $2^{4.7} - 5$)
through the algebra, and round only on the last line. The exponent need not be a
whole number. Then check the rounded value against the domain.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\ln(x + 3) = 2$ to the nearest hundredth.
\step{A single natural logarithm equals a number, so convert with base $e$ and round only at the end.}
\step{Domain $x > -3$; \ $x + 3 = e^2$, so $x = e^2 - 3 = 7.389\ldots - 3 \approx \ans{4.39}$, which satisfies $x > -3$ $\checkmark$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_3(x + 1) = 2.4$ to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \approx 12.97}$}
\end{tryitbox}
