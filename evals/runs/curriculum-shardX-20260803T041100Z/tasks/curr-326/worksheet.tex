\wstitleblock{Computing Distance and Midpoint from Coordinates}{High School Geometry}{}

\noindent\textbf{Where the two formulas come from.} Drop a horizontal leg and a vertical
leg from the segment's endpoints. The segment is the hypotenuse of a right triangle whose
legs are $|x_2 - x_1|$ and $|y_2 - y_1|$, so the Pythagorean theorem gives the length. The
midpoint is not a length at all --- it is the \emph{average} of the two endpoints, taken one
coordinate at a time.

  \draw[->] (-0.4,0) -- (6.6,0) node[right] {$x$};

  \draw[dashed, red] (1,1) -- (5,1) -- (5,4);
  \fill (1,1) circle (3pt); \node[below left] at (1,1) {$A(x_1, y_1)$};
  \fill (5,4) circle (3pt); \node[above right] at (5,4) {$B(x_2, y_2)$};
  \fill[blue] (3,2.5) circle (3pt); \node[above left] at (3,2.5) {$M$};
  \node[below] at (3,1) {horizontal leg};
  \node[right] at (5,2.5) {vertical leg};
\end{tikzpicture}
\end{center}

\noindent\small\textit{How every segment on this sheet is labelled: $A(x_1, y_1)$ and
$B(x_2, y_2)$ are the endpoints and $M$ is the midpoint. No values are shown here --- use
the coordinates each problem gives you.}\normalsize

\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Distance: \ $AB = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$
  \item Midpoint: \ $M = \left( \dfrac{x_1 + x_2}{2},\ \dfrac{y_1 + y_2}{2} \right)$
  \item Leave a distance in exact radical form unless the problem says otherwise; show the
        substitution step before you simplify.
\end{itemize}
\noindent\rule{\linewidth}{0.4pt}

\problem[5cm]{Find the distance between $A(1, 2)$ and $B(4, 6)$.
Show the substitution into the distance formula, and name the two legs of the right
triangle you are using.}

\problem[5cm]{Find the midpoint of the segment joining $A(2, -3)$ and $B(8, 7)$.
Write it as an ordered pair.}

\problem[5.5cm]{Find the exact distance between $A(-3, 1)$ and $B(2, 13)$.
Watch the signs when you subtract the negative $x$-coordinate.}

\problem[5.5cm]{Find the midpoint of $A(-5, 4)$ and $B(3, -2)$.
Then state which quadrant the midpoint lies in.}

\problem[5.5cm]{Find the distance between the origin $A(0, 0)$ and $B(6, 6)$.
Give the answer in exact radical form, fully simplified --- not a rounded decimal.}

\problem[6cm]{$M(3, 5)$ is the midpoint of $\overline{AB}$, and one endpoint is
$A(-1, 2)$. Find the coordinates of $B$.

\noindent (Think of the midpoint formula as two averaging equations, one for $x$ and one
for $y$, and solve each for the missing coordinate. Then check your $B$ by recomputing
the midpoint.)}

\problem[1cm]{Triangle $PQR$ has vertices $P(1, 1)$, $Q(7, 1)$, and $R(4, 6)$.
\begin{enumerate}[label=(\alph*), leftmargin=1.2cm, itemsep=2.6cm, topsep=4pt]
  \item Find the midpoint $M$ of side $\overline{PQ}$. \ansline
  \item Find the length of the median $\overline{RM}$ --- the segment from $R$ to that
        midpoint. \ansline
\end{enumerate}}

\problem[7.5cm]{\textbf{Coordinate proof.} Quadrilateral $ABCD$ has vertices
$A(0, 0)$, $B(6, 2)$, $C(8, 7)$, and $D(2, 5)$. Its diagonals are $\overline{AC}$ and
$\overline{BD}$.

\noindent Compute the midpoint of each diagonal. Then write a short proof: state what you
found and explain why sharing a midpoint proves that the diagonals \textbf{bisect each
other}. Two or three complete sentences.\noansline}
