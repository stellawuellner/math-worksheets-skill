\wstitleblock{Finding the Missing Factor}{Grades 2--3 \quad Operations and Algebraic Thinking}{}

\begin{center}
\textit{Finding a missing factor from an equal-groups picture, from a number
sentence, and from a story that names its units}
\end{center}

\noindent\textbf{How to hunt for a missing factor.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A missing factor asks: \emph{how many in each equal group}, or
\emph{how many equal groups}.
\item Think of the matching division fact. If a number sentence says
$\square \times 6 = 30$, then $30 \div 6$ names the number in the box.
\item Check your answer by multiplying back. The product must be the total you
started with.
\item When the problem tells a story, write the unit next to your answer, such
as \emph{seats} or \emph{bags}.
\end{itemize}

\vspace{0.2cm}

\problem[3.5cm]{The tiles below sit in $4$ equal rows. There are $20$ tiles in
all. Write the missing factor in the box, then say how many tiles are in each
row.
\begin{center}\Large
\begin{tabular}{ccccc}
$\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ \\[0.2cm]
$\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ \\[0.2cm]
$\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ \\[0.2cm]
$\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ & $\blacksquare$ \\
\end{tabular}
\end{center}
\begin{center}\Large
$4 \times \underline{\hspace{1.2cm}} = 20$
\end{center}}

\problem[4cm]{This division sentence is missing a number. Find it, then write

\problem[4cm]{Mia had to find the missing factor in $4 \times \square = 36$.
She wrote $32$ in the box. Explain in one sentence what Mia did instead of
finding a missing factor, then write the correct missing factor.}

\problem[5cm]{Ben lays out $56$ counters in equal rows with none left over.

(a) He uses $7$ rows. How many counters are in each row?

(b) He gathers the same $56$ counters and lays them in $8$ rows instead. How
many counters are in each row now?

Write both missing-factor equations, then explain in one sentence why the
number in each row got smaller when the number of rows got bigger.}
