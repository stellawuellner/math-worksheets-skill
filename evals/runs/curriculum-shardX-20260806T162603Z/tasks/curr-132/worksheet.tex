\wstitleblock{Multiplying Fractions and Mixed Numbers}{Grade 4--5}{}
\begin{center}
\textit{Fractions times whole numbers, fractions, and mixed numbers}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Multiply numerators together and denominators together. Rewrite any
        mixed number as an improper fraction first.
  \item Give every answer in simplest form; use a mixed number when the
        answer is bigger than 1.
\end{itemize}

\problem[3.5cm]{\textbf{Follow the pattern} $\;n \times \tfrac{a}{b} =
\tfrac{n \times a}{b}$: \quad multiply $4 \times \tfrac{2}{3}$ and write the
answer as a mixed number.}

\problem[3.5cm]{Multiply: $\tfrac{1}{2} \times \tfrac{1}{3}$}

\problem[3.5cm]{Multiply and simplify: $\tfrac{2}{3} \times \tfrac{3}{4}$}

\problem[3.5cm]{Multiply: $\tfrac{2}{5} \times 15$}

\problem[3.5cm]{Multiply: $1\tfrac{1}{2} \times 4$}

\problem[4cm]{Multiply and simplify: $\tfrac{3}{8} \times \tfrac{4}{9}$}

\problem[4.5cm]{Multiply: $2\tfrac{1}{3} \times 1\tfrac{1}{2}$}

\problem{One batch of muffins uses $\tfrac{3}{4}$ cup of milk. How many cups
of milk do 5 batches use?
\par\vspace*{4.5cm}\answerline{cups}}

\problem[4.5cm]{Find the missing factor:
$\square \times \tfrac{3}{4} = \tfrac{1}{2}$}

\problem{A shelf is built from 3 boards laid end to end. Each board is
$2\tfrac{1}{4}$ ft long. How many feet long is the shelf?
\par\vspace*{4.5cm}\answerline{ft}}

\problem[5cm]{Multiply: $1\tfrac{1}{2} \times 2\tfrac{1}{2}$. Write your
answer as a mixed number or a decimal.}

\problem{\textbf{Challenge.} A rectangular garden is $3\tfrac{1}{2}$ m long
and $2\tfrac{1}{4}$ m wide. What is its area, in square meters?
\par\vspace*{5.5cm}\answerline{m$^2$}}
