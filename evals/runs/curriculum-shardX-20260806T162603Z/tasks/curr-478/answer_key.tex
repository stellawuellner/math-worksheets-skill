\problem{Decompose $\dfrac{1}{(x-1)(x+3)}$.}

\textbf{Solution.} Write $\dfrac{1}{(x-1)(x+3)} = \dfrac{A}{x-1} + \dfrac{B}{x+3}$,
so $1 = A(x+3) + B(x-1)$. Setting $x = 1$ gives $4A = 1$; setting $x = -3$ gives
$-4B = 1$.
\[ \boxed{\frac{1}{4(x - 1)} - \frac{1}{4(x + 3)}} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\int \dfrac{5}{(x-2)(x+3)}dx$.}

\textbf{Solution.} $5 = A(x+3) + B(x-2)$ gives $A = 1$ at $x = 2$ and $B = -1$
at $x = -3$, so the integrand is $\dfrac{1}{x-2} - \dfrac{1}{x+3}$.
\[ \boxed{\ln|x - 2| - \ln|x + 3| + C} \]
Equivalently $\ln\left|\dfrac{x-2}{x+3}\right| + C$.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$r(x) = \dfrac{6}{(x+1)(x+4)}$ g/min. (a) Exact mass in the first
$2$ minutes. (b) To three decimals.}

\textbf{Solution (a).} $6 = A(x+4) + B(x+1)$ gives $A = 2$ and $B = -2$, so
\[ \int_0^2\left(\frac{2}{x+1} - \frac{2}{x+4}\right)dx
   = \left[2\ln\frac{x+1}{x+4}\right]_0^2
   = 2\ln\frac{3}{6} - 2\ln\frac{1}{4}. \]
\[ \boxed{2\ln 2\ \text{g}} \]

\textbf{Solution (b).}
\[ \boxed{1.386} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\int \dfrac{x+4}{(x+2)^2}dx$.}

\textbf{Solution.} A repeated factor needs both powers:
$\dfrac{x+4}{(x+2)^2} = \dfrac{A}{x+2} + \dfrac{B}{(x+2)^2}$, so
$x + 4 = A(x+2) + B$, giving $A = 1$ and $B = 2$.
\[ \int\left(\frac{1}{x+2} + \frac{2}{(x+2)^2}\right)dx \]
\[ \boxed{\ln|x + 2| - \frac{2}{x + 2} + C} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\int \dfrac{1}{P(10-P)}dP$.}

\textbf{Solution.} $1 = A(10 - P) + BP$ gives $A = B = \dfrac{1}{10}$, so
\[ \int\frac{1}{P(10-P)}dP
   = \frac{1}{10}\int\left(\frac{1}{P} + \frac{1}{10-P}\right)dP . \]
\[ \boxed{\frac{\ln|P|}{10} - \frac{\ln|10 - P|}{10} + C} \]
This is the step that turns the logistic differential equation into something
solvable for $P$.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\int \dfrac{x^2+2x+4}{x(x^2+4)}dx$.}

\textbf{Solution.} The quadratic is irreducible, so its numerator is linear:
\[ \frac{x^2+2x+4}{x(x^2+4)} = \frac{A}{x} + \frac{Bx + C}{x^2+4}. \]
Clearing denominators, $x^2 + 2x + 4 = A(x^2+4) + (Bx + C)x$. At $x = 0$,
$4A = 4$, so $A = 1$; matching $x^2$ terms gives $B = 0$ and matching $x$ terms
gives $C = 2$.
\[ \int\left(\frac{1}{x} + \frac{2}{x^2+4}\right)dx \]
\[ \boxed{\ln|x| + \arctan\frac{x}{2} + C} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$r(t) = \dfrac{8}{(t+1)(t+3)}$ kg/yr. (a) Exact mass in $3$ years.
(b) To three decimals.}

\textbf{Solution (a).} $8 = A(t+3) + B(t+1)$ gives $A = 4$ and $B = -4$:
\[ \int_0^3\left(\frac{4}{t+1} - \frac{4}{t+3}\right)dt
   = \left[4\ln\frac{t+1}{t+3}\right]_0^3
   = 4\ln\frac{4}{6} - 4\ln\frac{1}{3}. \]
\[ \boxed{4\ln 2\ \text{kg}} \]

\textbf{Solution (b).}
\[ \boxed{2.773} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$r(x) = \dfrac{2}{x(x+2)}$ kg/yr. (a) Mass from $x = 1$ to $3$.
(b) The limit. (c) What it means.}

\textbf{Solution (a).} $2 = A(x+2) + Bx$ gives $A = 1$, $B = -1$, so
\[ \int_1^3\left(\frac{1}{x} - \frac{1}{x+2}\right)dx
   = \left[\ln\frac{x}{x+2}\right]_1^3 = \ln\frac{3}{5} - \ln\frac{1}{3}. \]
\[ \boxed{\ln\frac{9}{5}} \]

\textbf{Solution (b).} As $b \rightarrow \infty$, $\dfrac{b}{b+2}\rightarrow 1$
and its logarithm goes to zero:
\[ \boxed{\ln 3} \]

\textbf{Solution (c) --- instructor-judged.} The total mass removed over all
future time approaches a finite number, $\ln 3$ kilograms, instead of growing
without bound: the improper integral converges. So the filter has a lifetime
ceiling on what it can ever take out, no matter how long it runs. Full credit
needs that finite-total conclusion, not merely the value of the limit.
