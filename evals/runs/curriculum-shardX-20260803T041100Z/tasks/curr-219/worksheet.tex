\wstitleblock{Slope and Rate of Change --- Diagnosing Inverted and Sign-Reversed Calculations}{Grade 8 / Pre-Algebra}{}

\noindent\small Slope is \textbf{rise over run}, and both differences must be
taken in the \textbf{same order}:
\[ m = \frac{y_2 - y_1}{x_2 - x_1} \]
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Flipping the fraction gives the \textbf{inverted} slope (run over rise).
  \item Starting with point $1$ on top and point $2$ on the bottom gives the
        \textbf{sign-reversed} slope.
  \item In each problem, decide which of those two errors was made (if any), then
        compute the correct value. Show the subtraction you used.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{Warm up. Find the slope of the line through $(2, 3)$ and $(6, 11)$.
Label your rise and your run before you divide.}

\problem[5cm]{\textbf{Find and fix.} Devi finds the slope through $(-3, 5)$ and
$(2, -5)$. She writes:
\[ m = \frac{5 - (-5)}{2 - (-3)} = \frac{10}{5} = 2 \]
Devi's arithmetic is correct, but her answer is not. Name the error in her set-up,
then give the correct slope.}

\problem[5cm]{\textbf{Find and fix.} Mika finds the slope through $(1, 4)$ and
$(5, 6)$. He writes:
\[ m = \frac{5 - 1}{6 - 4} = \frac{4}{2} = 2 \]
Name Mika's error, then give the correct slope as a fraction.}

\problem[5cm]{A tank is draining. The table shows the water level, in litres, at
four times.\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Hours} & $0$ & $2$ & $4$ & $6$ \\ \hline
\textbf{Litres} & $46$ & $40$ & $34$ & $28$ \\ \hline
\end{tabular}
\end{center}
Find the \textbf{change} in the water level from hour $0$ to hour $6$. A change
that goes down is negative, so write your answer with its sign.}

\problem[5cm]{Use the same draining tank.\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Hours} & $0$ & $2$ & $4$ & $6$ \\ \hline
\textbf{Litres} & $46$ & $40$ & $34$ & $28$ \\ \hline
\end{tabular}
\end{center}
Find the rate of change of the water level, in litres per hour, between hour $0$
and hour $6$. Keep the level difference on top and the time difference underneath,
both taken in the same order.}

\problem[5cm]{\textbf{Find and fix.} Ty finds the slope through $(5, 9)$ and
$(1, 1)$. He writes:
\[ m = \frac{1 - 9}{5 - 1} = \frac{-8}{4} = -2 \]
Both of Ty's differences came out of the same two points, so his arithmetic looks
fine --- but he mixed the order. Name the error and give the correct slope.
Then say whether the line rises or falls from left to right.}

\problem[6cm]{A candle burns at a steady rate. At $3$ minutes it is $18$ cm tall,
and at $11$ minutes it is $6$ cm tall.\par
Jo reports the rate of change as $\dfrac{11 - 3}{6 - 18} \approx -0.67$ cm per
minute. Explain what Jo inverted, then find the correct rate of change of the
candle's height, in centimetres per minute.}

\problem[6.5cm]{\textbf{Challenge.} A line passes through $(a, 4)$ and $(7, -2)$,
and its slope is $-3$. Priya sets up
\[ \frac{7 - a}{-2 - 4} = -3 \]
and reports $a = -11$.\par
Diagnose Priya's set-up error, write the correct equation, and solve it for $a$.}
