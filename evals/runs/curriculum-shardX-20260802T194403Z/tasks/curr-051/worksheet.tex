\ifnum#1>0\relax\foreach \bj in {1,...,#1}{\btflat\hspace{3pt}}\hspace{12pt}\fi%

\wstitleblock{Reading Base-Ten Models: Hundreds, Tens, and Ones}{Grade 2--3 Place Value}{}

\noindent\textbf{How to read the pictures.} Every picture below is built from three
kinds of blocks:

\begin{center}
\begin{tabular}{ccl}
\btflat & \quad & \textbf{flat} --- a square of one hundred small cubes $=$ one hundred \\[6pt]
\btrod  & \quad & \textbf{rod} --- a stick of ten small cubes $=$ one ten \\[6pt]
\btone  & \quad & \textbf{unit} --- one small cube $=$ one one \\
\end{tabular}
\end{center}

\noindent Count the flats, then the rods, then the units. Say it out loud
(``two hundreds, four tens, three ones''), write it as an addition, and then
write the number. Show your counting in the space under each picture.

\vspace{0.3cm}

\problem[3.4cm]{Count the blocks and write the number.
\btmodel{2}{4}{3}
\underline{\hspace{1.2cm}} hundreds \quad \underline{\hspace{1.2cm}} tens \quad
\underline{\hspace{1.2cm}} ones}

\problem[3.4cm]{Careful --- one place is empty here. Count the blocks and write the number.
\btmodel{3}{0}{6}
\underline{\hspace{1.2cm}} hundreds \quad \underline{\hspace{1.2cm}} tens \quad
\underline{\hspace{1.2cm}} ones}

\problem[3.6cm]{Write this model as an addition of hundreds, tens, and ones
(expanded form), then write the number.
\btmodel{5}{7}{2}
Expanded form: \underline{\hspace{1.4cm}} $+$ \underline{\hspace{1.4cm}} $+$
\underline{\hspace{1.4cm}}}

\problem[3.6cm]{Write this model in expanded form, then write the number.
\btmodel{1}{8}{0}
Expanded form: \underline{\hspace{1.4cm}} $+$ \underline{\hspace{1.4cm}} $+$
\underline{\hspace{1.4cm}}}

\problem[3.6cm]{This model shows four flats, six rods, and seven units.
\btmodel{4}{6}{7}
The rods all together are worth how much? Write the \textbf{value of the rods}
(not how many rods there are).}

\problem[4.2cm]{Marcus built this model. There are more than nine rods, so he can
trade ten rods for one flat before he reads the number.
\btmodel{2}{13}{5}
Trade first, then write the number the blocks show.}

\problem[4.2cm]{Write the number for each model, then compare them with $<$ or $>$.
\btmodel{3}{4}{2}
Model A $=$ \underline{\hspace{1.6cm}}
\btmodel{3}{2}{8}
Model B $=$ \underline{\hspace{1.6cm}} \\[4pt]
Write $<$ or $>$: \quad Model A \underline{\hspace{1.4cm}} Model B}

\problem[4.6cm]{\textbf{Challenge.} Rosa says Model A is bigger because it has more
flats. Trade rods for flats where you can, write both numbers, then compare.
\btmodel{4}{2}{6}
Model A $=$ \underline{\hspace{1.6cm}}
\btmodel{3}{12}{9}
Model B $=$ \underline{\hspace{1.6cm}} \\[4pt]
Write $<$ or $>$: \quad Model A \underline{\hspace{1.4cm}} Model B}
