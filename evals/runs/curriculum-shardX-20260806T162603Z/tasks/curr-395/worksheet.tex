\wstitleblock{Binomial Coefficients and the Structure of an Expansion}{Algebra 2}{}

\begin{center}
\small\textit{Expanding binomial powers, picking out a named term, using the structure of Pascal's triangle, and summing coefficients}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item In $(A + B)^n$ written in descending powers of $A$, the term at index $k$ is $\binom{n}{k}A^{n-k}B^{k}$, and $k$ starts at $0$.
  \item Not every problem wants a full expansion. Read the question first: sometimes one term, one coefficient, or one total is all that is asked for.
  \item Leave answers exact, and keep signs attached to the number they belong to.
\end{itemize}

\vspace{0.1cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.6cm]{Expand $(x + 1)^3$ completely.}

\problem[4.6cm]{Expand $(x - 4)^3$ completely.}

\problem[4.8cm]{Find the coefficient of $x^3$ in the expansion of $(x + 2)^5$. Do not expand the whole thing.}

\problem[4.8cm]{Find the sum of all the coefficients in the expansion of $(2x + 3)^4$, without expanding it.}

\problem[5cm]{Evaluate $\displaystyle\sum_{k=0}^{5} \binom{5}{k}$, the total of row $5$ of Pascal's triangle.}

\problem[5.2cm]{Expand $(2x - 1)^4$ completely.}

\problem[5.2cm]{Find the coefficient of $x^2$ in the expansion of $(3x - 2)^5$.}

\problem{Every row of Pascal's triangle reads the same forwards and backwards, so binomial coefficients come in mirror-image pairs.
\par\smallskip
(a)~Compute $\dbinom{9}{3}$.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~Explain why $\dbinom{9}{3}$ and $\dbinom{9}{6}$ must be equal, without computing the second one.
\par\vspace*{2.6cm}\ansline}

\problem[5.2cm]{For each whole number $n$, the coefficients of $(1 + x)^n$ add up to $2^n$. Evaluate $\displaystyle\sum_{n=0}^{5} 2^n$, the total of those sums for $n = 0$ through $n = 5$.}

\problem[4.8cm]{Find the coefficient of $x^4y^2$ in the expansion of $(x + y)^6$.}

\problem[5.2cm]{Expand $(x^2 + 3)^3$ completely, in descending powers of $x$.}

\problem{The expansion of $(1 + x)^4$ is the key to all three parts.
\par\smallskip
(a)~Expand $(1 + x)^4$ completely.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~Use your expansion to evaluate $3^4$ by substituting a suitable value of $x$.
\par\vspace*{2.6cm}\ansline
\par\smallskip
(c)~For some whole number $n$, the coefficients of $(1 + x)^n$ add up to $512$. Find $n$, and explain why substituting $x = 1$ gives the sum of the coefficients.
\par\vspace*{3.2cm}\ansline}
