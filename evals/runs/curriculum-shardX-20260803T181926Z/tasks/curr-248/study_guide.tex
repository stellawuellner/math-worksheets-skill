\sstitleblock{Two-Way Tables: Comparing Categorical Frequencies}

\noindent\small Every example below reads this table. A vet clinic recorded
100 pets by species and by where they live:\normalsize
\par\smallskip
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{|l|c|c|c|}
\hline
 & \textbf{Indoor} & \textbf{Outdoor} & \textbf{Row total} \\
\hline
\textbf{Cats} & 21 & 9 & 30 \\
\hline
\textbf{Dogs} & 14 & 56 & 70 \\
\hline
\textbf{Column total} & 35 & 65 & 100 \\
\hline
\end{tabular}
\end{center}

\skillheading{Skill 1 --- Reading a two-way table}

\begin{formulabox}
\textbf{Rule:} a cell is where one row meets one column, so it counts the
individuals in \emph{both} categories at once. A \textbf{row total} is that
row's cells added up; a \textbf{column total} is that column's cells added up;
the \textbf{grand total} is every inner cell added up, and it must equal both
the sum of the row totals and the sum of the column totals.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad How many pets are dogs that live outdoors?
\step{``Dogs'' and ``outdoors'' are two categories at once, so the answer is a
single cell --- find the row, then the column.}
\step{Read across the Dogs row and down the Outdoor column; they meet at one
number.}

$\ans{56}$ dogs live outdoors.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad How many pets are cats that live outdoors?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 2 --- Row relative frequency}

\begin{formulabox}
\textbf{Rule:} divide a cell by \emph{its own row} total, then multiply by 100.
Row relative frequency answers ``of the ones in this row, what percent are in
this column?'' The percents along one row always add to 100.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Of the cats, what percent live indoors?
\step{The question starts with ``of the cats,'' so the group being described is
the Cats row --- that row's total is the base.}
\step{The cell is 21 and the Cats row total is 30, so
$100 \times \frac{21}{30} = 100 \times 0.7$.}

$\ans{70\%}$ of the cats live indoors.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Of the dogs, what percent live indoors?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20\%}$}
\end{tryitbox}

\skillheading{Skill 3 --- Column relative frequency}

\begin{formulabox}
\textbf{Rule:} divide the same kind of cell by \emph{its own column} total,
then multiply by 100. Column relative frequency answers the reversed question:
``of the ones in this column, what percent are in this row?'' Same cell,
different base, different answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Of the indoor pets, what percent are cats?
\step{This time the group is ``the indoor pets,'' which is a column, so the
Indoor column total is the base --- not the Cats row total.}
\step{The cell is 21 and the Indoor column total is 35, so
$100 \times \frac{21}{35} = 100 \times 0.6$.}

$\ans{60\%}$ of the indoor pets are cats.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Of the indoor pets, what percent are dogs?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40\%}$}
\end{tryitbox}

\skillheading{Skill 4 --- Comparing to judge association}

\begin{formulabox}
\textbf{Rule:} compute the same relative frequency for both rows, then subtract
to get the gap in \textbf{percentage points}. A large gap means the two
variables are \textbf{associated} --- knowing one changes what you expect of the
other. A gap near zero means little or no association. Never compare raw counts
across rows of different sizes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 70 percent of cats live indoors and 20 percent of dogs
do. Is species associated with where the pet lives?
\step{The two rows hold different numbers of pets, so compare the two row
percents, not the counts 21 and 14.}
\step{The gap is $70 - 20$ percentage points, which is large.}

$\ans{50}$ percentage points --- yes, there is an association.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad In another clinic 45 percent of cats and 42 percent of
dogs live indoors. Find the gap, then say whether there is an
association.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ points --- no real association}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The most costly mistake on this topic is using the wrong
total. The same cell can give three different percents --- out of its row, out
of its column, or out of the grand total --- and they answer three different
questions. Before you divide, read the phrase ``of the \dots'' in the question:
whatever follows it names the total you need.
\end{watchoutbox}
