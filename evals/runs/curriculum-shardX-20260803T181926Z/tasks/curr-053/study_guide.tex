\sstitleblock{Comparing Three-Digit Numbers}

\noindent\small Two three-digit numbers are never compared digit by digit from
the right. Start at the \textbf{left}: hundreds first, then tens, then ones.
The first place where the digits differ decides everything.
\normalsize

\skillheading{1. Compare with a place-value chart}
\begin{formulabox}
\textbf{Rule:} write one digit in each box, then walk left to right.
Same hundreds? Look at tens. Same tens? Look at ones.\\[3pt]
$>$ means ``is greater than''; $<$ means ``is less than''. The point of the
symbol always aims at the smaller number.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Compare $573$ and $537$.
\step{Both numbers have three digits, so the compare starts at the hundreds
place --- that is the place worth the most.}
\step{Hundreds: $5$ and $5$, tied. Tens: $7$ against $3$, so the first number
is ahead and the ones digits are never needed.}
\par
$\ans{573 > 537}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare $248$ and $284$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{248 < 284}$}
\end{tryitbox}

\skillheading{2. Compare base-ten blocks}
\begin{formulabox}
\textbf{Rule:} count flats, then rods, then cubes --- the same left-to-right
order as the chart.
\begin{center}
\begin{tabular}{ccc}
\begin{tikzpicture}[scale=0.42]
  \draw[thick] (0,0) rectangle (2,2);
  \foreach \i in {0.4,0.8,1.2,1.6} {\draw (\i,0) -- (\i,2); \draw (0,\i) -- (2,\i);}
\end{tikzpicture}
&
\begin{tikzpicture}[scale=0.42]
  \draw[thick] (0,0) rectangle (0.4,2);
  \foreach \i in {0.4,0.8,1.2,1.6} {\draw (0,\i) -- (0.4,\i);}
\end{tikzpicture}
&
\begin{tikzpicture}[scale=0.42]
  \draw[thick] (0,0) rectangle (0.4,0.4);
\end{tikzpicture}
\\
\footnotesize flat = one hundred & \footnotesize rod = one ten
& \footnotesize cube = one
\end{tabular}
\end{center}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad What number is $4$ flats, $1$ rod, $6$ cubes?
\step{Each kind of block names one place, so give each pile its place value
instead of counting all the pieces.}
\step{$4 \times 100 = 400$, \ $1 \times 10 = 10$, \ $6 \times 1 = 6$, and
$400 + 10 + 6$ fills hundreds, tens, ones.}
\par
$\ans{416}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad What number is $3$ flats, $0$ rods, $7$ cubes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{307}$}
\end{tryitbox}

\skillheading{3. Put three numbers in order}
\begin{formulabox}
\textbf{Rule:} sort by hundreds first and only break ties with tens, then ones.
Writing the three numbers in a chart, one under the other, makes the ties easy
to see.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Order $316$, $361$, $136$ from least to greatest.
\step{Ordering is just comparing more than once, so sort on the hundreds digit
before looking anywhere else.}
\step{Hundreds are $3$, $3$, $1$, so $136$ is smallest. The two with $3$
hundreds split on tens: $1$ ten against $6$ tens.}
\par
$\ans{136, 316, 361}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Order $820$, $802$, $280$ from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{280, 802, 820}$}
\end{tryitbox}

\skillheading{4. Comparing counts in a story}
\begin{formulabox}
\textbf{Rule:} pull the two numbers out of the story, compare them the usual
way, then answer the question that was asked --- ``who has more'' needs a
\emph{name}, not just a symbol.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Room A read $734$ pages, Room B read $743$ pages. Which
room read more?
\step{The question is a comparison of two counts, so chart the numbers and
compare from the left.}
\step{Hundreds tie at $7$; tens are $3$ against $4$, so Room B is ahead.}
\par
$\ans{734 < 743}$, so Room B read more.
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A shop sold $509$ apples and $590$ pears. Were there
fewer apples or fewer pears?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{509 < 590}$, fewer apples}
\end{tryitbox}
