\wstitleblock{Congruent or Similar? Evidence from Transformations}{Grade 8 / Pre-Algebra \quad Geometry}{}

\begin{center}
\textit{Using side lengths and slopes as evidence that a translation,
reflection, rotation, or dilation maps one figure onto another}
\end{center}

\noindent\textbf{Directions.} Saying two figures ``look the same'' is not
evidence. On this sheet you show it with numbers: compute lengths with the
distance formula and directions with slope, then say what those numbers prove.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=3pt]
\item \textbf{Distance:} $d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$. \quad
\textbf{Slope:} $m = \dfrac{y_2 - y_1}{x_2 - x_1}$ (undefined when the segment
is vertical).
\item \textbf{Congruent} means a sequence of \emph{rigid motions} ---
translation, reflection, rotation --- maps one figure exactly onto the other.
Every corresponding length stays the same.
\item \textbf{Similar} means a dilation may be used as well. Corresponding
lengths are then all multiplied by the same scale factor $k$.
\item \textbf{Slopes tell you which motion.} A translation keeps every slope
unchanged. A half turn does too. A reflection across $y = x$ swaps a slope for
its reciprocal. A vertical segment has no slope at all.
\end{itemize}

\vspace{0.1cm}
\noindent Round nothing: every answer on this sheet comes out exact. Show the
substitution into the formula, not just the result.

\problem{Triangle $ABC$ has vertices $A(1, 1)$, $B(5, 1)$, $C(5, 4)$. It
is translated to triangle $A'B'C'$ with $A'(4, 3)$, $B'(8, 3)$, $C'(8, 6)$.

Find $AB$ and $A'B'$, and state what the two lengths show.
\par\vspace*{5.5cm}\answerline{units}}

\problem{Segment $DE$ has endpoints $D(2, 3)$ and $E(6, 6)$. It is
reflected across the $x$-axis to $D'(2, -3)$ and $E'(6, -6)$.

Find $DE$ and $D'E'$, and state what the two lengths show.
\par\vspace*{5.5cm}\answerline{units}}

\problem{Segment $FG$ has endpoints $F(0, 0)$ and $G(3, 4)$. A dilation
centred at the origin sends it to $F'(0, 0)$ and $G'(6, 8)$.

Find $FG$ and $F'G'$. Are the segments congruent? What is the scale factor?
\par\vspace*{6cm}\answerline{units}}

\problem{Segment $HJ$ has endpoints $H(1, 2)$ and $J(4, 6)$. A rotation
about the origin sends it to $H'(-2, 1)$ and $J'(-6, 4)$.

Find $HJ$ and $H'J'$. Does this rotation preserve length?
\par\vspace*{6cm}\answerline{units}}

\problem[6cm]{Segment $KL$ has endpoints $K(-2, 1)$ and $L(2, 9)$. Its image
under a translation is $K'(1, -1)$ and $L'(5, 7)$.

Find the slope of $KL$ and the slope of $K'L'$. What do the two slopes tell you
about the two segments?}

\problem{Triangle $MNP$ has vertices $M(0, 0)$, $N(4, 0)$, $P(0, 3)$. A
dilation centred at the origin sends it to $M'(0, 0)$, $N'(6, 0)$,
$P'(0, 4.5)$.

Find $MN$, $M'N'$, $NP$, and $N'P'$. Then compute both ratios
$\frac{M'N'}{MN}$ and $\frac{N'P'}{NP}$ and say what they show about the two
triangles.
\par\vspace*{7cm}\answerline{units}}

\problem{Triangle $QRS$ has vertices $Q(1, 1)$, $R(4, 1)$, $S(1, 5)$.
Its image is $Q'(1, 1)$, $R'(1, 4)$, $S'(5, 1)$.

Find all three side lengths of each triangle: $QR$ and $Q'R'$, $RS$ and
$R'S'$, $QS$ and $Q'S'$. Then state the congruence criterion your six numbers
establish.
\par\vspace*{7.5cm}\answerline{units}}

\problem[7cm]{Segment $TU$ has endpoints $T(2, 1)$ and $U(6, 4)$. A rotation of
half a turn about the origin sends it to $T'(-2, -1)$ and $U'(-6, -4)$.

\textbf{(a)} Find $TU$ and $T'U'$.

\textbf{(b)} Find the slope of $TU$ and the slope of $T'U'$.

\textbf{(c)} Most rotations change a segment's slope. Explain, using your two
slopes, why this one does not.}

\problem[7cm]{Ana says: ``Triangle $ABC$ with $A(0, 0)$, $B(4, 0)$, $C(0, 3)$
and triangle $A'B'C'$ with $A'(0, 0)$, $B'(8, 0)$, $C'(0, 6)$ are congruent,
because both are right triangles with a vertex at the origin.''

\textbf{(a)} Find $AB$ and $A'B'$.

\textbf{(b)} Use those two lengths to explain in one sentence why Ana is wrong.

\textbf{(c)} Name the relationship that \emph{does} hold between the two
triangles, and give the transformation that produces it.}
