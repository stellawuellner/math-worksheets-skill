\wstitleblock{Interpreting Correlation and Least-Squares Regression}{Precalculus / Statistics}{}

\begin{center}
\small Reading the data, using the least-squares line, meaning of slope and intercept, and correlation
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item All problems use the study below. $x$ is study time in hours; $y$ is the test score in points.
  \item The least-squares line for these six students is $\hat{y} = 50 + 6x$.
  \item A residual is $y - \hat{y}$: actual minus predicted. Keep the sign.
  \item Round any decimal answer to four decimal places unless told otherwise.
\end{itemize}

\vspace{0.2cm}
\begin{center}
\textbf{Six students: study time and test score}\\[4pt]
\begin{tabular}{lcccccc}
\toprule
\textbf{Study time $x$ (hours)} & 1 & 2 & 3 & 4 & 5 & 6 \\
\textbf{Test score $y$ (points)} & 59 & 59 & 68 & 74 & 77 & 89 \\
\bottomrule
\end{tabular}\\[4pt]
\small For this data set: \ $\sum(x-\bar{x})(y-\bar{y}) = 105$, \quad
$\sum(x-\bar{x})^2 = 17.5$, \quad $\sum(y-\bar{y})^2 = 666$.
\end{center}
\vspace{0.2cm}

\problem[4cm]{Read from the table: what score did the student who studied $4$ hours earn?}

\problem[4.5cm]{Use the least-squares line to predict the score of a student who studies
$3.5$ hours.}

\problem[0.4cm]{The slope of the line is $6$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] By how many points does the line predict the score changes for $2$ additional hours of study? \quad \ansblank
  \item[(b)] Interpret the slope of $6$ in the context of this study, in a sentence.
\end{itemize}
\vspace*{2.8cm}}

\problem[4.5cm]{The student who studied $5$ hours scored $77$ points. Find the residual for
that student.}

\problem[5cm]{Using the three sums given above, compute the correlation coefficient
\ $r = \dfrac{\sum(x-\bar{x})(y-\bar{y})}{\sqrt{\sum(x-\bar{x})^2 \cdot \sum(y-\bar{y})^2}}$.}

\problem[0.4cm]{The coefficient of determination is $r^2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $r^2$ for this data set. \quad \ansblank
  \item[(b)] Interpret that number in the context of this study, in a sentence.
\end{itemize}
\vspace*{2.8cm}}

\problem[0.4cm]{The intercept of the line is $50$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What score does the line predict for a student who studies $0$ hours? \quad \ansblank
  \item[(b)] Explain whether that number is a trustworthy prediction for these data.
\end{itemize}
\vspace*{2.8cm}}

\problem[0.4cm]{Consider the six students as a group.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] One of the six is chosen at random. Find the probability that the chosen student studied more than $4$ hours. Give an exact fraction. \quad \ansblank
  \item[(b)] The correlation here is about $0.97$. Explain why that strong correlation does not show that studying causes the higher scores.
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{A teacher wants to use the line for a student planning to study $8$ hours.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What score does the line predict at $x = 8$? \quad \ansblank
  \item[(b)] Explain why this particular prediction should be treated with caution.
\end{itemize}
\vspace*{2.8cm}}

\problem[0.4cm]{\textbf{Synthesis.} A seventh student studies $4$ hours and scores $62$ points.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find this student's residual against the existing line $\hat{y} = 50 + 6x$. \quad \ansblank
  \item[(b)] Explain what a residual of that size means, and how adding this point would change the correlation and the fitted line.
\end{itemize}
