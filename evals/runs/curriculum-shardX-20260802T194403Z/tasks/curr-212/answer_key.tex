\noindent\small Each solution shows the same four moves in the same order: clear fractions,
distribute, collect, isolate. Where a step is the one students most often get wrong, it is called
out. Substituting the answer back into the \emph{original} equation is the check to insist on.
\normalsize

\problem{Solve $3(x + 4) = 21$.}
\textbf{Solution:} Distribute the $3$ across both terms in the brackets, then undo.
\begin{align*}
3x + 12 &= 21 \\
3x &= 9 \\
x &= 3
\end{align*}
\ans{x = 3}
Check: $3(3 + 4) = 3 \cdot 7 = 21$. \emph{Faster alternative:} divide both sides by $3$ first,
giving $x + 4 = 7$.

\problem{Solve $-2(x - 5) = 14$.}
\textbf{Solution:} The factor is $-2$, so both products change sign.
\begin{align*}
-2x + 10 &= 14 \\
-2x &= 4 \\
x &= -2
\end{align*}
\ans{x = -2}
\emph{Common error:} writing $-2x - 10 = 14$ (sign not distributed to the $-5$) gives $x = -12$.

\problem{Solve $\dfrac{x}{3} + \dfrac{x}{6} = 5$.}
\textbf{Solution:} Denominators $3$ and $6$: the LCD is $6$. Multiply \emph{every} term by $6$.
\begin{align*}
6 \cdot \tfrac{x}{3} + 6 \cdot \tfrac{x}{6} &= 6 \cdot 5 \\
2x + x &= 30 \\
3x &= 30 \\
x &= 10
\end{align*}
\ans{x = 10}
Check: $\tfrac{10}{3} + \tfrac{10}{6} = \tfrac{20}{6} + \tfrac{10}{6} = 5$.

\problem{Solve $4 - 2(x + 3) = 10$.}
\textbf{Solution:} The $2$ being subtracted distributes as $-2$; only then combine the constants.
\begin{align*}
4 - 2x - 6 &= 10 \\
-2x - 2 &= 10 \\
-2x &= 12 \\
x &= -6
\end{align*}
\ans{x = -6}
\emph{Common error:} subtracting first ($2 \cdot (x+3) = 10$, i.e. treating $4-2$ as $2$) gives
$x = 2$ --- distribution comes before that subtraction.

\problem{Multiply $\dfrac{x}{4} + \dfrac{2}{3}$ by the LCD, $12$, and write it without fractions.}
\textbf{Solution:} Multiply each term separately:
$12 \cdot \dfrac{x}{4} = 3x$ \ and \ $12 \cdot \dfrac{2}{3} = 8$.
\ans{3x + 8}
\emph{Why it works:} $12$ is a multiple of both $4$ and $3$, so each denominator divides out
exactly. Multiplying only the first term is the usual slip.

\problem{Solve $3(x - 2) = 2(x + 4)$.}
\textbf{Solution:} Distribute on both sides, then collect the variable on the left.
\begin{align*}
3x - 6 &= 2x + 8 \\
x - 6 &= 8 \\
x &= 14
\end{align*}
\ans{x = 14}
Check: $3(14 - 2) = 36$ and $2(14 + 4) = 36$.

\problem{Solve $\dfrac{3x}{4} - 2 = \dfrac{x}{2}$.}
\textbf{Solution:} LCD of $4$ and $2$ is $4$; multiply every term, including the $-2$.
\begin{align*}
3x - 8 &= 2x \\
x - 8 &= 0 \\
x &= 8
\end{align*}
\ans{x = 8}
\emph{Common error:} forgetting to multiply the whole-number term gives $3x - 2 = 2x$ and $x = 2$.

\problem{Solve $\dfrac{1}{2}(6x - 8) = 11$.}
\textbf{Solution:} A fraction in front of brackets still distributes: halve each term.
\begin{align*}
3x - 4 &= 11 \\
3x &= 15 \\
x &= 5
\end{align*}
\ans{x = 5}
\emph{Equivalent route:} multiply both sides by $2$ first, giving $6x - 8 = 22$.

\problem{Expand and simplify $7 - 3(2x - 5)$.}
\textbf{Solution:} Distribute $-3$ over both terms, then combine the constants.
\[ 7 - 6x + 15 \;=\; 22 - 6x \]
\ans{22 - 6x}
\emph{Common error:} $7 - 6x - 15 = -8 - 6x$ --- the second sign was not changed.

\problem{Solve $\dfrac{2}{3}(x + 6) = \dfrac{1}{2}(x + 2) + 5$.}
\textbf{Solution:} LCD of $3$ and $2$ is $6$. Multiply every term by $6$ (keeping the brackets),
then distribute.
\begin{align*}
4(x + 6) &= 3(x + 2) + 30 \\
4x + 24 &= 3x + 6 + 30 \\
4x + 24 &= 3x + 36 \\
x &= 12
\end{align*}
\ans{x = 12}
Check: $\tfrac{2}{3}(18) = 12$ and $\tfrac{1}{2}(14) + 5 = 12$.

\problem{Solve $\dfrac{x + 3}{2} - \dfrac{x - 5}{4} = 5$.}
\textbf{Solution:} The bar groups the whole numerator, so keep brackets when multiplying by the
LCD $4$.
\begin{align*}
2(x + 3) - (x - 5) &= 20 \\
2x + 6 - x + 5 &= 20 \\
x + 11 &= 20 \\
x &= 9
\end{align*}
\ans{x = 9}
\emph{Common error:} $2x + 6 - x - 5$ --- the minus in front of the second fraction must reach both
terms of its numerator; that slip gives $x = 19$.

\problem{Challenge: solve $\dfrac{2}{5}(10x - 15) = 3(x - 1) + 4$.}
\textbf{Solution:} $\tfrac{2}{5}$ distributes over both terms; $\tfrac{2}{5} \cdot 10x = 4x$ and
$\tfrac{2}{5} \cdot 15 = 6$.
\begin{align*}
4x - 6 &= 3x - 3 + 4 \\
4x - 6 &= 3x + 1 \\
x &= 7
\end{align*}
\ans{x = 7}
Check in the original: $\tfrac{2}{5}(70 - 15) = \tfrac{2}{5}(55) = 22$, and
$3(7 - 1) + 4 = 18 + 4 = 22$. Both sides agree, so $x = 7$ is confirmed.
