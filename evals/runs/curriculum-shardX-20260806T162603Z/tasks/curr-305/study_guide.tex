\sstitleblock{Building Multi-Step Arguments from Givens}

\skillheading{Skill 1 --- Putting the steps in a working order}

\begin{formulabox}
\textbf{Every argument has the same shape:} start at a given, apply a definition to turn words into symbols, apply a postulate or theorem to get a relationship, then use properties of equality to finish.\par
\textbf{Ordering test:} a line may be written only if everything it needs is already above it. If a line needs something below it, it is in the wrong place.\par
\textbf{Two reasons that always come as a pair:} the \emph{definition of congruent} segments or angles turns a congruence mark into an equation; the \emph{definition} that produced the congruence comes first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two lines cross. Given that a pair of vertical angles measure $(5x - 4)^\circ$ and $(3x + 12)^\circ$, find $x$.
\step{The given intersection lets me name the pair vertical, the vertical angles theorem then gives congruence, and the definition of congruent angles turns that into an equation --- three steps in that order.}
\step{$5x - 4 = 3x + 12$, so $2x = 16$.}

$\Rightarrow\quad \ans{x = 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Vertical angles measure $(6x + 1)^\circ$ and $(4x + 15)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Segment arguments}

\begin{formulabox}
\textbf{Definition of midpoint:} $M$ on $\overline{AB}$ with $\overline{AM} \cong \overline{MB}$; then $AM = MB$ and $AB = 2\,AM$.\par
\textbf{Segment Addition Postulate:} $B$ between $A$ and $C$ gives $AB + BC = AC$.\par
\textbf{Overlapping segments:} add the shared middle stretch to both sides of a given equality, then rename each sum with the postulate. Subtracting it runs the same argument backwards.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $M$ is the midpoint of $\overline{AB}$ with $AM = 3x + 5$ cm and $MB = x + 17$ cm. Find $x$.
\step{The midpoint given produces congruent pieces, and congruent segments have equal lengths --- that is the step that turns the picture into an equation I can solve.}
\step{$3x + 5 = x + 17$, so $2x = 12$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $M$ is the midpoint of $\overline{AB}$ with $AM = 5x - 4$ cm and $MB = 2x + 11$ cm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Angle arguments with two links}

\begin{formulabox}
\textbf{Linear Pair Postulate:} adjacent angles whose outer sides form a line have measures adding to $180$.\par
\textbf{Vertical Angles Theorem:} the opposite pair formed by two intersecting lines are congruent.\par
\textbf{Definition of an angle bisector:} the whole angle is \emph{twice} either half.\par
Chaining two of these is the usual multi-step task: use the bisector to write the whole angle in terms of the marked half, then feed that into the linear-pair sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\overrightarrow{QS}$ bisects $\angle PQR$ with $m\angle PQS = (3x + 5)^\circ$, and $\angle PQR$ and $\angle RQT$ form a linear pair with $m\angle RQT = (4x)^\circ$. Find $x$.
\step{The bisector lets me write the whole angle as twice the marked half, and the linear pair then supplies one equation containing only $x$ --- two links, one unknown.}
\step{$2(3x + 5) + 4x = 180$, so $6x + 10 + 4x = 180$ and $10x = 170$.}

$\Rightarrow\quad \ans{x = 17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same picture with $m\angle PQS = (2x + 10)^\circ$ and $m\angle RQT = (6x)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 16}$}
\end{tryitbox}
