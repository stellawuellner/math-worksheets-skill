\sstitleblock{Factor Detective: Factors, Multiples, and Patterns}

\skillheading{1. Factor pairs, primes and composites}

\begin{formulabox}
A \textbf{factor pair} is two numbers that multiply to make the target. Hunt
for them in order --- $1$, $2$, $3$, $4 \ldots$ --- and stop when the two
numbers in a pair meet.\par
\textbf{Prime:} exactly two factors, $1$ and itself. \textbf{Composite:} more
than two. ($1$ is neither.)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Complete the factor pair $3 \times
\underline{\hspace{0.8cm}} = 18$, and say whether $18$ is prime.
\step{A factor pair asks how many equal groups fit exactly, so this is a
division question in disguise.}
\step{Count by threes: $3, 6, 9, 12, 15, 18$ --- six threes, so
$18 \div 3 = 6$.}

The partner factor is $\ans{6}$, and since $18$ has the extra pair
$3 \times 6$, it is composite.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Complete $7 \times \underline{\hspace{0.8cm}} = 56$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{2. Divisibility rules: test without dividing}

\begin{formulabox}
Divisible by \textbf{2} if the last digit is even; by \textbf{5} if it ends in
$0$ or $5$; by \textbf{10} if it ends in $0$.\par
Divisible by \textbf{3} if the \emph{sum of the digits} is divisible by $3$ ---
and by \textbf{9} if that digit sum is divisible by $9$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $258$ divisible by $3$?
\step{Dividing would work, but the digit-sum rule answers it in one line, so
add the digits first.}
\step{$2 + 5 + 8 = 15$, and $15 = 3 \times 5$ is divisible by $3$.}

The digit sum is $\ans{15}$, so $258$ \emph{is} divisible by $3$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add the digits of $471$ to test it for $3$.\\[2pt]
\rotatebox{180}{\footnotesize check: digit sum $\ans{12}$, so yes}
\end{tryitbox}

\skillheading{3. Multiples and the first common multiple}

\begin{formulabox}
The \textbf{multiples} of a number are its skip-count: $4, 8, 12, 16 \ldots$
Every number is a multiple of each of its factors.\par
To find the \textbf{first common multiple} of two numbers, skip-count both and
stop at the first number that appears in both lists.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the first common multiple of $4$ and $6$.
\step{Two skip-counts have to land on the same number, so list them side by
side and watch for the first match.}
\step{Fours: $4, 8, 12$. Sixes: $6, 12$. The first match is the third multiple
of $4$.}

$4 \times 3 = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the first common multiple of $3$ and $5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\skillheading{4. Number patterns: find the step, then jump}

\begin{formulabox}
Subtract two neighbours to find the \textbf{step}. Then remember that reaching
the $n$th number takes $n - 1$ steps, because the first number is already
there.\par
$n$th number $=$ first number $+$ (number of jumps) $\times$ step.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at $5$ and adds $6$ each time. What is
the $7$th number?
\step{Writing out seven terms works, but counting jumps is quicker and does not
drift --- so find how many jumps are needed first.}
\step{Reaching the $7$th number takes $6$ jumps: $5 + 6 \times 6 = 5 + 36$.}

The $7$th number is $\ans{41}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at $2$ and adds $9$ each time. Find the
$6$th number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{47}$}
\end{tryitbox}
