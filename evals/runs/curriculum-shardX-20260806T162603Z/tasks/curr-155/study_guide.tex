\sstitleblock{Selecting Ratio Representations for Comparisons}

% =================== SKILL 1 ===================
\skillheading{Comparing with Unit Rates}

\begin{formulabox}
\textbf{Rule:} When two offers or speeds use \textbf{different} amounts, divide
to get each \textbf{unit rate} (per one pound, per one hour), then compare the
unit rates directly. Choose this tool when the question says ``per'' or ``which
is cheaper/faster.''
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Store P sells 4 lb of apples for 10 dollars; store Q
sells 3 lb for 9 dollars. Which is cheaper per pound?
\step{Different amounts, so compare unit prices: $10 \div 4 = 2.5$ dollars per
pound and $9 \div 3 = 3$ dollars per pound.}
\step{Comparing the unit prices, store P costs less per pound:
$\ans{2.5 < 3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Store R sells 5 lb for 12 dollars; store S sells 4 lb for
10 dollars. Write the comparison of their price-per-pound fractions with $<$ or
$>$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{12}{5} < \tfrac{10}{4}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Scaling to a Common Amount}

\begin{formulabox}
\textbf{Rule:} A ratio table or double number line lets you scale two ratios to
the \textbf{same} amount (same time, same servings) and compare the other
quantity. Choose this tool when both ratios can reach a shared amount easily.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Ava walks 8 km every 2 hours; Ben walks 9 km every 3
hours. Who goes farther in 6 hours?
\step{Scale both to 6 hours: Ava $8 \times 3 = 24$ km, Ben $9 \times 2 = 18$
km.}
\step{Over the same 6 hours, Ava goes farther: $\ans{24 > 18}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Cara bikes 10 km every 2 hours; Dan bikes 12 km every 4
hours. Compare their km-per-hour fractions with $<$ or $>$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{10}{2} > \tfrac{12}{4}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Never compare raw numbers from different-sized batches
— 210 km ``beats'' 140 km until you notice the times differ. Scale to a common
amount (or a unit rate) first.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Splitting a Total with a Tape Diagram}

\begin{formulabox}
\textbf{Rule:} When a \textbf{total} is shared in a ratio, draw a tape: one box
per ratio part. Add the parts, divide the total by the number of boxes to value
one box, then multiply. Choose this tool when the problem gives a total (or a
difference).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A 45-marble bag is split in the ratio $2{:}7$. How many
marbles is the larger share?
\step{The tape has $2 + 7 = 9$ boxes for 45 marbles, so one box is
$45 \div 9 = 5$ marbles.}
\step{The larger share has 7 boxes: $7 \times 5 = 35$, so it is $\ans{35}$
marbles.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A 36-sticker sheet is split in the ratio $5{:}4$. How
many stickers is the larger share?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Unit rate:] the amount per one unit
  \item[Common amount:] a shared quantity two ratios are scaled to
  \item[Tape diagram:] equal boxes splitting a total by ratio parts
  \item[Representation:] the tool you choose to model a ratio
\end{description}
\end{multicols}
