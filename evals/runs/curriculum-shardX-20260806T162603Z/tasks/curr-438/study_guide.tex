\sstitleblock{Conditional and Independent-Event Probability}

\skillheading{Skill 1 --- Reading a probability off a two-way table}

\begin{formulabox}
\textbf{Rule:} every probability from a table is one count divided by another.
The question tells you which total goes on the bottom: ``chosen at random'' means the
grand total; ``chosen from the students who \dots'' means that row or column total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In a table of $90$ people, $24$ walk to work. Find $P(\text{walks})$.
\step{The person is drawn from everyone, so the denominator is the grand total, not a row total.}
\step{$P = \dfrac{24}{90}$, and dividing numerator and denominator by $6$ reduces it.}

$\Rightarrow\quad P(\text{walks}) = \ans{\tfrac{4}{15}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $50$ surveyed shoppers, $15$ used a coupon. Find $P(\text{coupon})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $P = \ans{\tfrac{3}{10}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Conditional probability shrinks the pool}

\begin{formulabox}
\textbf{Rule:}\quad $P(A \mid B) = \dfrac{P(A \text{ and } B)}{P(B)}$, which on a table is simply
\[ P(A \mid B) = \frac{\text{count in both } A \text{ and } B}{\text{total count in } B}. \]
The condition after the bar replaces the whole group as the new denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Of $36$ people who drive, $24$ also cycle. Find $P(\text{cycles} \mid \text{drives})$.
\step{The condition ``drives'' names the new pool, so the denominator is the count of drivers, not everyone.}
\step{$P = \dfrac{24}{36}$, which reduces by $12$.}

$\Rightarrow\quad P(\text{cycles} \mid \text{drives}) = \ans{\tfrac{2}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $25$ students who take Latin, $15$ also take music. Find $P(\text{music} \mid \text{Latin})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $P = \ans{\tfrac{3}{5}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Testing whether two events are independent}

\begin{formulabox}
\textbf{Test:} $A$ and $B$ are independent exactly when
$P(A \text{ and } B) = P(A) \cdot P(B)$ --- equivalently when $P(A \mid B) = P(A)$.
Compute both sides and compare. Independence is a numerical fact about the data, never an
assumption you may make because the two things ``feel unrelated''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Of $100$ people, $50$ own a bike and $40$ own a car. Compute the product $P(\text{bike}) \cdot P(\text{car})$.
\step{The product is what independence would predict for the joint probability, so build it first and compare it with the real joint count afterwards.}
\step{$\dfrac{50}{100} \cdot \dfrac{40}{100} = \dfrac{1}{2} \cdot \dfrac{2}{5}$.}

$\Rightarrow\quad P(\text{bike}) \cdot P(\text{car}) = \ans{\tfrac{1}{5}}$, so independence predicts $20$ people owning both.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $60$ people, $30$ read fiction and $20$ read poetry. Find the product of the two probabilities.\\[2pt]
\rotatebox{180}{\footnotesize check: product $= \ans{\tfrac{1}{6}}$}
\end{tryitbox}

\skillheading{Skill 4 --- The multiplication rule}

\begin{formulabox}
\textbf{Rule:}\quad $P(A \text{ and } B) = P(A) \cdot P(B \mid A)$. If the first outcome does not
change the second, $P(B \mid A) = P(B)$ and the rule collapses to a plain product. Without
replacement the pool shrinks, so both the numerator and the denominator of the second factor
drop by one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A bag holds $10$ marbles, $4$ of them red. Two are drawn without replacement. Find $P(\text{both red})$.
\step{The draws are not independent --- taking a red marble out changes the second draw --- so use the conditional form of the rule.}
\step{$P = \dfrac{4}{10} \cdot \dfrac{3}{9}$, since one red and one marble overall have been removed.}

$\Rightarrow\quad P(\text{both red}) = \ans{\tfrac{2}{15}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A box holds $12$ pens, $5$ of them blue. Two are drawn without replacement. Find $P(\text{both blue})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $P = \ans{\tfrac{5}{33}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $P(A \mid B)$ and $P(B \mid A)$ are different numbers. A test that
catches $90\%$ of a rare condition does not mean a positive result is $90\%$ likely to be real
--- the rarity of the condition is in the other denominator.
\end{watchoutbox}
