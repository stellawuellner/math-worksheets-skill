\sstitleblock{Comparison, Integral, Ratio, and Alternating-Series Tests}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Limit comparison}

\begin{formulabox}
\textbf{Use it for} terms that are a ratio of polynomials (or behave like one). Keep only the leading power on top and bottom to pick the comparison series $b_n$, then compute $L = \lim a_n/b_n$. If $L$ is finite \emph{and positive}, the two series live or die together.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum \frac{2n+1}{n^{3}+7}$ against $\displaystyle\sum\frac{1}{n^{2}}$.
\step{The terms are a ratio of polynomials, so limit comparison is the tool; leading powers give $n/n^3 = 1/n^2$ as the partner.}
\step{$\dfrac{(2n+1)/(n^{3}+7)}{1/n^{2}} = \dfrac{2n^{3}+n^{2}}{n^{3}+7}$, whose limit is finite and positive.}

$\Rightarrow\quad L = \ans{2}$, and since $\sum 1/n^2$ converges, so does the series
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $L$ for $\displaystyle\sum \frac{5n-2}{n^{3}+1}$ against $\displaystyle\sum\frac{1}{n^{2}}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$, so it converges}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Integral test}

\begin{formulabox}
\textbf{Use it when} $a_n = f(n)$ for a function you can integrate. Check three things first: $f$ positive, continuous, and decreasing on $[N,\infty)$. Then $\sum a_n$ and $\int_N^{\infty} f$ converge together. The integral is \emph{not} the sum --- it only decides the question.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}}$ with $f(x) = 1/x^{2}$.
\step{The terms come from a function that is easy to integrate and is positive, continuous and decreasing past $x = 1$, so the integral test applies.}
\step{$\int_1^{\infty} x^{-2}dx = \left[-x^{-1}\right]_1^{\infty} = 0 + 1$, a finite value.}

$\Rightarrow\quad \ans{1}$, so the series converges
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_1^{\infty}\frac{dx}{x^{3}}$ and say what it decides.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{1}{2}}$, so it converges}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Ratio test}

\begin{formulabox}
\textbf{Use it for} factorials and $n$th powers. Compute $L = \lim |a_{n+1}/a_{n}|$. Then $L < 1$ means convergence, $L > 1$ (including infinite) means divergence, and $L = 1$ means the test is silent --- which is exactly what happens on rational terms, so do not start there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $\displaystyle\sum_{n=0}^{\infty}\frac{3^{n}}{n!}$.
\step{A factorial in the denominator makes consecutive terms easy to divide, so the ratio test is the natural first move.}
\step{$\dfrac{a_{n+1}}{a_{n}} = \dfrac{3^{n+1}}{(n+1)!}\cdot\dfrac{n!}{3^{n}} = \dfrac{3}{n+1}$, and that shrinks to nothing.}

$\Rightarrow\quad L = \ans{0} < 1$, so the series converges
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\displaystyle\sum\frac{n!}{2^{n}}$ the ratio is $\frac{n+1}{2}$. Find $L$ and say what it decides.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\infty}$, so it diverges}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Alternating series and its error bound}

\begin{formulabox}
\textbf{Use it when} the signs alternate. Two conditions: the term sizes decrease, and their limit is $0$. Then the series converges, and the error after $N$ terms is at most the size of the \emph{first term left out} --- so the bound is $a_{N+1}$, read straight off the formula.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{2}}$, bound the error after $20$ terms.
\step{Signs alternate and the sizes $1/n^2$ decrease to zero, so the remainder bound applies: find the first omitted term.}
\step{Twenty terms run to $n = 20$, so the first omitted term is the one with $n = 21$, of size $1/21^{2}$.}

$\Rightarrow\quad \ans{\frac{1}{441}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{3}}$, bound the error after $10$ terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{1}{1331}}$}
\end{tryitbox}
