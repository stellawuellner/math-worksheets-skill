\wstitleblock{Conditional and Independent-Event Probabilities}{Precalculus / Advanced Statistics --- Discrete Mathematics}{}

\noindent\small Conditional probability from a two-way table, the multiplication rule, testing independence, and independent trials in a binomial model

\medskip
\noindent\textbf{Rules used on this sheet:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item $P(A \mid B) = \dfrac{P(A \cap B)}{P(B)}$ --- in a table, restrict to $B$'s row or column and divide within it.
  \item Multiplication rule: $P(A \cap B) = P(B) \cdot P(A \mid B)$. When $A$ and $B$ are \textbf{independent}, $P(A \mid B) = P(A)$ and this becomes $P(A) \cdot P(B)$.
  \item $A$ and $B$ are independent exactly when $P(A \cap B) = P(A) \cdot P(B)$.
  \item Give exact fractions in lowest terms unless a decimal is requested.
\end{itemize}

\medskip
\noindent\textbf{Survey table --- used by problems 1, 2, 3 and 6.} All 200 students at a
school were asked whether they ride the bus and whether they play a school sport.
\begin{center}
\begin{tabular}{|l|c|c|c|}\hline
 & \textbf{Plays a sport} & \textbf{No school sport} & \textbf{Total} \\\hline
\textbf{Rides the bus} & 48 & 32 & 80 \\\hline
\textbf{Does not ride the bus} & 72 & 48 & 120 \\\hline
\textbf{Total} & 120 & 80 & 200 \\\hline
\end{tabular}
\end{center}
\normalsize

\problem[5cm]{A student is chosen at random from the 200 surveyed.
Given that the student rides the bus, find the probability that the student plays a
school sport. Write the conditional probability you are computing in
$P(A \mid B)$ notation before you divide.}

\problem[5cm]{Using the same survey, find the probability that a student rides the bus,
given that the student plays a school sport.
Explain in one sentence why this answer differs from your answer to problem 1 even
though both use the 48 students in the same cell.}

\problem[1cm]{Still using the survey table, test whether ``plays a school sport'' and
``rides the bus'' are independent events.
\begin{enumerate}[label=(\alph*), itemsep=2.6cm, leftmargin=1.5cm, topsep=4pt]
  \item Find $P(\text{plays a sport})$ for a randomly chosen student. \ansline
  \item Find $P(\text{plays a sport and rides the bus})$. \ansline
  \item Multiply $P(\text{plays a sport}) = 0.6$ by $P(\text{rides the bus}) = 0.4$, compare with your answer to (b), and state whether the two events are independent. \ansline
\end{enumerate}}

\problem[5.5cm]{On a production line each memory chip passes inspection with probability
$0.98$, and the chips are tested independently of one another.
Two chips are selected. Find the probability that \emph{both} pass inspection.
Give your answer as a decimal.}

\problem[6cm]{A drawer holds $12$ identical pens, of which $5$ write in red.
Two pens are drawn at random \textbf{without replacement}.
Find the probability that both pens write in red.
Show the two factors you multiplied, and state the conditional probability that the
second factor represents.}

\problem[5.5cm]{Return to the survey table.
Given that a student does \emph{not} play a school sport, find the probability that the
student rides the bus.
State the total you divided by and why that total, not 200, is the correct denominator.}

\problem[1cm]{For two events $A$ and $B$ it is known that
$P(A) = 0.4$, $P(B) = 0.5$ and $P(A \cap B) = 0.25$.
\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=4pt]
  \item Compute the product $P(A) \cdot P(B)$. \ansline
  \item Compute $P(A \mid B)$, then state whether $A$ and $B$ are independent and justify the decision with your two numbers. \ansline
\end{enumerate}}

\problem[1cm]{A factory gets $60\%$ of its parts from line 1 and $40\%$ from line 2.
Of the parts from line 1, $2\%$ are defective; of the parts from line 2, $5\%$ are
defective. A part is chosen at random.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the probability that the part came from line 1 \emph{and} is defective. \ansline
  \item Find the probability that the part is defective. \ansline
\end{enumerate}}

\problem[6cm]{A basketball player takes $3$ free throws. The throws are independent;
write $a$ for the probability of making one throw and $b$ for the probability of missing
one, so $a + b = 1$.
Expand $(a + b)^3$ completely, then state which term gives the probability of making
exactly two of the three throws, and explain why that term carries a coefficient
greater than $1$.}

\problem[1cm]{The same player makes each free throw with probability $0.8$, and the
three throws are independent.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the probability that she makes exactly two of the three throws. Give a decimal. \ansline
  \item Explain in two or three complete sentences why multiplying $0.8 \times 0.8 \times 0.2$ is valid only when the throws are independent, and what would have to change in the calculation if they were not. \noansline
\end{enumerate}}
