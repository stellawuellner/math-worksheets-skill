% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Two Cube Patterns}

\begin{formulabox}
\textbf{Difference:} $a^{3} - b^{3} = (a - b)(a^{2} + ab + b^{2})$\\[3pt]
\textbf{Sum:} $a^{3} + b^{3} = (a + b)(a^{2} - ab + b^{2})$\\[3pt]
The binomial keeps the sign you started with; inside the trinomial the middle
sign is the opposite one, and the last sign is always a plus. The trinomial
never factors again.\\[3pt]
\textbf{Cubes to know:} $1, 8, 27, 64, 125$ --- and $8x^{3}$ is $(2x)^{3}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^{3} + 8$
\step{Two terms and both are perfect cubes, which is the signal for the sum-of-cubes pattern --- there is no common factor to pull out first.}
\step{With $a = x$ and $b = 2$: the binomial is $x + 2$, and the trinomial is $x^{2} - 2x + 4$ because the middle sign flips and $b^{2} = 4$.}

$\Rightarrow\quad \ans{(x+2)(x^2-2x+4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $125x^{3} - 27$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(5x-3)(25x^2+15x+9)}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Four Terms: Factor by Grouping}

\begin{formulabox}
\textbf{Method:} split the four terms into two pairs, pull the common factor out
of each pair, and the same bracket should be left over both times. That shared
bracket is one factor; what you pulled out makes the other.\\[3pt]
\textbf{Then look again:} the second factor is often a difference of squares, so
"factor completely" usually means one more step.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^{3} - 3x^{2} - 4x + 12$
\step{Four terms with no common factor across all of them, so try grouping in pairs and watch for a shared bracket.}
\step{$x^{2}(x - 3) - 4(x - 3)$ leaves the shared bracket $x - 3$, giving $(x - 3)(x^{2} - 4)$, and $x^{2} - 4$ is a difference of squares.}

$\Rightarrow\quad \ans{(x-3)(x-2)(x+2)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $2x^{3} + x^{2} - 18x - 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2x+1)(x-3)(x+3)}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- A Quadratic in Disguise}

\begin{formulabox}
\textbf{Shape:} an $x^{4}$ term, an $x^{2}$ term and a constant. Factor it as if
$x^{2}$ were a single letter, then break each resulting factor down if you
can.\\[3pt]
A factor like $x^{2} + 4$ is a \textit{sum} of squares and does not factor over
the real numbers --- leaving it whole is the complete answer, not an unfinished
one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^{4} - 10x^{2} + 9$
\step{The three terms are $x^{4}$, $x^{2}$ and a constant, so this factors like a quadratic with $x^{2}$ playing the role of the variable.}
\step{Product $9$ and sum $-10$ come from $-1$ and $-9$, giving $(x^{2} - 1)(x^{2} - 9)$; each of those is a difference of squares.}

$\Rightarrow\quad \ans{(x-1)(x+1)(x-3)(x+3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $x^{4} - 5x^{2} - 36$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-3)(x+3)(x^2+4)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} pull out a common factor \textit{first} --- $2x^{3} - 16$
is not a difference of cubes until the $2$ comes out. And with six powers, such
as $x^{6} - 64$, split it as a difference of \textit{squares} first; going
straight to cubes stops one factor short.
\end{watchoutbox}
