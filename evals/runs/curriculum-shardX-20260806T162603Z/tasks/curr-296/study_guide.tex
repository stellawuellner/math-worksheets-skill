\sstitleblock{Evaluating Piecewise and Absolute-Value Functions}

\skillheading{Skill 1 --- Absolute value is a distance}

\begin{formulabox}
\textbf{Meaning:} \quad $|a|$ is how far $a$ is from zero, so $|{-7}| = 7$ and $|7| = 7$. It is never negative.\par\smallskip
\textbf{Order:} finish everything \emph{inside} the bars, take the distance, then do what is outside. A minus sign in front of the bars is outside, so it acts last --- and it can make the whole value negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = |x-5| + 1$, find $f(2)$.
\step{This is a substitution, so the only decision is the order --- inside the bars first, because the bars group like brackets.}
\step{$2 - 5 = -3$; the distance of $-3$ from zero is $3$; then add the $1$ outside.}
\par
$f(2) = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = -3|x+2| + 10$, find $g(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $g(1) = \ans{1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $-3|x+2|$ does \emph{not} mean $|-3(x+2)|$. The bars close before the $-3$ is applied, so the result of the bars is positive and the $-3$ then makes it negative.
\end{watchoutbox}

\skillheading{Skill 2 --- Pick the rule before you compute}

\begin{formulabox}
A piecewise function is several rules, each with a condition. \textbf{Test the input against the conditions first}, then use only the rule it satisfies.\par\smallskip
$<$ excludes the boundary and $\le$ includes it, so exactly one rule ever owns a boundary value --- and it is the one written with $\le$ or $\ge$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 2x+3$ when $x < 1$, and $x^2+1$ when $x \ge 1$, find $f(0)$ and $f(3)$.
\step{Nothing can be computed until each input is matched to a condition, so read the conditions before looking at the rules.}
\step{$0 < 1$, so use $2x+3$: $2(0)+3$. And $3 \ge 1$, so use $x^2+1$: $3^2+1$.}
\par
$f(0) = \ans{3}$, \quad $f(3) = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For the same $f$, find $f(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $f(1) = \ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Working backwards to the input}

\begin{formulabox}
Given the output, find the inputs. Undo the operations \emph{outside} the bars first, until the absolute value stands alone; then split into two equations, because two numbers sit at the same distance from zero.\par\smallskip
$|A| = k$ (with $k > 0$) becomes $A = k$ \ \emph{or} \ $A = -k$. Solving only the first loses half the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $|x-4| = 6$.
\step{The bars are already alone, so the job is the split --- distance $6$ can be reached from either side, so expect two answers.}
\step{$x - 4 = 6$ gives $x = 10$; \ $x - 4 = -6$ gives $x = -2$.}
\par
$x = \ans{-2}$ \ or \ $x = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2|x+3| = 8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-7}$ or $x = \ans{1}$}
\end{tryitbox}
