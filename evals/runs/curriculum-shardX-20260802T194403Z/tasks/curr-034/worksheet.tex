\wstitleblock{Teen Numbers --- Fixing Swapped Tens and Ones}{Kindergarten--Grade 1}{}

\noindent\small Every teen number is made of \textbf{1 ten} and some \textbf{ones}.
The \textbf{tens} digit is written \textbf{first}. Some children below flipped their
digits. Read the blocks, then write the number the blocks really show.\normalsize

\problem[3cm]{Count the dots. One ten-frame is full, so it is \textbf{1 ten}.
The loose dots are the \textbf{ones}.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\end{tabular}\qquad
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & ~ \\\hline
~ & ~ & ~ & ~ & ~ \\\hline
\end{tabular}
\end{center}
There is \textbf{1 ten} and there are \textbf{4 ones}. Write the number.}

\problem[3cm]{Leo builds \textbf{1 ten}-stick and \textbf{6 ones}.
Leo writes \textbf{61}.\\[4pt]
Leo put his digits in the wrong places. Write the number Leo's blocks really show.}

\problem[3cm]{Nina is writing the number \textbf{eighteen}.
Eighteen is \textbf{1 ten} and \textbf{8 ones}. Nina writes \textbf{81}.\\[4pt]
Circle the digit Nina put in the wrong place. Then write eighteen the correct way.}

\problem[3cm]{Sam builds \textbf{1 ten} and \textbf{3 ones}.
Ty builds \textbf{3 tens} and \textbf{1 one}.\\[4pt]
Build or draw both. Then write $<$ or $>$ in the blank:\\[8pt]
\hspace*{1cm}\Large Sam's number \ansblank Ty's number}

\problem[3cm]{Ana counts these dots and writes \textbf{71}.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\end{tabular}\qquad
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
$\bullet$ & $\bullet$ & ~ & ~ & ~ \\\hline
\end{tabular}
\end{center}
Ana counted \textbf{1 ten} and \textbf{7 ones}. Tell why \textbf{71} is wrong,
then write the number the dots really show.}

\problem[3.5cm]{Jo means to write \textbf{1 ten} and \textbf{9 ones}.
Instead she swaps the two digits.\\[4pt]
Write $>$ or $<$ in the blank, then say which digit sits in the tens place in
Jo's swapped number:\\[8pt]
\hspace*{1cm}\Large Jo's swapped number \ansblank the number Jo meant}
