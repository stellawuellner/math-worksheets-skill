% =================== SKILL 1 ===================
\skillheading{Skill 1 --- From Dots to a Numeral}

\begin{formulabox}
\textbf{Rule:} Count the objects one at a time. The last number you say is
\emph{how many}, and the \textbf{numeral} is that number written with digits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Count: $\bullet\ \bullet\ \bullet\ \bullet\ \bullet$
--- write the numeral.
\step{Count the dots one-to-one, then write the last number said as a numeral.}
\step{``One, two, three, four, five'' --- the count ends at five, so the numeral is $\ans{5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Count: $\bullet\ \bullet\ \bullet\ \bullet$ --- write
the numeral.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Number Words}

\begin{formulabox}
\textbf{Rule:} Every numeral has a name you can spell:\par
\begin{center}\small
\begin{tabular}{ccccc}
\emph{one} 1 & \emph{two} 2 & \emph{three} 3 & \emph{four} 4 & \emph{five} 5\\
\emph{six} 6 & \emph{seven} 7 & \emph{eight} 8 & \emph{nine} 9 & \emph{ten} 10\\
\end{tabular}\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The card says \emph{eight}. Write the numeral.
\step{A number word and a numeral are two ways to write the same number --- find \emph{eight} in the counting sequence.}
\step{\emph{Eight} comes after seven and before nine, so the numeral is $\ans{8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The card says \emph{three}. Write the numeral.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} \emph{Six} and \emph{seven} start with the same sound.
Check the whole word before you write the numeral.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From a Numeral to a Quantity}

\begin{formulabox}
\textbf{Rule:} A numeral tells you \emph{how many to show}. A ten-frame helps:
fill the top row of five first, then keep going in the bottom row.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Show the numeral 6 as dots on a ten-frame.
\step{Fill the top row of five first, then count on until the count reaches the numeral.}
\step{Top row: five dots. One more makes $5 + 1 = \ans{6}$ --- six dots in all.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad To show the numeral 9, you fill the top row of five.
How many dots go in the bottom row?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}
