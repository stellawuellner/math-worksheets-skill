\wstitleblock{Composing Piecewise and Nonlinear Functions}{Precalculus \quad Advanced Transformations and Composition}{}

\begin{center}
\textit{Evaluating a composition, simplifying one into a single rule, choosing
the right piece of a piecewise function, and solving composition equations}
\end{center}

\noindent\textbf{The functions used throughout this worksheet.}
\begin{center}
$f(x) = x^2 - 3$ \qquad $g(x) = 2x + 1$ \qquad $h(x) = \sqrt{x-4}$
\\[0.25cm]
$p(x) = \begin{cases} x^2, & x < 2 \\ 5 - x, & x \ge 2 \end{cases}$
\qquad\qquad
$q(x) = \begin{cases} -x, & x < 0 \\ x^2 + 1, & x \ge 0 \end{cases}$
\end{center}

\noindent\small\textbf{Directions.} $(f \circ g)(x)$ means $f(g(x))$: the
\emph{inner} function acts first. With a piecewise outer function, choose the
piece using the \textbf{inner function's output}, not the original input.
Simplify exactly --- keep radicals and fractions --- and write the requested
value or rule on the answer line.\normalsize

\problem[5cm]{Find $(f \circ g)(2)$.

\smallskip
Show the inner value first, then substitute it into the outer function.}

\problem[5cm]{Find a single simplified rule for $(f \circ g)(x) = f(g(x))$.

\smallskip
Expand fully and collect like terms.}

\problem[5.5cm]{Find $p(g(-2))$.

\smallskip
\textbf{(a)} Compute the inner value $g(-2)$. \textbf{(b)} State which piece of
$p$ that value selects, and why. \textbf{(c)} Give $p(g(-2))$.}

\problem[6cm]{Solve $(f \circ g)(x) = 13$ for $x$.

\smallskip
Use your simplified rule from problem 2, set it equal to $13$, and solve the
resulting quadratic exactly. List every real solution on the answer line.}

\problem[5cm]{Find $(g \circ f)(3)$.

\smallskip
Compare your answer with problem 1: the same two functions in the other order.
Say in one line what that comparison shows.}

\problem[5cm]{Find a single simplified rule for $(g \circ f)(x) = g(f(x))$.

\smallskip
Then state how this rule differs from the rule you found in problem 2, and what
that says about whether composition is commutative.}

\problem[6cm]{Consider $(p \circ g)(x) = p(2x+1)$.

\smallskip
\textbf{(a)} The rule for $p$ changes when its input reaches $2$. Solve
$2x + 1 = 2$ to find the value of $x$ where the composition switches pieces, and
write that value on the answer line.

\smallskip
\textbf{(b)} Using that cut point, write $(p \circ g)(x)$ as a piecewise rule in
$x$, with the correct inequality on each piece.}

\problem[5.5cm]{Find $(q \circ f)(1)$.

\smallskip
\textbf{(a)} Compute $f(1)$. \textbf{(b)} Explain which piece of $q$ that value
selects --- be careful, the test compares the \emph{inner output} with $0$, not
$x$ with $0$. \textbf{(c)} Give the value of the composition.}

\problem[6cm]{Solve $(g \circ f)(x) = 45$ for $x$.

\smallskip
Use your rule from problem 6. List every real solution on the answer line, and
state in one line why this equation has two solutions while a linear composition
would have one.}

\problem[6cm]{Find a single simplified rule for $(f \circ h)(x) = f(h(x))$.

\smallskip
\textbf{(a)} Simplify the rule as far as possible and write it on the answer
line. \textbf{(b)} The simplified rule would make sense for every real $x$, but
the composition does not. State the domain of $(f \circ h)$ and explain what the
simplification threw away.}

\problem[6cm]{Consider $(h \circ f)(x) = \sqrt{x^2 - 7}$.

\smallskip
\textbf{(a)} A square root needs a non-negative radicand. Solve $x^2 - 7 = 0$
and write both boundary values on the answer line.

\smallskip
\textbf{(b)} Use those boundaries to write the domain of $(h \circ f)$ in
interval notation, and say why the domain comes in two pieces.}

\problem[7cm]{Solve $(q \circ g)(x) = 5$ for $x$, where $q$ is piecewise.

\smallskip
\textbf{(a)} \emph{First piece.} $q$ uses $-u$ when its input $u = 2x+1$ is
negative. Solve $-(2x+1) = 5$, then check whether your solution really does make
$2x+1$ negative.

\smallskip
\textbf{(b)} \emph{Second piece.} $q$ uses $u^2 + 1$ when $u = 2x+1 \ge 0$.
Solve $(2x+1)^2 + 1 = 5$, then test each candidate the same way.

\smallskip
\textbf{(c)} Write the complete solution set on the answer line, and explain why
one candidate had to be thrown out.}
