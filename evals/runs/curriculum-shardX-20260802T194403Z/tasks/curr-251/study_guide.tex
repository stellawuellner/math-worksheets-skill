\skillheading{Skill 1 --- Read an area model as terms}
\begin{formulabox}\textbf{Model $\rightarrow$ notation:}\\[2pt]
Side lengths are \textbf{factors}; each cell inside is one \textbf{term} of
the product. Cells with the same power of $x$ are like terms and combine, so a
four-cell model usually gives three terms.\par
$(x + p)(x + q) = x^2 + (p + q)x + pq$
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Expand $(x + 2)(x + 4)$ with a model.
\step{Each side is a factor, so I multiply every part of one side by every
part of the other and read off the cells.}
\step{Cells: $x^2$, $4x$, $2x$, $8$. The middle two are like terms
($4x + 2x = 6x$): $\ans{x^2 + 6x + 8}$}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Expand $(x + 1)(x + 7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 8x + 7}$}\end{tryitbox}

\skillheading{Skill 2 --- Name the factors of a polynomial}
\begin{formulabox}\textbf{Rule:}\quad Factoring rewrites a \emph{sum of terms}
as a \emph{product of factors} --- the same value, a different form.\par
Always look for a common factor first: every term of $ax^m + bx^n$ shares the
greatest common numerical factor and the lowest power of $x$. Degrees of
factors \textbf{add} to the degree of the product.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Factor $x^2 + 9x + 20$.
\step{The terms share no common factor, so I look for two numbers whose
product is the constant term and whose sum is the middle coefficient.}
\step{$4 \cdot 5 = 20$ and $4 + 5 = 9$, so $\ans{(x + 4)(x + 5)}$ ---
two factors of degree $1$, product of degree $2$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Factor $4x^2 - 6x$ completely (pull out
the greatest common factor).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x(2x - 3)}$}\end{tryitbox}

\skillheading{Skill 3 --- Coefficient and degree}
\begin{formulabox}\textbf{Vocabulary on one term, $-7x^3$:}\quad coefficient
$-7$ (the sign belongs to it), variable $x$, degree $3$ (the exponent).\par
\textbf{Polynomial:} its \textbf{degree} is the largest term degree; its
\textbf{leading coefficient} is the coefficient of that highest-degree term
--- found by writing the polynomial in standard form (degrees decreasing),
not by reading the first term as typed. A constant has degree $0$; $x$ alone
has coefficient $1$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Put $5 - 2x^2 + 7x^3$ in standard
form; give its degree and leading coefficient.
\step{Leading coefficient is defined from standard form, so I reorder the
terms by degree before reading anything off.}
\step{Degrees are $0$, $2$, $3$, so descending order gives
$\ans{7x^3 - 2x^2 + 5}$ --- degree $3$, leading coefficient $7$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $3x - 8 + x^4$ in standard form.
What is its leading coefficient?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^4 + 3x - 8}$, leading coefficient $1$}\end{tryitbox}

\skillheading{Skill 4 --- Same expression, two forms (equivalent-forms)}
\begin{formulabox}\textbf{Rule:}\quad Two expressions are \textbf{equivalent}
when they give the same value for \emph{every} $x$ --- proved by algebra
(distributing or factoring), never by checking a few inputs in a table.\par
Counting terms does not decide it: $3(2x + 5)$ is one term made of two
factors, and $6x + 15$ is two terms; they are the same expression written
two ways.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Show $2(3x - 4)$ as a sum of terms.
\step{The expression is a product, so distributing the outside factor turns it
into a sum without changing its value.}
\step{$2 \cdot 3x = 6x$ and $2 \cdot (-4) = -8$, giving $\ans{6x - 8}$ ---
one term became two, the value did not change.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Expand $(x + 5)(x - 5)$. How many terms
does the answer have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 - 25}$, two terms}\end{tryitbox}
