\problem{Evaluate: \quad $-9 + 4$}

\textbf{Solution:} The signs differ, so subtract the absolute values ($9 - 4 = 5$) and
keep the sign of the number farther from zero, $-9$. \ans{-5}

\problem{Evaluate: \quad $4 - 11$}

\textbf{Solution:} Rewrite as adding the opposite: $4 - 11 = 4 + (-11)$. Different
signs: $11 - 4 = 7$, and $-11$ is farther from zero, so the result is negative. \ans{-7}

\problem{Evaluate: \quad $(-4)(-7)$}

\textbf{Solution:} A negative times a negative is positive: $4 \cdot 7 = 28$. \ans{28}

\problem{Evaluate: \quad $-\tfrac{2}{5} + \left(-\tfrac{1}{5}\right)$}

\textbf{Solution:} Same denominators and same signs: add the absolute values,
$\tfrac{2}{5} + \tfrac{1}{5} = \tfrac{3}{5}$, and keep the negative sign. \ans{-\tfrac{3}{5}}

\problem{Evaluate: \quad $-7 - (-9)$}

\textbf{Solution:} Subtracting $-9$ is adding $9$: $-7 - (-9) = -7 + 9 = 2$. \ans{2}

\problem{Evaluate, writing the answer as a fraction in simplest form: \quad $-36 \div 8$}

\textbf{Solution:} Different signs make the quotient negative:
$\tfrac{36}{8} = \tfrac{9}{2}$ after dividing top and bottom by 4, so
$-36 \div 8 = -\tfrac{9}{2}$. \ans{-\tfrac{9}{2}}

\problem{Evaluate: \quad $-\tfrac{5}{6} + \tfrac{1}{4}$}

\textbf{Solution:} Use the common denominator 12: $-\tfrac{5}{6} = -\tfrac{10}{12}$ and
$\tfrac{1}{4} = \tfrac{3}{12}$. Different signs: $\tfrac{10}{12} - \tfrac{3}{12} = \tfrac{7}{12}$,
negative because $-\tfrac{10}{12}$ is farther from zero. \ans{-\tfrac{7}{12}}

\problem{Evaluate: \quad $1.5 - (-2.25)$}

\textbf{Solution:} Subtracting $-2.25$ is adding $2.25$: $1.5 + 2.25 = 3.75$. \ans{3.75}

\problem{Evaluate: \quad $\left(-\tfrac{2}{3}\right)\!\left(\tfrac{9}{8}\right)$}

\textbf{Solution:} Multiply numerators and denominators:
$\tfrac{2 \cdot 9}{3 \cdot 8} = \tfrac{18}{24} = \tfrac{3}{4}$. One factor is negative,
so the product is negative. \ans{-\tfrac{3}{4}}

\problem{Write $<$ or $>$ in the blank: \ $-\tfrac{7}{8} \;\square\; -0.9$}

\textbf{Solution:} Convert to the same form: $-\tfrac{7}{8} = -0.875$. On the number
line $-0.875$ sits to the \emph{right} of $-0.9$ (closer to zero), so
\ans{-\tfrac{7}{8} \;>\; -0.9}

\problem{Evaluate: \quad $-2\left(\tfrac{3}{4} - \tfrac{5}{4}\right)$}

\textbf{Solution:} Parentheses first: $\tfrac{3}{4} - \tfrac{5}{4} = -\tfrac{2}{4} = -\tfrac{1}{2}$.
Then a negative times a negative: $-2 \cdot \left(-\tfrac{1}{2}\right) = 1$. \ans{1}

\problem{\textbf{Challenge.} Evaluate: \quad
$\left(-\tfrac{3}{4} + \tfrac{1}{2}\right) \div \left(-\tfrac{5}{8}\right)$}

\textbf{Solution:} Parentheses first: $-\tfrac{3}{4} + \tfrac{1}{2} = -\tfrac{3}{4} + \tfrac{2}{4} = -\tfrac{1}{4}$.
Dividing by $-\tfrac{5}{8}$ means multiplying by its reciprocal $-\tfrac{8}{5}$:
\[ -\tfrac{1}{4} \cdot \left(-\tfrac{8}{5}\right) = \tfrac{8}{20} = \tfrac{2}{5}. \]
Two negatives give a positive answer. \ans{\tfrac{2}{5}}
