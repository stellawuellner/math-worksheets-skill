\wstitleblock{Unit Rates from Fractions and Mixed Numbers}{Grades 6--7 Mathematics}{}

\begin{center}
{\small Rates from fractions, rates with mixed numbers, comparing unit rates, and using a unit rate to solve}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item A unit rate is one quantity divided by the other: \ $\text{rate} = \dfrac{\text{amount}}{\text{time or size}}$. When both are fractions, dividing means multiplying by the reciprocal.
  \item Change every mixed number to an improper fraction \emph{before} dividing.
  \item Leave answers as exact fractions or mixed numbers in lowest terms --- do not round.
\end{itemize}

\vspace{0.2cm}

\problem[3.4cm]{A runner covers $\dfrac{3}{4}$ of a mile in $\dfrac{1}{2}$ of an hour. What is the runner's speed in miles per hour?
\answerline{mph}}

\problem[3.4cm]{A cyclist rides $\dfrac{5}{8}$ of a kilometre in $\dfrac{1}{4}$ of an hour. What is the cyclist's speed in kilometres per hour?
\answerline{km/h}}

\problem[3.4cm]{A toy car travels $\dfrac{7}{8}$ of a metre in $\dfrac{1}{4}$ of a second. What is its speed in metres per second?
\answerline{m/s}}

\problem[3.4cm]{A cart rolls $2\dfrac{1}{2}$ feet in $\dfrac{3}{4}$ of a second. What is its speed in feet per second?
\answerline{ft/s}}

\problem[3.4cm]{A printer produces $3\dfrac{1}{3}$ pages in $\dfrac{2}{5}$ of a minute. How many pages does it produce per minute?}

\problem[3.8cm]{One runner covers $\dfrac{7}{8}$ of a mile in $\dfrac{3}{4}$ of an hour. A second runner covers $\dfrac{5}{6}$ of a mile in $\dfrac{2}{3}$ of an hour.
\begin{itemize}
  \item[(a)] Speed of the first runner: \answerline{mph}
  \item[(b)] Speed of the second runner: \answerline{mph}
  \item[(c)] Write $<$ or $>$ between the first runner's speed and the second runner's speed. \ansline
\end{itemize}}

\problem[3.6cm]{A pump fills $\dfrac{5}{6}$ of a tank in $\dfrac{1}{3}$ of an hour, working at a constant rate. How many hours does it take to fill one whole tank?
\answerline{hours}}

\problem[3.6cm]{A car uses $\dfrac{2}{5}$ of a gallon of fuel to travel $9\dfrac{1}{3}$ miles. How many miles does it travel per gallon?}

\problem[3.8cm]{One shop sells $\dfrac{3}{4}$ of a kilogram of cashews for 6 dollars. Another shop sells $\dfrac{2}{3}$ of a kilogram for 5 dollars.
\begin{itemize}
  \item[(a)] Price per kilogram at the first shop, in dollars: \ansline
  \item[(b)] Price per kilogram at the second shop, in dollars: \ansline
  \item[(c)] Write $<$ or $>$ between the first shop's price and the second shop's price. \ansline
\end{itemize}}

\problem[3.6cm]{A pump moves $\dfrac{7}{8}$ of a pool in $1\dfrac{3}{4}$ hours at a constant rate. How many hours does it take to move the whole pool?
\answerline{hours}}

\problem[3.6cm]{A recipe uses $1\dfrac{1}{2}$ cups of flour to make $\dfrac{2}{3}$ of a batch. How many cups of flour does one whole batch need?}

\problem[4cm]{A cyclist rides $5\dfrac{1}{4}$ miles in $\dfrac{3}{4}$ of an hour, then rides a further $2\dfrac{1}{2}$ miles in $\dfrac{1}{2}$ of an hour.
\begin{itemize}
  \item[(a)] Speed on the first leg: \answerline{mph}
  \item[(b)] Speed on the second leg: \answerline{mph}
  \item[(c)] Speed for the whole ride: \answerline{mph}
  \item[(d)] Explain why the whole-ride speed is not the average of your answers to (a) and (b).
\end{itemize}
