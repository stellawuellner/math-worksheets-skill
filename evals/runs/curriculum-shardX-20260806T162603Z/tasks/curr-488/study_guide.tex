\sstitleblock{Volumes from Known Cross Sections and Cylindrical Shells}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Slicing into known cross sections}

\begin{formulabox}
\textbf{One idea:} $V = \displaystyle\int_a^b A(x)\,dx$, where $A(x)$ is the area of the slice at position $x$. The base region only tells you the \emph{side} $s(x)$ of that slice; the shape of the cross section turns it into an area.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item square on the base: $A = s^2$ \qquad semicircle on the base (diameter $s$): $A = \tfrac{\pi}{8}s^2$
  \item equilateral triangle on the base: $A = \tfrac{\sqrt{3}}{4}s^2$
\end{itemize}
If lengths are in centimetres, $A$ is in square centimetres and $V$ is in cubic centimetres.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Base: region under $y = 2x$ from $x = 0$ to $x = 3$; cross sections perpendicular to the $x$-axis are squares. Find $V$.
\step{The slices are named by $x$, so integrate in $x$: read the side of the square off the base, then square it to get the area.}
\step{The side runs from the axis to the line, so $s = 2x$ and $A(x) = 4x^2$; then $V = \int_0^3 4x^2\,dx = \left[\tfrac{4}{3}x^3\right]_0^3$.}

$\Rightarrow\quad \ans{V = 36}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Base: region under $y = 3x$ from $x = 0$ to $x = 2$; square cross sections. Find $V$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{V = 24}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Cylindrical shells}

\begin{formulabox}
\textbf{Shell formula:} $V = \displaystyle\int_a^b 2\pi\,(\text{radius})(\text{height})\,dx$. For a region revolved about the $y$-axis the radius is $x$ and the height is the function value, so the integral stays in $x$ and the curve needs no rewriting. Think of unrolling each thin cylinder into a rectangle of width $2\pi r$, height $h$, thickness $dx$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The region under $y = x$ from $x = 0$ to $x = 2$ is revolved about the $y$-axis. Find $V$.
\step{Revolving about a vertical axis with the region described by $y$ in terms of $x$ makes shells the setup that needs no algebra.}
\step{Radius $x$, height $x$, so $V = \int_0^2 2\pi x\cdot x\,dx = 2\pi\left[\tfrac{x^3}{3}\right]_0^2$.}

$\Rightarrow\quad \ans{V = \frac{16\pi}{3} \approx 16.755}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The region under $y = x$ from $x = 0$ to $x = 3$, revolved about the $y$-axis.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{V = 18\pi \approx 56.549}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Radius when the axis moves}

\begin{formulabox}
\textbf{The radius is a distance, not a coordinate.} For revolution about the vertical line $x = k$ with the region to the \emph{left} of it, the shell at position $x$ is $k - x$ away; with the region to the \emph{right}, it is $x - k$. Check an endpoint before integrating: the radius must be largest for the shell furthest from the axis and zero for a shell on it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The region under $y = x^2$ from $x = 0$ to $x = 2$ is revolved about $x = 3$. Find $V$.
\step{The axis lies to the right of the region, so the shell radius is the gap $3 - x$; the height is still the function value.}
\step{$V = \int_0^2 2\pi(3-x)x^2\,dx = 2\pi\left[x^3 - \tfrac{x^4}{4}\right]_0^2 = 2\pi(8 - 4)$.}

$\Rightarrow\quad \ans{V = 8\pi \approx 25.133}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The same region under $y = x^2$ on $[0,2]$, now revolved about $x = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{V = \frac{56\pi}{3} \approx 58.643}$}
\end{tryitbox}
