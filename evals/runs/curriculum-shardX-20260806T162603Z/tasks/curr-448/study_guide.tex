\sstitleblock{Interpreting Correlation and Least-Squares Regression}

\skillheading{Skill 1 --- Reading a value out of the data}

\begin{formulabox}
\textbf{Rule:} before any line is fitted, be sure which column is the explanatory variable
$x$ and which is the response $y$, and read values off the table or scatterplot in those
units. Every interpretation later has to be phrased in those same units.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table records deliveries per day: Mon $4$, Tue $7$, Wed $9$, Thu $12$. How many on Wednesday?
\step{The question names a single category, so this is a direct read of that column, not a computation.}
\step{Find the Wed column and read the count beneath it.}

$\Rightarrow\quad \ans{9}$ deliveries
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From the same table, what is the largest daily count?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{Skill 2 --- Predicting and finding residuals}

\begin{formulabox}
\textbf{Rule:}\quad predicted value $\hat{y} = a + bx$; \quad residual $= y - \hat{y}$
(actual minus predicted). A positive residual means the point sits \emph{above} the line, a
negative one below. Least squares chooses the line that makes the sum of the squared residuals
as small as possible, which is why the residuals always sum to zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\hat{y} = 12 + 2.5x$, predict $y$ when $x = 6$.
\step{A prediction is a straight substitution into the fitted line --- no data point is needed.}
\step{$\hat{y} = 12 + 2.5(6) = 12 + 15$.}

$\Rightarrow\quad \hat{y} = \ans{27}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The actual value at $x = 6$ turned out to be $30$. Find the residual.\\[2pt]
\rotatebox{180}{\footnotesize check: residual $= \ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- What the slope and intercept mean}

\begin{formulabox}
\textbf{Slope:} for each one-unit increase in $x$, the line predicts a change of $b$ units in
$y$. Say ``is associated with'', not ``causes''. \quad \textbf{Intercept:} the predicted $y$
when $x = 0$ --- meaningful only if $x = 0$ lies inside the range of the data. Predicting
outside that range is \emph{extrapolation} and is not supported by the data.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\hat{y} = 12 + 2.5x$, how much change does the line predict over $4$ extra units of $x$?
\step{The slope is a rate, so the predicted change is the slope times the change in $x$ --- the intercept plays no part.}
\step{$2.5 \times 4$.}

$\Rightarrow\quad$ predicted change $= \ans{10}$ units of $y$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same line. What change does it predict over $10$ extra units of $x$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25}$}
\end{tryitbox}

\skillheading{Skill 4 --- Correlation and $r^2$}

\begin{formulabox}
\textbf{Formula:}
\[ r = \frac{\sum(x-\bar{x})(y-\bar{y})}{\sqrt{\sum(x-\bar{x})^2\,\sum(y-\bar{y})^2}} \]
$r$ runs from $-1$ to $1$ and measures the strength and direction of a \emph{linear} pattern.
$r^2$ is the fraction of the variation in $y$ accounted for by that linear relationship with
$x$. Neither says anything about cause: a lurking variable can drive both.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A data set has $\sum(x-\bar{x})(y-\bar{y}) = 48$, $\sum(x-\bar{x})^2 = 20$, $\sum(y-\bar{y})^2 = 128$. Find $r$.
\step{All three sums the formula needs are given, so substitute rather than returning to the raw data.}
\step{$r = \dfrac{48}{\sqrt{20 \cdot 128}} = \dfrac{48}{\sqrt{2560}}$.}

$\Rightarrow\quad r \approx \ans{0.9487}$, a strong positive linear association.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For the same three sums, find $r^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r^2 = \ans{0.9}$}
\end{tryitbox}
