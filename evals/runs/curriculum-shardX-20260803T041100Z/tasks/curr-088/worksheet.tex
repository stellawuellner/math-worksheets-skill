\wstitleblock{One-Step Mass and Liquid Volume Problems}{Grades 2--3 \quad Measurement and Data}{}

\begin{center}
\textit{One-step problems with mass in grams, liquid volume in millilitres, and
measurements read from a record table}
\end{center}

\noindent\textbf{How to work each problem.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Write down the two numbers you are given, \emph{with their units}
(g for mass, mL for liquid volume).
\item Decide on \textbf{one} operation: put together (add), take away
(subtract), equal groups (multiply), or share equally (divide).
\item Write the answer with its unit. A number with no unit does not answer a
measurement question.
\end{itemize}

\vspace{0.15cm}

\problem{A bag of rice has a mass of $250$ g. A bag of beans has a mass of
$300$ g. The two bags are put on the same scale.

What is the total mass of the two bags?
\par\vspace*{3.4cm}\answerline{g}}

\problem{A pitcher holds $800$ mL of water. Ravi pours out $350$ mL of the
water into a bowl.

How much water is still in the pitcher?
\par\vspace*{3.4cm}\answerline{mL}}

\problem{Maya keeps a measurement record while she makes juice.
\par
\begin{center}
\begin{tabular}{|l|r|}
\hline
\textbf{Measurement record} & \textbf{Amount} \\
\hline
Juice already in the jug & $350$ mL \\
\hline
Juice Maya pours in & $150$ mL \\
\hline
\end{tabular}
\end{center}
Read both amounts from the record. How much juice is in the jug now?
\par\vspace*{3.4cm}\answerline{mL}}

\problem{Every apple in a basket has the same mass, $150$ g. There are $4$
apples in the basket.

What is the mass of all $4$ apples together?
\par\vspace*{3.8cm}\answerline{g}}

\problem{One measuring cup holds $450$ mL of milk. A second measuring cup
holds $275$ mL of milk. Both cups are emptied into one bowl.

How much milk is in the bowl?
\par\vspace*{3.8cm}\answerline{mL}}

\problem{A class weighed three fruits and recorded the masses.
\par
\begin{center}
\begin{tabular}{|l|r|}
\hline
\textbf{Fruit} & \textbf{Mass} \\
\hline
melon & $900$ g \\
\hline
pineapple & $650$ g \\
\hline
mango & $400$ g \\
\hline
\end{tabular}
\end{center}
How much heavier is the melon than the pineapple?
\par\vspace*{3.4cm}\answerline{g}}

\problem{A box of books has a mass of $850$ g. One book with a mass of $275$ g
is taken out of the box.

What is the mass of the box now?
\par\vspace*{3.8cm}\answerline{g}}

\problem{A pot holds $600$ mL of soup. The soup is shared equally into $4$
bowls, so every bowl gets the same amount.

How much soup goes into each bowl?
\par\vspace*{3.8cm}\answerline{mL}}

\problem{Noah kept this record while emptying a water bottle.
\par
\begin{center}
\begin{tabular}{|l|r|}
\hline
\textbf{Measurement record} & \textbf{Amount} \\
\hline
Water in the bottle at the start & $750$ mL \\
\hline
Water Noah poured out & $300$ mL \\
\hline
Noah's answer for what is left & $1050$ mL \\
\hline
\end{tabular}
\end{center}
Noah's answer cannot be right. Explain in one sentence what he did wrong, then
write the correct amount of water left in the bottle.
\par\vspace*{4.2cm}\answerline{mL}}

\problem{A large pot holds $950$ mL of soup. Grandma pours the soup equally
into $5$ cups, so every cup gets the same amount.

How much soup goes into each cup?
\par\vspace*{4.2cm}\answerline{mL}}
