% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Terms, coefficients and degree}

\begin{formulabox}
\textbf{The vocabulary, on one polynomial:} $\ 3x^4 - x^2 + 5x - 2$
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Terms} are the pieces joined by $+$ and $-$; the sign belongs to the term.
  \item The \textbf{coefficient} is the number multiplying the power. A lone $x$ has coefficient $1$; a lone number has degree $0$.
  \item The \textbf{degree} of the polynomial is the largest exponent that appears.
  \item Handy check: $P(1)$ is the sum of all the coefficients.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $P(x) = 3x^4 - x^2 + 5x - 2$, list the coefficients and find $P(1)$.
\step{The coefficients are what I want, and substituting $1$ turns every power of $x$ into $1$, so the sum of the coefficients is exactly the value $P(1)$.}
\step{Coefficients in order: $3$, $-1$, $5$, $-2$ (degree $4$, and the constant has degree $0$).}
\step{$P(1) = 3 - 1 + 5 - 2$}

$\Rightarrow\quad P(1) = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $R(x) = 2x^3 + 4x^2 - x + 6$, list the coefficients and find $R(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- From a product to a sum (area model)}

\begin{formulabox}
\textbf{Rule:} every cell of the rectangle is one partial product, and the total area is their sum.
\[
  (x + p)(x + q) \;=\; x^2 + (p+q)\,x + pq
\]
The middle coefficient is the \emph{sum} of the two numbers; the constant is their \emph{product}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(x+5)(x+2)$.
\step{Two binomials multiplied means four partial products, so I draw the two-by-two area model rather than guessing the middle term.}
\step{Cells: $x\cdot x = x^2$, \ $x\cdot 5 = 5x$, \ $2\cdot x = 2x$, \ $2\cdot 5 = 10$.}
\step{Add, combining the like terms $5x$ and $2x$: $x^2 + 7x + 10$.}

$\Rightarrow\quad \ans{x^2 + 7x + 10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(x+3)(x+4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 7x + 12}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From a sum back to a product}

\begin{formulabox}
\textbf{Always look for a common factor first:} pull out the largest number that divides every
coefficient and the lowest power of $x$ that appears in every term.
\textbf{Then, for $x^2 + bx + c$:} find two numbers with product $c$ and sum $b$; they are the
constants of the two binomial factors.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $10x^3 + 15x^2$, then factor $x^2 - 2x - 15$.
\step{The first has a common factor in every term, so that is a common-factor job; the second has no common factor but is a trinomial, so it is a product-and-sum job.}
\step{$10$ and $15$ share $5$, and both terms carry $x^2$: $10x^3 + 15x^2 = 5x^2(2x + 3)$.}
\step{For $x^2 - 2x - 15$: two numbers with product $-15$ and sum $-2$ are $-5$ and $3$.}

$\Rightarrow\quad \ans{5x^2(2x + 3)} \quad\text{and}\quad \ans{(x - 5)(x + 3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $x^2 - 4x - 21$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x - 7)(x + 3)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the degree of a \emph{product} is the sum of the degrees of its factors,
so $(3x^2-1)(x+5)$ has degree $3$ --- you never have to expand to find it. And a missing number
is never a missing coefficient: in $x^2 + x$, the coefficient of $x$ is $1$.
\end{watchoutbox}
