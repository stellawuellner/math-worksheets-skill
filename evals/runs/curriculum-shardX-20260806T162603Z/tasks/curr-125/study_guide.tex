% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Ordering with benchmarks}

\begin{formulabox}
\textbf{Benchmark test:} sort fractions by first placing each one against an
easy landmark like $\tfrac{1}{2}$ or 1. A fraction is more than
$\tfrac{1}{2}$ when its numerator is more than half its denominator. Often
the benchmarks alone settle the whole order.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Order $\tfrac{2}{5}$, $\tfrac{1}{2}$, $\tfrac{5}{8}$
from least to greatest.
\step{Place each fraction against the benchmark one half instead of rewriting all three.}
\step{Half of 5 is $2.5$, so $\tfrac{2}{5}$ is below the benchmark; half of 8 is 4, so $\tfrac{5}{8}$ is above it: $\ans{\tfrac{2}{5}} < \ans{\tfrac{1}{2}} < \ans{\tfrac{5}{8}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Order $\tfrac{5}{12}$, $\tfrac{1}{2}$,
$\tfrac{7}{10}$ from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{12}} < \ans{\tfrac{1}{2}} < \ans{\tfrac{7}{10}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Ordering with a common denominator}

\begin{formulabox}
\textbf{Plan:} when benchmarks leave two fractions on the same side, rewrite
all the fractions with one common denominator and sort by numerators ---
same-size pieces sort honestly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Order $\tfrac{2}{3}$, $\tfrac{3}{4}$,
$\tfrac{7}{12}$ from least to greatest.
\step{All three denominators divide into 12, so rewrite everything in twelfths.}
\step{$\tfrac{2}{3} = \tfrac{\ans{8}}{12}$ and $\tfrac{3}{4} = \tfrac{\ans{9}}{12}$, so sorting 7, 8, 9 twelfths gives $\ans{\tfrac{7}{12}} < \ans{\tfrac{2}{3}} < \ans{\tfrac{3}{4}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Order $\tfrac{5}{6}$, $\tfrac{3}{4}$, $\tfrac{2}{3}$
from least to greatest (use twelfths).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{2}{3}} < \ans{\tfrac{3}{4}} < \ans{\tfrac{5}{6}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Mixed numbers and improper fractions}

\begin{formulabox}
\textbf{Rule:} a mixed number and an improper fraction are two names for one
value. To compare or order them, put everything in the same form: each whole
is worth one denominator's count of pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $\tfrac{9}{4}$ as a mixed number.
\step{Divide to find the wholes, and the remainder stays as fourths.}
\step{$9 \div 4 = 2$ remainder 1, so $\tfrac{9}{4} = \ans{2\tfrac{1}{4}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $\tfrac{11}{3}$ as a mixed number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3\tfrac{2}{3}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never compare a mixed number and an improper
fraction by their front numbers. $2\tfrac{1}{4}$ and $\tfrac{9}{4}$ start
with 2 and 9 --- and are exactly equal.
\end{watchoutbox}
