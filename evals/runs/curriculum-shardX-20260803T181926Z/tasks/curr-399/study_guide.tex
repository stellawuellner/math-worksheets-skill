\sstitleblock{Reading a Conic Equation Without Losing a Sign or a Square Root}

\skillheading{Skill 1 --- Centre and radius of a circle}

\begin{formulabox}
\textbf{Standard form:} \quad $(x-h)^2 + (y-k)^2 = r^2$, \ centre $(h,k)$,
radius $r$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The form \emph{subtracts} the centre, so read $h$ and $k$ by setting
        each binomial to zero. A printed $+$ gives a negative coordinate.
  \item The right-hand side is $r^2$. The radius is its square root.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the centre and radius of $(x-2)^2 + (y+5)^2 = 81$.
\step{The two things asked for are the centre and the radius, and both are hidden
by an operation --- a subtraction in the binomials and a square on the right ---
so undo one at a time.}
\step{Set each binomial to zero: $x - 2 = 0$ and $y + 5 = 0$, so the centre is
$\ans{(2,\ -5)}$.}
\step{The right side is $r^2$, so take its square root: $r = \ans{9}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the centre and radius of $(x+6)^2 + (y-1)^2 = 25$.\\[2pt]
\rotatebox{180}{\footnotesize check: centre $\ans{(-6,\ 1)}$, \ radius $\ans{5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(x+6)$ does \emph{not} mean the centre is at $6$. Ask
what value of $x$ makes the bracket zero --- that is the coordinate.
\end{watchoutbox}

\skillheading{Skill 2 --- Reading an ellipse's denominators}

\begin{formulabox}
\textbf{Standard form:} \quad
$\dfrac{(x-h)^2}{a^2} + \dfrac{(y-k)^2}{b^2} = 1$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Each denominator is a \emph{square}: the semi-axis is its square root.
  \item The \emph{larger} denominator marks the major axis, whichever term it
        sits under --- not the term written first.
  \item Full axis length $=$ twice the semi-axis.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Give both semi-axes of
$\dfrac{x^2}{36} + \dfrac{y^2}{16} = 1$.
\step{Semi-axis lengths are asked for and the denominators are squares, so this
is a square-root reading, not a copy.}
\step{Under $x^2$: the semi-axis is $\sqrt{36} = \ans{6}$, and since $36$ is the
larger denominator this is the semi-\emph{major} axis, lying horizontally.}
\step{Under $y^2$: the semi-minor axis is $\sqrt{16} = \ans{4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Give the semi-major and semi-minor axes of
$\dfrac{x^2}{9} + \dfrac{y^2}{64} = 1$, and say which way the major axis
runs.\\[2pt]
\rotatebox{180}{\footnotesize check: semi-major $\ans{8}$ (vertical), \ semi-minor $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Which way a hyperbola opens}

\begin{formulabox}
\textbf{Standard forms:} \quad
$\dfrac{(x-h)^2}{a^2} - \dfrac{(y-k)^2}{b^2} = 1$ opens left and right; \quad
$\dfrac{(y-k)^2}{b^2} - \dfrac{(x-h)^2}{a^2} = 1$ opens up and down.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The two squared terms are \emph{subtracted}. The branches open along the
        variable whose term is \textbf{positive}.
  \item The vertex distance is the square root of the \emph{positive} term's
        denominator. The subtracted term's denominator never gives a vertex.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Which way does $\dfrac{x^2}{25} - \dfrac{y^2}{4} = 1$
open, and where are its vertices?
\step{The $x$ term carries the plus sign, so the branches open left and right;
the vertices then sit on the $x$-axis and are found by setting $y = 0$.}
\step{With $y = 0$ the equation becomes $\dfrac{x^2}{25} = 1$, so $x^2 = 25$ and
$x = \ans{-5}$ or $x = \ans{5}$: the vertices are five units either side of the
centre.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Which way does $\dfrac{y^2}{49} - \dfrac{x^2}{9} = 1$
open, and where are its vertices?\\[2pt]
\rotatebox{180}{\footnotesize check: opens up and down; $y = \ans{-7}$ or $y = \ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Reaching for the denominator under the \emph{subtracted}
term puts the vertices where the curve has no points at all. Find the plus sign
first, every time.
\end{watchoutbox}
