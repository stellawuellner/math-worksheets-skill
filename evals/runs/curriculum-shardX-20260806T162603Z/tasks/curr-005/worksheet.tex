\wstitleblock{Compare Sets by Counting On}{Kindergarten--Grade 1}{}
\begin{center}\textit{Which has more, how many more, and making sets equal}\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Count each group of dots one at a time and write its number.
  \item To find \emph{how many more}, start at the smaller number and count on
        to the bigger one. Each hop is one more.
\end{itemize}

\problem[2.8cm]{Count each row. Write the numbers.\par\vspace{0.2cm}
\begin{center}
Row A: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet$}\\[4pt]
Row B: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet$}
\end{center}
\vspace{0.2cm}\par Row A: \ansblank\qquad Row B: \ansblank\qquad
Which row has more? Write A or B: \ansblank}

\problem[2.8cm]{These rows look different --- but do they have the same number
of dots? Count each row.\par\vspace{0.2cm}
\begin{center}

\vspace{0.2cm}\par Row A: \ansblank\qquad Row B: \ansblank\qquad

\problem[3cm]{Row A has 3 dots. Row B has 8 dots.\par\vspace{0.2cm}
\begin{center}

\end{center}
\vspace{0.2cm}\par Count on from 3 to 8. Write each number you say on a hop
line: \scratchblank\ \scratchblank\ \scratchblank\ \scratchblank\ \scratchblank
\par\vspace{0.2cm} How many more dots does Row B have? \ansblank}

\problem[3.5cm]{\textbf{Challenge.} Count each row. Then find how many more
dots Row B has by counting on.\par\vspace{0.2cm}
\begin{center}
Row A: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet$}\\[4pt]
Row B: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet$}
\end{center}
\vspace{0.2cm}\par Row A: \ansblank\qquad Row B: \ansblank\qquad
How many more in Row B? \ansblank\par\vspace{0.2cm}
Then say how you counted on --- tell a grown-up or write it below.}
