\wstitleblock{Two Forms of a Linear Equation}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Standard form to slope-intercept form, back again, and the intercepts
in between}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Standard form is $Ax + By = C$; slope-intercept form is $y = mx + b$.
        The two describe the same relationship.
  \item To go from standard to slope-intercept, isolate $y$: subtract the
        $x$-term from both sides, then divide \emph{every} term by the
        coefficient of $y$.
  \item To go the other way, clear the fraction and move the $x$-term to the
        left so both variables sit on one side.
  \item Keep the quantities and units the story gives you attached to your
        answer.
\end{itemize}

\vspace{0.15cm}

\problem[5cm]{A school spends \$36 on supplies, buying $x$ notebooks at \$3
each and $y$ pens at \$2 each: $3x + 2y = 36$. Rewrite the relationship in
slope-intercept form, solved for $y$.}

\problem[5cm]{A club sells $x$ child tickets at \$4 and $y$ adult tickets at
\$5, taking \$40 in all: $4x + 5y = 40$. Rewrite the relationship in
slope-intercept form, solved for $y$.}

\problem[5cm]{A caterer's plan for $x$ trays of fruit and $y$ trays of bread
is $y = -\frac{2}{3}x + 8$. Rewrite it in the standard form $2x + 3y = C$ by
finding the constant $C$.}

\problem{A printing budget of \$60 buys $x$ posters at \$5 each and $y$ flyers
at \$4 each: $5x + 4y = 60$.
\begin{itemize}
  \item[(a)] Find the $x$-intercept: how many posters if no flyers are bought?
        \par\vspace*{2.6cm}\ansline
  \item[(b)] Find the $y$-intercept: how many flyers if no posters are bought?
        \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[5cm]{A fundraiser's profit relationship is $2x - y = 7$, where $x$ is
the number of boxes sold and $y$ is the profit in dollars. Rewrite it in
slope-intercept form, solved for $y$.}

\problem[5cm]{A hiking club's plan for $x$ hours of walking and $y$ hours of
rest is $y = \frac{3}{4}x - 5$. Rewrite it in the standard form
$3x - 4y = C$ by finding the constant $C$.}

\problem{A community fund of \$18 pays for $x$ packets of seeds at \$6 each
and $y$ bags of soil at \$3 each: $6x + 3y = 18$.
\begin{itemize}
  \item[(a)] Find the $x$-intercept.
        \par\vspace*{2cm}\ansline
  \item[(b)] Find the $y$-intercept.
        \par\vspace*{2cm}\ansline
  \item[(c)] Use the two intercept points to find the slope of the line.
        \par\vspace*{2cm}\ansline
\end{itemize}}

\problem{A booth takes exactly \$48 selling $x$ small drinks and $y$ large
drinks, which is the relationship $3x + 4y = 48$. Some combinations are listed
below.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Small drinks $x$} & 0 & 4 & 8 & 12 \\ \hline
\textbf{Large drinks $y$} & 12 & 9 & 6 & 3 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] By how much does the number of large drinks change as $x$ goes
        from 0 to 12?
        \par\vspace*{2cm}\ansline
  \item[(b)] Rewrite $3x + 4y = 48$ in slope-intercept form, solved for $y$.
        \par\vspace*{2.3cm}\ansline
\end{itemize}}

\problem[5cm]{A paint mixture uses $x$ litres of blue and $y$ litres of white
according to $y = -\frac{5}{2}x + 7$. Rewrite it in the standard form
$5x + 2y = C$ by finding the constant $C$.}

\problem{A community garden has \$120 to spend, with tomato plants at \$8 each
and pepper plants at \$6 each: $8x + 6y = 120$, where $x$ is the number of
tomato plants and $y$ the number of pepper plants.
\begin{itemize}
  \item[(a)] Rewrite the equation in slope-intercept form, solved for $y$.
        \par\vspace*{2.6cm}\ansline
  \item[(b)] Explain what the slope you found means for the gardener choosing
        between the two kinds of plant.
        \par\vspace*{2.4cm}\ansline
\end{itemize}}
