\sstitleblock{Integrating Rational Functions With Partial Fractions}

\begin{formulabox}
\textbf{The whole method in one line:} a rational function with distinct linear
factors downstairs equals a sum of constants over those factors, and each piece
integrates to a logarithm --- $\int \frac{dx}{x-r} = \ln|x-r| + C$.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \textbf{Check the degree first.} If the numerator's degree is not lower
than the denominator's, do polynomial division before splitting.
\item \textbf{Find the constants by substitution.} Clear denominators, then put
$x$ equal to each root in turn: every term but one vanishes.
\item \textbf{Keep the absolute values} and combine differences of logarithms
into one logarithm at the end.
\end{itemize}
\end{formulabox}

\skillheading{1. Splitting into distinct linear factors}

\begin{formulabox}
Write $\frac{N(x)}{(x-r_1)(x-r_2)} = \frac{A}{x-r_1} + \frac{B}{x-r_2}$, clear
denominators, and substitute $x = r_1$ then $x = r_2$. Check the split by
recombining over the common denominator --- faster than differentiating.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $\displaystyle\int \frac{4}{(x-1)(x+3)}\,dx$.
\step{The factors are distinct and linear and the numerator's degree is lower,
so constants over each factor are all that is needed.}
\step{$4 = A(x+3) + B(x-1)$: at $x = 1$, $4 = 4A$ so $A = 1$; at $x = -3$,
$4 = -4B$ so $B = -1$. Integrating each piece gives the logarithms.}
\[ \ans{\ln|x-1| - \ln|x+3| + C} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $\displaystyle\int \frac{6}{(x-2)(x+1)}\,dx$.

\rotatebox{180}{\footnotesize check: $\ans{2\ln|x-2| - 2\ln|x+1| + C}$}
\end{tryitbox}

\skillheading{2. Definite integrals after the split}

\begin{formulabox}
Split first, antidifferentiate, then substitute the limits. Combine the
logarithms into one before evaluating: it turns four log values into a single
quotient and removes most of the arithmetic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $\displaystyle\int_{1}^{2} \frac{3}{x(x+3)}\,dx$.
\step{A definite integral changes nothing about the algebra --- decompose
exactly as before, and save the limits for the last line.}
\step{$3 = A(x+3) + Bx$ gives $A = 1$, $B = -1$, so the antiderivative is
$\ln\left|\frac{x}{x+3}\right|$, and evaluating gives
$\ln\frac{2}{5} - \ln\frac{1}{4}$.}
\[ \ans{\ln\frac{8}{5}} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\displaystyle\int_{0}^{1}
\frac{2}{(x+1)(x+3)}\,dx$.

\rotatebox{180}{\footnotesize check: $\ans{\ln\frac{3}{2}}$}
\end{tryitbox}

\skillheading{3. Improper integrals: split, then take a limit}

\begin{formulabox}
$\int_{a}^{\infty} f = \lim_{b\to\infty}\int_{a}^{b} f$. Decompose, evaluate on
$[a, b]$, \textbf{combine into a single logarithm}, and only then let
$b \to \infty$. A quotient like $\frac{b}{b+1}$ tends to 1, so its logarithm
tends to 0; separate logarithms would give $\infty - \infty$ and decide nothing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $\displaystyle\int_{1}^{\infty}
\frac{1}{x(x+1)}\,dx$.
\step{An infinite limit is not a number to substitute, so the integral is
defined as the limit of the proper integral on $[1, b]$.}
\step{$\frac{1}{x} - \frac{1}{x+1}$ integrates to
$\ln\left|\frac{x}{x+1}\right|$; on $[1, b]$ that is
$\ln\frac{b}{b+1} + \ln 2$, and the first term tends to $\ln 1 = 0$.}
\[ \ans{\ln 2} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\displaystyle\int_{2}^{\infty}
\frac{4}{x(x+2)}\,dx$.

\rotatebox{180}{\footnotesize check: $\ans{\ln 4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{Watch out.} A coefficient belongs outside its logarithm:
$2\ln|x-2|$ is $\ln(x-2)^2$, not $\ln|2x-4|$. And decaying to zero does not make
an improper integral converge --- $\int_{0}^{\infty}\frac{dt}{t+1}$ diverges
even though the integrand tends to 0. What matters is how fast.
\end{watchoutbox}
