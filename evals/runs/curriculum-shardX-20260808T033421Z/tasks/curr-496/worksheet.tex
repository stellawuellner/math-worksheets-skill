\wstitleblock{Recognising Geometric Series, p-Series, and the nth-Term Test}{AP Calculus BC}{}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Three tools, three questions. \emph{Is it geometric?} Look for a constant ratio $r$; it converges when $|r| < 1$, to $\dfrac{\text{first term}}{1-r}$. \emph{Is it a $p$-series?} $\sum \frac{1}{n^p}$ converges exactly when $p > 1$. \emph{Do the terms go to zero?} If not, the series diverges.
  \item The nth-term test only ever proves \emph{divergence}. Terms shrinking to zero proves nothing on its own.
  \item Where a table of partial sums is given, use it as a picture of what the symbols are doing --- then answer with the algebra.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{The table shows the first few partial sums of $\displaystyle\sum_{n=0}^{\infty}\left(\frac{1}{3}\right)^{n}$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{l|cccc}
\textbf{last index $n$} & $0$ & $1$ & $2$ & $3$ \\ \hline
\textbf{partial sum} & $1$ & $1.3333$ & $1.4444$ & $1.4815$ \\
\end{tabular}
\end{center}
\par
Find the exact sum of the whole series, and say what the table suggests about it.}

\problem[5cm]{Find the exact sum of $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}}$. (This is the most famous convergent $p$-series; you are expected to know its value.)}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{3n^{2}+1}{5n^{2}+2n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty}\frac{3n^{2}+1}{5n^{2}+2n}$.
  \item[(b)] State what the nth-term test lets you conclude about the series, and explain what it would \emph{not} have told you had the limit been zero.
\end{itemize}}

\problem{A ball is dropped from a height of $2$ metres. Each time it hits the floor it rebounds to $\frac{3}{5}$ of the height it fell from, and this continues forever. Find the total vertical distance the ball travels (down, then up and down again on every bounce).
\par\vspace*{6cm}\answerline{m}}

\problem[6.5cm]{The table shows partial sums of $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{4}}$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{l|cccc}
\textbf{last index $n$} & $1$ & $2$ & $3$ & $4$ \\ \hline
\textbf{partial sum} & $1$ & $1.0625$ & $1.0748$ & $1.0788$ \\
\end{tabular}
\end{center}
\par
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the exact sum of the series.
  \item[(b)] Explain how the exponent alone tells you this series converges, and connect that to what the table is doing.
\end{itemize}}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty} n\sin\!\left(\frac{1}{n}\right)$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty} n\sin\!\left(\frac{1}{n}\right)$.
  \item[(b)] State the conclusion the nth-term test gives, and explain why the terms fail to shrink even though $\sin\!\left(\frac{1}{n}\right)$ does.
\end{itemize}}

\problem[7cm]{The repeating decimal $0.\overline{45} = 0.454545\ldots$ can be written as the geometric series $\displaystyle\sum_{n=1}^{\infty}\frac{45}{100^{\,n}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Sum the series to write $0.\overline{45}$ as a fraction in lowest terms.
  \item[(b)] Explain why the same argument shows that \emph{every} repeating decimal is a rational number.
\end{itemize}}

\problem[7cm]{Here are three series. Exactly one converges.
\[ \text{(i) } \sum_{n=0}^{\infty}\left(\frac{2}{3}\right)^{n} \qquad
   \text{(ii) } \sum_{n=1}^{\infty}\frac{1}{\sqrt{n}} \qquad
   \text{(iii) } \sum_{n=1}^{\infty}\frac{n}{n+4} \]
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the exact sum of series (i).
  \item[(b)] Find $\displaystyle\lim_{n\to\infty}\frac{n}{n+4}$.
  \item[(c)] Classify all three series, naming the deciding quantity in each case.
\end{itemize}}
