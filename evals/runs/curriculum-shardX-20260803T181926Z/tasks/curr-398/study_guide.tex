\sstitleblock{Vertices, Foci and Asymptotes of Ellipses and Hyperbolas}

\noindent\small Both curves are written with $x^2$ and $y^2$ over
denominators. Which curve you have is decided by the sign between the two
terms: a plus sign is an ellipse (a closed loop), a minus sign is a hyperbola
(two branches). Which direction it runs is decided by the term that comes
first, or is positive. The number under the transverse or major term is always
$a^2$.\normalsize

\skillheading{Ellipse: vertices and foci from the equation}
\begin{formulabox}
\textbf{Rule:} For $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ with $a > b$, the
major axis is horizontal: vertices $(\pm a, 0)$, co-vertices $(0, \pm b)$,
foci $(\pm c, 0)$ with $c^2 = a^2 - b^2$.\par
\textbf{Subtract} for an ellipse: the foci sit \emph{inside} the loop, so
$c < a$. If the bigger denominator is under $y^2$ instead, everything turns a
quarter-turn: vertices $(0, \pm a)$ and foci $(0, \pm c)$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find the foci of $\dfrac{x^2}{100} + \dfrac{y^2}{36} = 1$.
\step{The bigger denominator, $100$, is under $x^2$, so this is a horizontal
ellipse and the foci are on the $x$-axis --- use $c^2 = a^2 - b^2$.}
\step{$c^2 = 100 - 36 = 64$, so $c = \sqrt{64}$.}

Foci at $(\pm c, 0)$ with $c = \ans{8}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $c$ for $\dfrac{x^2}{225} + \dfrac{y^2}{144} = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Hyperbola: vertices and foci}
\begin{formulabox}
\textbf{Rule:} For $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ the branches open
left and right: vertices $(\pm a, 0)$, foci $(\pm c, 0)$ with
$c^2 = a^2 + b^2$.\par
\textbf{Add} for a hyperbola: the foci sit \emph{beyond} the vertices, so
$c > a$ always. The positive term names the direction --- if the $y^2$ term is
the positive one, the branches open up and down and $a^2$ is the number
under $y^2$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find the vertices and foci of
$\dfrac{x^2}{9} - \dfrac{y^2}{16} = 1$.
\step{The $x^2$ term is positive, so the branches open sideways and both
vertices and foci are on the $x$-axis --- use $c^2 = a^2 + b^2$.}
\step{$a = \sqrt{9} = 3$, so the vertices are $(\pm 3, 0)$; and
$c^2 = 9 + 16 = 25$.}

Foci at $(\pm c, 0)$ with $c = \ans{5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $c$ for $\dfrac{x^2}{64} - \dfrac{y^2}{36} = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Hyperbola: slopes of the asymptotes}
\begin{formulabox}
\textbf{Rule:} Draw the central rectangle: $a$ each way along the transverse
axis, $b$ each way along the other. The asymptotes are its two diagonals
extended.\par
Opening sideways: $y = \pm \dfrac{b}{a}x$. Opening up and down:
$y = \pm \dfrac{a}{b}x$. Either way the number from under the \emph{positive}
term goes on the bottom when the curve opens sideways and on the top when it
opens up and down. An ellipse has no asymptotes at all.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find the asymptote slopes of
$\dfrac{x^2}{25} - \dfrac{y^2}{4} = 1$.
\step{The $x^2$ term is positive, so this hyperbola opens sideways and the
slopes are $\pm \frac{b}{a}$ --- rise $b$ over run $a$ on the central
rectangle.}
\step{$a = \sqrt{25} = 5$ and $b = \sqrt{4} = 2$, so the slope is
$\frac{2}{5}$, and the asymptotes are $y = \pm \frac{2}{5}x$.}

Slope $= \ans{\frac{2}{5}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find the asymptote slope of
$\dfrac{x^2}{49} - \dfrac{y^2}{9} = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{3}{7}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not read $a$ as ``the number under $x^2$''. For an
ellipse $a^2$ is the \emph{larger} denominator; for a hyperbola $a^2$ is the
denominator under the \emph{positive} term. Getting that backwards inverts
every asymptote slope. And never take the square root of the whole equation:
the denominators are $a^2$ and $b^2$, so take roots of those numbers alone.
\end{watchoutbox}
