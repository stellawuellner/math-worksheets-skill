\wstitleblock{Making Ten}{Kindergarten--Grade 1 \quad Operations and Algebraic Thinking}{}

\begin{center}
\textit{Finding the missing part of ten, taking a part away from ten, and using
ten to add a bigger number}
\end{center}

\noindent\textbf{How to use a ten-frame.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A full ten-frame holds $10$. A full top row holds $5$.
\item Count the dots you can see. Then count the empty squares. Those two
numbers are the two \emph{parts} of $10$.
\item To add a big number, first fill the frame to $10$, then count on with
whatever is left over.
\end{itemize}

\bigskip

$10$?
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.3cm]\hline
$\bullet$ & & & & \\[0.3cm]\hline
\end{tabular}
\end{center}
$6 + \underline{\hspace{1.2cm}} = 10$
\par\vspace*{3.5cm}\ansline}

\problem{This ten-frame has $3$ dots. How many more dots would fill it to $10$?
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & & \\[0.3cm]\hline
 & & & & \\[0.3cm]\hline
\end{tabular}
\end{center}
$3 + \underline{\hspace{1.2cm}} = 10$
\par\vspace*{3.5cm}\ansline}

\problem{A tower is built from $10$ cubes. Nina takes $8$ cubes off the top.
How many cubes are still in the tower?
\begin{center}\Large
$10 - 8 = \underline{\hspace{1.2cm}}$
\end{center}
\par\vspace*{4cm}\ansline}

\problem{Ravi has $10$ stickers. He puts $7$ of them in his left pocket and the
rest in his right pocket. How many stickers are in his right pocket?
\begin{center}\Large
$7 + \underline{\hspace{1.2cm}} = 10$
\end{center}
\par\vspace*{4cm}\ansline}

\problem{Ellie has $8$ counters on her ten-frame and $4$ more counters in her
hand.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.3cm]\hline
$\bullet$ & $\bullet$ & $\bullet$ & & \\[0.3cm]\hline
\end{tabular}
\qquad in her hand: $\bullet$ \; $\bullet$ \; $\bullet$ \; $\bullet$
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item How many of her counters does she need to fill the frame to $10$?
        \ansline
  \item Now count on from $10$. How many counters does Ellie have altogether?
        \ansline
\end{enumerate}}

\problem{A carton holds $10$ eggs. Dad uses $6$ eggs to bake. How many eggs are
left in the carton?
\begin{center}\Large
$10 - 6 = \underline{\hspace{1.2cm}}$
\end{center}
\par\vspace*{4cm}\ansline}

\problem{The whole is $10$. One part is $2$. Fill in the other part.
\begin{center}\Large
\begin{tabular}{|c|}\hline
whole: $10$ \\[0.25cm]\hline
\end{tabular}
\\[0.35cm]
\begin{tabular}{|c|c|}\hline
part: $2$ & part: $?$ \\[0.35cm]\hline
\end{tabular}
\end{center}
\par\vspace*{3.5cm}\ansline}

\problem{Solve $7 + 5$ by making a ten first.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.3cm]\hline
$\bullet$ & $\bullet$ & & & \\[0.3cm]\hline
\end{tabular}
\qquad in her hand: $\bullet$ \; $\bullet$ \; $\bullet$ \; $\bullet$ \; $\bullet$
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item How many of the $5$ are used to fill the frame to $10$? \ansline
  \item What is $7 + 5$? \ansline
\end{enumerate}}

\problem{There are $10$ seats at story time.

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item $4$ children sit down. How many seats are still empty? \ansline
  \item Then $3$ more children sit down. How many seats are empty now?
        \ansline
\end{enumerate}}

\problem{\textbf{Put it together.} Sam has $10$ stickers. He gives $3$ of them
to a friend. Later his teacher gives him $6$ new stickers.

\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=6pt]
  \item How many stickers does Sam have after he gives $3$ away? \ansline
  \item Now make a ten to add the $6$ new stickers. How many stickers does Sam
        have in the end? \ansline
\end{enumerate}}
