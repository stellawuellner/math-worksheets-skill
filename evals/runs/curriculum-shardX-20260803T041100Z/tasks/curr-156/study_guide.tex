\sstitleblock{Finding Unit Rates with Whole-Number Quantities}

\skillheading{Skill 1 --- Finding a unit rate by dividing}

\begin{formulabox}\textbf{Rule:}\ A \emph{unit rate} tells you how much of one
quantity goes with \textbf{exactly one} of the other.\par
\centerline{unit rate $=\ \dfrac{\text{total of the first quantity}}{\text{number of the second quantity}}$}
The quantity named after the word \emph{per} is always the divisor: ``dollars
per bottle'' means dollars $\div$ bottles.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad A 6-pack of juice costs 18 dollars.
Find the cost per bottle.
\step{``Per bottle'' names bottles as the divisor, so I divide the dollars by
the number of bottles rather than the other way round.}
\step{$18 \div 6 = 3$, and the units come along: dollars per bottle.}
$\Rightarrow\quad \ans{3}$ dollars per bottle
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad A printer prints 40 pages in 8 minutes.
How many pages per minute?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$ pages per minute}
\end{tryitbox}

\skillheading{Skill 2 --- Reading a unit rate out of a ratio table}

\begin{formulabox}\textbf{Rule:}\ Every column of a ratio table holds the same
rate. To reach the ``per one'' column, divide \textbf{both} rows by the same
number; to leave it, multiply both rows by the same number.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad A car covers 208 miles in 4 hours.
\ \begin{tabular}{|l|c|c|}\hline
\textbf{Hours} & 4 & 1 \\ \hline
\textbf{Miles} & 208 & ? \\ \hline
\end{tabular}
\step{The table already pairs the two quantities, so I just scale the hours
column down to 1 and do the same thing to the miles.}
\step{$4 \div 4 = 1$, so $208 \div 4 = 52$.}
$\Rightarrow\quad \ans{52}$ miles per hour
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad 7 identical boxes hold 91 crayons.
How many crayons per box?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$ crayons per box}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two rates fairly}

\begin{formulabox}\textbf{Rule:}\ Two totals can only be compared when they buy
the same amount. Bring \emph{both} offers down to one unit, then compare the
unit rates. Subtracting the two unit rates says by how much, and a difference
of $0$ means the two rates are equal.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad 3 lb for 12 dollars, or 5 lb for
15 dollars? By how much per pound?
\step{The two offers weigh different amounts, so the totals are not comparable
--- I reduce each to dollars per pound first.}
\step{$12 \div 3 = 4$ and $15 \div 5 = 3$; the difference is $4 - 3$.}
$\Rightarrow\quad \ans{1}$ dollar more per pound at the first store
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad 4 lb for 32 dollars, or 5 lb for
30 dollars? Subtract the second rate from the first.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$ dollars more per pound}
\end{tryitbox}

\skillheading{Skill 4 --- Using a unit rate to scale up}

\begin{formulabox}\textbf{Rule:}\ Once you have the rate for one, multiply it
by however many you want. The same work written as a proportion is
$\dfrac{p}{q} = \dfrac{x}{n}$, which cross-multiplies to $qx = pn$.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad 9 notebooks cost 45 dollars. What do
13 notebooks cost?
\step{The two orders differ only in how many notebooks, so one notebook is the
bridge between them.}
\step{$45 \div 9 = 5$ dollars each, then $5 \times 13 = 65$; as a proportion,
$9x = 45 \cdot 13$ gives the same $x$.}
$\Rightarrow\quad \ans{65}$ dollars
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad A printer prints 8 pages in 2 minutes.
How many minutes for 52 pages?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$ minutes}
\end{tryitbox}
