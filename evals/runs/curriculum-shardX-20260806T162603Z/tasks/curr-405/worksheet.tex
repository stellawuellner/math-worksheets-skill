\wstitleblock{Transformations, Compositions and Inverses: A Synthesis Review}{Precalculus}{}

\begin{center}
\small\textit{Transforming a graph, composing two functions, and inverting the result}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--3 are a warm-up, one on each of the three tools. After that the tools are mixed, so decide which one the problem is asking for before you start writing.
  \item Give exact answers; leave fractions as fractions.
  \item Where a domain restriction is stated, use it --- it decides a sign.
\end{itemize}
\vspace{0.15cm}

\problem[4.5cm]{Let $f(x) = x^{2}$ and $g(x) = f(x - 2) + 3$. Find $g(5)$.}

\problem[4.5cm]{Let $f(x) = 2x + 1$ and $g(x) = x^{2}$. Find $(f \circ g)(3)$.}

\problem[4.5cm]{Let $f(x) = 5x - 3$. Find $f^{-1}(12)$.}

\problem[5cm]{Let $f(x) = \sqrt{x}$ and $g(x) = -2f(x + 1)$. Find $g(8)$.}

\problem[5.5cm]{Let $f(x) = x^{2} - 1$ and $g(x) = x + 4$. Write $(f \circ g)(x)$ as a simplified polynomial.}

\problem[5.5cm]{Let $f(x) = \dfrac{x - 7}{2}$. Write the rule for $f^{-1}$, then find $f^{-1}(4)$.}

\problem[6cm]{Let $f(x) = \sqrt{x - 1}$ and $g(x) = x^{2} + 1$. Simplify $(f \circ g)(x)$, then solve $(f \circ g)(x) = 3$.}

\problem[0pt]{Let $f(x) = x^{2} - 17$ and $g(x) = 3f(x - 2) + 5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $g(6)$.
  \par\vspace*{2.8cm}\ansline
  \item[(b)] Describe, in order, the transformations that take $y = f(x)$ to $y = g(x)$.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[6.5cm]{Let $f(x) = (x - 3)^{2} + 1$ with the domain restricted to $x \geq 3$. Find $f^{-1}(17)$.}

\problem[0pt]{Let $f(x) = \dfrac{1}{x + 2}$ and $g(x) = \dfrac{1}{x} - 2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Simplify $(f \circ g)(x)$.
  \par\vspace*{3.0cm}\ansline
  \item[(b)] Explain what your answer says about the relationship between $f$ and $g$.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[6.5cm]{Let $f(x) = 2x + 1$ and $h(x) = f^{-1}(x - 4)$. Find $h(11)$.}

\problem[0pt]{Let $f(x) = 3x - 5$ and $g(x) = x^{2}$ with the domain of $g$ restricted to $x \geq 0$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $(g \circ f)^{-1}(9)$.
  \par\vspace*{4.0cm}\ansline
  \item[(b)] Explain why $(g \circ f)^{-1}$ equals $f^{-1} \circ g^{-1}$ rather than $g^{-1} \circ f^{-1}$.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}
