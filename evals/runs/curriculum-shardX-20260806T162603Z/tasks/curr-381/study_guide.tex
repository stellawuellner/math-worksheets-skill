\sstitleblock{Composing Functions from Symbolic Rules}

\skillheading{Skill 1 --- Composing at a number}

\begin{formulabox}
\textbf{Notation:} $(f \circ g)(a) = f\bigl(g(a)\bigr)$. The function written
\emph{closest to the input} runs first. Two machines in a row: work out the inner
output, then feed it into the outer rule.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 4x-1$ and $g(x) = x^2$, find
$(f \circ g)(3)$.
\step{Only one number has to travel through the chain, so evaluate the inner
machine at the input and pass its output on --- no algebra is needed.}
\step{$g(3) = 3^2 = 9$, then $f(9) = 4(9) - 1 = 36 - 1$.}

$\Rightarrow\quad \ans{35}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = x+7$ and $g(x) = 2x$, find
$(f \circ g)(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17}$}
\end{tryitbox}

\skillheading{Skill 2 --- Composing into a new rule}

\begin{formulabox}
\textbf{Method:} replace every $x$ in the \emph{outer} rule by the whole
\emph{inner} rule, in brackets, then simplify. The brackets are what protect a
squaring or a negative sign from being applied to only part of the inner rule.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = x^2-2$ and $g(x) = x+3$, find
$(f \circ g)(x)$.
\step{The whole inner rule is the input to $f$, so it goes in wherever $f$'s
formula shows an $x$ --- bracketed, because $f$ squares its input.}
\step{$f(g(x)) = (x+3)^2 - 2 = x^2 + 6x + 9 - 2$.}

$\Rightarrow\quad \ans{x^2 + 6x + 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 5x+2$ and $g(x) = x-4$, find
$(f \circ g)(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x - 18}$}
\end{tryitbox}

\skillheading{Skill 3 --- The two orders are different}

\begin{formulabox}
\textbf{Fact:} $f \circ g$ and $g \circ f$ are usually \emph{not} the same
function. Composition is not like multiplication: swapping the order changes
which operation happens first, and that changes the rule. Always check which
function the notation puts closest to the input.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 3x-2$ and $g(x) = x+5$, build both orders.
\step{Each order is a separate substitution, so build them one at a time and
compare the finished rules rather than guessing.}
\step{$f(g(x)) = 3(x+5) - 2 = 3x + 15 - 2$, while
$g(f(x)) = (3x-2) + 5$.}

$f \circ g:\ \ans{3x + 13}$ \qquad $g \circ f:\ \ans{3x + 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 4x+1$ and $g(x) = x-2$, build both
orders.\\[2pt]
\rotatebox{180}{\footnotesize check: $f \circ g: \ans{4x - 7}$, \ $g \circ f: \ans{4x - 1}$}
\end{tryitbox}

\skillheading{Skill 4 --- Solving with a composition}

\begin{formulabox}
\textbf{Method:} build the composed rule \emph{first}, so the equation becomes an
ordinary one in $x$, then solve it as usual. Finish by running your answer
through both machines to check it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 3x+1$ and $g(x) = x-2$, solve
$(f \circ g)(x) = 10$.
\step{Nothing can be solved while the unknown is still inside two functions, so
compose into a single rule before touching the equation.}
\step{$f(g(x)) = 3(x-2)+1 = 3x-5$, so $3x - 5 = 10$ and $3x = 15$. Check:
$g(5) = 3$ and $f(3) = 10$.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 2x-3$ and $g(x) = x+6$, solve
$(f \circ g)(x) = 21$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}
