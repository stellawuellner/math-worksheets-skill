\sstitleblock{Derivatives: Power, Product, and Quotient Rules}

\skillheading{1. Power rule (rewrite first)}

\begin{formulabox}
\textbf{Rule.} $\dfrac{d}{dx}x^n = nx^{n-1}$ for every real $n$, and
$\dfrac{d}{dx}\left[c\,f(x)\right] = c\,f'(x)$; the derivative of a constant is
$0$.\\[3pt]
\textbf{Rewrite before you differentiate:} $\sqrt{x} = x^{1/2}$, \
$\sqrt[3]{x} = x^{1/3}$, \ $\dfrac{1}{x^n} = x^{-n}$. The power rule cannot see
a radical or a reciprocal until it is a power.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = 3x^4 - \dfrac{5}{x}$.
\step{The second term is not written as a power, so rewrite it as $-5x^{-1}$
and then every term is power-rule ready.}
\step{$3x^4 \rightarrow 12x^3$; \ $-5x^{-1} \rightarrow +5x^{-2}$ (the
exponent $-1$ comes out front and drops to $-2$).}

$\Rightarrow\quad \ans{y' = 12x^3 + \dfrac{5}{x^2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = 2\sqrt{x} + x^3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y' = \dfrac{1}{\sqrt{x}} + 3x^2}$}
\end{tryitbox}

\skillheading{2. Product rule}

\begin{formulabox}
\textbf{Rule.} $(uv)' = u'v + uv'$ --- ``derivative of the first times the
second, plus the first times the derivative of the second''.\\[3pt]
The derivative of a product is \textbf{never} $u'v'$. Label $u$, $v$, $u'$,
$v'$ before substituting, and you will not lose a term.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = (x + 4)(x^2 - 2)$.
\step{Two factors, each of which is easy to differentiate on its own --- that
is the product rule's signature.}
\step{$u = x + 4$, $u' = 1$; $v = x^2 - 2$, $v' = 2x$. Then
$y' = 1(x^2 - 2) + (x + 4)(2x) = x^2 - 2 + 2x^2 + 8x$.}

$\Rightarrow\quad \ans{y' = 3x^2 + 8x - 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = (3x - 1)(x^2 + 5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y' = 9x^2 - 2x + 15}$}
\end{tryitbox}

\skillheading{3. Quotient rule}

\begin{formulabox}
\textbf{Rule.} $\left(\dfrac{u}{v}\right)' = \dfrac{u'v - uv'}{v^2}$ ---
``low $\cdot$ d-high minus high $\cdot$ d-low, all over low squared''.\\[3pt]
The subtraction makes the \textbf{order matter}: swapping the two products
flips the sign of the whole answer. Leave $v^2$ factored; expanding it hides
the structure and helps nothing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $f(x) = \dfrac{x + 2}{x - 3}$.
\step{A single fraction with a variable in the denominator, and no easy
rewrite as a power --- use the quotient rule.}
\step{$u = x + 2$, $u' = 1$; $v = x - 3$, $v' = 1$. Numerator:
$1(x - 3) - (x + 2)(1) = x - 3 - x - 2 = -5$.}

$\Rightarrow\quad \ans{f'(x) = \dfrac{-5}{(x - 3)^2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $f(x) = \dfrac{2x}{x + 1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f'(x) = \dfrac{2}{(x + 1)^2}}$}
\end{tryitbox}

\skillheading{4. Where the rules come from: the limit definition}

\begin{formulabox}
\textbf{Definition.} $f'(x) = \displaystyle\lim_{h \to 0}
\frac{f(x + h) - f(x)}{h}$.\\[3pt]
The plan is always the same: expand $f(x + h)$, subtract $f(x)$ so the
constant terms cancel, factor $h$ out of what is left, cancel it, and only
then let $h \to 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $f'(x)$ from the definition for $f(x) = x^2$.
\step{Substituting $h = 0$ straight away gives $0/0$, so the $h$ in the
denominator has to be cancelled first.}
\step{Numerator: $(x + h)^2 - x^2 = 2xh + h^2 = h(2x + h)$. Cancel the $h$ to
get $2x + h$, then let $h \to 0$.}

$\Rightarrow\quad \ans{f'(x) = 2x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $f'(x)$ from the definition for $f(x) = 4x^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f'(x) = 8x}$}
\end{tryitbox}
