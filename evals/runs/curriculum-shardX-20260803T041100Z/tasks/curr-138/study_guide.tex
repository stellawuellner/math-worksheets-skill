\sstitleblock{Multiplying and Dividing Decimals by Place Value}

\skillheading{Skill 1 --- Multiply or divide by 10, 100, 1000}
\begin{formulabox}\textbf{Rule:} the digits move, the decimal point stays put.\\[4pt]
$\times 10$ moves every digit \textbf{one place left} (each digit is worth ten times more);
$\times 100$ moves it two places left. $\div 10$ moves every digit \textbf{one place right};
$\div 100$ moves it two places right. Fill empty places with $0$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $3.6 \times 100$
\step{Multiplying by 100 makes each digit worth 100 times as much, so every digit slides two places to the left.}
\step{The 3 ones become 3 hundreds and the 6 tenths become 6 tens: $\ans{360}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $47.5 \div 100$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.475}$}\end{tryitbox}

\skillheading{Skill 2 --- Multiply two decimals}
\begin{formulabox}\textbf{Rule:} multiply the digits as whole numbers, then place the point by place value.\\[4pt]
tenths $\times$ tenths $=$ hundredths \quad and \quad hundredths $\times$ tenths $=$ thousandths.\par
Shortcut: the product has as many decimal places as the two factors have together.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $0.6 \times 0.4$
\step{Both factors are tenths, so multiply the digits first and expect an answer in hundredths.}
\step{$6 \times 4 = 24$, and 24 hundredths is $\ans{0.24}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $1.2 \times 0.5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.6}$}\end{tryitbox}

\skillheading{Skill 3 --- Divide by a decimal}
\begin{formulabox}\textbf{Rule:} rewrite both numbers in the \textbf{same place-value unit}, then divide whole numbers.\\[4pt]
$3.5 \div 0.7$ becomes ``how many 7 tenths fit in 35 tenths?''\par
Dividing by a number smaller than 1 gives a quotient \emph{larger} than the number you started with.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $3.5 \div 0.7$
\step{Both numbers can be said in tenths, and counting how many 7 tenths fit inside 35 tenths turns this into whole-number division.}
\step{$35 \div 7 = 5$, and the check $5 \times 0.7 = 3.5$ works, so the quotient is $\ans{5}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $5.6 \div 0.8$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}\end{tryitbox}

\skillheading{Skill 4 --- Estimate first, then judge the size}
\begin{formulabox}\textbf{Rule:} round each number to the nearest whole, multiply or divide those, and use the result to test where the decimal point belongs.\par
Multiplying by a number \emph{smaller than one} makes the result smaller;
dividing by a number \emph{smaller than one} makes the result larger.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Estimate $5.8 \times 3.1$
\step{An estimate only needs whole numbers, so round each factor to the nearest whole before multiplying.}
\step{$5.8$ rounds to 6 and $3.1$ rounds to 3, giving $6 \times 3 = \ans{18}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Estimate $7.2 \times 4.9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} A product of two numbers below 1 must be
smaller than both of them. If your answer is bigger, the decimal point moved the wrong way.
\end{watchoutbox}
