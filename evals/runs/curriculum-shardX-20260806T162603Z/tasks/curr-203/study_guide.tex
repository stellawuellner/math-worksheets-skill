\sstitleblock{Approximating Square Roots}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Bracketing a root between whole numbers}

\begin{formulabox}
\textbf{Rule:} squaring keeps order. If $p^2 < N < q^2$ then $p < \sqrt{N} < q$.
\par\smallskip
\textbf{Method:} find the perfect square just below the radicand and the one just above
it. Their roots are the two consecutive whole numbers the root lies between.
\par\smallskip
\textbf{Squares to have ready:} $1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121, 144$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Between which whole numbers does $\sqrt{52}$ lie?
\step{The radicand is not a perfect square, so no exact value exists --- the useful answer is the pair of squares it falls between.}
\step{$49 < 52 < 64$, and taking square roots across the inequality keeps the order.}

$\Rightarrow\quad \ans{7} < \sqrt{52} < \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Between which whole numbers does $\sqrt{85}$ lie?\\[2pt]
\rotatebox{180}{\footnotesize check: between $\ans{9}$ and $\ans{10}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Squeezing a root to a decimal}

\begin{formulabox}
\textbf{Method:} once the root is bracketed, test decimals inside the bracket by
\emph{squaring} them. When two neighbouring decimals trap the radicand, the root is
between them --- round to whichever is asked for.
\par\smallskip
Squaring a candidate is the check. Never round a root you have not squared back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Give $\sqrt{40}$ to the nearest hundredth.
\step{The radicand sits between two perfect squares, so I squeeze inside that bracket by squaring tenths, then hundredths.}
\step{$6.3^2 = 39.69$ and $6.4^2 = 40.96$, so the root is in the low sixes; then $6.32^2 = 39.9424$ and $6.33^2 = 40.0689$ trap it.}

$\Rightarrow\quad \sqrt{40} \approx \ans{6.32}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Give $\sqrt{55}$ to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7.42}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} halving the radicand is not a square root. Always square your
estimate and compare it with the number under the radical --- that one check catches
almost every wrong answer on this topic.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Using the estimate in a measurement}

\begin{formulabox}
\textbf{Rule:} a square of area $A$ has side $\sqrt{A}$, in the matching length unit ---
square feet give feet, square metres give metres.
\par\smallskip
Round only at the end, and write the unit on the answer. When material has to
\emph{cover} the area, round \textbf{up} to the next whole size.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A square patio covers $90$ square feet. Give its side to the
nearest tenth of a foot.
\step{Area of a square is side times side, so the side is the square root of the area --- and the root is irrational, so the answer is an estimate.}
\step{$9^2 = 81$ and $10^2 = 100$ bracket it; squeezing with tenths, $9.4^2 = 88.36$ and $9.5^2 = 90.25$ trap the area.}

$\Rightarrow\quad \text{side} \approx \ans{9.5 \text{ ft}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A square deck covers $130$ square feet. Give its side to the
nearest tenth of a foot.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11.4 \text{ ft}}$}
\end{tryitbox}
