\wstitleblock{Rigid Motions and Constructions --- Planning and Justifying Compass-and-Straightedge Work}{High School Geometry}{}

\begin{center}\small\textit{Writing a construction as an ordered plan, justifying it with
the congruent triangles the arcs create, and checking the finished claim on
coordinates}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A \textbf{plan} is a numbered list of moves. Each step must say where the compass
        point sits, whether the opening changes, and what gets drawn. ``Draw an arc'' is
        not a step until it says \emph{centred where} and \emph{with what radius}.
  \item Legal tools only: an unmarked straightedge and a compass. No ruler measurements,
        no protractor.
  \item A \textbf{justification} names the congruent triangles or the equal radii that
        force the result, and the criterion (SSS, SAS, ASA) or definition being used.
        ``It looks right'' justifies nothing.
  \item The \textbf{coordinate check} problems do not ask you to construct anything. They
        hand you the output of a construction and ask you to test the property that
        construction is supposed to guarantee. Say which property you are testing before
        you compute, and round lengths to the nearest hundredth.
  \item The figures below are for reference only. They carry no measurements: use the
        letters, and take any numbers from the problem you are on.
\end{itemize}
\normalsize

\bigskip

\problem[8cm]{Write the plan for constructing the \textbf{perpendicular bisector} of
segment $AB$ with compass and straightedge.

\smallskip
Number your steps. For every arc, say where the compass point sits and whether the opening
changed from the step before. Then state the one condition the opening must satisfy for the
construction to work at all, and say what goes wrong if it does not.\noansline}

\problem[8cm]{Write the plan for constructing the \textbf{bisector of angle} $DEF$.

\smallskip
Number your steps. Mark clearly which two arcs must share the same compass opening, and
say why the construction fails if they do not.\noansline}

\problem[8cm]{A perpendicular bisector was constructed on segment $AB$, where
$A(-2, 1)$ and $B(6, 7)$. The construction produced the point $P(-1, 8)$.

\smallskip
Name the property a perpendicular bisector guarantees about $P$, then test it: compute
$PA$ and $PB$.}

\problem[9cm]{\textbf{Justify the perpendicular-bisector construction.} In the finished
figure the two arc intersections are called $M$ and $N$, and the endpoints of the segment
are $A$ and $B$.

\smallskip
(a) Name two congruent triangles in that figure and the criterion that makes them
congruent, saying exactly which lengths are equal \emph{by construction}.

\smallskip
(b) Explain how that congruence forces the line $MN$ to be both perpendicular to $AB$ and
to pass through its midpoint.\noansline}

\problem[8cm]{A regular hexagon is inscribed in a circle of radius 10 centred at the
origin, by walking the compass --- still open to the radius --- around the circle. Two
consecutive vertices land at $(10, 0)$ and $(5, 8.66)$.

\smallskip
State the claim that construction depends on, then test it by computing the distance
between those two vertices.}

\problem[9cm]{Write the plan for constructing the line through point $P$ that is
\textbf{parallel to line} $\ell$, using the copy-an-angle method.

\smallskip
Number your steps. Say which step creates the pair of corresponding angles, and say why a
protractor may not be used to make the copy.\noansline}

\problem[8cm]{A segment $AB$ with $A(1, 2)$ and $B(7, 10)$ is copied onto a ray starting
at $C(3, -4)$ by opening the compass to $AB$ and swinging that opening from $C$. The copy
lands with endpoint $D(11, 2)$.

\smallskip
Name the property the copy construction guarantees, then test it: compute $AB$ and $CD$.}

\problem[8cm]{\textbf{Justify the equilateral-triangle construction.} Given segment $AB$,
you open the compass to $AB$, draw an arc centred at $A$, draw a second arc centred at $B$
with the \emph{same} opening, and call their intersection $C$.

\smallskip
Prove that triangle $ABC$ is equilateral. Your argument must use the definition of a
circle, not the appearance of the drawing.\noansline}

\problem[9cm]{The perpendicular bisectors of two sides of the triangle with vertices
$A(0, 0)$, $B(8, 0)$ and $C(0, 6)$ were constructed, and they meet at $O(4, 3)$.

\smallskip
State what the intersection of two perpendicular bisectors is guaranteed to be, then test
it: compute $OA$, $OB$ and $OC$.}

\problem[10cm]{\textbf{Challenge.} Given segment $AB$, plan a construction of the point
$Q$ on $AB$ with $AQ$ equal to one quarter of $AB$.

\smallskip
(a) Write the full plan, numbered.

\smallskip
(b) Justify it: for each bisection you use, name the congruence or equidistance argument
that guarantees the point you get really is a midpoint.

\smallskip
(c) A classmate says the same repeated-bisection idea gives a way to trisect any segment.
Explain why it does not.\noansline}
