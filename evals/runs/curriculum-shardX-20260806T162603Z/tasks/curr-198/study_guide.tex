\sstitleblock{Random Samples and Simple Probability Models}

% =================== SKILL 1 ===================
\skillheading{A Simple Probability Model}

\begin{formulabox}
\textbf{Rule:} when every outcome is equally likely,
\[
  P(\text{event}) = \frac{\text{favourable outcomes}}{\text{total outcomes}}
\]
Every probability lands between 0 and 1. Reduce the fraction. ``Not the event''
has probability $1 - P(\text{event})$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tray holds 15 identical cards and 6 of them are
yellow. One card is drawn at random. Find the probability it is yellow.
\step{The cards are identical, so every card is equally likely --- count the
favourable ones over the total.}
\step{Six of the fifteen cards are yellow, so the probability is
$\frac{6}{15}$, which reduces to $\ans{2/5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A box holds 24 pens and 9 of them are red. Find the
probability of drawing a red pen.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3/8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Scaling a Random Sample up to a Population}

\begin{formulabox}
\textbf{Rule:} a random sample estimates the whole population by keeping the
same fraction:
\[
  \text{estimate} = \frac{\text{count in the sample}}{\text{size of the sample}}
  \times \text{size of the population}
\]
Write the sample fraction first, then multiply. Always state the units of the
answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In a random sample of 40 books from a library of 500, 12
were poetry. Estimate how many poetry books the library holds.
\step{The sample was random, so the fraction of poetry in it is a fair estimate
of the fraction in the library.}
\step{The sample fraction is $\frac{12}{40} = 0.3$, so the estimate is
$0.3 \times 500 = \ans{150}$ books.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In a random sample of 35 apples from a crop of 400, 7
were bruised. Estimate the number of bruised apples in the crop.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{80}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Judging Whether a Sample Can Be Trusted}

\begin{formulabox}
\textbf{Rule:} a sample is \textbf{random} when every member of the population
has the same chance of being chosen. If it does not, the estimate describes the
group you asked, not the population --- and no amount of correct arithmetic
repairs that.\par
\textbf{Warning signs:} the group was chosen because of the very thing being
measured, or people chose themselves by turning up, replying, or volunteering.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A town of 450 residents is sampled by numbering every
resident and drawing 30 numbers at random. What chance does one named resident
have of being chosen?
\step{Every resident is numbered, so every one is equally likely --- that is
what makes the sample random, and it also makes the chance easy to count.}
\step{Thirty of the 450 residents are drawn, so a named resident has probability
$\frac{30}{450} = \ans{1/15}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A school numbers all 500 students and draws 25 at random.
What is the probability that one named student is chosen?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1/20}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} surveying the basketball team about the school's
favourite sport gives a perfectly correct calculation and a useless estimate.
Check how the sample was chosen \emph{before} you scale it up.
\end{watchoutbox}
