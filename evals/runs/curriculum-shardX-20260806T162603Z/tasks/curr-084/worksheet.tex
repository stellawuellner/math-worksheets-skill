\wstitleblock{Fractions: Equal Parts Fix-It}{Grade 3}{}

\noindent\textbf{Directions:}
\begin{itemize}
  \item A fraction can only be named when the whole is cut into \emph{equal} parts.
  \item Some students in these problems made a mistake. Find it, fix it, and show your thinking in the work space.
\end{itemize}

% ---------------------------------------------------------------- Problem 1
\problem[2.6cm]{This rectangle is cut into 4 equal parts, and 1 part is shaded.
Write the fraction of the rectangle that is shaded.
\begin{center}
\begin{tikzpicture}
  \fill[blue!20] (0,0) rectangle (1.5,1.2);
  \draw[thick] (0,0) rectangle (6,1.2);
  \draw (1.5,0) -- (1.5,1.2);
  \draw (3,0) -- (3,1.2);
  \draw (4.5,0) -- (4.5,1.2);
\end{tikzpicture}
\end{center}}

% ---------------------------------------------------------------- Problem 2
\problem[3cm]{Mia cut this circle into three parts and shaded one part.
The parts are \emph{not} all the same size. Mia says the shaded part is one third
of the circle. Say why Mia cannot call the shaded part one third yet.
\begin{center}
\begin{tikzpicture}
  \fill[blue!20] (0,0) -- (90:1.2) arc (90:180:1.2) -- cycle;
  \draw[thick] (0,0) circle (1.2);
  \draw (0,0) -- (90:1.2);
  \draw (0,0) -- (180:1.2);
  \draw (0,0) -- (330:1.2);
\end{tikzpicture}
\end{center}}

% ---------------------------------------------------------------- Problem 3
\problem[3cm]{Rectangle A and Rectangle B are the same size. Each one is cut into
four parts, and each one has one part shaded.
\begin{center}
\begin{tikzpicture}
  % Rectangle A: equal quarters
  \fill[blue!20] (0,0) rectangle (1.125,1.2);
  \draw[thick] (0,0) rectangle (4.5,1.2);
  \draw (1.125,0) -- (1.125,1.2);
  \draw (2.25,0) -- (2.25,1.2);
  \draw (3.375,0) -- (3.375,1.2);
  \node[below] at (2.25,-0.1) {A};
  % Rectangle B: unequal parts
  \fill[blue!20] (8.2,0) rectangle (10.0,1.2);
  \draw[thick] (5.5,0) rectangle (10.0,1.2);
  \draw (6.1,0) -- (6.1,1.2);
  \draw (7.0,0) -- (7.0,1.2);
  \draw (8.2,0) -- (8.2,1.2);
  \node[below] at (7.75,-0.1) {B};
\end{tikzpicture}
\end{center}
(a) What fraction of Rectangle A is shaded? \ansblank\par
(b) Circle the letter of the rectangle that \emph{truly} shows one fourth,
and tell how you know.}

% ---------------------------------------------------------------- Problem 4
\problem[3cm]{Deshawn split this rectangle into three parts using the heavy lines,
and he shaded the small part. He says: ``The shaded part is $\tfrac{1}{3}$
because it is 1 part out of 3.''
\begin{center}
\begin{tikzpicture}[scale=1.1]
  \fill[blue!20] (0,1) rectangle (1,2);
  \draw[gray!55, thin] (1,0) -- (1,1);
  \draw[gray!55, thin] (2,0) -- (2,2);
  \draw[gray!55, thin] (0,1) -- (3,1);
  \draw[very thick] (1,1) -- (1,2);
  \draw[very thick] (0,1) -- (2,1);
  \draw[very thick] (2,0) -- (2,1);
  \draw[thick] (0,0) rectangle (3,2);
\end{tikzpicture}
\end{center}
(a) Count the small equal squares. How many small squares make the whole
rectangle? \ansblank\par
(b) What fraction of the rectangle is \emph{really} shaded? \ansblank\par
(c) Name Deshawn's mistake.}

% ---------------------------------------------------------------- Problem 5
\problem{Priya drew a number line from 0 to 1. Her line is 8 cm long. She made
tick marks that are \emph{not} evenly spaced, and she labeled the first mark
$\tfrac{1}{4}$.
\begin{center}
\begin{tikzpicture}
  \draw[thick] (0,0) -- (8,0);
  \draw[thick] (0,-0.15) -- (0,0.15) node[above] {$0$};
  \draw[thick] (8,-0.15) -- (8,0.15) node[above] {$1$};
  \draw (1.2,-0.12) -- (1.2,0.12);
  \node[below] at (1.2,-0.12) {$\tfrac{1}{4}$};
  \draw (3.0,-0.12) -- (3.0,0.12);
  \draw (4.4,-0.12) -- (4.4,0.12);
\end{tikzpicture}
\end{center}
(a) To show fourths correctly, the tick marks must cut the line into 4 equal
lengths. How many centimeters apart should Priya's tick marks be?
\par\vspace*{2.2cm}\answerline{cm}\par
(b) Say what was wrong with Priya's number line.
\par\vspace*{2.2cm}}

% ---------------------------------------------------------------- Problem 6
\problem{Ben folded a paper strip into two parts. He says each part is one half.
Then he measures: one part is 12 cm long and the other part is 8 cm long.
\begin{center}
\begin{tikzpicture}
  \draw[thick] (0,0) rectangle (5,1);
  \draw[thick] (3,0) -- (3,1);
  \node at (1.5,0.5) {12 cm};
  \node at (4,0.5) {8 cm};
\end{tikzpicture}
\end{center}
(a) Is Ben right? Say how you know.
\par\vspace*{2.2cm}
(b) If the same strip were folded into 2 \emph{equal} parts, how long would
each part be?
\par\vspace*{2.2cm}\answerline{cm}}

% ---------------------------------------------------------------- Problem 7
\problem[3.2cm]{Tia's garden is split by heavy lines into 4 sections that are
\emph{not} all the same size. The shaded tomato section covers 4 small squares.
Tia says: ``Tomatoes take up $\tfrac{1}{4}$ of the garden, because tomatoes are
1 section out of 4.''
\begin{center}
\begin{tikzpicture}[scale=1.1]
  \fill[blue!20] (0,0) rectangle (2,2);
  \draw[gray!55, thin] (1,0) -- (1,2);
  \draw[gray!55, thin] (3,0) -- (3,2);
  \draw[gray!55, thin] (0,1) -- (4,1);
  \draw[very thick] (2,0) -- (2,2);
  \draw[very thick] (2,1) -- (4,1);
  \draw[very thick] (3,1) -- (3,2);
  \draw[thick] (0,0) rectangle (4,2);
  \node at (1,1) {\small tomatoes};
\end{tikzpicture}
\end{center}
(a) Use the small squares. What fraction of the garden do the tomatoes
\emph{really} cover? \ansblank\par
(b) Explain what Tia got wrong.}

% ---------------------------------------------------------------- Problem 8
\problem[2.6cm]{Now draw your own correct model. The grid below shows 3 rows of
4 small squares.
\begin{center}
\begin{tikzpicture}[scale=1.1]
  \draw[gray!55, thin] (1,0) -- (1,3);
  \draw[gray!55, thin] (2,0) -- (2,3);
  \draw[gray!55, thin] (3,0) -- (3,3);
  \draw[gray!55, thin] (0,1) -- (4,1);
  \draw[gray!55, thin] (0,2) -- (4,2);
  \draw[thick] (0,0) rectangle (4,3);
\end{tikzpicture}
\end{center}
(a) Draw straight lines to cut the grid into six \emph{equal} parts, then shade
one part.\par
(b) How many small squares are in each equal part? \ansblank\par
(c) What fraction of the whole rectangle is one small square? \ansblank}
