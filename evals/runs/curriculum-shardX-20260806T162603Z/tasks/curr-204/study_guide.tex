% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Testing whether two roots may be added}

\begin{formulabox}
\textbf{The rule that is broken most often:} a radical does \emph{not} distribute across
addition.
\[ \sqrt{a} + \sqrt{b} \;\neq\; \sqrt{a + b} \]
\textbf{How to test it:} pick numbers whose roots you know, work out both sides, and
compare. One counterexample settles the question for good.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $\sqrt{4} + \sqrt{9}$ the same as $\sqrt{4 + 9}$?
\step{The claim is a general one, so a single pair of numbers with known roots is enough to decide it --- pick perfect squares and compute both sides.}
\step{The left side is two whole roots added; the right side is one root of the total, which is not a perfect square and lands near three and a half.}

Left side $= \ans{5}$, \quad right side $= \sqrt{13} \approx \ans{3.61}$. Not equal.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Work out $\sqrt{25} + \sqrt{144}$, then $\sqrt{169}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17}$ and $\ans{13}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Combining like radical terms}

\begin{formulabox}
\textbf{Rule:} treat the radical as a name. Terms with the \emph{same} radical part are
like terms, so add or subtract only the counts in front:
\[ p\sqrt{k} + q\sqrt{k} = (p + q)\sqrt{k} \]
The radical part never changes when like terms are collected. Terms with different
radical parts simply do not combine.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $6\sqrt{7} + 2\sqrt{7}$.
\step{Both terms carry the same radical part, so they are like terms and only the counts in front are added.}
\step{Counting: six of them plus two more of them makes eight of them, and the radical part is carried along unchanged.}

$\Rightarrow\quad 6\sqrt{7} + 2\sqrt{7} = \ans{8\sqrt{7}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $9\sqrt{5} - 4\sqrt{5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5\sqrt{5}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the two classic slips are adding the numbers \emph{under} the
radicals, and doing that on top of adding the counts in front. Check every answer by
asking: is the radical part exactly what it was before?
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Simplify first, then combine}

\begin{formulabox}
\textbf{Method:} radical parts that look different can become the same after simplifying.
Pull the largest perfect-square factor out of each radicand, then collect like terms.
\[ \sqrt{p \cdot k} = \sqrt{p} \cdot \sqrt{k} \quad\text{when } p \text{ is a perfect square} \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\sqrt{12} + \sqrt{27}$.
\step{The two radical parts differ, so nothing can be combined yet --- pull out perfect-square factors first and see whether they match.}
\step{Each radicand is a perfect square times three, so each term becomes a whole-number count of the same radical, and the two counts then add.}

$\Rightarrow\quad \sqrt{12} + \sqrt{27} = \ans{5\sqrt{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\sqrt{20} + \sqrt{45}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5\sqrt{5}}$}
\end{tryitbox}
