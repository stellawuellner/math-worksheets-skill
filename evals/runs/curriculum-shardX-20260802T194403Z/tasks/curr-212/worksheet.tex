\wstitleblock{Linear Equations with Distribution and Fractions}{Grade 8 / Pre-Algebra --- Expressions and Equations}{}

\noindent\small\textbf{Skills practised:} Distributing first, clearing fractions with the LCD,
variables on both sides, and equivalent rewrites.
\smallskip

\noindent\textbf{Worked pattern --- follow this order every time.}
\smallskip

\noindent\begin{minipage}{\linewidth}\small
\textbf{Solve} $\dfrac{x}{2} + \dfrac{1}{3} = 2(x - 1)$.\par\vspace{2pt}
\textbf{Step 1 --- clear fractions.} The denominators are $2$ and $3$, so the LCD is $6$.
Multiply \emph{every term on both sides} by $6$: \quad $3x + 2 = 12(x - 1)$.\par\vspace{1pt}
\textbf{Step 2 --- distribute.} Multiply through the brackets: \quad $3x + 2 = 12x - 12$.
\par\vspace{1pt}
\textbf{Step 3 --- collect.} Variable terms to one side, numbers to the other: \quad $14 = 9x$.
\par\vspace{1pt}
\textbf{Step 4 --- isolate and check.} $x = \dfrac{14}{9}$. Substitute back into the
\emph{original} equation to check.
\end{minipage}
\smallskip

\noindent\small Show every step. Leave answers as exact fractions where they are not whole numbers.
\normalsize

\problem[4.5cm]{Solve for $x$: \quad $3(x + 4) = 21$}

\problem[4.5cm]{Solve for $x$: \quad $-2(x - 5) = 14$}

\problem[5cm]{Solve for $x$: \quad $\dfrac{x}{3} + \dfrac{x}{6} = 5$
\par\smallskip\noindent\small (Start by multiplying every term by the LCD.)\normalsize}

\problem[5cm]{Solve for $x$: \quad $4 - 2(x + 3) = 10$
\par\smallskip\noindent\small (The subtraction sign in front of the brackets belongs to the $2$.)
\normalsize}

\problem[5cm]{The expression $\dfrac{x}{4} + \dfrac{2}{3}$ appears on one side of an equation.
Multiply it by the LCD, $12$, and write the equivalent expression with no fractions left in it.}

\problem[5cm]{Solve for $x$: \quad $3(x - 2) = 2(x + 4)$}

\problem[5cm]{Solve for $x$: \quad $\dfrac{3x}{4} - 2 = \dfrac{x}{2}$}

\problem[5cm]{Solve for $x$: \quad $\dfrac{1}{2}(6x - 8) = 11$}

\problem[5cm]{Expand and simplify $7 - 3(2x - 5)$, writing your answer in the form
$a + bx$ or $bx + a$.}

\problem[6cm]{Solve for $x$: \quad $\dfrac{2}{3}(x + 6) = \dfrac{1}{2}(x + 2) + 5$}

\problem[6cm]{Solve for $x$: \quad $\dfrac{x + 3}{2} - \dfrac{x - 5}{4} = 5$
\par\smallskip\noindent\small (A fraction bar groups the whole numerator --- keep the brackets when
you multiply.)\normalsize}

\problem[6cm]{\textbf{Challenge.} Solve for $x$, then check your solution in the original equation:
\quad $\dfrac{2}{5}(10x - 15) = 3(x - 1) + 4$}
