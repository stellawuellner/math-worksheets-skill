\wstitleblock{Numerical and Algebraic Expressions --- Distributing and Combining Without Breaking the Rules}{Grades 6--7}{}

\noindent\small Two rules do most of the work on this page, and breaking either
one is the most common way to lose an expression problem.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Distributing} means multiplying \emph{every} term inside the
        parentheses --- including the sign in front of each one.
  \item \textbf{Like terms} have exactly the same letter part. $5x$ and $2x$ are
        like; $5x$ and $3y$ are not, and neither are $5x$ and $5$.
  \item \textbf{The substitution test:} if two expressions are really
        equivalent, they give the same value for every number you put in. One
        disagreement proves they are different expressions.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{Write $4(x+7)$ without parentheses. Multiply every term inside by
$4$.}

\problem[5cm]{\textbf{Find and fix.} Kim writes $3(x+5) = 3x+5$.\par
Test both expressions at $x = 4$. Kim's version gives $17$. Work out the value
of $3(x+5)$ at $x = 4$ and write it on the answer line, then say in a few words
which term Kim forgot to multiply.}

\problem[5cm]{\textbf{Find and fix.} Devon writes $5x+3y = 8xy$.\par
Test both expressions at $x = 2$ and $y = 3$. Devon's version gives $48$. Work
out the value of $5x+3y$ at those values and write it on the answer line, then
say in a few words why $5x$ and $3y$ cannot be pulled into one term.}

\problem[5cm]{Simplify by combining like terms only:\par
\vspace{0.1cm}
\hspace{1cm}$7a + 4 - 2a + 9$}

\problem[6cm]{\textbf{Find and fix.} Simplify $8-2(x-3)$.\par
Nadia writes $8-2x-6$, which is $2-2x$, and at $x=1$ her version gives $0$.\par
The $-2$ must be multiplied by \emph{both} terms in the parentheses. Work out
the correct value of $8-2(x-3)$ at $x=1$ and write it on the answer line.}

\problem[6cm]{Simplify completely:\par
\vspace{0.1cm}
\hspace{1cm}$5(2x-3) + 4x$\par
\vspace{0.1cm}
Distribute first, then combine only the terms that are alike.}

\problem[6cm]{Factor $6x+15$ by taking out the greatest common factor. Then
check your answer by distributing it back --- the check should return $6x+15$
exactly.}

\problem[8cm]{\textbf{Explain, then simplify.}\par
\textbf{(a)} A classmate insists that $5x+3y$ ``must be able to become one
term.'' Explain, in your own words, why two unlike terms cannot be combined.
Remember that a letter stands for a number nobody has told you yet.\par
\textbf{(b)} Simplify $3(2x-4) - 2(x-5)$, then check your simplified expression
by substituting $x = 3$ into it. Write that value on the answer line. (A
classmate who keeps both signs inside the second bracket gets $-10$ here.)}
