\wstitleblock{Selecting Ratio Representations for Comparisons}{Grade 6}{}

\begin{center}
\textit{Choosing a representation: unit-rate comparisons, scaling to a common amount, and tape-diagram totals}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item You know three ratio tools: \textbf{unit rates}, \textbf{ratio tables /
        double number lines} (scaling to a common amount), and \textbf{tape
        diagrams} (splitting totals). For each problem, pick the tool that fits
        and show it in your work space.
  \item After the warm-up, the problem types are mixed on purpose — part of the
        job is choosing the method.
\end{itemize}

\problem[4cm]{\textbf{Warm-up.} A box of 6 pencils costs 90 cents. How many
cents does one pencil cost?}

\problem[5cm]{\textbf{Warm-up.} A cocoa recipe uses 2 cups of cocoa for every 5
cups of milk. Scale the recipe up: how many cups of cocoa go with 20 cups of
milk?}

\problem[5.5cm]{\textbf{Warm-up.} Two friends split 42 trading cards in the
ratio $3{:}4$. How many cards are in the \textbf{larger} share?}

\problem[5cm]{Brand A sells 4 pounds of rice for 12 dollars. Brand B sells 5
pounds for 10 dollars.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] Brand A, dollars per pound: \ansblank
  \item[(b)] Brand B, dollars per pound: \ansblank
  \item[(c)] The cheaper brand per pound is \ansblank
\end{itemize}}

\problem[5.5cm]{Mia reads 20 pages in 8 minutes. Noah reads 27 pages in 12
minutes. Scale both readers to the same 24 minutes to compare
them.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] pages Mia reads in 24 minutes: \ansblank
  \item[(b)] pages Noah reads in 24 minutes: \ansblank
  \item[(c)] the faster reader is \ansblank
\end{itemize}}

\problem[6cm]{Team A splits a 60-dollar prize in the ratio $2{:}3$. Team B
splits a 63-dollar prize in the ratio $4{:}5$.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] larger share on team A, in dollars: \ansblank
  \item[(b)] larger share on team B, in dollars: \ansblank
  \item[(c)] Which of those two larger shares is bigger? \ansblank
\end{itemize}}

\problem[5.5cm]{Hiker A covers 9 kilometers every 2 hours. Hiker B covers 12
kilometers every 3 hours. Both hike for 6 hours.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] hiker A's distance in 6 hours, in km: \ansblank
  \item[(b)] hiker B's distance in 6 hours, in km: \ansblank
  \item[(c)] Who is ahead after 6 hours? \ansblank
\end{itemize}}

\problem[5cm]{A pack of 8 yogurts costs 6 dollars (600 cents). A single yogurt
costs 80 cents.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] cents per yogurt when buying the pack: \ansblank
  \item[(b)] the better buy is \ansblank
\end{itemize}}

\problem[5cm]{Shelter A houses cats and dogs in the ratio $5{:}3$, with 16
animals in all. Shelter B houses 9 cats.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] number of cats at shelter A: \ansblank
  \item[(b)] Which shelter has more cats? \ansblank
\end{itemize}}

\problem[6.5cm]{A punch mixes syrup and soda in the ratio $3{:}5$, and a full
bowl holds 32 cups of punch. How many cups of syrup are in the bowl? In the work
space, solve it with the representation of your choice, and say which
representation you chose and why it fits this problem.}

\problem[5.5cm]{The express train covers 210 kilometers in 3 hours. The local
train covers 140 kilometers in 2 hours. A classmate says the express train must
be faster because 210 is bigger than 140.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] express train speed, km per hour: \ansblank
  \item[(b)] local train speed, km per hour: \ansblank
  \item[(c)] Write $<$, $>$, or $=$ to compare the speeds: \ansblank
\end{itemize}}

\problem[7cm]{\textbf{Synthesis challenge.} Recipe A mixes 2 cups of lemon juice
for every 3 cups of water. Recipe B mixes 3 cups of lemon juice for every 5 cups
of water.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] Which recipe tastes stronger? (Compare the juice-per-water
             fractions.) \ansblank
  \item[(b)] You make 40 cups (juice plus water) of the \textbf{stronger}
             recipe. How many cups of lemon juice do you need? \ansblank
  \item[(c)] In the work space, explain which representation you used for each
             part.
\end{itemize}}
