\sstitleblock{Estimating Differences and Checking Reasonableness}

\skillheading{1. Estimate a difference by rounding}

\begin{formulabox}
\textbf{The rule:} round \emph{both} numbers to the same place, then subtract the
rounded numbers. Write $\approx$ (``is about''), not $=$.\\[3pt]
\textbf{Nearest hundred:} look at the \emph{tens} digit --- $5$ or more rounds up,
less than $5$ rounds down.\\[2pt]
\textbf{Nearest ten:} look at the \emph{ones} digit, same rule.\\[3pt]
Rounding only one of the two numbers is the most common mistake; the estimate
then drifts far from the real difference.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Estimate $583 - 216$ to the nearest hundred.
\step{The question asks about how many, not exactly how many, so round each
number first and subtract the easy numbers instead of regrouping.}
\step{$583 \to 600$ (tens digit $8$); $216 \to 200$ (tens digit $1$).}
\step{$600 - 200 = \ans{400}$, so the exact difference should land near $400$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Estimate $742 - 385$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{300}$}
\end{tryitbox}

\skillheading{2. Is my exact answer reasonable?}

\begin{formulabox}
\textbf{The rule:} the estimate is a \emph{safety net}. Find the exact
difference, then hold it next to the estimate.\\[3pt]
Close to the estimate $\Rightarrow$ the answer is reasonable.\\[2pt]
Far from the estimate $\Rightarrow$ a trade was probably missed --- go back and
rework the subtraction before you write the answer down.\\[3pt]
``Close'' means a few tens away when you rounded to the nearest hundred.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A jar holds $405$ beads. $168$ beads are taken out. How
many beads are left, and is the answer reasonable?
\step{Estimate first so there is something to compare the exact answer with,
then subtract exactly.}
\step{Estimate: $400 - 200 = 200$ beads. Exact: trade across the zero ---
$405$ becomes $3$ hundreds, $9$ tens, $15$ ones; $15 - 8 = 7$, $9 - 6 = 3$,
$3 - 1 = 2$.}
\step{The exact answer $\ans{237}$ beads sits close to the estimate of $200$
beads, so it is reasonable.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $612$ marbles, $289$ given away. Find the exact number
left and check it against your estimate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{323}$}
\end{tryitbox}

\skillheading{3. Check a difference by adding}

\begin{formulabox}
\textbf{The rule:} addition undoes subtraction.\\[3pt]
If $\text{start} - \text{taken away} = \text{answer}$, then
$\text{answer} + \text{taken away} = \text{start}$.\\[3pt]
Add your answer to the number you subtracted. Getting the starting number back
proves every trade was right; getting anything else proves it was not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $486 - 259 = 227$ correct?
\step{Checking a difference means running it backwards, so add the answer to the
number that was taken away.}
\step{$227 + 259$: \ $7 + 9 = 16$ (write $6$, carry $1$); \ $2 + 5 + 1 = 8$;
\ $2 + 2 = 4$.}
\step{The check gives $\ans{486}$, the number we started from, so $227$ is
correct.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Check $350 - 176 = 174$ by adding $174 + 176$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{350}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When rounding to the nearest hundred, look at the
\emph{tens} digit, not the ones digit. $349$ rounds \emph{down} to $300$ even
though it ends in $9$.
\end{watchoutbox}
