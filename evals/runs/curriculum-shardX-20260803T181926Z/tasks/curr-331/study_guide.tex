\sstitleblock{Classifying Quadrilaterals from Defining Properties}

\skillheading{1. Read the marks, then name the most specific shape}

\begin{formulabox}
\textbf{Definitions (each one adds a condition to the one above):}
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item \textbf{Trapezoid} --- at least one pair of parallel sides.
  \item \textbf{Parallelogram} --- \emph{both} pairs of opposite sides parallel.
  \item \textbf{Rhombus} --- a parallelogram with four congruent \emph{sides}.
  \item \textbf{Rectangle} --- a parallelogram with four right \emph{angles}.
  \item \textbf{Square} --- both at once: four congruent sides \emph{and} four
  right angles.
\end{itemize}
\textbf{Parallelogram angle facts:} opposite angles are congruent; consecutive
angles are supplementary. Every quadrilateral's angles sum to $360^\circ$.\\[2pt]
\textbf{(!)} Name what the marks \emph{force}, not what the drawing resembles,
and always give the \emph{most specific} name that the marks reach.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Name $ABCD$, then find $x$.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \coordinate (A) at (0,0);
  \coordinate (B) at (4.4,0);
  \coordinate (C) at (5.047,2.415);
  \coordinate (D) at (0.647,2.415);
  \draw[thick] (A)--(B)--(C)--(D)--cycle;
  \node[below left] at (A) {$A$};
  \node[below right] at (B) {$B$};
  \node[above right] at (C) {$C$};
  \node[above left] at (D) {$D$};
  \coordinate (M) at ($(A)!0.55!(B)$);
  \draw[thick] ($(M)!4pt!135:(B)$) -- (M) -- ($(M)!4pt!225:(B)$);
  \coordinate (N) at ($(D)!0.55!(C)$);
  \draw[thick] ($(N)!4pt!135:(C)$) -- (N) -- ($(N)!4pt!225:(C)$);
  \coordinate (P) at ($(B)!0.45!(C)$);
  \coordinate (Q) at ($(B)!0.65!(C)$);
  \draw[thick] ($(P)!4pt!135:(C)$) -- (P) -- ($(P)!4pt!225:(C)$);
  \draw[thick] ($(Q)!4pt!135:(C)$) -- (Q) -- ($(Q)!4pt!225:(C)$);
  \coordinate (R) at ($(A)!0.45!(D)$);
  \coordinate (S) at ($(A)!0.65!(D)$);
  \draw[thick] ($(R)!4pt!135:(D)$) -- (R) -- ($(R)!4pt!225:(D)$);
  \draw[thick] ($(S)!4pt!135:(D)$) -- (S) -- ($(S)!4pt!225:(D)$);
  \pic [draw, angle radius=6mm, angle eccentricity=1.9, font=\scriptsize,
        "$(3x + 8)^\circ$"] {angle = B--A--D};
  \pic [draw, angle radius=6mm, angle eccentricity=1.7, font=\scriptsize,
        "$62^\circ$"] {angle = D--C--B};
\end{tikzpicture}
\end{center}
\step{Two pairs of chevrons and no side or angle marks --- that is the
parallelogram definition and nothing more, so use a \emph{parallelogram} angle
fact.}
\step{$\angle A$ and $\angle C$ are opposite, so they are congruent:
$3x + 8 = 62$, giving $3x = 54$.}

Parallelogram, and $\ans{x = 18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In parallelogram $PQRS$, $\angle P$ and $\angle Q$ are
consecutive, with $m\angle P = (2x)^\circ$ and $m\angle Q = (x + 30)^\circ$.
Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 50}$}
\end{tryitbox}

\skillheading{2. Classify from what the diagonals do}

\begin{formulabox}
\textbf{Diagonal tests --- each one names a family on its own:}
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item Diagonals \textbf{bisect each other} $\Rightarrow$ parallelogram.
  \item \ldots\ and are also \textbf{congruent} $\Rightarrow$ rectangle.
  \item \ldots\ and are also \textbf{perpendicular} $\Rightarrow$ rhombus.
  \item Congruent \emph{and} perpendicular \emph{and} bisecting $\Rightarrow$
  square.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The diagonals of $ABCD$ bisect each other at $E$, with
$AE = 3x - 4$ and $EC = x + 6$. Find $x$.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \coordinate (A) at (0,0.5);
  \coordinate (B) at (4.6,0);
  \coordinate (C) at (5.2,2.7);
  \coordinate (D) at (0.6,3.2);
  \draw[thick] (A)--(B)--(C)--(D)--cycle;
  \draw[thick] (A)--(C);
  \draw[thick] (B)--(D);
  \fill ($(A)!0.5!(C)$) circle (2pt);
  \node[below left] at (A) {$A$};
  \node[below right] at (B) {$B$};
  \node[above right] at (C) {$C$};
  \node[above left] at (D) {$D$};
  \node[below right, font=\scriptsize] at ($(A)!0.5!(C)$) {$E$};
\end{tikzpicture}
\end{center}
\step{Bisecting means $E$ is the midpoint of $\overline{AC}$, so the two halves
of that diagonal are equal --- set them equal rather than adding them.}
\step{$3x - 4 = x + 6$, so $2x = 10$.}

$\ans{x = 5}$, and the bisecting diagonals make $ABCD$ a parallelogram
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The diagonals of rectangle $PQRS$ are congruent, with
$PR = 4x + 1$ and $QS = 2x + 15$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}

\skillheading{3. Classify from vertex coordinates}

\begin{formulabox}
\textbf{Slope} decides parallel and perpendicular; \textbf{distance} decides
congruent.\\[3pt]
Equal slopes $\Rightarrow$ parallel. \quad Slopes multiplying to $-1$
$\Rightarrow$ perpendicular, so a right angle.\\[2pt]
Both pairs of opposite sides parallel $\Rightarrow$ parallelogram; add a right
angle for a rectangle, add equal adjacent sides for a rhombus, add both for a
square.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $W(0, 0)$, $X(4, 2)$ and $Y(6, -2)$ are three vertices.
Is the angle at $X$ a right angle?
\step{A right angle at $X$ is a claim about the two sides meeting there, so
compare the slopes of $\overline{WX}$ and $\overline{XY}$.}
\step{Slope of $\overline{WX}$ is $\dfrac{2 - 0}{4 - 0}$, and slope of
$\overline{XY}$ is $\dfrac{-2 - 2}{6 - 4}$.}

$\ans{\tfrac{1}{2}}$ and $\ans{-2}$ multiply to $-1$, so yes --- the angle at
$X$ is right
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope of the segment joining $(2, -1)$ and
$(4, 3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}
