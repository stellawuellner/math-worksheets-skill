\wstitleblock{Solving Right Triangles: Elevation and Depression}{High School Geometry}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item An \textbf{angle of elevation} is measured up from a horizontal line;
        an \textbf{angle of depression} is measured down from one. Because the
        two horizontal lines are parallel, the angle of depression from the top
        equals the angle of elevation from the bottom.
  \item Work in degree mode. Round every length to two decimal places and every
        angle to the nearest tenth of a degree.
  \item Each problem carries its own diagram drawn to scale from that problem's
        givens. Write the unit on every answer.
\end{itemize}

\problem{A $6$ m ladder leans against a wall and makes an angle of $62^\circ$
with the level ground. How far up the wall does the top of the ladder reach, in
metres?
\par\vspace{0.1cm}
\probfig{1}
\par\vspace*{3.2cm}
\answerline{m}}

\problem{A surveyor stands $40$ ft from the foot of a flagpole on level ground.
The angle of elevation from the surveyor's eye to the top of the pole is
$35^\circ$. Find the height of the pole above eye level, in feet.
\par\vspace{0.1cm}
\probfig{2}
\par\vspace*{3.2cm}
\answerline{ft}}

\problem{From the top of a $25$ m lighthouse the angle of depression to a boat

\par\vspace{0.1cm}
\probfig{3}
\par\vspace*{3.2cm}
\answerline{m}}

\problem{A guy wire runs from an anchor on level ground $12$ m from the base of

\par\vspace{0.1cm}
\probfig{4}
\par\vspace*{3.2cm}
\answerline{degrees}}

\problem{From a window $15$ m above level ground, the angle of elevation to the

\problem{A zip line runs from a platform $22$ m above level ground down to a
landing point on the ground. Along the zip line the angle of depression is
$27^\circ$.
\par\vspace{0.1cm}
\probfig{8}
\par\vspace{0.1cm}
(a) Find the length of the zip line, in metres. \answerline{m}

(b) Find the horizontal distance from the base of the platform to the landing
