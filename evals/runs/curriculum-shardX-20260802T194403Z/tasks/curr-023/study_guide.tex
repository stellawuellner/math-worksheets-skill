\sstitleblock{Add-To Stories Within 10}

\skillheading{1. Turning the story into a number sentence (story-to-number-sentence)}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
An \emph{add-to} story has three parts: what you \textbf{start} with, what is
\textbf{added}, and how many there are \textbf{in all}.\\[3pt]
\hspace*{0.5cm}start $+$ added $=$ in all \qquad (the ``in all'' number is the one you are hunting)\\[3pt]
Words that mean add: \emph{more, gives you, joins, lands, puts in, altogether, in all}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 3 ducks are in the pond. 4 more ducks swim in. How many ducks in all?
\step{The pond starts with 3 and 4 more join, so the two parts are known and the whole is missing --- write start $+$ added.}
\step{$3 + 4$: count on from 3 --- ``4, 5, 6, 7'' --- four counts.}
$3 + 4 = \ans{7}$ ducks in all.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 5 toy cars are on the rug. 3 more cars are rolled over.
How many cars in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{2. Counting on from the bigger part (count-on-to-add)}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
Put the \textbf{bigger} part in your head, then count \textbf{one number for
each} of the smaller part. Say the next number first --- the starting number is
already counted.\\[3pt]
$6 + 2$: hold \textbf{6}, say ``7, 8'' \quad (2 counts) \quad $\Rightarrow 8$
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Start with 6 shells. Find 2 more.
\step{Both parts are small and one is much bigger, so counting on beats counting everything again.}
\step{Hold 6, then say two numbers: ``7, 8.'' Stop after two counts.}
$6 + 2 = \ans{8}$ shells.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Start with 7 marbles. Find 2 more.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do \emph{not} say the starting number as your first
count. For $6 + 2$ say ``7, 8'' --- saying ``6, 7'' lands you one too low.
\end{watchoutbox}

\skillheading{3. Showing the story on a ten-frame (ten-frame-add)}

\begin{formulabox}
\textbf{Model:}\\[4pt]
Fill the top row first, left to right, then the bottom row. A full row is
\textbf{5}, and a full frame is \textbf{10}, so you can see the total without
counting every dot.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The frame shows 4 counters. Put on 5 more.
\step{The picture already holds the start number, so add the new counters to the frame and read the whole frame.}
\step{4 fills most of the top row; 5 more finishes that row (that is 5) and puts 4 in the bottom row: $5 + 4$.}
$4 + 5 = \ans{9}$ counters in all.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The frame shows 3 counters. Put on 5 more.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}
