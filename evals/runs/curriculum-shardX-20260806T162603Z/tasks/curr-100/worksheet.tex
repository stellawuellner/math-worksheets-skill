\problem{Use Bar Graph A. A quadrilateral is a shape with four straight sides, so the squares, the
rectangles and the trapezoids are all quadrilaterals --- the triangles are not. How many
quadrilaterals are in the shape box?
\par\vspace*{3.6cm}\answerline{shapes}}

\problem[3cm]{Use the Shape Set. How many pairs of parallel sides does shape T have?}

\problem[3cm]{Use Line Plot B. How many cut-outs have $4$ sides?}

\problem{Use Bar Graph A. The shapes with four square corners are the squares and the rectangles.
How many shapes in the box have four square corners?
\par\vspace*{3.8cm}\answerline{shapes}}

\problem[3.4cm]{Use the Shape Set.
\begin{itemize}
  \item[(a)] How many of the four shapes have four equal sides? \quad \ansblank
  \item[(b)] All four shapes have four sides, but they are not all in that group. Explain why having
        four sides is not enough.
\end{itemize}}

\problem[3cm]{Use Bar Graph A. How many more rectangles are there than triangles?}

\problem[3.2cm]{Use Line Plot B. Every cut-out with four sides is a quadrilateral. How many more
quadrilateral cut-outs are there than five-sided cut-outs?}

\problem[4.4cm]{Use Line Plot B. The class makes $2$ more cut-outs, and both of them are squares.
\begin{itemize}
  \item[(a)] How many cut-outs will have $4$ sides then? \quad \ansblank
  \item[(b)] Explain what happens to the tallest column, and why the number of three-sided cut-outs
        does not change.
\end{itemize}}
