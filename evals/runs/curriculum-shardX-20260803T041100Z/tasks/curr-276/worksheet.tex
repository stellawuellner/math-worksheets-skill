\wstitleblock{Distinguishing Exponential from Linear Change}{Algebra 1 \quad Exponential Growth and Decay}{}

\begin{center}
\textit{Deciding from a table whether change is by a constant difference or a
constant ratio, writing the matching rule, comparing a linear model with an
exponential one, and classifying real situations before computing}
\end{center}

\noindent\textbf{The one idea behind every problem.} Two patterns look alike for
a step or two and then part company forever. Linear change \textbf{adds} the
same amount each step; exponential change \textbf{multiplies} by the same factor
each step. Every problem here asks the same first question: is the step an
addition or a multiplication?

\begin{center}
\fbox{\parbox{0.92\linewidth}{\centering
\textbf{Linear:} constant \emph{difference} \quad
$f(n) = a + d \cdot n$
\\[3pt]
\textbf{Exponential:} constant \emph{ratio} \quad
$f(n) = a \cdot r^{\,n}$
\\[3pt]
$a$ is the starting value, $d$ is what is added, $r$ is what you multiply by}}

\smallskip
\small\textit{Reference card. No values are filled in --- use the numbers given
in each problem.}\normalsize
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item For every table, write the difference row \emph{and} the ratio row before
deciding. One of them will be constant, or neither will be.
\item Say which model you chose and why, then compute.
\item Round money answers to the nearest cent. Write your final answer on the
answer line.
\end{itemize}

\problem[6cm]{Here is a table of values for $f$.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$n$ & 0 & 1 & 2 & 3 \\
\hline
$f(n)$ & 5 & 8 & 11 & 14 \\
\hline
\end{tabular}
\end{center}
\par
\textbf{(a)} Write the differences between consecutive values, and the ratios.
Which one is constant?

\smallskip
\textbf{(b)} Is $f$ linear or exponential? Write its rule.

\smallskip
\textbf{(c)} Find $f(6)$.}

\problem[6.5cm]{Here is a table of values for $g$.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$n$ & 0 & 1 & 2 & 3 \\
\hline
$g(n)$ & 4 & 12 & 36 & 108 \\
\hline
\end{tabular}
\end{center}
\par
\textbf{(a)} The differences here are 8, 24, 72 --- not constant. Compute the
ratios instead. What do you find?

\smallskip
\textbf{(b)} Is $g$ linear or exponential? Write its rule.

\smallskip
\textbf{(c)} Find $g(5)$.}

\problem[6cm]{Rosa opens a savings account with 200 dollars and adds 25 dollars
at the end of every month. No interest is paid.

\smallskip
\textbf{(a)} Is the balance changing by a constant difference or a constant
ratio? Say which, in one sentence.

\smallskip
\textbf{(b)} Write the rule for the balance after $n$ months.

\smallskip
\textbf{(c)} How many dollars are in the account after 8 months?}

\problem[6.5cm]{A bacteria colony starts with 6 cells, and the number of cells
doubles every hour.

\smallskip
\textbf{(a)} ``Doubles'' names an operation. Is it addition or multiplication?
What does that make the model?

\smallskip
\textbf{(b)} Write the rule for the number of cells after $n$ hours.

\smallskip
\textbf{(c)} How many cells are there after 7 hours?}

\problem[7cm]{Two savings plans both start at 100 dollars.

\begin{itemize}[leftmargin=1.0cm, itemsep=1pt, topsep=2pt]
\item \textbf{Plan L} adds 50 dollars each step.
\item \textbf{Plan E} multiplies by 1.5 each step.
\end{itemize}

\textbf{(a)} Write a rule for each plan.

\smallskip
\textbf{(b)} Plan E is behind Plan L at step 1. Explain in one sentence how it
can still end up ahead.

\smallskip
\textbf{(c)} At step 5, how many dollars is Plan E ahead of Plan L? Round to the
nearest cent.}

\problem[7cm]{Not every pattern is one of the two. Here is a table for $h$.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$n$ & 1 & 2 & 3 & 4 \\
\hline
$h(n)$ & 2 & 8 & 18 & 32 \\
\hline
\end{tabular}
\end{center}
\par
\textbf{(a)} Compute the first ratio $h(2) \div h(1)$.

\smallskip
\textbf{(b)} Compute the next ratio $h(3) \div h(2)$ as a decimal.

\smallskip
\textbf{(c)} The differences are 6, 10, 14 --- not constant either. What do your
two ratios prove about $h$? Write both ratios on the answer line.}

\problem[6.5cm]{A pattern is known to be linear, and two of its values are
$f(2) = 17$ and $f(5) = 32$.

\smallskip
\textbf{(a)} What is the constant rate of change?

\smallskip
\textbf{(b)} Use it to write the rule for $f(n)$.

\smallskip
\textbf{(c)} If instead the pattern were exponential through those same two
points, would the rate of change still be constant? Answer in one sentence.}

\problem[7.5cm]{A car is worth 24000 dollars today. Two dealers describe how it
will lose value.

\begin{itemize}[leftmargin=1.0cm, itemsep=1pt, topsep=2pt]
\item \textbf{Dealer A:} it loses 12 percent of its current value each year.
\item \textbf{Dealer B:} it loses 2400 dollars each year.
\end{itemize}

\textbf{(a)} Which description is exponential and which is linear? How do you
know?

\smallskip
\textbf{(b)} Write a rule for each value after $n$ years.

\smallskip
\textbf{(c)} After 5 years, how many dollars more is the car worth under Dealer
A than under Dealer B? Round to the nearest cent.}

\problem[7.5cm]{A population grows exponentially. It is 500 at time $0$ and 4000
at time 3.

\smallskip
\textbf{(a)} Find the growth factor $r$ per unit of time. Show the equation you
solved.

\smallskip
\textbf{(b)} Use your rule to find the population at time 5.

\smallskip
\textbf{(c)} Write both answers on the answer line, in the order (a) then (b).}

\problem[8cm]{\textbf{Challenge.} A summer job offers two pay plans for a
20-day stretch.

\begin{itemize}[leftmargin=1.0cm, itemsep=1pt, topsep=2pt]
\item \textbf{Flat plan:} 1000 dollars on every day.
\item \textbf{Doubling plan:} 0.01 dollars on day 1, and the daily pay doubles
each day after that.
\end{itemize}

\textbf{(a)} Write a rule for the doubling plan's pay on day $n$. Be careful
with the exponent: day 1 has been doubled zero times.

\smallskip
\textbf{(b)} On day 10, which plan pays more that day? A one-line estimate is
enough.

\smallskip
\textbf{(c)} On day 20, how many dollars more does the doubling plan pay that
day than the flat plan? Round to the nearest cent.}
