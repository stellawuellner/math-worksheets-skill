\begin{center}\textit{Counting rows, arrays, and ten-frames}\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Touch each dot one time as you count. Say one number for each dot.
  \item The last number you say tells \emph{how many}.
  \item Write the number on the blank.
\end{itemize}
