\sstitleblock{Modeling Discrete Linear Change and Arithmetic Sums}

\skillheading{Skill 1 --- Turning a situation into a term rule}

\begin{formulabox}
\textbf{Set up first, compute second.} Say in words what $n$ counts (chairs, weeks, rows) and what $a_n$ measures, with its unit. A situation that changes by the same amount at every stage is arithmetic.\par
\textbf{Term rule:} $a_n = a_1 + (n-1)d$. The $n-1$ is there because stage $n$ is $n-1$ steps past stage $1$.\par
\textbf{Careful:} a one-off charge that is \emph{not} repeated (a joining fee) sits outside the step: that total is $a_1 + dn$, not a term of the sequence.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ladder costs \$25 for the first day of hire and \$14 for each day after. What does a $6$-day hire cost?
\step{The daily charge is constant after the first day, so the cost by day $n$ is arithmetic with $a_1 = 25$ and $d = 14$.}
\step{Six days is five extra days: $25 + 5(14) = 25 + 70$.}

cost $= \ans{95}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tool costs \$40 for the first day and \$12 for each day after. What does a $9$-day hire cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{136}$ dollars}
\end{tryitbox}

\skillheading{Skill 2 --- Finding the stage number}

\begin{formulabox}
\textbf{Rule:} when the question gives you a \emph{value} and asks \emph{when}, set the term rule equal to that value and solve for $n$.\par
$a_1 + (n-1)d = \text{target} \;\Longrightarrow\; n - 1 = \dfrac{\text{target} - a_1}{d} \;\Longrightarrow\; n = \dfrac{\text{target} - a_1}{d} + 1$.\par
\textbf{Sanity check:} a stage number must be a whole number. A fraction means the target is never hit exactly --- report the first stage past it, and say so.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $a_n = 7 + 5(n-1)$, which term equals $62$?
\step{The value is given and the stage number is unknown, so this is an equation to solve, not a substitution.}
\step{$7 + 5(n-1) = 62$, so $5(n-1) = 55$ and $n - 1 = 11$.}

$n = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $a_n = 100 - 8(n-1)$, which term equals $44$?\\[2pt]
\rotatebox{180}{\footnotesize check: $n = \ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} do not forget the $+1$ at the end. Dividing gives the number of \emph{steps}; the stage number is one more than that.
\end{watchoutbox}

\skillheading{Skill 3 --- Adding up a finite run}

\begin{formulabox}
\textbf{Rule:} $S_n = \dfrac{n}{2}\,(a_1 + a_n)$. Pair the smallest term with the largest, the next-smallest with the next-largest, and every pair has the same total.\par
\textbf{Order of work:} find $a_n$ first, then average it with $a_1$, then multiply by how many terms there are --- and $n$ is a count of terms, not a count of steps.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $a_n = 6 + 4(n-1)$, find the sum of the first $9$ terms.
\step{A total over a fixed run of stages is a finite arithmetic sum, so pair the ends rather than adding nine numbers.}
\step{Last term: $6 + 8(4) = 38$.}
\step{Nine terms, so $\frac{9}{2}(6 + 38) = 9(22)$.}

$S_9 = \ans{198}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $a_n = 3 + 7(n-1)$, find the sum of the first $10$ terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $S_{10} = \ans{345}$}
\end{tryitbox}
