\sstitleblock{Groups, Arrays, and Products}

\skillheading{1. Adding the same number again and again}

\begin{formulabox}
When the \emph{same} number is added over and over, count how many times it
appears --- that is the number of groups. \quad
\textbf{groups $\times$ size of a group $=$ total.}\par
$4 + 4 + 4$ is three groups of four, so it is $3 \times 4$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $4 + 4 + 4$ as a multiplication.
\step{Every group is the same size, so counting the groups is quicker than
counting everything one by one.}
\step{There are $3$ fours, so this is $3 \times 4$: count $4, 8, 12$.}

$3 \times 4 = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $5 + 5 + 5 + 5$ as a multiplication and find the
total.\\[2pt]
\rotatebox{180}{\footnotesize check: $4 \times 5 = \ans{20}$}
\end{tryitbox}

\skillheading{2. Reading an array: rows and row size}

\begin{formulabox}
An array has rows that all hold the same number of things. Read it as
\textbf{rows $\times$ how many in each row}.\par
Turn the array sideways and you read the same picture the other way, so
$3 \times 5$ and $5 \times 3$ give the same total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An array has $3$ rows with $5$ dots in each row.
\step{Counting all the dots works but is slow --- the rows are equal, so count
rows and row size instead.}
\step{$3$ rows of $5$ is $3 \times 5$: count by fives, $5, 10, 15$.}

$3 \times 5 = \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An array has $4$ rows with $6$ tiles in each row. How
many tiles?\\[2pt]
\rotatebox{180}{\footnotesize check: $4 \times 6 = \ans{24}$}
\end{tryitbox}

\skillheading{3. Turning a product back into an addition}

\begin{formulabox}
$5 \times 2$ is read as \emph{5 groups of 2}. The \textbf{second} number is
what repeats, and the \textbf{first} number says how many times.\par
So $5 \times 2 = 2 + 2 + 2 + 2 + 2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $5 \times 2$ as a repeated addition.
\step{Say the multiplication out loud as ``five groups of two'' --- that
sentence tells you which number to write down over and over.}
\step{Write $2$ five times: $2 + 2 + 2 + 2 + 2$, and count by twos:
$2, 4, 6, 8, 10$.}

$5 \times 2 = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $3 \times 6$ as a repeated addition and find the
total.\\[2pt]
\rotatebox{180}{\footnotesize check: $6 + 6 + 6 = \ans{18}$}
\end{tryitbox}

\skillheading{4. Finding the missing factor}

\begin{formulabox}
$3 \times \underline{\hspace{0.8cm}} = 18$ asks: three of \emph{what} makes
$18$? Skip-count by the number you know and stop when you reach the total ---
the number of counts you made is the missing factor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fill in $3 \times \underline{\hspace{0.8cm}} = 18$.
\step{One factor and the total are known, so skip-counting by the known factor
will run straight into the answer.}
\step{Count by threes: $3, 6, 9, 12, 15, 18$ --- that is six threes.}

The missing factor is $\ans{6}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Fill in $\underline{\hspace{0.8cm}} \times 4 = 28$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $4 \times 6$ is not $4 + 6$. The two numbers have
different jobs --- one counts the groups, the other says how big each group is.
\end{watchoutbox}
