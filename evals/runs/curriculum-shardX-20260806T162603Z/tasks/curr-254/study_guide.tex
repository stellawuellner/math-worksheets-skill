% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Distributing a minus sign}

\begin{formulabox}
\textbf{Rule:} a minus sign in front of a bracket is a factor of $-1$, so it changes the sign of
\emph{every} term inside:
\[
  A - (b + c - d) \;=\; A - b - c + d .
\]
Rewrite the bracket with all its signs flipped \emph{before} you collect anything. The term that
gets missed is almost always the last one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(6x + 5) - (2x - 8)$.
\step{A whole bracket is being subtracted, so my first move is to flip every sign inside it rather than start collecting.}
\step{$-(2x - 8)$ becomes $-2x + 8$; the $-8$ turns into $+8$.}
\step{Collect: $6x - 2x = 4x$ and $5 + 8 = 13$.}

$\Rightarrow\quad \ans{4x + 13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(9x - 1) - (4x - 6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x + 5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Combining only like terms}

\begin{formulabox}
\textbf{Rule:} two terms are alike only when they have the \emph{same variable to the same power}.
Combining them adds the coefficients and leaves the power alone: $ax^n + bx^n = (a+b)x^n$.
So $3x^2 + 4x$ is already as simple as it gets, and $x^3$ never joins $x^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $4x^2 + 7x - x^2 + 2x$.
\step{Nothing is being subtracted as a bracket here, so the only question is which terms share a power.}
\step{Squares: $4x^2 - x^2 = 3x^2$. \quad Linear: $7x + 2x = 9x$.}

$\Rightarrow\quad \ans{3x^2 + 9x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $5x^3 + 2x - 3x^3 + 6x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x^3 + 8x}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Simplifying in stages, then checking}

\begin{formulabox}
\textbf{The test-value check:} equivalent expressions agree at \emph{every} number. So pick an easy
value such as $x = 1$ or $x = 2$, substitute it into the original expression and into your
simplified answer, and compare. One disagreement proves a mistake; agreement is strong evidence
you are right. Avoid $x = 0$, which hides errors in the variable terms.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(2x^2 - x + 4) - (x^2 + 3x - 5)$, then check it at $x = 1$.
\step{A bracket is subtracted, so flip its signs first; then I will check the result with a test value instead of trusting the arithmetic.}
\step{$2x^2 - x + 4 - x^2 - 3x + 5$, which collects to $x^2 - 4x + 9$.}
\step{Test at $x = 1$: the original gives $(2 - 1 + 4) - (1 + 3 - 5) = 5 - (-1) = 6$, and the answer gives $1 - 4 + 9 = 6$.}

$\Rightarrow\quad \ans{x^2 - 4x + 9}$, and both sides give $\ans{6}$ at $x = 1$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(5x^2 + 2x - 3) - (2x^2 - 4x + 1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^2 + 6x - 4}$}
\end{tryitbox}
