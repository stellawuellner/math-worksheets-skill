\sstitleblock{The Derivative as a Limit and as a Rate}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Derivative as a Limit of Slopes}

\begin{formulabox}
\textbf{Definition:}
\[
  f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}
\]
The fraction is the slope of the line through $(a, f(a))$ and
$(a+h, f(a+h))$. Letting $h \to 0$ slides the second point into the first, and
the secant slope becomes the tangent slope. You may never substitute $h = 0$
straight away --- that is $\frac{0}{0}$. Simplify until the $h$ downstairs
cancels.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^2$. Find $f'(5)$.
\step{Substituting $h=0$ in the difference quotient gives $\frac{0}{0}$, so expand the top first and cancel the $h$ --- that is the only move available.}
\step{$\dfrac{(5+h)^2-25}{h} = \dfrac{25+10h+h^2-25}{h} = \dfrac{h(10+h)}{h} = 10+h$, and as $h \to 0$ that tends to $10$.}

$\Rightarrow\quad f'(5) = $ $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^2$. Find $f'(4)$ from the definition.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Average Rate Versus Instant Rate}

\begin{formulabox}
\textbf{Two different questions.} Average rate over $[a,b]$ is
$\frac{f(b)-f(a)}{b-a}$: one slope, between two points, describing a whole
interval. Instantaneous rate at $a$ is $f'(a)$: the slope of the curve at a
single point. Read the wording --- ``from \dots\ to \dots'' asks for the first,
``at \dots'' asks for the second, and they are almost never equal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $s(t) = t^2$ metres at $t$ seconds. Find the average
velocity from $t = 2$ to $t = 5$.
\step{``From 2 to 5'' names an interval, so this is the two-point slope, not a derivative --- no limit is needed.}
\step{$\dfrac{s(5)-s(2)}{5-2} = \dfrac{25-4}{3} = \dfrac{21}{3} = 7$ metres per second.}

$\Rightarrow\quad$ $\ans{7}$ metres per second
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $s(t) = t^2$. Find the average velocity from
$t = 1$ to $t = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$ metres per second}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- What a Derivative Value Means}

\begin{formulabox}
\textbf{Reading $f'(a)$ in three parts.} The \textbf{sign} says the direction:
positive means the output is increasing at that moment, negative means it is
decreasing. The \textbf{size} says how fast. The \textbf{units} are always output
units per input unit --- litres per minute, metres per second, dollars per item.
A derivative is a rate at an instant, never a total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A balloon's volume is $V(t) = t^2 + 1$ litres at $t$
minutes. Find $V'(3)$ and say what it means.
\step{The question asks for a rate at one instant, so differentiate and then substitute --- the derivative function is a machine that turns an input into a slope.}
\step{$V'(t) = 2t$, so $V'(3) = 2(3) = 6$: positive, size $6$, units litres per minute --- the balloon is filling at $6$ litres per minute at that instant.}

$\Rightarrow\quad V'(t) = $ $\ans{2t}$ \quad and \quad $V'(3) = $ $\ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^2 + 5x$. Find $f'(x)$, then $f'(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x+5}$ \ and \ $\ans{9}$}
\end{tryitbox}
