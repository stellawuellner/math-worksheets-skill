\sstitleblock{Where Is the Shape? Position Words on a Shelf}

\noindent\small A position word tells \emph{where} something is, not what it
looks like. On a numbered shelf you can always check a position word with
numbers: shelves are counted from the bottom up, and spots are counted from the
left. Point at both shapes, read their numbers, then compare.\normalsize

\skillheading{Skill 1 --- Above and below}

\begin{formulabox}
\textbf{Rule:} Shelf 1 is the bottom shelf. The \textbf{bigger} shelf number is
\textbf{higher up}. So if the first shape's shelf number is bigger, it is
\emph{above}; if it is smaller, it is \emph{below}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A ball is on shelf 4. A book is on shelf 2. Is the ball
above or below the book?
\step{The question is about high and low, so look at the shelf numbers and
compare them.}
\step{The ball is 4 and the book is 2. Four is more than two.}

$\ans{4 > 2}$ \quad so the ball is \textbf{above} the book.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A hat is on shelf 1. A cup is on shelf 3. Compare the two
shelf numbers with $<$ or $>$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1 < 3}$ --- the hat is below}
\end{tryitbox}

\skillheading{Skill 2 --- Left and right}

\begin{formulabox}
\textbf{Rule:} Spot 1 is farthest left. The \textbf{bigger} spot number is
farther \textbf{right}. Shelf numbers do not help here, and spot numbers do not
tell you anything about high and low.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A mug is in spot 2. A plate is in spot 5. Is the mug to
the left or the right of the plate?
\step{The question is about left and right, so this time look at the spot
numbers.}
\step{The mug is 2 and the plate is 5. Two is less than five.}

$\ans{2 < 5}$ \quad so the mug is to the \textbf{left} of the plate.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A bag is in spot 6. A boot is in spot 3. Compare the two
spot numbers with $<$ or $>$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6 > 3}$ --- the bag is to the right}
\end{tryitbox}

\skillheading{Skill 3 --- Putting shapes in order}

\begin{formulabox}
\textbf{Rule:} To line shapes up, write their position numbers in order.
\emph{Bottom to top} and \emph{left to right} both go smallest first.
\emph{Top to bottom} and \emph{right to left} both go biggest first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Three shapes are on shelves 3, 1 and 2. Put them in
order from the bottom shelf to the top shelf.
\step{Bottom to top means lowest shelf first, so the smallest number leads.}
\step{The numbers are 3, 1 and 2. Counting order puts 1 first, then 2, then 3.}

$\ans{1, \; 2, \; 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Three shapes are in spots 2, 4 and 1. Put the spot numbers
in order from right to left.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4, \; 2, \; 1}$}
\end{tryitbox}

\skillheading{Skill 4 --- Telling what is between}

\begin{formulabox}
\textbf{Rule:} A shape is \emph{between} two others when its position number
lands in the \textbf{middle} once all three numbers are put in order. Both
neighbours must be on the same shelf, or ``between'' does not apply.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad On one shelf, a shape sits in spot 4, another in spot 1,
and a third in spot 2. Which one is between the other two?
\step{``Between'' is about the middle, so put all three spot numbers in order
first and then look at the one in the middle.}
\step{The numbers 4, 1 and 2 in counting order are 1, then 2, then 4, and 2 is
the middle one.}

$\ans{1, \; 2, \; 4}$ \quad so the shape in spot 2 is \textbf{between} the
other two.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad On one shelf, shapes sit in spots 5, 2 and 3. Put the
numbers in order, then say which spot is in the middle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2, \; 3, \; 5}$ --- spot 3 is between}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Answer in the order the question asks. If it says
``the oval and the hexagon,'' write the oval's number first --- swapping them
flips $<$ and $>$ and turns \emph{below} into \emph{above}. And a shape can be
above one shape and below another at the same time, so always name which shape
you are comparing it with.
\end{watchoutbox}
