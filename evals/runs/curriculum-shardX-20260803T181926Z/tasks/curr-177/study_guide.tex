\sstitleblock{Two-Step Equations and Checking Solutions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Solving Two-Step Equations}

\begin{formulabox}
\textbf{Rule --- undo in reverse order.} \ An equation of the form $ax + b = c$ was built by multiplying, \emph{then} adding. Take it apart the other way round:
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item \textbf{Step 1:} add or subtract to remove $b$ from the $x$-side.
  \item \textbf{Step 2:} divide both sides by the coefficient $a$.
\end{itemize}
Whatever you do to one side, do to the other --- that is what keeps the equation true.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4x + 5 = 29$

\step{The $x$ was multiplied by $4$ and then $5$ was added, so peel off the addition first and the multiplication second.}
\step{Subtract $5$ from both sides: $4x = 24$. Now divide both sides by $4$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-2x + 13 = 3$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the sign belongs to the coefficient. In $-2x + 13 = 3$ you divide by $-2$, not by $2$ --- and a negative divided by a negative gives a positive answer.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- When the Variable Is Divided}

\begin{formulabox}
\textbf{Rule.} \ In $\dfrac{x}{a} + b = c$ the variable is \emph{divided}, so the last move is to \emph{multiply}. Clear the added number first, then multiply both sides by $a$. Division and multiplication undo each other exactly as addition and subtraction do.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{5} + 2 = 7$

\step{The $x$ is divided by $5$ and then $2$ is added, so remove the added $2$ first and undo the division last.}
\step{Subtract $2$ from both sides: $\dfrac{x}{5} = 5$. Multiply both sides by $5$.}

$\Rightarrow\quad \ans{x = 25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{n}{3} - 4 = 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 18}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Checking a Solution by Substituting}

\begin{formulabox}
\textbf{Rule.} \ A solution is a number that makes the two sides \emph{equal}. To check one, put it back into the \textbf{original} equation, work out the left side on its own, and compare with the right side. Equal means correct; not equal means go back and find the step that broke the balance.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $x = 4$ a solution of $3x - 5 = 7$?

\step{Checking means testing the candidate, not re-solving --- substitute $4$ for $x$ in the left side and see what it gives.}
\step{$3(4) - 5 = 12 - 5$, and the right side is $7$, so the two sides agree.}

$\Rightarrow\quad \text{left side} = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $y = -2$ a solution of $5y + 8 = -2$? Work out the left side.\\[2pt]
\rotatebox{180}{\footnotesize check: left side $= \ans{-2}$, so yes}
\end{tryitbox}
