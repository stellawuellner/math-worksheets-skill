\wstitleblock{Solving Exponential Equations with Common Bases}{Precalculus}{}

\begin{center}
\small\textit{Matching bases, rewriting to a common base, and reading exponential tables}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Rewrite both sides as powers of the \emph{same} base, then set the exponents equal.
  \item Show the common-base line before you set exponents equal --- that line is the method.
  \item Give exact answers. Leave fractions as fractions; do not round.
\end{itemize}
\vspace{0.15cm}

\problem[5cm]{Solve for $x$: \qquad $3^{\,x+2} = 3^{\,5}$}

\problem[5cm]{Solve for $x$: \qquad $4^{\,x} = 64$}

\problem[0pt]{The table gives values of $f(x) = 2^{\,x}$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $1$ & $2$ & $3$ & $4$ \\ \hline
$f(x)$ & $2$ & $4$ & $8$ & $16$ \\ \hline
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Extend the pattern one row further: find $f(5)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Write $256$ as a power of $2$ and solve $2^{\,x} = 256$.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[5cm]{Solve for $x$: \qquad $5^{\,2x-1} = 5^{\,x+4}$}

\problem[6cm]{Solve for $x$: \qquad $8^{\,x} = 32$}

\problem[0pt]{A bacteria culture is modelled by $P(t) = 50 \cdot 2^{\,t}$, where $P$ is the number of cells and $t$ is the time in hours. The table shows the first four hours.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$t$ (hours) & $0$ & $1$ & $2$ & $3$ \\ \hline
$P(t)$ (cells) & $50$ & $100$ & $200$ & $400$ \\ \hline
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] How many cells are there after $5$ hours? Find $P(5)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Solve $50 \cdot 2^{\,t} = 3200$ to find when the culture reaches $3200$ cells.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[6cm]{Solve for $x$: \qquad $9^{\,x-1} = 27^{\,x}$}

\problem[7cm]{Solve for $x$: \qquad $7^{\,x^{2}} = 7^{\,3x+4}$}

\problem[0pt]{The table gives values of $h(x) = 3^{\,x}$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $-1$ & $0$ & $1$ & $2$ \\ \hline
$h(x)$ & $\tfrac{1}{3}$ & $1$ & $3$ & $9$ \\ \hline
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Continue the table to the left: find $h(-3)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Solve $3^{\,x} = \dfrac{1}{81}$.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}
