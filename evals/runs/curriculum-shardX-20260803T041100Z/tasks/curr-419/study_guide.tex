\sstitleblock{Unit Circle --- Degree-Radian and Quadrant-Sign Errors}

\skillheading{Skill 1 --- Reference angle, then quadrant sign}

\begin{formulabox}
\textbf{Two-part rule:} an exact value is
\emph{(value of the reference angle)} with \emph{(the quadrant's sign)}.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Reference angle: the acute angle to the nearest part of the $x$-axis.
        In quadrant II use $\pi - \theta$; in III, $\theta - \pi$; in IV,
        $2\pi - \theta$ (degrees: $180^\circ - \theta$, $\theta - 180^\circ$,
        $360^\circ - \theta$).
  \item Signs (ASTC): quadrant I all $+$; II only $\sin +$; III only $\tan +$;
        IV only $\cos +$. Cosine reads the $x$-coordinate and sine the
        $y$-coordinate, so the signs are just the signs of the coordinates.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the exact value of $\cos\dfrac{5\pi}{6}$.
\step{The angle is between $\dfrac{\pi}{2}$ and $\pi$, so it sits in quadrant II
where cosine is negative --- decide that before looking anything up.}
\step{Reference angle $\pi - \dfrac{5\pi}{6} = \dfrac{\pi}{6}$, and
$\cos\dfrac{\pi}{6} = \dfrac{\sqrt{3}}{2}$; attach the quadrant's minus sign.}
$\cos\dfrac{5\pi}{6} = \ans{-\frac{\sqrt{3}}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the exact value of $\sin\dfrac{5\pi}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\frac{\sqrt{2}}{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Getting the angle mode right}

\begin{formulabox}
\textbf{Conversions:}
\[ \text{degrees} \rightarrow \text{radians}: \ \times \frac{\pi}{180}
   \qquad
   \text{radians} \rightarrow \text{degrees}: \ \times \frac{180}{\pi} \]
\textbf{Size check that catches a reversed conversion:} a full turn is
$360^\circ$ but only $2\pi \approx 6.28$ radians. Any angle under one turn has a
radian measure below $6.28$; a converted answer in the hundreds or thousands means
the factor was upside down.\par
\smallskip
\textbf{Mode check:} a bare number with no degree symbol means radians. If a
calculator returns a sign the quadrant forbids, suspect the mode before the
mathematics.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Convert $300^\circ$ to radians.
\step{Going from degrees to radians, so multiply by $\dfrac{\pi}{180}$ --- the
factor that makes the number smaller.}
\step{$300 \cdot \dfrac{\pi}{180} = \dfrac{300\pi}{180}$, and dividing both parts
by $60$ gives the simplified form.}
$\ans{\frac{5\pi}{3}}$, about $5.24$ radians --- under $2\pi$ $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Convert $135^\circ$ to radians, in exact form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{3\pi}{4}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Solving on an interval}

\begin{formulabox}
\textbf{Method:} isolate the trig ratio, take the reference angle from the
\textbf{size} of the ratio, then use the \textbf{sign} to choose the quadrants ---
never the other way round.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $\sin\theta > 0$: quadrants I, II. \quad $\sin\theta < 0$: III, IV.
  \item $\cos\theta > 0$: quadrants I, IV. \quad $\cos\theta < 0$: II, III.
  \item Expect \textbf{two} solutions per full turn for a sine or cosine equation
        (one at each of two quadrants), and answer in the unit the interval uses.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2\sin\theta - 1 = 0$ for
$0^\circ \le \theta < 360^\circ$.
\step{Isolate first: $\sin\theta = \dfrac{1}{2}$ is positive, so the solutions are
in quadrants I and II --- the sign picks the quadrants, the size picks the
reference angle.}
\step{Reference angle $30^\circ$; quadrant I gives $30^\circ$ and quadrant II
gives $180^\circ - 30^\circ$.}
$\ans{\theta = 30^\circ}$ and $\ans{\theta = 150^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2\cos\theta + 1 = 0$ for
$0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = 120^\circ}$, $\ans{\theta = 240^\circ}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A mode error and a quadrant-sign error both return a
number of believable size, so recomputing never exposes them. Predict the sign
from the quadrant \emph{before} you evaluate, and reject any answer that
contradicts the prediction.
\end{watchoutbox}
