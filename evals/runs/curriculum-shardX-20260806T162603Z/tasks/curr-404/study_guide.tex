\sstitleblock{Diagnosing Transformation-Order and Inverse-Domain Errors}

\skillheading{Skill 1 --- Inside the function or outside it}

\begin{formulabox}
\textbf{Outside $f$} ($a\,f(x) + k$): stretches and shifts the \emph{output}, vertically, exactly as written.\par
\textbf{Inside $f$} ($f(bx + c)$): changes the \emph{input}, horizontally, and always in the \emph{opposite} sense --- $f(x+3)$ moves left, $f(2x)$ compresses.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2}$. Evaluate $g(x) = f(x + 3)$ at $x = 1$.
\step{The $+3$ sits inside $f$, so it changes the input before squaring --- it is not added to the answer.}
\step{$g(1) = f(1 + 3) = f(4) = 4^{2}$.}

$\Rightarrow\quad \ans{g(1) = 16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $f$. Evaluate $h(x) = f(x) + 3$ at $x = 1$, and notice how far apart the two answers are.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h(1) = 4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Ordering the outside operations}

\begin{formulabox}
\textbf{Rule.} A vertical stretch multiplies everything already present, so \emph{stretch and reflect first, shift last}.\par
Written out: $a\,f(x) + k$ means take $f(x)$, multiply by $a$, \emph{then} add $k$. Doing the shift first turns it into $a\bigl(f(x) + k\bigr)$, which moves the graph by $ak$ instead of $k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Stretch $y = |x|$ vertically by $4$, then shift down $5$. Find $g(3)$.
\step{The shift is last, so it must stay outside the multiplication --- writing a bracket around both would multiply the shift too.}
\step{$g(x) = 4|x| - 5$, so $g(3) = 4(3) - 5 = 12 - 5$.}

$\Rightarrow\quad \ans{g(3) = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Stretch $y = x^{2}$ vertically by $2$, then shift up $3$. Find $g(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g(4) = 35}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a reflection is a stretch by a negative number, so it obeys the same rule --- reflect before shifting, or the shift gets negated too.
\end{watchoutbox}

\skillheading{Skill 3 --- Letting the restriction pick the branch}

\begin{formulabox}
\textbf{Rule.} A squared rule sends two inputs to one output, so it has an inverse only on one side of its vertex.\par
The kept side decides the sign: a domain \emph{above} the vertex gives $h + \sqrt{\cdots}$, a domain \emph{below} it gives $h - \sqrt{\cdots}$. Check by asking whether your answer lands inside the original domain.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = (x-1)^{2} + 2$ on $x \geq 1$. Find $f^{-1}(11)$.
\step{The domain starts at the vertex and runs right, so the inverse must return values at or above $1$ --- that fixes the plus sign before any arithmetic.}
\step{$f^{-1}(x) = 1 + \sqrt{x - 2}$, so $f^{-1}(11) = 1 + \sqrt{9} = 1 + 3$.}

$\Rightarrow\quad \ans{f^{-1}(11) = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = (x+2)^{2} - 1$ on $x \leq -2$. Find $f^{-1}(24)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f^{-1}(24) = -7}$}
\end{tryitbox}
