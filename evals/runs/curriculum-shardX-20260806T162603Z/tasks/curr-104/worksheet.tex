\wstitleblock{Find the Error: Rounding and Place Value}{Grades 4--5 Mathematics}{}

\noindent\textbf{Directions:} Some problems on this sheet show another
student's work that contains a mistake. Be the detective: find the error, say
what went wrong, and fix it. Watch for two famous traps --- rounding across a 9,
and forgetting the zeros that hold a place.

\problem[3cm]{Round $2{,}963$ to the nearest hundred.}

\problem[3cm]{Round $39{,}725$ to the nearest thousand.}

\problem[4cm]{Leo rounds $4{,}972$ to the nearest hundred and writes
$4{,}900$. Explain Leo's mistake, then write the correct
answer.~\ansblank}

\problem[4cm]{Mia multiplies $30 \times 40$ and writes $120$. Explain what Mia
forgot, then write the correct product.~\ansblank}

\problem[3cm]{Round $9{,}984$ to the nearest hundred.}

\problem[4cm]{Jon writes: \ $50{,}000 + 300 + 7 = 5{,}307$. Explain Jon's
error, then write the correct number.~\ansblank}

\problem[4cm]{Circle all of the numbers that round to $700$ when rounded to
the nearest hundred. Below each number, write what it \emph{does} round to.
\par\vspace{4pt}
\begin{center}
{\large $748$ \qquad $651$ \qquad $762$ \qquad $685$}
\end{center}
\noansline}

\problem[4.5cm]{Ava rounds $452$ and $448$ to the nearest ten and announces:
``They round to the same number, so $452$ and $448$ are equal.''
\par\vspace{2pt}\noindent
$452$ rounds to~\ansblank\quad and \quad $448$ rounds to~\ansblank
\par\vspace{2pt}\noindent
Explain what is wrong with Ava's conclusion.}
