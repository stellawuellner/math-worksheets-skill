\wstitleblock{Riemann Sums, Accumulation, and the Fundamental Theorem}{AP Calculus AB/BC \quad Integration}{}

\begin{center}
\textit{Riemann sum estimates from tables and functions, antiderivatives,
evaluating definite integrals by the Fundamental Theorem, and accumulation
functions}
\end{center}

\noindent\textbf{The chain this sheet follows.} A Riemann sum estimates
accumulated change from finitely many samples. Its limit is the definite
integral. The Fundamental Theorem then evaluates that integral exactly from an
antiderivative, and read the other way it says the accumulation function
$F(x) = \int_a^x f(t)\,dt$ has $F'(x) = f(x)$.

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Decide first whether the problem wants an \emph{estimate} (a sum), an
\emph{exact value} (an antiderivative), or a \emph{conclusion about a function
defined by an integral}. The three alternate.
\item Show the sum or the evaluated antiderivative, not just the number. Write
$F(b) - F(a)$ explicitly on every FTC evaluation.
\item Include units where the problem gives them.
\end{itemize}

\problem[8cm]{Find the general antiderivative:
\[
\int \left(4x^3 - 2x\right)\,dx
\]
Reverse the power rule term by term and remember the constant of integration.}

\problem[9cm]{A car's velocity $v(t)$, in metres per second, is recorded every
2 seconds:

\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$t$ (s) & 0 & 2 & 4 & 6 & 8 \\
\hline
$v(t)$ (m/s) & 5 & 9 & 12 & 14 & 15 \\
\hline
\end{tabular}
\end{center}
\smallskip
Estimate the distance travelled on $[0, 8]$ using a \textbf{left} Riemann sum
with 4 subintervals of equal width.

\smallskip
Write out the sum before evaluating it, and state $\Delta t$.}

\problem[6cm]{Evaluate exactly:
\[
\int_1^3 3x^2 \,dx
\]
Find an antiderivative first, then write the $F(b) - F(a)$ subtraction before
simplifying.}

\problem[7cm]{Define \; $F(x) = \displaystyle\int_0^x (2t + 1)\,dt$.

\smallskip
\textbf{(a)} Find $F(3)$.

\smallskip
\textbf{(b)} In one sentence, say what $F(3)$ measures about the graph of
$y = 2t+1$ between $t = 0$ and $t = 3$.}

\problem[8cm]{Using the same velocity data:

\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$t$ (s) & 0 & 2 & 4 & 6 & 8 \\
\hline
$v(t)$ (m/s) & 5 & 9 & 12 & 14 & 15 \\
\hline
\end{tabular}
\end{center}
\smallskip
Estimate the distance travelled on $[0,8]$ with a \textbf{trapezoidal} sum using
all four subintervals.

\smallskip
Write the trapezoidal formula with its coefficients before substituting, and say
in one sentence why the interior readings are counted twice.}

\problem[6cm]{Find the general antiderivative:
\[
\int \left(\cos x + 4x\right)\,dx
\]
Handle the trigonometric term and the power term separately.}

\problem[7cm]{Evaluate exactly:
\[
\int_0^2 \left(x^3 - 4x\right)\,dx
\]
\textbf{(a)} Write $F(2) - F(0)$ and simplify.

\smallskip
\textbf{(b)} Your answer is negative. Explain in one sentence what that says
about the graph of $y = x^3 - 4x$ on $[0,2]$, and why it is not a contradiction
for an ``area'' calculation.}

\problem[8cm]{Let \; $g(x) = \displaystyle\int_1^x f(t)\,dt$, \; where
$f(t) = 3t^2 - 6t$.

\smallskip
\textbf{(a)} Find $g(4)$. Put this value on the answer line.

\smallskip
\textbf{(b)} Find $g'(3)$. Do this \emph{without} integrating, and name the
theorem that lets you.

\smallskip
\textbf{(c)} Is $g$ increasing or decreasing at $x = 3$? Justify from your answer
to (b).}

\problem[8cm]{Let $f(x) = x^2 + 1$ on the interval $[0, 2]$.

\smallskip
\textbf{(a)} Estimate $\displaystyle\int_0^2 f(x)\,dx$ using a \textbf{right}
Riemann sum with $n = 4$ subintervals. The four right-endpoint values of $f$ are
1.25, 2, 3.25 and 5. Put the estimate on the answer line.

\smallskip
\textbf{(b)} The exact value of the integral is $\tfrac{14}{3}$. Is your estimate
an overestimate or an underestimate?

\smallskip
\textbf{(c)} Justify your answer to (b) from the behaviour of $f$ on $[0,2]$,
not from the two numbers. Your argument should work for \emph{any} increasing
function.}

\problem[10cm]{\textbf{Challenge.} Water flows into a tank at the rate
\[
r(t) = 4t + 3 \quad \text{gallons per minute}, \qquad 0 \le t \le 6 .
\]
At time $t = 0$ the tank already holds 10 gallons. Let $A(t)$ be the number of
gallons in the tank at time $t$.

\smallskip
\textbf{(a)} How many gallons enter the tank during the first 6 minutes? Set up
and evaluate a definite integral, and put the value on the answer line.

\smallskip
\textbf{(b)} How many gallons are in the tank at $t = 6$?

\smallskip
\textbf{(c)} Explain, using the Fundamental Theorem, why the integral in part
(a) gives the \emph{change} in the amount rather than the amount itself, and why
the initial 10 gallons has to be added separately. Write
$A(t) = A(0) + \int_0^t r(u)\,du$ and say what each piece contributes.}
