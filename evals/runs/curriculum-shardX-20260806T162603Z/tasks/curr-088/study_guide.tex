% =================== SKILL 1: add-amounts ===================
\skillheading{Put Amounts Together: Add}

\begin{formulabox}
\textbf{Rule:} When two amounts are put together (``in all'', ``total'',
``altogether''), \textbf{add} them. Both amounts must use the \emph{same unit}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A box holds 250 g of pasta and a jar holds 130 g of rice.
What is the total mass?
\step{Both masses are in grams and the question asks for the amount all
together --- that means add.}
\step{$250 + 130 = 380$.}

The total mass is $\ans{380\ \text{g}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A can holds 320 g of soup and a cup holds 165 g of yogurt.
What is the total mass?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{485\ \text{g}}$}
\end{tryitbox}

% =================== SKILL 2: subtract-compare ===================
\skillheading{Take Away or Compare: Subtract}

\begin{formulabox}
\textbf{Rule:} When an amount is \emph{taken away}, or the question asks
``how much more'' or ``how much is left'', \textbf{subtract}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A bottle holds 600 mL of water. Nia pours out 240 mL.
How much water is left?
\step{An amount is poured away and the question asks what is left --- that
means subtract.}
\step{$600 - 240 = 360$.}

There are $\ans{360\ \text{mL}}$ left.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A barrel holds 48 L of rainwater. 19 L drains out. How many
liters are left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{29\ \text{L}}$}
\end{tryitbox}

% =================== SKILL 3: multiply-groups ===================
\skillheading{Equal Groups: Multiply}

\begin{formulabox}
\textbf{Rule:} When there are \emph{equal groups} and you want the total,
\textbf{multiply}: (groups) $\times$ (amount in each group).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} One tin of paint has a mass of 3 kg. What is the mass of
6 tins?
\step{Every tin has the same mass, so this is equal groups --- multiply the
number of tins by the mass of one tin.}
\step{$6 \times 3 = 18$.}

The 6 tins have a mass of $\ans{18\ \text{kg}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} One can of milk holds 4 L. How many liters do 5 cans hold?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20\ \text{L}}$}
\end{tryitbox}

% =================== SKILL 4: divide-share ===================
\skillheading{Equal Shares: Divide}

\begin{formulabox}
\textbf{Rule:} When one total is split into \emph{equal shares} and you want
the size of each share, \textbf{divide}: (total) $\div$ (number of shares).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A pitcher holds 18 L of tea, poured equally into 6 jugs.
How many liters go into each jug?
\step{One total is split into equal shares and the question asks for each
share --- that means divide.}
\step{$18 \div 6 = 3$.}

Each jug gets $\ans{3\ \text{L}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A sack of 45 g of seeds is split equally into 9 pots. How
many grams go into each pot?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5\ \text{g}}$}
\end{tryitbox}
