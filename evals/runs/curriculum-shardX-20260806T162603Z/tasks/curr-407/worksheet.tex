\wstitleblock{Locating Zeros and Turning-Point Constraints}{Precalculus}{}

\begin{center}
\small\textit{Factoring to locate zeros, reading multiplicity against degree, and bounding the number of turning points}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Factor completely before listing zeros --- a zero you cannot see in the factored form is a zero you will miss.
  \item The multiplicities of the zeros add up to the degree, and a polynomial of degree $n$ has at most $n - 1$ turning points.
  \item Give exact zeros. Leave fractions as fractions.
\end{itemize}
\vspace{0.15cm}

\problem[4.5cm]{Find all real zeros of $p(x) = x^{2} - 7x + 12$.}

\problem[4.5cm]{A polynomial has degree $6$. What is the greatest possible number of turning points its graph can have?}

  \par\vspace*{2.6cm}\ansline
  \item[(b)] State the multiplicity of each zero and whether the curve crosses or touches the axis there.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}
