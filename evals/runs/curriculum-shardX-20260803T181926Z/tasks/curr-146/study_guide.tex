\sstitleblock{Reading and Plotting Points on a Grid}

\skillheading{Skill 1 --- Reading a point: across first, then up}

\begin{formulabox}
\textbf{An ordered pair is written $(x, y)$.} The first number says how far to travel
\textbf{across} from the corner marked $0$; the second says how far to travel
\textbf{up}.\par
\smallskip
Always in that order. $(3, 7)$ and $(7, 3)$ use the same two numbers and land in completely
different places, which is why they are called \emph{ordered} pairs.\par
\smallskip
A coordinate of $0$ is a real instruction: it means \emph{do not travel that way at all}.
The corner itself is $(0, 0)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write point $P$ as an ordered pair.
\begin{center}
\begin{tikzpicture}[scale=0.36]
  \draw[gray!35, very thin] (0,0) grid (9,9);
  \draw[->, thick] (0,0) -- (9.6,0) node[right, font=\tiny] {$x$};
  \draw[->, thick] (0,0) -- (0,9.6) node[above, font=\tiny] {$y$};
  \foreach \t in {1,...,9} {\node[below, font=\tiny] at (\t,0) {\t}; \node[left, font=\tiny] at (0,\t) {\t};}
  \node[below left, font=\tiny] at (0,0) {0};
  \fill[blue!80] (4,7) circle (3pt);
  \node[above right, font=\small] at (4,7) {$P$};
  \fill[green!45!black] (8,2) circle (3pt);
  \node[above right, font=\small] at (8,2) {$Q$};
  \draw[red, thick, ->] (0,0) -- (4,0);
  \draw[red, thick, ->] (4,0) -- (4,7);
\end{tikzpicture}
\end{center}
\step{A point is read as a journey from the corner, and the journey always goes across
before it goes up --- so trace the red arrows in that order.}
\step{Across the bottom to sit under $P$: four squares. Then straight up to $P$: seven
squares.}

$\Rightarrow\quad P = (\ans{4},\, \ans{7})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write the green point $Q$ on the same grid as an ordered pair.\\[2pt]
\rotatebox{180}{\footnotesize check: $Q = (\ans{8},\, \ans{2})$}
\end{tryitbox}

\skillheading{Skill 2 --- Plotting a point and measuring a line}

\begin{formulabox}
\textbf{To plot $(x, y)$:} put your finger on $0$, slide across $x$ squares, then go up $y$
squares, and mark the spot.\par
\smallskip
\textbf{To measure a straight line between two points:} if the line is \emph{upright}, both
points have the same $x$, so subtract the two $y$-coordinates. If the line is \emph{flat},
both points have the same $y$, so subtract the two $x$-coordinates. Only one of the two
coordinates does any work.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plot $R(5, 2)$ and $S(5, 7)$. How many squares long is
$\overline{RS}$?
\step{Both points are five squares across, so the line joining them is upright and only the
up-and-down numbers can differ --- subtract those.}
\step{$y$ of $S$ minus $y$ of $R$: \quad $7 - 2$.}

$\Rightarrow\quad \overline{RS} = \ans{5}$ squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Plot $T(3, 3)$ and $U(3, 9)$. How many squares long is
$\overline{TU}$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$ squares}
\end{tryitbox}

\skillheading{Skill 3 --- Two rules make a list of points}

\begin{formulabox}
\textbf{Run two counting rules side by side.} The number from the first rule is the $x$, the
number from the second rule is the $y$, and together they make one ordered pair.\par
\smallskip
Plot the pairs in order and they fall on a straight line. If one rule adds $1$ each time
and the other adds $3$, then $y$ climbs three times as fast as $x$ --- so $y$ is always
three times $x$, and you can jump straight to any pair without writing out the rows in
between.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rule A: start at $0$, add $1$. Rule B: start at $0$, add $3$. What
is $y$ when $x = 2$?
\step{Rule B climbs three steps for every one step of Rule A, so instead of listing rows,
multiply the $x$ by three.}
\step{The pairs run $(0, 0)$, $(1, 3)$, $(2, 6)$, and $3 \times 2 = 6$ agrees.}

$\Rightarrow\quad y = \ans{6}$, so the pair is $(2, 6)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same two rules. What is $y$ when $x = 4$?\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{12}$, so the pair is $(4, 12)$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the commonest mistake on a grid is doing the two journeys in the
wrong order --- up first, then across. Say the pair out loud as you plot it: ``across four,
up seven.'' If a point lands somewhere unexpected, check the order before checking
anything else.
\end{watchoutbox}
