\wstitleblock{Unit Fractions in Area Models}{Grade 3 --- Fractions as Numbers}{}

\vspace{0.1cm}
\noindent\textbf{Directions:}
\begin{itemize}
  \item A \textbf{unit fraction} names \emph{one} of the equal pieces a whole is cut into.
  \item For each model, count how many \emph{equal} pieces make the whole shape. That count is the
        bottom number of the fraction; the top number stays $1$.
  \item Write fractions in the form $\frac{1}{\Box}$, and write your final answer on the answer line.
\end{itemize}

\vspace{0.15cm}

\problem[5cm]{Both shapes below were cut into four pieces, and one piece of each is shaded.
Only one of them shows \emph{fourths}.
\par\vspace{0.15cm}
