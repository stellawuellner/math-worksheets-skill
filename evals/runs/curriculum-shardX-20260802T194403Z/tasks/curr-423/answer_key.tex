\problem{Solve $2\sin x - 1 = 0$ on $[0, 2\pi)$.

\textbf{Solution.} Isolate the trig function: $\sin x = \tfrac{1}{2}$. The
reference angle is $\tfrac{\pi}{6}$, and sine is positive in quadrants I and II,
so the two solutions in one revolution are $\tfrac{\pi}{6}$ and
$\pi - \tfrac{\pi}{6} = \tfrac{5\pi}{6}$.
$\ans{x = \tfrac{\pi}{6},\ \tfrac{5\pi}{6}}$
\noindent \emph{Count check:} the graph crosses $y = \tfrac{1}{2}$ twice on
$[0, 2\pi)$, matching the two answers.}

\problem{Solve $2\cos x + \sqrt{3} = 0$ on $[0, 2\pi)$.

\textbf{Solution.} $\cos x = -\tfrac{\sqrt{3}}{2}$. The reference angle is
$\tfrac{\pi}{6}$; cosine is negative in quadrants II and III, giving
$\pi - \tfrac{\pi}{6} = \tfrac{5\pi}{6}$ and
$\pi + \tfrac{\pi}{6} = \tfrac{7\pi}{6}$.
$\ans{x = \tfrac{5\pi}{6},\ \tfrac{7\pi}{6}}$}

\problem{Solve $\tan x + 1 = 0$ on $[-\pi, \pi)$.

\textbf{Solution.} $\tan x = -1$, reference angle $\tfrac{\pi}{4}$. Tangent has
period $\pi$, so its solutions are spaced $\pi$ apart:
$x = -\tfrac{\pi}{4} + k\pi$. The members of that family lying in
$[-\pi, \pi)$ are $-\tfrac{\pi}{4}$ (from $k = 0$) and
$-\tfrac{\pi}{4} + \pi = \tfrac{3\pi}{4}$ (from $k = 1$).
$\ans{x = -\tfrac{\pi}{4},\ \tfrac{3\pi}{4}}$
\noindent \emph{Common error:} using period $2\pi$ and reporting only one
solution, or listing $\tfrac{7\pi}{4}$, which lies outside this interval.}

\problem{Verify $\cos 2x = 1 - 2\sin^2 x$.

\textbf{Solution.} Start from the difference form of the double-angle identity
and replace $\cos^2 x$ using the Pythagorean identity:
\begin{align*}
\cos 2x &= \cos^2 x - \sin^2 x \\
        &= (1 - \sin^2 x) - \sin^2 x \\
        &= 1 - 2\sin^2 x
\end{align*}
$\ans{\cos 2x = 1 - 2\sin^2 x}$
\noindent This is the form to reach for whenever an equation mixes $\cos 2x$
with $\sin x$: it makes every term a function of the same angle.}

\problem{Solve $\cos 2x + 3\sin x = 2$ on $[0, 2\pi)$.

\textbf{Solution.} Substitute the identity from problem 4 so that every term is
a function of $\sin x$:
\begin{align*}
(1 - 2\sin^2 x) + 3\sin x &= 2 \\
-2\sin^2 x + 3\sin x - 1 &= 0 \\
2\sin^2 x - 3\sin x + 1 &= 0 \\
(2\sin x - 1)(\sin x - 1) &= 0
\end{align*}
So $\sin x = \tfrac{1}{2}$, giving $x = \tfrac{\pi}{6}$ and $\tfrac{5\pi}{6}$;
or $\sin x = 1$, giving $x = \tfrac{\pi}{2}$.
$\ans{x = \tfrac{\pi}{6},\ \tfrac{\pi}{2},\ \tfrac{5\pi}{6}}$}

\problem{Solve $2\cos^2 x - \cos x - 1 = 0$ on $[0, 2\pi)$.

\textbf{Solution.} Treat $\cos x$ as the variable and factor:
\[ (2\cos x + 1)(\cos x - 1) = 0 \]
$\cos x = -\tfrac{1}{2}$ gives the quadrant II and III angles
$\tfrac{2\pi}{3}$ and $\tfrac{4\pi}{3}$; $\cos x = 1$ gives $x = 0$ (and
$2\pi$, which is excluded because the interval is half-open).
$\ans{x = 0,\ \tfrac{2\pi}{3},\ \tfrac{4\pi}{3}}$
\noindent \emph{Grading note:} an answer that includes $2\pi$ has misread the
interval; an answer missing $x = 0$ has divided by $(\cos x - 1)$ instead of
factoring.}

\problem{Ferris wheel height at $t = 10$ seconds, $h(t) = 9 - 8\cos(\pi t/15)$.

\textbf{Solution.} $\dfrac{\pi (10)}{15} = \dfrac{2\pi}{3}$, and
$\cos\dfrac{2\pi}{3} = -\dfrac{1}{2}$, so
$h(10) = 9 - 8\left(-\tfrac{1}{2}\right) = 9 + 4$.
$\ans{13\ \text{m}}$
\noindent \emph{Sanity check:} the model has centre height $9$ m and radius
$8$ m, so every value must lie between $1$ m and $17$ m; $13$ m does.}

\problem{Find every $t \in [0, 30)$ seconds with $h(t) = 13$ metres.

\textbf{Solution.} Set the model equal to the target and isolate the cosine:
\begin{align*}
9 - 8\cos\!\left(\frac{\pi t}{15}\right) &= 13 \\
\cos\!\left(\frac{\pi t}{15}\right) &= -\frac{1}{2}
\end{align*}
Let $u = \dfrac{\pi t}{15}$. As $t$ runs over $[0, 30)$, $u$ runs over
$[0, 2\pi)$ --- exactly one revolution --- so $u = \tfrac{2\pi}{3}$ or
$u = \tfrac{4\pi}{3}$. Converting back, $t = \dfrac{15u}{\pi}$ gives
$t = 10$ and $t = 20$.
$\ans{t = 10\ \text{s},\ 20\ \text{s}}$
\noindent The two times are symmetric about $t = 15$ s, the top of the wheel ---
the rider passes $13$ m once going up and once coming down.}

\problem{Solve $\sin 2x = \cos x$ on $[0, 2\pi)$.

\textbf{Solution.} Rewrite the double angle, move everything to one side, and
factor --- never divide by $\cos x$, which would delete the solutions where
$\cos x = 0$:
\begin{align*}
2\sin x \cos x - \cos x &= 0 \\
\cos x\,(2\sin x - 1) &= 0
\end{align*}
$\cos x = 0$ gives $x = \tfrac{\pi}{2}$ and $\tfrac{3\pi}{2}$;
$\sin x = \tfrac{1}{2}$ gives $x = \tfrac{\pi}{6}$ and $\tfrac{5\pi}{6}$.
$\ans{x = \tfrac{\pi}{6},\ \tfrac{\pi}{2},\ \tfrac{5\pi}{6},\ \tfrac{3\pi}{2}}$
\noindent \emph{Grading note:} exactly the two answers a student loses by
dividing through by $\cos x$ are $\tfrac{\pi}{2}$ and $\tfrac{3\pi}{2}$.}

\problem{Solve $2\sin^2\theta - \sin\theta - 1 = 0$ on $[0^\circ, 360^\circ)$.

\textbf{Solution.} Factor the quadratic in $\sin\theta$:
\[ (2\sin\theta + 1)(\sin\theta - 1) = 0 \]
$\sin\theta = 1$ gives $\theta = 90^\circ$. $\sin\theta = -\tfrac{1}{2}$ has
reference angle $30^\circ$ and is negative in quadrants III and IV, giving
$180^\circ + 30^\circ = 210^\circ$ and $360^\circ - 30^\circ = 330^\circ$.
$\ans{\theta = 90^\circ,\ 210^\circ,\ 330^\circ}$
\noindent \emph{Grading note:} $\sin\theta = 1$ contributes only one solution
per revolution because $1$ is the maximum of sine --- a student who reports two
angles there has applied the quadrant rule without checking the extreme value.}
