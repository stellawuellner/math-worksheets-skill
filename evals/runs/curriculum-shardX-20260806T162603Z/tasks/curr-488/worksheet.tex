\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every length in these problems is a real measurement. The units are stated in each problem; every volume is therefore a cubic unit, and the answer line names which one.
  \item Write the integral before you evaluate it: state the cross-sectional area $A(x)$, or the shell radius and height, and the limits.
  \item Give an exact answer (a fraction, a multiple of $\pi$, or a radical) and then a decimal to two places.
\end{itemize}

\vspace{0.2cm}

\problem{The base of a solid is the region in the first quadrant bounded by the line $y = x$, the $x$-axis, and the line $x = 4$, where $x$ and $y$ are measured in metres. Every cross section perpendicular to the $x$-axis is a square with one side in the base. Find the volume of the solid.
\par\vspace*{5.5cm}\answerline{m$^3$}}

\problem{The base of a solid is the region bounded by $y = \sqrt{x}$, the $x$-axis, and the line $x = 9$, where $x$ and $y$ are in centimetres. Every cross section perpendicular to the $x$-axis is a semicircle whose diameter runs from the $x$-axis to the curve. Find the volume of the solid.
\par\vspace*{5.5cm}\answerline{cm$^3$}}

\problem{The region bounded by $y = x^2$, the $x$-axis, and the line $x = 3$, with $x$ and $y$ in feet, is revolved about the $y$-axis. Use cylindrical shells to find the volume of the resulting solid.
\par\vspace*{5.5cm}\answerline{ft$^3$}}

\problem{The base of a solid is the triangular region bounded by the line $y = 4 - x$ and the two coordinate axes, where $x$ and $y$ are in centimetres. Every cross section perpendicular to the $x$-axis is an equilateral triangle with one side in the base. (Recall that an equilateral triangle of side $s$ has area $\frac{\sqrt{3}}{4}s^2$.) Find the volume of the solid.
\par\vspace*{6cm}\answerline{cm$^3$}}

\problem{The region bounded by $y = 4 - x^2$ and the $x$-axis, for $x \ge 0$, is revolved about the $y$-axis. All lengths are in metres. Use cylindrical shells to find the volume of the resulting solid.
\par\vspace*{6cm}\answerline{m$^3$}}

\problem{The region bounded by $y = \sqrt{x}$, the $x$-axis, and the line $x = 9$ (lengths in centimetres) is revolved about the $y$-axis.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Use cylindrical shells to find the volume of the solid.
  \item[(b)] Explain, in one or two sentences, why the shell setup uses the curve exactly as written while a washer setup would first have to rewrite it.
\end{itemize}
\par\vspace*{6cm}\answerline{cm$^3$}}

\problem{The base of a solid is the region enclosed by the curves $y = x$ and $y = x^2$, with $x$ and $y$ in metres. Every cross section perpendicular to the $x$-axis is a square.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the two $x$-values where the curves meet. \ansline
  \item[(b)] Find the volume of the solid.
\end{itemize}
\par\vspace*{5.5cm}\answerline{m$^3$}}

\problem{The region bounded by $y = x^2$, the $x$-axis, and the line $x = 2$ (lengths in centimetres) is revolved about the vertical line $x = 4$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Use cylindrical shells to find the volume of the resulting solid.
  \item[(b)] Explain why the shell radius is $4 - x$ and not $x$, and check your expression at both ends of the interval.
\end{itemize}
\par\vspace*{6cm}\answerline{cm$^3$}}
