\wstitleblock{Multiplying Fractions and Mixed Numbers}{Grade 4--5 Fractions}{}

\noindent\small
Multiplying a fraction by a fraction, a fraction by a whole number, mixed numbers
by rewriting them as improper fractions, and finding a missing factor.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=3pt]
  \item Show the multiplication you set up, not just the final number.
  \item Give every answer in simplest form. Improper fractions are fine.
  \item Rewrite a mixed number as an improper fraction \emph{before} you multiply.
\end{itemize}
\normalsize

\problem[5cm]{Multiply. Simplify your answer.
\[ \frac{2}{3} \times \frac{3}{5} \]}

\problem[5cm]{Multiply. Simplify your answer.
\[ \frac{3}{4} \times \frac{1}{2} \]}

\problem[5cm]{Multiply a fraction by a whole number. Simplify your answer.
\[ \frac{5}{6} \times 4 \]}

\problem[5cm]{Multiply. Look for a common factor you can cancel before you multiply.
\[ \frac{3}{8} \times \frac{2}{9} \]}

\problem[6cm]{Rewrite the mixed number as an improper fraction first, then multiply.
\[ 2\frac{1}{2} \times \frac{3}{5} \]}

\problem[6cm]{Find the missing factor. What number times $\dfrac{3}{4}$ gives $\dfrac{9}{20}$?
\[ \frac{3}{4} \times x = \frac{9}{20} \]}

\problem[6cm]{Multiply two mixed numbers. Simplify your answer.
\[ 1\frac{1}{3} \times 2\frac{1}{4} \]}

\problem[6cm]{A batch of trail mix uses $\dfrac{3}{8}$ of a cup of raisins.
Ravi makes 12 batches. How many cups of raisins does he use in all?}

\problem[6cm]{Multiply. Cancel common factors before multiplying --- the numbers stay small.
\[ \frac{7}{10} \times \frac{5}{14} \]}

\problem[7cm]{A rectangular doormat measures $2\frac{3}{4}$ feet by $1\frac{1}{2}$ feet.
Find its area. Give the answer as a decimal number of square feet.
\answerline{ft$^2$}}

\problem[6cm]{Find the missing factor.
\[ x \times 2\frac{1}{2} = 4 \]}

\problem[8cm]{A trail is $4\frac{1}{2}$ kilometres long. On Saturday Maya walks
$\dfrac{2}{3}$ of the whole trail. On Sunday she walks $\dfrac{1}{2}$ of the part
that was left. How far does she walk on Sunday?
\answerline{km}}
