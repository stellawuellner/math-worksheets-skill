\sstitleblock{Right-Triangle Trigonometry --- Choosing and Checking}

\skillheading{Skill 1 --- Pick the ratio that fits your two sides}

\begin{formulabox}
\textbf{SOH-CAH-TOA:}\quad $\sin A = \text{opp}/\text{hyp}$, \quad
$\cos A = \text{adj}/\text{hyp}$, \quad $\tan A = \text{opp}/\text{adj}$.\par
\smallskip
\textbf{Method:} label the sides \emph{from the angle you are working from} ---
opposite, adjacent, hypotenuse --- then pick the ratio that contains the side
you have and the side you want. The hypotenuse is always opposite the right
angle; it is never the adjacent side.\par
\smallskip
\textbf{Special triangles:} $30$-$60$-$90$ has sides $1 : \sqrt{3} : 2$
(shortest opposite the $30^\circ$); $45$-$45$-$90$ has $1 : 1 : \sqrt{2}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The hypotenuse is $15$ and the angle is $28^\circ$. Find
the side opposite that angle.
\step{Want the opposite side, know the hypotenuse --- opposite with hypotenuse
is sine, the S of SOH.}
\step{$\sin 28^\circ = x/15$, so $x = 15\sin 28^\circ = 7.0420\ldots$ with the
calculator in degree mode.}
$\ans{x \approx 7.04}$, and it is shorter than the hypotenuse, as a leg must be.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The hypotenuse is $18$ and the angle is $40^\circ$. Find
the leg adjacent to that angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13.79}$}
\end{tryitbox}

\skillheading{Skill 2 --- Inverse trig for a missing angle}

\begin{formulabox}
\textbf{Two sides given, angle wanted:} build the ratio first, then apply the
inverse: $A = \sin^{-1}(\text{opp}/\text{hyp})$, \quad
$A = \cos^{-1}(\text{adj}/\text{hyp})$, \quad
$A = \tan^{-1}(\text{opp}/\text{adj})$.\par
\smallskip
\textbf{Two different things:} $\sin^{-1}$ is the INVERSE (it returns an angle);
$(\sin A)^{-1}$ is the RECIPROCAL, $1/\sin A$ (it returns a number). The
calculator button marked $\sin^{-1}$ is the inverse.\par
\smallskip
\textbf{Upside-down check:} the bigger side always faces the bigger angle. If
the side opposite your angle is the shorter one, your answer must be under
$45^\circ$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The side opposite $A$ is $5$ and the hypotenuse is $13$.
Find $\angle A$.
\step{Two sides are known and an angle is wanted, and the pair is
opposite-with-hypotenuse, so it is the inverse sine that undoes it.}
\step{$\sin A = 5/13$, so $A = \sin^{-1}(5/13) = 22.6198\ldots$ degrees.}
$\ans{A \approx 22.62^\circ}$, under $45^\circ$ as the short opposite side
demands.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The side adjacent to $A$ is $8$ and the hypotenuse is
$17$. Find $\angle A$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{61.93^\circ}$}
\end{tryitbox}

\skillheading{Skill 3 --- Degree mode and the sanity check}

\begin{formulabox}
\textbf{Set the mode first.} Every angle on this sheet is in degrees, so the
screen must read DEG. In radian mode the calculator answers a different question
and gives no warning.\par
\smallskip
\textbf{Two checks that catch a mode error instantly:} $\sin 30^\circ$ must
read exactly $0.5$, and a leg must come out shorter than the hypotenuse. A
negative length, or a length larger than the hypotenuse, is a mode error until
proven otherwise.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The leg adjacent to a $25^\circ$ angle is $30$. Find the
opposite leg.
\step{Opposite over adjacent is tangent, so the setup is a tangent --- and
tangent is where a mode error does the most damage, so check DEG before
pressing enter.}
\step{$30\tan 25^\circ = 13.9892\ldots$ in degree mode; the same keystrokes in
radian mode return $-4.01$, which no length can be.}
$\ans{13.99}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The leg adjacent to a $63^\circ$ angle is $12$. Find the
opposite leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{23.55}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a wrong ratio usually returns a believable number ---
sine and cosine of the same angle both give a length shorter than the
hypotenuse. Size checks catch mode errors; only naming opposite and adjacent
catches a wrong ratio.
\end{watchoutbox}
