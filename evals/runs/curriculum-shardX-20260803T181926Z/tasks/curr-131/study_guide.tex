\sstitleblock{A Fraction of a Whole Number or Set}

\skillheading{1. One equal group: a unit fraction of a set}

\begin{formulabox}
\textbf{The bottom number makes the groups.} It tells you how many
\emph{equal} groups to break the whole set into, so you \textbf{divide} by
it.\\[3pt]
$\frac{1}{5}$ of $20$ means: break $20$ into $5$ equal groups
($20 \div 5 = 4$) and take one group. The answer is how many things are
\emph{in} a group, not how many groups there are.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is $\frac{1}{6}$ of $30$ counters?
\step{The top number is $1$, so this is just ``make the groups and take one'' ---
only the bottom number does any work.}
\step{Break $30$ into $6$ equal groups: $30 \div 6 = 5$ in each group.}
\begin{center}
\begin{tikzpicture}[x=0.42cm, y=0.42cm]
  \fill[blue!35] (0,0) rectangle (1,1);
  \draw[thick] (0,0) rectangle (6,1);
  \foreach \i in {1,2,3,4,5} \draw[thick] (\i,0) -- (\i,1);
\end{tikzpicture}
\end{center}

One group holds $\ans{5}$ counters
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What is $\frac{1}{4}$ of $28$ marbles?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{2. Several equal groups: divide, then multiply}

\begin{formulabox}
\textbf{Divide by the bottom, then multiply by the top.} The top number says
how many of the equal groups to take.\\[3pt]
$\frac{3}{4}$ of $20$: \quad $20 \div 4 = 5$ in each group, then
$3 \times 5 = 15$.\\[2pt]
Always in that order. Dividing first keeps the numbers small and keeps the
picture true to the words.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is $\frac{3}{5}$ of $40$ stamps?
\step{The bottom number $5$ says five equal groups and the top number $3$ says
take three of them, so this needs both moves, not just the division.}
\step{Divide: $40 \div 5 = 8$ stamps in one group. Multiply:
$3 \times 8$.}
\begin{center}
\begin{tikzpicture}[x=0.42cm, y=0.42cm]
  \fill[teal!35] (0,0) rectangle (3,1);
  \draw[thick] (0,0) rectangle (5,1);
  \foreach \i in {1,2,3,4} \draw[thick] (\i,0) -- (\i,1);
\end{tikzpicture}
\end{center}

Three groups of eight: $\ans{24}$ stamps
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What is $\frac{2}{7}$ of $42$ beads?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{3. Working backwards to find the whole}

\begin{formulabox}
When the \emph{part} is known and the whole is missing, run the two moves in
reverse:\\[3pt]
\textbf{Divide the part by the top number} to get one group, then
\textbf{multiply by the bottom number} to rebuild the whole.\\[2pt]
Forwards you divide then multiply; backwards you divide then multiply
too --- but by the other numbers. Draw the bar and it is hard to mix up.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\frac{2}{3}$ of a number is $18$. What is the number?
\step{The $18$ covers two groups, not one, so it cannot be multiplied by $3$
straight away --- find one group first.}
\step{One group: $18 \div 2 = 9$. Whole: $3 \times 9$, since the bottom number
says the whole is three groups.}

The number is $\ans{27}$ \quad (check: $27 \div 3 = 9$ and
$2 \times 9 = 18$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\frac{3}{4}$ of a number is $21$. What is the
number?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} do not divide by the top number by mistake --- the
\emph{bottom} number always makes the groups. And do not stop after dividing:
if the top number is bigger than $1$, there is a multiplication still to do.
\end{watchoutbox}
