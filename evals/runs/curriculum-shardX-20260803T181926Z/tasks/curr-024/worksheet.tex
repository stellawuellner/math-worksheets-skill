\wstitleblock{Adding Within 10 --- Finding and Fixing a Counting-On Mistake}{Kindergarten--Grade 1}{}

\noindent\large When we \textbf{count on}, we hold the first number in our head
and take one hop for each thing we add.
\begin{itemize}[leftmargin=0.8cm, itemsep=2pt, topsep=3pt]
  \item Say the first number \emph{quietly}. Do \textbf{not} count it as a hop.
  \item Start counting with the \textbf{next} number. Take one hop for each dot
        you are adding.
  \item Each friend below made the same kind of mistake. Find it, then write the
        answer that is really right.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{Sam has $6$ stickers. He gets $3$ more.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,6} {\fill[black!70] (\i,0) circle (0.22);}
  \foreach \i in {1,...,3} {\draw[thick] (\i+7.2,0) circle (0.22);}
\end{tikzpicture}
\end{center}
Sam counts on like this: ``six \ldots\ seven \ldots\ eight.'' He writes $8$.
\par\smallskip
\textbf{(a)} Circle the number Sam should \emph{not} have counted as a hop.
\par
\textbf{(b)} Count on the right way. How many stickers does Sam have?}

\problem[5cm]{Mia has $4$ blocks. She gets $2$ more.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,4} {\fill[black!70] (\i,0) circle (0.22);}
  \foreach \i in {1,...,2} {\draw[thick] (\i+5.2,0) circle (0.22);}
\end{tikzpicture}
\end{center}
Mia counts on like this: ``four \ldots\ five.'' She writes $5$.
\par\smallskip
\textbf{(a)} How many hops should Mia take?
\par
\textbf{(b)} Count on the right way. How many blocks does Mia have?}

\problem[5.5cm]{Ravi has $3$ toy cars. He gets $5$ more.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,3} {\fill[black!70] (\i,0) circle (0.22);}
  \foreach \i in {1,...,5} {\draw[thick] (\i+4.2,0) circle (0.22);}
\end{tikzpicture}
\end{center}
Ravi starts at his little number and tries to take five hops. He loses one hop
along the way and writes $7$.
\par\smallskip
\textbf{(a)} Which number should Ravi start from so he has fewer hops to take?
\par
\textbf{(b)} Count on that way. How many cars does Ravi have?}

\problem[6cm]{Jo has $5$ shells. She wants $9$ shells.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,5} {\fill[black!70] (\i,0) circle (0.22);}
  \draw[thick, dashed, rounded corners] (6.1,-0.5) rectangle (9.6,0.5);
  \node at (7.85,0) {\large ?};
\end{tikzpicture}
\end{center}
Jo counts out loud: ``five \ldots\ six \ldots\ seven \ldots\ eight \ldots\
nine.'' She counted five number names, so she writes $5$.
\par\smallskip
\textbf{(a)} Jo should count her \textbf{hops}, not the words she says. How many
hops did she really take?
\par
\textbf{(b)} How many more shells does Jo need?}

\problem[6cm]{Ana has $2$ marbles. She gets $7$ more.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,2} {\fill[black!70] (\i,0) circle (0.22);}
  \foreach \i in {1,...,7} {\draw[thick] (\i+3.2,0) circle (0.22);}
\end{tikzpicture}
\end{center}
Ana starts at her little number and writes $8$.
\par\smallskip
\textbf{(a)} Show a start number that needs only two hops.
\par
\textbf{(b)} Count on that way. How many marbles does Ana have?}

\problem[6.5cm]{Ben did two problems. He made the same mistake in both.
\begin{center}
\begin{tikzpicture}[x=0.7cm, y=0.7cm]
  \foreach \i in {1,...,6} {\fill[black!70] (\i,0) circle (0.22);}
  \foreach \i in {1,...,4} {\draw[thick] (\i+7.2,0) circle (0.22);}
  \foreach \i in {1,...,2} {\fill[black!70] (\i,-1.3) circle (0.22);}
  \foreach \i in {1,...,6} {\draw[thick] (\i+3.2,-1.3) circle (0.22);}
\end{tikzpicture}
\end{center}
Ben has $6$ pencils and gets $4$ more. He writes $9$.
\par\smallskip
Ben has $2$ crayons and gets $6$ more. He writes $7$.
\par\smallskip
\textbf{(a)} Tell what Ben did wrong both times.
\par
\textbf{(b)} Write the right answer for the pencils and the right answer for the
crayons.}
