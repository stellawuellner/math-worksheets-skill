\sstitleblock{Expanded \& Factored Forms}

\skillheading{Skill 1 --- Expanding (distribute)}

\begin{formulabox}
\textbf{Rule:} \quad $a(b + c) = ab + ac$ \quad --- multiply the outside factor by
every term inside. Expanded form has \emph{no parentheses}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $2(4x + 3)$
\step{To remove the parentheses, distribute the outside factor to each term inside, one product at a time.}
\step{$2 \cdot 4x = 8x$ and $2 \cdot 3 = 6$.}

$\Rightarrow\quad 2(4x + 3) = \ans{8x + 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $5(3x + 2)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15x + 10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Combining like terms}

\begin{formulabox}
\textbf{Rule:} like terms share the same variable part; add or subtract their
coefficients and keep the variable part. Constants combine only with constants.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $7x + 8 - 3x - 2$
\step{Group the terms by kind before combining: the $x$-terms together, the plain numbers together.}
\step{$x$-terms: $7x - 3x = 4x$. \ Constants: $8 - 2 = 6$.}

$\Rightarrow\quad 7x + 8 - 3x - 2 = \ans{4x + 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $10x + 1 - 6x + 4$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x + 5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Factoring out the GCF}

\begin{formulabox}
\textbf{Rule:} factoring undoes expanding: find the greatest common factor of the
terms and pull it outside: $ab + ac = a(b + c)$. Check by distributing back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $15x + 20$
\step{Find the largest number dividing both coefficients; it becomes the outside factor and the quotients stay inside.}
\step{The GCF of $15$ and $20$ is $5$: \ $15x = 5 \cdot 3x$ and $20 = 5 \cdot 4$.}

$\Rightarrow\quad 15x + 20 = \ans{5(3x + 4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $8x + 12$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4(2x + 3)}$}
\end{tryitbox}

\skillheading{Skill 4 --- Mixed rewrites: expand, then combine}

\begin{formulabox}
\textbf{Rule:} when an expression mixes a product with loose terms, expand the
product \emph{first}, then combine like terms. Work one rewrite at a time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $3(2x + 1) + 4x$
\step{The parentheses block the like terms from meeting, so expand the product first and only then combine.}
\step{$3(2x + 1) = 6x + 3$; then $6x + 3 + 4x = 10x + 3$.}

$\Rightarrow\quad 3(2x + 1) + 4x = \ans{10x + 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $2(3x + 4) - 5x$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x + 8}$}
\end{tryitbox}
