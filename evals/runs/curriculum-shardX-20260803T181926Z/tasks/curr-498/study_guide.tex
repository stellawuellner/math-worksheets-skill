\sstitleblock{Intervals of Convergence with Endpoint Checks}

\noindent\small For $\sum a_n (x-c)^n$ the ratio test always returns an open
interval $|x-c| < R$ and is \emph{silent} at $|x-c| = R$. So the work is always
the same four moves: ratio test for $R$, endpoints from $c \pm R$, substitute
each endpoint to get a numeric series, then test that series with a tool the
ratio test cannot supply.\normalsize

\skillheading{Skill 1 --- Ratio test and radius}

\begin{formulabox}
\textbf{Rule:} form $\left|a_{n+1}/a_n\right|$, cancel, and factor out $|x-c|$.
What remains is a number $L = \lim_{n\to\infty}$ of the $x$-free part.
Convergence needs $|x-c| \cdot L < 1$, so $R = 1/L$ (and $R = \infty$ when
$L = 0$). Divide top and bottom by the highest power of $n$ to take the limit.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $R$ for $\displaystyle\sum_{n=1}^{\infty}
\frac{(x-1)^n}{n \cdot 4^n}$.
\step{The terms carry a power of $(x-1)$, so the ratio test is the tool ---
build the ratio, cancel, and read off the $x$-free limit.}
\step{$\left|\frac{a_{n+1}}{a_n}\right| = |x-1| \cdot \frac{n}{4(n+1)}$, and
dividing by $n$ gives $\frac{1}{4(1+1/n)} \to \frac{1}{4}$.}

$\ans{L = \tfrac{1}{4}}$ \quad so $R = 1/L = 4$, centered at $c = 1$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $L$ for $\displaystyle\sum_{n=1}^{\infty}
\frac{(x+2)^n}{n \cdot 6^n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{L = \tfrac{1}{6}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Endpoints from center and radius}

\begin{formulabox}
\textbf{Rule:} $|x - c| < R$ is the open interval from $c - R$ to $c + R$, so
the two endpoints are exactly $c - R$ and $c + R$. Read $c$ off the
\emph{written} form $(x - c)^n$: in $(x+2)^n$ the center is $c = -2$, not $2$.
An infinite radius has no endpoints at all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A series centered at $c = 1$ has radius $R = 4$. Find its
endpoints.
\step{The ratio test gives the open set $|x - 1| < 4$; its boundary is the pair
of points where $|x-1|$ equals $4$, so just add and subtract the radius.}
\step{$c - R = 1 - 4$ and $c + R = 1 + 4$.}

$\ans{x = -3}$ \quad and \quad $\ans{x = 5}$, so the open part is $-3 < x < 5$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A series centered at $c = -2$ has radius $R = 6$. Give
both endpoints.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -8}$ and $\ans{x = 4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Substituting an endpoint}

\begin{formulabox}
\textbf{Rule:} put the endpoint value in for $x$ and simplify until no $x$ is
left. The powers always collapse: $(x-c)^n$ becomes $(\pm R)^n$, which cancels
the $R^n$ in the denominator and leaves either $1$ or $(-1)^n$ on top. That is
the point of the substitution --- it turns the power series into a numeric
series you already know.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Substitute the left endpoint $x = -3$ into
$\displaystyle\sum_{n=1}^{\infty} \frac{(x-1)^n}{n \cdot 4^n}$ and find the
exact sum.
\step{At an endpoint the ratio test is inconclusive, so substitute and identify
the resulting numeric series by name.}
\step{$x - 1 = -4$, so the terms are $\frac{(-4)^n}{n \cdot 4^n}
= \frac{(-1)^n}{n}$ --- the alternating harmonic series, convergent by the
alternating series test.}

$\ans{-\ln 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Substitute $x = -3$ into $\displaystyle\sum_{n=1}^{\infty}
\frac{(x-2)^n}{n^2 \cdot 5^n}$ and give the exact sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\dfrac{\pi^2}{12}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Deciding at each endpoint}

\begin{formulabox}
\textbf{Rule:} name the test. $\sum 1/n^p$ converges iff $p > 1$; an
alternating series with terms decreasing to $0$ converges; if the terms do not
approach $0$ the series diverges by the $n$th-term test. A converging endpoint
earns a square bracket, a diverging one a round bracket --- and the two
endpoints are decided \emph{separately}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Decide the right endpoint $x = 7$ of
$\displaystyle\sum_{n=1}^{\infty} \frac{(x-2)^n}{n^2 \cdot 5^n}$.
\step{Substitute, then match the numeric series to a test --- here the sign
pattern is constant, so a $p$-series comparison is the right tool.}
\step{$x - 2 = 5$, so the terms are $\frac{5^n}{n^2 \cdot 5^n} = \frac{1}{n^2}$:
a $p$-series with $p = 2 > 1$, so it converges, and $x = 7$ gets a square
bracket.}

$\ans{\dfrac{\pi^2}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Decide the right endpoint $x = 4$ of
$\displaystyle\sum_{n=1}^{\infty} \frac{(x-1)^n}{n \cdot 3^n}$: converges or
diverges?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{diverges}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $a_n \to 0$ never proves convergence --- the harmonic
series is the standing counterexample. Test each endpoint separately: the two
almost always behave differently, and a single bracket copied to both is the
most common lost point on this topic.
\end{watchoutbox}
