\sstitleblock{Algebraic Limits: Factoring and Rationalizing}

\skillheading{Skill 1 --- Limits by factoring (limits-by-factoring)}
\begin{formulabox}\textbf{Rule:}\\[4pt]
If direct substitution gives $\tfrac{0}{0}$, the numerator and denominator share
a factor of $(x-a)$. Factor, cancel it, then substitute.\\[2pt]
$a^2-b^2=(a-b)(a+b)$ \qquad $a^3-b^3=(a-b)(a^2+ab+b^2)$
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\displaystyle\lim_{x\to4}\frac{x^2-16}{x-4}$
\step{Substitution gives $\tfrac{0}{0}$, so factor rather than conclude.}
\step{$\dfrac{(x-4)(x+4)}{x-4}=x+4$ for $x\neq4$; at $x=4$ this is $\ans{8}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\displaystyle\lim_{x\to5}\frac{x^2-2x-15}{x-5}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}\end{tryitbox}

\skillheading{Skill 2 --- Limits by rationalizing (limits-by-rationalizing)}
\begin{formulabox}\textbf{Rule:}\\[4pt]
When a square root causes the $\tfrac{0}{0}$, multiply top and bottom by the
\textbf{conjugate}. $(\sqrt{u}-v)(\sqrt{u}+v)=u-v^2$ clears the radical from the
numerator and produces the cancelling factor.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\displaystyle\lim_{x\to0}\frac{\sqrt{x+1}-1}{x}$
\step{Multiply by $\dfrac{\sqrt{x+1}+1}{\sqrt{x+1}+1}$ so the numerator becomes $(x+1)-1=x$.}
\step{Cancel $x$ to get $\dfrac{1}{\sqrt{x+1}+1}$; at $x=0$ this is $\ans{1/2}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\displaystyle\lim_{x\to0}\frac{\sqrt{x+9}-3}{x}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1/6}$}\end{tryitbox}

\skillheading{Skill 3 --- Removable discontinuities (removable-discontinuity)}
\begin{formulabox}\textbf{Rule:}\\[4pt]
A piecewise $f$ is continuous at $x=a$ exactly when
$\lim_{x\to a}f(x)=f(a)$. To patch a hole, set the assigned value equal to the
limit you computed.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $f(x)=\dfrac{x^2-100}{x-10}$ for $x\neq10$,
$f(10)=c$. Find $c$.
\step{Cancel: $\dfrac{(x-10)(x+10)}{x-10}=x+10$, so the limit at 10 is 20.}
\step{Continuity forces $f(10)$ to equal that limit, so $c=\ans{20}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $f(x)=\dfrac{x^2-25}{x-5}$ for $x\neq5$, $f(5)=c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Cancelling changes the function at the
removed point, not the limit. Never write $\tfrac{0}{0}=1$, and never conclude a
limit fails to exist just because substitution is undefined.
\end{watchoutbox}
