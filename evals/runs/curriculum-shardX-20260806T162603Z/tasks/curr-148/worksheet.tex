\wstitleblock{Generating and Comparing Two Patterns}{Grades 4--5 Mathematics}{}

\begin{center}
{\small Generating terms, relating the rules, comparing terms, and turning patterns into ordered pairs}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Problems 1 to 5 and problem 10 use the two rules in the table below. Both patterns start at 0 and neither ever stops.
  \item State the terms you generated before answering. Include the unit whenever the problem gives one.
\end{itemize}

\vspace{0.15cm}
\begin{center}
\begin{tabular}{|l|c|c|c|c|c|c|}
\hline
\textbf{Term number} & 1 & 2 & 3 & 4 & 5 & 6 \\ \hline
\textbf{Pattern A} --- start at 0, add 4 & 0 & 4 & 8 & \rule{0.9cm}{0pt} & \rule{0.9cm}{0pt} & \rule{0.9cm}{0pt} \\[2pt] \hline
\textbf{Pattern B} --- start at 0, add 8 & 0 & 8 & 16 & \rule{0.9cm}{0pt} & \rule{0.9cm}{0pt} & \rule{0.9cm}{0pt} \\[2pt] \hline
\end{tabular}\\[0.2cm]
{\small \textbf{Two Rules Table} --- fill in the blanks as you work; problems 1 to 5 and 10 use it.}
\end{center}

\vspace{0.2cm}

\problem[3cm]{Write the 6th term of pattern A.}

\problem[3cm]{Write the 6th term of pattern B.}

\problem[3cm]{Compare the two rules. Each time pattern A adds its step, pattern B adds a step how many times as large?}

\problem[3.4cm]{How much greater is the 10th term of pattern B than the 10th term of pattern A?}

\problem[3.2cm]{Write the ordered pair (pattern A's 4th term, pattern B's 4th term).
\par\vspace{0.3cm}
$($~\ansblank~$,$~\ansblank~$)$}

\problem[3.6cm]{Maya has 12 dollars saved and adds 3 dollars every week. Liam has 4 dollars saved and adds 5 dollars every week. Neither of them spends anything. After how many weeks will they have the same amount?
\answerline{weeks}}

\problem[3.4cm]{How many dollars will Maya and Liam each have at the moment their savings are equal?}

\problem[3.4cm]{The table below gives five terms of pattern P and five terms of pattern Q.
\par\vspace{0.25cm}
\begin{center}
\begin{tabular}{|l|c|c|c|c|c|}
\hline
\textbf{Term number} & 1 & 2 & 3 & 4 & 5 \\ \hline
\textbf{Pattern P} & 7 & 10 & 13 & 16 & 19 \\ \hline
\textbf{Pattern Q} & 2 & 6 & 10 & 14 & 18 \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.25cm}
Write the ordered pair (pattern Q's 5th term, pattern P's 5th term). \quad $($~\ansblank~$,$~\ansblank~$)$}

\problem[3.6cm]{Use the table of patterns P and Q from problem 8.
\begin{itemize}
  \item[(a)] Write the mean of pattern P's five terms. \ansline
  \item[(b)] Write the mean of pattern Q's five terms. \ansline
\end{itemize}}

\problem[3.6cm]{Return to patterns A and B from the Two Rules Table.
\begin{itemize}
  \item[(a)] Write the 12th term of pattern A. \ansline
  \item[(b)] Write the 12th term of pattern B. \ansline
  \item[(c)] Describe the relationship between matching terms of the two patterns, and explain why that relationship must hold for every term, not just the ones you checked.
\end{itemize}
