\sstitleblock{Triangle Congruence: SSS, SAS, ASA, AAS and HL}

\skillheading{Skill 1 --- Name the criterion from the evidence}

\begin{formulabox}
\textbf{The five criteria} (three pairs each; position decides the name):
\begin{itemize}[leftmargin=1cm, itemsep=0pt, topsep=2pt]
  \item \textbf{SSS} --- three sides. \quad \textbf{SAS} --- angle \emph{between} the two sides.
  \item \textbf{ASA} --- side \emph{between} the two angles. \quad \textbf{AAS} --- side \emph{not} between them.
  \item \textbf{HL} --- right triangles only: hypotenuses and one leg.
\end{itemize}
\textbf{Not criteria:} SSA (see Skill 3) and AAA (that is similarity, not congruence).
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\triangle ABC \cong \triangle DEF$ by SSS, with $AB = 4x + 1$ and $DE = 6x - 9$
\step{Naming the criterion tells me which parts are congruent, so $\overline{AB}$ and $\overline{DE}$ are corresponding sides and their lengths are equal.}
\step{$4x + 1 = 6x - 9$ gives $10 = 2x$.}

$\Rightarrow\quad \ans{x = 5}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\triangle PQR \cong \triangle STU$, $PQ = 7x - 2$, $ST = 3x + 22$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Find the third pair the figure hides}

\begin{formulabox}
Two triangles in one figure usually share a part, and the shared part is the pair
nobody marks.
\begin{itemize}[leftmargin=1cm, itemsep=0pt, topsep=2pt]
  \item \textbf{Shared side or angle} --- reflexive property of congruence.
  \item \textbf{Crossing segments} --- the vertical angles at the crossing are congruent.
  \item \textbf{Parallel sides with a transversal} --- alternate interior angles are congruent.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Crossing segments give $m\angle 1 = 2x + 15$ and its vertical angle $4x - 5$
\step{The two angles are vertical angles, so the theorem already makes them congruent --- I set the measures equal rather than hunting for another given.}
\step{$2x + 15 = 4x - 5$ gives $20 = 2x$.}

$\Rightarrow\quad \ans{x = 10}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Vertical angles measure $5x + 4$ and $3x + 20$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- HL, and why SSA fails without it}

\begin{formulabox}
\textbf{SSA is not a criterion:} a side of fixed length swung from a fixed angle can
reach the far ray in two places, giving two different triangles from the same data.\par
\textbf{HL escapes this} because a right angle plus the hypotenuse forces the third
side: $\text{leg} = \sqrt{\text{hyp}^2 - \text{leg}^2}$. Both triangles then have all
three sides congruent, so HL is really SSS in disguise.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Right angle at $C$, hypotenuse $10$, leg $6$ --- find the other leg
\step{HL claims two parts determine the whole right triangle, so I check that by computing the third side from just those two.}
\step{$\sqrt{10^2 - 6^2} = \sqrt{100 - 36} = \sqrt{64}$, and no other value is possible.}

$\Rightarrow\quad \ans{8}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Right angle, hypotenuse $25$, leg $7$ --- find the other leg\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

\skillheading{Skill 4 --- Writing the two-column proof}

\begin{formulabox}
\textbf{Shape of the proof:} list the givens, add the hidden pair (shared part,
vertical angles, or alternate interior angles), then close with the criterion and the
three statements it uses. Every statement needs a reason; ``it looks that way'' is not
one. CPCTC comes \emph{after} the congruence, never before it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $ABCD$ has $\overline{AB} \cong \overline{CD}$ and $\overline{BC} \cong \overline{DA}$. Prove $\triangle ABC \cong \triangle CDA$.
\step{I have two pairs of sides and the two triangles share the diagonal, so a third side pair is free and SSS is the criterion to aim at.}
\step{1.~$\overline{AB} \cong \overline{CD}$ (given). \; 2.~$\overline{BC} \cong \overline{DA}$ (given). \; 3.~$\overline{AC} \cong \overline{CA}$ (reflexive). \; 4.~$\triangle ABC \cong \triangle CDA$ by SSS.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad In the same figure, prove $\triangle ABD \cong \triangle CDB$ using the diagonal $\overline{BD}$.\\[2pt]
\rotatebox{180}{\footnotesize check: same three steps with $\overline{BD} \cong \overline{DB}$ reflexive, then SSS}
\end{tryitbox}
