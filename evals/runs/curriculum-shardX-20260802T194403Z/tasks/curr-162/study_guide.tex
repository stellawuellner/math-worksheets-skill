\sstitleblock{Percent Applications --- Rate, Whole, and Percent Change}

\begin{formulabox}
\textbf{One relationship runs through all three skills:}
\[ \frac{\text{part}}{\text{whole}} \;=\; \frac{\text{percent}}{100} \]
Two of the three are always given. \textbf{Name the missing one first} --- that single
decision is what tells you which skill below you are using. The part is the piece being
described; the whole is what the percent is ``of''.
\end{formulabox}

\skillheading{Skill 1 --- Percent rate from a part and a whole (percent-rate-from-parts)}

\begin{formulabox}
\textbf{Missing: the percent.} Put the unknown on top of the 100:
$\dfrac{\text{part}}{\text{whole}} = \dfrac{p}{100}$, so $p = \dfrac{\text{part}}{\text{whole}} \times 100$.\par
Shortcut: if the whole divides 100 evenly (25, 20, 50, 4, 5), scale the fraction to a
denominator of 100 and read the percent straight off.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 27 of 45 tickets were sold. What percent were sold?
\step{The part (27) and the whole (45) are both given and the percent is missing, so the unknown goes over 100.}
\step{$\dfrac{27}{45} = \dfrac{p}{100} \Rightarrow 45p = 2700 \Rightarrow p = 60$. \ (Or simplify first: $\tfrac{27}{45} = \tfrac{3}{5} = \tfrac{60}{100}$.)}
$\Rightarrow\quad \ans{60\%}$ of the tickets were sold
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 51 of 60 seats are taken. What percent are taken?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{85\%}$}
\end{tryitbox}

\skillheading{Skill 2 --- The whole from a part and a percent (whole-from-part-and-rate)}

\begin{formulabox}
\textbf{Missing: the whole.} The unknown now sits in the \emph{denominator}:
$\dfrac{\text{part}}{n} = \dfrac{\text{percent}}{100}$, so $n = \dfrac{\text{part} \times 100}{\text{percent}}$.\par
\textbf{Size check:} when the percent is below 100, the whole must come out
\emph{bigger} than the part. If it does not, the two numbers were swapped.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 24 students is 30\% of the grade. How many students are in the grade?
\step{The percent (30) and the part (24) are given and the whole is missing, so the unknown belongs under the part.}
\step{$\dfrac{24}{n} = \dfrac{30}{100} \Rightarrow 30n = 2400 \Rightarrow n = 80$. Check: $0.30 \times 80 = 24$ $\checkmark$}
$\Rightarrow\quad$ the grade has $\ans{80}$ students
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 45 pages is 60\% of a book. How many pages does the book have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\skillheading{Skill 3 --- Percent increase or decrease (rate-from-change)}

\begin{formulabox}
\textbf{Missing: the rate of a change.} The part is the \emph{amount of change}; the whole
is always the \textbf{original} amount:
\[ \text{percent change} = \frac{|\text{new} - \text{original}|}{\text{original}} \times 100 \]
Going up is an increase, going down a decrease --- say which in your answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A club grew from 80 members to 92. What percent increase is that?
\step{This compares a change to a starting amount, so the change is the part and the original 80 is the whole --- not the new 92.}
\step{Change $= 92 - 80 = 12$, so $\dfrac{12}{80} = \dfrac{15}{100}$.}
$\Rightarrow\quad$ an increase of $\ans{15\%}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A price fell from 150 dollars to 120 dollars. What percent decrease?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20\%}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} (1) Dividing whole by part instead of part by whole gives a
nonsense percent --- 80 over 5.40 is over 1400\%, so let the size of the answer warn you.
(2) In a percent change, dividing by the \emph{new} amount answers a different question.
(3) After a 20\% increase, the new total is 120\% of the old one, so undo it by dividing by
1.20 --- never by taking 20\% off the new total.
\end{watchoutbox}
