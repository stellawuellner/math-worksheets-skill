\wstitleblock{Rewriting Expressions with Distributive Structure}{Algebra 1 \quad Expressions and Polynomial Operations}{}

\begin{center}
\textit{Distributing over a sum, expanding two binomials, factoring out the
greatest common factor, and checking that two forms are equivalent}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Every answer is an \emph{equivalent expression}, not a value of $x$ ---
unless the problem asks you to substitute a number.
\item Show the distribution step. Write the products before you combine.
\item ``Expanded form'' means no parentheses left. ``Factored form'' means the
greatest common factor is written outside one set of parentheses.
\item Where a problem states units, put the unit on your final answer.
\end{itemize}

\vspace{0.2cm}

\problem{A mural is painted on a wall that is $6$ ft tall. The mural's width is
$(2x + 5)$ ft, where $x$ is the number of feet the artist extended it past the
doorway. Write the area of the mural as an expression with no parentheses.
\par\vspace*{5cm}\answerline{ft$^2$}}

\problem[5cm]{A climbing gym bills each of its $9$ members the same amount this
month: $(4x + 15)$ dollars, where $x$ is the number of extra classes that
member took. Write the gym's total billing, in dollars, with no parentheses.}

\problem{A carpenter cuts $6$ boards to the same length, $(2x - 9)$ inches
each. Write the total length of board used, in inches, with no parentheses.
\par\vspace*{5cm}\answerline{in}}

\problem[5cm]{A rectangular tile panel has area $(18x + 24)$ square
centimetres and is $6$ centimetres tall. Write the area in factored form
$6(\;\underline{\hspace{1.4cm}}\;)$ so that the expression inside the parentheses is
the panel's width in centimetres.}

\problem{A square patio measures $x$ ft on each side. The owner extends it
$3$ ft in one direction and $8$ ft in the other, so the new patio is
$(x + 3)$ ft by $(x + 8)$ ft. Write the new area with no parentheses.
\par\vspace*{6cm}\answerline{ft$^2$}}

\problem[6cm]{Maya claims that $4(3x + 2) + 2x$ can be rewritten as $14x + 8$.
Rewrite the left-hand expression in the form $ax + b$ and state whether Maya is
correct.}

\problem[6cm]{A rectangle has area $(15x^2 + 10x)$ square units and width $5x$
units. Factor the area completely so that $5x$ appears outside the parentheses;
the expression left inside is the rectangle's length.}

\problem[6cm]{Devin wrote $5(2x - 3) = 10x - 3$. Substituting $x = 4$ into
Devin's expression gives $37$. Substitute $x = 4$ into the \emph{original}
expression $5(2x - 3)$ and report that value. Then state, in one sentence, what
Devin did wrong.}

\problem{A rug measures $(2x + 3)$ ft by $(x + 6)$ ft. Write its area with no
parentheses, in simplest form.
\par\vspace*{6cm}\answerline{ft$^2$}}

\problem[7cm]{A stage crew orders $6$ kits that each cost $(2x + 5)$ dollars
and $3$ kits that each cost $(4x - 1)$ dollars.

\smallskip
(a) Write the total cost $6(2x + 5) + 3(4x - 1)$ as a single expression of the
form $ax + b$.

\smallskip
(b) Rewrite that answer in factored form, with the greatest common factor
outside the parentheses.}
