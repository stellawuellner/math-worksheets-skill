\sstitleblock{Fraction Models: Are the Parts Equal?}

\skillheading{1. Check the parts first (equal-parts-check)}

\begin{formulabox}
\textbf{The rule:} a fraction names \emph{equal} parts of one whole.\\[3pt]
Before you write any fraction, ask: \textbf{are all the parts the same size?}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Yes} $\rightarrow$ the model names a fraction. Bottom number
        $=$ how many equal parts fill the whole; top number $=$ how many are
        shaded.
  \item \textbf{No} $\rightarrow$ counting pieces gives the \emph{wrong}
        fraction. Look for smaller equal parts inside the model and count
        those instead.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Sam broke this strip into two pieces and says the shaded
piece is one half.
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \fill[gray!35] (0,0) rectangle (2,1.1);
  \draw[gray] (4,0) -- (4,1.1);
  \draw[gray] (6,0) -- (6,1.1);
  \draw[thick] (0,0) rectangle (8,1.1);
  \draw[line width=1.4pt] (2,0) -- (2,1.1);
\end{tikzpicture}
\end{center}
\step{The two pieces are not the same size, so ``one out of two pieces'' is not
a half --- switch to the equal small parts the thin lines show.}
\step{The whole strip is $4$ equal small parts and the shaded piece covers
$1$ of them.}
$\Rightarrow\quad$ the shaded piece is $\ans{\tfrac{1}{4}}$, not one half.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A strip is made of $6$ equal small parts. It is broken into
two pieces, and the shaded piece covers $2$ of the small parts. What fraction
of the strip is shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{2}{6}=\tfrac{1}{3}}$}
\end{tryitbox}

\skillheading{2. Name the fraction (name-fraction-from-equal-units)}

\begin{formulabox}
\textbf{How to write it:}
\[
\frac{\text{how many equal parts are shaded}}{\text{how many equal parts fill the whole}}
\]
Count \emph{every} equal part in the whole for the bottom number --- including
the ones that are not shaded.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} What fraction of this bar is shaded?
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \fill[gray!35] (0,0) rectangle (2.4,1.1);
  \foreach \i in {1,2,3,4,5,6,7,8,9} {\draw[gray] (0.8*\i,0) -- (0.8*\i,1.1);}
  \draw[thick] (0,0) rectangle (8,1.1);
\end{tikzpicture}
\end{center}
\step{All the parts are the same width, so this model does name a fraction ---
now just count.}
\step{There are $10$ equal parts in the whole bar, and $3$ of them are shaded.}
$\Rightarrow\quad \ans{\tfrac{3}{10}}$ of the bar is shaded.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A bar is cut into $6$ equal parts and $5$ of them are shaded.
What fraction of the bar is shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{6}}$}
\end{tryitbox}

\skillheading{3. Fix a model that has unequal parts (fix-the-model)}

\begin{formulabox}
\textbf{To show halves, thirds, fourths $\ldots$} cut the whole into that many
parts \emph{of the same size}.\\[3pt]
If the whole is already made of equal small squares:
\[
\text{squares in each part} \;=\; \text{squares in the whole} \div \text{number of parts}
\]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A bar of $12$ equal squares must be cut into fourths. Where
do the cuts go?
\step{Fourths means $4$ parts that are all the same size, so share the squares
out evenly instead of guessing where to cut.}
\step{$12 \div 4 = 3$, so each fourth is $3$ squares wide (cut after the
$3$rd, $6$th and $9$th square).}
$\Rightarrow\quad$ each fourth holds $\ans{3}$ squares.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A bar of $10$ equal squares is cut into $2$ equal halves. How
many squares are in each half?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{4. More equal parts means smaller parts (compare-unit-fractions)}

\begin{formulabox}
\textbf{Same whole, more parts, smaller pieces.} If two identical wholes are
cut into equal parts, the one cut into \emph{more} parts has the
\emph{smaller} pieces --- there is only so much to share out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Two identical ribbons. One is cut into thirds, the other into
fifths. Which single piece is longer?
\step{Both wholes are the same length, so compare the number of equal parts
rather than the numbers on the bottom of the fractions.}
\step{Thirds share the ribbon among fewer parts, so each third is longer than
each fifth.}
$\Rightarrow\quad \ans{\tfrac{1}{3} > \tfrac{1}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Two identical ribbons, one cut into eighths and one cut into
sixths. Write $>$ or $<$ between the two pieces.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{8} < \tfrac{1}{6}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} counting pieces is not the same as counting equal
parts. A shape cut into three \emph{unequal} pieces has no thirds in it at all.
\end{watchoutbox}
