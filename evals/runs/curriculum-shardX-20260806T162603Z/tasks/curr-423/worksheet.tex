\wstitleblock{Solving Trigonometric Equations on a Restricted Interval}{Precalculus}{}

\begin{center}\small\textit{Solving linear, factorable, identity-first and multiple-angle trigonometric equations on a restricted interval}\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Give \emph{every} solution inside the stated interval, and no solution outside it. List them in increasing order.
  \item Angles are in degrees wherever a degree symbol appears; problem 6 is in hours.
  \item For a multiple-angle equation, widen the interval for the inside angle first, solve there, then divide.
\end{itemize}
\vspace{0.15cm}

\problem{Solve $2\cos\theta + 1 = 0$ for $0^\circ \le \theta < 360^\circ$.
\par\vspace*{4.8cm}\answerline{degrees}}

\problem{Solve $\sqrt{2}\,\sin\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.
\par\vspace*{4.8cm}\answerline{degrees}}

\problem{Solve $\sin\theta\,(2\sin\theta - 1) = 0$ for $0^\circ \le \theta < 360^\circ$. Give all four solutions.
\par\vspace*{5.4cm}\answerline{degrees}}

\problem{A cam on a press returns to its marked position whenever the crank angle $\theta$ satisfies $2\sin 2\theta - \sqrt{3} = 0$, with $\theta$ measured in degrees. Find every crank angle in one full turn, $0^\circ \le \theta < 360^\circ$, at which this happens.
\par\vspace*{6cm}\answerline{degrees}}

\problem{Consider $2\sin^2\theta + 3\cos\theta - 3 = 0$ on $0^\circ \le \theta < 360^\circ$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Use a Pythagorean identity to rewrite the left-hand side as a product of two factors. \ansline
  \vspace*{2.8cm}
  \item[(b)] Solve the equation on the given interval. \answerline{degrees}
  \vspace*{3.4cm}
\end{itemize}}

\problem{A buoy rides a swell so that its height above the mean water line is $h(t) = 6\sin\!\left(\dfrac{\pi t}{6}\right)$, where $h$ is in metres and $t$ is the number of hours after midnight.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find every time $t$ with $0 \le t < 12$ at which the buoy is exactly $3$ metres above the mean line. \answerline{hours}
  \vspace*{3.4cm}
  \item[(b)] Explain what those two times mean for the buoy's motion during those twelve hours. \ansline
  \vspace*{2.6cm}
\end{itemize}}

\problem{Solve $2\cos 3\theta - 1 = 0$ for $0^\circ \le \theta < 180^\circ$.
\par\vspace*{6cm}\answerline{degrees}}

\problem{Consider $2\cos^2\theta - \sin\theta - 1 = 0$ on $0^\circ \le \theta < 360^\circ$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Use a Pythagorean identity to rewrite the left-hand side as a product of two factors. \ansline
  \vspace*{2.8cm}
  \item[(b)] Solve the equation on the given interval. \answerline{degrees}
  \vspace*{3.4cm}
\end{itemize}}

\problem{A searchlight sweeps a wall. The brightness at a fixed point on the wall is $B(\theta) = 40\sin 2\theta$ lux, where $\theta$ is the beam angle in degrees. Find every beam angle with $0^\circ \le \theta < 180^\circ$ at which the brightness is exactly $20$ lux.
\par\vspace*{6cm}\answerline{degrees}}

\problem{Consider $(2\sin\theta - \sqrt{3})(2\sin\theta + \sqrt{3}) = 0$ on $0^\circ \le \theta < 360^\circ$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Find all solutions on the interval. \answerline{degrees}
  \vspace*{4cm}
  \item[(b)] The equation $2\sin\theta - \sqrt{3} = 0$ has only two solutions on the same interval. Explain why the equation in part (a) has four. \ansline
  \vspace*{3cm}
\end{itemize}}
