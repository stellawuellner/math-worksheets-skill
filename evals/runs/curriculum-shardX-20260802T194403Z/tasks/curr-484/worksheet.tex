\wstitleblock{Diagnosing Separation and Initial-Condition Errors}{AP Calculus AB/BC $\bullet$ Error Analysis}{}

\begin{center}
Separating the variables before integrating, carrying the constant of
integration correctly, and applying the initial condition at the right step
\end{center}

\noindent\small\textbf{Directions.} Separable equations go wrong in three
predictable places. Watch for all three below.
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt, topsep=2pt]
  \item integrating before the variables are actually separated;
  \item losing the constant of integration, or adding it at the wrong moment;
  \item applying the initial condition too early or too late.
\end{itemize}
Where a student's work is shown, \textbf{(i)} name the first wrong line,
\textbf{(ii)} say what rule it breaks, \textbf{(iii)} solve correctly. Round
decimal answers to two places, and check each solution by substituting it back
into the differential equation.\normalsize

\problem[6cm]{Find the general solution of
$\dfrac{dy}{dx} = 8x^{3} - 6x$. Include the constant of integration, and say in
one phrase what that constant represents on a slope field.}

\problem[7cm]{The same equation, $\dfrac{dy}{dx} = 8x^{3} - 6x$, now carries the
initial condition $y(1) = 4$. Find the particular solution, then evaluate
$y(2)$. State the step at which the initial condition may first be used.}

\problem[8cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = 2xy$ with
$y(0) = 3$, then find $y(1)$. Wren's work:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
$\displaystyle\int dy = \int 2xy\,dx \ \Rightarrow \ y = x^{2} + C$;
\quad $y(0) = 3 \Rightarrow C = 3$; \quad $y(1) = 4$ \\[0.35cm]\hline
\end{tabular}
\end{center}
\noindent Name the first wrong line, say what Wren did to the $y$ on the right
side, and give the correct $y(1)$ to two decimal places.}

\problem[8cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = 5y$ with
$y(0) = 6$, then find $y(1)$. Dev's work:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
$\ln|y| = 5x + C \ \Rightarrow \ y = e^{5x} + C$;
\quad $y(0) = 6 \Rightarrow C = 5$; \quad $y(1) \approx 153.41$ \\[0.35cm]\hline
\end{tabular}
\end{center}
\noindent Dev's first line is correct. Name the line that is not, explain what
exponentiating does to a \emph{sum}, and find $y(1)$ correctly.}

\problem[8cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = \dfrac{x}{y}$ with
$y(0) = 4$, then find $y(3)$. Rosa separates correctly and reaches
$\dfrac{y^{2}}{2} = \dfrac{x^{2}}{2} + C$, then writes:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
``take the square root of both sides'': $y = x + C$;
\quad $y(0) = 4 \Rightarrow C = 4$; \quad $y(3) = 7$ \\[0.35cm]\hline
\end{tabular}
\end{center}
\noindent Say what is wrong with that square root, then solve correctly for
$y(3)$.}

\problem[8cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = y^{2}$ with
$y(0) = 2$, then find $y(0.25)$. Sam's work:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
$\displaystyle\int y^{-2}dy = \int dx \ \Rightarrow \ \dfrac{1}{y} = x + C$;
\quad $y(0) = 2 \Rightarrow C = 0.5$; \quad $y(0.25) \approx 1.33$ \\[0.35cm]\hline
\end{tabular}
\end{center}
\noindent One sign is missing. Fix it and give $y(0.25)$. Then say what the
corrected solution does as $x$ approaches $0.5$, and why that behaviour is
invisible in Sam's version.}

\problem[8cm]{Solve $\dfrac{dy}{dx} = \dfrac{6x}{x^{2}+1}$ with $y(0) = 2$ and
evaluate $y(2)$ to two decimal places. Then explain why applying the initial
condition \emph{before} integrating would be meaningless, and what answer a
student who forgets the constant of integration entirely would report.}

\problem[8cm]{A classmate argues: ``A constant is a constant --- it makes no
difference whether you add it before or after exponentiating.'' Using
$\dfrac{dy}{dx} = 5y$ from problem 4, refute this. Differentiate
$y = e^{5x} + 5$, compare the result with $5y$, and explain what the constant
becomes when it is carried through the exponential correctly.
