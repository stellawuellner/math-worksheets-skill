\sstitleblock{Naming a Conic and Reading Its Key Features}

\skillheading{Skill 1 --- Name the family before anything else}

\begin{formulabox}
\textbf{General form:} $Ax^2 + Bxy + Cy^2 + Dx + Ey + F = 0$. Move every term to one side
first, then read $A$, $B$, $C$ straight off.\par
\smallskip
\textbf{Indicator:} $B^2 - 4AC$.
\begin{itemize}[label={$\bullet$}, leftmargin=1.0cm, itemsep=0pt, topsep=1pt]
  \item negative and $A = C$: \textbf{circle} \quad$\cdot$\quad negative and $A \neq C$:
        \textbf{ellipse}
  \item zero (only one squared term): \textbf{parabola} \quad$\cdot$\quad positive
        (squares of opposite sign): \textbf{hyperbola}
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Name the conic $16x^2 - y^2 = 16$.
\step{Only the family is wanted, and the family is decided by $B^2 - 4AC$ alone --- so
read off $A$, $B$, $C$ and compute nothing else.}
\step{One side: $16x^2 - y^2 - 16 = 0$, so $A = 16$, $B = 0$, $C = -1$, and
$B^2 - 4AC = 0 - 4(16)(-1)$.}

$\Rightarrow\quad \ans{64}$, positive, so it is a \textbf{hyperbola}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Name the conic $x^2 + 9y^2 = 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-36}$, negative with $A \neq C$, an ellipse}
\end{tryitbox}

\skillheading{Skill 2 --- Circle: centre and radius}

\begin{formulabox}
\textbf{From} $x^2 + y^2 + Dx + Ey + F = 0$ \textbf{to} $(x-h)^2 + (y-k)^2 = r^2$:
complete the square on the $x$ terms and on the $y$ terms.\par
\smallskip
Shortcut: $h = -D/2$, \quad $k = -E/2$, \quad $r = \sqrt{h^2 + k^2 - F}$.\par
\smallskip
The bracket signs are \emph{minus} $h$ and \emph{minus} $k$, so $(y+3)^2$ means $k = -3$.
And the right-hand side is $r^2$ --- take the square root before writing the radius.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Centre and radius of $x^2 + y^2 - 4x + 6y - 3 = 0$.
\step{Both squares are present with equal coefficients, so it is a circle and the shortcut
formulas apply directly --- no need to complete the square by hand.}
\step{$h = -(-4)/2$, \quad $k = -6/2$, \quad $r = \sqrt{h^2 + k^2 + 3} = \sqrt{4 + 9 + 3}
= \sqrt{16}$.}

$\Rightarrow\quad$ centre $(\ans{2}, \ans{-3})$, \quad radius $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Centre and radius of $x^2 + y^2 + 2x - 8y - 19 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: centre $(\ans{-1}, \ans{4})$, radius $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Parabola: axis and vertex}

\begin{formulabox}
\textbf{From} $y = ax^2 + bx + c$: the axis of symmetry is the vertical line
$x = -b/(2a)$; the vertex is that $x$ together with the $y$ you get by substituting it
back. Opens \textbf{up} when $a > 0$, \textbf{down} when $a < 0$.\par
\smallskip
The $x$-intercepts (if any) come from solving $ax^2 + bx + c = 0$, and the axis always
sits exactly halfway between them --- a free check on both answers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Axis and vertex of $y = x^2 - 8x + 12$.
\step{The vertex is the one point on the axis of symmetry, so find the axis first and the
vertex falls out of it --- no completing the square needed.}
\step{$a = 1$, $b = -8$, so $x = -(-8)/2 = 4$; then $y = 4^2 - 8(4) + 12 = 16 - 32 + 12$.}

$\Rightarrow\quad$ axis $x = \ans{4}$, \quad vertex $(4, \ans{-4})$, opening up
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Axis and vertex of $y = x^2 + 6x + 11$.\\[2pt]
\rotatebox{180}{\footnotesize check: axis $x = \ans{-3}$, vertex $(-3, \ans{2})$}
\end{tryitbox}

\skillheading{Skill 4 --- Ellipse: vertices and foci}

\begin{formulabox}
\textbf{Standard form:} $x^2/a^2 + y^2/b^2 = 1$ with the centre at the origin. The
denominators are squares of the distances, so take square roots before plotting.\par
\smallskip
Vertices $(\pm a, 0)$, co-vertices $(0, \pm b)$, foci $(\pm c, 0)$ with
$c^2 = a^2 - b^2$ --- when the larger denominator sits under $x^2$, all of this runs
along the $x$-axis. A hyperbola $x^2/a^2 - y^2/b^2 = 1$ uses the same $a$ for its
vertices, but $c^2 = a^2 + b^2$ instead.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Vertices and foci of $x^2/25 + y^2/16 = 1$.
\step{Every key feature here is a distance, and the equation stores distances as squares
--- so the whole job is three square roots.}
\step{$a = \sqrt{25}$, \quad $b = \sqrt{16}$, \quad $c = \sqrt{25 - 16} = \sqrt{9}$.}

$\Rightarrow\quad a = \ans{5}$, $b = \ans{4}$, $c = \ans{3}$: vertices $(\pm 5, 0)$,
foci $(\pm 3, 0)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $a$, $b$ and $c$ for $x^2/100 + y^2/64 = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $a = \ans{10}$, $b = \ans{8}$, $c = \ans{6}$}
\end{tryitbox}
