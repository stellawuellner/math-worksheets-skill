\wstitleblock{Matching Division Stories, Drawings, and Equations}{Grades 2--3 \quad Operations and Algebraic Thinking}{}

\begin{center}
\textit{Matching division stories to equations, reading equal-group and array
drawings, choosing the story an equation tells, and checking with
multiplication fact families}
\end{center}

\noindent\textbf{The one idea behind every problem.} A division sentence always
says the same three things: \textbf{how many in all}, \textbf{how many groups},
and \textbf{how many in each group}. A story, a drawing, and an equation are
three ways of telling that one idea. Your job is to move between them.

\begin{center}
\fbox{\parbox{0.92\linewidth}{\centering
total in all \; $\div$ \; number of groups \; $=$ \; how many in each group
\\[2pt]
total in all \; $\div$ \; how many in each group \; $=$ \; number of groups}}

\small\textit{Reference card. No numbers are filled in --- use the numbers given
in each problem.}\normalsize
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Every problem asks you to show the match, not just the answer: write the
division equation, or draw the groups, or say which story fits.
\item You may draw dots, circles, or an array in the work space --- pictures are
part of the work here.
\item Write your final answer on the answer line.
\end{itemize}

\problem[5cm]{Maya has 12 stickers. She shares them equally with 3 friends at
her table, so that each friend gets the same number.

\smallskip
Write the division equation this story tells, then answer it: how many stickers
does each friend get?}

\problem[5cm]{The picture below shows 20 counters that have already been sorted
into 4 equal groups.

\smallskip
\begin{center}
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet$}\quad
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet$}\quad
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet$}\quad
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet$}
\end{center}
\smallskip
Write the division equation that matches the picture, then answer it: how many
counters are in each group?}

\problem[5cm]{Here is a multiplication sentence with a missing number:
\[
3 \times \square \;=\; 18
\]
\textbf{(a)} What number belongs in the box?

\smallskip
\textbf{(b)} Write the division sentence that belongs to the same fact family
and asks for that same missing number.}

\problem[5.5cm]{A teacher has 24 crayons. She packs them into boxes, and each
box holds 6 crayons.

\smallskip
This story does \emph{not} tell you the number of groups --- it tells you how
many go in each group. Write the division equation the story tells, then answer
it: how many boxes does she fill?}

\problem[5.5cm]{Look at this equation:
\[
15 \div 5 \;=\; \square
\]
Which story below matches the equation? Circle A or B, tell in one sentence how
you know, then answer the equation.

\smallskip
\textbf{A.} There are 15 shells. They are put into 5 equal piles.

\smallskip
\textbf{B.} There are 15 shells in each of 5 buckets.}

\problem[5.5cm]{The picture below shows 18 counters sorted into 3 equal groups.

\smallskip
\begin{center}
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$}\quad
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$}\quad
\fbox{$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$}
\end{center}
\smallskip
Write the division equation that matches the picture, then answer it. Then write
the multiplication equation that matches the same picture.}

\problem[5.5cm]{A fact family uses the same three numbers over and over.

\smallskip
\textbf{(a)} Fill in the missing number: \quad $7 \times \square = 28$

\smallskip
\textbf{(b)} Write all four sentences of this fact family --- two multiplication
sentences and two division sentences.

\smallskip
Put the missing number from part (a) on the answer line.}

\problem[5.5cm]{Read carefully --- one of these numbers is the total and one is
the size of each group.

\smallskip
A farm stand has 32 apples. The apples are put into baskets with 8 apples in
each basket.

\smallskip
Write the division equation this story tells, then answer it: how many baskets
are used?}

\problem[6cm]{Look at this equation:
\[
42 \div 6 \;=\; \square
\]
\textbf{(a)} Only one of these stories matches the equation. Circle it.

\smallskip
\textbf{A.} A baker makes 42 muffins and puts 6 muffins in each tray.

\smallskip
\textbf{B.} A baker makes 6 trays of muffins with 42 muffins on each tray.

\smallskip
\textbf{(b)} Explain in one sentence why the other story does not match.

\smallskip
\textbf{(c)} Answer the equation and write it on the answer line.}

\problem[6cm]{The array below shows 36 counters arranged in 4 equal rows.

\smallskip
\begin{center}
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[2pt]
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[2pt]
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[2pt]
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$
\end{center}
\smallskip
Write the division equation that matches the array, then answer it: how many
counters are in each row? Be careful --- the question asks about a row, not
about the number of rows.}

\problem[6cm]{This time the missing number is the one you divide \emph{by}:
\[
45 \div \square \;=\; 9
\]
\textbf{(a)} What number belongs in the box?

\smallskip
\textbf{(b)} Write a short story that this equation could tell, using your
answer from part (a).

\smallskip
Put the missing number from part (a) on the answer line.}

\problem[7cm]{\textbf{Challenge.} A classroom has 5 packs of pencils, and every
pack holds 6 pencils. All of the pencils are shared equally among 3 tables.

\smallskip
\textbf{(a)} What do you have to find \emph{before} you can divide? Write that
step as an equation.

\smallskip
\textbf{(b)} Now write the division equation and answer it: how many pencils
does each table get?

\smallskip
\textbf{(c)} Draw a picture that shows your division. Label what the groups are.}
