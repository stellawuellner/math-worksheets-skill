\sstitleblock{Solving for Angle Measures on Parallel Lines}

\skillheading{Skill 1 --- Congruent pairs: set the expressions equal}

\begin{formulabox}
On \textbf{parallel} lines cut by a transversal, three named pairs are \emph{congruent}:\par
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=1pt]
  \item \textbf{corresponding} --- matching corners at the two intersections
  \item \textbf{alternate interior} --- both between the lines, opposite sides of $t$
  \item \textbf{alternate exterior} --- both outside the lines, opposite sides of $t$
\end{itemize}
Each of these gives the equation ``first expression $=$ second expression''. Substitute your answer back into both to check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$ and corresponding angles measure $(3x + 7)^\circ$ and $(5x - 23)^\circ$. Find $x$.
\step{The pair is corresponding on parallel lines, so the angles are congruent --- which turns the picture into an equality of the two expressions.}
\step{$3x + 7 = 5x - 23$, so $30 = 2x$.}

$\Rightarrow\quad \ans{x = 15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Alternate interior angles measure $(2x + 18)^\circ$ and $(4x - 6)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 12}$}
\end{tryitbox}

\skillheading{Skill 2 --- Supplementary pairs: add to 180}

\begin{formulabox}
Two named pairs are \emph{supplementary}:\par
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=1pt]
  \item \textbf{same-side interior} --- both between the lines, same side of $t$ (needs parallel lines)
  \item \textbf{linear pair} --- adjacent at one vertex, outer sides forming a line (needs nothing)
\end{itemize}
Each gives ``first expression $+$ second expression $= 180$''. Adding when the pair is congruent, or equating when it is supplementary, is the error this topic is built to catch.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$ and same-side interior angles measure $(5x + 12)^\circ$ and $(3x + 8)^\circ$. Find $x$.
\step{Both angles lie between the parallel lines on the same side of the transversal, so the pair is supplementary --- the equation is a sum equal to $180$, not an equality.}
\step{$(5x + 12) + (3x + 8) = 180$, so $8x + 20 = 180$ and $8x = 160$.}

$\Rightarrow\quad \ans{x = 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same-side interior angles measure $(4x + 9)^\circ$ and $(2x + 15)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 26}$}
\end{tryitbox}

\skillheading{Skill 3 --- Two-step chains}

\begin{formulabox}
When two angles are not a named pair, travel through a third. \textbf{Corresponding} or \textbf{vertical} moves you to the other intersection; a \textbf{linear pair} moves you around a vertex.\par
Track what each link does: congruent then congruent leaves the ends congruent; congruent then supplementary leaves the ends \emph{supplementary}. Count the supplementary links --- an odd number means subtract from $180$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$ and one angle measures $124^\circ$. Find the angle reached by a corresponding step followed by a linear pair.
\step{The corresponding step copies the measure to the other intersection, and the linear pair then subtracts it from $180$ --- one supplementary link, so the answer is the supplement.}
\step{$180 - 124$.}

$\Rightarrow\quad \ans{56}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The same chain starting from an angle of $71^\circ$. Find the measure at the end.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{109}$}
\end{tryitbox}
