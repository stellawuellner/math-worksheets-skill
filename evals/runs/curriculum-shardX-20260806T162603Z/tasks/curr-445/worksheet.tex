\wstitleblock{Connecting Numerical, Graphical and Algebraic Ideas of a Limit}{Precalculus}{}

\begin{center}
\small Limits from tables of values, from graphs, from algebra, and as the sum of a series
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Give exact values (fractions, not decimals) for every limit and every sum.
  \item Where a problem asks for a table value, round to four decimal places.
  \item The four approaches are mixed on purpose after problems 1--2: decide which view the problem is asking for before you start.
\end{itemize}

\vspace{0.15cm}

\problem[4.5cm]{Evaluate $\displaystyle\lim_{x \to 3} \frac{x^2 - 9}{x - 3}$ algebraically.}

\problem[3.5cm]{Let $f(x) = \dfrac{x - 4}{\sqrt{x} - 2}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $f(3.9)$ to four decimal places. \quad \ansblank
  \item[(b)] Compute $f(4.1)$ to four decimal places. \quad \ansblank
  \item[(c)] Give the exact value of $\displaystyle\lim_{x \to 4} f(x)$. \quad \ansblank
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{Let $g(x) = \begin{cases} x + 1, & x < 2 \\ 6 - x, & x \ge 2 \end{cases}$
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find $\displaystyle\lim_{x \to 2^-} g(x)$. \quad \ansblank
  \item[(b)] Find $\displaystyle\lim_{x \to 2^+} g(x)$. \quad \ansblank
  \item[(c)] Sketch $g$ on the grid, marking open and closed endpoints, and say whether $\displaystyle\lim_{x \to 2} g(x)$ exists.
\end{itemize}

\problem[4.5cm]{Evaluate $\displaystyle\lim_{x \to -1} \frac{x^2 + 3x + 2}{x + 1}$.}

\problem[3.5cm]{Consider the geometric series $\displaystyle\sum_{n=1}^{\infty} 3\left(\tfrac{1}{2}\right)^{n-1}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the fourth partial sum $S_4$ as an exact fraction. \quad \ansblank
  \item[(b)] Find the sum of the whole series. \quad \ansblank
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{Let $h(x) = \dfrac{\sin x}{x}$, with $x$ in radians.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $h(0.1)$ to four decimal places. \quad \ansblank
  \item[(b)] Compute $h(0.5)$ to four decimal places. \quad \ansblank
  \item[(c)] Give the exact value of $\displaystyle\lim_{x \to 0} h(x)$. \quad \ansblank
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{Let $p(x) = \dfrac{x - 2}{x^2 - 4}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find $\displaystyle\lim_{x \to 2} p(x)$ as an exact fraction. \quad \ansblank
  \item[(b)] Sketch $p$ near $x = 2$ on the grid, marking what happens at $x = 2$ itself, and say whether the graph has a hole or a vertical asymptote there.
\end{itemize}

\problem[5cm]{Evaluate $\displaystyle\lim_{x \to 0} \frac{\sqrt{x + 9} - 3}{x}$ as an exact
fraction. (Multiplying by the conjugate helps.)}

\problem[3.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\left(\frac{1}{n} - \frac{1}{n+1}\right)$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the sum of the series. \quad \ansblank
  \item[(b)] Write the partial sum $S_N$ in closed form and explain why the sum of the series is a limit rather than an addition.
\end{itemize}
\vspace*{3cm}}

\problem[3.5cm]{Let $q(n) = \left(1 + \dfrac{1}{n}\right)^{n}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $q(100)$ to four decimal places. \quad \ansblank
  \item[(b)] Compute $q(10000)$ to four decimal places. \quad \ansblank
  \item[(c)] Name the number these values are approaching, and explain why a table alone cannot prove that is the limit.
\end{itemize}
\vspace*{3cm}}

\problem[3.5cm]{Consider $\displaystyle\lim_{x \to 2} \frac{x^3 - 8}{x^2 - 4}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the limit. \quad \ansblank
  \item[(b)] Explain why direct substitution does not settle this, and why cancelling the common factor is legitimate inside a limit.
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{\textbf{Synthesis.} Let
$r(x) = \begin{cases} \dfrac{x^2 - 1}{x - 1}, & x \ne 1 \\[4pt] 3, & x = 1 \end{cases}$
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find $\displaystyle\lim_{x \to 1} r(x)$. \quad \ansblank
  \item[(b)] Sketch $r$ on the grid and explain why $r$ is not continuous at $x = 1$ even though the limit there exists.
\end{itemize}
