\sstitleblock{Zero and Negative Exponents}

\skillheading{1. What zero and negative exponents mean}

\begin{formulabox}
\textbf{Definitions.} \ $a^{0} = 1$ for every $a \neq 0$, \quad and \quad
$a^{-n} = \dfrac{1}{a^{n}}$ \ (so $a^{-n}$ is the \emph{reciprocal} of
$a^{n}$).\\[3pt]
\textbf{Why:} $\dfrac{a^{3}}{a^{3}} = a^{3-3} = a^{0}$, and a number divided
by itself is $1$. Keep subtracting and $a^{-1}$ has to mean $\dfrac{1}{a}$ for
the pattern to hold.\\[3pt]
\textbf{A negative exponent never makes a value negative.} It makes it small.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $3^{-2}$.
\step{The exponent is negative, so the definition says take the reciprocal of
the positive power --- nothing here is negative.}
\step{$3^{-2} = \dfrac{1}{3^{2}}$ and $3^{2} = 9$.}

$\Rightarrow\quad \ans{\dfrac{1}{9}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $2^{-3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{8}}$}
\end{tryitbox}

\skillheading{2. Rewriting with positive exponents}

\begin{formulabox}
\textbf{Rule.} A factor with a negative exponent changes \emph{level} and
turns positive: from the numerator it moves down, from the denominator it
moves up. In symbols, $\dfrac{1}{a^{-n}} = a^{n}$.\\[3pt]
\textbf{Only the factor that carries the exponent moves.} In $3x^{-2}$ the
$3$ has no negative exponent, so it stays where it is:
$3x^{-2} = \dfrac{3}{x^{2}}$, not $\dfrac{1}{3x^{2}}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $5x^{-3}$ with positive exponents.
\step{Ask which factor carries the negative exponent: only $x$ does, so only
$x$ moves.}
\step{Send $x^{-3}$ to the denominator, where it becomes $x^{3}$, and leave
the coefficient $5$ on top.}

$\Rightarrow\quad \ans{\dfrac{5}{x^{3}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $\dfrac{2x^{-2}}{y^{-3}}$ with positive
exponents.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{2y^{3}}{x^{2}}}$}
\end{tryitbox}

\skillheading{3. The exponent laws still work}

\begin{formulabox}
\textbf{Laws (same base).} \ $a^{m} \cdot a^{n} = a^{m+n}$, \quad
$\dfrac{a^{m}}{a^{n}} = a^{m-n}$, \quad $\left(a^{m}\right)^{n} = a^{mn}$.\\[3pt]
These hold for negative and zero exponents too --- just do the integer
arithmetic on the exponents, then rewrite any negative result as a
reciprocal at the very end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $x^{3} \cdot x^{-7}$.
\step{Same base, multiplied --- that is the product law, so add the
exponents.}
\step{$3 + (-7) = -4$, giving $x^{-4}$, which then has to be rewritten with a
positive exponent.}

$\Rightarrow\quad \ans{\dfrac{1}{x^{4}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\left(x^{-3}\right)^{2} \cdot x^{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{x^{2}}}$}
\end{tryitbox}

\skillheading{4. Comparing powers}

\begin{formulabox}
\textbf{Rule.} Evaluate first, compare second. Comparing the exponents
instead of the values is the usual trap: $2^{-3}$ is bigger than $3^{-2}$
even though $-3$ is smaller than $-2$.\\[3pt]
For unit fractions, the \textbf{smaller denominator} is the larger number.
And two different-looking powers can be equal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $2^{-4}$ and $4^{-2}$.
\step{The bases differ, so nothing can be read off the exponents --- write
each one as a fraction.}
\step{$2^{-4} = \dfrac{1}{16}$ and $4^{-2} = \dfrac{1}{16}$: the same number.}

$\Rightarrow\quad \ans{=}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $>$, $<$ or $=$: \quad
$3^{0} \underline{\hspace{0.9cm}} 3^{-2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}
