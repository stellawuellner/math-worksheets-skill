\wstitleblock{Graphing Basic Polar Curves from Equations}{Precalculus}{}

\noindent\small
Evaluating r at a given angle, finding the angles where r is zero, locating the
maximum r that fixes a curve's orientation, and identifying and sketching basic
polar curves.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=3pt]
  \item Work in degrees. Where an interval is asked for, use
        $0^\circ \le \theta < 360^\circ$ and list every solution in that window.
  \item Give exact values, not decimals.
  \item A negative $r$ is legal: plot $|r|$ in the direction opposite $\theta$.
  \item Show the equation you solved, not just the angles.
\end{itemize}
\normalsize

\problem[5cm]{For the circle $r = 6\cos\theta$, find $r$ when $\theta = 60^\circ$.}

\problem[5cm]{For the circle $r = 4\sin\theta$, find $r$ when $\theta = 30^\circ$.}

\problem[5cm]{For the cardioid $r = 1 + \cos\theta$, find $r$ when
$\theta = 90^\circ$. This value is the curve's distance from the pole straight up
the $90^\circ$ ray.}

\problem[8cm]{The rose $r = 4\sin(2\theta)$ returns to the pole between petals.
\par Find every $\theta$ in $0^\circ \le \theta < 360^\circ$ for which $r = 0$.
These angles are the lines the petals are tucked between.}

\problem[5cm]{For the limacon $r = 2 + 4\cos\theta$, find $r$ when
$\theta = 180^\circ$. Read your answer carefully: the sign matters.}

\problem[8cm]{For the cardioid $r = 1 + \cos\theta$, find every $\theta$ in
$0^\circ \le \theta < 360^\circ$ with $r = 0$.
\par This single angle is where the cardioid's cusp sits.}

\problem[8cm]{The rose $r = 2\sin(3\theta)$ reaches its greatest distance from the
pole when $r = 2$.
\par Solve $2\sin(3\theta) = 2$ for every $\theta$ in
$0^\circ \le \theta < 360^\circ$. These angles point along the petal tips.}

\problem[5cm]{For the limacon $r = 3 - 2\cos\theta$, find $r$ when
$\theta = 120^\circ$.}

\problem[8cm]{The limacon $r = 2 + 4\cos\theta$ passes through the pole twice, and
those two angles bound its inner loop.
\par Find every $\theta$ in $0^\circ \le \theta < 360^\circ$ for which $r = 0$.}

\problem[8cm]{For the circle $r = 4\cos\theta$, the maximum value of $r$ is 4.
\par Solve $4\cos\theta = 4$ on $0^\circ \le \theta < 360^\circ$. The single angle
you find is the direction in which the circle is offset from the pole.}

\problem[3cm]{\textbf{Sketch.} Graph the cardioid $r = 1 + \cos\theta$ on the polar
grid below. Plot the four points at $\theta = 0^\circ$, $90^\circ$, $180^\circ$, and
$270^\circ$ first, then join them smoothly. Label the cusp.
\par \emph{Each ring is 1 unit; the outermost ring is $r = 5$. The heavy ray is the
polar axis, $\theta = 0^\circ$.}
\begin{center}
\begin{tikzpicture}[scale=0.62]
  \foreach \rad in {1,2,3,4,5} \draw[gray!45] (0,0) circle (\rad);
  \foreach \ang in {0,30,60,90,120,150,180,210,240,270,300,330}
     \draw[gray!45] (0,0) -- (\ang:5);
  \draw[->, thick] (0,0) -- (5.6,0);
  \fill (0,0) circle (1.6pt);
\end{tikzpicture}
\end{center}
\noansline}

\problem[5cm]{\textbf{Identify, sketch, and justify.} Consider
$r = 2 + 4\cos\theta$.
\par Name the family this curve belongs to, sketch it on the grid below, and explain
--- using the constant term and the coefficient of $\cos\theta$ --- why this curve has
an inner loop while $r = 4 + 2\cos\theta$ would not. Use the pole-crossing angles you
found earlier as evidence.
\par \emph{Each ring is 1 unit; the outermost ring is $r = 6$. The heavy ray is the
polar axis, $\theta = 0^\circ$.}
\begin{center}
\begin{tikzpicture}[scale=0.52]
  \foreach \rad in {1,2,3,4,5,6} \draw[gray!45] (0,0) circle (\rad);
  \foreach \ang in {0,30,60,90,120,150,180,210,240,270,300,330}
     \draw[gray!45] (0,0) -- (\ang:6);
  \draw[->, thick] (0,0) -- (6.6,0);
  \fill (0,0) circle (1.6pt);
\end{tikzpicture}
\end{center}
\noansline}
