\sstitleblock{Rounding and Ordering on the Number Line}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Rounding to the nearest ten}

\begin{formulabox}
\textbf{Rule:} Put the number between its two neighbor tens on the line.
The \emph{ones} digit decides: 0--4 stays down, 5--9 goes up.
Halfway (5 ones) always rounds \emph{up}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round 73 to the nearest ten.
\step{Trap 73 between its neighbor tens on the line: it sits between 70 and 80.}
\step{The ones digit is three, before the halfway mark at seventy-five, so 73 is closer to 70.}
$\Rightarrow\quad \ans{70}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round 86 to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{90}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Rounding to the nearest hundred}

\begin{formulabox}
\textbf{Rule:} Same line, bigger jumps: the neighbors are whole hundreds and
the \emph{tens} digit decides --- 0--4 tens stays down, 5--9 tens goes up.
The ones digit never gets a vote.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round 238 to the nearest hundred.
\step{Trap 238 between its neighbor hundreds, 200 and 300, with the halfway mark at two hundred fifty.}
\step{The tens digit is three, so 238 sits before the halfway mark and stays down.}
$\Rightarrow\quad \ans{200}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round 561 to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{600}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} For the nearest hundred, read the \emph{tens} digit.
A number like 249 rounds \emph{down} to 200 --- its 9 ones cannot outvote the
4 tens.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Ordering numbers on a line}

\begin{formulabox}
\textbf{Rule:} On a number line, \emph{left is less} and \emph{right is
greater}. To order numbers, mark each one, then read the marks left to
right. To place a mark, compare hundreds, then tens, then ones.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Order least to greatest: 462, 426, 264.
\step{Mark by hundreds first: 264 has only two hundreds, so it sits far left of the other two.}
\step{462 and 426 tie at four hundreds; 426 has two tens and 462 has six tens, so 426 marks left of 462.}
$\Rightarrow\quad \ans{264,\ 426,\ 462}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Order least to greatest: 519, 591, 159.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{159,\ 519,\ 591}$}
\end{tryitbox}
