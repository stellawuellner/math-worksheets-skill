\sstitleblock{Signed Rational Numbers: Mixed Operations}

\skillheading{Skill 1 --- Adding signed numbers}

\begin{formulabox}
\textbf{Rule:} same signs --- add the absolute values, keep the sign.
Different signs --- subtract the absolute values, keep the sign of the number
farther from zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $-8 + 3$
\step{The two numbers have different signs, so subtract the absolute values and take the sign of the number farther from zero.}
\step{$8 - 3 = 5$, and $-8$ is farther from zero than $3$, so the sum is negative.}

$\Rightarrow\quad -8 + 3 = \ans{-5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $-6 + 10$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtracting signed numbers}

\begin{formulabox}
\textbf{Rule:} \quad $a - b = a + (-b)$ \quad --- rewrite every subtraction as adding
the opposite, then use the addition rules. Two minus signs in a row make a plus.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $-2 - (-9)$
\step{The number being subtracted is negative, so change the double negative into addition first, then apply the addition sign rules.}
\step{$-2 - (-9) = -2 + 9 = 7$, because $9 - 2 = 7$ and $9$ is farther from zero.}

$\Rightarrow\quad -2 - (-9) = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $3 - 8$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Multiplying and dividing signed numbers}

\begin{formulabox}
\textbf{Rule:} multiply or divide the absolute values, then set the sign:
same signs $\rightarrow$ positive, \ different signs $\rightarrow$ negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $(-5)(6)$
\step{This is a product, so multiply the absolute values first and decide the sign separately using the same-sign/different-sign rule.}
\step{$5 \cdot 6 = 30$; the factors have different signs, so the product is negative.}

$\Rightarrow\quad (-5)(6) = \ans{-30}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $-42 \div (-7)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 4 --- Mixed operations: choose the method}

\begin{formulabox}
\textbf{Rule:} work parentheses first, then multiply or divide, then add or subtract.
Before comparing or combining, rewrite the numbers in one form (all fractions with a
common denominator, or all decimals) --- pick whichever form is easier.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $-3\left(\tfrac{1}{2} - 2\right)$
\step{Order of operations says to simplify inside the parentheses first, then multiply by the signed factor using the sign rules.}
\step{$\tfrac{1}{2} - 2 = -\tfrac{3}{2}$, and $-3 \cdot \left(-\tfrac{3}{2}\right)$ has two negative factors, so it is positive: $\tfrac{9}{2}$.}

$\Rightarrow\quad -3\left(\tfrac{1}{2} - 2\right) = \ans{\tfrac{9}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $-2\left(\tfrac{5}{2} - 3\right)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}
