\sstitleblock{Reading a Limit Three Ways}

\begin{formulabox}
\textbf{One limit, three kinds of evidence.} A graph shows you the shape, a
table shows you the trend, and algebra settles the value. Use the first two to
predict and to catch mistakes; use the third to prove. Whenever two disagree,
the algebra wins --- and one of the other two was misread.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading a Limit Off a Graph}

\begin{formulabox}
\textbf{Rule:} trace the curve toward $x = c$ from each side and read the
\emph{height} it heads for. An open circle is a height the graph reaches but does
not occupy --- it still counts for the limit. A solid dot is $f(c)$ and does
\textbf{not} count. If the two sides head to different heights, the limit does
not exist.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Near $x = 2$ the graph of $f$ follows the line
$y = 3x - 1$ on both sides, with an open circle on that line at $x = 2$ and a
solid dot at $(2, 1)$. Find $\displaystyle\lim_{x \to 2} f(x)$.
\step{Both sides follow the same line, so the limit is the height that line reaches at $2$ --- the solid dot is the value $f(2)$, which a limit ignores.}
\step{On the line, the height at $x = 2$ is $3(2) - 1 = 5$, so both sides head to $5$.}

$\Rightarrow\quad$ $\ans{5}$ \quad (while $f(2) = 1$, so $f$ is not continuous at $2$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Near $x = -1$ the graph follows $y = x^2 + 2$ on both
sides, with an open circle there and a solid dot at $(-1, 0)$. Find
$\displaystyle\lim_{x \to -1} f(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Reading a Limit Off a Table}

\begin{formulabox}
\textbf{Rule:} inputs must close in from \emph{both} sides. Read the outputs and
ask what number they are squeezing toward --- not what the last entry says. Two
sides settling on different numbers is a jump. A column that keeps growing is an
infinite limit. A table can only ever predict: it samples finitely many inputs,
so it is evidence, never proof.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table for $h(x) = \frac{x^2-16}{x-4}$ gives $7.9$ at
$x = 3.9$ and $8.1$ at $x = 4.1$. Find $\displaystyle\lim_{x \to 4} h(x)$.
\step{The two sides are closing in on $8$ from below and above, so predict $8$ --- then prove it, because a table is not a proof.}
\step{$\dfrac{x^2-16}{x-4} = \dfrac{(x-4)(x+4)}{x-4} = x+4$ for $x \neq 4$, and $4+4 = 8$, matching the prediction.}

$\Rightarrow\quad$ $\ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table for $\dfrac{x^2-100}{x-10}$ near $x = 10$ gives
values just under and just over $20$. Find the exact limit.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Proving It With Algebra}

\begin{formulabox}
\textbf{Rule:} substitute first. A number is the answer; $\frac{0}{0}$ means
rewrite. Three rewrites cover almost everything: factor and cancel, multiply by
the conjugate when a square root blocks the cancelling, and split $|x - c|$ by
side. After the rewrite, substitute again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to 0} \frac{\sqrt{x+9}-3}{x}$
\step{Substituting gives $\frac{0}{0}$ and nothing factors, so clear the root with the conjugate $\sqrt{x+9}+3$.}
\step{$\dfrac{(x+9)-9}{x\left(\sqrt{x+9}+3\right)} = \dfrac{1}{\sqrt{x+9}+3}$, and at $x=0$ that is $\dfrac{1}{6}$.}

$\Rightarrow\quad$ $\ans{\tfrac{1}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to 0} \frac{\sqrt{x+1}-1}{x}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2}}$}
\end{tryitbox}
