\wstitleblock{Comparing Fractions Without Being Fooled}{Grades 4--5 Mathematics}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item A fraction is a \emph{count} of pieces (the top) and a \emph{size} of piece (the bottom). Comparing only one of the two is the mistake this sheet is about.
  \item Show the equivalent fractions or the benchmark you used before writing a sign.
  \item Where a problem asks you to explain, write full sentences on the lines.
\end{itemize}

\vspace{0.2cm}

\problem[3cm]{Fill in the missing numerator: \quad $\dfrac{3}{4} = \dfrac{\rule{0.7cm}{0.4pt}}{12}$}

\problem[3cm]{Write $<$ or $>$ between the two fractions: \quad $\dfrac{4}{9} \qquad \dfrac{4}{7}$}

\problem[3.5cm]{Dev writes: ``$\dfrac{3}{8}$ is greater than $\dfrac{3}{5}$, because 8 is greater than 5.''
\begin{itemize}
  \item[(a)] Rewrite $\dfrac{3}{8}$ with denominator 40 and write its new numerator. \ansline
  \item[(b)] Rewrite $\dfrac{3}{5}$ with denominator 40 and write its new numerator. \ansline
  \item[(c)] Write the correct sign, $<$ or $>$, between $\dfrac{3}{8}$ and $\dfrac{3}{5}$. \ansline
  \item[(d)] Explain what went wrong in Dev's reasoning.
\end{itemize}
