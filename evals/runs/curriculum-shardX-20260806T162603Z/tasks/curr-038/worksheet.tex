\wstitleblock{Find the Missing Number}{Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Each problem has a missing number. Write it on the blank.
  \item The pictures and part--whole boxes are there to help you see the parts.
\end{itemize}

\problem[3cm]{The ten frame shows 8 counters. How many more counters are
needed to fill the frame?\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\ \hline
$\bullet$ & $\bullet$ & $\bullet$ & \phantom{$\bullet$} & \phantom{$\bullet$} \\ \hline
\end{tabular}
\end{center}
$8 + \ansblank = 10$}

\problem[3cm]{Use the part--whole box. One part is missing.\par
\begin{center}
\begin{tabular}{|c|c|}
\hline
\multicolumn{2}{|c|}{whole: 12} \\ \hline
part: 7 & part: ? \\ \hline
\end{tabular}
\end{center}
$7 + \ansblank = 12$}

\problem[3.5cm]{Jo has 9 grapes on her plate. She wants 15 grapes for her
snack. How many more grapes does she need?\par
$9 + \ansblank = 15$ \quad grapes}

\problem[3.5cm]{6 kids are riding the bus. At the next stop, more kids get on.
Now 11 kids are riding. How many kids got on at the stop?\par
$6 + \ansblank = 11$ \quad kids}

\problem[3cm]{Use the part--whole box. One part is missing.\par
\begin{center}
\begin{tabular}{|c|c|}
\hline
\multicolumn{2}{|c|}{whole: 16} \\ \hline
part: 9 & part: ? \\ \hline
\end{tabular}
\end{center}
$9 + \ansblank = 16$}

\problem[3.5cm]{This time the first part is missing.\par
$\ansblank + 8 = 14$}

\problem[4cm]{Missing numbers and subtraction are best friends.\par
(a) Find the missing number: \ $7 + \ansblank = 13$\par
(b) Now write the subtraction that checks your answer: \ $13 - 7 = \ansblank$}

\problem[4.5cm]{Lia had some shells in her pail. She found 6 more shells on
the beach. Now she has 14 shells. How many shells did she start with?\par
$\ansblank + 6 = 14$ \quad shells}
