\sstitleblock{Rigid Motions on the Coordinate Plane}

\skillheading{Skill 1 --- Translations: slide every point}

\begin{formulabox}
\textbf{Rule:} $(x, y) \rightarrow (x + h,\, y + k)$. Positive $h$ moves right, negative
$h$ moves left; positive $k$ moves up, negative $k$ moves down.\par
\smallskip
Apply it to \emph{every} vertex separately --- the rule is about points, and a figure moves
because all of its points do. Nothing rotates, nothing flips, and every length survives,
because both endpoints of a segment shift by the same amount and the gap between them
cannot change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Image of $A(-2, 5)$ under $(x, y) \rightarrow (x + 7,\, y - 4)$.
\step{The rule is already in coordinate form, so there is nothing to set up --- work one
coordinate at a time and let the signs do the moving.}
\step{$x' = -2 + 7$, \quad $y' = 5 - 4$.}

$\Rightarrow\quad A' = (\ans{5},\, \ans{1})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Image of $B(3, -6)$ under the same rule.\\[2pt]
\rotatebox{180}{\footnotesize check: $B' = (\ans{10},\, \ans{-10})$}
\end{tryitbox}

\skillheading{Skill 2 --- Reflections: which coordinate flips}

\begin{formulabox}
\textbf{Over the $x$-axis:} $(x, y) \rightarrow (x,\, -y)$. \quad
\textbf{Over the $y$-axis:} $(x, y) \rightarrow (-x,\, y)$. \quad
\textbf{Over the line $y = x$:} $(x, y) \rightarrow (y,\, x)$.\par
\smallskip
The trick for keeping the first two apart: the axis you reflect \emph{over} is the
coordinate that stays put. A horizontal mirror flips you up and down, so it changes $y$; a
vertical mirror flips you sideways, so it changes $x$.\par
\smallskip
The mirror is always the perpendicular bisector of the segment joining a point to its
image --- a free check on any answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect $C(4, -7)$ over the $x$-axis.
\step{The mirror is horizontal, so it can only change how far up or down the point is ---
that is the $y$-coordinate, and $x$ is left alone.}
\step{$x' = 4$, \quad $y' = -(-7)$.}

$\Rightarrow\quad C' = (\ans{4},\, \ans{7})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Reflect $D(-5, 2)$ over the $x$-axis.\\[2pt]
\rotatebox{180}{\footnotesize check: $D' = (\ans{-5},\, \ans{-2})$}
\end{tryitbox}

\skillheading{Skill 3 --- Rotations about the origin}

\begin{formulabox}
\textbf{$90^\circ$ counterclockwise:} $(x, y) \rightarrow (-y,\, x)$. \quad
\textbf{$180^\circ$:} $(x, y) \rightarrow (-x,\, -y)$. \quad
\textbf{$90^\circ$ clockwise:} $(x, y) \rightarrow (y,\, -x)$.\par
\smallskip
For the quarter turns: \emph{swap the coordinates, then fix one sign}. Which sign is the
only thing the direction changes, so decide the direction first and read the rule second.
\par
\smallskip
A rotation about the origin keeps every point the same distance from the origin, so
$x^2 + y^2$ is unchanged --- swapping and negating cannot alter a sum of squares.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rotate $E(3, 8)$ by $90^\circ$ counterclockwise about the origin.
\step{A quarter turn always swaps the two coordinates; counterclockwise is the direction
that makes the new first coordinate the negated one.}
\step{Swap to get $(8, 3)$, then negate the first: $x' = -8$, \quad $y' = 3$.}

$\Rightarrow\quad E' = (\ans{-8},\, \ans{3})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rotate $F(-2, 6)$ by $90^\circ$ counterclockwise about the origin.\\[2pt]
\rotatebox{180}{\footnotesize check: $F' = (\ans{-6},\, \ans{-2})$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} all three of these are \textbf{rigid} motions --- they preserve
every distance and every angle, so a figure and its image are always congruent. That is
why you never need to measure all three sides to justify a congruence: naming the motion
is the argument.
\end{watchoutbox}
