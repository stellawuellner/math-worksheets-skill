\sstitleblock{Precise Definitions and Counterexamples}

\skillheading{Skill 1 --- Reading a definition exactly}

\begin{formulabox}
\textbf{A definition must be testable.} Every word has to name something you can check on a figure: a location (``on the segment''), a relation (``congruent''), or a measure (``$90^\circ$'').\par
\textbf{Words that are not conditions:} ``looks like'', ``about the same'', ``roughly''. If a definition rests on one of these, it cannot be used in a proof.\par
\textbf{Perpendicular lines:} two lines that intersect \emph{to form a right angle}. The angle clause is the whole content --- ``two lines that intersect'' is every crossing pair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Lines $j$ and $k$ are perpendicular, and one of the four angles they form measures $(4x + 10)^\circ$. Find $x$.
\step{The definition of perpendicular says the angle formed is a right angle, so I can replace the word ``perpendicular'' with the equation ``that measure is $90$''.}
\step{$4x + 10 = 90$, so $4x = 80$.}

$\Rightarrow\quad \ans{x = 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Lines $p$ and $q$ are perpendicular, and one angle they form measures $(6x - 12)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 17}$}
\end{tryitbox}

\skillheading{Skill 2 --- Turning a definition into an equation}

\begin{formulabox}
\textbf{Midpoint} of $\overline{AB}$: the point $M$ \emph{on} $\overline{AB}$ with $\overline{AM} \cong \overline{MB}$ \ $\Rightarrow$ \ $AM = MB$.\par
\textbf{Angle bisector} of $\angle ABC$: the ray $\overrightarrow{BD}$ with $\angle ABD \cong \angle DBC$ \ $\Rightarrow$ \ $m\angle ABD = m\angle DBC$.\par
\textbf{Congruent segments:} $\overline{PQ} \cong \overline{RS}$ \ $\Rightarrow$ \ $PQ = RS$.\par
Each definition turns one congruence mark into one equation. Write the equation first, then solve it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $M$ is the midpoint of $\overline{AB}$, with $AM = 4x + 3$ and $MB = 6x - 9$. Find $x$.
\step{Midpoint means the two pieces are congruent, and congruent segments have equal lengths --- so the definition hands me the equation $AM = MB$.}
\step{$4x + 3 = 6x - 9$, so $12 = 2x$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\overrightarrow{BD}$ bisects $\angle ABC$, with $m\angle ABD = 5x - 2$ and $m\angle DBC = 3x + 8$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Building a counterexample}

\begin{formulabox}
To disprove ``if $P$ then $Q$'', produce one case that \textbf{satisfies $P$} and \textbf{fails $Q$}. One case is enough, and it must do both jobs --- always check the hypothesis numerically before claiming the conclusion fails.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Disprove: ``If two angles are supplementary, then each one is acute.'' Two angles measure $a$ and $b$ degrees.
\step{I need a pair that meets the hypothesis --- supplementary means the measures add to $180$ --- and breaks the conclusion, so at least one must be $90$ or more; try $a = 120$ and $b = 60$.}
\step{Hypothesis check: $a + b = 120 + 60$, and $120$ is obtuse, so the conclusion fails.}

$\Rightarrow\quad$ the sum is $\ans{180}$, so the pair really is supplementary, and the claim is false.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Disprove ``if two angles are complementary, then they are congruent'' with the pair $a = 30$ and $b = 60$: check the sum the hypothesis needs.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{90}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a counterexample that fails the ``if'' part proves nothing. Check the hypothesis first, then show the conclusion breaks.
\end{watchoutbox}
