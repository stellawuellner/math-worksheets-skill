\sstitleblock{Circular Reasoning and Unsupported Statements}

\skillheading{Skill 1 --- When a reason just repeats the claim}
\begin{formulabox}\textbf{Test every reason with one question:} was this fact available
\emph{before} this step? A reason may be a given, a definition, a postulate, or a
theorem already proved. If the reason is the statement itself --- or the theorem you
were asked to prove --- the step is \textbf{circular} and proves nothing.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Prove vertical angles congruent. A student
writes: (1) the angles are vertical --- Given; (2) their measures are equal --- vertical
angles are congruent; (3) the angles are congruent --- definition of congruence.
\step{Read each reason and ask whether it was available: step 2 cites the very theorem being proved, so the proof assumes its conclusion.}
\step{Repair it with facts that come earlier: each angle is supplementary to the same third angle (two linear pairs), and angles supplementary to the same angle are congruent.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad ``$\overline{AB} \cong \overline{CD}$ because they
are congruent segments.'' Which rule does that reason break?\\[2pt]
\rotatebox{180}{\footnotesize check: it restates the claim --- cite the given or the definition of congruent segments}\end{tryitbox}

\skillheading{Skill 2 --- Use only what the givens support}
\begin{formulabox}\textbf{Watch the words you were handed.} ``Linear pair'' gives you
supplementary ($+ = 180^\circ$). ``Vertical angles'' gives you congruent ($=$).
``Corresponding angles'' gives you congruent only when parallel lines are given. Picking
the wrong one changes the equation you solve, so it changes the answer.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\angle PQR$ and $\angle RQS$ form a linear
pair, $m\angle PQR = 4x + 10$ and $m\angle RQS = 6x - 20$. Find $x$.
\step{The given is a linear pair, which supports supplementary --- so add the measures to $180$, do not set them equal.}
\step{$(4x + 10) + (6x - 20) = 180$ gives $10x - 10 = 180$, so $10x = 190$ and $\ans{x = 19}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Two angles forming a linear pair measure $3x + 5$
and $7x + 15$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 16}$}\end{tryitbox}

\skillheading{Skill 3 --- Name the postulate that does the work}
\begin{formulabox}\textbf{Segment Addition Postulate:} if $M$ is between $A$ and $B$,
then $AM + MB = AB$. Betweenness alone gives you this. \textbf{Midpoint} is a stronger,
separate given: only then may you write $AM = MB$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $M$ is between $A$ and $B$ with $AM = 3x - 1$,
$MB = x + 5$, and $AB = 28$. Find $x$.
\step{The given is betweenness, not a midpoint, so the postulate that applies is segment addition --- add the parts to the whole.}
\step{$(3x - 1) + (x + 5) = 28$ gives $4x + 4 = 28$, so $4x = 24$ and $\ans{x = 6}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $M$ is between $A$ and $B$ with $AM = 2x + 1$,
$MB = 3x - 6$, and $AB = 30$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}\end{tryitbox}
