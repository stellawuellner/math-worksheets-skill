\sstitleblock{Triangle Similarity --- Reversed Correspondence and Scale-Factor Errors}

\skillheading{Skill 1 --- Reading the correspondence from the statement}

\begin{formulabox}
\textbf{Rule:} the letters of a similarity statement are matched \emph{in order}.
$\triangle ABC \sim \triangle DEF$ means
$A \leftrightarrow D$, \ $B \leftrightarrow E$, \ $C \leftrightarrow F$, and
therefore
\[ \frac{AB}{DE} = \frac{BC}{EF} = \frac{CA}{FD} \]
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A side is named by its two endpoints, so its partner is named by the two
        matching letters --- $\overline{BC}$ pairs with $\overline{EF}$, never with
        $\overline{DE}$.
  \item Write the pairs down before writing any fraction. A scrambled statement
        such as $\triangle ABC \sim \triangle FED$ is legal and means
        $A \leftrightarrow F$, $B \leftrightarrow E$, $C \leftrightarrow D$.
  \item Keep every ratio pointing the same way: first triangle on top in
        \emph{both} fractions, or second on top in both.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\triangle ABC \sim \triangle DEF$ with $AB = 4$, $BC = 7$
and $DE = 10$. Find $EF$.
\step{$\overline{AB}$ and $\overline{DE}$ are a corresponding pair and both lengths
are known, so they fix the factor; $\overline{BC}$ pairs with the unknown
$\overline{EF}$.}
\step{$DEF$ is the larger triangle ($10 > 4$), so multiply by
$\dfrac{10}{4}$: \ $EF = 7 \cdot \dfrac{10}{4}$.}
$EF = \ans{17.5}$, longer than $BC$ as expected.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\triangle ABC \sim \triangle DEF$ with $AB = 3$, $BC = 5$
and $DE = 12$. Find $EF$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

\skillheading{Skill 2 --- Pointing the scale factor the right way}

\begin{formulabox}
\textbf{Rule:} write the scale factor as
\[ k = \frac{\text{side of the triangle you are going TO}}
            {\text{matching side of the triangle you are coming FROM}} \]
\textbf{Then predict before you multiply:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Going to the \emph{smaller} triangle, $k < 1$ and every answer shrinks.
  \item Going to the \emph{larger} triangle, $k > 1$ and every answer grows.
  \item An answer on the wrong side of the side you started from is a reversed
        factor --- and the two wrong answers are reciprocals of each other, never
        near misses.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\triangle RST \sim \triangle XYZ$ with $RS = 18$,
$XY = 6$ and $ST = 15$. Find $YZ$.
\step{The known pair $RS = 18$ and $XY = 6$ says $XYZ$ is the smaller triangle, so
the factor must be less than $1$ and $YZ$ must be shorter than $15$.}
\step{$k = \dfrac{6}{18} = \dfrac{1}{3}$, so $YZ = 15 \cdot \dfrac{6}{18}$.}
$YZ = \ans{5}$, which is indeed less than $15$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\triangle RST \sim \triangle XYZ$ with $RS = 21$,
$XY = 6$ and $ST = 14$. Find $YZ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Scaling an area, not a side}

\begin{formulabox}
\textbf{Rule:} if corresponding sides are in the ratio $k$, then
\[ \frac{\text{larger area}}{\text{smaller area}} = k^2 \]
An area is a product of two lengths, and \emph{both} lengths get multiplied by
$k$, so the area is multiplied by $k \cdot k$. Using $k$ alone on an area is the
most common scale-factor error after reversing the direction.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Corresponding sides are in the ratio $3 : 7$ and the smaller
triangle's area is $18$. Find the larger area.
\step{The question moves between \emph{areas}, so the side ratio has to be squared
before it is used.}
\step{$k = \dfrac{7}{3}$, so $k^2 = \dfrac{49}{9}$, and the larger area is
$18 \cdot \dfrac{49}{9}$.}
The larger area is $\ans{98}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Corresponding sides are in the ratio $2 : 5$ and the smaller
triangle's area is $12$. Find the larger area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A reversed correspondence and a reversed scale factor both
produce a clean-looking number, so re-doing the arithmetic never catches them.
Catch them by size instead: decide which triangle is larger \emph{first}, and
reject any answer that moves the wrong way.
\end{watchoutbox}
