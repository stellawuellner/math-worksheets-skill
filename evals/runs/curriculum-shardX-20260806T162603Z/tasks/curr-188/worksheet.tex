\begin{center}
\large Reading unit rates from graphs, writing equations, and graphing from tables
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item A proportional relationship graphs as a \textbf{straight line through
        the origin} $(0,0)$, and its unit rate is the $y$-value at $x = 1$.
  \item Show your work beside each graph. Units are stated in each question.
\end{itemize}
