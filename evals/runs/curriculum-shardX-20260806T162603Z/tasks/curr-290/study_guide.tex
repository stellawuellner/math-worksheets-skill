\sstitleblock{Choosing and Combining Factoring Techniques}

\skillheading{Skill 1 --- Take out the common factor first}

\begin{formulabox}
\textbf{Ask three questions, in this order:} (1) Is there a factor in \emph{every} term? Take it out. (2) How many terms are left --- two or three? (3) Which pattern fits? Two terms with a minus between squares is a difference of squares; three terms is a trinomial.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $10x^2 - 15x$ completely.
\step{Question 1 first: both terms carry a factor, so the common factor comes out before anything else is considered.}
\step{The greatest common factor of $10x^2$ and $15x$ is $5x$, and $10x^2 = 5x(2x)$, $15x = 5x(3)$.}
\par
$10x^2 - 15x = $ $\ans{5x(2x-3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $8x^3 + 12x^2$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x^2(2x+3)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``completely'' means keep going. $4x^2 - 36 = 4(x^2-9)$ is not the end --- the leftover is a difference of squares and factors again.
\end{watchoutbox}

\skillheading{Skill 2 --- Trinomials that start with a plain $x^2$}

\begin{formulabox}
\textbf{Rule:} \quad $x^2 + bx + c = (x+p)(x+q)$ \ where $p \cdot q = c$ and $p + q = b$.\par\smallskip
\textbf{Signs:} if $c$ is positive, $p$ and $q$ share the sign of $b$; if $c$ is negative, they have opposite signs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^2 + 9x + 20$.
\step{Three terms, no common factor, and the leading coefficient is $1$ --- that is the product-and-sum method, not grouping.}
\step{Product $20$ and sum $9$: the pair is $4$ and $5$, both positive because the product is positive and the sum is positive.}
\par
$x^2 + 9x + 20 = $ $\ans{(x+4)(x+5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $x^2 - 2x - 15$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-5)(x+3)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Trinomials with a number in front}

\begin{formulabox}
\textbf{Grouping method for} $ax^2 + bx + c$: find two numbers with product $a \cdot c$ and sum $b$; split the middle term into those two pieces; group the four terms in pairs and take a common factor out of each pair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $3x^2 + 11x + 6$.
\step{Three terms, no common factor, and the leading coefficient is not $1$, so the product-and-sum shortcut does not apply --- use grouping.}
\step{Product $3 \cdot 6 = 18$, sum $11$: the pair is $9$ and $2$. Split and group: $3x^2 + 9x + 2x + 6 = 3x(x+3) + 2(x+3)$.}
\par
$3x^2 + 11x + 6 = $ $\ans{(3x+2)(x+3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $2x^2 - 7x + 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2x-3)(x-2)}$}
\end{tryitbox}

\skillheading{Skill 4 --- Difference of two squares}

\begin{formulabox}
\textbf{Rule:} \quad $a^2 - b^2 = (a-b)(a+b)$. Two terms only, both perfect squares, joined by a minus sign. A \emph{sum} of two squares does not factor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^2 - 100$.
\step{Two terms rather than three, both perfect squares, with a minus between them --- that rules out every trinomial method and picks this pattern.}
\step{$a = \sqrt{x^2} = x$ and $b = \sqrt{100} = 10$, so the factors are $(a-b)$ and $(a+b)$.}
\par
$x^2 - 100 = $ $\ans{(x-10)(x+10)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $36x^2 - 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(6x-1)(6x+1)}$}
\end{tryitbox}
