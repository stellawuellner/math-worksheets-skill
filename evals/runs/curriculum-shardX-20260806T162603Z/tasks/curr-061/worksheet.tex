\wstitleblock{Subtracting Two-Digit Numbers with Place-Value Blocks}{Grades 2--3}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item In every picture, a tall rod is one ten and a small square is one.
  \item Cross out blocks to take away. Start with the ones.
  \item If there are not enough ones, open one rod into ten ones first --- that is regrouping. The total does not change, only how it is shown.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

  }
  \foreach \k in {0,...,6} { \draw[line width=0.7pt] (9,\k) rectangle (10,\k+1); }
\end{tikzpicture}

\par\vspace{0.15cm}
