\sstitleblock{Factor Instead of Dividing, and Check Every Candidate}

\skillheading{Skill 1 --- Factor, never divide}

\begin{formulabox}
\textbf{Zero-product property:} if $PQ = 0$ then $P = 0$ \emph{or} $Q = 0$ ---
both branches, always.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Dividing by an expression assumes it is not zero, which quietly deletes
        every solution that makes it zero.
  \item Move everything to one side, factor, then solve each branch separately.
  \item The lost solutions are exactly the zeros of whatever was divided by ---
        useful for diagnosing someone else's work.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2\sin\theta\cos\theta = -\sin\theta$ on
$0^\circ \le \theta < 360^\circ$.
\step{Both sides share a $\sin\theta$, which is the temptation to divide, so
move everything to one side and factor instead to keep that branch alive.}
\step{$\sin\theta\,(2\cos\theta + 1) = 0$, so $\sin\theta = 0$ or
$\cos\theta = -\frac{1}{2}$}

The solutions are $\ans{\theta = 0^\circ,\; 120^\circ,\; 180^\circ,\; 240^\circ}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2\sin\theta\cos\theta = \cos\theta$ on
$0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = 30^\circ,\; 90^\circ,\; 150^\circ,\; 270^\circ}$}
\end{tryitbox}

\skillheading{Skill 2 --- Only factors cancel}

\begin{formulabox}
\textbf{The rule:} $\dfrac{ab}{a} = b$, but $\dfrac{a + b}{a} \neq b$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A denominator divides the \emph{whole} numerator, so it cancels only
        against a factor common to every term.
  \item Factor first (difference of squares, common factor), then cancel; or
        split the fraction term by term, which is always legal.
  \item One numerical test kills a wrong simplification: pick a convenient value
        and compare the two forms.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $\dfrac{\cos^2 x - 1}{\cos x - 1}$ equal to $\cos x$?
Test it where $\cos x = 0.6$.
\step{A suspected illegal cancellation is fastest to settle numerically, so
evaluate the original expression at one convenient value and compare.}
\step{$\dfrac{0.36 - 1}{0.6 - 1} = \dfrac{-0.64}{-0.4}$, while $\cos x$ would
give $0.6$}

The true value is $\ans{1.6}$, so the cancellation is wrong; factoring gives
$\cos x + 1$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Evaluate $\dfrac{\cos^2 x - 1}{\cos x - 1}$ where
$\cos x = 0.25$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1.25}$}
\end{tryitbox}

\skillheading{Skill 3 --- Keep both signs, check every candidate}

\begin{formulabox}
\textbf{Square roots:} $\sqrt{u^2} = |u|$, so $u^2 = k$ gives
$u = \pm\sqrt{k}$ --- two branches. \qquad
\textbf{Squaring both sides} can add solutions: $A^2 = B^2$ only forces
$A = \pm B$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Safer than rooting: factor as a difference of squares, which shows both
        branches without any sign to remember.
  \item Squaring \emph{adds} solutions; dividing \emph{removes} them. Only the
        first can be repaired afterwards --- by testing every candidate in the
        original equation.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $4\cos^2\theta = 1$ on
$0^\circ \le \theta < 360^\circ$.
\step{Rooting both sides risks dropping the negative branch, so factor the
difference of squares and let each factor supply its own sign.}
\step{$(2\cos\theta - 1)(2\cos\theta + 1) = 0$, so $\cos\theta = \pm\frac{1}{2}$}

The solutions are $\ans{\theta = 60^\circ,\; 120^\circ,\; 240^\circ,\; 300^\circ}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $4\sin^2\theta = 2$ on
$0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = 45^\circ,\; 135^\circ,\; 225^\circ,\; 315^\circ}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} If your solution set is a strict subset of a classmate's
and both of you did correct arithmetic, look for a division --- the missing
angles are the zeros of whatever you divided by.
\end{watchoutbox}
