\wstitleblock{Interior and Exterior Polygon Angle Sums}{High School Geometry}{}

\noindent\small
Interior angle sums, a missing angle inside a polygon, exterior angle sums, and
the interior and exterior angles of a regular polygon.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=3pt]
  \item Every polygon here is convex. Let $n$ be the number of sides.
  \item State the theorem you are using before you substitute.
  \item Give angle answers in degrees; give side-count answers as a whole number.
\end{itemize}
\normalsize

{\small \textbf{How angles are named here.} No values are shown: use the numbers
given in each problem. An \emph{interior} angle sits inside the polygon at a
vertex; its \emph{exterior} angle is formed by extending one side, and the two
together form a straight line.}
\end{center}

\problem[5cm]{Find the sum of the interior angles of a convex hexagon.
\answerline{degrees}}

\problem[5cm]{Find the sum of the interior angles of a convex 15-gon.
\answerline{degrees}}

\problem[5cm]{A regular octagon has all of its interior angles equal. Find the
measure of one interior angle.
\answerline{degrees}}

\problem[5cm]{A regular 12-gon has all of its exterior angles equal. Find the
measure of one exterior angle.
\answerline{degrees}}

\problem[6cm]{Four of the interior angles of a convex pentagon measure
$102^\circ$, $118^\circ$, $96^\circ$, and $131^\circ$. Find the measure $x$ of the
fifth interior angle.
\answerline{degrees}}

\problem[6cm]{Each exterior angle of a regular polygon measures $24^\circ$.
How many sides $n$ does the polygon have?}

\problem[6cm]{Each interior angle of a regular polygon measures $150^\circ$.
How many sides $n$ does the polygon have? Set up the equation from the interior
angle sum rather than guessing.}

\problem[6cm]{The four interior angles of a convex quadrilateral are in the ratio
$1 : 2 : 3 : 4$, so they can be written $x$, $2x$, $3x$, and $4x$. Find $x$.
\answerline{degrees}}

\problem[6cm]{The interior angles of a convex polygon add to $1980^\circ$.
How many sides $n$ does the polygon have?}

\problem[9cm]{\textbf{Justify.} Prove or explain in complete sentences why the
exterior angles of \emph{every} convex polygon add to $360^\circ$, no matter how
many sides it has.
\par Use the interior angle sum $(n-2)180^\circ$ and the fact that an interior
angle and its exterior angle at the same vertex form a linear pair. Your
explanation must work for a general $n$, not just for one example.
\noansline}
