% =================== SKILL 1: equal-parts-check ===================
\skillheading{Check First: Are the Parts Equal?}

\begin{formulabox}
\textbf{Rule:} A fraction only names \emph{equal} parts. Before writing any
fraction, ask: \textbf{are all the parts the same size?} If one part is bigger
than another, the parts do not have a fraction name yet.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A paper strip is cut into parts that are 3 cm, 3 cm, and
6 cm long. Do the parts show thirds?
\step{To call each part one third, all three parts must be the same length ---
so compare the part lengths before naming any fraction.}
\step{Two parts are 3 cm, but the last is 6 cm: one part is longer than the
others.}

The lengths are not all the same, $\ans{3 < 6}$, so the strip does
\emph{not} show thirds.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A ribbon is cut into parts that are 4 cm, 5 cm, and 4 cm
long. Do the parts show thirds?\\[2pt]
\rotatebox{180}{\footnotesize check: no --- $\ans{4 < 5}$, the parts are not equal}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Equal parts can have \emph{different shapes}. Equal
means the same amount of space (or length), not the same look.
\end{watchoutbox}

% =================== SKILL 2: write-the-fraction ===================
\skillheading{Write the Fraction from an Equal-Parts Model}

\begin{formulabox}
\textbf{Rule:} When the parts \emph{are} equal:
$\text{fraction shaded} = \dfrac{\text{shaded parts}}{\text{equal parts in all}}$
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle is cut into 8 equal parts, and 3 parts are
shaded. What fraction is shaded?
\step{The parts are equal, so a fraction is allowed: count shaded parts for
the top number and all the parts for the bottom number.}
\step{Shaded parts: 3. Equal parts in all: 8.}

So the shaded fraction is $\ans{\tfrac{3}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A circle is cut into 6 equal parts, and 5 parts are shaded.
What fraction is shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{6}}$}
\end{tryitbox}

% =================== SKILL 3: fix-with-equal-units ===================
\skillheading{Fix an Unequal Model: Count Equal Squares}

\begin{formulabox}
\textbf{Rule:} If the parts are \emph{not} equal, do not count the parts!
Cover the whole with same-size small squares, then count:
$\text{true fraction} = \dfrac{\text{squares shaded}}{\text{squares in all}}$
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle made of 2 rows of 5 small squares is split by
heavy lines into four parts that are \emph{not} equal. The shaded part covers
2 small squares. What fraction is really shaded?
\step{The parts are not equal, so parts cannot be counted --- count the
same-size small squares instead.}
\step{Squares in all: $2 \times 5 = 10$. Squares shaded: 2, so the fraction is
$\tfrac{2}{10}$, which is the same as one fifth.}

The true shaded fraction is $\ans{\tfrac{1}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle made of 3 rows of 4 small squares is split into
five unequal parts. The shaded part covers 3 small squares. What fraction is
really shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{4}}$}
\end{tryitbox}
