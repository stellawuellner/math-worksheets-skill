\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every problem shows work that is wrong somewhere. Find the error, then repair it.
  \item Part (a) always wants a corrected value. Put it on the Answer line.
  \item Part (b) always wants a sentence or two naming the error --- not just ``it is different'', but which rule was broken. Write it in the space above the Answer line.
  \item Round only at the last step, to the number of decimal places asked for.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{A student separates $\dfrac{dy}{dx} = \dfrac{x}{y}$ correctly as $y\,dy = x\,dx$, then integrates to get $y^2 = x^2 + C$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Give the correct result of integrating both sides, in the form $\frac{1}{2}y^2 = G(x) + C$.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[6.5cm]{To solve $\dfrac{dy}{dx} = 6x^2 y$ with $y(0) = 4$, a student writes $\ln|y| = 6x^3 + C$ and reports $y(1) = 4e^{6} \approx 1613.715$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $y(1)$, to three decimal places.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[6.5cm]{For $\dfrac{dy}{dx} = \dfrac{y}{5}$ with $y(0) = 7$, a student integrates correctly to $\ln|y| = \dfrac{x}{5} + C$, then writes $y = e^{x/5} + C$. From $y(0) = 7$ they get $C = 6$ and report $y(10) = e^{2} + 6 \approx 13.389$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $y(10)$, to three decimal places.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[6.5cm]{A student solves $\dfrac{dy}{dx} = \dfrac{4x}{y}$ with $y(0) = 3$ and $y > 0$. Their work reads: $y\,dy = 4x\,dx$, then $y^2 = 2x^2 + C$, then $C = 9$, and finally $y(2) \approx 4.123$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $y(2)$.
  \item[(b)] Name the error the student made, and say why the value of $C$ they found is wrong as a consequence of it.
\end{itemize}}

\problem[6.5cm]{A savings balance $P$, in dollars, satisfies $\dfrac{dP}{dt} = 0.06P$ with $P(0) = 2000$, where $t$ is in years. A student writes $P = e^{0.06t} + 2000$ and reports $P(15) \approx 2002.46$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $P(15)$, in dollars, to two decimal places.
  \item[(b)] Name the error, and describe the quick sanity check on the story that should have caught it.
\end{itemize}}

\problem[6.5cm]{A student looks at $\dfrac{dy}{dx} = x + xy$ and says the equation cannot be separated because the right side is a sum.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Separate the equation anyway, solve it with $y(0) = 2$, and give $y(1)$ to three decimal places.
  \item[(b)] Explain the test that decides whether a right side containing a sum is separable.
\end{itemize}}

\problem[7cm]{Two students solve $\dfrac{dy}{dx} = \dfrac{\cos x}{2y}$ with $y(0) = 3$ and $y > 0$. One writes $y^2 = \sin x + 9$; the other writes $y^2 = \sin x + 3$ and reports $y\!\left(\frac{\pi}{2}\right) = 2$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $y$ at $x = \frac{\pi}{2}$, to three decimal places.
  \item[(b)] Explain which of the two solutions is wrong and exactly what was done incorrectly.
\end{itemize}}

\problem[7cm]{Here is a complete student solution to $\dfrac{dy}{dx} = \dfrac{2x}{y+1}$ with $y(0) = 3$ and $y > -1$. Exactly one line is wrong.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=2pt, label={}]
  \item[Line 1:] $(y+1)\,dy = 2x\,dx$
  \item[Line 2:] $\dfrac{(y+1)^2}{2} = x^2 + C$
  \item[Line 3:] substituting $(0, 3)$ gives $C = 16$
  \item[Line 4:] $y = -1 + \sqrt{2x^2 + 32}$, so $y(2) \approx 5.325$
\end{itemize}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Give the corrected value of $y(2)$, to three decimal places.
  \item[(b)] Identify the wrong line and explain the mistake in it.
\end{itemize}}
