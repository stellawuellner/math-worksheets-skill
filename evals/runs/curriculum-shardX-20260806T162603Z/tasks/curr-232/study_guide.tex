\sstitleblock{Evaluating Functions from Rules, Tables, and Graphs}

\skillheading{Skill 1 --- Putting an input into a rule}

\begin{formulabox}
\textbf{What the notation means:}\\[3pt]
$f(3)$ is the \textit{output} of $f$ when the \textit{input} is $3$. The letter in the
brackets is the number you substitute --- it is never a multiplication.
To evaluate, replace every $x$ in the rule with the input, in brackets, then simplify.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 5x - 4$. \quad Find $f(3)$.
\step{The input is a number and the rule is written in $x$, so substitute $3$ for $x$ and simplify.}
\step{$f(3) = 5(3) - 4 = 15 - 4$}
$\Rightarrow\quad f(3) = \ans{11}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 5x - 4$. \quad Find $f(-2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-14}$}
\end{tryitbox}

\skillheading{Skill 2 --- Reading an output out of a table}

\begin{formulabox}
\textbf{Rule:} a table pairs each input with its output, one column at a time.\\[3pt]
To find $g(3)$: look along the input row for $3$, then read the number directly below (or
beside) it in the output row. No arithmetic is needed --- the value is already there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $g(3)$ from this table.
\par\vspace{0.1cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $1$ & $2$ & $3$ & $4$ \\ \hline
$g(x)$ & $6$ & $10$ & $14$ & $18$ \\ \hline
\end{tabular}
\end{center}
\step{The value is stored in the table, so this is a lookup: find the input column, read the output.}
\step{The input $3$ sits in the third column; the output row holds $14$ there.}
$\Rightarrow\quad g(3) = \ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $g(4)$ from the table $x = 0, 2, 4, 6$ with $g(x) = 5, 9, 13, 17$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reading an output off a graph}

\begin{formulabox}
\textbf{Rule:} on a graph, the input is the horizontal coordinate and the output is the
vertical one. To find $f(a)$: go to $a$ on the horizontal axis, move up or down to the
plotted point, then read across to the vertical axis. That point is $(a, f(a))$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $f(2)$ from this graph.
\par\vspace{0.05cm}
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=center, xmin=-0.5, xmax=3.6, ymin=-1, ymax=11.5,
  xtick={0,1,2,3}, ytick={0,2,4,6,8,10}, grid=both,
  grid style={line width=0.2pt, draw=gray!55},
  major grid style={line width=0.35pt, draw=gray!55},
  tick label style={font=\scriptsize},
  xlabel={$x$}, ylabel={$y$}, width=5.6cm, height=4.1cm, enlargelimits=false]
\addplot[thick, blue, mark=*, mark size=1.5pt] coordinates {(0,1)(1,4)(2,7)(3,10)};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The output is not written anywhere, so read it off the picture: input across, output up.}
\step{Above $x = 2$ the plotted point is $(2, 7)$, so the output is the second coordinate.}
$\Rightarrow\quad f(2) = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From the same graph, find $f(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 4 --- Comparing two functions at the same input}

\begin{formulabox}
\textbf{Rule:} you can only compare outputs that come from the \textit{same} input.\\[3pt]
Evaluate each function at that input --- whichever form it is given in --- then compare the two
numbers with $<$, $>$, or $=$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 2x + 5$, and a table gives $h(4) = 12$. Compare $f(4)$ and $h(4)$.
\step{The two functions are given in different forms, so evaluate each one separately before comparing.}
\step{Rule: $f(4) = 2(4) + 5 = 13$. \quad Table: $h(4) = 12$. \quad Now compare $13$ with $12$.}
$\Rightarrow\quad \ans{13 > 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 2x + 5$ and a table gives $h(2) = 10$. Compare $f(2)$ and $h(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9 < 10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f(-4)$ and $-f(4)$ are different. In $f(-4)$ the negative goes
\textit{into} the rule; in $-f(4)$ you evaluate first and flip the sign afterwards.
\end{watchoutbox}
