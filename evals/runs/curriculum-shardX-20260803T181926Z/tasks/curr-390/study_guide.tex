% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Turning a story into a system}

\begin{formulabox}
\textbf{Rule:} one unknown quantity $=$ one letter; one sentence of the story that
compares quantities $=$ one equation. You need as many independent equations as letters.\par
\textbf{Two useful patterns:} a \emph{count} equation ($x + y = $ total number) and a
\emph{value} equation (price, mass or percent per item $\times$ number $=$ total).\par
\textbf{Solve by elimination:} scale one equation so a letter has matching coefficients, then subtract.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 2 large boxes and 3 small boxes weigh 31 kg; 1 large box and 1 small
box weigh 13 kg. Find each mass.
\step{Two unknown masses and two comparisons, so I want a 2-by-2 system: let $x$ be the large mass and $y$ the small mass, then eliminate one letter.}
\step{$2x + 3y = 31$ and $x + y = 13$. Times 2: $2x + 2y = 26$. Subtracting: $y = 31 - 26 = 5$, so $x = 13 - 5 = 8$.}
$\Rightarrow\quad \ans{x = 8}$ kg large, $\ans{y = 5}$ kg small
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 3 large boxes and 2 small boxes weigh 31 kg; 1 large box and 1 small
box weigh 12 kg. Find each mass.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$, $\ans{y = 5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Solving a 2-by-2 system with a matrix}

\begin{formulabox}
\textbf{Matrix form:} $ax + by = e$ and $cx + dy = f$ become $AX = B$ with
$A = \begin{bmatrix} a & b \\ c & d \end{bmatrix}$,
$X = \begin{bmatrix} x \\ y \end{bmatrix}$,
$B = \begin{bmatrix} e \\ f \end{bmatrix}$.\par
\textbf{Determinant:} $\det A = ad - bc$ (down-diagonal minus up-diagonal).\par
\textbf{Cramer's rule:} $x = \dfrac{\det A_x}{\det A}$, $y = \dfrac{\det A_y}{\det A}$, where
$A_x$ and $A_y$ are $A$ with that variable's column replaced by $B$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $2x + y = 11$ and $3x + 4y = 24$ with a matrix.
\step{The coefficients are messy for substitution but the system is square, so I put it in matrix form and use Cramer's rule.}
\step{$\det A = (2)(4) - (1)(3) = 5$; $\det A_x = (11)(4) - (1)(24) = 20$; $\det A_y = (2)(24) - (11)(3) = 15$.}
$\Rightarrow\quad \ans{x = 4}$, $\ans{y = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $3x + 2y = 12$ and $x + 4y = 14$ with a matrix.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 2}$, $\ans{y = 3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} if $\det A = 0$ you cannot divide by it and $A$ has no inverse.
That is not a mistake --- it means the two equations describe parallel lines
(no solution) or the very same line (infinitely many). Compare the equations directly
to say which.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Where a line meets a curve}

\begin{formulabox}
\textbf{Rule:} solve the linear equation for one letter, substitute it into the
nonlinear equation, and you are left with ONE quadratic equation in one letter.\par
\textbf{Then:} solve the quadratic, and put each root back into the \emph{line} to get
its partner coordinate. Two roots give two intersection points, one root gives a
tangent touch, no real root means the graphs never meet.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Where does $y = x^2 - 2x - 3$ meet $y = x + 1$?
\step{Both equations already give $y$, so I want the $x$ values where the two outputs agree --- set the right-hand sides equal and solve the quadratic that appears.}
\step{$x^2 - 2x - 3 = x + 1 \Rightarrow x^2 - 3x - 4 = 0 \Rightarrow (x - 4)(x + 1) = 0$, so $x = 4$ or $x = -1$; the line gives $y = 5$ and $y = 0$.}
$\Rightarrow\quad \ans{(4,5)}$ and $\ans{(-1,0)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Where does $y = x^2 + 2x - 1$ meet $y = 3x + 1$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,7)}$ and $\ans{(-1,-2)}$}
\end{tryitbox}
