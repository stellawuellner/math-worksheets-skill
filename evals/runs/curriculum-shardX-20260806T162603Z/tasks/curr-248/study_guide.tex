\sstitleblock{Comparing Groups with Two-Way Tables}

\skillheading{Skill 1 --- Reading a cell and a total}

\begin{formulabox}
\textbf{How a two-way table is built:} one variable labels the rows, the other labels the columns,
and each inner cell counts the people who are in \textit{both} of those categories.\\[3pt]
The last column holds each row's total, the last row holds each column's total, and the corner
holds the grand total. Adding a row across, or a column down, must reproduce those totals.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many people are in group $B$ and answered yes?
\par\vspace{0.05cm}
\begin{center}\small
\begin{tabular}{|l|c|c|c|}
\hline
 & \textbf{Yes} & \textbf{No} & \textbf{Total} \\ \hline
\textbf{Group A} & $14$ & $6$ & $20$ \\ \hline
\textbf{Group B} & $9$ & $11$ & $20$ \\ \hline
\textbf{Total} & $23$ & $17$ & $40$ \\ \hline
\end{tabular}
\end{center}
\step{Two categories at once means one inner cell, so find the row first and then the column.}
\step{Run along the Group B row to the Yes column; the cell where they meet holds the count.}
$\Rightarrow\quad \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In a table, Group A has $21$ yes and $9$ no, and Group B has $13$ yes and
$17$ no. How many are in group A and answered no?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 2 --- Turning a count into a row percentage}

\begin{formulabox}
\textbf{Rule:} \quad row percentage $= \dfrac{\text{cell}}{\text{that row's total}} \times 100$\\[3pt]
The denominator is the total of the row the cell sits in --- never the grand total, unless the
question really is about everybody. Choosing the wrong denominator is the most common mistake
with these tables.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A row of $30$ people contains $12$ who said yes. What percent of that row
said yes?
\step{The question is about that row only, so the row total is the denominator.}
\step{$\dfrac{12}{30} \times 100 = 0.4 \times 100$}
$\Rightarrow\quad \ans{40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A row of $24$ people contains $18$ who said yes. What percent of that row
said yes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two groups fairly}

\begin{formulabox}
\textbf{Rule:} compare \textit{fractions of each row}, not the raw counts.\\[3pt]
Two rows almost never hold the same number of people, so a bigger count can come from a bigger
group rather than a stronger tendency. Write each group's share as a fraction of its own row
total, simplify, then compare.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Group A: $12$ out of $30$ said yes. Group B: $18$ out of $24$ said yes.
Which share is larger?
\step{The row totals differ, so the counts cannot be compared directly --- build a fraction for each row.}
\step{$\dfrac{12}{30} = \dfrac{2}{5} = 0.4$ and $\dfrac{18}{24} = \dfrac{3}{4} = 0.75$, so group B has the larger share.}
$\Rightarrow\quad \ans{\tfrac{2}{5} < \tfrac{3}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Group A: $21$ out of $30$. Group B: $26$ out of $40$. Which share is
larger?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{21}{30} > \tfrac{26}{40}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Deciding whether there is an association}

\begin{formulabox}
\textbf{Rule:} work out each row's percentage for the \textit{same} column and look at the gap.\\[3pt]
A large gap means the two variables appear \textbf{associated} --- knowing the row tells you
something about the answer. A gap of a percentage point or two means little. An association is
not proof that one thing causes the other.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Row one: $42$ of $50$ passed. Row two: $30$ of $50$ passed. How many
percentage points separate them?
\step{Association is judged from the gap between row percentages, so compute both and subtract.}
\step{$\dfrac{42}{50} \times 100 = 84$ and $\dfrac{30}{50} \times 100 = 60$, so the gap is $84 - 60$.}
$\Rightarrow\quad \ans{24}$ percentage points --- a large gap, so the variables appear associated
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Row one: $45$ of $60$ liked it. Row two: $24$ of $60$ liked it. How many
percentage points separate them?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}
