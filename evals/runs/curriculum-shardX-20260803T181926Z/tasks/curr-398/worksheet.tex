\wstitleblock{Ellipses and Hyperbolas: Reading a Model's Key Features}{Algebra 2}{}

\noindent\small
Vertices and foci of ellipses, vertices and foci of hyperbolas, and the
asymptotes of a hyperbola. Every problem below gives you a real object modelled
by a conic in standard position, with the units stated in the problem.
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item For an ellipse $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ with $a > b$:
        vertices $(\pm a, 0)$, foci $(\pm c, 0)$, and $c^2 = a^2 - b^2$.
  \item For a hyperbola $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$: vertices
        $(\pm a, 0)$, foci $(\pm c, 0)$, $c^2 = a^2 + b^2$, and asymptotes
        $y = \pm \dfrac{b}{a}\,x$.
  \item Keep the units of the model in your answer, and show the arithmetic
        that produced each number.
\end{itemize}
\normalsize
\vspace{0.2cm}

\problem{The floor of a whispering gallery is an ellipse. With the centre of
the room at the origin and both distances measured in feet, the wall follows
$\dfrac{x^2}{169} + \dfrac{y^2}{144} = 1$. Two people standing at the foci can
hear each other whisper. How far from the centre of the room is each focus?
\par\vspace*{5cm}\answerline{ft}}

\problem[5cm]{Two radio towers stand on a hyperbola of positions given by
$\dfrac{x^2}{36} - \dfrac{y^2}{64} = 1$, where $x$ and $y$ are in kilometres.
Give the coordinates of the two vertices and of the two foci of this
hyperbola.}

\problem[6cm]{A lawn sprinkler wets an elliptical patch described by
$\dfrac{(x-3)^2}{25} + \dfrac{(y+2)^2}{9} = 1$, with $x$ and $y$ measured in
metres from a corner stake. Give the coordinates of the two endpoints of the
major axis, then the coordinates of the two foci.}

\problem[5cm]{The narrow waist of a cooling tower is modelled by the hyperbola
$\dfrac{x^2}{144} - \dfrac{y^2}{81} = 1$, with $x$ and $y$ in metres. Far above
and far below the waist the tower wall follows the asymptotes almost exactly.
Find the slope of each asymptote, then write the equations of both asymptotes.}

\problem[8cm]{A satellite travels an elliptical orbit with the centre of its
planet at one focus. Its closest approach to that focus measures $4200$
kilometres and its farthest distance measures $9800$ kilometres. The closest
distance equals $a - c$ and the farthest equals $a + c$, where $a$ is the
semi-major axis and $c$ is the centre-to-focus distance. Write a system of two
equations and solve it for $a$ and $c$.}

\problem[6cm]{Two navigation stations lie $20$ kilometres apart, at the two
foci of a hyperbola on which a ship sits. The ship's distances to the two
stations differ by $12$ kilometres, which is the vertex-to-vertex distance
$2a$. Find $a$, then find $b^2$ for the hyperbola
$\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ that models the ship's possible
positions.}

\problem[6cm]{A hyperbolic mirror is ground so that its cross-section has
vertices $(\pm 5, 0)$, with distances in centimetres, and asymptotes
$y = \pm \dfrac{2}{5}\,x$. Find $b$, then find the centre-to-focus distance
$c$, rounded to two decimal places.}

\problem[8cm]{An architect models a bridge arch as the upper half of an ellipse
centred at the origin, with $x$ and $y$ in metres. The survey shows the arch
passes through the point $(3, 4)$ and through the point $(6, 2)$. Substituting
each point into $\dfrac{x^2}{m} + \dfrac{y^2}{n} = 1$ gives a system in $m$ and
$n$; solve it for $m = a^2$ and $n = b^2$, then give the coordinates of the two
foci.}

\problem[6cm]{A tunnel cross-section follows $\dfrac{y^2}{49} - \dfrac{x^2}{16}
= 1$, with $x$ and $y$ in metres. Jae reports that the asymptotes have slopes
$\pm 0.57$. Explain what Jae did wrong, then give the correct slope of each
asymptote.}
