\wstitleblock{Bounds, Signs, and the Variable of Integration: Find the Error}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Most problems below show a student's work. The antiderivatives are
        correct --- the mistakes are in the \emph{limits}, the \emph{sign}, or
        the \emph{variable}.
  \item When a part asks you to name or describe an error, write a sentence.
  \item Give exact values (fractions, radicals) unless a decimal is requested.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Evaluate $\displaystyle\int_0^3 (2x - 4)\,dx$. Your answer is a
signed value, so keep the sign.}

\problem{Asked for $\displaystyle\int_5^2 2x\,dx$, a student writes
\emph{``$\left[x^2\right]$ from $2$ to $5$ is $25 - 4 = 21$.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Give the corrected value.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem{Asked for $\dfrac{d}{dx}\displaystyle\int_1^x (3t^2 + 1)\,dt$, a
student answers \emph{``$3t^2 + 1$, so at $x = 2$ the value is $4$.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Give the corrected value at $x = 2$.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem{Asked for $\displaystyle\int_0^2 x\left(x^2 + 1\right)^3 dx$, a student
substitutes $u = x^2 + 1$, $du = 2x\,dx$, and writes
\[ \frac{1}{2}\int_0^2 u^3\,du = \frac{1}{8}\left[u^4\right]_0^2 = 2. \]
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Give the corrected value.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{A particle moves with velocity $v(t) = t^2 - 4$ metres per second for
$0 \le t \le 3$. A student reports the \emph{total distance travelled} as
$\displaystyle\int_0^3 (t^2 - 4)\,dt = -3$ metres.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Describe the error.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Find $\displaystyle\int_0^2 (4 - t^2)\,dt$.
    \par\vspace*{2.0cm}\ansline
  \item[(c)] Find $\displaystyle\int_2^3 (t^2 - 4)\,dt$.
    \par\vspace*{2.0cm}\ansline
  \item[(d)] Give the total distance travelled.
    \par\vspace*{2.0cm}\ansline
\end{itemize}}

\problem{Asked for $\displaystyle\int_0^{\sqrt{\pi/2}} x\cos\left(x^2\right)dx$,
a student substitutes $u = x^2$, replaces $x\,dx$ by $du$, and reports the
value $1$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Give the corrected value.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{Asked for the \emph{area} between $y = x^3$ and the $x$-axis on
$[-2, 2]$, a student computes $\displaystyle\int_{-2}^{2} x^3\,dx = 0$ and
concludes the area is $0$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Describe the error.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Give the corrected area.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{Asked for $\displaystyle\int_1^4 \frac{1}{\sqrt{x}}\,dx$, a student
substitutes $u = \sqrt{x}$, so $du = \dfrac{1}{2\sqrt{x}}\,dx$, and writes
$\displaystyle\int_1^4 2\,du = 2(4 - 1) = 6$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Give the corrected value.
    \par\vspace*{2.4cm}\ansline
  \item[(c)] Describe the habit that prevents this error from happening again.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}
