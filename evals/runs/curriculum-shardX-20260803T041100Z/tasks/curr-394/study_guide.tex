\sstitleblock{Counting Steps, Finding Ratios, and Knowing When a Sum Exists}

\skillheading{Skill 1 --- Count steps, not terms}

\begin{formulabox}
\textbf{Arithmetic:} $a_n = a_1 + (n-1)d$. \qquad
\textbf{Geometric:} $a_n = a_1 r^{\,n-1}$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The first term costs no steps, so reaching term $n$ takes $n-1$ of them.
  \item \textbf{The $n=1$ test:} substitute $n = 1$. A correct formula returns
        $a_1$ exactly; a formula using $nd$ returns the second term instead.
  \item Sum of $n$ arithmetic terms: $S_n = \frac{n}{2}(a_1 + a_n)$ --- and
        $a_n$ here must be the $n$th term, not the one after it.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad An arithmetic sequence has $a_1 = 5$ and $d = 3$. Find the
$20$th term.
\step{The formula that jumps straight to a term is the explicit one, and its
whole risk is the step count, so write $(n-1)$ before substituting anything.}
\step{$a_{20} = 5 + (20-1)(3) = 5 + 57$}

The $20$th term is $\ans{62}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad An arithmetic sequence has $a_1 = 8$ and $d = 6$. Find the
$15$th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{92}$}
\end{tryitbox}

\skillheading{Skill 2 --- Ratios divide, and only some sums exist}

\begin{formulabox}
\textbf{Common ratio:} $r = \dfrac{a_{k+1}}{a_k}$ --- divide, never subtract, and
confirm on a second pair before trusting it.\par
\smallskip
\textbf{Infinite sum:} $S_\infty = \dfrac{a_1}{1-r}$, valid \emph{only} when
$|r| < 1$. The denominator is $1-r$; dividing by $r$ instead is the most common
mis-copy. If $|r| \geq 1$ the series diverges and no sum exists.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the sum of the infinite geometric series with
$a_1 = 12$ and $r = \frac{1}{4}$.
\step{Check convergence first: $\left|\frac14\right| < 1$, so the sum exists and
the formula may be used at all.}
\step{$S_\infty = \dfrac{12}{1 - \frac14} = \dfrac{12}{\frac34}$}

The sum is $\ans{16}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the sum of the infinite geometric series with
$a_1 = 20$ and $r = \frac{1}{5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25}$}
\end{tryitbox}

\skillheading{Skill 3 --- Which $k$ does a binomial term want?}

\begin{formulabox}
\textbf{Binomial theorem:} $(x+p)^n = \sum \binom{n}{k}x^{\,n-k}p^{\,k}$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $k$ counts the \emph{constant's} power, not $x$'s. For the $x^m$ term,
        solve $n - k = m$, so $k = n - m$.
  \item The two exponents always add to $n$ --- check every term against that.
  \item $\binom{n}{k} = \binom{n}{n-k}$, so the coefficient alone will not tell
        you that $k$ is wrong. Only the power of $p$ will.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the coefficient of $x^3$ in $(x+2)^5$.
\step{Set the index from the exponent that is missing: $n - k = 3$ with $n = 5$
forces $k = 2$, so the constant is squared, not cubed.}
\step{$\binom{5}{2}x^3 2^2 = 10 \cdot x^3 \cdot 4$}

The coefficient is $\ans{40}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the coefficient of $x^4$ in $(x+2)^6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{60}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An off-by-one index usually produces a clean, plausible
number --- that is why it survives. Test at $n=1$, and substitute your answer
back into the original sequence before you trust it.
\end{watchoutbox}
