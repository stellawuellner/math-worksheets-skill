\wstitleblock{Evaluating Functions from Rules, Tables, and Graphs}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Substituting into rules, reading tables, reading graphs, and comparing outputs}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $f(3)$ means ``the output of $f$ when the input is $3$''. It is never $f$ multiplied by $3$.
  \item Show the substitution you made, or point to the table row or plotted point you read.
  \item In Part B every problem has its own graph. Read values only from the marked points on that graph.
\end{itemize}
