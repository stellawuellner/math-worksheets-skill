\problem{Solve $2\cos\theta + 1 = 0$ for $0^\circ \le \theta < 360^\circ$.}

\textbf{Solution:} isolate the cosine, then use the reference angle and the two quadrants where cosine is negative.
\[ \cos\theta = -\frac{1}{2}, \qquad \text{reference angle } 60^\circ, \qquad \text{quadrants II and III}. \]
So $\theta = 180^\circ - 60^\circ = 120^\circ$ and $\theta = 180^\circ + 60^\circ = 240^\circ$. \ans{\theta = 120^\circ,\ 240^\circ}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.15cm}

\problem{Solve $\sqrt{2}\,\sin\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.}

\textbf{Solution:} $\sin\theta = \dfrac{1}{\sqrt{2}} = \dfrac{\sqrt{2}}{2}$, whose reference angle is $45^\circ$. Sine is positive in quadrants I and II, so
$\theta = 45^\circ$ and $\theta = 180^\circ - 45^\circ = 135^\circ$. \ans{\theta = 45^\circ,\ 135^\circ}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.15cm}

\problem{Solve $\sin\theta\,(2\sin\theta - 1) = 0$ for $0^\circ \le \theta < 360^\circ$.}

\textbf{Solution:} the product is zero exactly when one factor is zero, so treat two equations.
\begin{align*}
  \sin\theta = 0 \quad&\Longrightarrow\quad \theta = 0^\circ,\ 180^\circ \\
  2\sin\theta - 1 = 0 \ \Longrightarrow\ \sin\theta = \tfrac{1}{2} \quad&\Longrightarrow\quad \theta = 30^\circ,\ 150^\circ
\end{align*}
Note that $\theta = 360^\circ$ is excluded by the half-open interval. \ans{\theta = 0^\circ,\ 30^\circ,\ 150^\circ,\ 180^\circ}

\problem{Cam: solve $2\sin 2\theta - \sqrt{3} = 0$ for $0^\circ \le \theta < 360^\circ$.}

\textbf{Solution:} let $u = 2\theta$. As $\theta$ runs over $[0^\circ, 360^\circ)$, $u$ runs over $[0^\circ, 720^\circ)$ --- widen the interval \emph{before} solving, or half the answers are lost.
\[ \sin u = \frac{\sqrt{3}}{2} \quad\Longrightarrow\quad u = 60^\circ,\ 120^\circ,\ 420^\circ,\ 480^\circ. \]
Dividing each by $2$: $\theta = 30^\circ, 60^\circ, 210^\circ, 240^\circ$. All four lie in the required interval. \ans{\theta = 30^\circ,\ 60^\circ,\ 210^\circ,\ 240^\circ}

\problem{$2\sin^2\theta + 3\cos\theta - 3 = 0$ on $0^\circ \le \theta < 360^\circ$. (a) Factor. (b) Solve.}

\textbf{(a)} The equation mixes $\sin^2$ with $\cos$, so replace $\sin^2\theta$ by $1 - \cos^2\theta$ to leave one function:
\begin{align*}
  2(1 - \cos^2\theta) + 3\cos\theta - 3 &= -2\cos^2\theta + 3\cos\theta - 1 \\
  &= -(2\cos^2\theta - 3\cos\theta + 1) = (1 - \cos\theta)(2\cos\theta - 1).
\end{align*}
\ans{(1 - \cos\theta)(2\cos\theta - 1)}

\textbf{(b)} Set each factor to zero: $\cos\theta = 1$ gives $\theta = 0^\circ$; $\cos\theta = \dfrac{1}{2}$ gives $\theta = 60^\circ$ and $\theta = 300^\circ$. \ans{\theta = 0^\circ,\ 60^\circ,\ 300^\circ}

\problem{Buoy: $h(t) = 6\sin\!\left(\dfrac{\pi t}{6}\right)$ metres. (a) Times with $h = 3$ on $0 \le t < 12$. (b) Interpret them.}

\textbf{(a)} $6\sin\!\left(\dfrac{\pi t}{6}\right) = 3$, so $\sin\!\left(\dfrac{\pi t}{6}\right) = \dfrac{1}{2}$. As $t$ runs over $[0, 12)$ the inside angle runs over $[0, 2\pi)$ exactly once, and sine equals $\dfrac{1}{2}$ there at $\dfrac{\pi}{6}$ and $\dfrac{5\pi}{6}$:
\[ \frac{\pi t}{6} = \frac{\pi}{6} \ \Rightarrow\ t = 1, \qquad \frac{\pi t}{6} = \frac{5\pi}{6} \ \Rightarrow\ t = 5. \]
\ans{t = 1,\ 5\text{ hours}}

\textbf{(b)} \emph{Instructor-judged.} A correct response interprets rather than restates: the buoy passes $3$ metres twice per cycle, once rising (at $t = 1$) and once falling (at $t = 5$), because the height peaks at $6$ metres at $t = 3$ and returns. Credit an answer that says which crossing is rising and which is falling; repeating the two numbers is not enough.

\problem{Solve $2\cos 3\theta - 1 = 0$ for $0^\circ \le \theta < 180^\circ$.}

\textbf{Solution:} let $u = 3\theta$, so $u$ runs over $[0^\circ, 540^\circ)$.
\[ \cos u = \frac{1}{2} \quad\Longrightarrow\quad u = 60^\circ,\ 300^\circ,\ 420^\circ. \]
(The next value, $660^\circ$, is past $540^\circ$.) Dividing by $3$ gives $\theta = 20^\circ, 100^\circ, 140^\circ$. \ans{\theta = 20^\circ,\ 100^\circ,\ 140^\circ}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.15cm}

\problem{$2\cos^2\theta - \sin\theta - 1 = 0$ on $0^\circ \le \theta < 360^\circ$. (a) Factor. (b) Solve.}

\textbf{(a)} Replace $\cos^2\theta$ by $1 - \sin^2\theta$ so only sine remains:
\begin{align*}
  2(1 - \sin^2\theta) - \sin\theta - 1 &= -2\sin^2\theta - \sin\theta + 1 \\
  &= (1 - 2\sin\theta)(\sin\theta + 1).
\end{align*}
\ans{(1 - 2\sin\theta)(\sin\theta + 1)}

\textbf{(b)} $1 - 2\sin\theta = 0$ gives $\sin\theta = \dfrac{1}{2}$, so $\theta = 30^\circ$ and $150^\circ$; $\sin\theta + 1 = 0$ gives $\sin\theta = -1$, so $\theta = 270^\circ$ --- a single solution, since $-1$ is the minimum of sine and is reached once per turn. \ans{\theta = 30^\circ,\ 150^\circ,\ 270^\circ}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.15cm}

\problem{Searchlight: solve $40\sin 2\theta = 20$ for $0^\circ \le \theta < 180^\circ$.}

\textbf{Solution:} $\sin 2\theta = \dfrac{20}{40} = \dfrac{1}{2}$. With $u = 2\theta$, $u$ runs over $[0^\circ, 360^\circ)$, where $\sin u = \dfrac{1}{2}$ at $u = 30^\circ$ and $u = 150^\circ$. Halving: $\theta = 15^\circ$ and $\theta = 75^\circ$. \ans{\theta = 15^\circ,\ 75^\circ}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.15cm}

\problem{$(2\sin\theta - \sqrt{3})(2\sin\theta + \sqrt{3}) = 0$ on $0^\circ \le \theta < 360^\circ$. (a) Solve. (b) Explain the count.}

\textbf{(a)} Each factor gives its own equation.
\begin{align*}
  \sin\theta = \frac{\sqrt{3}}{2} \quad&\Longrightarrow\quad \theta = 60^\circ,\ 120^\circ \\
  \sin\theta = -\frac{\sqrt{3}}{2} \quad&\Longrightarrow\quad \theta = 240^\circ,\ 300^\circ
\end{align*}
\ans{\theta = 60^\circ,\ 120^\circ,\ 240^\circ,\ 300^\circ}

\textbf{(b)} \emph{Instructor-judged.} A correct response explains that the product form allows \emph{two} values of $\sin\theta$, $\dfrac{\sqrt{3}}{2}$ and $-\dfrac{\sqrt{3}}{2}$, and each value is attained in two quadrants during one full turn --- two cases times two quadrants is four angles. The single linear equation permits only the positive value, so it yields two. Full credit requires that two-cases-times-two-quadrants reason; noticing only that one list is longer is not enough.
