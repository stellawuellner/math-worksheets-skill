\wstitleblock{Number Path Subtraction}{Kindergarten--Grade 1}{}

\begin{center}
\textit{Counting back on the path, counting up on the path, and finding the
missing hop}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Every problem has its own number path. Put your finger on the start
        number.
  \item To subtract, hop \textbf{back} --- or start at the smaller number
        and hop \textbf{up}. Count your hops!
  \item You may draw your hops on the path any time it helps.
\end{itemize}

\problem[2.5cm]{Here is the pattern. Start at 7. Hop back 2: you land on 6,
then 5. So $7-2=5$. Now trace it yourself and write the answer.
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.62, line width=0.8pt]
  \foreach \n in {1,...,10} { \draw (\n-1,0) rectangle (\n,1); \node at (\n-0.5,0.5) {\n}; }
\end{tikzpicture}
\end{center}
$7\;-\;2\;=$~\ansblank}
