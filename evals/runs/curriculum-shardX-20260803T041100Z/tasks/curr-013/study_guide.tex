\sstitleblock{Ordering Three Amounts: Least to Greatest}

\skillheading{Skill 1 --- Ordering three numbers}
\begin{formulabox}\textbf{The rule:}\\[4pt]
\textbf{Least} means the smallest amount. \textbf{Greatest} means the largest amount.
When you count up, you say the least number \emph{first} and the greatest number \emph{last}.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Find the least of the three. Write it in the first blank.
  \item Of the two that are left, find the least. Write it in the middle blank.
  \item The last one goes in the third blank.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Put 8, 3 and 6 in order from least to greatest.
\step{These are three plain numbers, so use counting order: whichever is said first when counting up is the least.}
\step{Count up: 3 comes first, then 6, then 8. So the order is $\ans{3, 6, 8}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Put 5, 9 and 2 in order from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2, 5, 9}$}\end{tryitbox}

\skillheading{Skill 2 --- Count each group, then order}
\begin{formulabox}\textbf{The rule:}\\[4pt]
When the amounts are shown as pictures (dots, ten-frames, cube trains), you cannot
compare them until every group has a \emph{number}. Count each group, write its number
next to the picture, and only then put the numbers in order.
A full ten-frame is 10; count on from 10 for the extra counters.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad One full ten-frame and 3 extra counters; a group of 6; a full ten-frame.
\step{Nothing can be ordered yet --- turn each picture into a number first.}
\step{Full frame and 3 more: count on 11, 12, 13. The others are 6 and 10. In counting order: $\ans{6, 10, 13}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad One full ten-frame and 2 extra counters; a group of 5; a group of 8.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5, 8, 12}$}\end{tryitbox}

\skillheading{Skill 3 --- Work it out, then order}
\begin{formulabox}\textbf{The rule:}\\[4pt]
Sometimes one amount is hidden in the story: ``3 more than'' or ``scored 5 more points''.
Do that addition \textbf{first}, so you have three real numbers, and then order them.
Never order the numbers the story happens to say --- order the amounts each person really has.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Ana has 4 shells and finds 3 more. Ben has 5 shells. Cal has 9 shells.
\step{Ana's amount is hidden, so work out Ana's total before comparing anybody.}
\step{Ana has $4+3=7$ shells. Now order 7, 5 and 9: $\ans{5, 7, 9}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Ana has 6 marbles and gets 2 more. Ben has 4 marbles. Cal has 10 marbles.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4, 8, 10}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} 14 and 41 use the same two digits but they
are not the same amount. Look at the \textbf{tens} digit first: 14 has 1 ten, 41 has 4 tens,
so 14 is the smaller number.
\end{watchoutbox}
