\sstitleblock{Product, Quotient, and Power Rules for Logarithms}

\skillheading{Skill 1 --- The product rule}

\begin{formulabox}
\textbf{Rule:} $\log_b(MN) = \log_b M + \log_b N$, for the same base $b$ and
positive $M$, $N$. Read left to right it expands a product; read right to left it
condenses a sum into one logarithm.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Give the exact value of $\log_6 4 + \log_6 9$.
\step{Neither logarithm is a whole number on its own, but the bases match, so
condensing with the product rule may produce a friendly input.}
\step{$\log_6 4 + \log_6 9 = \log_6(4 \cdot 9) = \log_6 36$, and $6^2 = 36$.}

$\Rightarrow\quad \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Give the exact value of $\log_2 16 + \log_2 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 2 --- The quotient rule}

\begin{formulabox}
\textbf{Rule:} $\log_b\!\dfrac{M}{N} = \log_b M - \log_b N$. The numerator's
logarithm comes first; subtracting in the wrong order gives the negative of the
right answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Give the exact value of $\log_3 108 - \log_3 4$.
\step{A difference of same-base logarithms condenses to one logarithm of a
quotient, and dividing first is what turns an awkward input into a power of the
base.}
\step{$\log_3\!\dfrac{108}{4} = \log_3 27$, and $3^3 = 27$.}

$\Rightarrow\quad \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Give the exact value of $\log_5 250 - \log_5 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- The power rule}

\begin{formulabox}
\textbf{Rule:} $\log_b\!\left(M^{\,p}\right) = p\log_b M$. An exponent inside
comes out in front; a coefficient in front goes back inside as an exponent. A
coefficient of $\tfrac12$ is a square root, and a negative coefficient is a
reciprocal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $\log_b\!\left(x^4\right)$ so no exponent is left
inside.
\step{The input is a single power, so the power rule alone does the whole job ---
no product or quotient is involved.}
\step{The exponent $4$ moves out in front of the logarithm.}

$\Rightarrow\quad \ans{4\log_b x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $\log_b\!\left(y^7\right)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7\log_b y}$}
\end{tryitbox}

\skillheading{Skill 4 --- Using all three together}

\begin{formulabox}
\textbf{Order that works:} to \emph{condense}, move every coefficient inside with
the power rule first, then combine sums into products and differences into
quotients. To \emph{expand}, split the product or quotient first, then bring each
exponent out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Condense $2\log_b x + \log_b 3 - \log_b y$.
\step{Coefficients must go inside before anything can be combined, because the
product and quotient rules only apply to plain logarithms.}
\step{$2\log_b x = \log_b\!\left(x^2\right)$, so the expression becomes
$\log_b\!\left(x^2\right) + \log_b 3 - \log_b y$, which condenses to one
logarithm.}

$\Rightarrow\quad \ans{\log_b\!\dfrac{3x^2}{y}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Condense $\log_b 5 + 3\log_b x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\log_b\!\left(5x^3\right)}$}
\end{tryitbox}
