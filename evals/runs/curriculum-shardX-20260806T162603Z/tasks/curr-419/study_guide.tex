\sstitleblock{Degree--Radian and Quadrant--Sign Errors}

\skillheading{Skill 1 --- Choosing the conversion factor}

\begin{formulabox}
\textbf{One fact, two uses:} $180^\circ = \pi$ radians.\par
degrees $\to$ radians: multiply by $\dfrac{\pi}{180}$. \quad radians $\to$ degrees: multiply by $\dfrac{180}{\pi}$.\par
\textbf{Self-check:} converting \emph{from} radians, the $\pi$ must cancel. A $\pi$ still standing means the factor was upside down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert $135^\circ$ to radians.
\step{Starting in degrees and wanting radians, so the factor is the one that puts $\pi$ on top.}
\step{$135 \cdot \dfrac{\pi}{180} = \dfrac{135\pi}{180} = \dfrac{3\pi}{4}$}
$\Rightarrow\quad 135^\circ = \ans{\tfrac{3\pi}{4}}$ radians.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $300^\circ$ to radians.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5\pi}{3}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Attaching the sign from the quadrant}

\begin{formulabox}
\textbf{ASTC (``All Students Take Calculus''):} in quadrant I \emph{all} are positive; in II only \emph{sine}; in III only \emph{tangent}; in IV only \emph{cosine}.\par
\textbf{Order of work:} find the reference angle, read its value from the first quadrant, then attach the sign the quadrant demands. The quadrant never changes \emph{which} ratio you read.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\cos 225^\circ$ exactly.
\step{Want cosine of an angle past $180^\circ$, so find the reference angle first and let the quadrant supply the sign.}
\step{$225 - 180 = 45$, so the reference angle is $45^\circ$ and the angle is in quadrant III, where cosine is negative.}
$\Rightarrow\quad \cos 225^\circ = -\cos 45^\circ = \ans{-\tfrac{\sqrt{2}}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\sin 300^\circ$ exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\tfrac{\sqrt{3}}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an inverse-trig key returns one angle only --- $\sin^{-1}$ and $\tan^{-1}$ from $-90^\circ$ to $90^\circ$, $\cos^{-1}$ from $0^\circ$ to $180^\circ$. On a $360^\circ$ interval there are usually \emph{two} answers, and the calculator's may not even be one of them.
\end{watchoutbox}

\skillheading{Skill 3 --- Reading the reference angle's value}

\begin{formulabox}
\textbf{The three you must know:}\par
$\sin 30^\circ = \dfrac{1}{2}$, \ $\cos 30^\circ = \dfrac{\sqrt{3}}{2}$; \quad $\sin 45^\circ = \cos 45^\circ = \dfrac{\sqrt{2}}{2}$; \quad $\sin 60^\circ = \dfrac{\sqrt{3}}{2}$, \ $\cos 60^\circ = \dfrac{1}{2}$.\par
Tangent is the quotient: $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\cos\dfrac{4\pi}{3}$ exactly.
\step{Want cosine, so I need the cosine of the reference angle --- the sine value is the classic wrong pick here.}
\step{$\dfrac{4\pi}{3} = \pi + \dfrac{\pi}{3}$: reference angle $\dfrac{\pi}{3}$, quadrant III, cosine negative, and $\cos\dfrac{\pi}{3} = \dfrac{1}{2}$.}
$\Rightarrow\quad \cos\dfrac{4\pi}{3} = \ans{-\tfrac{1}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\tan\dfrac{5\pi}{3}$ exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\sqrt{3}}$}
\end{tryitbox}
