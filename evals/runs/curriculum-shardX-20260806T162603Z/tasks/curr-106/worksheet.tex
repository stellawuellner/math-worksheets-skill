\wstitleblock{Area Models for Two-Digit Multiplication}{Grades 4--5 Mathematics}{}

\begin{center}\textit{Splitting factors, finding partial products, adding to
the total, and estimating to check}\end{center}

\noindent\textbf{Directions:} An area model splits each factor into tens and
ones, so one big multiplication becomes four easy cells. Labeled models are
given in the first problems; later problems give you an empty model to label
and fill in yourself.

\problem[2.5cm]{The model shows $14 \times 23$, with 14 split into $10 + 4$ and
23 split into $20 + 3$. Find each cell, then add:

\begin{center}
\begin{tikzpicture}[scale=1.0]
  \draw[thick] (0,0) rectangle (5.2,2.6);
  \draw[thick] (3.8,0) -- (3.8,2.6);
  \draw[thick] (0,0.9) -- (5.2,0.9);
  \node[above] at (1.9,2.6) {$20$};
  \node[above] at (4.5,2.6) {$3$};
  \node[left] at (0,1.75) {$10$};
  \node[left] at (0,0.45) {$4$};
\end{tikzpicture}
\end{center}

$10 \times 20 = $~\ansblank\quad $10 \times 3 = $~\ansblank\quad
$4 \times 20 = $~\ansblank\quad $4 \times 3 = $~\ansblank
\par\vspace{4pt}\noindent
