\wstitleblock{Building Bigger Shapes from Smaller Ones}{Kindergarten--Grade 1}{}

\begin{center}
\small counting the small squares in a composed figure, naming the new shape, comparing two builds, and finishing a rectangle
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Small shapes can be fitted together to build a bigger shape. The pieces must not overlap and must leave no gaps.
  \item The thin lines show the small squares. The thick line shows the outside of the built shape.
  \item Touch each small square once as you count, so none is counted twice.
\end{itemize}
