\wstitleblock{Ten-Frames: How Many Do You See?}{Kindergarten--Grade 1 \quad Counting and Cardinality}{}

\begin{center}
\textit{Telling how many a ten-frame shows, finding how many more make ten,
and comparing two ten-frames}
\end{center}

\noindent\textbf{How to read a ten-frame.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A ten-frame has two rows of five boxes. A full row is \textbf{five}. A
full frame is \textbf{ten}.
\item When the top row is full, you do not have to start at one. Say
``five'', then count on for the dots below: six, seven, \ldots
\item Look before you count. Check whether the top row really is full ---
sometimes it is not.
\end{itemize}

\problem[3cm]{\large The top row is full. Say ``five'', then count on.
How many dots does this ten-frame show?
\begin{center}\tenframe{5}{1}\end{center}}

\problem[3cm]{\large How many dots does this ten-frame show?
\begin{center}\tenframe{5}{4}\end{center}}

\problem[3cm]{\large Neither row is full this time. Count the top row, then keep
counting the bottom row. How many dots in all?
\begin{center}\tenframe{4}{4}\end{center}}

\problem[3cm]{\large How many boxes are still \textbf{empty} in this ten-frame?
\begin{center}\tenframe{5}{2}\end{center}}

\problem[3cm]{\large How many dots does this ten-frame show?
\begin{center}\tenframe{5}{5}\end{center}}

\problem[4cm]{\large Look at both ten-frames. Count each one. Then write
$>$ or $<$ in the blank so the sentence is true.
\begin{center}
\begin{tabular}{cc}
Frame A & Frame B \\[0.15cm]
\tenframe{5}{2} & \tenframe{5}{4} \\
\end{tabular}
\end{center}
\hspace{1cm}\large Frame A \underline{\hspace{1.2cm}} Frame B}

\problem[3cm]{\large How many more dots are needed to fill this ten-frame?
\begin{center}\tenframe{2}{2}\end{center}}

\problem[3cm]{\large Careful --- the top row is not full. How many dots does this
ten-frame show?
\begin{center}\tenframe{3}{2}\end{center}}

\problem[4cm]{\large Count both ten-frames. Then write $>$, $<$ or $=$ in the blank

\tenframe{4}{4} & \tenframe{5}{3} \\

\hspace{1cm}\large Frame A \underline{\hspace{1.2cm}} Frame B}

\problem[4cm]{\large This time count the \textbf{empty} boxes in each frame. Which

\begin{tabular}{cc}

\tenframe{5}{1} & \tenframe{5}{4} \\
\end{tabular}
\end{center}
\hspace{0.4cm}\large empty boxes in A \underline{\hspace{1.2cm}} empty boxes in B}
