\sstitleblock{Generating and Comparing Two Patterns}

\skillheading{Skill 1 --- Generating a term without listing them all}

\begin{formulabox}
\textbf{Rule:} The first term is the start, so reaching term $n$ takes $n - 1$ steps.\\[3pt]
term $n$ $=$ start $+$ step size $\times (n - 1)$\\[3pt]
Off-by-one is the usual slip here: term 6 is 5 steps out, not 6.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at 3 and adds 5 each time. Write the 7th term.
\step{Listing seven terms works but invites a miscount, so I count steps instead: the 7th term is six steps past the start.}
\step{$3 + 5 \times 6 = 3 + 30$, and listing agrees: 3, 8, 13, 18, 23, 28, 33.}
7th term: $\ans{33}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at 2 and adds 6 each time. Write the 5th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{26}$}
\end{tryitbox}

\skillheading{Skill 2 --- Relating one rule to the other}

\begin{formulabox}
\textbf{Rule:} When two patterns start at the same place, compare their \emph{step sizes}, not their terms.\\[3pt]
If both start at 0 and one step is $k$ times the other, then every matching term is $k$ times the other as well --- after the same number of steps, one pattern has added $k$ times as much.\\[3pt]
This fails if the two patterns start at different places, because the head start never gets multiplied.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Pattern C starts at 0 and adds 3. Pattern D starts at 0 and adds 9. Each D term is how many times the matching C term?
\step{Both start at 0, so the whole relationship is carried by the steps and I can compare the step sizes alone.}
\step{$9 \div 3 = 3$, and checking term 4: C gives 9 and D gives 27.}
Multiplier: $\ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One pattern starts at 0 and adds 5; another starts at 0 and adds 20. Each term of the second is how many times the first?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing matching terms}

\begin{formulabox}
\textbf{Rule:} To compare term $n$ of two patterns, generate both and subtract.\\[3pt]
gap $=$ (start$_1$ $+$ step$_1 \times (n-1)$) $-$ (start$_2$ $+$ step$_2 \times (n-1)$)\\[3pt]
A pattern that begins behind can still finish ahead: the gap closes by (step$_1 -$ step$_2$) every step.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Pattern E starts at 10 and adds 2. Pattern F starts at 4 and adds 4. How much greater is F's 8th term than E's 8th term?
\step{The starts differ, so a step-size ratio will not answer this --- I build both 8th terms and subtract.}
\step{F: $4 + 4 \times 7 = 32$. \ E: $10 + 2 \times 7 = 24$. \ Gap: $32 - 24$.}
F is greater by $\ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Pattern G starts at 6 and adds 3; pattern H starts at 1 and adds 5. How much greater is H's 9th term than G's 9th term?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

\skillheading{Skill 4 --- Writing the two patterns as ordered pairs}

\begin{formulabox}
\textbf{Rule:} Pair the terms that share a term number: (term $n$ of the first pattern, term $n$ of the second).\\[3pt]
The pattern named first supplies the across number. Plotted on a grid, matching terms of two constant-step patterns always fall on a straight line through the pairs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Pattern J starts at 0 and adds 2. Pattern K starts at 0 and adds 5. Write the ordered pair for term 6.
\step{The question names J first, so J's term is the across number and K's is the up number --- I generate each to term 6 separately.}
\step{J: $0 + 2 \times 5 = 10$. \ K: $0 + 5 \times 5 = 25$.}
Ordered pair: $(\ans{10}, \ans{25})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Pattern L starts at 0 and adds 3; pattern M starts at 0 and adds 7. Write the ordered pair for term 5.\\[2pt]
\rotatebox{180}{\footnotesize check: $(\ans{12}, \ans{28})$}
\end{tryitbox}
