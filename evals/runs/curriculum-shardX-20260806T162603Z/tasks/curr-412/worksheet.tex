\wstitleblock{Solving Logarithmic Equations with Domain Checks}{Precalculus}{}

\begin{center}
\small\textit{Converting to exponential form, equating arguments, and condensing logarithms with a domain check}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Every logarithm needs a \emph{positive} argument. A value that makes any argument zero or negative is not a solution, however cleanly it came out of the algebra.
  \item Show the exponential form (or the condensed single logarithm) before you solve.
  \item Give exact answers; leave fractions as fractions.
\end{itemize}
\vspace{0.15cm}

\problem[4.5cm]{Solve for $x$: \qquad $\log_{2} x = 5$}

\problem[4.5cm]{Solve for $x$: \qquad $\log_{6}(4x) = \log_{6} 20$}

\problem[5cm]{Solve for $x$: \qquad $\log_{3} x + \log_{3} 4 = \log_{3} 24$}

\problem[5cm]{Solve for $x$: \qquad $\log_{3}(x - 1) = 4$}

\problem[5cm]{Solve for $x$: \qquad $\log_{5}(2x - 3) = \log_{5}(x + 4)$}

\problem[0pt]{Consider \qquad $\log x + \log(x - 3) = 1$ \qquad (base $10$).
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Before solving, write the set of $x$-values for which \emph{both} logarithms are defined. Give your answer as an inequality.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Now solve the equation, keeping only the value your part (a) allows.
  \par\vspace*{3.4cm}\ansline
\end{itemize}}

\problem[5cm]{Solve for $x$: \qquad $\log_{1/2} x = -3$}

\problem[6cm]{Solve for $x$: \qquad $\log_{6}\!\left(x^{2} + 5x\right) = \log_{6} 24$ \\[2pt]
\small (Test every candidate in the original equation --- a negative candidate is not automatically wrong here.)}

\problem[0pt]{Consider \qquad $\log_{4}(x + 6) - \log_{4} x = 1$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Write the inequality that both arguments force on $x$. Give the single inequality that satisfies both.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Condense the left side with the quotient rule and solve.
  \par\vspace*{3.4cm}\ansline
\end{itemize}}

\problem[5.5cm]{Solve for $x$: \qquad $\log_{9} x = \dfrac{3}{2}$}

\problem[0pt]{Consider \qquad $\log_{3} x + \log_{3}(x - 8) = 2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Write the inequality that both arguments force on $x$.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Condense, convert to exponential form, and solve.
  \par\vspace*{3.6cm}\ansline
\end{itemize}}

\problem[0pt]{Consider \qquad $\ln(x + 3) + \ln(x - 3) = \ln 16$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Write the inequality that both arguments force on $x$.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Condense both sides to a single natural logarithm and solve.
  \par\vspace*{3.6cm}\ansline
\end{itemize}}
