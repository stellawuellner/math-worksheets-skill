\wstitleblock{Powers of Ten and Place Value}{Grades 4--5 Mathematics}{}

\begin{center}\textit{Multiplying and dividing by powers of ten, exponent form,
and place-value shifts}\end{center}

\noindent\textbf{Directions:} Problem 1 shows a worked pattern to get you
started. After that, work each problem on your own --- watch whether you are
multiplying or dividing, and by which power of ten.

\problem[2.5cm]{Complete the pattern. The first line is done for you:
\par\vspace{2pt}\noindent
\hspace*{1cm}$7 \times 10 = 70$
\par\vspace{2pt}\noindent
\hspace*{1cm}$7 \times 100 = $~\ansblank
\par\vspace{2pt}\noindent
\hspace*{1cm}$7 \times 1{,}000 = $~\ansblank}

\problem[2.5cm]{$26 \times 10 = $~\ansblank}

\problem[2.5cm]{$40 \times 1{,}000 = $~\ansblank}

\problem[2.5cm]{$10^3 = 10 \times 10 \times 10 = $~\ansblank}

\problem[2.5cm]{$3{,}500 \div 10 = $~\ansblank}

\problem[2.5cm]{$62{,}000 \div 100 = $~\ansblank}

\problem[3cm]{$45 \times 10^2 = $~\ansblank}

\problem[3cm]{$8{,}000 \div 10^3 = $~\ansblank}

\problem[3.5cm]{In $66{,}000$, the 6 in the ten-thousands place is how many
\emph{times} the value of the 6 in the thousands place?~\ansblank
\par\vspace{2pt}\noindent
Explain how you know.}

\problem[3.5cm]{Compare the two 5s:
\par\vspace{2pt}\noindent
The digit 5 in $52{,}600$ has a value of~\ansblank.
\par\vspace{2pt}\noindent
The digit 5 in $5{,}260$ has a value of~\ansblank.
\par\vspace{2pt}\noindent
The first value is~\ansblank~times the second value.}

\problem[3.5cm]{How many times as large as $10^2$ is $10^5$?}

\problem[4.5cm]{One box holds 10 markers. One case holds 100 boxes. How many
markers are in 6 cases?}
