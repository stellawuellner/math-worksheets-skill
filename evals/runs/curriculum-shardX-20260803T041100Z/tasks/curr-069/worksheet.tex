\wstitleblock{Reading Arrays: Rows, Groups, and the Total}{Grade 2--3 Multiplication}{}

\noindent\textbf{Three different questions --- read carefully which one is asked:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{How many rows?} $\rightarrow$ count the rows going down the side.
  \item \textbf{How many in each row?} $\rightarrow$ count across just one row.
  \item \textbf{How many in all (the total)?} $\rightarrow$ rows $\times$ in each row.
\end{itemize}
\noindent The total is almost always \emph{bigger} than the number of rows.
If your total equals the number of rows, you answered the wrong question.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3cm]{Look at this array of dots.
\begin{center}\large
\begin{tabular}{ccccc}
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[2pt]
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[2pt]
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$
\end{tabular}
\end{center}
There are \textbf{3} rows with \textbf{5} dots in each row.
How many dots are there \textbf{in all}?}

\problem[3.5cm]{This array has \textbf{4} rows with \textbf{6} squares in each row.
\begin{center}\large
\begin{tabular}{cccccc}
$\square$ & $\square$ & $\square$ & $\square$ & $\square$ & $\square$ \\[2pt]
$\square$ & $\square$ & $\square$ & $\square$ & $\square$ & $\square$ \\[2pt]
$\square$ & $\square$ & $\square$ & $\square$ & $\square$ & $\square$ \\[2pt]
$\square$ & $\square$ & $\square$ & $\square$ & $\square$ & $\square$
\end{tabular}
\end{center}
Write the multiplication sentence for the picture, then find the
\textbf{total number of squares}.}

\problem[3.5cm]{A garden has \textbf{6} rows of carrots with \textbf{3} carrots in
each row.

\noindent Circle the words that tell you the \emph{total}, then find how many
carrots are in the garden altogether.}

\problem[4cm]{\textbf{Find and fix the mistake.} Jonah's array has \textbf{5} rows
with \textbf{4} stickers in each row.
\begin{center}\large
\begin{tabular}{cccc}
$\star$ & $\star$ & $\star$ & $\star$ \\[2pt]
$\star$ & $\star$ & $\star$ & $\star$ \\[2pt]
$\star$ & $\star$ & $\star$ & $\star$ \\[2pt]
$\star$ & $\star$ & $\star$ & $\star$ \\[2pt]
$\star$ & $\star$ & $\star$ & $\star$
\end{tabular}
\end{center}
Jonah writes: ``There are \textbf{5} stickers in all, because I count 5 rows.''

\noindent What did Jonah count instead of the total? Write the correct total.}

\problem[4cm]{\textbf{Find and fix the mistake.} Mia sets out \textbf{2} rows of
shells with \textbf{8} shells in each row. She writes:
$2 + 8 = \textbf{10}$ shells in all.

\noindent Which operation should Mia have used, and what is the correct total
number of shells?}

\problem[4cm]{A baker puts \textbf{28} muffins into \textbf{4} equal rows.

\noindent How many muffins are in \textbf{each row}? (This is \emph{not} 28 --- that
is the total for the whole tray.)}

\problem[4cm]{Ella lines up \textbf{30} beads in rows with \textbf{5} beads in
each row.

\noindent How many \textbf{rows} of beads does she make?}

\problem[4.5cm]{\textbf{Challenge.} A checkerboard game uses \textbf{36} counters
arranged in equal rows of \textbf{9} counters each. Sam says, ``There must be 36
rows, because there are 36 counters.''

\noindent Explain in one sentence what Sam mixed up, then find the real number of
rows.}
