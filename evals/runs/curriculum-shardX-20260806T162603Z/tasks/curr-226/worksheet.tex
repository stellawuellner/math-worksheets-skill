\wstitleblock{Where Two Lines Meet}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{The point that makes both equations true at once}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item A \emph{solution of a system} is a pair of values that makes
        \emph{every} equation true --- not just one of them.
  \item On a graph that pair is the point where the two lines cross. In a
        table it is the row the two tables share.
  \item Write every answer as an ordered pair, and say what the two numbers
        stand for when the problem comes from a story.
\end{itemize}

\problem[3.5cm]{Find the pair of values that makes \emph{both} of these
equations true: $x + y = 5$ and $x - y = 3$. Write your answer as an ordered
pair.}

\problem[3.5cm]{Find the pair of values that makes both of these equations
true: $2x + y = 11$ and $x - y = 1$. Write your answer as an ordered pair.}

\problem{Lines $A$ and $B$ in the Graph Bank are the graphs of $y = x + 1$ and
$y = 7 - x$.
\begin{itemize}
  \item[(a)] Write the coordinates of the point where the two lines cross.
        \par\vspace*{1.9cm}\ansline
  \item[(b)] Explain what that point means for the two equations.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{Two bike-rental plans are shown in the tables below. Plan A follows
$y = 4h + 8$ dollars and plan B follows $y = 2h + 16$ dollars, where $h$ is
the number of hours.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Plan A hours} & 1 & 2 & 3 & 4 \\ \hline
\textbf{Plan A cost (\$)} & 12 & 16 & 20 & 24 \\ \hline
\end{tabular}
\quad
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Plan B hours} & 1 & 2 & 3 & 4 \\ \hline
\textbf{Plan B cost (\$)} & 18 & 20 & 22 & 24 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What does plan A cost at 4 hours?
        \par\vspace*{1.6cm}\ansline
  \item[(b)] What does plan B cost at 4 hours?
        \par\vspace*{1.6cm}\ansline
  \item[(c)] Write the (hours, cost) pair the two tables share --- the
        solution of the system.
        \par\vspace*{1.6cm}\ansline
\end{itemize}}

\problem{Gym P charges \$25 to join plus \$8 for each class. Gym Q charges
\$40 to join plus \$8 for each class. Their cost equations form a system.
\begin{itemize}
  \item[(a)] Find the number of classes at which the two gyms cost the same,
        or state that there is no such number.
        \par\vspace*{2cm}\ansline
  \item[(b)] Explain what your answer means about the two gyms.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem[4.5cm]{Lines $A$ and $C$ in the Graph Bank are the graphs of
$y = x + 1$ and $y = 2x - 4$. Write the coordinates of the point where they
cross.}

\problem{A student claims that $(3, 2)$ is the solution of the system
$3x + y = 11$ and $x + 2y = 12$.
\begin{itemize}
  \item[(a)] Find the actual solution of the system.
        \par\vspace*{2cm}\ansline
  \item[(b)] Explain what happens when the claimed pair is substituted into
        each equation, and why that settles the question.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{Plan J's data charges are in the table below; its equation is
$y = 6x + 20$ dollars for $x$ gigabytes. Plan K starts at \$32 and follows
$y = 2x + 32$ dollars.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Gigabytes} & 0 & 1 & 2 & 3 \\ \hline
\textbf{Plan J cost (\$)} & 20 & 26 & 32 & 38 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What does plan J cost at 3 gigabytes?
        \par\vspace*{1.8cm}\ansline
  \item[(b)] Write the (gigabytes, cost) pair at which the two plans agree.
        \par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem{Lines $A$ and $D$ in the Graph Bank are the graphs of $y = x + 1$ and
$y = x + 4$.
\begin{itemize}
  \item[(a)] Write the solution of this system, or state that there is none.
        \par\vspace*{1.9cm}\ansline
  \item[(b)] Explain how the Graph Bank shows that your answer to part (a) is
        right.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{A community pool sells single visits for \$5 each, so $v$ visits cost
$y = 5v$ dollars. A membership costs \$30 plus \$2 for each visit, so it costs
$y = 2v + 30$ dollars.
\begin{itemize}
  \item[(a)] Write the (visits, cost) pair at which the two options cost the
        same.
        \par\vspace*{1.9cm}\ansline
  \item[(b)] Explain what that pair means for a swimmer deciding between the
        two options.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}
