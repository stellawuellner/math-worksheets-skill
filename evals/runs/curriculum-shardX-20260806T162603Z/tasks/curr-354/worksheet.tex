\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item Four of these problems show a student's work that contains exactly one
    mistake. For those, do both jobs: work the problem correctly yourself, and
    then say what the student did wrong.
  \item A correction with no diagnosis earns half the marks. Name the step and
    the rule, not just the right answer.
  \item Write every value in standard form $a + bi$, and remember that
    $i^{2} = -1$.
\end{itemize}

\problem{A student wrote:
\par\vspace{0.15cm}
\hspace*{0.8cm}$(8 + 2i) - (5 - 6i) = 8 - 5 + 2i - 6i = 3 - 4i$
\par\vspace{0.15cm}
(a) Work the subtraction correctly.
\par\vspace*{3.2cm}\ansline
\par\vspace{0.15cm}
(b) Name the error in the student's line.
\par\vspace*{2.4cm}\ansline}

\problem[5cm]{Simplify: \quad $\dfrac{10}{1 + 3i}$}

\problem{A student's work for $\dfrac{14 + 8i}{3 - 2i}$:
\par\vspace{0.15cm}
\hspace*{0.8cm}$\dfrac{14 + 8i}{3 - 2i} \cdot \dfrac{3 + 2i}{3 + 2i}
  = \dfrac{26 + 52i}{9 - 4} = \dfrac{26 + 52i}{5}$
\par\vspace{0.15cm}
(a) Work the division correctly.
\par\vspace*{3.4cm}\ansline
\par\vspace{0.15cm}
(b) Name the error. The numerator is right, so say which value is wrong and what it should be.
\par\vspace*{2.4cm}\ansline}

\problem{A student expanded a square like this:
\par\vspace{0.15cm}

\par\vspace{0.15cm}
(a) Expand the square correctly.
\par\vspace*{3.2cm}\ansline
\par\vspace{0.15cm}
(b) Name the error.
\par\vspace*{2.4cm}\ansline}

\problem[6cm]{Simplify: \quad $(3 - 4i) - (-6 + i) + (2 - 5i)$}

\problem{Here is a student's complete solution to $\dfrac{20 + 15i}{2 + i}$.
\par\vspace{0.15cm}
\hspace*{0.8cm}Line 1: \ multiply above and below by $\dfrac{2 - i}{2 - i}$
\par\vspace{0.1cm}
\hspace*{0.8cm}Line 2: \ $= \dfrac{20 + 15i}{(2 + i)(2 - i)} = \dfrac{20 + 15i}{5}$
\par\vspace{0.1cm}
\hspace*{0.8cm}Line 3: \ $= 4 + 3i$
\par\vspace{0.15cm}
(a) Give the corrected quotient.
\par\vspace*{3.6cm}\ansline
\par\vspace{0.15cm}
(b) Say which line is the first one that is wrong, and state the rule it breaks.
\par\vspace*{2.6cm}\ansline}
