\wstitleblock{Graphing Quadratics from Vertex or Standard Form}{Algebra 1}{Vertex form, standard form, and transformations of the parent parabola}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Vertex form} is $y = a(x-h)^2 + k$: the vertex is $(h, k)$, and the sign of $a$ says which way the parabola opens.
  \item \textbf{Standard form} is $y = ax^2 + bx + c$: the axis of symmetry is $x = -\dfrac{b}{2a}$, the $y$-intercept is $c$, and the $x$-intercepts come from solving $y = 0$.
  \item Every part is worth writing down: a graph is only as good as the points you found first.
  \item The last three problems supply a grid. Plot your points, then join them with a \emph{smooth curve} --- a parabola never has a sharp corner.
\end{itemize}

\par\smallskip

\problem[5cm]{$y = -(x+1)^2 + 4$
\par\smallskip
(a) Find the $x$-intercepts. \quad (b) Does this parabola open up or down, and is its vertex a maximum or a minimum? Explain how the equation tells you.}

\problem[5cm]{$y = x^2 - 6x + 5$
\par\smallskip

\par\smallskip

\problem[5cm]{The graph of $y = x^2$ is shifted $3$ units to the left and $2$ units down.
\par\smallskip
(a) Write the equation of the new parabola in vertex form. \quad (b) Find the $y$-intercept of the new parabola.}

\problem[5cm]{The graph of $y = x^2$ is reflected in the $x$-axis and then shifted $9$ units up.
\par\smallskip
