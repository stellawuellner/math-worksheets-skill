\sstitleblock{Using Dilations to Reason About Similarity}

\skillheading{Skill 1 --- Coordinates of a dilated point}
\begin{formulabox}\textbf{Rules:}\\[4pt]
Centre at the origin: $(x, y) \longmapsto (kx, ky)$.\par
Centre at $(h, m)$: $(x, y) \longmapsto \bigl(h + k(x-h),\ m + k(y-m)\bigr)$ --- measure
from the centre, multiply by $k$, step back out.\par
$k > 1$ enlarges, $0 < k < 1$ reduces, and $k = 1$ leaves the figure alone.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad A dilation about the origin with $k = 3$ maps
$P(4, -2)$ to $P'$. Find the $x$-coordinate of $P'$.
\step{The centre is the origin, so each coordinate is just multiplied by $k$; take the $x$-coordinate on its own.}
\step{$3 \times 4 = \ans{12}$ (the $y$-coordinate is handled the same way).}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad A dilation about the origin with $k = 0.5$ maps
$Q(5, -4)$ to $Q'$. Find the $y$-coordinate of $Q'$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$}\end{tryitbox}

\skillheading{Skill 2 --- Scale factor from two lengths}
\begin{formulabox}\textbf{Rule:}\\[4pt]
\[ k = \frac{\text{image length}}{\text{original length}} \]
Image over original, in that order: $k$ answers ``how many times as long is the image?''
Lengths on a grid come from the distance formula.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad A model edge is 12 cm long. On the enlarged
copy the matching edge is 18 cm. Find $k$.
\step{The two edges correspond, so divide the image length by the original length.}
\step{$18 \div 12 = \ans{1.5}$, so every length on the copy is 1.5 times the model's.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad An 8 cm edge becomes 20 cm on the enlarged copy.
Find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.5}$}\end{tryitbox}

\skillheading{Skill 3 --- Slope under a dilation}
\begin{formulabox}\textbf{Rule:}\\[4pt]
A dilation maps a segment to a \textbf{parallel} segment, so the image has the
\emph{same slope}. Equal slopes are the coordinate evidence that the angles did not
change --- which is why the figure and its image are similar, not just the same size.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Find the slope of the segment from $(1, 2)$ to
$(3, 6)$. Its image under $k = 4$ runs from $(4, 8)$ to $(12, 24)$.
\step{Slope is rise over run, and the claim to test is that the image gives the same number.}
\step{$(6-2)/(3-1) = 4/2 = \ans{2}$, and the image gives $(24-8)/(12-4) = 16/8 = 2$ as well.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Find the slope of the segment from $(2, 1)$ to
$(8, 4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.5}$}\end{tryitbox}

\skillheading{Skill 4 --- Similar figures: sides and area}
\begin{formulabox}\textbf{Rules:}\\[4pt]
In similar figures every pair of matching sides has the \emph{same} ratio $k$, so a
missing side is $k \times$ (matching side).\par
Lengths scale by $k$, but area scales by $k^2$: both dimensions grew, so the area grew twice.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad A triangle has sides 5 cm and 7 cm. A similar
triangle has 15 cm matching the 5 cm side. Find the side matching 7 cm.
\step{Matching sides share one ratio, so find $k$ from the pair that is fully known, then apply it to the other side.}
\step{$k = 15 \div 5 = 3$, so the missing side is $3 \times 7 = \ans{21}$ cm.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad A triangle has sides 4 cm and 9 cm. A similar
triangle has 10 cm matching the 4 cm side. Find the side matching 9 cm (in cm).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{22.5}$}\end{tryitbox}

\begin{examplebox}\textbf{Example:}\quad A figure has area 15 square units. It is dilated
with $k = 2$. Find the area of the image.
\step{Area is a product of two lengths and each one is multiplied by $k$, so multiply the area by $k^2$, not by $k$.}
\step{$15 \times 2^2 = 15 \times 4 = \ans{60}$ square units.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad A figure has area 24 square units and is dilated
with $k = 3$. Find the area of the image (in square units).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{216}$}\end{tryitbox}
