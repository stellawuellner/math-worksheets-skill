\wstitleblock{Two-Digit Multiplication with Area Models}{Grade 4--5 --- Number and Operations in Base Ten}{}

\noindent\textbf{Build the area model, estimate the product, find a missing partial product}

\vspace{0.15cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item Each rectangle is split by place value: the tens part and the ones part
        of one factor across the top, the tens part and the ones part of the
        other factor down the side.
  \item Write the product of the row label and the column label inside each
        region. Those four numbers are the \emph{partial products}.
  \item Add the four partial products, then write the multiplication sentence
        on the answer line.
\end{itemize}
