\sstitleblock{Equations with Variables on Both Sides}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Collecting the variable terms}

\begin{formulabox}
\textbf{The balance rule:} whatever is done to one side must be done to the other.
\par\smallskip
\textbf{Order of moves:} clear brackets, take the \emph{smaller} variable term off both
sides, undo the constant, undo the coefficient. Taking the smaller variable term off keeps
the surviving coefficient positive, which is why that choice is worth making.
\par\smallskip
Every move uses the \textbf{inverse} operation: a subtraction is undone by adding.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $5x + 2 = 3x + 10$.
\step{Variables sit on both sides, so nothing can be undone until one side is variable-free --- remove the smaller variable term from both sides first.}
\step{That leaves a two-step equation: subtract the constant from both sides, then divide both sides by the surviving coefficient.}

$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $6x - 1 = 2x + 11$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Building the equation from a situation}

\begin{formulabox}
\textbf{Method:} name the unknown with a letter and say what it counts. Write an
expression for each side --- a fixed amount plus a per-unit amount times the unknown ---
and set them equal, because "the same" is what an equals sign means.
\par\smallskip
Check the answer in the \emph{story}, not only in the equation: an answer must make sense
as a count of classes, hours, or metres.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One fair charges $6$ to enter plus $2$ a ride; another charges $2$
to enter plus $3$ a ride. After how many rides is the cost the same?
\step{The question asks when two totals are equal, so the unknown is the number of rides and the equation sets the two cost expressions equal.}
\step{Each side is an entry charge plus a rate times the rides; taking the smaller rides term off both sides leaves a one-step equation.}

$\Rightarrow\quad \ans{r = 4}$ rides
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One pool charges $10$ plus $2$ a swim; another charges $4$ plus $4$
a swim. After how many swims is the cost equal?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{r = 3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- When the variable disappears}

\begin{formulabox}
\textbf{Two special cases.} Collecting the variable terms sometimes removes them
completely, and what is left decides everything:
\par\smallskip
\textbf{True statement left} (such as $6 = 6$): every number works --- infinitely many
solutions. \par
\textbf{False statement left} (such as $1 = -3$): no number works --- no solution.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3(x + 2) = 3x + 6$, then $4x + 1 = 4x - 3$.
\step{Both have matching variable terms, so I clear brackets and collect, then read whatever numerical statement survives.}
\step{The first distributes to give the two sides identical; the second loses its variable terms and leaves two different numbers claimed equal.}

First: $\ans{3x + 6 = 3x + 6}$, true for every $x$. \quad
Second: $\ans{\text{no solution}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many solutions has $5(x - 2) = 5x - 10$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x - 10 = 5x - 10}$, infinitely many}
\end{tryitbox}
