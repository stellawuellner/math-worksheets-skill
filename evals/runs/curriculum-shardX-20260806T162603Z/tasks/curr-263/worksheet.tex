\wstitleblock{Solving Linear Systems by Elimination}{Algebra 1}{}

\begin{center}
\small Scaling one equation, scaling both equations, and modeling with systems
\end{center}
\vspace{0.15cm}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Solve every system by \textbf{elimination}. Multiply one or both equations by a constant first so that one pair of like terms becomes opposites (or matches).
  \item Show the scaled equations you actually added or subtracted.
  \item Write each answer as an ordered pair, or as a labelled pair of values for the word problems.
  \item Check your solution in \emph{both} original equations before writing it down.
\end{itemize}

\problem[5cm]{Solve by elimination.
\[ 3x + 2y = 16 \qquad\text{and}\qquad 5x - 4y = 12 \]}

\problem[5cm]{Solve by elimination.
\[ 4x + 3y = 18 \qquad\text{and}\qquad 2x - y = 4 \]}

\problem[5.5cm]{Solve by elimination. Neither pair of like terms matches yet, so scale \emph{both} equations.
\[ 3x + 4y = 25 \qquad\text{and}\qquad 2x + 3y = 18 \]}

\problem[0.2cm]{The drama club sells two kinds of tickets. Three adult tickets and two student tickets cost \$37. Five adult tickets and four student tickets cost \$65. Let $a$ be the price of one adult ticket in dollars and let $s$ be the price of one student ticket in dollars.
\par\vspace*{0.15cm}
(a) Write a system for the two sales and solve it by elimination. Give both prices in dollars.
\par\vspace*{4.6cm}\ansline
\par\vspace*{0.25cm}
(b) Explain why multiplying the first equation by $2$ is enough to eliminate $s$, with no second multiplication needed.
\par\vspace*{2.4cm}}

\problem[5.5cm]{Solve by elimination.
\[ 5x + 3y = 21 \qquad\text{and}\qquad 2x + 5y = 16 \]}

\problem[6cm]{An office places two orders from the same supplier. Four notebooks and three folders cost \$27. Six notebooks and five folders cost \$43. Let $n$ be the price of one notebook in dollars and let $f$ be the price of one folder in dollars. Write a system and solve it by elimination.}

\problem[5.5cm]{Solve by elimination.
\[ 7x - 2y = 1 \qquad\text{and}\qquad 3x + 4y = 15 \]}

\problem[0.2cm]{A school buys pencils and pens by the box. Five boxes of pencils and two boxes of pens cost \$46. Three boxes of pencils and four boxes of pens cost \$50. Let $p$ be the price of one box of pencils in dollars and let $q$ be the price of one box of pens in dollars.
\par\vspace*{0.15cm}
(a) Write a system and solve it by elimination. Give both prices in dollars.
\par\vspace*{4.6cm}\ansline
\par\vspace*{0.25cm}
(b) Explain why multiplying the first equation by $2$ does not change which pairs $(p,q)$ solve the system.
\par\vspace*{2.4cm}}

\problem[6cm]{Solve by elimination. Watch the signs --- one of the answers is negative.
\[ 6x + 5y = 3 \qquad\text{and}\qquad 8x - 3y = 33 \]}

\problem[0.2cm]{A market sells cashews and almonds by the kilogram. Three kilograms of cashews and two kilograms of almonds cost \$57. Five kilograms of cashews and four kilograms of almonds cost \$101. Let $c$ be the price of one kilogram of cashews in dollars and let $d$ be the price of one kilogram of almonds in dollars.
\par\vspace*{0.15cm}
(a) Write a system and solve it by elimination. Give both prices in dollars.
\par\vspace*{5cm}\ansline
\par\vspace*{0.25cm}
(b) There are two ways to scale here: match the almond terms, or match the cashew terms. Describe both routes and justify which one takes fewer multiplications.
\par\vspace*{3cm}}
