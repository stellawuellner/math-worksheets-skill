\begin{center}
Limit evidence from graphs, from tables, and from algebra
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item A limit can be read off a graph, estimated from a table, or computed by algebra. The three must agree --- and when they seem not to, the algebra is what settles it.
  \item Problems 1, 2, 5 and 8 all refer to \textbf{Graph A} below. The rest carry their own table or expression.
  \item Write $+\infty$, $-\infty$, or DNE where a finite value does not exist, and say which.
\end{itemize}

\problem[3.4cm]{\textbf{Graph A.} Read the two one-sided limits at $x = 1$.
\par\vspace{0.15cm}
\textbf{(a)} $\displaystyle\lim_{x \to 1^-} f(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 1^+} f(x) = $ \ansblank}

\problem[3.4cm]{\textbf{Graph A.} The graph passes through $x = 3$ with no break
and no hole. Read $\displaystyle\lim_{x \to 3} f(x)$.}

\problem[4cm]{The table lists values of $g(x) = \dfrac{x^3 - 27}{x - 3}$ near
$x = 3$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $2.9$ & $2.99$ & $3.01$ & $3.1$ \\ \hline
$g(x)$ & $26.11$ & $26.9101$ & $27.0901$ & $27.91$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.1cm}
Use the table to predict $\displaystyle\lim_{x \to 3} g(x)$, then confirm the
prediction by factoring $x^3 - 27$ and cancelling.}

\problem[5cm]{Compute algebraically --- no table, no graph:
\[
  \lim_{x \to 0} \frac{\sqrt{x + 16} - 4}{x}
\]}

\problem[5.2cm]{\textbf{Synthesis challenge.} Let
$r(x) = \dfrac{x^2 - 9}{|x - 3|}$. Here is a table of values near $x = 3$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $2.9$ & $2.99$ & $3.01$ & $3.1$ \\ \hline
$r(x)$ & $-5.9$ & $-5.99$ & $6.01$ & $6.1$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.1cm}
\textbf{(a)} $\displaystyle\lim_{x \to 3^-} r(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 3^+} r(x) = $ \ansblank
\par\vspace{0.1cm}
Confirm both values by splitting $|x-3|$ into its two cases and cancelling ---
the table predicts, the algebra proves.
\par\vspace{0.15cm}
\textbf{(c)} Sketch what the graph of $r$ must look like just left and just
right of $x = 3$, and state whether $\displaystyle\lim_{x \to 3} r(x)$ exists.}
