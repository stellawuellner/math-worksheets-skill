\problem{$f(x) = 2x - 5$, $g(x) = x^2 + 1$; find $(f \circ g)(3)$.\\
\textbf{Step 1.} $(f \circ g)(3)$ means $f\bigl(g(3)\bigr)$ --- the function
written on the \emph{right} acts first.\\
\textbf{Step 2.} $g(3) = 3^2 + 1 = 10$.\\
\textbf{Step 3.} $f(10) = 2(10) - 5 = 15$. \ans{15}}

\problem{Same $f$ and $g$; find $(g \circ f)(3)$.\\
\textbf{Step 1.} Now $f$ acts first: $(g \circ f)(3) = g\bigl(f(3)\bigr)$.\\
\textbf{Step 2.} $f(3) = 2(3) - 5 = 1$.\\
\textbf{Step 3.} $g(1) = 1^2 + 1 = 2$.\\
\textbf{Comparison to state:} $15 \neq 2$, so composition is \emph{not}
commutative --- swapping the order changes the answer, and one counterexample
settles it. \ans{2}}

\problem{$f(x) = 3x + 7$; find $f^{-1}(22)$.\\
\textbf{Step 1.} $f^{-1}(22)$ is the input whose output is $22$, so solve
$3x + 7 = 22$ rather than substituting $22$ into $f$.\\
\textbf{Step 2.} $3x = 15$, so $x = 5$.\\
\textbf{Step 3.} Check forward: $f(5) = 15 + 7 = 22$ $\checkmark$ \ans{x = 5}}

\problem{\textbf{Find and fix.} $f(x) = 3x + 1$, $g(x) = x - 6$;
$(f \circ g)(4)$.\\
\textbf{Step 1.} Nia's first move was $f(4) = 13$, then she subtracted $6$. That
is $g\bigl(f(4)\bigr) = (g \circ f)(4) = 7$ --- the reverse composition.\\
\textbf{Step 2.} The correct order feeds $4$ to $g$ first:
$g(4) = 4 - 6 = -2$.\\
\textbf{Step 3.} Then $f(-2) = 3(-2) + 1 = -5$.\\
\textbf{One-sentence diagnosis to accept:} ``She applied $f$ first; in
$f \circ g$ the inner function $g$ is applied first.'' \ans{-5}}

\problem{$f(x) = 3x + 7$, $f^{-1}(x) = \dfrac{x - 7}{3}$; show
$f\bigl(f^{-1}(x)\bigr) = x$.\\
\textbf{Step 1.} Substitute the whole inverse expression into $f$:
$f\bigl(f^{-1}(x)\bigr) = 3 \cdot \dfrac{x - 7}{3} + 7$.\\
\textbf{Step 2.} The $3$ and the denominator $3$ cancel: $(x - 7) + 7$.\\
\textbf{Step 3.} $x - 7 + 7 = x$ for every real $x$, which is exactly the
identity that certifies an inverse. \ans{f(f^{-1}(x)) = x}}

\problem{\textbf{Find and fix.} $h(x) = (x - 3)^2$ on $x \ge 3$; Sam's
$h^{-1}(x) = 3 - \sqrt{x}$ gives $h^{-1}(16) = -1$.\\
\textbf{Step 1.} An inverse must return values \emph{inside the domain of $h$},
and $-1 \ge 3$ is false, so Sam's output cannot be an input of $h$.\\
\textbf{Step 2.} Build the rule honestly: from $y = (x - 3)^2$,
$\sqrt{y} = |x - 3| = x - 3$ because $x \ge 3$ makes $x - 3$ non-negative. So
$h^{-1}(x) = 3 + \sqrt{x}$ --- the plus branch is forced by the restriction.\\
\textbf{Step 3.} $h^{-1}(16) = 3 + 4 = 7$, and checking forward,
$h(7) = (7-3)^2 = 16$ $\checkmark$\\
\textbf{Diagnosis to accept:} ``He kept the minus branch of the square root,
which leaves the restricted domain $x \ge 3$.'' \ans{7}}

\problem{$g(x) = x^2$ on all of $\mathbb{R}$; solve $x^2 = 9$.\\
\textbf{Step 1.} $x^2 - 9 = 0$ factors as $(x - 3)(x + 3) = 0$, so the equation
has two real solutions, not one.\\
\textbf{Step 2.} $x = 3$ and $x = -3$ both square to $9$.\\
\textbf{What it proves:} two different inputs share the output $9$, so $g$ is
not one-to-one on $\mathbb{R}$ and has no inverse there. Writing
$g^{-1}(x) = \sqrt{x}$ silently keeps only the $x \ge 0$ half of the domain; the
claim is legitimate only after $g$ is restricted to $x \ge 0$.
\ans{x = 3 \ \text{or}\ x = -3}}

\problem{$f(x) = \sqrt{x - 1}$, $g(x) = x^2 + 1$; Marcus's claim.\\
\textbf{Step 1.} The step $\sqrt{x^2} = x$ is the flaw: $\sqrt{x^2} = |x|$, which
equals $x$ only when $x \ge 0$. For $x = -4$, $f(g(-4)) = \sqrt{17 - 1} = 4$,
not $-4$.\\
\textbf{Step 2.} He checked only one direction. The second check,
$g\bigl(f(x)\bigr) = (\sqrt{x-1})^2 + 1 = x$, needs $x - 1 \ge 0$, i.e.
$x \ge 1$ --- and that is the restriction his one-sided check hid.\\
\textbf{Step 3.} Domains that make the pair genuine inverses: $f$ has domain
$x \ge 1$ with range $y \ge 0$; $g$ must be restricted to $x \ge 0$, where its
range is $y \ge 1$. Each function's domain is the other's range, which is the
structural signature of an inverse pair.\\
\textbf{Model answer (accept any equivalent wording):} ``$\sqrt{x^2} = |x|$, so
$f(g(x)) = x$ only for $x \ge 0$. He never checked $g(f(x))$, which forces
$x \ge 1$. With $f: x \ge 1 \to y \ge 0$ and $g: x \ge 0 \to y \ge 1$ the two are
genuine inverses.''}
