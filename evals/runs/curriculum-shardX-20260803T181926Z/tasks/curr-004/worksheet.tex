\wstitleblock{Missing and Mixed-Up Numbers}{Kindergarten / Grade 1 --- Counting sequences and cardinality}{}

\noindent\textbf{How to use this page:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=2pt]
  \item Say the numbers out loud as you read each row.
  \item When a number is missing, say the number just before the blank, then say the next one. That is the number that goes in the blank.
  \item Two rows have a mistake in them. Find the mistake, then write the row the right way.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3.5cm]{Write the number that is missing.
\[ 5 \qquad 6 \qquad \underline{\hspace{1.4cm}} \qquad 8 \qquad 9 \]}

\problem[4cm]{Write the two numbers that are missing.
\[ 11 \qquad 12 \qquad \underline{\hspace{1.4cm}} \qquad 14 \qquad \underline{\hspace{1.4cm}} \qquad 16 \]}

\problem[3.5cm]{Circle the number that is greater. Then write it on the line.
\[ 21 \qquad\qquad 12 \]}

\problem[4.5cm]{\textbf{Find the mistake.} Ben counted his blocks out loud. He said:
\[ 6 \qquad 7 \qquad 9 \qquad 10 \qquad 11 \]
Ben left a number out. Write the number Ben missed on the line. Then write his whole count the right way.}

\problem[4.5cm]{\textbf{Find the mistake.} Mia put her number cards in a row:
\[ 13 \qquad 14 \qquad 16 \qquad 15 \qquad 17 \]
Two cards traded places. Write all five numbers in counting order.}

\problem[5.5cm]{\textbf{Show how you know.} Which number is missing?
\[ 17 \qquad 18 \qquad \underline{\hspace{1.4cm}} \qquad 20 \]
Draw a number line, or draw that many counters, to show how you found it. Then tell a grown-up how you checked your answer. \noansline}
