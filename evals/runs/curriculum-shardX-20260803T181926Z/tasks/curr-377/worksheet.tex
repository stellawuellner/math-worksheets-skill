\wstitleblock{Properties of Logarithms}{Algebra 2}{}

\begin{center}
\small\itshape Applying the product, quotient, and power properties of logarithms to expand, condense, evaluate, and solve
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Use only the product, quotient, and power properties --- no calculator except where a decimal answer is asked for.
  \item Every logarithm here has a positive argument, so the properties apply throughout.
  \item Name the property you use at each step; a correct answer with no visible property is only half the work.
\end{itemize}

\vspace{0.2cm}

\problem[4cm]{Evaluate using the product property. \qquad $\log_2 8 + \log_2 4$}

\problem[4.5cm]{Expand into separate terms, writing the constant term as a multiple of $\ln 2$.
\qquad $\ln(8x)$}

\problem[5cm]{Write as a single logarithm, then evaluate it.
\qquad $2\log_3 6 - \log_3 4$}

\problem[4.5cm]{Evaluate using the product property. \qquad $\log_6 4 + \log_6 9$}

\problem[5cm]{Expand into separate terms using the quotient property.
\qquad $\log_2\!\left(\dfrac{x}{32}\right)$}

\problem[5cm]{Solve for $x$. \qquad $\log_3 x + \log_3 4 = \log_3 20$}

\problem[5cm]{Write as a single logarithm. \qquad $3\ln 2 + 2\ln 5$}

\problem[6cm]{\textbf{(a)} Expand $\log_2\!\left(\dfrac{x^5}{16}\right)$ using the quotient and power properties.
\par\vspace{0.2cm}
\textbf{(b)} Use your expanded form to evaluate the expression when $x = 4$.}

\problem[5cm]{Solve for $x$. \qquad $\log x - \log 3 = \log 7$}

\problem[6cm]{A calculator gives $\log 2 \approx 0.3010$ and $\log 3 \approx 0.4771$.
Using only the properties of logarithms and those two values, approximate $\log 48$ to three decimal places.}

\problem[6cm]{\textbf{(a)} Write $3\ln x - \ln y - 2\ln z$ as a single logarithm.
\par\vspace{0.2cm}
\textbf{(b)} Evaluate your single logarithm when $x = 2$, $y = 4$, and $z = 1$. Leave the answer in exact form.}

\problem[5cm]{Solve for $x$. \qquad $3\log_2 x = \log_2 27$}
