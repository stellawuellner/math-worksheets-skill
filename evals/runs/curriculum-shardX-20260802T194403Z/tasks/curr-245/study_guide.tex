\sstitleblock{Pythagorean Theorem and Coordinate Distance}

\noindent\small Every problem on the worksheet is the same triangle in
disguise. Find the right angle, name the longest side $c$, and only then decide
whether you are adding squares (looking for $c$) or subtracting them (looking
for a leg).\normalsize

\skillheading{1. Hypotenuse from two legs (hypotenuse-from-legs)}
\begin{formulabox}
\textbf{Theorem:} in a right triangle with legs $a$, $b$ and hypotenuse $c$,
\[ a^2 + b^2 = c^2 \qquad\Longrightarrow\qquad c = \sqrt{a^2 + b^2} \]
The hypotenuse is always opposite the right angle and always the longest side.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A rectangle is $9$ cm by $12$ cm. How long is its
diagonal?
\step{The diagonal cuts the rectangle into two right triangles whose legs are
the sides, and I want the side opposite the right angle --- add the squares.}
\step{$c = \sqrt{9^2 + 12^2} = \sqrt{81 + 144} = \sqrt{225}$}
$\ans{15\ \text{cm}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find the diagonal of an $8$ cm by $15$ cm rectangle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17\ \text{cm}}$}
\end{tryitbox}

\skillheading{2. A missing leg from the hypotenuse}
\begin{formulabox}
\textbf{Rearranged:}\quad $a = \sqrt{c^2 - b^2}$ \quad --- subtract when the
side you want is \emph{not} the longest one.\\[3pt]
Ladders, ropes, guy wires and taut strings are hypotenuses: they run from a
high point to a point on the ground away from the base.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A $20$ ft ladder rests with its base $16$ ft from a
wall. How high does it reach?
\step{The ladder is the longest side, so it is $c$ and the height is a leg ---
subtract the squares.}
\step{$h = \sqrt{20^2 - 16^2} = \sqrt{400 - 256} = \sqrt{144}$}
$\ans{12\ \text{ft}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A $41$ ft wire runs from the top of a pole to a stake
$9$ ft from its base. How tall is the pole?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40\ \text{ft}}$}
\end{tryitbox}

\skillheading{3. Distance between two points (coordinate-distance)}
\begin{formulabox}
\textbf{Rule:} the horizontal gap $|x_2 - x_1|$ and the vertical gap
$|y_2 - y_1|$ are the legs, so
\[ d = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2} \]
This is not a new formula --- it is the theorem applied to a triangle you draw
on the grid. Subtracting a negative coordinate \emph{adds} to the gap.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Distance from $(1, 2)$ to $(7, 10)$.
\step{Two points on a grid: build the right triangle by moving across, then
up, and the distance is its hypotenuse.}
\step{Legs $7 - 1 = 6$ and $10 - 2 = 8$, so
$d = \sqrt{6^2 + 8^2} = \sqrt{36 + 64} = \sqrt{100}$}
$\ans{10\ \text{units}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find the distance from $(-3, 4)$ to $(2, -8)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13\ \text{units}}$}
\end{tryitbox}

\skillheading{4. Multistep and composite figures (multistep-composite)}
\begin{formulabox}
\textbf{Method:} break the picture into right triangles and straight runs,
compute each piece, then combine at the end.\\[3pt]
Horizontal and vertical segments need no square root. Save every rounding for
the final line, and check that a shortcut is shorter than the path it replaces.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Perimeter of the triangle with vertices $(0,0)$,
$(9,0)$, $(9,12)$.
\step{Two sides lie along grid lines, so only the slanted side needs the
theorem; then add all three.}
\step{Legs $9$ and $12$ give
$\sqrt{9^2 + 12^2} = 15$, so the perimeter is $9 + 12 + 15$.}
$\ans{36\ \text{units}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A field is $30$ m by $40$ m. How much shorter is the
diagonal than walking along two sides?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20\ \text{m}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Add squares only when hunting the hypotenuse. If the
answer comes out \emph{longer} than the ladder, rope, or string in the problem,
you subtracted in the wrong order. And $\sqrt{a^2 + b^2}$ is never $a + b$.
\end{watchoutbox}
