\sstitleblock{Perimeter and Area of Coordinate Polygons}

\skillheading{Skill 1 --- Perimeter from coordinates}

\begin{formulabox}
\textbf{Rule.} Perimeter is the sum of the side lengths, and every side length
comes from the distance formula
$d = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}$.\par
Two shortcuts: a \emph{horizontal} side's length is the difference of the
$x$-values, and a \emph{vertical} side's length is the difference of the
$y$-values --- no square roots needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the perimeter of the triangle with vertices $(1,1)$,
$(7,1)$ and $(7,9)$.
\step{Perimeter means every side, so take them one at a time --- two of these
sides are horizontal and vertical, so only the slanted one needs the full
formula.}
\step{horizontal side $6$, vertical side $8$, slanted side
$\sqrt{6^2+8^2} = 10$; adding, $6+8+10$}

perimeter $= \ans{24}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the perimeter of the triangle with vertices $(2,2)$,
$(2,7)$ and $(14,7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$ units}
\end{tryitbox}

\skillheading{Skill 2 --- Area by the shoelace method}

\begin{formulabox}
\textbf{Rule.} List the vertices once around the boundary in order, then
returning to the first. Multiply each $x$ by the \emph{next} $y$ and subtract
each next $x$ times the current $y$; add all those terms and halve the size of
the total:
\[ \text{area} = \tfrac{1}{2}\left| \sum \left( x_i y_{i+1} - x_{i+1} y_i \right) \right| . \]
Going round the wrong way only changes the sign, which the size bars remove.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the area of the quadrilateral $(0,0)$, $(6,0)$, $(8,5)$,
$(2,5)$.
\step{The sides are not all horizontal or vertical, so there is no easy
rectangle here --- the shoelace sum handles any polygon given in order.}
\step{terms $0$, $30$, $30$, $0$ add to $60$, and half of $60$ is the area}

area $= \ans{30}$ square units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the area of the quadrilateral $(1,0)$, $(9,0)$, $(7,4)$,
$(3,4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$ square units}
\end{tryitbox}

\skillheading{Skill 3 --- Area by splitting the shape}

\begin{formulabox}
\textbf{Rule.} A polygon whose sides are all horizontal or vertical can be cut
into rectangles: find each rectangle's area from its width and height, then add.
This is the fastest check on a shoelace answer, because the two methods share no
arithmetic --- if they agree, both are almost certainly right.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the area of the L-shaped hexagon $(0,0)$, $(6,0)$,
$(6,3)$, $(3,3)$, $(3,6)$, $(0,6)$.
\step{Every side runs along a grid line, so cutting into rectangles is easier
than the shoelace sum --- one horizontal cut at height $3$ does it.}
\step{bottom rectangle $6 \times 3 = 18$, upper-left rectangle
$3 \times 3 = 9$, and $18 + 9$ is the total}

area $= \ans{27}$ square units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the area of the L-shaped hexagon $(0,0)$, $(10,0)$,
$(10,2)$, $(4,2)$, $(4,6)$, $(0,6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$ square units}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} equal side lengths do not make a square. A rhombus with
four sides of $5$ has area $15$, not $25$ --- only a rectangle's area is the
product of two adjacent sides.
\end{watchoutbox}
