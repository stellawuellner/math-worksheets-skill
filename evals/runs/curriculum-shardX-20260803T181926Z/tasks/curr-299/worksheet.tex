\wstitleblock{Absolute-Value, Piecewise, and Linear-Quadratic Models --- Boundaries and Open/Closed Endpoints}{Algebra 1}{}

\noindent\small Every rule below has a \textbf{boundary}: the input where one
formula stops and another starts. Almost every mistake on this page happens at
that one input.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A \textbf{closed} inequality ($\le$ or $\ge$) \emph{owns} the boundary
        point. A \textbf{strict} one ($<$ or $>$) does not.
  \item Exactly one rule may own each input. Two rules that both own it give two
        outputs, and neither owning it gives none.
  \item Read the inequality \emph{before} you substitute. The formula that looks
        nearest is not always the one that applies.
\end{itemize}
Show the rule you chose, not just the number.
\normalsize

\vspace{0.15cm}

\problem[4.5cm]{Warm up. A model is defined by
\[ F(x) = \begin{cases} 2x + 1, & x < 3 \\ x^2 - 4, & x \ge 3 \end{cases} \]
Which rule owns the input $x = 1$? Name it, then evaluate $F(1)$.}

\problem[5cm]{\textbf{Find and fix.} Using the same model
\[ F(x) = \begin{cases} 2x + 1, & x < 3 \\ x^2 - 4, & x \ge 3 \end{cases} \]
Mia evaluates at the boundary and writes $F(3) = 7$. Explain in one sentence which
rule Mia used and why that rule does not own $x = 3$. Then give the correct value
of $F(3)$.}

\problem[5cm]{\textbf{Find and fix.} A different model reverses which side is
closed:
\[ G(x) = \begin{cases} 5 - x, & x \le -2 \\ x^2 + 1, & x > -2 \end{cases} \]
Leo has learned to ``use the second rule at the boundary'' and writes
$G(-2) = 5$. Say what is wrong with that habit, then give the correct value of
$G(-2)$.}

\problem[5cm]{\textbf{Find and fix.} A student writes this definition:
\[ H(x) = \begin{cases} x + 4, & x \le 1 \\ 3x, & x \ge 1 \end{cases} \]
Substituting $x = 1$ into the two rules gives two different outputs, so $H$ is not
a function as written. Repair the definition by changing exactly one inequality
sign so that the \emph{top} rule keeps the boundary, and then give $H(1)$.}

\problem[5.5cm]{A second student writes:
\[ K(x) = \begin{cases} 2x, & x < 0 \\ x + 2, & x > 0 \end{cases} \]
\textbf{(a)} Explain why this definition fails at $x = 0$, and give two different
repairs, each changing exactly one inequality sign. \emph{(Open response --- your
teacher checks this part.)}
\textbf{(b)} Repair it so that the \emph{top} rule owns the boundary, then give
$K(0)$.}

\problem[5.5cm]{A machine part passes inspection when its measurement $x$, in
centimetres, satisfies $|2x - 5| \le 3$. Priya reports the passing range as
$1 < x < 4$ and tells the operator that a part measuring exactly $1$ does not
pass. Solve $|2x - 5| = 3$ to find the two endpoints, then state whether each
endpoint passes inspection and correct Priya's range.}

\problem[5.5cm]{A ramp is modelled by
\[ M(x) = \begin{cases} 4x + k, & x < 2 \\ x^2 - 1, & x \ge 2 \end{cases} \]
The builder wants no jump at the boundary: the value the top rule approaches as
$x$ climbs to $2$ must equal the value the bottom rule actually takes at $x = 2$.
Find the value of $k$ that does this.}

\problem[6cm]{\textbf{Challenge.} A designer wants a piecewise model built from
the line $y = x - 1$ and the parabola $y = x^2 - 4x + 3$, switching from one to
the other at a point where the two graphs meet, so the model has no jump. Find
every point where the line and the parabola meet. Then state which of the two
$x$-values must be written with a closed inequality if the line is the rule used
first.}
