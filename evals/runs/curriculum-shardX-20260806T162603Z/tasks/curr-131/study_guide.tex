\sstitleblock{Finding a Fraction of a Number or Set}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- A unit fraction of a number}

\begin{formulabox}
\textbf{Rule:} $\tfrac{1}{d}$ of a number means: split the number into $d$
equal groups and take one group. In symbols, divide by the denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\tfrac{1}{6}$ of 30.
\step{One sixth means 6 equal groups and I take one --- so divide 30 by 6.}
\step{$30 \div 6 = 5$, so $\tfrac{1}{6}$ of 30 is $\ans{5}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\tfrac{1}{7}$ of 28.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Any fraction of a number}

\begin{formulabox}
\textbf{Plan:} to find $\tfrac{a}{d}$ of a number, first find
$\tfrac{1}{d}$ of it (divide by the denominator), then take $a$ of those
groups (multiply by the numerator).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\tfrac{3}{4}$ of 20.
\step{First find one fourth by dividing, then take three of those groups.}
\step{$20 \div 4 = 5$, and $3 \times 5 = 15$, so $\tfrac{3}{4}$ of 20 is $\ans{15}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\tfrac{2}{5}$ of 35.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Finding the whole from a part}

\begin{formulabox}
\textbf{Plan:} this is the backward question. If you know what a fraction of
the number is, first find what \emph{one} group holds, then multiply by the
number of groups to rebuild the whole.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\tfrac{2}{3}$ of a number is 12. What is the number?
\step{The 12 fills 2 of the 3 equal groups, so one group is 12 divided by 2, and the whole is 3 groups.}
\step{$12 \div 2 = 6$ in each group, and $3 \times 6 = 18$, so the number is $\ans{18}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\tfrac{1}{5}$ of a number is 9. What is the
number?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``of'' does not always mean multiply the two numbers
you see. $\tfrac{1}{4}$ of 12 is $12 \div 4 = 3$, not $12 \times 4$.
Drawing the equal groups keeps the two moves straight.
\end{watchoutbox}
