\sstitleblock{Finding and Verifying Inverse Functions}

\begin{formulabox}
\textbf{What an inverse is.} $f^{-1}$ undoes $f$: if $f(a) = b$ then
$f^{-1}(b) = a$. So $f^{-1}(f(x)) = x$ and $f(f^{-1}(x)) = x$.\\[3pt]
\textbf{Notation warning.} $f^{-1}$ is the inverse \emph{function}. It is not
$\dfrac{1}{f(x)}$.\\[3pt]
\textbf{Domain and range swap.} The domain of $f^{-1}$ is the range of $f$, and
its range is the domain of $f$. Graphically, $f^{-1}$ is the reflection of $f$
across the line $y = x$.\\[3pt]
\textbf{Only one-to-one functions have inverses.} If a horizontal line meets
the graph of $f$ twice, two inputs share an output and there is nothing for
$f^{-1}$ to return --- restrict the domain first.
\end{formulabox}

\skillheading{1. Find the inverse: swap, then solve}

\begin{formulabox}
Write $y = f(x)$; swap $x$ and $y$; solve the new equation for $y$; rename it
$f^{-1}(x)$. Solving means undoing the operations \textbf{in reverse order} ---
whatever was done last to $x$ is undone first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the inverse of $f(x) = 4x + 3$.
\step{The question asks which input produces a given output, so swap the roles
of the variables and solve for the new $y$.}
\step{$y = 4x + 3$ becomes $x = 4y + 3$. Subtract $3$: $x - 3 = 4y$. Divide by
$4$: $y = \dfrac{x-3}{4}$.}
$f^{-1}(x) = \ans{\dfrac{x-3}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the inverse of $f(x) = 6x - 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f^{-1}(x) = \tfrac{x+1}{6}}$}
\end{tryitbox}

\skillheading{2. Verify a pair with composition}

\begin{formulabox}
Two functions are inverses only when \textbf{both} $f(g(x)) = x$ and
$g(f(x)) = x$. Substitute one whole function into the other and simplify; the
variable must survive alone, with every constant cancelling.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Are $f(x) = \dfrac{x}{2} + 4$ and $g(x) = 2x - 8$ inverses?
\step{``Verify'' means composition, not re-deriving --- substitute $g$ into $f$
and simplify, then repeat the other way round.}
\step{$f(g(x)) = \dfrac{2x-8}{2} + 4 = (x - 4) + 4$, and the constants cancel.}
The composition collapses to $\ans{x}$, and $g(f(x))$ does the same, so the
pair is inverse.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Simplify $f(g(x))$ for $f(x) = 3x - 1$ and
$g(x) = \dfrac{x+1}{3}$. What does it come out to?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x}$ --- so they are inverses}
\end{tryitbox}

\skillheading{3. Inverse values, domain and restriction}

\begin{formulabox}
$f^{-1}(b)$ asks: which input of $f$ gives the output $b$? You can answer it
either by building the formula for $f^{-1}$ and evaluating, or by solving
$f(x) = b$ directly --- the two routes must agree.\\[2pt]
When $f$ is not one-to-one, restrict its domain first (for $f(x) = x^2$ take
$x \ge 0$); the inverse then inherits its domain from the range of the
restricted $f$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} For $f(x) = 2x + 7$, find $f^{-1}(9)$.
\step{Only one value is wanted, so either build the inverse formula or solve
$f(x) = 9$ --- the formula route also leaves you something reusable.}
\step{$x = 2y + 7$ gives $f^{-1}(x) = \dfrac{x-7}{2}$, so
$f^{-1}(9) = \dfrac{9-7}{2} = \dfrac{2}{2}$.}
$f^{-1}(9) = \ans{1}$, and indeed $f(1) = 2 + 7 = 9$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} For $f(x) = 5x - 2$, find $f^{-1}(13)$. Check your answer by
putting it back into $f$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$, since $f(3) = 13$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} undo operations in reverse order. For
$f(x) = 3x + 5$ the inverse is $\dfrac{x-5}{3}$, not $\dfrac{x}{3} - 5$ --- the
$5$ comes off before the division, because it was added last.
\end{watchoutbox}
