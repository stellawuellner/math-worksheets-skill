\wstitleblock{Find the Mistake: Composition Order and Inverse Domains}{Algebra 2}{}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item In $(f \circ g)(x) = f\bigl(g(x)\bigr)$ the \emph{inner} rule $g$ acts on the input first.
  \item A rule is the inverse of $f$ only if \emph{both} round trips return the input, and only on the inputs where that actually happens.
  \item When a problem quotes a wrong answer, first produce the correct one, then say exactly which move produced the wrong one.
\end{itemize}

\problem{Let $f(x) = 3x - 1$ and $g(x) = x^2 + 2$.
\par\smallskip
(a)~Find $f\bigl(g(2)\bigr)$.
\par\vspace*{2.7cm}\ansline
\par\smallskip
(b)~Find $g\bigl(f(2)\bigr)$.
\par\vspace*{2.7cm}\ansline}

\problem{A student was asked for $(f \circ g)(6)$, where $f(x) = \dfrac{x}{2} + 7$ and $g(x) = 8 - x$. The student wrote $-2$.
\par\smallskip
(a)~Work out the correct value of $(f \circ g)(6)$.
\par\vspace*{2.9cm}\ansline
\par\smallskip
(b)~Explain which composition the student actually carried out, and why it produces $-2$.
\par\vspace*{2.6cm}\ansline}

\problem{For $f(x) = 3x + 12$, a student claims the inverse rule is $\dfrac{x}{3} + 12$, saying ``divide by $3$ instead of multiplying, and leave the $12$ alone.''
\par\smallskip
(a)~Compute $f(5)$, then feed that output into the student's claimed rule. What comes out?
\par\vspace*{2.9cm}\ansline
\par\smallskip
(b)~Write the correct rule for $f^{-1}(x)$.
\par\vspace*{2.9cm}\ansline}

\problem{A student writes that the inverse of $h(x) = x^2 + 1$ is $\sqrt{x - 1}$, and says it works for every input.
\par\smallskip
(a)~Compute $h(-3)$, then feed that output into $\sqrt{x - 1}$. Write the number that comes out.
\par\vspace*{2.9cm}\ansline
\par\smallskip
(b)~Explain what the round trip shows about the claim.
\par\vspace*{2.7cm}\ansline}

\problem{Asked for a simplified rule for $(g \circ f)(x)$ with $f(x) = x - 4$ and $g(x) = x^2$, a student wrote $x^2 - 4$.
\par\smallskip
(a)~Write the correct simplified rule for $(g \circ f)(x)$.
\par\vspace*{2.6cm}\ansline
\par\smallskip
(b)~Evaluate the correct rule at $x = 7$.
\par\vspace*{2.2cm}\ansline
\par\smallskip
(c)~Explain the error in the student's rule.
\par\vspace*{2.2cm}\ansline}

\problem{Let $k(x) = \sqrt{x} + 3$.
\par\smallskip
(a)~Find the rule for $k^{-1}(x)$.
\par\vspace*{2.9cm}\ansline
\par\smallskip
(b)~The rule from part (a) is not the whole answer: it must carry a restriction on its inputs. State the restriction and justify it.
\par\vspace*{2.9cm}\ansline}

\problem{A student claims that the inverse of $f(x) = \dfrac{x + 5}{x - 2}$ is $\dfrac{x - 2}{x + 5}$ --- ``just turn it upside down.''
\par\smallskip
(a)~Find the correct rule for $f^{-1}(x)$.
\par\vspace*{3.4cm}\ansline
\par\smallskip
(b)~Evaluate your correct rule at $x = 3$.
\par\vspace*{2.2cm}\ansline
\par\smallskip
(c)~Explain why turning a rule upside down is not the same as inverting it.
\par\vspace*{2.2cm}\ansline}

\problem{Let $f(x) = x + 3$ and $g(x) = x^2$, both defined for every real number. One composition simplifies to $x^2 + 3$.
\par\smallskip
(a)~Write the simplified rule for the \emph{other} composition, $(g \circ f)(x)$.
\par\vspace*{2.4cm}\ansline
\par\smallskip
(b)~The two composed rules agree at exactly one input. Find it.
\par\vspace*{2.6cm}\ansline
\par\smallskip
(c)~Explain why they can agree at only one input and not at two.
\par\vspace*{2.4cm}\ansline}
