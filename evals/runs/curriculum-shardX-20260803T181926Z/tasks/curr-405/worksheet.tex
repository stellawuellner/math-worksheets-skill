\wstitleblock{Transformations, Compositions and Inverses Together}{Precalculus \quad Advanced function transformations and composition}{}

\begin{center}
\textit{Transforming a rule, composing two functions, and inverting a function}
\end{center}

\noindent\textbf{Before you start.} All three ideas on this sheet are the same
idea seen from different sides: a transformation composes $f$ with a simple
linear map, and an inverse undoes a composition.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Inside the function} changes $x$ and acts \emph{horizontally}, in
the direction opposite to the sign: $f(x-h)$ shifts right by $h$, $f(kx)$
compresses by a factor $1/k$.
\item \textbf{Outside the function} changes $y$ and acts \emph{vertically},
exactly as written: $af(x)+c$ stretches by $a$ and shifts up by $c$.
\item \textbf{Composition works inside out}: $(f\circ g)(x)=f(g(x))$, so
evaluate $g$ first.
\item \textbf{An inverse undoes}: $f^{-1}(f(x))=x$ and $f(f^{-1}(x))=x$. To
find it, swap $x$ and $y$ and solve for $y$.
\end{itemize}

\medskip

\problem[6cm]{Let $f(x)=x^2$. Write the rule for $g(x)=f(x-2)+5$ as a single
expanded polynomial, and say which direction each shift goes.}

\problem[6cm]{Let $f(x)=2x-3$ and $g(x)=x^2+1$. Find $(f\circ g)(2)$. Show the
inner value before you use it.}

\problem[7cm]{Let $f(x)=2x-3$.

\smallskip
(a) Find $f^{-1}(x)$ by swapping $x$ and $y$ and solving.

\smallskip
(b) Check your answer by simplifying $f\bigl(f^{-1}(x)\bigr)$, and evaluate
$f^{-1}(7)$.}

\problem[6cm]{Let $f(x)=x^2-x$. Write the rule for $h(x)=-f(2x)$ as an expanded
polynomial, and name the two transformations involved.}

\problem[6cm]{Let $f(x)=x+4$ and $g(x)=x^2$. Write $(g\circ f)(x)$ as an
expanded polynomial. Then say how the graph of $g\circ f$ is related to the
graph of $g$.}

\problem[8cm]{Let $f(x)=\dfrac{2x+1}{x-3}$.

\smallskip
(a) Find $f^{-1}(x)$. (Swap $x$ and $y$, clear the denominator, then collect
the $y$ terms on one side.)

\smallskip
(b) Verify by simplifying $f\bigl(f^{-1}(x)\bigr)$, and evaluate $f^{-1}(4)$.}

\problem[7cm]{The graph of $h$ is the graph of $f(x)=\sqrt{x}$ stretched
vertically by a factor of $3$, then shifted left $4$ and down $1$.

\smallskip
(a) Write the rule for $h(x)$.

\smallskip
(b) Find $h(5)$.}

\problem[7cm]{Let $f(x)=\dfrac{1}{x-2}$ and $g(x)=x+5$. Find
$(f\circ g)(-1)$, and state the one input value that $f\circ g$ must exclude
from its domain.}

\problem[7cm]{Let $f(x)=3x-7$. For which value of $x$ is $f^{-1}(x)=2$? Answer
without ever writing $f^{-1}$ if you can see the shortcut, then check with the
formula.}

\problem[8cm]{Let $f(x)=2x$ and $g(x)=x+3$.

\smallskip
(a) Write $(f\circ g)(x)$ and then find its inverse.

\smallskip
(b) Separately, compute $g^{-1}\bigl(f^{-1}(x)\bigr)$. Show that the two
results agree, and say in one sentence why the order of $f$ and $g$ reverses.}

\problem[8cm]{A table gives two values of $f$: \; $f(1)=4$ and $f(3)=-2$.
Let $h(x)=-2f(x-1)+3$.

\smallskip
(a) Find $h(2)$.

\smallskip
(b) Describe, in order, the three transformations that carry the graph of $f$
to the graph of $h$.}

\problem[10cm]{\textbf{Synthesis.} Let $f(x)=(x-1)^2$ with the domain
restricted to $x\ge 1$.

\smallskip
(a) Explain why this restriction gives $f$ an inverse, when $f$ on all real
numbers has none.

\smallskip
(b) Find $f^{-1}(x)$ and verify it by simplifying $f\bigl(f^{-1}(x)\bigr)$.

\smallskip
(c) Compute $f^{-1}\bigl(f(5)\bigr)$, and say what that value shows about
reflecting the graph across the line $y=x$.}
