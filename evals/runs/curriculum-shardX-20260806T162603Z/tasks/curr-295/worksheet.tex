\wstitleblock{Choosing a Method and Reading the Roots}{Algebra 1}{Square roots, factoring, the quadratic formula, and what the roots mean in a story}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Three methods are in play, and \textbf{nobody tells you which to use}: take square roots when the squared part is already alone; factor when the expression factors quickly; use the quadratic formula when it does not.
  \item Before each problem, write one short line saying which method you picked. That line is part of the practice.
  \item ``Solve'' means \textbf{list every solution}, unless the problem says otherwise.
  \item In the story problems, solve the equation first and interpret afterwards --- a solution can be perfectly correct and still not answer the question that was asked.
\end{itemize}

\problem[7cm]{A ball is kicked straight up from the ground. Its height above the ground, in feet, after $t$ seconds is $h = -16t^2 + 32t$.
\par\medskip
(a) Solve $-16t^2 + 32t = 0$ and list both solutions, in seconds.
\par\smallskip
(b) In the space above, say which solution tells you when the ball lands, and explain what the other solution describes.}

\problem[7cm]{A rectangular vegetable garden is $4$ metres longer than it is wide. Its area is $45$ square metres. Writing $w$ for the width in metres, the area gives $w(w+4) = 45$.
\par\medskip
(a) Solve $w(w+4) = 45$ and list both solutions.
\par\smallskip
(b) In the space above, state the width of the garden in metres and explain why the other solution is rejected.}
