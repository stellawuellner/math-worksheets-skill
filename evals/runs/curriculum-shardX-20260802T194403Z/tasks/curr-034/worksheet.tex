\wstitleblock{Teen Numbers --- Fixing Swapped Tens and Ones}{Kindergarten--Grade 1}{}

\noindent\small Every teen number is \textbf{1 ten} and some \textbf{ones}.
The \textbf{tens} digit is always written \textbf{first}. Some children below
flipped their two digits. Read the blocks or the words, then write the number
that is really being shown.\normalsize

\begin{itemize}[leftmargin=1.4cm, itemsep=1pt, topsep=3pt]
  \item A full ten-frame is one ten. Loose dots are ones.
  \item Build it with blocks or draw it before you write.
  \item Check: does the first digit match how many full tens you see?
\end{itemize}

\problem[3cm]{Count the dots. One ten-frame is full, so that is \textbf{1 ten}.
The loose dots are the \textbf{ones}.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline

$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\end{tabular}\qquad
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
~ & ~ & ~ & ~ & ~ \\\hline
\end{tabular}
\end{center}
There is \textbf{1 ten} and there are \textbf{5 ones}. Write the number.}

\problem[3cm]{Mia builds \textbf{1 ten}-stick and \textbf{3 ones}.
Mia writes \textbf{31}.\\[4pt]
Mia put her digits in the wrong places. Circle the digit that is in the wrong
place, then write the number Mia's blocks really show.}

\problem[3cm]{Ben is writing the number \textbf{seventeen}.
Seventeen is \textbf{1 ten} and \textbf{7 ones}. Ben writes \textbf{71}.\\[4pt]
Tell why \textbf{71} is too big, then write seventeen the correct way.}

\problem[3cm]{Sam builds \textbf{1 ten} and \textbf{2 ones}.
Ty builds \textbf{2 tens} and \textbf{1 one}.\\[4pt]
Draw both. Then write $<$ or $>$ in the blank:\\[8pt]
\hspace*{1cm}\Large Sam's number \ansblank Ty's number}

\problem[3cm]{Ana counts these dots and writes \textbf{61}.

\problem[3.5cm]{Jo means to write \textbf{1 ten} and \textbf{8 ones}.
Instead she swaps her two digits.\\[4pt]
Write $>$ or $<$ in the blank, then say which digit sits in the tens place of
Jo's swapped number:\\[8pt]
\hspace*{1cm}\Large Jo's swapped number \ansblank the number Jo meant}
