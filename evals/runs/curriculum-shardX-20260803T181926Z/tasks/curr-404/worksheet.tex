\wstitleblock{Diagnosing Transformation-Order and Inverse-Domain Errors}{Precalculus --- Advanced function transformations and composition}{}

\noindent\textbf{Two habits this sheet is built to break:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Reading the inside left to right.} Everything applied to the input acts in the \emph{reverse} of the order it is written, and a horizontal shift can only be read after the inside is factored: $f(2x+8)$ is not a shift of $8$.
  \item \textbf{Treating an inverse as pure algebra.} Swapping $x$ and $y$ produces a formula; it produces a \emph{function} only on the range of the original. The domain of $f^{-1}$ is not an afterthought and it is not all reals by default.
\end{itemize}

\noindent On every find-and-fix item, quote the line of the student's work that
fails and say which of the two habits produced it, then redo the problem from
that line on. Where a decimal is requested, give three places.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4cm]{Let $f(x) = x^2$ and let $g(x) = f(x - 3) + 2$. Write $g(x)$ in
expanded polynomial form, and name in words the two transformations that carry
the graph of $f$ to the graph of $g$.}

\problem[5cm]{Let $f(x) = x^2$ and let $h(x) = f(2x - 6)$.
\par\vspace{2pt}
(a) Rewrite $h(x)$ in the form $a(x - c)^2$ by factoring the inside first, and
state the horizontal transformations in the order they actually occur.
\par\vspace{2pt}
(b) Evaluate $h(5)$.}

\problem[5.5cm]{Let $f(x) = 2x + 1$ and $g(x) = x^2 - 3$. Find $(f \circ g)(x)$
and $(g \circ f)(x)$, each in expanded form, and state in one line why the two
results are different functions.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} Let $f(x) = \sqrt{x + 1}$ and
$g(x) = 3x - 2$. Asked for $(f \circ g)(2)$, a student computes $f(2) = \sqrt{3}$
first, then substitutes into $g$, and reports $3\sqrt{3} - 2 \approx 3.196$.
\par\vspace{2pt}
Name the error, write $(f \circ g)(x)$ in simplified form, and give
$(f \circ g)(2)$ to three decimal places.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student describes the graph
of $y = f(2x + 8)$ as ``the graph of $f$ shifted left $8$, then compressed
horizontally by a factor of $\tfrac{1}{2}$''.
\par\vspace{2pt}
Rewrite the inside in factored form, state the correct horizontal shift, and
find the $x$-value at which the inside expression equals $0$.}

\problem[5.5cm]{Let $f(x) = \sqrt{x - 4} + 1$.
\par\vspace{2pt}
(a) Find $f^{-1}(x)$ and give it in completed-square form.
\par\vspace{2pt}
(b) State the domain of $f^{-1}$ and say where that restriction comes from.
\par\vspace{2pt}
(c) Check your inverse by evaluating $f^{-1}(3)$.}

\problem[6cm]{\textbf{Find and fix the mistake.} Let $f(x) = x^2 - 6x + 5$ with
the domain restricted to $x \ge 3$. A student writes
\[ f^{-1}(x) = 3 \pm \sqrt{x + 4}, \qquad \text{domain: all real numbers} \]
\par\vspace{2pt}
Rewrite $f$ in vertex form, explain which branch of the $\pm$ survives and why,
give the correct domain of $f^{-1}$, and check the result by evaluating
$f^{-1}(5)$.}

\problem[5cm]{\textbf{Explain.} Write a short paragraph for a classmate who
reads $f(3x - 12)$ as ``stretch by $3$, then shift left $12$'' and who believes
every inverse has domain all reals. Explain why transformations applied to the
input act in the reverse of the written order, why the inside must be factored
before a shift can be read off, and why the domain of $f^{-1}$ is the range of
$f$. \noansline}
