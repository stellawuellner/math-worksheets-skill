\wstitleblock{Reading and Comparing Multi-Digit Numbers}{Grades 4--5 Mathematics}{}

\begin{center}\textit{Value of a digit, number forms, and comparing numbers}\end{center}

\noindent\textbf{Directions:} Use the place-value charts and models to help you.
Show your thinking in the space under each problem, and write each final answer
on its answer blank.

\problem[2.5cm]{The place-value chart shows the number $5{,}872$.

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Thousands} & \textbf{Hundreds} & \textbf{Tens} & \textbf{Ones} \\
\hline
5 & 8 & 7 & 2 \\
\hline
\end{tabular}
\end{center}

Using the chart, what is the \emph{value} of the digit 8?}

\problem[3cm]{What is the value of the digit 6 in $46{,}205$? (It may help to
write the number in a place-value chart first.)}

\problem[2.5cm]{Write the number in standard form:
\par\vspace{2pt}\noindent
$30{,}000 + 7{,}000 + 400 + 9 = $~\ansblank}

\problem[3cm]{Write this number in standard form: \emph{sixty-two thousand,
four hundred eighteen}.}

\problem[2.5cm]{Write $<$, $>$, or $=$ to make the statement true:
\par\vspace{2pt}\noindent
\hspace*{1.5cm}$46{,}205$~~\ansblank~~$46{,}502$}

\problem[2.5cm]{Write $<$, $>$, or $=$ to make the statement true:
\par\vspace{2pt}\noindent
\hspace*{1.5cm}$72{,}341$~~\ansblank~~$72{,}314$}

\problem[3.5cm]{Order these numbers from least to greatest:
$23{,}450$; \ $23{,}045$; \ $23{,}504$
\par\vspace{2pt}\noindent
least~\ansblank\quad middle~\ansblank\quad greatest~\ansblank}

\problem[3.5cm]{The digit 4 appears twice in $44{,}175$.

\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Ten-thousands} & \textbf{Thousands} & \textbf{Hundreds} & \textbf{Tens} & \textbf{Ones} \\
\hline
4 & 4 & 1 & 7 & 5 \\
\hline
\end{tabular}
\end{center}

The 4 in the ten-thousands place is how many \emph{times} the value of the 4 in
the thousands place?}

\problem[3.5cm]{Maya says $9{,}876 > 12{,}345$ ``because 9 is greater than 1.''
Write the correct comparison symbol: \ $9{,}876$~~\ansblank~~$12{,}345$.
Then explain in one sentence what Maya forgot to check first.}

\problem[4cm]{A stadium counted $68{,}742$ fans on Saturday and $68{,}472$ fans
on Sunday.
\par\vspace{2pt}\noindent
(a) Which day had the greater attendance?~\ansblank
\par\vspace{2pt}\noindent
(b) How many more fans attended that day?~\ansblank}
