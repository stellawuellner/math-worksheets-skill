\wstitleblock{Linear Representations and Equations --- Telling Slope and $y$-Intercept Apart}{Grade 8 / Pre-Algebra}{}

\noindent\small Slope and $y$-intercept answer two different questions, and
mixing them up is the single most common error in this unit.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Slope $m$} answers ``how much does $y$ change for each $1$ that
        $x$ increases?'' It is a \emph{rate}: $m = \dfrac{\text{change in } y}{\text{change in } x}$.
  \item \textbf{$y$-intercept $b$} answers ``what is $y$ when $x$ is $0$?'' It is
        a single \emph{output}, at one specific input.
  \item In $y = mx + b$ the slope is the number \emph{attached to} $x$ and the
        intercept is the number standing alone --- no matter which one is
        written first.
  \item In each problem: name the error in words first, then show corrected work.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{\textbf{Find and fix.} A line is given by $y = 3x + 5$. Lin says
``the slope is $5$ and the $y$-intercept is $3$,'' then finds $y$ when $x = 4$ by
computing $5(4) + 3 = 23$.
\par\smallskip
\textbf{(a)} State the real slope and the real $y$-intercept.
\par
\textbf{(b)} Find the correct value of $y$ when $x = 4$.}

\problem[5.5cm]{\textbf{Find and fix.} A line is given by $y = 12 - 2x$. Rae says
``the first number is the slope, so $m = 12$ and $b = -2$,'' then finds $y$ when
$x = 5$ by computing $58$.
\par\smallskip
\textbf{(a)} Explain why the order the numbers are written in does not decide
which is which.
\par
\textbf{(b)} Find the correct value of $y$ when $x = 5$.}

\problem[5.5cm]{\textbf{Find and fix.} A line passes through $(2, 3)$ and
$(6, 11)$. Dev writes
$\;m = \dfrac{6 - 2}{11 - 3} = \dfrac{1}{2}$.
\par\smallskip
\textbf{(a)} Say which change Dev put on top, and which one belongs there.
\par
\textbf{(b)} Find the correct slope.}

\problem[6.5cm]{\textbf{Find and fix.} A linear relationship is shown in this
table.
\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $1$ & $2$ & $3$ & $4$ \\ \hline
$y$ & $13$ & $17$ & $21$ & $25$ \\ \hline
\end{tabular}
\end{center}
\par\smallskip
Sofia says ``the rate of change is $13$, because that is the first number in the
$y$ row.''
\par\smallskip
\textbf{(a)} Explain what a rate of change actually measures here.
\par
\textbf{(b)} Find the true change in $y$ for each step of $1$ in $x$.}

\problem[6cm]{\textbf{Find and fix.} Use the same table as problem $4$. Sofia now
says ``the $y$-intercept is $13$, because that is where the table starts.''
\par\smallskip
\textbf{(a)} Say what input the $y$-intercept belongs to, and why the table does
not show it.
\par
\textbf{(b)} Using the step size of $4$ you found in problem $4$, work backwards
one step and give the $y$-intercept.}

\problem[6cm]{\textbf{Find and fix.} A line has slope $4$ and passes through the
point $(3, 20)$. Owen writes $y = 4x + 20$, ``because $20$ is the $y$-value.''
\par\smallskip
\textbf{(a)} Test Owen's equation at $x = 3$ and say what it gives.
\par
\textbf{(b)} Write the equation $4(3) + b = 20$, solve it for $b$, and state the
correct $y$-intercept.}

\problem[7cm]{\textbf{Find and fix.} A pool is draining at a steady rate. After
$2$ minutes it holds $46$ gallons; after $8$ minutes it holds $22$ gallons.
\par\smallskip
A student reports ``the rate of change is $-0.25$ gallons per minute, and the
pool started with $46$ gallons.''
\par\smallskip
\textbf{(a)} Both numbers are wrong for the same underlying reason. Find the
correct rate of change in gallons per minute.
\par
\textbf{(b)} Find how many gallons the pool held at $0$ minutes.}

\problem[7.5cm]{\textbf{Explain.}
\par\smallskip
\textbf{(a)} In your own words, state the question the slope answers and the
question the $y$-intercept answers. Say why one is a rate and the other is a
single value.
\par
\textbf{(b)} Explain why swapping $m$ and $b$ in $y = mx + b$ gives a completely
different line, even though the same two numbers appear.
\par
\textbf{(c)} Describe how you would find the slope and the $y$-intercept in each
of these: a table of values, a graph, and an equation with the constant written
first.
