\sstitleblock{When a Number Rule Stops Working}

\skillheading{Skill 1 --- Testing a rule instead of trusting it}

\begin{formulabox}
\textbf{Rule:} A claim that starts with \emph{every}, \emph{all} or \emph{always} is only safe until one case breaks it.\\[3pt]
A few examples that agree prove nothing. One example that disagrees --- a \textbf{counterexample} --- settles it.\\[3pt]
So: keep generating terms past the ones you were shown, and check each against the claim.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at 5 and adds 6 each time. Someone looks at the first few terms and claims every term is prime. Find the first term that is not prime.
\step{The claim says every term, so I keep listing terms and test each one for factors instead of stopping where the examples stopped.}
\step{The terms are 5, 11, 17, 23, 29, then the 6th term: $5 + 5 \times 6 = 35$, and $35 = 5 \times 7$, so it is not prime.}
First term that breaks the claim: $\ans{35}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at 2 and adds 3 each time. Find the first term that is not prime.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Last digits do not decide primes}

\begin{formulabox}
\textbf{Rule:} A number is prime when its only factors are 1 and itself. The last digit cannot tell you that.\\[3pt]
The last digit rules out some primes (a number ending in 0, 2, 4, 5, 6 or 8 above 5 is never prime), but it can never guarantee one. To settle it, hunt for a factor pair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Someone claims every number ending in 3 is prime. Find the smallest counterexample greater than 3.
\step{To break an ``every'' claim I need one number ending in 3 with a factor pair other than 1 and itself, so I test them in order.}
\step{13 is prime and 23 is prime, but the next one splits: $3 \times 11 = 33$.}
Smallest counterexample: $\ans{33}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Someone claims every number ending in 9 is prime. Find the smallest counterexample greater than 9.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{39}$}
\end{tryitbox}

\skillheading{Skill 3 --- Repairing a pattern that drifted}

\begin{formulabox}
\textbf{Rule:} In an add-the-same-number pattern, every step must be identical.\\[3pt]
term $n$ $=$ first term $+$ (step size) $\times$ ($n - 1$)\\[3pt]
Check a written list by subtracting neighbours: the differences must all match. The first place they do not is where the mistake is, and every term after it inherits the error.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at 3 and adds 5 each time. Someone wrote 3, 8, 13, 20. Find the correct 4th term.
\step{The differences in the written list are 5, 5, then 7, so the list drifted at the last step --- I rebuild that term from the rule rather than from the wrong list.}
\step{The 4th term is 3 steps past the start: $3 + 3 \times 5 = 18$.}
Correct 4th term: $\ans{18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at 7 and adds 4 each time. Someone wrote 7, 11, 15, 20. Find the correct 4th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Terms that share a property --- all odd, all prime so far, all two-digit --- do not tell you the rule that made them. Ask what is \emph{done} to get from one term to the next, then test that on a term nobody showed you.
\end{watchoutbox}
