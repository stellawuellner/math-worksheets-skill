\wstitleblock{Reading the Evidence for Triangle Congruence}{High-school geometry}{}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Matching tick marks mean congruent \emph{sides}; matching arcs mean congruent \emph{angles}. Read the marks before you name anything.
  \item The five criteria are \textbf{SSS}, \textbf{SAS} (the angle must be \emph{between} the two sides), \textbf{ASA} (the side must be \emph{between} the two angles), \textbf{AAS} (the side is \emph{not} between them), and \textbf{HL} (right triangles only: hypotenuse and one leg).
  \item Side-side-angle is not on the list. Two triangles can match that way and still differ.
\end{itemize}
