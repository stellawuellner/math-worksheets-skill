\sstitleblock{Area or Perimeter? How to Tell, and How to Find Each One}

\skillheading{Skill 1 --- Perimeter: add all the way around}

\begin{formulabox}
\textbf{Perimeter} is the distance around the outside of a shape. Walk the whole
edge and \textbf{add} every side you walk along.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Rectangle: long $+$ short $+$ long $+$ short.
  \item Square: four equal sides, so add the side four times.
  \item A perimeter is a \emph{length}, so it is measured in cm, m or in.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A rectangle is $7$ cm long and $2$ cm wide. How much
ribbon goes around it?
\step{``Around it'' means we travel the outside edge, so this is perimeter ---
add the sides instead of multiplying them.}
\step{Two long sides and two short sides: $7 + 2 + 7 + 2$, which is
$14 + 4$.}
The ribbon is $\ans{18\ \text{cm}}$ long.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A rectangle is $9$ cm long and $5$ cm wide. How much ribbon
goes all the way around it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28\ \text{cm}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Area: multiply to cover the inside}

\begin{formulabox}
\textbf{Area} is how many unit squares cover the inside of a shape. In a
rectangle the squares sit in equal rows, so you \textbf{multiply}.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Rectangle: long side $\times$ short side $=$ rows $\times$ squares in a row.
  \item An area counts \emph{squares}, so it is measured in cm$^2$, m$^2$ or in$^2$.
  \item If your answer has no ``square'' in its unit, you did not find an area.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A rectangle is $6$ cm long and $3$ cm wide. How many
one-centimetre squares cover it?
\step{``Cover it'' means we are counting squares inside, so this is area --- and
for a rectangle that means multiply.}
\step{The squares make $3$ rows with $6$ squares in each row: $6 \times 3$.}
It takes $\ans{18\ \text{cm}^2}$ to cover the rectangle.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A rectangle is $7$ cm long and $4$ cm wide. How many
one-centimetre squares cover it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28\ \text{cm}^2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Deciding which one a question wants}

\begin{formulabox}
\textbf{Read the question for its deciding word before you compute.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \emph{Around, fence, ribbon, edging, border, trim} $\rightarrow$ perimeter (add).
  \item \emph{Cover, fill, tiles, paint, carpet, space inside} $\rightarrow$ area (multiply).
  \item Then check the unit: a plain unit means perimeter, a square unit means area.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A garden bed is $12$ m long and $5$ m wide. How much
edging goes around the bed?
\step{The deciding word is ``around'', so the question wants perimeter, not the
space inside the bed.}
\step{Add the four sides: $12 + 5 + 12 + 5$, which is $24 + 10$.}
The edging needed is $\ans{34\ \text{m}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A floor is $6$ m long and $5$ m wide and must be covered
with square-metre tiles. How many tiles are needed?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30\ \text{m}^2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The same rectangle has two different answers, one for
each question --- so ``I got a number'' is never proof you answered the right
one. Two shapes can share a perimeter and still have different areas, which is
why you cannot get one from the other.
\end{watchoutbox}
