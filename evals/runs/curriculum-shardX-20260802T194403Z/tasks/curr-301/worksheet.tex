\wstitleblock{Precise Definitions and Counterexamples}{High School Geometry \quad Definitions, Postulates, and Proof Logic}{}

\noindent\textbf{Two jobs a definition has to do.} \small A precise definition
must (1) be true of \emph{every} object it names and (2) be true of
\emph{nothing else}. A statement that fails job (2) is not a definition --- it is
at best a property, and the way you prove that is to exhibit one object that
satisfies the statement and is not the thing: a \textbf{counterexample}. One
counterexample is enough.\normalsize

\smallskip
\noindent\textbf{Model: definitions become coordinate tests.} Every definition
below can be checked with arithmetic, which is what makes a counterexample
something you can \emph{compute} rather than argue about.

\begin{center}\small
\begin{tabular}{|p{3.0cm}|p{5.2cm}|p{4.6cm}|}
\hline
\textbf{Word} & \textbf{Precise definition} & \textbf{Coordinate test (notation)} \\
\hline
midpoint of $\overline{AB}$ & the point $M$ on $\overline{AB}$ with $AM = MB$ &
$M = \left(\dfrac{x_1+x_2}{2},\ \dfrac{y_1+y_2}{2}\right)$ \\
\hline
congruent segments & segments of equal length &
$AB = CD$, each by $\sqrt{(\Delta x)^2 + (\Delta y)^2}$ \\
\hline
parallel lines & coplanar lines with no point in common &
slopes equal: $m_1 = m_2$ \\
\hline
perpendicular lines & lines meeting at a right angle &
$m_1 \cdot m_2 = -1$ \\
\hline
\end{tabular}
\end{center}

\noindent\textbf{Directions.} Use the definition, not the picture in your head.
Show the arithmetic that settles each question, and write your final answer on
the answer line.

\problem[5cm]{A student writes: ``The midpoint of a segment is a point near the
middle of it.''

\smallskip
Use the \emph{precise} definition --- the point $M$ on $\overline{AB}$ with
$AM = MB$ --- to find the midpoint of $\overline{AB}$ where $A(-6, 2)$ and
$B(4, 8)$. Then state in one sentence why the student's wording fails to name
one particular point.}

\problem[5.5cm]{By definition, a triangle is \emph{isosceles} when at least two
of its sides are congruent. Triangle $PQR$ has $P(0,0)$, $Q(6,0)$, $R(3,4)$, and
$PQ = 6$.

\smallskip
Compute $PR$. Then decide whether $\triangle PQR$ is isosceles, and say which
wording --- ``at least two congruent sides'' or ``exactly two congruent
sides'' --- decides whether an equilateral triangle counts as isosceles.}

\problem[6cm]{Kai writes: ``The perpendicular bisector of $\overline{AB}$ is the
line through the midpoint of $\overline{AB}$.''

\smallskip
Let $A(-4, 1)$ and $B(2, 5)$, so the midpoint is $M(-1, 3)$ and the slope of
$\overline{AB}$ is $\tfrac{2}{3}$. Kai draws $\overleftrightarrow{CM}$ through
$C(3, -2)$ and $M$.

\smallskip
Compute the slope of $\overleftrightarrow{CM}$. Is the product of the two slopes
$-1$? Explain what your answer proves about Kai's statement.}

\problem[6cm]{Consider the true statement: ``If a quadrilateral is a square,
then it has four right angles.''

\smallskip
\textbf{(a)} Write its converse. \quad \textbf{(b)} Rectangle $PQRS$ has
$P(0,0)$, $Q(6,0)$, $R(6,3)$, $S(0,3)$; all four of its angles are right angles
and $PQ = 6$. Compute $QR$. \quad \textbf{(c)} Explain why $PQRS$ makes the
converse false.}

\problem[6cm]{Consider the claim: ``If two segments have the same length, then
they are parallel.''

\smallskip
$\overline{AB}$ runs from $A(0,0)$ to $B(3,4)$ and $\overline{CD}$ runs from
$C(0,0)$ to $D(-4,3)$. Both have length $5$, and the slope of $\overline{AB}$ is
$\tfrac{4}{3}$.

\smallskip
Compute the slope of $\overline{CD}$, then explain in one sentence why these two
segments form a counterexample.}

\problem[6cm]{By definition, two non-vertical lines are perpendicular exactly
when the product of their slopes is $-1$.

\smallskip
Line 1 passes through $(0,0)$ and $(4,3)$, so its slope is $\tfrac{3}{4}$.
Line 2 passes through $(0,0)$ and $(3, m)$.

\smallskip
Find the value of $m$ that makes Line 2 perpendicular to Line 1. Set up the
defining equation before you solve it.}

\problem[6.5cm]{A student writes: ``A rhombus is a quadrilateral whose diagonals
are perpendicular.''

\smallskip
Quadrilateral $WXYZ$ has $W(0,0)$, $X(3,4)$, $Y(0,10)$, $Z(-3,4)$. Its diagonal
$\overline{WY}$ is vertical and its diagonal $\overline{XZ}$ is horizontal, so
the diagonals are perpendicular. Its side lengths are $WX = 5$, $XY = \sqrt{45}$,
$YZ = \sqrt{45}$, $ZW = 5$.

\smallskip
\textbf{(a)} Explain why $WXYZ$ satisfies the student's statement but is not a
rhombus. \quad \textbf{(b)} Write a precise definition of \emph{rhombus}.
\quad \textbf{(c)} The student's statement is still true of every rhombus. What
is it, if not a definition?
\noansline}
