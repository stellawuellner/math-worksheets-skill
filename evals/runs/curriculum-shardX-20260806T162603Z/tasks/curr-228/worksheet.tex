\wstitleblock{Solving Systems Without a Graph}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Substitution, elimination, and choosing between them}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Name your two variables and what each one stands for before you write
        any equations.
  \item Use \emph{substitution} when one equation is already solved for a
        variable; use \emph{elimination} when the same term appears in both
        equations.
  \item Give both values with the units the story uses, and check them in both
        equations.
\end{itemize}

\vspace{0.15cm}

\problem[6cm]{A school trip has 12 people in all: $a$ adults and $c$ children,
so $a + c = 12$. There are 4 more adults than children, so $a = c + 4$. Find
the number of adults and the number of children.}

\problem[6cm]{At a snack bar, 2 hot dogs and 1 drink cost \$11, so
$2h + d = 11$; 2 hot dogs and 3 drinks cost \$17, so $2h + 3d = 17$. Here $h$
is the price of a hot dog and $d$ the price of a drink, both in dollars. Find
each price.}

\problem[6.5cm]{A school buys $p$ pens and $n$ notebooks, 30 items in total,
so $p + n = 30$. Pens cost \$2 each and notebooks \$5 each, and the bill was
\$90, so $2p + 5n = 90$. Find how many of each were bought.}

\problem[6.5cm]{A courier weighs two loads. Two large boxes and three small
ones weigh 31 kg: $2l + 3s = 31$. Two large boxes and five small ones weigh
41 kg: $2l + 5s = 41$. Here $l$ and $s$ are the masses in kilograms of one
large and one small box. Find each mass.}

\problem{At a cinema, 4 adult tickets and 3 child tickets cost \$71, so
$4a + 3c = 71$; 2 adult tickets and 3 child tickets cost \$49, so
$2a + 3c = 49$. Prices are in dollars.
\begin{itemize}
  \item[(a)] Find the price of an adult ticket and of a child ticket.
        \par\vspace*{3.2cm}\ansline
  \item[(b)] Explain which method you used and why it was the quicker one
        here.
        \par\vspace*{2cm}\ansline
\end{itemize}}

\problem[7cm]{A carpenter cuts a 26-metre board into two pieces of lengths $x$
and $y$ metres, so $x + y = 26$. The longer piece is 2 metres more than twice
the shorter one, so $y = 2x + 2$. Find the length of each piece.}

\problem[8cm]{At a bakery, 3 bagels and 2 muffins cost \$17, so
$3b + 2m = 17$; 2 bagels and 5 muffins cost \$26, so $2b + 5m = 26$. Prices
are in dollars. Find the price of a bagel and the price of a muffin.}

\problem{Gym P charges $y = 12x + 30$ dollars for $x$ months of membership;
gym Q charges $y = 18x$ dollars.
\begin{itemize}
  \item[(a)] Find the (months, cost) pair at which the two gyms charge the
        same.
        \par\vspace*{3.2cm}\ansline
  \item[(b)] Explain which method you used and why it suits this pair of
        equations.
        \par\vspace*{2cm}\ansline
\end{itemize}}

\problem{A caterer quotes two orders. Two trays of fruit and three trays of
bread cost \$46, so $2f + 3b = 46$. Four trays of fruit and six trays of bread
cost \$100, so $4f + 6b = 100$. Prices are in dollars per tray.
\begin{itemize}
  \item[(a)] Solve the system, or state that it has no solution.
        \par\vspace*{3.4cm}\ansline
  \item[(b)] Explain what your answer means about the caterer's two quotes.
        \par\vspace*{2cm}\ansline
\end{itemize}}

\problem{A boat travels on a river. Going downstream the current helps it, and
it covers 20 km in one hour, so $b + c = 20$. Going upstream the current holds
it back, and it covers 12 km in one hour, so $b - c = 12$. Here $b$ is the
boat's own speed and $c$ the current's speed, both in kilometres per hour.
\begin{itemize}
  \item[(a)] Find the boat's speed and the current's speed.
        \par\vspace*{3.2cm}\ansline
  \item[(b)] Explain what makes adding the two equations a good first move
        here.
        \par\vspace*{2cm}\ansline
\end{itemize}}
