\wstitleblock{Three Kinds of Evidence About a Limit}{AP Calculus AB/BC \quad Limits and continuity}{}

\begin{center}
\textit{Reading limits from a graph, estimating them from a table of values, and
confirming them with algebra}
\end{center}

\noindent\textbf{Before you start.} A limit claim can be supported three ways,
and a good answer says which one you used.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Graphical} --- follow the curve in from each side and see what
height it heads for. A filled dot tells you the function \emph{value}; it does
not tell you the limit.
\item \textbf{Numerical} --- evaluate at inputs closing in from both sides. A
table suggests a limit; it never proves one.
\item \textbf{Symbolic} --- factor, rationalize, or clear the compound
fraction, then substitute. This is the one that settles the matter.
\end{itemize}

\problem[9cm]{\textbf{Synthesis.} Let $\displaystyle p(x)=\frac{x^2+x-6}{x-2}$
for $x\neq 2$, with $p(2)$ not yet defined.

\smallskip

\smallskip
(b) Find $\displaystyle\lim_{x\to 2}p(x)$ algebraically, and state the value of

\smallskip

sentences saying how the table, the algebra and your sketch agree.}
