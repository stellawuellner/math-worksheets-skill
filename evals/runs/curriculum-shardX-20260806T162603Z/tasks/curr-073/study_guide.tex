\sstitleblock{Division: Finding the Unknown Factor}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Solving a missing-factor equation}

\begin{formulabox}
\textbf{Rule:}\quad To solve $b \times n = $ total, ask: ``$b$ times \emph{what} makes the total?''\par
Run through the $b$ times table until you land on the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $6 \times n = 30$.
\step{Ask: 6 times what makes 30? Skip count by 6 until you reach 30.}
\step{$6, 12, 18, 24, 30$ --- that is five 6s, so $6 \times 5 = 30$.}

$\Rightarrow\quad\ans{n = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $4 \times n = 28$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Division as an unknown factor}

\begin{formulabox}
\textbf{Rule:}\quad total $\div\ b$ asks the same question as $b \times \textbf{?} = $ total.\par
A related multiplication fact you already know gives the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $45 \div 5$.
\step{Rewrite the division as a missing-factor question: $5 \times \textbf{?} = 45$.}
\step{From the fives table, $5 \times 9 = 45$, so $45 \div 5 = 9$.}

$\Rightarrow\quad\ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $32 \div 4$ by thinking $4 \times \textbf{?} = 32$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Equal-group story problems}

\begin{formulabox}
\textbf{Rule:}\quad Find the total and the given group number first.\par
total $\div$ number in each group $=$ how many groups;\quad total $\div$ number of groups $=$ how many in each.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} There are 28 pencils, packed 4 to a pack. How many packs?
\step{The total is 28 and each group holds 4, so ask: 4 times what makes 28?}
\step{$4 \times 7 = 28$, so $28 \div 4 = 7$.}

$\Rightarrow\quad\ans{7}$ packs
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} There are 54 books shared equally on 6 shelves. How many books on each shelf?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Read carefully \emph{which} number the story gives you --- the number of groups or the number in each group. The division is the same, but the unit of your answer (bags? apples?) comes from what the question asks.
\end{watchoutbox}
