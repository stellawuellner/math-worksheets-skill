\sstitleblock{Solving Trigonometric Equations on a Restricted Interval}

\skillheading{Skill 1 --- Linear equations in one ratio}

\begin{formulabox}
\textbf{Method:} isolate the ratio, find the reference angle from the positive value, then place one solution in \emph{each} quadrant the sign allows.\par
Sine positive: I, II. \ Sine negative: III, IV. \ Cosine positive: I, IV. \ Cosine negative: II, III. \ Tangent positive: I, III. \ Tangent negative: II, IV.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\sin\theta - \sqrt{3} = 0$ for $0^\circ \le \theta < 360^\circ$.
\step{One ratio, first power --- so isolate it and read off the two quadrants where that sign occurs.}
\step{$\sin\theta = \dfrac{\sqrt{3}}{2}$: reference angle $60^\circ$, sine positive in I and II.}
$\Rightarrow\quad \theta = \ans{60}^\circ$ or $\theta = \ans{120}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\cos\theta + \sqrt{2} = 0$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{135}^\circ$ or $\ans{225}^\circ$}
\end{tryitbox}

\skillheading{Skill 2 --- Equations that factor}

\begin{formulabox}
\textbf{Rule:} a product is zero exactly when one of its factors is zero. Set each factor to zero and solve it as a separate linear equation, then merge the lists.\par
\textbf{Never divide by a trigonometric factor} --- dividing by $\cos\theta$ throws away every solution with $\cos\theta = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\cos\theta\,(2\cos\theta + 1) = 0$ for $0^\circ \le \theta < 360^\circ$.
\step{A product equals zero, so split it into two easier equations rather than expanding.}
\step{$\cos\theta = 0 \Rightarrow 90^\circ, 270^\circ$; \ $\cos\theta = -\dfrac{1}{2} \Rightarrow 120^\circ, 240^\circ$.}
$\Rightarrow\quad \theta = \ans{90}^\circ, \ans{120}^\circ, \ans{240}^\circ, \ans{270}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sin\theta\,(2\sin\theta + 1) = 0$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{0}^\circ, \ans{180}^\circ, \ans{210}^\circ, \ans{330}^\circ$}
\end{tryitbox}

\skillheading{Skill 3 --- Identity first, then factor}

\begin{formulabox}
\textbf{When two different ratios appear}, use $\sin^2\theta + \cos^2\theta = 1$ to rewrite the squared one, leaving a quadratic in a single ratio. Then factor and use Skill 2.\par
$\sin^2\theta = 1 - \cos^2\theta$ \qquad $\cos^2\theta = 1 - \sin^2\theta$
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\sin^2\theta + \cos\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.
\step{Two ratios in one equation cannot be solved directly, so trade the squared sine for cosine and get a quadratic in cosine alone.}
\step{$2(1 - \cos^2\theta) + \cos\theta - 1 = -(2\cos\theta + 1)(\cos\theta - 1)$, so $\cos\theta = -\dfrac{1}{2}$ or $\cos\theta = 1$.}
$\Rightarrow\quad \theta = \ans{0}^\circ, \ans{120}^\circ, \ans{240}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\cos^2\theta + \sin\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{90}^\circ, \ans{210}^\circ, \ans{330}^\circ$}
\end{tryitbox}

\skillheading{Skill 4 --- Multiple angles: widen first}

\begin{formulabox}
\textbf{For $\sin(k\theta)$ or $\cos(k\theta)$:} substitute $u = k\theta$ and multiply the interval's endpoints by $k$. Solve for $u$ over that wider interval, then divide every answer by $k$.\par
Expect about $k$ times as many solutions --- if you get the same number as a plain equation, you widened too late.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\cos 2\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.
\step{The angle inside is doubled, so the inside variable sweeps twice as far --- widen the interval before solving, not after.}
\step{$u = 2\theta$ on $[0^\circ, 720^\circ)$, $\cos u = \dfrac{1}{2} \Rightarrow u = 60^\circ, 300^\circ, 420^\circ, 660^\circ$; halve each.}
$\Rightarrow\quad \theta = \ans{30}^\circ, \ans{150}^\circ, \ans{210}^\circ, \ans{330}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\sin 3\theta - 1 = 0$ for $0^\circ \le \theta < 180^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{10}^\circ, \ans{50}^\circ, \ans{130}^\circ, \ans{170}^\circ$}
\end{tryitbox}
