\sstitleblock{Reading and Sketching Slope Fields}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading a Slope from the Field}

\begin{formulabox}
\textbf{What a field is.} At each lattice point $(x, y)$ the picture draws a
short segment whose slope is the value of $\dfrac{dy}{dx}$ there. Nothing is
solved --- the equation is simply evaluated, one point at a time.\par
\textbf{Two tells.} Parallel segments across a \emph{row} mean the slope depends
only on $y$; parallel down a \emph{column} means it depends only on $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\dfrac{dy}{dx} = x + y$, find the slope at $(2, 1)$.
\step{A slope field asks nothing more than the value of the right-hand side, so substitute the coordinates straight into the equation.}
\step{$\dfrac{dy}{dx} = 2 + 1$}

$\Rightarrow\quad \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same equation. Find the slope at $(3, -1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Sketching a Solution Curve}

\begin{formulabox}
\textbf{Rule.} Start at the given point and move so the curve is always
\emph{tangent} to the segments beneath it. Never cut across a segment, and never
let two solution curves touch.\par
\begin{center}
\begin{tikzpicture}[scale=0.4]
  \draw[gray!45, thin] (-2.2,-2.2) grid (2.2,2.2);
  \foreach \x in {-2,-1,0,1,2} \foreach \y in {-2,-1,0,1,2} {
    \pgfmathsetmacro{\m}{\x/2}
    \pgfmathsetmacro{\hx}{0.36/sqrt(1+(\m)*(\m))}
    \pgfmathsetmacro{\hy}{\m*\hx}
    \draw[blue!65!black, thick] ({\x-\hx},{\y-\hy}) -- ({\x+\hx},{\y+\hy});
  }
  \draw[red, thick] plot[domain=-2:2, samples=40] (\x,{\x*\x/4-1});
\end{tikzpicture}
\end{center}
The red curve is one solution of $\dfrac{dy}{dx} = \dfrac{x}{2}$: flat where the
segments are flat, steeper where they are steeper.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad To sketch a solution of $\dfrac{dy}{dx} = \dfrac{x}{2}$,
find the slopes at $x = 2$ and $x = 4$.
\step{A sketch is only as good as the slopes it is built from, so read a few off the equation before drawing anything.}
\step{$\dfrac{2}{2}$ and $\dfrac{4}{2}$}

$\Rightarrow\quad \ans{1}$ \quad and \quad $\ans{2}$: the curve doubles its steepness between them.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\dfrac{dy}{dx} = \dfrac{x}{3}$, find the slopes at
$x = 3$ and $x = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$, $\ans{2}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From the Field to a Formula}

\begin{formulabox}
\textbf{When the slope depends only on $x$,} the equation is just an
antiderivative: $\dfrac{dy}{dx} = f(x)$ gives $y = F(x) + C$, and every solution
curve in the field is a vertical shift of one another.\par
A point on the curve fixes $C$: substitute the coordinates and solve.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{dy}{dx} = 3x^2$ through the point
$(1, 5)$.
\step{The slope depends on $x$ alone, so antidifferentiate first and let the given point pin down the constant.}
\step{$y = x^3 + C$, and $5 = 1 + C$.}

$\Rightarrow\quad \ans{y = x^3 + C}$ \quad with \quad $\ans{C = 4}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{dy}{dx} = 4x^3$ through $(1, 7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = x^4 + C}$, $\ans{C = 6}$}
\end{tryitbox}
