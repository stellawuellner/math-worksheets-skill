\begin{center}
Power, product, and quotient rules for derivatives
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Decide which rule the expression calls for \emph{before} you differentiate. A sum of powers needs only the power rule; a product of two non-constant factors needs the product rule; one expression divided by another needs the quotient rule.
  \item Rewrite negative and fractional powers first: $\dfrac{5}{x} = 5x^{-1}$ and $\sqrt{x} = x^{1/2}$.
  \item Simplify your answer, and keep a quotient-rule result over a single squared denominator.
\end{itemize}

\problem[4.6cm]{Differentiate: \quad $y = 4x^5 - 3x^2 + 7$}

\problem[5cm]{Differentiate: \quad $y = 2x^3 + \dfrac{5}{x}$
\quad {\small (rewrite the second term as a power first)}}

\problem[5.4cm]{Differentiate: \quad $y = \left(x^2 + 1\right)(3x - 4)$
\quad {\small (product rule; then expand and collect)}}

\problem[5.6cm]{Differentiate: \quad $y = \dfrac{2x + 1}{x - 3}$
\quad {\small (quotient rule; leave the denominator squared)}}

\problem[5.6cm]{Differentiate: \quad $y = \sqrt{x} + \dfrac{6}{x^3}$}

\problem[6cm]{Differentiate: \quad $y = \left(3x^2 - 1\right)\left(x^3 + 2x\right)$
\quad {\small (product rule, then collect like terms)}}

\problem[6cm]{Differentiate: \quad $y = \dfrac{x^2}{x^2 + 1}$
\quad {\small (quotient rule; the numerator simplifies a long way)}}
