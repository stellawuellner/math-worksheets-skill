\wstitleblock{Composing Piecewise and Nonlinear Functions}{Precalculus}{}

\begin{center}
\small\textit{Evaluating a composition, writing the composed rule, and solving a composition equation}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $(f \circ g)(x)$ means $f\bigl(g(x)\bigr)$: the inner function runs first.
  \item With a piecewise rule, the branch is chosen by the number going \emph{into} that rule --- check it every time you use the rule.
  \item Simplify composed rules fully, and give exact solutions.
\end{itemize}
\vspace{0.15cm}

\problem[4.5cm]{Let $f(x) = x^{2} + 1$ and $g(x) = 3x - 2$. Find $(f \circ g)(4)$.}

\problem[5cm]{Let $p(x) = 2x + 5$ and $q(x) = x^{2} - 3$. Write $(p \circ q)(x)$ as a simplified polynomial.}

\problem[5.5cm]{Let $u(x) = x - 4$ and $v(x) = x^{2}$. Solve $(v \circ u)(x) = 49$.}

\problem[0pt]{Let $g(x) = x^{2} - 5$ and
\[ f(x) = \begin{cases} 2x + 1, & x < 3 \\[2pt] x^{2} - 4, & x \geq 3 \end{cases} \]
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $(f \circ g)(4)$.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Which rule for $f$ did you use, and what comparison told you to use it?
  \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = \dfrac{2}{x - 1}$ and $g(x) = \dfrac{1}{x}$. Write $(f \circ g)(x)$ as a single simplified fraction.}

\problem[6cm]{Let $f(x) = \sqrt{x + 4}$ and $g(x) = 2x - 1$. Solve $(f \circ g)(x) = 5$.}

\problem[0pt]{Let $f(x) = 3x - 2$ and
\[ g(x) = \begin{cases} -x, & x \leq 0 \\[2pt] x^{2}, & x > 0 \end{cases} \]
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $(f \circ g)(-5)$.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Find $(g \circ f)(-5)$.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = x^{2} - 6x$ and $g(x) = x + 3$. Write $(f \circ g)(x)$ as a simplified polynomial.}

\problem[6cm]{Let $f(x) = x^{2} + 2x$ and $g(x) = x - 3$. Solve $(f \circ g)(x) = 0$.}

\problem[0pt]{Let \[ f(x) = \begin{cases} x + 7, & x < 2 \\[2pt] 3x, & x \geq 2 \end{cases} \]
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $(f \circ f)(-1)$.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Describe why the two passes through $f$ do not use the same rule.
  \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[6.5cm]{Let $f(x) = \dfrac{x}{x + 1}$ and $g(x) = \dfrac{1}{x - 2}$. Write $(f \circ g)(x)$ as a single simplified fraction.}

\problem[0pt]{Let $f(x) = \sqrt{x}$ and $g(x) = x^{2} - 3x$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Solve $(f \circ g)(x) = 2$.
  \par\vspace*{3.4cm}\ansline
  \item[(b)] Explain why neither of your answers is extraneous, even though one of them is negative.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}
