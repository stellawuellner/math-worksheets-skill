\sstitleblock{Identifying Conics and Their Key Features}

\skillheading{Skill 1 --- Naming a conic from its equation}

\begin{formulabox}
\textbf{Put both squared terms on one side} and call their coefficients $A$ (for $x^2$) and
$C$ (for $y^2$). Then the product $A \cdot C$ names the curve: positive and equal
coefficients means a circle, positive and unequal means an ellipse, negative means a
hyperbola. If one squared term is missing altogether, it is a parabola.\par
\textbf{Why:} same signs let the two terms trade off against a fixed total, so the curve
closes; opposite signs let one grow without bound, so it opens into two branches.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Name the conic $3x^2 + 3y^2 = 27$.
\step{Both squared terms are present, so the sign test decides it --- multiply the two coefficients before looking at anything else.}
\step{$A = 3$ and $C = 3$, so $A \cdot C = 9$, which is positive, and the coefficients are equal.}

$\Rightarrow\quad A \cdot C = \ans{9}$, positive with $A = C$: a \textbf{circle} (radius $3$ after dividing through).
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compute $A \cdot C$ for $5x^2 - 2y^2 = 20$ and name the conic.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-10}$, negative, so a hyperbola}
\end{tryitbox}

\skillheading{Skill 2 --- Centre and radius of a circle}

\begin{formulabox}
\textbf{Standard form:} $(x-h)^2 + (y-k)^2 = r^2$, centre $(h, k)$, radius $r$. The
right-hand side is $r^2$, so take a square root before reporting the radius.\par
\textbf{From the expanded form} $x^2 + y^2 + px + qy + f = 0$: halve and negate each linear
coefficient to get the centre, then $r^2 = (p/2)^2 + (q/2)^2 - f$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the centre and radius of $x^2 + y^2 - 4x + 8y + 4 = 0$.
\step{The squared terms have no visible centre, so complete the square in each variable --- half of each linear coefficient is what moves inside the bracket.}
\step{Half of $-4$ is $-2$ and half of $8$ is $4$: \ $(x-2)^2 + (y+4)^2 = -4 + 4 + 16$.}

$\Rightarrow\quad$ centre $\bigl(\ans{2},\, \ans{-4}\bigr)$, \quad radius $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the centre and radius of $x^2 + y^2 - 6x + 2y - 6 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: centre $\bigl(\ans{3}, \ans{-1}\bigr)$, radius $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- The vertex of a parabola}

\begin{formulabox}
\textbf{Axis of symmetry first:} for $y = ax^2 + bx + c$ the vertex sits at
$x = -\dfrac{b}{2a}$. Substitute that $x$ back into the equation to get the $y$-coordinate ---
never guess it from the constant term.\par
\textbf{Vertex form} $y = a(x-h)^2 + k$ then reads the vertex off directly as $(h, k)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the vertex of $y = x^2 - 4x + 7$.
\step{The vertex lies on the axis of symmetry, and that axis depends only on $a$ and $b$, so compute it first and let the equation supply the height.}
\step{$x = -\dfrac{-4}{2(1)} = 2$, then $y = 2^2 - 4(2) + 7 = 4 - 8 + 7$.}

$\Rightarrow\quad$ vertex $\bigl(\ans{2},\, \ans{3}\bigr)$, so $y = (x-2)^2 + 3$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the vertex of $y = x^2 + 10x + 21$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\bigl(\ans{-5},\, \ans{-4}\bigr)$}
\end{tryitbox}

\skillheading{Skill 4 --- Ellipse and hyperbola features}

\begin{formulabox}
\textbf{Ellipse} $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$: vertices at distance $a$ along the
axis with the larger denominator, and foci at $c$ with $c^2 = a^2 - b^2$ --- \emph{subtract},
because the foci sit inside.\par
\textbf{Hyperbola} $\dfrac{y^2}{a^2} - \dfrac{x^2}{b^2} = 1$: the positive term names the
opening direction, vertices at distance $a$, asymptote slopes $\pm a/b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\dfrac{x^2}{16} + \dfrac{y^2}{7} = 1$, find the semi-major axis and the focal distance.
\step{The larger denominator sits under $x^2$, so the long axis is horizontal --- that decides which letter is $a$ before any arithmetic.}
\step{$a = \sqrt{16}$, and $c = \sqrt{16 - 7} = \sqrt{9}$.}

$\Rightarrow\quad a = \ans{4}, \qquad c = \ans{3}$ \quad (foci inside the vertices, as required)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\dfrac{y^2}{9} - \dfrac{x^2}{16} = 1$, find the vertex distance and the positive asymptote slope.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ and $\ans{\tfrac{3}{4}}$}
\end{tryitbox}
