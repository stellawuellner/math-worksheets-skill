\sstitleblock{Precise Definitions and Counterexamples}

\skillheading{1. Turn a definition into a coordinate test (test-a-definition-with-coordinates)}

\begin{formulabox}
\textbf{A definition you can compute with is a definition you can check.}\\[3pt]
\begin{tabular}{@{}p{4.0cm}@{}p{9.3cm}@{}}
congruent segments & equal length: $AB = \sqrt{(\Delta x)^2 + (\Delta y)^2}$\\
midpoint of $\overline{AB}$ & $M = \left(\frac{x_1+x_2}{2},\ \frac{y_1+y_2}{2}\right)$ --- the one point with $AM = MB$\\
parallel & slopes equal: $m_1 = m_2$\\
perpendicular & slopes multiply to $-1$: $m_1 m_2 = -1$\\
\end{tabular}\\[3pt]
Read the definition, name the quantity it talks about, compute that quantity.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $\triangle PQR$ isosceles? $P(0,0)$, $Q(6,0)$, $R(3,4)$.
\step{``Isosceles'' is defined by \emph{at least two congruent sides}, and
congruent means equal length --- so this is a distance-formula question, not a
looks-like-it question.}
\step{$PR = \sqrt{(3-0)^2+(4-0)^2} = \sqrt{9+16} = \ans{PR = 5}$}
\step{$QR = \sqrt{(3-6)^2+4^2} = 5$ too, and $PQ = 6$: two sides match, so yes,
isosceles.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\triangle ABC$ has $A(-5,0)$, $B(5,0)$, $C(0,12)$. Compute
$AC$ to test whether it is isosceles.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{AC = 13}$}
\end{tryitbox}

\skillheading{2. Kill a claim with one counterexample (find-a-counterexample)}

\begin{formulabox}
\textbf{A counterexample to ``if $P$ then $Q$'' is one object where $P$ is true
and $Q$ is false.} One is enough --- you never need a second, and you never need
to explain why the claim ``usually'' fails.\\[3pt]
\textbf{Where to hunt:} the claim names one property; look for an object with
that property and nothing else. Equal \emph{lengths} say nothing about
\emph{direction}; equal \emph{angles} say nothing about \emph{size}.\\[2pt]
\textbf{Converse warning:} ``if $P$ then $Q$'' being true tells you nothing about
``if $Q$ then $P$''. Each needs its own proof or its own counterexample.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Disprove: ``If two segments have the same length, they are
parallel.'' $\overline{AB}$: $(0,0)$ to $(6,8)$, length $10$, slope $\frac{4}{3}$.
$\overline{CD}$: $(0,0)$ to $(8,6)$, length $10$.
\step{Both segments already satisfy the hypothesis (equal length), so all that is
left is to show the conclusion fails --- compare slopes.}
\step{slope of $\overline{CD} = \dfrac{6-0}{8-0} = \ans{\tfrac{3}{4}}$}
\step{$\tfrac{3}{4} \ne \tfrac{4}{3}$, so they are not parallel. One object,
claim dead.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\overline{EF}$: $(0,0)$ to $(5,12)$ and $\overline{GH}$:
$(0,0)$ to $(12,5)$ both have length $13$. Compute the slope of $\overline{GH}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{12}}$}
\end{tryitbox}

\skillheading{3. Repair a loose definition (write-a-precise-definition)}

\begin{formulabox}
\textbf{A definition must do two jobs:} (1) be true of every object it names,
and (2) be true of \emph{nothing else}. A statement that only does job (1) is a
\textbf{property} (a theorem), not a definition.\\[3pt]
\textbf{Repair recipe:} name the general category, then add exactly the
conditions that rule everything else out --- and test the repair by hunting for
one more counterexample.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad ``The perpendicular bisector of $\overline{AB}$ is a line
through the midpoint of $\overline{AB}$.'' Test it with $A(0,0)$, $B(8,4)$,
midpoint $M(4,2)$, and the line through $M$ and $P(6,5)$.
\step{To show the wording admits too much, find one line that fits it and is not
the perpendicular bisector --- so compute the candidate's slope.}
\step{slope of $\overleftrightarrow{MP} = \dfrac{5-2}{6-4} = \ans{\tfrac{3}{2}}$}
\step{$\overline{AB}$ has slope $\tfrac{1}{2}$, so perpendicular would need
$-2$. The line goes through $M$ but is not perpendicular: the wording fails job
(2). \textbf{Repair:} the line through the midpoint of $\overline{AB}$
\emph{and perpendicular to} $\overline{AB}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad ``A median goes from a vertex to the opposite side'' is too
loose. In $\triangle ABC$ with $A(-2,0)$, $B(8,0)$, $C(1,7)$, find the midpoint
of $\overline{AB}$ --- the endpoint the real median must hit.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\,0)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Not a definition'' does not mean ``false''. Every
rhombus really does have perpendicular diagonals --- that statement is a true
property. It fails only as a \emph{definition}, because kites have perpendicular
diagonals too.
\end{watchoutbox}
