\sstitleblock{Multiplying Fractions and Mixed Numbers}

\skillheading{Skill 1 --- Multiply fractions (and whole numbers)}

\begin{formulabox}
\textbf{Rule:} \ $\dfrac{a}{b} \times \dfrac{c}{d} = \dfrac{a \times c}{b \times d}$
--- multiply straight across. No common denominator is needed.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A whole number $n$ is $\dfrac{n}{1}$, so the same rule covers it.
  \item Cancel a common factor from any top with any bottom \emph{before} you
        multiply --- the numbers stay small and the answer lands simplified.
  \item ``of'' means $\times$: $\dfrac{2}{3}$ of a cup is $\dfrac{2}{3} \times 1$ cup.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\dfrac{4}{9} \times \dfrac{3}{8}$
\step{Two fractions multiplied: multiply across, but cancel first so nothing needs
      reducing at the end.}
\step{3 divides into 9 and 4 divides into 8: $\dfrac{1}{3} \times \dfrac{1}{2}
      = \dfrac{1}{6}$.}
$\Rightarrow\quad \ans{\,\dfrac{1}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $6 \times \dfrac{2}{9}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\,\dfrac{4}{3}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Multiply mixed numbers}

\begin{formulabox}
\textbf{Rule:} \ Convert first, multiply second:
\ $a\dfrac{b}{c} = \dfrac{a \times c + b}{c}$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Never multiply the whole parts and the fraction parts separately.
  \item Estimate to check: $2\frac{3}{4} \times 1\frac{1}{2}$ must be a bit more
        than $3 \times 1$, so an answer near 4 is believable.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $1\dfrac{2}{3} \times 2\dfrac{1}{4}$
\step{Mixed numbers cannot be multiplied part by part, so rewrite both as improper
      fractions and then use the across rule.}
\step{$1\dfrac{2}{3} = \dfrac{5}{3}$ and $2\dfrac{1}{4} = \dfrac{9}{4}$;
      cancel 3 into 9: $\dfrac{5}{1} \times \dfrac{3}{4} = \dfrac{15}{4}$.}
$\Rightarrow\quad \ans{\,\dfrac{15}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $3\dfrac{1}{2} \times \dfrac{4}{7}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Find a missing factor}

\begin{formulabox}
\textbf{Rule:} \ If $\dfrac{a}{b} \times x = p$, then
$x = p \div \dfrac{a}{b} = p \times \dfrac{b}{a}$ --- divide by a fraction by
multiplying by its \textbf{reciprocal} (flip it).\\[3pt]
Always check by multiplying your answer back into the original sentence.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $\dfrac{2}{5} \times x = \dfrac{6}{25}$
\step{One factor and the product are known, so undo the multiplication --- divide the
      product by the known factor.}
\step{$x = \dfrac{6}{25} \times \dfrac{5}{2} = \dfrac{3}{5}$, and
      $\dfrac{2}{5} \times \dfrac{3}{5} = \dfrac{6}{25}$ $\checkmark$}
$\Rightarrow\quad \ans{x = \dfrac{3}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $x \times 1\dfrac{1}{2} = \dfrac{9}{4}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \dfrac{3}{2}}$}
\end{tryitbox}
