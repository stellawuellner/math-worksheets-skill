\sstitleblock{Seeing Small Numbers Without Counting}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Knowing the Dot Patterns}

\begin{formulabox}
\textbf{Rule:} Some arrangements are pictures of a number: a pair is 2, a
triangle is 3, a square is 4, and the die's X is 5. When you know the
pattern, you know the number \emph{without counting}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A card shows dots in a square arrangement --- one dot
at each corner. How many?
\step{Name the arrangement first: a square of dots is the picture of four.}
\step{One dot per corner of a square --- the card shows $\ans{4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A card shows four corner dots \emph{plus} one dot in
the middle --- the die pattern. How many?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Seeing the Parts in a Pattern}

\begin{formulabox}
\textbf{Rule:} A new pattern is easier when you spot \textbf{small parts} you
already know, then put the parts together.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A domino shows a triangle of dots on one half and a
pair on the other. How many in all?
\step{Spot the two known parts: the triangle is 3 and the pair is 2.}
\step{Put the parts together: $3 + 2 = \ans{5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A card shows a triangle of dots and one extra dot. How
many in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} If you find yourself touching every dot, pause and
look for a part you know first. Counting one by one is the backup plan, not
the goal here.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Five-Frame Flash}

\begin{formulabox}
\textbf{Rule:} A five-frame has exactly 5 boxes. A \textbf{full frame is 5},
and each \textbf{empty box} means one less than 5.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A five-frame has 1 empty box. How many dots?
\step{Use the frame, not a count: every empty box takes one away from five.}
\step{$5 - 1 = \ans{4}$ --- the frame shows four at a glance.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A five-frame has 2 empty boxes. How many dots?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}
