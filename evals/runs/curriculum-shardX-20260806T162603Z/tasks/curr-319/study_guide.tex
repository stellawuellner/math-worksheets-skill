\sstitleblock{Correspondence and Scale Factor in Similar Triangles}

\skillheading{Skill 1 --- Reading the correspondence}

\begin{formulabox}
\textbf{Rule.} $\triangle ABC \sim \triangle DEF$ pairs the vertices in the
order written: $A \leftrightarrow D$, $B \leftrightarrow E$,
$C \leftrightarrow F$. Matching sides are the ones between matching vertices,
so $\overline{AB} \leftrightarrow \overline{DE}$ and
$\overline{BC} \leftrightarrow \overline{EF}$.\par
Write every proportion the same way round --- first triangle on top on
\emph{both} sides:
\[ \frac{AB}{DE} = \frac{BC}{EF}. \]
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \sim \triangle DEF$ with $AB = 4$ cm,
$BC = 7$ cm and $DE = 10$ cm. Find $EF$.
\step{Want $EF$, and I know the side matching it ($BC$) plus one complete pair
($AB$ with $DE$) --- so one proportion with first triangle on top on both sides
does it.}
\step{$\dfrac{4}{10} = \dfrac{7}{EF} \;\Rightarrow\; 4 \cdot EF = 7 \cdot 10$}

$EF = \ans{17.5}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle ABC \sim \triangle DEF$ with $AB = 5$ cm,
$BC = 8$ cm and $DE = 15$ cm. Find $EF$.\\[2pt]
\rotatebox{180}{\footnotesize check: $EF = \ans{24}$ cm}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the unknown must sit on the same level as the other side
of its own triangle. If $\overline{DE}$ is on the bottom, $\overline{EF}$ is on
the bottom too.
\end{watchoutbox}

\skillheading{Skill 2 --- Using the scale factor correctly}

\begin{formulabox}
\textbf{Rule.} The scale factor \emph{from} one triangle \emph{to} another is
\[ k = \frac{\text{length in the target triangle}}{\text{matching length in the starting triangle}}, \]
and every side of the starting triangle is multiplied by that same $k$.
Enlarging gives $k > 1$; reducing gives $k < 1$. Reversing the direction
replaces $k$ by $1/k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle PQR \sim \triangle STU$ with $PQ = 8$ cm,
$ST = 20$ cm and $QR = 6$ cm. Find $TU$.
\step{Want a side of the target triangle, and I have one complete pair to build
the factor from --- so find $k$ first, then multiply.}
\step{$k = \dfrac{20}{8} = 2.5$, so $TU = 6 \cdot 2.5$}

$TU = \ans{15}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle PQR \sim \triangle STU$ with $PQ = 9$ mm,
$ST = 27$ mm and $QR = 4$ mm. Find $TU$.\\[2pt]
\rotatebox{180}{\footnotesize check: $TU = \ans{12}$ mm}
\end{tryitbox}

\skillheading{Skill 3 --- Spotting the error in a solution}

\begin{formulabox}
\textbf{Two checks that catch almost every setup mistake.}
\begin{itemize}[leftmargin=0.6cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Same order twice.} Each ratio must be first triangle over second
        triangle. If one side of the equation is flipped, the answer comes out
        by a factor of $k^2$.
  \item \textbf{Size sense.} Enlarging must make every length longer, reducing
        must make every length shorter. An answer that moves the wrong way means
        the factor was inverted.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \sim \triangle DEF$ with $AB = 6$, $BC = 9$ and
$DE = 15$. Some work reads $\frac{6}{15} = \frac{EF}{9}$. Redo it correctly.
\step{The unknown $EF$ belongs to the second triangle, so it must sit on the
bottom like $DE$ does --- that tells me the written ratio is upside down before
any arithmetic.}
\step{$\dfrac{6}{15} = \dfrac{9}{EF} \;\Rightarrow\; 6 \cdot EF = 9 \cdot 15$}

$EF = \ans{22.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle ABC \sim \triangle DEF$ with $AB = 8$, $BC = 20$ and
$DE = 12$. Some work reads $\frac{8}{12} = \frac{EF}{20}$. Redo it
correctly.\\[2pt]
\rotatebox{180}{\footnotesize check: $EF = \ans{30}$}
\end{tryitbox}
