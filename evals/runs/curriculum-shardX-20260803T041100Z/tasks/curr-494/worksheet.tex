\wstitleblock{Parametric, Polar and Vector-Valued Calculus --- Diagnosing Parameter, Polar-Bound and Speed-versus-Velocity Errors}{AP Calculus BC}{}

\noindent\small Every problem below contains work that is arithmetically clean and
conceptually wrong. Three faults recur:
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Parameter:} $\dfrac{dy}{dx} = \dfrac{dy/dt}{dx/dt}$, evaluated at
        the $t$ that \emph{reaches} the point --- never at the point's own
        coordinate.
  \item \textbf{Polar bounds:} area is
        $\frac{1}{2}\displaystyle\int r^2\,d\theta$ taken over the $\theta$-interval
        that traces the region \emph{once}. Find where $r = 0$ before choosing
        limits.
  \item \textbf{Speed vs velocity:} velocity is the vector
        $\langle x'(t),\, y'(t)\rangle$; speed is its magnitude
        $\sqrt{x'(t)^2 + y'(t)^2}$, and a speed is never negative.
  \item For each problem: name the error in words, then show the corrected work.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{\textbf{Find and fix.} A curve is given by $x(t) = t^2 + 2t$ and
$y(t) = t^3 - 4t$. Ana wants $\dfrac{dy}{dx}$ at $t = 1$ and computes
$\dfrac{dx/dt}{dy/dt}$, obtaining $-4$.
\par\smallskip
Name Ana's error in one sentence, then compute $\dfrac{dy}{dx}$ at $t = 1$.}

\problem[5.5cm]{\textbf{Find and fix.} A particle moves along
$x(t) = t^2 - 3t$, $\;y(t) = t^3$. Bo reports the speed at $t = 2$ as $13$,
having added $x'(2)$ and $y'(2)$.
\par\smallskip
Say what quantity Bo's number actually describes, then find the speed at $t = 2$,
rounded to the nearest hundredth.}

\problem[6cm]{\textbf{Find and fix.} The rose $r = 4\cos(3\theta)$ has three
petals. Cal wants the area of \emph{one} petal, integrates
$\frac{1}{2} r^2$ from $0$ to $2\pi$, and reports $25.13$.
\par\smallskip
\textbf{(a)} Solve $r = 0$ to find the two consecutive $\theta$-values that bound
one petal.
\par
\textbf{(b)} Using those limits, compute the exact area of one petal. Explain in
one sentence what Cal's limits actually measured.}

\problem[6.5cm]{\textbf{Find and fix.} A particle's position is
$x(t) = t^2$, $\;y(t) = \frac{2}{3}t^3$ for $0 \le t \le 1$. Dev finds the total
distance travelled by integrating $x'(t)$ and $y'(t)$ separately and adding the
two results, obtaining $1.667$.
\par\smallskip
Write the correct integral for total distance, evaluate it exactly, and say in
one sentence why Dev's method cannot give a distance.}

\problem[6.5cm]{\textbf{Find and fix.} The curve $x(t) = t^2 + 1$,
$\;y(t) = t^3 - 3t$ passes through the point $(2, -2)$. Elle finds the slope of
the tangent there by substituting $t = 2$, ``because $x = 2$'', and reports
$2.25$.
\par\smallskip
\textbf{(a)} Find every $t$ with $x(t) = 2$, and say which one reaches the point
$(2, -2)$.
\par
\textbf{(b)} Compute the correct slope $\dfrac{dy}{dx}$ at that point.}

\problem[6cm]{\textbf{Find and fix.} A limaçon is given by
$r = 1 + 2\cos\theta$. Ravi is asked for every $\theta$ in $[0, 2\pi)$ at which
the curve passes through the pole. He writes $\cos\theta = \frac{1}{2}$ and
answers $\theta = \frac{\pi}{3}$.
\par\smallskip
Ravi solved the wrong equation. Set up the correct one and give every solution in
$[0, 2\pi)$, in exact form.}

\problem[6.5cm]{\textbf{Find and fix.} A particle follows the vector-valued path
$\mathbf{r}(t) = \langle \sin 2t,\; \cos t \rangle$. Nia computes the speed at
$t = \frac{\pi}{3}$ by adding the two components of $\mathbf{r}'(t)$, and reports
$-1.87$.
\par\smallskip
Give one reason Nia's answer must be wrong before doing any arithmetic. Then
compute the speed at $t = \frac{\pi}{3}$, rounded to the nearest hundredth.}

\problem[7cm]{\textbf{Explain.}
\par\smallskip
\textbf{(a)} Distinguish, in your own words, the velocity vector, the speed, and
the slope $\dfrac{dy}{dx}$ for a parametric curve. Say which are vectors, which
are scalars, and what each one tells you about the motion.
\par
\textbf{(b)} Explain why a component of velocity may be negative but a speed
never can.
\par
\textbf{(c)} Explain why integrating $\frac{1}{2} r^2$ over a full turn overcounts the
area of the rose in problem 3, and state a habit for choosing polar limits.
\par\noansline}
