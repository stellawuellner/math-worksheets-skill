\sstitleblock{Common Denominators for Unlike Fractions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Rewriting a fraction with a new denominator}

\begin{formulabox}
\textbf{Rule:} multiply the top and the bottom by the \emph{same} number.
The fraction keeps its value because you are multiplying by a form of 1.
The smallest denominator that works for two fractions is the smallest number
that both denominators divide into evenly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $\tfrac{2}{3}$ with denominator 6.
\step{I want sixths and I have thirds --- ask what number times 3 gives 6, then multiply top and bottom by it.}
\step{$3 \times 2 = 6$, so multiply top and bottom by 2: $\tfrac{2 \times 2}{3 \times 2}$ --- the new numerator is $\ans{4}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $\tfrac{3}{5}$ with denominator 10. What is the
new numerator?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Adding unlike fractions}

\begin{formulabox}
\textbf{Plan:} you can only add pieces that are the same size.
First rewrite both fractions with a common denominator, then add the
numerators and keep the denominator. Simplify at the end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $\tfrac{1}{4} + \tfrac{1}{6}$.
\step{The pieces are different sizes --- find a common denominator first. The smallest number both 4 and 6 divide into is 12.}
\step{Rewrite: $\tfrac{1}{4} = \tfrac{3}{12}$ and $\tfrac{1}{6} = \tfrac{2}{12}$, so the sum is $\tfrac{3}{12} + \tfrac{2}{12} = \ans{\tfrac{5}{12}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $\tfrac{1}{3} + \tfrac{1}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{7}{12}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Subtracting unlike fractions}

\begin{formulabox}
\textbf{Plan:} same idea as adding --- rewrite both fractions with a common
denominator, then subtract the numerators and keep the denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $\tfrac{3}{4} - \tfrac{2}{3}$.
\step{Different denominators --- rewrite both in twelfths, since 12 is the smallest number 4 and 3 both divide into.}
\step{$\tfrac{3}{4} = \tfrac{9}{12}$ and $\tfrac{2}{3} = \tfrac{8}{12}$, so the difference is $\tfrac{9}{12} - \tfrac{8}{12} = \ans{\tfrac{1}{12}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $\tfrac{7}{8} - \tfrac{1}{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{8}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never add or subtract the denominators. In
$\tfrac{1}{4} + \tfrac{1}{6}$ the answer is \emph{not} $\tfrac{2}{10}$ ---
the denominator tells the size of the pieces, and only same-size pieces can
be counted together.
\end{watchoutbox}
