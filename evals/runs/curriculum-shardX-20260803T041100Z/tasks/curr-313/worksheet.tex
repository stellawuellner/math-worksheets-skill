\wstitleblock{Triangle Congruence: Solving for Corresponding Parts}{High School Geometry}{}

\noindent\small\textbf{Directions}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item A congruence statement is read \textbf{in order}: in $\triangle ABC \cong \triangle DEF$ the first letters correspond ($A \leftrightarrow D$), the second ($B \leftrightarrow E$), the third ($C \leftrightarrow F$). Corresponding sides are named by corresponding letter pairs: $\overline{AB} \leftrightarrow \overline{DE}$.
  \item By \textbf{CPCTC}, corresponding parts of congruent triangles are congruent, so the two matched measures are \emph{equal}. Write that equation first, then solve it.
  \item Every length below is in the unit stated in its problem, and every angle measure is in degrees. Show the equation you solved.
\end{itemize}
\normalsize

\problem[5cm]{$\triangle ABC \cong \triangle DEF$. The side $\overline{AB}$ measures
$3x + 2$ centimetres and the side $\overline{DE}$ measures $17$ centimetres.
Write the equation CPCTC gives you, and solve for $x$.}

\problem[5cm]{$\triangle PQR \cong \triangle STU$. The angle $\angle P$ measures
$(4y - 10)$ degrees and the angle $\angle S$ measures $50$ degrees. Solve for $y$.}

\problem[5.5cm]{$\triangle JKL \cong \triangle MNP$. The side $\overline{JK}$ measures
$2x + 7$ inches and the corresponding side $\overline{MN}$ measures $4x - 9$ inches.
Solve for $x$.}

\problem[6cm]{$\triangle ABC \cong \triangle DEF$. In $\triangle ABC$, the angle
$\angle A$ measures $52$ degrees and the angle $\angle B$ measures $61$ degrees.
Find the measure of $\angle F$ in degrees. (Which angle of $\triangle ABC$ corresponds
to $\angle F$?)}

\problem{$\triangle RST \cong \triangle XYZ$. The angle $\angle R$ measures
$(5x - 3)$ degrees and the angle $\angle X$ measures $(2x + 12)$ degrees.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Solve for $x$. \ansline
  \item Find the measure of $\angle R$. \answerline{degrees}
\end{enumerate}}

\problem{\textbf{Bridge truss.} Two triangular panels of a truss are built congruent:
$\triangle ABC \cong \triangle DEF$. The steel member $\overline{AC}$ measures
$4x + 1$ metres and the corresponding member $\overline{DF}$ measures $2x + 13$ metres.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Solve for $x$. \ansline
  \item How long is member $\overline{AC}$? \answerline{m}
\end{enumerate}}

\problem{Points $A$, $B$, $C$ and $D$ are arranged so that $\overline{AB} \cong \overline{CD}$
and $\angle BAC \cong \angle DCA$. The segment $\overline{AC}$ belongs to both
$\triangle BAC$ and $\triangle DCA$.
\begin{enumerate}[label=(\alph*), itemsep=5.5cm]
  \item Prove $\triangle BAC \cong \triangle DCA$. Write a two-column proof; the
        third pair of parts you need is the shared side. \noansline
  \item Using that congruence, $\overline{BC}$ measures $3x - 4$ centimetres and its
        corresponding side $\overline{DA}$ measures $20$ centimetres. Solve for $x$. \ansline
\end{enumerate}}

\problem{\textbf{Find and fix the mistake.} Kim is given $\triangle ABC \cong \triangle DFE$
and writes ``$\overline{AB}$ corresponds to $\overline{DE}$''.
\begin{enumerate}[label=(\alph*), itemsep=4.5cm]
  \item Explain why Kim's correspondence is wrong, and name the side that really
        corresponds to $\overline{AB}$. \ansline
  \item The side $\overline{AB}$ measures $2x + 5$ centimetres and the side
        $\overline{DF}$ measures $31$ centimetres. Write the correct equation and
        solve for $x$. \ansline
\end{enumerate}}
