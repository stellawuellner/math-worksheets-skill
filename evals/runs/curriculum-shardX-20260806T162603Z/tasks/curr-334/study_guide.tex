\sstitleblock{One-Way Implications Between Quadrilaterals}

\begin{formulabox}
\textbf{The family tree --- read the arrows in one direction only:}\par
square $\Rightarrow$ rectangle $\Rightarrow$ parallelogram $\Rightarrow$ trapezoid \quad and \quad square $\Rightarrow$ rhombus $\Rightarrow$ kite\par
\textbf{Every} square is a rhombus. \textbf{Not every} rhombus is a square. Going \emph{down} the tree is free; going \emph{up} always costs one extra condition, and you must say which one you have.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Angle tests in a parallelogram}

\begin{formulabox}
\textbf{What a parallelogram guarantees:} opposite angles congruent, and consecutive angles supplementary (they add to $180^\circ$).\par
\textbf{What it does not guarantee:} right angles. That is the extra condition that upgrades it to a rectangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In a parallelogram, $m\angle A = (4x + 10)^\circ$ and $m\angle B = (2x + 50)^\circ$ are consecutive. Find $x$.
\step{Consecutive angles of a parallelogram are supplementary --- so add the two expressions and set the total to a straight angle, rather than setting either one to a right angle.}
\step{$(4x + 10) + (2x + 50) = 180$, so $6x + 60 = 180$ and $6x = 120$}

$\Rightarrow\quad x = $ $\ans{20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Consecutive angles measure $(3x + 25)^\circ$ and $(x + 15)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Side-length tests}

\begin{formulabox}
\textbf{Rules:} all four sides congruent $\Rightarrow$ rhombus. Two pairs of \emph{adjacent} sides congruent $\Rightarrow$ kite. One pair of opposite sides both parallel and congruent $\Rightarrow$ parallelogram.\par
Congruent sides never produce right angles, so no side test alone can reach rectangle or square.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rhombus has $PQ = 5x - 2$ and $QR = 3x + 8$. Find $x$.
\step{All sides of a rhombus are congruent, so the two expressions describe the same length --- set them equal instead of adding them.}
\step{$5x - 2 = 3x + 8$, so $2x = 10$}

$\Rightarrow\quad x = $ $\ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A rhombus has $PQ = 7x - 3$ and $QR = 4x + 18$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Diagonal tests}

\begin{formulabox}
\textbf{Rules:} diagonals bisect each other $\Rightarrow$ parallelogram. Add \emph{congruent} diagonals $\Rightarrow$ rectangle. Add \emph{perpendicular} diagonals $\Rightarrow$ rhombus. Both extras together $\Rightarrow$ square.\par
In a rhombus the diagonals are perpendicular bisectors, so each half-diagonal pair gives a right triangle whose hypotenuse is a side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rhombus has half-diagonals $5$ and $12$ at the centre. Find a side.
\step{The diagonals of a rhombus are perpendicular, so a side is the hypotenuse of a right triangle on the two half-diagonals --- that is the pythagorean theorem, not an addition.}
\step{side $= \sqrt{5^2 + 12^2} = \sqrt{169}$}

$\Rightarrow\quad$ $\ans{13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A rhombus has half-diagonals $8$ and $15$. Find a side.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17}$}
\end{tryitbox}
