\wstitleblock{Adding and Subtracting Fractions with Like Denominators}{Grades 4--5}{}

\begin{center}\small\textit{Adding, subtracting, and finding a missing numerator --- all with like denominators}\end{center}

\noindent\textbf{The big idea.} When two fractions have the \textbf{same denominator},
the pieces are the \emph{same size}, so you can just count them.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Add or subtract the \textbf{numerators} (the counts of pieces).
  \item Keep the \textbf{denominator} (the size of one piece) exactly as it is.
  \item Shade or cross out on the bar model first, then write the number sentence.
  \item Write your final answer in simplest form when it can be simplified.
\end{itemize}
