% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Combining two translations}

\begin{formulabox}
\textbf{Rule:} translations add. Following $(x + a,\ y + b)$ by $(x + c,\ y + d)$ gives the single translation $(x + a + c,\ \ y + b + d)$.\par
The order does not matter here --- two slides give the same result either way, which is not true once a reflection or rotation joins in.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Translate by $(x + 4,\ y + 2)$, then by $(x - 9,\ y + 6)$. Find the single translation.
\step{Both steps are slides, so nothing turns or flips --- the components simply add, one direction at a time.}
\step{horizontal: $4 + (-9)$ \quad and \quad vertical: $2 + 6$}

$\Rightarrow\quad (x + $ $\ans{-5}$ $,\ y + $ $\ans{8}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Translate by $(x - 3,\ y)$, then by $(x + 11,\ y)$. Give the horizontal component of the single translation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Symmetry of a regular polygon}

\begin{formulabox}
\textbf{Rules:} a regular polygon with $n$ sides has $n$ lines of symmetry, and it has rotational symmetry of order $n$ about its centre.\par
The smallest rotation that maps it onto itself is $\dfrac{360^\circ}{n}$, and every multiple of that angle works too. Read the relation backwards to go from a given angle to $n$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the smallest rotation mapping a regular octagon onto itself.
\step{The eight vertices are equally spaced around one full turn, so the smallest turn that lands vertex on vertex is one eighth of that turn.}
\step{$360^\circ / 8$}

$\Rightarrow\quad$ $\ans{45^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A regular polygon's smallest rotation is $30^\circ$. Find its number of sides.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Two reflections in a row}

\begin{formulabox}
\textbf{Parallel mirrors:} the result is a translation perpendicular to them, through \emph{twice} the distance between the lines.\par
\textbf{Intersecting mirrors:} the result is a rotation about the crossing point, through \emph{twice} the angle between the lines.\par
The doubling is the whole point --- reflecting back and forth covers the gap twice.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect in $x = 2$, then in $x = 7$. Find the translation distance.
\step{The two mirrors are parallel, so the composition is a translation, and the rule doubles the gap between the lines rather than using it once.}
\step{$2(7 - 2) = 2(5)$}

$\Rightarrow\quad$ $\ans{10}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two mirrors through the origin meet at $40^\circ$. Find the angle of the resulting rotation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{80^\circ}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Naming the single equivalent motion}

\begin{formulabox}
\textbf{Method:} apply the composition to one point, compare the image with the original, and match the pattern to a rule.\par
both coordinates negated $\Rightarrow$ half turn about the origin \quad $y$ negated only $\Rightarrow$ reflection in the $x$-axis \quad $x$ negated only $\Rightarrow$ reflection in the $y$-axis \quad both shifted by fixed amounts $\Rightarrow$ translation
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect $(7, 3)$ in the $x$-axis, then in the $y$-axis. Name the single motion.
\step{Tracking one point is enough to identify the pattern, so apply both steps to it and then compare the result with where it started.}
\step{$(7, 3) \to (7, -3) \to (-7, -3)$: both coordinates have changed sign}

$\Rightarrow\quad ($ $\ans{-7}$ $,$ $\ans{-3}$ $)$, which is a half turn about the origin.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Reflect $(4, -9)$ in the $x$-axis. Give the $y$-coordinate of the image.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}
