\sstitleblock{Diagnosing Separation and Initial-Condition Errors}

\skillheading{Skill 1 --- Integrating both sides}
\begin{formulabox}\textbf{When the right side contains no $y$}, the equation is
already separated: integrate straight across.\\[3pt]
$\dfrac{dy}{dx} = f(x) \ \Longrightarrow \ y = \displaystyle\int f(x)\,dx + C$
\\[3pt]
One constant is enough --- a $C$ on each side collapses into their difference.
The $+C$ is not decoration: it is the entire family of solution curves, and the
initial condition's only job is to choose one of them.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Solve $\dfrac{dy}{dx} = 8x^{3} - 3$.
\step{The right side has no $y$ in it, so no separation step is needed --- this
is a plain antiderivative question.}
\step{Integrate term by term: $\displaystyle\int (8x^{3} - 3)\,dx$, giving
$\ans{y = 2x^{4} - 3x + C}$. Check by differentiating back.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $\dfrac{dy}{dx} = 10x - 6x^{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 5x^{2} - 2x^{3} + C}$}\end{tryitbox}

\skillheading{Skill 2 --- Separating the variables}
\begin{formulabox}\textbf{Rule:} you may integrate only when \emph{every} $y$
sits with $dy$ and \emph{every} $x$ sits with $dx$:\\[2pt]
$\dfrac{dy}{dx} = f(x)\,g(y) \ \Longrightarrow \
\dfrac{1}{g(y)}\,dy = f(x)\,dx \ \Longrightarrow \
\displaystyle\int \frac{dy}{g(y)} = \int f(x)\,dx$.\\[3pt]
$y$ is a \emph{function} of $x$, so a stray $y$ on the $dx$ side can never be
treated as a constant. Two integrals worth memorising:
$\displaystyle\int \frac{dy}{y} = \ln|y|$ and
$\displaystyle\int y^{-2}\,dy = -\frac{1}{y}$ (the minus sign is lost constantly).
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\dfrac{dy}{dx} = \dfrac{x}{y}$, \ $y(0) = 4$. Find $y(3)$.
\step{There is a $y$ on the right, so nothing may be integrated until it is
moved across --- multiply both sides by $y$ first.}
\step{$y\,dy = x\,dx$ gives $\frac{y^{2}}{2} = \frac{x^{2}}{2} + C$. Apply
$y(0) = 4$ here, in the form you derived: $C = 8$, so $y^{2} = x^{2} + 16$ and
$y = \sqrt{x^{2} + 16}$; then $y(3) = \sqrt{9 + 16} = \ans{5}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\dfrac{dy}{dx} = \dfrac{x}{y}$, \ $y(0) = 12$. Find $y(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}\end{tryitbox}
\begin{watchoutbox}\textbf{(!) Watch out:} never square-root term by term.
$\sqrt{x^{2} + 16} \neq x + 4$. Apply the initial condition to the equation you
actually derived, before any rearranging that changes what $C$ means.
\end{watchoutbox}

\skillheading{Skill 3 --- Where the constant of integration goes}
\begin{formulabox}\textbf{The exponential trap.} Solving $\dfrac{dy}{dx} = ky$
gives $\ln|y| = kx + C$. Exponentiating turns the sum into a
\textbf{product}:\\[2pt]
$|y| = e^{kx + C} = e^{C}e^{kx} \ \Longrightarrow \ y = Ae^{kx}$, \ where
$A = e^{C}$.\\[3pt]
So the constant is a \textbf{multiplier}, never a term: $y = e^{kx} + C$ is not
a solution for any $C \neq 0$. With $y(0) = y_{0}$ the answer is always
$y = y_{0}e^{kx}$.\\[2pt]
\textbf{One-line safety check:} differentiate your answer and compare it with
the original right-hand side.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\dfrac{dy}{dx} = 5y$, \ $y(0) = 7$. Find $y(0.2)$.
\step{The rate is proportional to the amount, so the solution is exponential and
the constant must be carried through as a factor, not added afterwards.}
\step{$y = Ae^{5x}$ with $y(0) = A = 7$, so $y = 7e^{5x}$ and
$y(0.2) = 7e^{1} = \ans{19.03}$. Check: $y' = 35e^{5x} = 5y$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\dfrac{dy}{dx} = 2y$, \ $y(0) = 6$. Find $y(0.5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16.31}$}\end{tryitbox}
