\sstitleblock{Canceled-Factor Holes vs Vertical Asymptotes}

\skillheading{Skill 1 --- Factor and Reduce a Rational Expression}

\begin{formulabox}
\textbf{Always factor before you decide anything.} A zero of the original denominator
is a break in the domain; \emph{which kind} of break depends on whether that factor
cancels.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Difference of squares: $x^2 - c^2 = (x-c)(x+c)$
  \item Monic trinomial: find the factor pair of the constant that sums to the middle coefficient
  \item Reducing loses information --- carry the excluded $x$-values forward yourself
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Reduce $\dfrac{x^2-4}{x^2-x-6}$ and state the restriction.
\step{Both parts are quadratic and the question is which factors are shared, so factor completely before cancelling anything.}
\step{$\dfrac{(x-2)(x+2)}{(x-3)(x+2)} = \ans{\dfrac{x-2}{x-3}}$, valid for $x \ne -2$ and $x \ne 3$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Reduce $\dfrac{x^2-1}{x^2+4x-5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x+1}{x+5}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Locate Holes and Vertical Asymptotes}

\begin{formulabox}
\textbf{After reducing:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Factor that \textbf{cancelled} $\rightarrow$ \emph{hole} at its zero. Height $=$ the \emph{reduced} form evaluated there.
  \item Factor that \textbf{survived} $\rightarrow$ \emph{vertical asymptote} at its zero.
  \item Reduce first, substitute once. Mixing the original numerator with the reduced denominator gives a false height of $0$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Height of the hole of $\dfrac{x^2-16}{x^2-2x-8}$ at $x = 4$.
\step{The factor $x-4$ cancels, so the two-sided limit exists and equals the reduced form's value --- evaluate the reduced form, never the original.}
\step{$\dfrac{(x-4)(x+4)}{(x-4)(x+2)} = \dfrac{x+4}{x+2}$, so the height is $\dfrac{4+4}{4+2} = \ans{\tfrac{4}{3}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Height of the hole of $\dfrac{x^2-1}{x^2-3x+2}$ at $x = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Tell a Hole from an Asymptote}

\begin{formulabox}
\textbf{Count multiplicities, do not just match factors.} A shared factor cancels only
as many times as it appears on top.\par
$\dfrac{(x-c)}{(x-c)^2}$ leaves one $(x-c)$ downstairs $\rightarrow$ \emph{asymptote}.\quad
$\dfrac{(x-c)^2}{(x-c)}$ cancels out fully $\rightarrow$ \emph{hole}.\par
\textbf{(!) A perfect-square denominator is the usual trap:} $x^2 - 6x + 9 = (x-3)^2$ holds the factor twice.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A student says $\dfrac{x^2-9}{x^2-6x+9}$ has a hole at $x = 3$. Fix it.
\step{The factor $x-3$ appears once on top and twice on the bottom, so it cannot cancel completely --- compare multiplicities before deciding.}
\step{$\dfrac{(x-3)(x+3)}{(x-3)^2} = \ans{\dfrac{x+3}{x-3}}$: one factor survives, so $x = 3$ is a vertical asymptote.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Reduce $\dfrac{x^2-25}{x^2+10x+25}$ and say what sits at $x = -5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x-5}{x+5}}$, asymptote}
\end{tryitbox}
