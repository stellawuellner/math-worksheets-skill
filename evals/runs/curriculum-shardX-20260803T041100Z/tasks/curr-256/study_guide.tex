\sstitleblock{Explaining Legal Equation Transformations}

\skillheading{Skill 1 --- Naming the property behind a step}

\begin{formulabox}\textbf{Rule:}\ A step is \emph{legal} when the new equation
has exactly the same solutions as the old one. Two families do that:
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item do the same thing to \textbf{both sides} --- Addition, Subtraction,
        Multiplication, or Division Property of Equality (the multiplier or
        divisor must be nonzero);
  \item rewrite \textbf{one side} into an equal form --- Distributive Property,
        combining like terms.
\end{itemize}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Solve $2x - 5 = 11$ and justify.
\step{The $x$ is multiplied then has 5 subtracted, so I undo in reverse order:
the subtraction first, then the multiplication.}
\step{Add 5 to both sides (Addition Property): $2x = 16$. Divide both sides by
2, nonzero (Division Property).}
$\Rightarrow\quad \ans{x = 8}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Solve $5x + 4 = 24$, naming the property
on each line.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Deciding whether two equations match}

\begin{formulabox}\textbf{Rule:}\ Rewriting one side is legal only when the two
forms are equal for \textbf{every} value of the variable. Distribute to
\emph{all} terms inside the parentheses; a partial distribution is a different
expression, so the equation it sits in is a different equation.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Write $3(x-4)$ without parentheses.
\step{Nothing is being done to both sides here --- I am only rewriting one
side, so I need an identity, and the Distributive Property is the one that
removes parentheses.}
\step{Multiply the 3 by \emph{both} terms: $3 \cdot x$ and $3 \cdot (-4)$.}
$\Rightarrow\quad 3(x-4) = \ans{3x-12}$, equal for every $x$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Write $5(x+3)$ without parentheses.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x+15}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the step that is not legal}

\begin{formulabox}\textbf{Rule:}\ Read down someone's work and ask of each
line: \emph{what was done, and was it done to both sides in full?} The three
usual offenders are a change made to one side only, a partial distribution, and
a division by a quantity that could be zero. Then substitute the answer into
the \textbf{original} equation --- a legal chain always survives that test.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad A student writes
$4x + 9 = 25 \to 4x = 34$. Find the bad line and fix it.
\step{The 9 moved but got bigger, so the student added 9 to the right while
removing it from the left --- not the same operation on both sides.}
\step{Subtract 9 from both sides: $4x = 16$, then divide both sides by 4.}
$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad A student writes
$7x - 2 = 19 \to 7x = 17$. What went wrong, and what is $x$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\skillheading{Skill 4 --- Multiplying an inequality by a negative}

\begin{formulabox}\textbf{Rule:}\ Every equality property has an inequality
twin, with one change: multiplying or dividing both sides by a
\textbf{negative} number \emph{reverses} the symbol, because negating reflects
the number line and reverses order. Adding or subtracting never does.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Solve $-4x < 12$.
\step{The only step needed is dividing both sides by $-4$, and that divisor is
negative, so the symbol must turn around on that line.}
\step{$\dfrac{-4x}{-4} = x$ and $\dfrac{12}{-4} = -3$, with $<$ becoming $>$.}
$\Rightarrow\quad \ans{x > -3}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Solve $-5x > 20$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x < -4}$}
\end{tryitbox}
