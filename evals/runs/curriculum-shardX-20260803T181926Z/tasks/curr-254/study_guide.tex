\sstitleblock{Sign and Like-Term Errors in Polynomial Operations}

\skillheading{Skill 1 --- Subtracting and Combining Like Terms}

\begin{formulabox}
\textbf{Subtracting means adding the opposite:} $A - B = A + (-1)B$, so every term of $B$ changes sign.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Write the sign-flipped line before you combine anything. It is one extra line and it is where the error would have been.
  \item Terms are \emph{like} only when the variable parts match exactly: $4x^2$ and $-x^2$ combine, $4x^2$ and $-x$ do not.
  \item Add the coefficients; never touch the exponent. $3x + 5x = 8x$, not $8x^2$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Simplify $(6x^2 + 2x - 9) - (4x^2 - 5x + 3)$.
\step{The minus sign belongs to all three terms of the second polynomial, so rewrite the subtraction as adding the opposite before combining.}
\step{$6x^2 + 2x - 9 - 4x^2 + 5x - 3$, and grouping like terms gives $\ans{2x^2 + 7x - 12}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Simplify $(8x^2 - x + 5) - (3x^2 + 4x - 2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x^2 - 5x + 7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Distributing and Expanding}

\begin{formulabox}
\textbf{Every term, every time:} $-a(b + c) = -ab - ac$, and $(a - b)^2 = a^2 - 2ab + b^2$ --- the middle term is not optional.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Two negatives make a positive: $(-2x)(-7) = +14x$. The most-missed sign on any sheet.
  \item A square is a product of two identical factors. Write $(2x-3)(2x-3)$ out rather than squaring each piece.
  \item Multiply coefficients, add exponents: $(-3x)(2x^2) = -6x^3$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Expand $-4x(2x - 3) + 2(x + 6)$.
\step{Two separate distributions have to happen before any terms are collected, so handle each parenthesis on its own and keep the signs with the factors that produced them.}
\step{$-8x^2 + 12x$ from the first, $2x + 12$ from the second, and collecting the $x$ terms gives $\ans{-8x^2 + 14x + 12}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Expand $(3x - 5)^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9x^2 - 30x + 25}$}
\end{tryitbox}

\skillheading{Skill 3 --- Factoring with Signs}

\begin{formulabox}
\textbf{For $x^2 + bx + c$:} find two numbers whose product is $c$ and whose sum is $b$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item $c$ negative $\Rightarrow$ the two numbers have opposite signs, and the one with the larger size takes $b$'s sign.
  \item $c$ positive $\Rightarrow$ both numbers share $b$'s sign.
  \item Always check: substitute one value, such as $x = 1$, into the original and into your factors. They must agree.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Factor $x^2 - 2x - 15$.
\step{The constant is negative, so the two numbers have opposite signs, and the sum being negative means the larger one is the negative.}
\step{The pair $-5$ and $3$ has product $-15$ and sum $-2$, so the factors are $\ans{(x - 5)(x + 3)}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^2 - 3x - 28$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x - 7)(x + 4)}$}
\end{tryitbox}
