% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Average value of a function}

\begin{formulabox}
\textbf{Rule:} the average value of $f$ on $[a,b]$ is $\displaystyle f_{\text{avg}} = \frac{1}{b-a}\int_a^b f(x)\,dx$. Integrating alone gives an accumulated total; the division by the interval length is what turns it into a per-unit average. Sanity check: the answer must lie between the smallest and largest values of $f$ on the interval.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the average value of $f(x) = x^2$ on $[0,3]$.
\step{The question asks for a per-unit average rather than a total, so integrate and then divide by the length of the interval.}
\step{$\dfrac{1}{3}\displaystyle\int_0^3 x^2\,dx = \dfrac{1}{3}\left[\dfrac{x^3}{3}\right]_0^3 = \dfrac{9}{3}$.}

$\Rightarrow\quad \ans{f_{\text{avg}} = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the average value of $f(x) = x^2$ on $[0,6]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f_{\text{avg}} = 12}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Accumulated change from a rate}

\begin{formulabox}
\textbf{Rule:} if $r(t)$ is a rate, then $\displaystyle\int_a^b r(t)\,dt$ is the net amount accumulated between $t=a$ and $t=b$ --- no division. The units multiply out and tell you whether the setup is right: litres per minute times minutes gives litres. Do \emph{not} divide by the interval here; that would answer a different question.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tank fills at $r(t) = 6t + 1$ litres per minute. How much enters over the first $4$ minutes?
\step{The words "how much enters" ask for a total, so integrate the rate over the time interval and do not divide by anything.}
\step{$\displaystyle\int_0^4 (6t+1)\,dt = \left[3t^2 + t\right]_0^4 = 48 + 4$.}

$\Rightarrow\quad \ans{52}$ litres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tank fills at $r(t) = 8t + 3$ litres per minute. How much enters over the first $3$ minutes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$ litres}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Displacement, distance, average velocity}

\begin{formulabox}
\textbf{Three different questions from one velocity.} Displacement is $\int_a^b v\,dt$; total distance is $\int_a^b |v|\,dt$; average velocity is $\frac{1}{b-a}\int_a^b v\,dt$. Position needs a starting value: $x(t) = x(a) + \int_a^t v$. To handle $|v|$, first solve $v = 0$ to find where the sign changes, then integrate over each piece and add the absolute values.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $v(t) = t^2 - 3t + 2$ m/s. Find the total distance travelled on $[0,3]$.
\step{"Total distance" counts backward motion as travel, so integrate the absolute value --- which means first finding where the velocity changes sign.}
\step{$v = (t-1)(t-2)$ vanishes at $t = 1$ and $t = 2$; the three pieces measure $\tfrac{5}{6}$, $\tfrac{1}{6}$ and $\tfrac{5}{6}$, and adding them gives the distance.}

$\Rightarrow\quad \ans{\frac{11}{6} \approx 1.833}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $v(t) = t^2 - 3t + 2$, now on $[0,4]$. Find the total distance.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{17}{3} \approx 5.667}$}
\end{tryitbox}
