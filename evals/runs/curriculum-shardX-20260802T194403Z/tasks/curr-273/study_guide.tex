\sstitleblock{Arithmetic Sequences as Linear Functions}

\skillheading{Skill 1 --- A recursive rule from a listed sequence}

\begin{formulabox}
\textbf{Two lines, always:}\\[4pt]
$a_1 = \text{first term}$ \qquad $a_n = a_{n-1} + d \quad (n \ge 2)$\par
\smallskip
Find $d$ by subtracting \emph{consecutive} terms, always later minus earlier, so
a decreasing sequence gives a negative $d$. Check the same difference twice
before you commit: one matching gap does not make a sequence arithmetic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write the recursive rule for $9, 14, 19, 24, \ldots$ and
find $a_7$.
\step{Consecutive differences are constant, so this is arithmetic and the
recursive rule is the pair (first term, constant step).}
\step{$d = 14 - 9 = 5$, so $a_1 = 9$ and $a_n = a_{n-1} + 5$; term 7 is 6 steps
along, giving $9 + 6(5) = \ans{39}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write the recursive rule for $40, 33, 26, \ldots$ and find
$a_6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 2 --- A recursive rule from a situation}

\begin{formulabox}
\textbf{Read the context for two things:}\\[4pt]
the \textbf{starting amount} $\rightarrow a_1$ \qquad the \textbf{repeated
change} $\rightarrow d$\par
\smallskip
Signal words: ``each,'' ``every,'' ``per'' mark the step. Gains are positive,
losses negative. Decide what $a_1$ means \emph{before} computing: the amount
before any change is $a_0$, not $a_1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A candle is 21 cm tall and burns down 2 cm each hour. How
tall is it after 6 hours?
\step{``Each hour'' names a constant repeated change, so height is arithmetic
in the number of hours, with the burn counted as negative.}
\step{After $n$ hours the height is $21 - 2n$, so after 6 hours it is
$21 - 12 = \ans{9}$ cm.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A plant is 12 cm tall and grows 3 cm each week. How tall
is it after 8 weeks?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

\skillheading{Skill 3 --- Running a recursive rule forward}

\begin{formulabox}
\textbf{Count the steps, not the terms:}\\[4pt]
$a_n = a_1 + (n-1)d$\par
\smallskip
Between $a_1$ and $a_n$ there are $n-1$ arrows, so the step is applied $n-1$
times. Using $n$ jumps one term too far --- that single off-by-one is the most
common lost mark on this topic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $a_1 = 4$ and $a_n = a_{n-1} + 11$. Find $a_{10}$.
\step{Listing ten terms is slow and error-prone; count the applications of the
step instead.}
\step{From term 1 to term 10 is 9 steps, so
$a_{10} = 4 + 9(11) = 4 + 99 = \ans{103}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $a_1 = 90$ and $a_n = a_{n-1} - 6$. Find $a_{12}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

\skillheading{Skill 4 --- Recursive rule to explicit linear function}

\begin{formulabox}
\textbf{The step is the slope:}\\[4pt]
$a_n = a_1 + (n-1)d = dn + (a_1 - d)$\par
\smallskip
So the slope is $d$ and the value at $n = 0$ is $a_1 - d$ --- one step
\emph{back} from the first term. Writing $a_1$ as the intercept is the classic
error; check it by testing $n = 1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $a_1 = 8$ and $a_n = a_{n-1} + 5$. Write the explicit
rule and find $a_9$.
\step{A constant step means the sequence is a linear function of $n$, so
rewrite the rule as slope times $n$ plus an intercept.}
\step{$a_n = 8 + 5(n-1) = 5n + 3$, so $a_9 = 5(9) + 3 = \ans{48}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $a_1 = -2$ and $a_n = a_{n-1} + 7$. Write the explicit
rule and find $a_{11}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{68}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A recursive rule without its first line is not a
formula --- $a_n = a_{n-1} + 5$ fits every sequence that counts by fives. Always
write $a_1$, and always state the domain $n \ge 2$ on the step line.
\end{watchoutbox}
