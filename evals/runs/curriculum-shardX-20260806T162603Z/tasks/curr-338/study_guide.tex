\sstitleblock{Writing and Reading Circle Equations}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Writing the equation of a circle}

\begin{formulabox}
\textbf{Centre-radius form:} \ $(x-h)^2 + (y-k)^2 = r^2$, \ centre $(h, k)$, radius $r$.\par
It is the distance formula with the square roots undone: the left side is the squared distance from $(x, y)$ to the centre. So when a point on the circle is given, substitute it and read off $r^2$ --- you never need $r$ itself.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Centre $(3, -2)$, passing through $(7, 1)$. Write the equation.
\step{The right side of the form is the squared distance from the centre to any point on the circle, so substituting the known point gives $r^2$ in one step --- no square root needed.}
\step{$r^2 = (7-3)^2 + (1-(-2))^2 = 16 + 9$}

$\Rightarrow\quad (x-3)^2 + (y+2)^2 = $ $\ans{25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Centre $(-1, 5)$, passing through $(4, 17)$. Find $r^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{169}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Reading a circle equation}

\begin{formulabox}
\textbf{Reading the form:} the number after the minus sign inside each bracket is the coordinate. $(x + 5)^2$ means $x - (-5)$, so $h = -5$. The right side is $r^2$, never $r$.\par
\textbf{Inside / on / outside:} substitute the point into the left side. Less than $r^2$ is inside, equal is on, greater is outside.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the radius of $(x-6)^2 + (y+3)^2 = 100$.
\step{The right-hand side of centre-radius form is the radius squared, not the radius, so the one move needed is a square root.}
\step{$r = \sqrt{100}$}

$\Rightarrow\quad r = $ $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the radius of $(x+2)^2 + (y-7)^2 = 144$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Completing the square to find the centre}

\begin{formulabox}
\textbf{Method:} group the $x$ terms and the $y$ terms, move the constant to the right, then for each group add the square of half its linear coefficient --- \emph{to both sides}. Factor each group as a perfect square.\par
Half of the $x$-coefficient becomes $-h$ inside the bracket, so a $+8x$ ends up as $(x+4)^2$ and a centre coordinate of $-4$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Put $x^2 + y^2 - 8x + 2y - 8 = 0$ into centre-radius form.
\step{The equation has both squared terms but also linear terms, which centre-radius form never has --- that is the signal to complete the square on each variable.}
\step{Half of $-8$ is $-4$ and half of $2$ is $1$: add $16$ and $1$ to both sides, giving $(x-4)^2 + (y+1)^2 = 8 + 16 + 1$}

$\Rightarrow\quad \ans{(x-4)^2 + (y+1)^2 = 25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Put $x^2 + y^2 + 4x - 12y + 15 = 0$ into centre-radius form and give the right-hand side.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+2)^2 + (y-6)^2 = 25}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the right-hand side is $r^2$. A circle written $\ldots = 36$ has radius $6$, not $36$. And when you add a completing term to the left, you must add it to the right as well --- forgetting that is the single most common slip in this topic.
\end{watchoutbox}
