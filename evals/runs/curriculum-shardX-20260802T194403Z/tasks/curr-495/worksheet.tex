\wstitleblock{Parametric, Polar, and Vector-Valued Calculus}{AP Calculus BC}{}

\noindent\small
Parametric slopes, vector motion, polar intersections, and accumulated
quantities. The methods are deliberately mixed after the first three problems,
so read each stem and decide what is being asked for before you set anything up:
a slope $\frac{dy/dt}{dx/dt}$, a magnitude $|\mathbf{v}|$, an intersection, or
an accumulation $\int$. Give exact values where they are clean, and round
decimal answers to three places. Calculator allowed on problems 2, 3, 6, 8,
and 9.
\normalsize

\problem[5.5cm]{A curve is given by $x(t) = t^2 + 1$ and $y(t) = t^3 - 3t$.
Find $\dfrac{dy}{dx}$ at $t = 2$.}

\problem[6cm]{A particle moves in the plane with position vector
$\mathbf{r}(t) = \langle t^3 - 3t,\ t^2 \rangle$. Find the particle's speed at
$t = 2$, rounded to three decimal places.}

\problem[6.5cm]{A particle has position
$\mathbf{r}(t) = \left\langle t^2,\ \tfrac{2}{3}t^3 \right\rangle$ for
$0 \le t \le 2$. Find the total distance the particle travels over that
interval, rounded to three decimal places.}

\problem[6.5cm]{The cardioid $r = 2 + 2\cos\theta$ and the circle $r = 3$ are
graphed together. Find every value of $\theta$ in $[0, 2\pi)$ at which the two
curves intersect.}

\problem[6.5cm]{A curve is given by $x(t) = t^3 - 3t$ and $y(t) = t^2 - 4t$.
Find the value of $t$ at which the curve has a \emph{horizontal} tangent line,
and show that the tangent there is not also vertical.}

\problem[6.5cm]{Find the exact area enclosed by \emph{one petal} of the rose
$r = 3\sin(2\theta)$, then give a three-decimal approximation. (One petal is
traced as $\theta$ runs from $0$ to $\frac{\pi}{2}$.)}

\problem[6cm]{A particle moves along the $x$-axis with position
$x(t) = t^2 e^{-t}$ for $t \ge 0$. Find its velocity $v(t) = x'(t)$, written as
a single simplified product.}

\problem[7cm]{Find the area of the region that lies \emph{inside} the circle
$r = 3$ and \emph{outside} the cardioid $r = 2 + 2\cos\theta$. Use your
intersection values from problem 4 as the limits, and round to three decimal
places.}

\problem[6.5cm]{Find the length of the parametric curve $x = t^2$, $y = t^3$
from $t = 0$ to $t = 1$, rounded to three decimal places.}

\problem{\textbf{Synthesis challenge.} The particle from problem 2 moves
with $\mathbf{r}(t) = \langle t^3 - 3t,\ t^2 \rangle$ on $0 \le t \le 3$.
\begin{enumerate}[label=(\alph*), leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item \begin{minipage}[t]{\linewidth}Explain why the total distance the
  particle travels is \emph{not} the same as the magnitude of its displacement,
  and write a correct expression for each. Do not evaluate them.
  \par\vspace*{4.5cm}\noansline\end{minipage}
  \item \begin{minipage}[t]{\linewidth}Determine whether the particle is
  speeding up or slowing down at $t = 1$, and justify your answer using
  $\mathbf{v}(1)$ and $\mathbf{a}(1)$.
  \par\vspace*{4.5cm}\noansline\end{minipage}
\end{enumerate}}
