\problem{Find the slope of the line through $(2,3)$ and $(8,7)$.}

\noindent\textbf{Solution.} Rise over run, both differences in the same order:
$m = \dfrac{7-3}{8-2} = \dfrac{4}{6}$, which reduces, so \ans{m = \tfrac{2}{3}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the slope of the line through $(-4,5)$ and $(6,-1)$.}

\noindent\textbf{Solution.} $m = \dfrac{-1-5}{6-(-4)} = \dfrac{-6}{10}$, so
\ans{m = -\tfrac{3}{5}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the slope of the line through $(5,-2)$ and $(5,9)$.}

\noindent\textbf{Solution.} The run is $5 - 5 = 0$, so the slope would require
division by zero: the line is vertical and has no slope. \ans{\text{undefined}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Line $j$ through $(-2,1)$ and $(2,9)$; line $k$ through $(0,-5)$ and
$(3,1)$. (a) Slope of $j$. (b) Slope of $k$. (c) Parallel, perpendicular or
neither?}

\noindent\textbf{Solution (a).} $m_j = \dfrac{9-1}{2-(-2)} = \dfrac{8}{4}$, so
\ans{m_j = 2}

\noindent\textbf{Solution (b).} $m_k = \dfrac{1-(-5)}{3-0} = \dfrac{6}{3}$, so
\ans{m_k = 2}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response says
the slopes are equal, and that equal slopes mean the lines rise by the same
amount for every unit of run, so they never meet: the lines are
\emph{parallel}. Naming them parallel without citing the equal slopes earns
partial credit.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Line $p$ through $(-1,4)$ and $(3,6)$; line $q$ through $(2,-3)$ and
$(4,-7)$. (a) Slope of $p$. (b) Slope of $q$. (c) Parallel, perpendicular or
neither?}

\noindent\textbf{Solution (a).} $m_p = \dfrac{6-4}{3-(-1)} = \dfrac{2}{4}$, so
\ans{m_p = \tfrac{1}{2}}

\noindent\textbf{Solution (b).} $m_q = \dfrac{-7-(-3)}{4-2} = \dfrac{-4}{2}$, so
\ans{m_q = -2}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response
multiplies the slopes, $\tfrac{1}{2} \times (-2) = -1$, and states that slopes
whose product is $-1$ belong to perpendicular lines --- equivalently that each
slope is the negative reciprocal of the other. Full credit requires the product
or the negative-reciprocal relationship to be shown, not just the word
perpendicular.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the slope of any line parallel to the line through $(-6,2)$ and
$(2,-4)$.}

\noindent\textbf{Solution.} Parallel lines share a slope, so compute the given
line's: $m = \dfrac{-4-2}{2-(-6)} = \dfrac{-6}{8}$, giving
\ans{m = -\tfrac{3}{4}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the slope of any line perpendicular to the line through $(3,-5)$
and $(9,7)$.}

\noindent\textbf{Solution.} The given line has slope
$\dfrac{7-(-5)}{9-3} = \dfrac{12}{6} = 2$. A perpendicular line has the negative
reciprocal, which read straight off the coordinates is
$-\dfrac{9-3}{7-(-5)} = -\dfrac{6}{12}$, so \ans{m = -\tfrac{1}{2}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$ABCD$ with $A(-1,0)$, $B(5,2)$, $C(7,8)$, $D(1,6)$. (a) Slope of
$\overline{AB}$. (b) Slope of $\overline{DC}$. (c) Slope of $\overline{AD}$.
(d) Why is $ABCD$ a parallelogram?}

\noindent\textbf{Solution (a).} $\dfrac{2-0}{5-(-1)} = \dfrac{2}{6}$, so
\ans{m_{AB} = \tfrac{1}{3}}

\noindent\textbf{Solution (b).} $\dfrac{8-6}{7-1} = \dfrac{2}{6}$, so
\ans{m_{DC} = \tfrac{1}{3}}

\noindent\textbf{Solution (c).} $\dfrac{6-0}{1-(-1)} = \dfrac{6}{2}$, so
\ans{m_{AD} = 3}

\noindent\textbf{Solution (d) --- instructor-judged.} A complete response notes
that $\overline{AB}$ and $\overline{DC}$ have equal slopes, computes
$m_{BC} = \dfrac{8-2}{7-5} = 3$ and observes that it equals $m_{AD}$, so both
pairs of opposite sides are parallel --- which is the definition of a
parallelogram. Full credit requires both pairs to be addressed.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Triangle $PQR$ with $P(-2,1)$, $Q(4,3)$, $R(6,-3)$. (a) Slope of
$\overline{PQ}$. (b) Slope of $\overline{QR}$. (c) Right-angled at $Q$?}

\noindent\textbf{Solution (a).} $\dfrac{3-1}{4-(-2)} = \dfrac{2}{6}$, so
\ans{m_{PQ} = \tfrac{1}{3}}

\noindent\textbf{Solution (b).} $\dfrac{-3-3}{6-4} = \dfrac{-6}{2}$, so
\ans{m_{QR} = -3}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response
multiplies the slopes of the two sides that meet at $Q$:
$\tfrac{1}{3} \times (-3) = -1$, so $\overline{PQ} \perp \overline{QR}$ and the
angle at $Q$ is a right angle --- the triangle is right-angled at $Q$. Full
credit requires the product to be shown and tied to the vertex where the two
sides meet.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$EFGH$ with $E(-3,-1)$, $F(1,-3)$, $G(5,5)$, $H(1,7)$. (a) Slope of
$\overline{EF}$. (b) Slope of $\overline{HG}$. (c) Slope of $\overline{FG}$.
(d) What kind of quadrilateral is $EFGH$?}

\noindent\textbf{Solution (a).} $\dfrac{-3-(-1)}{1-(-3)} = \dfrac{-2}{4}$, so
\ans{m_{EF} = -\tfrac{1}{2}}

\noindent\textbf{Solution (b).} $\dfrac{5-7}{5-1} = \dfrac{-2}{4}$, so
\ans{m_{HG} = -\tfrac{1}{2}}

\noindent\textbf{Solution (c).} $\dfrac{5-(-3)}{5-1} = \dfrac{8}{4}$, so
\ans{m_{FG} = 2}

\noindent\textbf{Solution (d) --- instructor-judged.} A complete response uses
the equal slopes of $\overline{EF}$ and $\overline{HG}$, computes
$m_{EH} = \dfrac{7-(-1)}{1-(-3)} = 2$ and matches it to $m_{FG}$, so both pairs
of opposite sides are parallel; then multiplies
$m_{EF} \times m_{FG} = -\tfrac{1}{2} \times 2 = -1$, so adjacent sides meet at
right angles. A parallelogram with right angles is a \emph{rectangle}. Full
credit requires both the parallel argument and the perpendicular argument.
