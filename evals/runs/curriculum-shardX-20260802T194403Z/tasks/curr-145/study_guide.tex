\sstitleblock{Measurement Conversion, Volume, and Angle Measure}

\skillheading{1. Converting units (unit-conversion)}

\begin{formulabox}
\textbf{Rule:} A \emph{bigger} unit turned into a \emph{smaller} one gives
\emph{more} of them, so \textbf{multiply}. Going the other way,
\textbf{divide}.\\[3pt]
$1$ ft $=12$ in \quad $1$ yd $=3$ ft \quad $1$ m $=100$ cm \quad
$1$ km $=1000$ m\\
$1$ kg $=1000$ g \quad $1$ L $=1000$ mL \quad $1$ h $=60$ min \quad
$1$ min $=60$ s\\[3pt]
Mixed measures ($2$ kg $250$ g) convert the big part, then \textbf{add} the
leftover.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A ribbon is $5$ m long. How many centimetres is that?
\step{Metres are bigger than centimetres, so the number of centimetres must be larger --- that means multiply by the $100$ in $1$ m $=100$ cm.}
\step{$5 \cdot 100 = 500$}
The ribbon is $\ans{500}$ cm long.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A bench is $7$ ft long. How many inches is that?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{84}$ in}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Answer in the unit the question \emph{asks} for, not the
unit the numbers came in --- and never multiply two edges that are still in
different units.
\end{watchoutbox}

\skillheading{2. Volume of a rectangular prism (prism-volume)}

\begin{formulabox}
\textbf{Formula:} $V = \text{length} \cdot \text{width} \cdot \text{height}$,
which is the same as (area of the base) $\cdot$ height.\\[3pt]
Volume is measured in \textbf{cubic} units (cm$^3$, ft$^3$) because you are
counting unit cubes that fill the box.\\[3pt]
\textbf{Missing edge?} Divide: height $=V \div$ (base area).
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A box is $7$ cm long, $3$ cm wide, and $2$ cm high. Find its volume.
\step{All three edges are given in the same unit and the box is a prism, so multiply the three edges --- no conversion needed.}
\step{Bottom layer: $7 \cdot 3 = 21$ cubes. Two layers: $21 \cdot 2 = 42$.}
$V = \ans{42}$ cm$^3$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A crate is $9$ cm by $2$ cm by $4$ cm. Find its
volume.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{72}$ cubic cm}
\end{tryitbox}

\skillheading{3. Angle measure (angle-measure)}

\begin{formulabox}
\textbf{Rule:} Angles that sit side by side \textbf{add}: if ray $BD$ is inside
$\angle ABC$, then $\angle ABD + \angle DBC = \angle ABC$.\\[3pt]
Whole angles worth knowing: a right angle is $90^\circ$, a straight angle is
$180^\circ$, a full turn is $360^\circ$.\\[3pt]
\textbf{Missing part?} part $=$ whole $-$ known part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A right angle is split into two parts. One part is $52^\circ$. Find the other.
\step{The two parts fill a right angle, so their whole is $90^\circ$ and the missing part is found by subtracting, not by adding.}
\step{$90 - 52 = 38$}
The other part is $\ans{38}$ degrees.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Two angles together form a straight angle. One measures
$118^\circ$. Find the other.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{62}$ degrees}
\end{tryitbox}

\skillheading{4. Putting it together (multistep-synthesis)}

\begin{formulabox}
\textbf{Plan for a multistep problem:}\\[3pt]
\textbf{(1)} Underline the unit the question asks for. \quad
\textbf{(2)} Make every given measure match before you multiply.\\
\textbf{(3)} Do the volume or angle work. \quad
\textbf{(4)} Convert once, at the end: $1000$ cm$^3 = 1$ L.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A tank is $30$ cm by $20$ cm by $25$ cm. How many litres does it hold?
\step{The edges are in centimetres but the answer must be in litres, so find the volume first and convert at the very end.}
\step{$30 \cdot 20 \cdot 25 = 15000$ cm$^3$, then $15000 \div 1000 = 15$.}
The tank holds $\ans{15}$ L.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A tank is $50$ cm by $20$ cm by $40$ cm. How many litres
does it hold?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40}$ L}
\end{tryitbox}
