\sstitleblock{Place Value to 1,000: Three Ways to Write a Number}

\noindent\small Every number below is the same number three times. The place a
digit sits in tells you what it is worth: hundreds, tens, ones. Once you know
each digit's value, all three forms are just different ways of saying it.
\normalsize

\skillheading{1. Standard form to expanded form}
\begin{formulabox}
\textbf{Rule:} give each digit its place value, then join them with $+$.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item hundreds digit $\times\,100$ \quad tens digit $\times\,10$ \quad ones digit $\times\,1$
  \item A digit of $0$ contributes nothing --- leave that part out of the sum.
\end{itemize}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Write $638$ in expanded form.
\step{Expanded form asks for the value of each digit, so read the number one
place at a time instead of as a whole.}
\step{$6$ is in the hundreds place $\rightarrow 600$; \ $3$ is in the tens
place $\rightarrow 30$; \ $8$ is in the ones place $\rightarrow 8$.}
\par
$\ans{638 = 600 + 30 + 8}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Write $275$ in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{275 = 200 + 70 + 5}$}
\end{tryitbox}

\skillheading{2. Expanded form to standard form}
\begin{formulabox}
\textbf{Rule:} each part of the sum fills exactly one place. Add them up, or
just read off the leading digit of each part.\\[3pt]
If a place is missing from the sum, that place gets a $0$ --- the zero holds
the place open so the other digits do not slide over.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Write $800 + 50 + 1$ in standard form.
\step{Each part names one place, so the job is to put one digit in each
place rather than to do a long addition.}
\step{$800 \rightarrow 8$ hundreds; \ $50 \rightarrow 5$ tens; \
$1 \rightarrow 1$ one.}
\par
$\ans{851}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Write $400 + 9$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{409}$}
\end{tryitbox}

\skillheading{3. Number names (word form) to standard form}
\begin{formulabox}
\textbf{Rule:} break the name at the word ``hundred''.\\[3pt]
\textit{(number) hundred} $\rightarrow$ the hundreds digit; the rest of the
name is a two-digit number that fills the tens and ones places.\\[3pt]
Teen and tens words are \emph{one} two-digit chunk, not two digits written
side by side: ``fifteen'' is $15$, which is $1$ ten and $5$ ones.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Write three hundred sixty-two in standard form.
\step{The name already tells you the place values, so split it at
``hundred'' and translate each chunk.}
\step{``three hundred'' $\rightarrow 300$; \ ``sixty'' $\rightarrow 60$; \
``two'' $\rightarrow 2$; \ so $300 + 60 + 2$.}
\par
$\ans{362}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Write nine hundred four in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{904}$}
\end{tryitbox}

\skillheading{4. Comparing numbers written in different forms}
\begin{formulabox}
\textbf{Rule:} put \emph{both} numbers in standard form first, then compare
place by place from the left.\\[3pt]
Same hundreds? Compare tens. Same tens? Compare ones.
Use $<$ for ``is less than'' and $>$ for ``is greater than'' --- the small
point aims at the smaller number.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Which is greater: $500 + 20 + 7$ or five hundred
seventy-two?
\step{The two numbers are written in different forms, so nothing can be
compared until both are in standard form.}
\step{$500 + 20 + 7 = 527$ and five hundred seventy-two $= 572$; both have
$5$ hundreds, and $2$ tens is less than $7$ tens.}
\par
$\ans{527 < 572}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare six hundred eight and $600 + 80$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{608 < 680}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A zero in the middle is the trap. In $907$ the tens
place is empty, so the expanded form is $900 + 7$, not $90 + 7$. And in
$600 + 5$ the missing tens part still needs a $0$ in standard form: the
answer is $605$, never $65$.
\end{watchoutbox}
