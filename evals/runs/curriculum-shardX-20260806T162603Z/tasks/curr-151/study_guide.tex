% =================== SKILL 1 ===================
\skillheading{Writing a Ratio Three Ways}

\begin{formulabox}
\textbf{Rule:} A ratio compares two quantities \textbf{in order}. The same ratio
can be written three ways: with the word ``to'' ($3$ to $5$), with a colon
($3{:}5$), or as a fraction $\left(\tfrac{3}{5}\right)$. The quantity named
first always goes first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A team wins 4 games and loses 9. Write the ratio of wins
to losses three ways.
\step{Wins are named first, so the win count goes first in every form.}
\step{4 wins to 9 losses: $4$ to $9$, \ $4{:}9$, \ $\tfrac{4}{9}$ --- three ways
to write $\ans{\tfrac{4}{9}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A team wins 5 games and loses 7. Write the ratio of wins
to losses as a fraction.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{7}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Order matters! The ratio of dogs to cats is \textbf{not}
the same as the ratio of cats to dogs — flipping the order names a different
comparison.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Part-to-Part vs.\ Part-to-Whole}

\begin{formulabox}
\textbf{Rule:} A \textbf{part-to-part} ratio compares one group to another
group. A \textbf{part-to-whole} ratio compares one group to the total. To find
the whole, \textbf{add the parts first}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A box holds 3 red pens and 5 black pens. Write the ratio
of red pens to \textbf{all} pens.
\step{``All pens'' means the whole, so add the parts first: $3 + 5 = 8$ pens.}
\step{Red to all is part-to-whole: 3 out of 8, so the ratio is
$\ans{\tfrac{3}{8}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A jar holds 2 green buttons and 7 yellow buttons. Write
the ratio of green buttons to \textbf{all} buttons.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{2}{9}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Making and Checking Equivalent Ratios}

\begin{formulabox}
\textbf{Rule:} Equivalent ratios name the same comparison. Multiply or divide
\textbf{both} parts of a ratio by the same nonzero number to get an equivalent
ratio. Two ratios are equivalent when they simplify to the same simplest form.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the missing value: $\dfrac{5}{8} = \dfrac{?}{24}$.
\step{Find what the denominator was multiplied by: $8 \times 3 = 24$, so both
parts must be multiplied by 3.}
\step{Numerator: $5 \times 3 = 15$, so the missing value is $\ans{15}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the missing value: $\dfrac{2}{5} = \dfrac{?}{30}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Ratio:] a comparison of two quantities, in order
  \item[Part-to-whole:] a ratio comparing one group to the total
  \item[Equivalent ratios:] ratios that name the same comparison
  \item[Simplest form:] both parts share no common factor but 1
\end{description}
\end{multicols}
