\wstitleblock{Building Multi-Step Arguments from Givens}{High-school geometry}{}

\begin{center}\small\textit{ordering the steps, segment arguments, and angle arguments}\end{center}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item An argument starts at a \emph{given} and ends at the claim. Every line in between must follow from a line already written, by a definition, a postulate, or a property of equality.
  \item Some problems ask you to order and justify the steps; others ask you to run the chain and compute. Both need the reasons written down.
  \item Diagrams show labelling only; all measurements are in the problem text.
\end{itemize}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.4pt}

\problem{The four statements below make a valid argument, but they are out of order. Number them $1$ to $4$ in a working order and write the reason for each.
\par\vspace{0.15cm}
\begin{itemize}[leftmargin=1.0cm, itemsep=2pt, topsep=2pt, label={}]
  \item \underline{\hspace{0.7cm}} \ $m\angle FGH = 180 - m\angle HGK$
  \item \underline{\hspace{0.7cm}} \ $\angle FGH$ and $\angle HGK$ form a linear pair
  \item \underline{\hspace{0.7cm}} \ $\overrightarrow{GF}$ and $\overrightarrow{GK}$ are opposite rays, and $\overrightarrow{GH}$ is a third ray from $G$
  \item \underline{\hspace{0.7cm}} \ $m\angle FGH + m\angle HGK = 180$
\end{itemize}
\par\vspace*{5.0cm}\noansline}

\problem{The five statements below prove that a midpoint cuts a segment in half, but they are out of order. Number them $1$ to $5$ and write the reason for each.
\par\vspace{0.15cm}
\begin{itemize}[leftmargin=1.0cm, itemsep=2pt, topsep=2pt, label={}]
  \item \underline{\hspace{0.7cm}} \ $AM = MB$
  \item \underline{\hspace{0.7cm}} \ $AM + MB = AB$
  \item \underline{\hspace{0.7cm}} \ $M$ is the midpoint of $\overline{AB}$
  \item \underline{\hspace{0.7cm}} \ $2\,AM = AB$, so $AM = \frac{1}{2}AB$
  \item \underline{\hspace{0.7cm}} \ $\overline{AM} \cong \overline{MB}$
\end{itemize}
\par\vspace*{5.4cm}\noansline}

\problem{The statements below prove that two angles complementary to the same angle are congruent. They are out of order. Number them and write the reason for each.
\par\vspace{0.15cm}
\begin{itemize}[leftmargin=1.0cm, itemsep=2pt, topsep=2pt, label={}]
  \item \underline{\hspace{0.7cm}} \ $m\angle P = m\angle R$, so $\angle P \cong \angle R$
  \item \underline{\hspace{0.7cm}} \ $m\angle P + m\angle Q = m\angle R + m\angle Q$
  \item \underline{\hspace{0.7cm}} \ $\angle P$ and $\angle Q$ are complementary; $\angle R$ and $\angle Q$ are complementary
  \item \underline{\hspace{0.7cm}} \ $m\angle P + m\angle Q = 90$ and $m\angle R + m\angle Q = 90$
\end{itemize}
\par\vspace*{5.4cm}\noansline}
