\wstitleblock{Diagnosing Quadrant and Parameter-Orientation Errors}{Precalculus --- Polar and Parametric Representations}{}

\noindent\small\textbf{Focus:} Quadrant-correct polar angles, negative-r conversions, parametric
orientation, and complete parameter solutions.\par\vspace{2pt}
\textbf{Directions:} Several problems show a student's work that is \emph{wrong}. For those,
(i) name the error in one sentence, then (ii) give the corrected value. Angles are in degrees
unless a problem uses $t$ in radians; give angles in $[0^\circ, 360^\circ)$ and round decimals to
two places. A calculator's $\arctan$ always returns an angle in $(-90^\circ, 90^\circ)$ --- that
is where most of these errors begin.\normalsize

\vspace{0.15cm}

\problem{\textbf{Find and fix.} \ Devin converts the point $P(-3, 4)$ to polar form. He writes
$r = \sqrt{(-3)^2 + 4^2} = 5$, then $\theta = \arctan\!\left(\dfrac{4}{-3}\right) = -53.13^\circ$,
and reports $(5, -53.13^\circ)$.\\[3pt]
State what is wrong with his angle, then give the correct $\theta$ in $[0^\circ, 360^\circ)$.
\par\vspace*{6cm}\answerline{degrees}}

\problem{Convert $Q(2, -2)$ to polar form. A classmate reasons: ``the calculator gives
$-45^\circ$, and a negative angle is not allowed, so I add $180^\circ$ and get $135^\circ$.''\\[3pt]
Sketch the point, explain why $135^\circ$ cannot be right, and give the correct $\theta$ in
$[0^\circ, 360^\circ)$.
\par\vspace*{6cm}\answerline{degrees}}

\problem[6cm]{\textbf{Find and fix.} \ Sasha converts the polar point $(-4, 60^\circ)$ to
rectangular coordinates and reports $(2, 3.46)$.\\[3pt]
Explain what a \emph{negative} $r$ does to the location of the point, then compute the correct
$y$-coordinate.}

\problem[6cm]{A student claims that $x = 3\sin t$, $y = 3\cos t$ is ``the same as''
$x = 3\cos t$, $y = 3\sin t$, and evaluates the first at $t = \dfrac{\pi}{3}$ as $x = 1.5$.\\[3pt]
Find the correct $x$-coordinate at $t = \dfrac{\pi}{3}$ for $x = 3\sin t$, $y = 3\cos t$, and say
in one sentence how the two parametrizations differ in starting point and direction.}

\problem{\textbf{Find and fix.} \ The ellipse $x = 4\cos t$, $y = 2\sin t$ is traced for
$0^\circ \le t < 360^\circ$. Asked for every parameter value where the curve reaches $y = 1$, a
student answers ``$t = 30^\circ$'' and stops.\\[3pt]
Explain why one solution cannot be the whole answer for a curve that is traced all the way around,
then list \textbf{all} such $t$ in $[0^\circ, 360^\circ)$.
\par\vspace*{7cm}\answerline{degrees}}

\problem{\textbf{Find and fix.} \ Converting $R(-5, -12)$ to polar form, a student computes
$r = 13$ and $\theta = \arctan\!\left(\dfrac{-12}{-5}\right) = 67.38^\circ$, then reports
$(13, 67.38^\circ)$.\\[3pt]
Which quadrant does $R$ actually lie in? Name the error, then give the correct $\theta$ in
$[0^\circ, 360^\circ)$.
\par\vspace*{6cm}\answerline{degrees}}

\problem{Rewrite the polar point $(-6, 200^\circ)$ with a \textbf{positive} $r$ and an angle
in $[0^\circ, 360^\circ)$. A student writes $(6, 380^\circ)$.\\[3pt]
Explain why adding $180^\circ$ was the wrong move here, then give the correct angle.
\par\vspace*{6cm}\answerline{degrees}}

\problem{\textbf{Synthesis (explain and sketch).} \ Consider
\[ C_1:\ x = 2 + 3\cos t,\quad y = -1 + 3\sin t \]
\[ C_2:\ x = 2 + 3\cos(-t),\quad y = -1 + 3\sin(-t), \]
both for $0 \le t < 2\pi$.\\[3pt]
Show that $C_1$ and $C_2$ are the same circle but are traced in opposite directions. Sketch the
circle, mark the starting point, and put an arrow on each curve. Then explain how a small table of
$t$ values proves the orientation --- and why the Cartesian equation alone never can.
\par\vspace*{8cm}\noansline}
