\sstitleblock{Two Different Questions: a Rate and a Starting Value}

\skillheading{Skill 1 --- Read the roles, not the order}

\begin{formulabox}
\textbf{Slope-intercept form:} $y = mx + b$, where $m$ multiplies $x$ and $b$
stands alone.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $m$ = how much $y$ changes per $1$ of $x$ (a rate).
        $b$ = the value of $y$ when $x = 0$ (one output).
  \item Order does not decide roles: $12 - 2x$ has $m = -2$ and $b = 12$, exactly
        like $-2x + 12$.
  \item \textbf{The $x = 0$ test:} substitute $x = 0$. Whatever is left is $b$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $y = 6x + 2$, find $y$ when $x = 3$.
\step{Identify the roles before substituting: the number attached to $x$ is the
slope, so it is the one that gets multiplied.}
\step{$y = 6(3) + 2 = 18 + 2$}

The value is $\ans{20}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $y = 9 - 3x$, find $y$ when $x = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Slope from two points: $y$ on top}

\begin{formulabox}
\textbf{Slope formula:} $m = \dfrac{\text{change in } y}{\text{change in } x}$,
and subtract the two coordinates in the \emph{same} order top and bottom.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Read it as ``$y$ per $x$'' --- the units tell you which goes on top.
  \item An inverted slope is the reciprocal of the right one, so if your answer
        times the correct one is $1$, you divided the wrong way round.
  \item A $y$ coordinate is never a slope, and it is the intercept only when its
        point has $x = 0$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the slope through $(1, 4)$ and $(5, 16)$.
\step{Slope is $y$ per $x$, so put the change in $y$ on top and keep both
subtractions in the same order.}
\step{$m = \dfrac{16 - 4}{5 - 1} = \dfrac{12}{4}$}

The slope is $\ans{3}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the slope through $(3, 10)$ and $(7, 2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$}
\end{tryitbox}

\skillheading{Skill 3 --- The intercept a table does not show}

\begin{formulabox}
\textbf{Backing up to $x = 0$:} if the table starts at $x = 1$, then
$b = (\text{the } y \text{ at } x = 1) - (\text{the step size})$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Find the step size first: subtract consecutive $y$ values (when the $x$
        values go up by $1$).
  \item The first row of a table is only the intercept if that row is $x = 0$.
  \item Check by rebuilding $y = mx + b$ and testing it on a row you were given.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A table has $y = 11$ at $x = 1$ and goes up by $5$ for
each step of $1$ in $x$. Find the $y$-intercept.
\step{The intercept lives at $x = 0$, one step to the left of the first row, so
step backwards by the step size instead of reading the row itself.}
\step{$b = 11 - 5$}

The $y$-intercept is $\ans{6}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A table has $y = 20$ at $x = 1$ and goes up by $7$ for each
step of $1$ in $x$. Find the $y$-intercept.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} If an equation you wrote fails the very point it came
from, you almost certainly used a $y$ coordinate as the intercept. Substitute the
point back in --- it takes one line and catches the swap every time.
\end{watchoutbox}
