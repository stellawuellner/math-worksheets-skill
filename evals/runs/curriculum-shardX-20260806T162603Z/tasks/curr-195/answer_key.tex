\problem{Closed box $4 \times 3 \times 2$ cm: surface area.

\textbf{Solution:} A closed box has three pairs of matching faces:
$SA = 2(\ell w + \ell h + wh) = 2(4 \cdot 3 + 4 \cdot 2 + 3 \cdot 2)
= 2(12 + 8 + 6) = 2 \cdot 26$, so \ans{52\ \text{cm}^2}}

\problem{Cube with $5$-inch edges: surface area.

\textbf{Solution:} A cube has $6$ identical square faces:
$SA = 6s^2 = 6 \cdot 5^2 = 6 \cdot 25$, so \ans{150\ \text{in}^2}}

\problem{Blueprint scale $1$ cm $: 4$ m; wall $3.5$ cm on paper.

\textbf{Solution:} Each blueprint centimeter stands for $4$ real meters:
$3.5 \cdot 4$, so the wall is \ans{14\ \text{m}} long.}

\problem{L-shaped patio with corners $(0,0)$, $(8,0)$, $(8,4)$, $(5,4)$,
$(5,7)$, $(0,7)$.

\textbf{Solution:} Split at the height $y = 4$: a bottom rectangle $8 \times 4
= 32$ and a top-left rectangle $5 \times 3 = 15$. Total $32 + 15$, so
\ans{47\ \text{m}^2}}

\problem{Floor plan $1$ in $: 5$ ft; bedroom $2$ in by $3$ in.

\textbf{Solution:} Scale each side first: real length $2 \cdot 5 = 10$ ft, real
width $3 \cdot 5 = 15$ ft. Area $= 10 \cdot 15$, so \ans{150\ \text{ft}^2}
(Scaling the drawing area $2 \cdot 3 = 6$ by $5$ only once gives $30$ --- too
small, because \emph{both} sides stretch.)}

\problem{Gift box $6 \times 4 \times 3$ in: paper, then sheets.

\textbf{Solution (a):} $SA = 2(6 \cdot 4 + 6 \cdot 3 + 4 \cdot 3)
= 2(24 + 18 + 12) = 2 \cdot 54$, so \ans{108\ \text{in}^2}

\textbf{Solution (b):} $108 \div 60 = 1.8$ sheets, and partial sheets are not
sold, so round \emph{up}: buy \ans{2} sheets.}

\problem{T-shaped stage with corners $(0,0)$, $(10,0)$, $(10,3)$, $(7,3)$,
$(7,8)$, $(3,8)$, $(3,3)$, $(0,3)$.

\textbf{Solution:} Split at height $y = 3$: bottom bar $10 \cdot 3 = 30$ and
upper column $4 \cdot 5 = 20$ (width $7 - 3 = 4$, height $8 - 3 = 5$). Total
$30 + 20$, so \ans{50\ \text{m}^2}}

\problem{Bridge $120$ m real, scale $1$ cm $: 8$ m: drawing length.

\textbf{Solution:} Going from real to drawing means \emph{dividing} by the
scale: $8x = 120$, so $x = 120 \div 8$, giving \ans{15\ \text{cm}}
(Multiplying instead gives $960$ --- a drawing longer than the bridge.)}

\problem{Logo of area $7$ cm$^2$ enlarged by scale factor $3$.

\textbf{Solution:} Lengths triple, so area multiplies by $3^2 = 9$:
$7 \cdot 9$, giving \ans{63\ \text{cm}^2} (Multiplying by $3$ only once gives
$21$, which forgets that both dimensions grow.)}

\problem{Open-top aquarium $60 \times 30 \times 40$ cm: glass area.

\textbf{Solution:} Bottom: $60 \cdot 30 = 1800$. Front and back:
$2 \cdot 60 \cdot 40 = 4800$. Two ends: $2 \cdot 30 \cdot 40 = 2400$. No top.
Total $1800 + 4800 + 2400$, so \ans{9000\ \text{cm}^2} (Adding a top gives
$10800$ --- but this tank is open.)}

\problem{House-shaped sign with corners $(0,0)$, $(8,0)$, $(8,5)$, $(4,8)$,
$(0,5)$.

\textbf{Solution:} Rectangle below: $8 \cdot 5 = 40$. Triangle roof: base $8$,
height $8 - 5 = 3$, area $\tfrac{1}{2} \cdot 8 \cdot 3 = 12$. Total $40 + 12$,
so \ans{52\ \text{m}^2}}

\problem{Challenge: $1:20$ model of a $4$ m by $3$ m room.

\textbf{Solution (a):} Convert, then divide by the scale: $4$ m $= 400$ cm, and
$400 \div 20$ gives \ans{20} cm; $3$ m $= 300$ cm, and $300 \div 20$ gives
\ans{15} cm.

\textbf{Solution (b):} Model floor area $= 20 \cdot 15$, so \ans{300} cm$^2$.

\textbf{Solution (c):} \emph{(instructor-judged)} Both the length and the
width are divided by $20$, so the area is divided by $20 \cdot 20 = 400$, not
by $20$. Any explanation naming the squared effect of a scale factor on area
earns full credit.}
