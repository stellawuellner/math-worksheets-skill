% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Products and the Sign of $i$ Squared}

\begin{formulabox}
\textbf{Rule:} $i^{2} = -1$, so every $i^{2}$ term flips sign when it becomes
real.\\[3pt]
$(a + bi)^{2} = a^{2} - b^{2} + 2abi$ \qquad $(a + bi)(a - bi) = a^{2} + b^{2}$\\[3pt]
Notice the two signs pull opposite ways: a square \textit{subtracts} $b^{2}$, a
conjugate product \textit{adds} it. Both come from the same $i^{2} = -1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $(4 - 3i)^{2}$
\step{A square is a product of two identical binomials, so all four products must be written --- the middle term is the one that gets dropped.}
\step{$16 - 12i - 12i + 9i^{2}$, and $9i^{2} = -9$, so the real part is $16 - 9$ and the imaginary part is $-24$.}

$\Rightarrow\quad \ans{7 - 24i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $(2 + 5i)^{2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-21 + 20i}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Distributing a Subtraction Sign}

\begin{formulabox}
\textbf{Rule:} a minus sign in front of a complex number reverses \textit{both}
of its parts.\\[3pt]
$(a + bi) - (c + di) = (a - c) + (b - d)i$\\[3pt]
Rewrite the subtraction as an addition first --- $-(c + di)$ becomes
$-c - di$ --- and the sign error has nowhere left to hide.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(7 + 3i) - (2 - 9i)$
\step{Nothing needs multiplying; the only risk is the sign, so rewrite the second number with both parts reversed before collecting.}
\step{$7 + 3i - 2 + 9i$, so the real parts give $7 - 2$ and the imaginary parts give $3 + 9$.}

$\Rightarrow\quad \ans{5 + 12i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(4 - 6i) - (-3 + 8i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7 - 14i}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Choosing the Right Conjugate}

\begin{formulabox}
\textbf{Rule:} to clear an $i$ from a denominator, multiply above \textit{and}
below by the conjugate of the denominator --- the same number with the sign of
its imaginary part reversed.\\[3pt]
$(c + di)(c - di) = c^{2} + d^{2}$, always real and always a \textit{sum}.\\[3pt]
Two ways this goes wrong: using the denominator itself instead of its conjugate,
and multiplying only the bottom by the conjugate.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{25}{4 + 3i}$
\step{An $i$ downstairs means the quotient is not in standard form, so multiply top and bottom by the conjugate $4 - 3i$.}
\step{Bottom: $16 + 9 = 25$. Top: $25(4 - 3i)$. The two $25$s cancel, leaving the real part $4$ and the imaginary part $-3$.}

$\Rightarrow\quad \ans{4 - 3i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{34 + 17i}{4 - i}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7 + 6i}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} when you check someone else's work, test the
\textit{denominator} first --- $9 - 4$ where $9 + 4$ belongs is the single most
common conjugate slip, and every later line of their solution will look tidy.
\end{watchoutbox}
