\wstitleblock{Measuring Fix-It: Gaps and Overlaps}{Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item When we measure, the units must sit end to end --- no gaps, no
    overlaps.
  \item Some kids in these problems measured the wrong way. Fix their counts!
\end{itemize}

\problem{The pencil is 8 cubes long. The crayon is 5 cubes long. How many
cubes longer is the pencil than the crayon?\par\vspace*{3cm}\answerline{cubes}}

\problem[3.5cm]{Mia measured her ribbon twice with paper clips. When she left
gaps between the clips she counted 6 clips. When the clips touched end to end
she counted 9 clips.\par
(a) How many clips did the gaps hide? \ansblank\par
(b) Say which count is the ribbon's true length.}

\problem{Leo measured the table with sticks, but he overlapped some sticks,
so his count came out 2 sticks too many. Leo counted 10 sticks. Fix his
count: how many sticks long is the table really? Tell what went
wrong.\par\vspace*{3.5cm}\answerline{sticks}}

\problem{Zoe measured the rope with paper clips but left 3 gaps, and each gap
was exactly 1 clip of rope. Zoe counted only 4 clips. Fix her count: how many
paper clips long is the rope really? Tell how you
know.\par\vspace*{3.5cm}\answerline{paper clips}}

\problem[3cm]{The blue string is 11 clips long. The red string is 14 clips
long. Write $>$, $<$, or $=$ in the blank:\par
\begin{center} 11 \ansblank\ 14 \end{center}}

\problem[4cm]{Challenge! Kim measured the bench with blocks but overlapped
some, so her count was 3 blocks too many. Kim counted 15 blocks. Ali measured
the same bench with no overlaps and counted 12 blocks.\par
(a) Fix Kim's count: \ansblank\ blocks\par
(b) Say whether Kim and Ali now agree.}
