\begin{center}
Evaluating binomial coefficients, expanding a binomial power in full,
extracting one specific term, and reading the structure of a row of
coefficients
\end{center}

\noindent\small\textbf{Directions.} The problems below rotate between four
tools, so decide which one the question needs before you start writing.
\[ (p+q)^{n} = \sum_{r=0}^{n} \binom{n}{r} p^{\,n-r} q^{\,r},
   \qquad \binom{n}{r} = \frac{n!}{r!\,(n-r)!} \]
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt, topsep=2pt]
  \item A single term needs the coefficient \emph{and} both powers --- the
        binomial coefficient by itself is almost never the answer.
  \item Simplify factorials by cancelling before multiplying.
\end{itemize}

\noindent\textbf{Reference.} The first rows of Pascal's triangle, shown for
their pattern only. No problem's numbers are taken from this display; use the
values each problem gives you.

\problem[6cm]{\textbf{Warm-up.} Evaluate $\dbinom{7}{3}$ by cancelling
factorials before you multiply. Show the cancellation.}

\problem[6cm]{\textbf{Warm-up.} Evaluate $\dbinom{9}{2}$. A classmate answers
72. Say what 72 actually counts, and give the correct value of
$\dbinom{9}{2}$.}

\problem[7cm]{\textbf{Full expansion.} Expand $(x+3)^{3}$ completely, using the
binomial theorem rather than repeated multiplication. Write the four terms in
descending powers of $x$.}

\problem[7cm]{\textbf{One term only.} In the expansion of $(x+4)^{5}$, find the
coefficient of the term containing $x^{2}$. Do not expand the whole binomial ---
name the value of $r$ you need first.}

\problem[7cm]{\textbf{Row structure.} Find the sum of all the entries in row 6
of Pascal's triangle (the row whose second entry is 6). Then explain, using the
binomial theorem, why substituting 1 for both letters in $(p+q)^{n}$ gives that
sum.}

\problem[8cm]{\textbf{Full expansion.} Expand $(2x-1)^{4}$ completely. Watch
both the power on the 2 and the alternating sign.}

\problem[7cm]{\textbf{One term only.} In the expansion of $(3+y)^{6}$, find the
coefficient of $y^{4}$. State which factor of the term the exponent on 3 comes
from.}

\problem[7.5cm]{\textbf{Solve for $n$.} A tournament pairs every team with every
other team exactly once, so the number of games is $\dbinom{n}{2}$, which equals
$\dfrac{n(n-1)}{2}$. Solve $\dfrac{n(n-1)}{2} = 45$ for $n$. Give both solutions
of the equation, then say which one can be a number of teams and why.}

\problem[7cm]{\textbf{Row structure.} Find the sum of the coefficients of
$(1+2x)^{5}$ without expanding it. State the substitution that produces the sum
of the coefficients of any polynomial.}

\problem[8cm]{\textbf{Full expansion, two letters.} Expand $(a+2b)^{3}$
completely, in descending powers of $a$. Keep the power of 2 attached to the
correct factor in every term.}

\problem[7.5cm]{\textbf{One term only.} In the expansion of $(2x+y)^{5}$, find
the coefficient of the term $x^{3}y^{2}$. Say why the coefficient is not simply
the entry from Pascal's triangle.}

\problem[8.5cm]{\textbf{Synthesis.} Consider $(3x-2)^{4}$.\\[3pt]
Find the sum of its coefficients using the same substitution as before, and then
explain what that sum tells you about the value of the polynomial at a
particular input. Finish by saying why the answer is so much smaller than the
individual coefficients of the expansion.}
