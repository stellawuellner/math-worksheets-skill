\sstitleblock{Multiplying and Dividing Fractions Without Flip Mistakes}

\skillheading{Skill 1 --- Cancel top with bottom, then multiply (cancel-then-multiply)}
\begin{formulabox}\textbf{Rule:}\quad
$\dfrac{a}{b} \times \dfrac{c}{d} = \dfrac{a \times c}{b \times d}$ \quad
(multiply straight across).\par
Before multiplying you may divide \textbf{one top number and one bottom
number} by the same factor. That is legal because it is the same as reducing
the answer afterwards. \textbf{Never} cancel two top numbers with each other,
and never two bottom numbers --- that changes the value.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\dfrac{5}{6} \times \dfrac{3}{10}$
\step{This is a multiplication, so I look for a top number and a bottom number
that share a factor.}
\step{$5$ (top) and $10$ (bottom) share $5$; $3$ (top) and $6$ (bottom) share
$3$. That leaves $\dfrac{1}{2} \times \dfrac{1}{2} = \ans{\frac{1}{4}}$}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad
$\dfrac{4}{9} \times \dfrac{3}{8}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{1}{6}}$}\end{tryitbox}

\skillheading{Skill 2 --- Flip only the divisor (flip-only-the-divisor)}
\begin{formulabox}\textbf{Rule (keep, change, flip):}\quad
$\dfrac{a}{b} \div \dfrac{c}{d} = \dfrac{a}{b} \times \dfrac{d}{c}$\par
\textbf{Keep} the first fraction unchanged, \textbf{change} $\div$ to
$\times$, \textbf{flip} only the second fraction (the divisor). The reciprocal
of $\dfrac{c}{d}$ is $\dfrac{d}{c}$, and $\dfrac{c}{d} \times \dfrac{d}{c} =
1$ --- that is why flipping undoes the division. Cancel only \emph{after} the
flip.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\dfrac{2}{5} \div \dfrac{3}{4}$
\step{Dividing by a fraction means multiplying by its reciprocal, so the
second fraction is the one I turn upside down.}
\step{Keep $\dfrac{2}{5}$, change to $\times$, flip $\dfrac{3}{4}$:
$\dfrac{2}{5} \times \dfrac{4}{3} = \ans{\frac{8}{15}}$}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad
$\dfrac{5}{8} \div \dfrac{1}{2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{4}}$}\end{tryitbox}

\skillheading{Skill 3 --- Find and fix a wrong flip (find-and-fix-the-flip)}
\begin{formulabox}\textbf{Two questions that catch every flip mistake:}\par
1. Which fraction got turned upside down --- the one \emph{after} the division
sign? \quad 2. Is the answer on the right side of $1$? Dividing by a fraction
smaller than $1$ makes the result \textbf{bigger} than the fraction you
started with; dividing by a fraction bigger than $1$ makes it smaller.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Sam wrote
$\dfrac{3}{5} \div \dfrac{2}{7} = \dfrac{5}{3} \times \dfrac{2}{7} =
\dfrac{10}{21}$. Find and fix the error.
\step{I check which fraction was flipped: Sam turned over the first one, but
only the divisor may be flipped.}
\step{Correct: $\dfrac{3}{5} \times \dfrac{7}{2} = \ans{\frac{21}{10}}$ ---
bigger than $\dfrac{3}{5}$, as it must be when dividing by a fraction less
than $1$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Lee wrote
$\dfrac{4}{7} \div \dfrac{2}{3} = \dfrac{7}{4} \times \dfrac{2}{3} =
\dfrac{7}{6}$. What is the correct answer?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{6}{7}}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Flipping belongs to division only
--- there is no flip in a multiplication problem. And cross-cancelling belongs
to multiplication only: in a division, rewrite it as a multiplication
\emph{first}, then look for a top and a bottom to cancel.
\end{watchoutbox}
