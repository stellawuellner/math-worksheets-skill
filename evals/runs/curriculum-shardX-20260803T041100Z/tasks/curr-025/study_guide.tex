\sstitleblock{Choosing a Fast Way to Add Within 10}

\noindent\small Three fast ways cover almost every fact within ten. Before you
add, \textbf{look at the two parts} and ask which way fits.\normalsize

\skillheading{1. Count on from the larger part}

\begin{formulabox}
\textbf{Use it when:} one part is small ($1$, $2$, or $3$) and the other is not.\\[3pt]
\textbf{How:} hold the \emph{larger} part in your head. Say the \textbf{next}
number first, one number for each of the smaller part. Stop when you have said
that many numbers.\\[3pt]
Order never changes a sum, so you may always start from the larger part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $5 + 3$
\step{One part is only $3$, so counting on is faster than counting all --- hold the larger part, $5$.}
\step{Say the next three numbers: ``$6$, $7$, $8$.'' Three counts, one for each of the $3$.}
$5 + 3 = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $7 + 1$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{2. Use a double or a near double}

\begin{formulabox}
\textbf{Use it when:} the two parts are the same, or are neighbours.\\[3pt]
\textbf{Doubles within ten:} $1{+}1{=}2$, \; $2{+}2{=}4$, \; $3{+}3{=}6$, \;
$4{+}4{=}8$, \; $5{+}5{=}10$.\\[3pt]
\textbf{Near double:} neighbours are a double \emph{and one more}. Use the
double of the \emph{smaller} part, then add $1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $3 + 4$
\step{The parts are neighbours, so reach for the double you already know instead of counting.}
\step{Double the smaller part: $3 + 3 = 6$. The other part is one bigger, so add one more: $6 + 1 = 7$.}
$3 + 4 = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $2 + 3$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{3. Make ten with ten-partners}

\begin{formulabox}
\textbf{Use it when:} the two parts fill a whole ten-frame together.\\[3pt]
\textbf{The ten-partners:} $1$ and $9$, \; $2$ and $8$, \; $3$ and $7$, \;
$4$ and $6$, \; $5$ and $5$.\\[3pt]
Learn these as \emph{pairs}, both ways round. Then a missing part is a
lookup, not a count: if one part is $6$, the partner is $4$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $3 + 7$
\step{These two parts are a ten-partner pair, so the frame fills exactly --- recall beats counting on seven.}
\step{On a ten-frame, $3$ counters leave $7$ empty spaces; filling them completes the frame.}
$3 + 7 = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $8 + 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} when you count on, do \emph{not} say the number you are
holding as your first count. Holding $6$ and adding $2$, say ``$7$, $8$'' ---
saying ``$6$, $7$'' lands you one too low every time.
\end{watchoutbox}
