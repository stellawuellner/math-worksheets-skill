\sstitleblock{Teen Numbers: Numerals and Number Words}

\skillheading{Skill 1 --- A teen number is one ten and some ones}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
Every teen number is built the same way: \ \textbf{one ten} $+$ \textbf{some ones}.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Fill a ten-frame all the way: that is the ten.
  \item The extra dots are the ones.
  \item In the numeral, the left digit 1 always means \emph{one ten} --- never one.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad One full ten-frame and 5 extra dots. Write the numeral.
\step{The full frame is a whole ten, so start counting at ten instead of at one.}
\step{Count on 5 more: 11, 12, 13, 14, 15. That is $10 + 5$.}
$\Rightarrow\quad \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad One full ten-frame and 8 extra dots. Write the numeral.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18}$}
\end{tryitbox}

\skillheading{Skill 2 --- Reading and writing teen number words}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
The ending \textbf{-teen} means ``and one ten''. Hear the ones part first, then
add the ten.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item four\,+\,teen $\rightarrow$ 4 ones and one ten.
  \item Learn the three that hide their pattern: \textbf{eleven}, \textbf{twelve},
        and \textbf{thirteen} (three), \textbf{fifteen} (five).
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write 16 as a number word, and say how many ones it has.
\step{Reading a numeral into a word: name the ones, then say \emph{-teen} for the ten.}
\step{Take the ten off the numeral to find the ones: $16 - 10$ leaves 6, so
      the word is \textbf{sixteen}.}
$\Rightarrow\quad \ans{6}$ \ ones
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write the numeral for \textbf{nineteen}.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two teen numbers}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
Two teen numbers both have one ten, so the tens cannot decide it --- \textbf{compare
the ones}. Fewer ones means the smaller number.\\[3pt]
$<$ means \emph{is less than}. \quad $>$ means \emph{is greater than}.
The open side always faces the bigger number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write $<$ or $>$ between 12 and 16.
\step{Both are teen numbers with one ten each, so the tens tie and only the ones matter.}
\step{12 has 2 ones, 16 has 6 ones; 2 ones is fewer than 6 ones.}
$\Rightarrow\quad 12 \ans{<} 16$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write $<$ or $>$ between 18 and 11.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} In 15 the digit 1 is worth \emph{ten}, not one. A student
who reads it as ``one and five'' gets 6 or thinks 14 are loose when a ten is bagged.
Say the ten first, every time.
\end{watchoutbox}
