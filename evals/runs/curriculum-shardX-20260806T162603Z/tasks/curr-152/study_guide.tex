\sstitleblock{Building Equivalent Ratio Tables}

% =================== SKILL 1 ===================
\skillheading{Scaling a Ratio Table Up or Down}

\begin{formulabox}
\textbf{Rule:} Every column of a ratio table shows the \textbf{same} ratio. To
build a new column, multiply or divide \textbf{both} quantities by the same
number (the multiplier). Never add the same amount to both — adding changes the
ratio.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ratio table starts with 2 stickers for 5 cents. Find
the cents column for 6 stickers.
\step{Find the multiplier from the known column: $6 \div 2 = 3$, so both
quantities are multiplied by 3.}
\step{Cents: $5 \times 3 = 15$, so 6 stickers cost $\ans{15}$ cents.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table starts with 3 goals for 4 saves. Find the saves
value in the column for 9 goals.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The most common table mistake is \textbf{adding} the
same number to both rows. Check any new column by simplifying: it must reduce to
the original ratio.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Building a Unit-Rate Column}

\begin{formulabox}
\textbf{Rule:} The most useful column in a ratio table is the column for
\textbf{one} unit. Divide both quantities by the first quantity to build it;
then any other column is just the unit column times the amount you want.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 3 pens cost 12 dollars. Build the one-pen column.
\step{Divide both quantities by 3 so the pen count becomes one.}
\step{Cost: $12 \div 3 = 4$, so one pen costs $\ans{4}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 6 erasers cost 18 dollars. How many dollars does one
eraser cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Finding a Missing Value with a Proportion}

\begin{formulabox}
\textbf{Rule:} Two equivalent ratios written as fractions make a
\textbf{proportion}. To find a missing value, find what one part was multiplied
(or divided) by, and apply the same multiplier to the other part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the missing value: $\dfrac{4}{9} = \dfrac{?}{36}$.
\step{The denominator was multiplied by 4, because $9 \times 4 = 36$.}
\step{Multiply the numerator by the same 4: $4 \times 4 = 16$, so the missing
value is $\ans{16}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the missing value: $\dfrac{5}{6} = \dfrac{?}{42}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Ratio table:] a table whose columns all show the same ratio
  \item[Multiplier:] the number both quantities are scaled by
  \item[Unit rate:] the value paired with one unit
  \item[Proportion:] an equation stating two ratios are equal
\end{description}
\end{multicols}
