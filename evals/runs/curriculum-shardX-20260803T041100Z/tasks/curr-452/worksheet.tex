\wstitleblock{One-Sided, Infinite, and End-Behavior Limits}{AP Calculus AB / BC}{}

\noindent\small
One-sided limits, infinite limits at a vertical asymptote, end behavior at
infinity, and continuity of a piecewise function. Decide which of the three
situations you are in before computing:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{One-sided.} The superscript $+$ or $-$ tells you which branch
        of the function to use and which sign a small factor carries.
  \item \textbf{Infinite.} A nonzero numerator over a denominator going to
        zero is $\pm\infty$; get the sign from the sign of the denominator on
        that side. Write $\infty$ or $-\infty$, or ``does not exist'' with a
        reason --- never a number.
  \item \textbf{End behavior.} As $x \to \pm\infty$, compare degrees, or
        divide numerator and denominator by the highest power of $x$ in the
        denominator.
\end{itemize}
Show the reasoning line that decides the answer, not just the answer.
\normalsize

\vspace{0.15cm}

\problem[4.5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to 2^{+}} \frac{|x - 2|}{x - 2}$}

\problem[4.5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to \infty} \frac{3x^2 + 5x - 1}{x^2 - 4}$}

\problem[5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to 5^{-}} \frac{1}{x - 5}$
\par\smallskip
State the sign of $x - 5$ for values just below $5$ as part of your reasoning.}

\problem[5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to -\infty} \frac{4x^3 - x}{2x^3 + 7}$}

\problem[6.5cm]{Let
\[ f(x) = \begin{cases} x^2 + 1, & x < 2 \\[2pt] 5x - 3, & x \ge 2
   \end{cases} \]
(a) Find $\displaystyle\lim_{x \to 2^{-}} f(x)$. \quad
(b) Find $\displaystyle\lim_{x \to 2^{+}} f(x)$. \quad
(c) Does $\displaystyle\lim_{x \to 2} f(x)$ exist? Explain in one sentence.
\par\smallskip
(a) \underline{\hspace{2.4cm}} \qquad (b) \underline{\hspace{2.4cm}}\noansline}

\problem[5.5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to -4^{+}} \frac{2x}{x + 4}$}

\problem[5.5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to \infty} \frac{\sqrt{9x^2 + 4}}{2x + 1}$
\par\smallskip
Divide by the highest power of $x$ in the denominator, and be explicit about
what $\sqrt{x^2}$ equals when $x$ is large and positive.}

\problem[5.5cm]{Evaluate: \quad
$\displaystyle \lim_{x \to 0^{+}} \left( \frac{1}{x} - \frac{1}{x^2} \right)$
\par\smallskip
Combine over a common denominator first --- the two pieces separately are both
infinite, and their difference is not settled until they are combined.}

\problem[6cm]{Let
\[ g(x) = \begin{cases} x^2 + k, & x < 3 \\[2pt] 4x - 1, & x \ge 3
   \end{cases} \]
Find the value of the constant $k$ that makes $g$ continuous at $x = 3$. Write
the one-sided limits first, set them equal, and solve.}

\problem[7cm]{\textbf{Synthesis challenge.} Let
$\displaystyle h(x) = \frac{2x^2 - 3x}{x^2 - x - 6}$.
\par\smallskip
(a) Find $\displaystyle\lim_{x \to \infty} h(x)$. \quad
(b) Find $\displaystyle\lim_{x \to 3^{-}} h(x)$. \quad
(c) Find $\displaystyle\lim_{x \to 3^{+}} h(x)$. \quad
(d) Using your answers, describe every vertical and horizontal asymptote of
the graph of $h$ in words.
\par\smallskip
(a) \underline{\hspace{2.2cm}} \quad (b) \underline{\hspace{2.2cm}} \quad
(c) \underline{\hspace{2.2cm}}\noansline}
