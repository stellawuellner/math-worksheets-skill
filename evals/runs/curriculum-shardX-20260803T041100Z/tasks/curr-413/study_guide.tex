\sstitleblock{Fitting and Interpreting Exponential Models}

\noindent\small Every model here is $y = a \cdot b^{\,t}$: $a$ is the value at $t = 0$,
$b$ is the factor multiplied on each time step. Fitting finds $a$ and $b$ from data;
interpreting reads them back as a starting amount and a rate.\normalsize

\skillheading{Skill 1 --- Fitting $a$ and $b$ from two points}
\begin{formulabox}\textbf{Method:} write one equation per data point, then \emph{divide}
the equations so $a$ cancels:
$\dfrac{a b^{\,t_2}}{a b^{\,t_1}} = b^{\,t_2 - t_1} = \dfrac{y_2}{y_1}$.\\[3pt]
Take the $(t_2 - t_1)$-th root for $b$, then substitute back into either equation for $a$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Fit $y = a \cdot b^{\,t}$ through $(0, 50)$ and $(3, 86.4)$.
\step{Two unknowns, two points --- divide the equations so the unknown $a$ cancels and only $b$ is left.}
\step{$a = 50$ from the first point, so $50b^{\,3} = 86.4$, giving $b^{\,3} = 1.728$ and $\ans{b = 1.2}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Fit $y = a \cdot b^{\,t}$ through $(0, 80)$ and $(3, 270)$: find $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{b = 1.5}$}\end{tryitbox}

\skillheading{Skill 2 --- Interpreting $a$ and $b$}
\begin{formulabox}\textbf{Reading the model:} $a$ is the amount at $t = 0$ (in the units of
the quantity), and $b$ is the multiplier per time step.
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item $b > 1$: growth, with rate $r = b - 1$ per step.
  \item $b < 1$: decay, with rate $r = 1 - b$ per step.
\end{itemize}
The rate is a percentage \emph{of the current amount}, so the change per step is not a
constant number of units.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $A(t) = 600 \cdot (1.07)^{\,t}$. State the growth rate per step.
\step{The base is a multiplier, so compare it to $1 + r$ rather than reading it as an amount.}
\step{$1.07 = 1 + r$ gives $r = \ans{0.07}$, a 7 percent increase per step, applied to whatever the current amount is.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $A(t) = 900 \cdot (0.92)^{\,t}$. State the decay rate per step.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.08}$}\end{tryitbox}

\skillheading{Skill 3 --- Solving for time with logarithms}
\begin{formulabox}\textbf{Method:} isolate the power first, then take logs:
$a b^{\,t} = y \Rightarrow b^{\,t} = \dfrac{y}{a} \Rightarrow
t = \dfrac{\ln (y/a)}{\ln b}$.\\[3pt]
For doubling, $y/a = 2$, so $t = \ln 2 / \ln b$ --- independent of $a$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad When does $600 \cdot (1.07)^{\,t}$ reach 900?
\step{The unknown sits in the exponent, so isolate the power and take logarithms of both sides.}
\step{$(1.07)^{\,t} = 900/600$, so $t = \dfrac{\ln (900/600)}{\ln 1.07} = \ans{5.99}$ steps.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad When does $500 \cdot (1.2)^{\,t}$ reach 1000?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3.80}$}\end{tryitbox}

\skillheading{Skill 4 --- Rewriting the base}
\begin{formulabox}\textbf{To the natural base:} $b^{\,t} = e^{(\ln b)t}$, so
$a b^{\,t} = a e^{k t}$ with $k = \ln b$. Here $k > 0$ means growth and $k < 0$ decay.\\[3pt]
\textbf{To a new time step:} if one old step is $n$ new steps, the new factor is
$b^{1/n}$ (twelve months to a year: $b_{\text{month}} = b^{1/12}$).
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Write $(1.07)^{\,t}$ in the form $e^{k t}$.
\step{The two forms must agree for every $t$, so match the bases: $e^{k}$ has to equal the given base.}
\step{$e^{k} = 1.07$ means $k = \ln 1.07$, so $(1.07)^{\,t} = \ans{e^{t \ln 1.07}}$ (about $e^{0.0677t}$).}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $(0.92)^{\,t}$ in the form $e^{k t}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{e^{t \ln 0.92}}$}\end{tryitbox}
