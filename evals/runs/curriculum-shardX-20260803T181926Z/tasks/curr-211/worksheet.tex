\wstitleblock{Solving Linear Equations with Variables on Both Sides}{Grade 8 / Pre-Algebra}{}

\begin{center}\small\textit{Reading a balance or table model as an equation,
collecting the variable terms on one side, and distributing before you solve}\end{center}

\noindent\textbf{How to work each problem.}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item \textbf{The balance model.} In the pan pictures, $\square$ is a sealed bag
        holding the same unknown number of counters --- call it $x$ --- and
        $\bullet$ is one loose counter. The pans balance, so the two sides are
        equal. Removing the \emph{same} thing from both pans keeps the balance;
        that is exactly what a legal move on an equation does.
  \item \textbf{Collect, then isolate.} Move the variable terms to one side and
        the constants to the other by adding or subtracting the same quantity on
        both sides. Then divide by the coefficient.
  \item \textbf{Distribute first} when a side is a product, such as $2(x+5)$.
  \item \textbf{Check every answer} by substituting it back into the
        \emph{original} equation and evaluating both sides.
  \item Show the balanced move beside each line, for example
        ``$-\,2x$ on both sides''.
\end{itemize}

\problem[6cm]{\textbf{Two plans, one break-even point.} A club is choosing between
two ways to pay for a season of practice sessions. Let $x$ be the number of
sessions.

\begin{center}
\begin{tabular}{|l|l|}\hline
\textbf{Plan} & \textbf{Total cost, in dollars}\\\hline
Plan A: pay a joining fee, then a low rate per session & $40 + 12x$\\\hline
Plan B: no joining fee, higher rate per session & $20x$\\\hline
\end{tabular}
\end{center}

\noindent Write the equation that says the two plans cost the same, then solve it.
How many sessions is that?}

\problem[6cm]{\textbf{What happens here?} Distribute the left side, then collect the
variable terms.
\[ 3(2x + 5) = 6x + 15 \]
Write the equation you are left with after collecting, and describe every number
that makes the original equation true.}

\problem[6cm]{\textbf{Challenge --- and justify it.} Distribute the left side, then
collect the variable terms.
\[ 4(x + 2) = 4x + 5 \]
Write the equation you are left with after collecting. Describe the solution set,
and explain in one sentence why the variable disappearing tells you what it does.}
