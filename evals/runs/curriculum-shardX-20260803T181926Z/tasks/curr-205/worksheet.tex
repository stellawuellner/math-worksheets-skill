\begin{center}
\textit{Estimating a square root, comparing two quantities, and ordering a
mixed list}
\end{center}

\noindent\textbf{Before you start.} A list can mix fractions, decimals and
square roots. They can only be ordered once they are all written the same way.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Trap the root} between the two perfect squares on either side of
the radicand, then refine.
\item \textbf{Convert everything to a decimal} --- and carry enough decimal
places that no two values still look equal.
\item \textbf{Two positive numbers can also be compared by squaring} them, since
squaring keeps the order.
\item On the negative side of zero the order \textbf{reverses}: the number
furthest from zero is the least.
\end{itemize}

\medskip

\problem[5cm]{Between which two consecutive whole numbers does $\sqrt{50}$ lie?
Use that to check yourself, then write $\sqrt{50}$ rounded to the nearest
hundredth.}

\problem[4cm]{Write $>$, $<$ or $=$ between the two values.

\smallskip
\hspace{1cm}$\sqrt{9}$ \; \underline{\hspace{1.2cm}} \; $2.9$}

\problem[5cm]{Order these from least to greatest. Show the decimal you used for
each one.

\smallskip
\hspace{1cm}$3.5$, \quad $\sqrt{10}$, \quad $\dfrac{10}{3}$}

\problem[5cm]{Write $\sqrt{20}$ rounded to the nearest hundredth, and name the
two consecutive whole numbers it lies between.}

\problem[4cm]{Write $>$, $<$ or $=$ between the two values. Squaring both is a
quick way to be sure.

\smallskip
\hspace{1cm}$\sqrt{2}$ \; \underline{\hspace{1.2cm}} \; $1.5$}

\problem[5cm]{Order these from least to greatest.

\smallskip
\hspace{1cm}$\sqrt{30}$, \quad $5.5$, \quad $\dfrac{27}{5}$}

\problem[5cm]{Many people are taught that $\pi$ ``is'' $\dfrac{22}{7}$. Write
$>$, $<$ or $=$ between them, using enough decimal places to be certain.

\smallskip
\hspace{1cm}$\pi$ \; \underline{\hspace{1.2cm}} \; $\dfrac{22}{7}$}

\problem[5cm]{Write $\sqrt{85}$ rounded to the nearest hundredth. Say which
perfect square it is closer to, and why that tells you the answer is not
half-way between the whole numbers.}

\problem[5cm]{Order these from \emph{greatest} to \emph{least} --- read the
direction carefully.

\smallskip
\hspace{1cm}$\sqrt{45}$, \quad $6.7$, \quad $\dfrac{20}{3}$}

\problem[6cm]{Two values that a calculator display can make look equal.

\smallskip
(a) Write $\dfrac{17}{4}$ as a decimal.

\smallskip
(b) Now write $>$, $<$ or $=$ between the two values.

\smallskip
\hspace{1cm}$\sqrt{18}$ \; \underline{\hspace{1.2cm}} \; $\dfrac{17}{4}$}

\problem[6cm]{Order these four values from least to greatest. Two decimal
places will not separate all of them --- carry more.

\smallskip
\hspace{1cm}$\sqrt{2}$, \quad $1.42$, \quad $\dfrac{7}{5}$, \quad $1.41$}

\problem[7cm]{\textbf{Synthesis.} Order these five values from least to
greatest, then sketch a number line and mark each one on it.

\smallskip
\hspace{1cm}$-\sqrt{5}$, \quad $-2.3$, \quad $-\dfrac{9}{4}$, \quad
$\sqrt{6}$, \quad $2.4$}
