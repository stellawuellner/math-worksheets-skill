\wstitleblock{Perimeter and Area of Polygons on the Coordinate Plane}{High School Geometry}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Perimeter} is the sum of the side lengths, and each side length
        comes from the distance formula
        $d = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}$.
  \item \textbf{Area} can be found by the shoelace method: going once around the
        vertices in order, add every $x_i y_{i+1}$, subtract every
        $x_{i+1} y_i$, then halve the size of the result.
  \item Give exact values where they are whole numbers; otherwise round to two
        decimal places. Write the unit on every answer.
  \item Where a grid is provided, sketching the figure is optional --- it is
        there so you can check that your answer is sensible.
\end{itemize}

\problem{A city park is mapped as a quadrilateral with corners at $(0,0)$,
