\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every problem follows the same three moves: find the radius $R$ from the ratio of coefficients, then test each endpoint by substituting it back into the original series, then write the interval with the right brackets.
  \item The ratio test never decides an endpoint --- it returns $1$ there by construction. Each endpoint needs its own test.
  \item Write intervals in interval notation. A square bracket means that endpoint converges; a round bracket means it does not.
\end{itemize}

\problem[5.5cm]{Consider $\displaystyle\sum_{n=0}^{\infty}\frac{x^{n}}{n!}$. The ratio of coefficients $\left|c_{n}/c_{n+1}\right|$ simplifies to $n+1$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact sum of the series at $x = 2$.
  \item[(c)] State the interval of convergence and explain why no endpoint test is needed.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n\,3^{n}}$. The ratio of coefficients simplifies to $\dfrac{3(n+1)}{n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact sum of the series at $x = 1$.
  \item[(c)] Test both endpoints, naming the series each one produces, and state the interval of convergence.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{(x-2)^{n}}{n^{2}}$. The ratio of coefficients simplifies to $\dfrac{(n+1)^{2}}{n^{2}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact sum of the series at its right endpoint.
  \item[(c)] Test both endpoints and state the interval of convergence in interval notation.
\end{itemize}}

\problem[5.5cm]{Consider $\displaystyle\sum_{n=0}^{\infty} n!\,x^{n}$. The ratio of coefficients simplifies to $\dfrac{1}{n+1}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] State the interval of convergence and explain what happens for every other value of $x$.
\end{itemize}}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n}(x+1)^{n}}{n\,2^{n}}$. The ratio of coefficients simplifies to $\dfrac{2(n+1)}{n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact sum of the series at its right endpoint, $x = 1$.
  \item[(c)] Test both endpoints and state the interval of convergence.
\end{itemize}}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{\sqrt{n}}$. The ratio of coefficients simplifies to $\dfrac{\sqrt{n+1}}{\sqrt{n}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Find $\displaystyle\lim_{n\to\infty}\frac{1}{\sqrt{n}}$, the size of the terms at either endpoint.
  \item[(c)] The two endpoints give the same term sizes but different verdicts. Explain why, and state the interval of convergence.
\end{itemize}}

\problem[7cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{(3x-6)^{n}}{n}$. Writing $(3x-6)^{n} = 3^{n}(x-2)^{n}$ makes the coefficients $c_{n} = 3^{n}/n$, whose ratio simplifies to $\dfrac{n+1}{3n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact sum of the series at $x = \frac{11}{5}$.
  \item[(c)] Test both endpoints, state the interval, and explain what goes wrong if the factor of $3$ is not pulled out first.
\end{itemize}}

\problem[7cm]{A dimensionless response index for a medication is modelled by $\displaystyle S(d) = \sum_{n=1}^{\infty}\frac{(d-4)^{n}}{n\,5^{n}}$, where $d$ is the dose in milligrams. The ratio of coefficients simplifies to $\dfrac{5(n+1)}{n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the radius of convergence.
  \item[(b)] Write the exact value of $S(6)$, the index at a dose of $6$ milligrams.
  \item[(c)] Test both endpoints, state the interval of convergence, and say which doses in milligrams the model is actually defined for.
\end{itemize}}
