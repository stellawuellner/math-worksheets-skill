\sstitleblock{Multiplication: Rows, Arrays, and the Total}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Finding the total of an array}

\begin{formulabox}
\textbf{Rule:}\quad rows $\times$ number in each row $=$ total.\par
The total counts \emph{everything} in the array, so it is usually bigger than either number you multiply.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An array has 4 rows with 6 dots in each row. How many dots in all?
\step{The question asks for the total, so multiply the rows by the number in each row.}
\step{$4 \times 6 = 24$. Skip counting by 6 checks it: $6, 12, 18, 24$.}

$\Rightarrow\quad\ans{24}$ dots
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} An array has 3 rows with 7 dots in each row. How many dots in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Equal groups in a story}

\begin{formulabox}
\textbf{Rule:}\quad number of groups $\times$ number in each group $=$ total.\par
Bags, boxes, and shelves in a story work just like rows in an array.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} There are 5 bags with 8 marbles in each bag. How many marbles in all?
\step{The story means 5 equal groups of 8, so the total needs multiplication.}
\step{$5 \times 8 = 40$.}

$\Rightarrow\quad\ans{40}$ marbles
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} There are 6 boxes with 4 pencils in each box. How many pencils in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Fixing a rows-versus-total mix-up}

\begin{formulabox}
\textbf{Check yourself:}\quad Does my answer count \emph{everything}?\par
Adding the rows to the number in each row mixes two different counts --- it never gives the total. If your answer is one of the numbers from the problem (or their sum), look again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A student wrote $5 + 6 = 11$ for an array of 5 rows with 6 dots in each row. Fix the mistake.
\step{The sum 11 mixes the row count with the dots in one row --- it does not count all five rows of dots, so multiply instead.}
\step{$5 \times 6 = 30$.}

$\Rightarrow\quad\ans{30}$ dots, not 11
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A student wrote $2 + 9 = 11$ for an array of 2 rows with 9 dots in each row. What is the real total?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The two numbers in an array problem name the \emph{rows} and the \emph{number in each row}. Neither one is the total by itself --- the total comes from multiplying them.
\end{watchoutbox}
