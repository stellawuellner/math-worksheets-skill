\sstitleblock{Direction, Form, and Strength in Scatter Plots}

\skillheading{Skill 1 --- Direction: which way does the cloud go?}

\begin{formulabox}
\textbf{Ask:} as you move right along the horizontal axis, do the points go \textit{up} or
\textit{down}?\\[3pt]
Up together $\rightarrow$ \textbf{positive} association. \quad One up while the other goes down
$\rightarrow$ \textbf{negative}. \quad No tilt at all $\rightarrow$ \textbf{no association}.
Always say it about the two real quantities, not just ``up''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Read the value at $x = 4$, then name the direction.
\begin{center}
\begin{tikzpicture}
\begin{axis}[width=5.4cm, height=3.0cm, axis lines=left, xmin=0, xmax=6, ymin=0, ymax=15,
  tick label style={font=\tiny}, grid=both, grid style={line width=0.15pt, draw=gray!55},
  major grid style={line width=0.3pt, draw=gray!55}, enlargelimits=false]
\addplot[only marks, mark=*, mark size=1.2pt, blue] coordinates {(1,3)(2,5)(3,8)(4,10)(5,13)};
\end{axis}
\end{tikzpicture}
\end{center}
\step{Reading a value is a lookup; naming the direction means asking which way the points tilt.}
\step{Above $x = 4$ the point sits level with $10$, and the whole cloud rises left to right, so the direction is positive.}
$\Rightarrow\quad \ans{10}$, positive association
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has points $(1,14)$, $(2,11)$, $(3,9)$, $(4,6)$, $(5,3)$. Read the
value at $x = 3$ and name the direction.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$, negative association}
\end{tryitbox}

\skillheading{Skill 2 --- Form: straight line or curve?}

\begin{formulabox}
\textbf{Ask:} could one straight line follow \textit{all} the points?\\[3pt]
Yes $\rightarrow$ \textbf{linear}. No --- the pattern rises then falls, or bends steeply
$\rightarrow$ \textbf{non-linear}. A useful test: find the highest point. If it sits in the middle
rather than at an end, the form cannot be linear.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the input with the greatest output, then name the form.
\begin{center}
\begin{tikzpicture}
\begin{axis}[width=5.4cm, height=3.0cm, axis lines=left, xmin=0, xmax=6, ymin=0, ymax=13,
  tick label style={font=\tiny}, grid=both, grid style={line width=0.15pt, draw=gray!55},
  major grid style={line width=0.3pt, draw=gray!55}, enlargelimits=false]
\addplot[only marks, mark=*, mark size=1.2pt, blue] coordinates {(1,5)(2,9)(3,11)(4,9)(5,5)};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The form question is about shape, so locate the peak and see whether it sits at an end.}
\step{The tallest point is above $x = 3$, in the middle, and the points fall away on both sides --- an arch, not a line.}
$\Rightarrow\quad \ans{3}$, non-linear form
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has points $(1,2)$, $(2,5)$, $(3,8)$, $(4,11)$, $(5,14)$. Find the
input with the greatest output and name the form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$, linear form}
\end{tryitbox}

\skillheading{Skill 3 --- Strength: how tightly do the points fit?}

\begin{formulabox}
\textbf{Ask:} how far do the points stray from the pattern?\\[3pt]
Hugging the line $\rightarrow$ \textbf{strong}. Scattered widely around it $\rightarrow$
\textbf{weak}. A quick numerical clue: for inputs close together, a small \textit{range} of
outputs means the points are packed tightly, a large range means they are spread out.
A single far-away point is an \textbf{outlier} --- describe it separately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Five points have outputs $10, 11, 12, 13, 14$. Find the range and say
what it suggests about strength.
\step{Strength is about spread, so measure the spread of the outputs with the range.}
\step{Largest $14$, smallest $10$, so the range is $14 - 10$ --- a narrow band for five points.}
$\Rightarrow\quad \ans{4}$, a tightly packed, strong pattern
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Five points have outputs $4, 15, 7, 13, 9$. Find the range and say what it
suggests.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$, widely spread, so weak}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} direction, form and strength are three separate questions. A plot can be
positive (direction) and still be non-linear (form) and weak (strength) --- answer all three when
you are asked to describe an association.
\end{watchoutbox}
