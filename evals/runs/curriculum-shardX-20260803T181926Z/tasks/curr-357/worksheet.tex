\wstitleblock{Polynomial Long Division}{Algebra 2 \quad Polynomial Arithmetic}{}

\begin{center}
\textit{Dividing polynomials by long division: linear divisors, dividends with
a missing power that needs a zero placeholder, and divisors of degree two or
more}
\end{center}

\noindent\textbf{How to work these problems.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Write both polynomials in descending powers, and put a zero coefficient
in for every power that is missing. A gap in the columns is where the
algorithm goes wrong.
\item Each round: divide the leading term of what is left by the leading term
of the divisor, multiply the whole divisor by that quotient term, subtract,
and bring down the next term. Subtracting means changing every sign.
\item Stop when the leftover has lower degree than the divisor. That leftover
is the remainder.
\item Write the answer as
\emph{quotient} $+$ \emph{remainder} over \emph{divisor}, and write
\emph{quotient} alone when the remainder is zero.
\end{itemize}

\vspace{0.15cm}

\problem[6cm]{Divide $x^2 + 7x + 10$ by $x + 2$.}

\problem[6cm]{Divide $x^2 + 5x + 9$ by $x + 2$.}

\problem[7cm]{Divide $x^3 - 2x^2 - 5x + 6$ by $x - 1$.}

\problem[8cm]{Divide $2x^3 + 3x^2 - 5x + 4$ by $x + 3$. Then substitute
$x = -3$ into the dividend $2x^3 + 3x^2 - 5x + 4$. You should get the same
number as your remainder --- write it down as a check.}

\problem[7cm]{Divide $x^3 - 8$ by $x - 2$. Write the dividend with every power
showing before you start.}

\problem[8cm]{Divide $6x^3 - x^2 - 5x + 2$ by $2x - 1$.}

\problem[8cm]{Divide $2x^3 + 5x - 3$ by $x + 1$. Then substitute $x = -1$ into
the dividend $2x^3 + 5x - 3$ as a check on your remainder.}

\problem[9cm]{Divide $x^4 + 2x^3 - 3x^2 + 4x - 5$ by $x^2 + 1$. Remember that
the division stops as soon as what is left has degree below $2$.}

\problem[9cm]{Divide $x^4 - 5x^2 + 4$ by $x - 2$.}

\problem[9cm]{Divide $3x^3 - 2x^2 + x - 4$ by $x - 1$. Then check your work:
multiply the divisor by your quotient, add your remainder, and write out the
polynomial you get.}

\problem[10cm]{Divide $2x^4 - 3x^3 + x - 5$ by $x^2 - x + 1$.}

\problem[9cm]{Divide $x^3 - 4x^2 + x + 6$ by $x - 3$. The remainder is zero,
so $x - 3$ is a factor. Use your quotient to write $x^3 - 4x^2 + x + 6$ as a
product of three linear factors.}
