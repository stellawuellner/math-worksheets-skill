\sstitleblock{Powers of Ten and Place Value}

\skillheading{Skill 1 --- Multiplying by 10, 100, or 1,000}

\begin{formulabox}
\textbf{Rule:} multiplying by a power of ten shifts every digit \textbf{left} —
one place per factor of ten. In practice: attach that many zeros.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 \times 1{,}000$
\step{Three factors of ten, so shift the 8 three places left (attach three
zeros).}
\step{$8 \times 1{,}000 = 8{,}000$}
$\Rightarrow\quad \ans{8{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $32 \times 100 = $ ?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3{,}200}$}
\end{tryitbox}

\skillheading{Skill 2 --- Dividing by 10, 100, or 1,000}

\begin{formulabox}
\textbf{Rule:} dividing by a power of ten shifts every digit \textbf{right} —
one place per factor of ten. In practice: remove that many zeros.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $4{,}700 \div 100$
\step{Two factors of ten, so shift the digits two places right (remove two
zeros).}
\step{$4{,}700 \div 100 = 47$}
$\Rightarrow\quad \ans{47}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $9{,}000 \div 10 = $ ?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{900}$}
\end{tryitbox}

\skillheading{Skill 3 --- Exponent Form}

\begin{formulabox}
\textbf{Rule:} $10^n$ means $n$ factors of ten — a 1 followed by $n$ zeros.
The exponent counts the zeros, and it counts the place shifts too.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $6 \times 10^3$
\step{The exponent 3 says: shift the 6 three places left.}
\step{$6 \times 10^3 = 6 \times 1{,}000 = 6{,}000$}
$\Rightarrow\quad \ans{6{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $10^4 = $ ?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10{,}000}$}
\end{tryitbox}

\skillheading{Skill 4 --- Place-Value Shifts}

\begin{formulabox}
\textbf{Rule:} the same digit one place to the \textbf{left} is worth
\textbf{10 times} as much; one place to the \textbf{right} is worth one tenth
as much.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The 9 in $92{,}000$ compared with the 9 in $9{,}200$.
\step{Find each value from its place: $90{,}000$ (ten-thousands) and $9{,}000$
(thousands), one place apart.}
\step{$90{,}000 \div 9{,}000 = 10$}
$\Rightarrow\quad \ans{10}$ times as much
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The 2 in $2{,}500$ is how many times the 2 in $250$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}
