% =================== SKILL 1 ===================
\skillheading{Adding within 20: make a ten}

\begin{formulabox}
\textbf{Rule:} Break one number apart to make a ten first. Tens are easy to
add on.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 + 6$
\step{Both parts are known and we want the total --- this is adding, and making a ten makes it fast.}
\step{8 needs 2 to make 10. Break 6 into 2 and 4: \ $8 + 2 = 10$, then $10 + 4 = 14$.}

$\Rightarrow\quad 8 + 6 = \ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $7 + 6$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Subtracting within 20: count up}

\begin{formulabox}
\textbf{Rule:} To take away, you can count UP from the smaller number to the
bigger one. The jumps you count are the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $15 - 8$
\step{The whole and one part are known and we want what is left --- this is subtracting, and counting up avoids counting backwards.}
\step{From 8 up to 10 is 2 jumps; from 10 up to 15 is 5 more. \ $2 + 5 = 7$.}

$\Rightarrow\quad 15 - 8 = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $14 - 8$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Missing numbers: use the fact family}

\begin{formulabox}
\textbf{Rule:} Addition and subtraction share fact families. If a part is
missing, subtract the known part from the whole. If the whole is missing, add
the parts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $6 + \square = 14$
\step{First decide what is missing --- here the whole 14 is known, so the blank is a part: subtract to find it.}
\step{$14 - 6 = 8$. \ Check: $6 + 8 = 14$. It works!}

$\Rightarrow\quad 6 + \ans{8} = 14$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $9 + \square = 16$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Read before you work! In a mixed set, some problems
add, some take away, and some hide a number. Ask first: what is missing ---
a part, or the whole?
\end{watchoutbox}
