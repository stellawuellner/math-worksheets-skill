\sstitleblock{Getting the Bounds, the Signs and the Variable Right}

\skillheading{Skill 1 --- Bound order carries a sign}

\begin{formulabox}
\textbf{Reversal:} $\displaystyle\int_a^b f = -\int_b^a f$. \qquad
\textbf{Additivity:} $\displaystyle\int_a^c f + \int_c^b f = \int_a^b f$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The FTC subtracts in the order written: lower bound's value comes off the
        upper bound's value, even when the lower number is the larger one.
  \item If the variable sits at the \emph{lower} bound of an accumulation
        function, flip it first --- the flip is where the minus comes from.
  \item Under a substitution the bounds are values of the \emph{old} variable.
        Convert them the moment you write $u = \ldots$, or back-substitute to
        $x$ before evaluating.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Evaluate $\displaystyle\int_{4}^{1} 2x\,dx$.
\step{The bounds run backwards, so resist reordering them and subtract in the
order written, $F(1) - F(4)$.}
\step{$F(x) = x^2$, so the value is $1 - 16$}

The integral is $\ans{-15}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Evaluate $\displaystyle\int_{5}^{2} 3x^2\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-117}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtract the whole lower value}

\begin{formulabox}
\textbf{FTC Part 2:} $\displaystyle\int_a^b f(x)\,dx = F(b) - F(a)$, where
$F' = f$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Write $\bigl[F(x)\bigr]_a^b$ with real brackets, then
        $\bigl(F(b)\bigr) - \bigl(F(a)\bigr)$ with real parentheses: the minus
        reaches every term of $F(a)$.
  \item Antiderivative signs to keep straight:
        $\int \sin x\,dx = -\cos x$ and $\int \cos x\,dx = \sin x$.
  \item Sign check: if the integrand is positive across the interval and the
        bounds increase, the answer must be positive.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Evaluate $\displaystyle\int_{1}^{2} (4x - 3)\,dx$.
\step{Antidifferentiate first and keep the lower value inside parentheses, since
that is where the sign is usually lost.}
\step{$\bigl[2x^2 - 3x\bigr]_1^2 = (8 - 6) - (2 - 3) = 2 - (-1)$}

The integral is $\ans{3}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Evaluate $\displaystyle\int_{1}^{3} (2x - 5)\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$}
\end{tryitbox}

\skillheading{Skill 3 --- The variable of integration is a dummy}

\begin{formulabox}
\textbf{FTC Part 1:} if $F(x) = \displaystyle\int_a^x f(t)\,dt$ then
$F'(x) = f(x)$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Inside the integral, $t$ is consumed: it cannot appear in the value of a
        definite integral, and a bound is never substituted into the integrand.
  \item $F'$ is the integrand's \emph{rule} with the bound variable in place of
        $t$ --- no antidifferentiating and no evaluating required.
  \item With the variable at the lower bound, flip and negate first (Skill 1).
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Let $F(x) = \displaystyle\int_{1}^{x} (t^2 + 3)\,dt$. Find
$F'(4)$.
\step{The variable is already at the upper bound, so FTC Part 1 applies directly
and no antiderivative is needed at all.}
\step{$F'(x) = x^2 + 3$, so $F'(4) = 16 + 3$}

The value is $\ans{19}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Let $G(x) = \displaystyle\int_{2}^{x} (5t + 1)\,dt$. Find
$G'(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An answer that matches the expected size but not the sign
is almost never an arithmetic error --- check the bound order, the direction of
an accumulation, and the sign of the antiderivative before re-doing the algebra.
\end{watchoutbox}
