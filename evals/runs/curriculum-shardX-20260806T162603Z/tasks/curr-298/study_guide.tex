\sstitleblock{Writing and Using Piecewise Rules for Fees}

\skillheading{Skill 1 --- Writing the rule from a price list}

\begin{formulabox}
\textbf{Shape:} \quad $C(x) = \begin{cases} \text{first formula},& \text{first condition} \\ \text{second formula},& \text{second condition}\end{cases}$\par\smallskip
\textbf{The second branch starts where the first stopped.} If the boundary is at $x = B$ and the charge there is $P$, the second branch is $P + (\text{new rate})(x - B)$. Writing it as (new rate)$\times x$ throws away everything already paid --- the single most common mistake on this topic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A gym charges $\$30$ for up to $10$ visits a month, then $\$2$ for each extra visit. What does a month of $15$ visits cost?
\step{The price changes at $10$ visits, so this needs two branches, and $15$ is past the boundary --- use the second one.}
\step{At the boundary $\$30$ has been paid, so the rule is $30 + 2(v-10)$ for $v > 10$; at $v = 15$ that is $30 + 2(5)$.}
\par
$C(15) = \ans{40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A club charges $\$40$ for up to $12$ visits, then $\$3$ per extra visit. Find the cost of $20$ visits.\\[2pt]
\rotatebox{180}{\footnotesize check: $C(20) = \ans{64}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the conditions are part of the answer. A pair of formulas with no ``when'' attached does not say which one to use, and it is not a piecewise rule.
\end{watchoutbox}

\skillheading{Skill 2 --- Using the rule to find a charge}

\begin{formulabox}
\textbf{Two steps, in this order:} test the input against the conditions to see which branch owns it, \emph{then} substitute into that branch only.\par\smallskip
Check a boundary value in both branches when you build a rule: a sensible price list gives the same answer either way, so the cost never jumps.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A car park charges $\$6$ for the first hour and $\$4$ for each hour after that, so $C(h) = 6 + 4(h-1)$ for $h > 1$. What does a $5$-hour stay cost?
\step{Five hours is past the one-hour boundary, so the second branch is the one that applies --- test the input before substituting.}
\step{$C(5) = 6 + 4(5-1) = 6 + 16$.}
\par
$C(5) = \ans{22}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Water costs $\$0.15$ per unit for the first $100$ units, so $C(k) = 0.15k$ there. Find the cost of $80$ units.\\[2pt]
\rotatebox{180}{\footnotesize check: $C(80) = \ans{12}$}
\end{tryitbox}

\skillheading{Skill 3 --- Working back from a charge}

\begin{formulabox}
\textbf{Pick the branch first, using the boundary charge.} Work out what the fee is exactly at the boundary. If the fee you were given is larger, the input lies in the upper branch; if smaller, the lower one.\par\smallskip
Then solve that branch's equation --- and check at the end that your answer really satisfies the condition you assumed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad With $C(h) = 6 + 4(h-1)$ for $h > 1$, a stay cost $\$30$. How many hours was it?
\step{At the boundary of one hour the charge is only $\$6$, and $\$30$ is larger, so the upper branch is the one to solve.}
\step{$6 + 4(h-1) = 30$ gives $4(h-1) = 24$, so $h - 1 = 6$.}
\par
$h = \ans{7}$ hours, and $7 > 1$, so the branch was right.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad With $C(n) = 20 + 2.5(n-10)$ for $n > 10$, a bill came to $\$45$. Find $n$.\\[2pt]
\rotatebox{180}{\footnotesize check: $n = \ans{20}$}
\end{tryitbox}
