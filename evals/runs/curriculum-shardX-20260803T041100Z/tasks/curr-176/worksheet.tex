\wstitleblock{Solving One-Step Equations with Rational Coefficients}{Grades 6--7 \quad Expressions and Equations}{}

\begin{center}
\textit{Undoing a unit-fraction coefficient, multiplying by the reciprocal of a
non-unit fraction, dividing to undo a decimal coefficient, working with a
negative rational coefficient, and checking a solution by substitution}
\end{center}

\noindent\textbf{The one idea behind every problem.} A one-step equation says
``something was done to $x$, and the result is this.'' You undo that one thing
on \emph{both} sides. When the coefficient is a fraction, the fastest undo is
multiplying by its \textbf{reciprocal}; when it is a decimal, the undo is
division.

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Show the undo step you used, written on both sides of the equation.
\item Leave fraction answers as fractions in lowest terms unless the problem
gives decimals.
\item Write your final answer on the answer line.
\end{itemize}

\problem[6cm]{A tape diagram is cut into 3 equal parts. One part is worth 7.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\strut\;7\; & \qquad & \qquad \\
\hline
\end{tabular}
\end{center}
\par
\textbf{(a)} Write the equation the diagram shows, using $x$ for the whole bar.

\smallskip
\textbf{(b)} Solve it. What is $x$?}

\problem[6cm]{A jar of marbles is being emptied. After a scoop, $0.4$ of the
marbles that were in the jar are on the table, and that is 6 marbles.

\par\smallskip
\[
0.4x = 6
\]
\textbf{(a)} What single operation undoes multiplying by $0.4$?

\smallskip
\textbf{(b)} Do it to both sides and solve for $x$.}

\problem[6.5cm]{A tape diagram is cut into 3 equal parts, and \emph{two} of
those parts together are worth 8.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\strut\;$\ast$\; & \;$\ast$\; & \qquad \\
\hline
\end{tabular}
\par\smallskip
{\small the two starred parts together are worth 8}
\end{center}
\par
\textbf{(a)} Explain in one sentence why $x$ cannot be 4.

\smallskip
\textbf{(b)} Write and solve the equation $\dfrac{2}{3}x = 8$.}

\problem[6cm]{A recipe uses one quarter of a bag of flour, which is 9 ounces.

\par\smallskip
\[
\frac{x}{4} = 9
\]
\textbf{(a)} Sketch a bar cut into 4 equal parts and label one of them.

\smallskip
\textbf{(b)} Solve for $x$, the number of ounces in the whole bag.}

\problem[6.5cm]{Mia claims that $x = 20$ is the solution of
$\dfrac{3}{5}x = 12$.

\par\smallskip
Do not re-solve the equation. \emph{Test} her answer instead: substitute 20 for
$x$ and evaluate the left side.

\smallskip
\textbf{(a)} What value does the left side give?

\smallskip
\textbf{(b)} Is Mia right? Say in one sentence how the substitution proves it.}

\problem[6.5cm]{Solve the equation, where the coefficient is negative.

\par\smallskip
\[
-\frac{3}{5}x = 12
\]
\textbf{(a)} Before solving, decide whether $x$ is positive or negative, and say
why.

\smallskip
\textbf{(b)} Multiply both sides by the reciprocal --- sign included --- and
solve.}

\problem[6.5cm]{A phone plan charges $1.25$ dollars per gigabyte, and this
month's data charge was $7.5$ dollars.

\par\smallskip
\[
1.25x = 7.5
\]
\textbf{(a)} Will $x$ be larger or smaller than $7.5$? Say why in one sentence
before you compute.

\smallskip
\textbf{(b)} Solve for $x$, the number of gigabytes used.}

\problem[7cm]{Both sides of this equation are fractions.

\par\smallskip
\[
\frac{5}{6}x = \frac{2}{3}
\]
\textbf{(a)} Write the reciprocal of the coefficient.

\smallskip
\textbf{(b)} Multiply both sides by it and simplify. Leave your answer as a
fraction in lowest terms.}

\problem[7cm]{This equation mixes a fraction coefficient with a decimal
constant.

\par\smallskip
\[
\frac{3}{8}x = 1.5
\]
\textbf{(a)} Rewrite $1.5$ as a fraction.

\smallskip
\textbf{(b)} Solve by multiplying both sides by the reciprocal of the
coefficient. Give your answer as a whole number.}

\problem[8cm]{\textbf{Challenge --- find and fix the mistake.} Devon solved
$\dfrac{2}{3}x = 18$ like this:

\par\smallskip
\begin{center}
\fbox{\parbox{0.8\linewidth}{\centering
$\dfrac{2}{3}x = 18$, so $x = \dfrac{2}{3}\cdot 18 = 12$.}}
\end{center}
\par
\textbf{(a)} Substitute Devon's answer, $x = 12$, into $\dfrac{2}{3}x$. What
value do you get?

\smallskip
\textbf{(b)} Name the mistake in one sentence.

\smallskip
\textbf{(c)} Solve the equation correctly. Write both your part (a) value and
your corrected solution on the answer line.}
