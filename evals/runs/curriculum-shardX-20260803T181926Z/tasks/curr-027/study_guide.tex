\sstitleblock{Subtracting Within 10 on a Number Path}

\noindent\small A number path is a row of squares, one for each number. Put a
finger on a square and hop: left takes away, right adds on. Say the number of
the square you land on, never the one you start from.\normalsize
\minipath

\skillheading{1. Hop back to take away}
\begin{formulabox}
\textbf{How to do it:} start on the \emph{first} number. Hop one square to the
\textbf{left} for each one you take away. The square you land on is the answer.\\[3pt]
Say the start number \emph{before} you hop, then count each hop: for
$7 - 3$ say ``seven'' first, then ``one, two, three'' as you hop.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $7 - 3$
\step{Taking away means going down, so start on the bigger number and hop to
the left.}
\step{Start on $7$. Hop to $6$, hop to $5$, hop to $4$. That is three hops.}

$7 - 3 = \ans{4}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $9 - 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{2. Hop up when the numbers are close}
\begin{formulabox}
\textbf{How to do it:} start on the \emph{smaller} number and hop to the
\textbf{right} until you reach the bigger one. \textbf{Count the hops} --- the
number of hops is the answer.\\[3pt]
Use this when the two numbers are near each other, like $9 - 7$ or $10 - 8$.
Fewer hops means fewer chances to miscount.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $8 - 6$
\step{$6$ and $8$ are close together, so hopping up from $6$ is quicker than
hopping back six times.}
\step{Start on $6$. Hop to $7$, hop to $8$. Stop. That is two hops.}

$8 - 6 = \ans{2}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $10 - 7$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{3. Find the missing number}
\begin{formulabox}
\textbf{How to do it:} when the middle number is missing, you know where you
start and where you land. Put one finger on each square and \textbf{count the
hops between them}. That count is the missing number.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $6 - \underline{\hspace{0.8cm}} = 2$
\step{The start and the landing square are both given, so the missing part is
how far apart they are.}
\step{One finger on $6$, one on $2$. Hop from $6$ to $5$, $4$, $3$, $2$: four
hops.}

$6 - \ans{4} = 2$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $8 - \underline{\hspace{0.8cm}} = 5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} do not count the starting square as a hop. For
$6 - 2$, say ``six'' on the square and then hop twice --- landing on $4$, not
$5$.
\end{watchoutbox}
