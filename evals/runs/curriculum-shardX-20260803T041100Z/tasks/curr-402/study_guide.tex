\sstitleblock{Composing Piecewise and Nonlinear Functions}

\begin{formulabox}
\textbf{Definition.} $(f \circ g)(x) = f(g(x))$ --- the \textbf{inner} function
$g$ acts first, and its output becomes the input of $f$.\\[3pt]
\textbf{Two rules that cause most errors.}
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=1pt]
\item Order matters: $f \circ g$ and $g \circ f$ are different functions.
\item Domain: the input must be legal for $g$, \emph{and} $g(x)$ must be legal
for $f$. Simplifying can hide a restriction, so read the domain off the
composition before simplifying.
\end{itemize}
\textit{Examples on this card use} $a(x) = 3x-1$, \; $b(x) = x^2+2$, \; and
$w(x) = \begin{cases} -x, & x < 0 \\ x + 6, & x \ge 0 \end{cases}$
\end{formulabox}

\skillheading{1. Evaluating a composition at a number}

\begin{formulabox}
Work \textbf{inside out}. Compute the inner value, write it down, then substitute
that single number into the outer function. Never substitute into both at once.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $(a \circ b)(2)$.
\step{The composition names $b$ as the inner function, so the only legal first
move is to evaluate $b$ at $2$ and carry that output outward.}
\step{$b(2) = 2^2 + 2 = 6$, and then $a(6) = 3(6) - 1$.}
\[ (a \circ b)(2) = \ans{17} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $(b \circ a)(2)$ --- the same two functions, other order.
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{2. Simplifying a composition into one rule}

\begin{formulabox}
Substitute the whole inner \emph{rule} into the outer function's variable slot,
keeping it in brackets, then expand:
$a(b(x)) = 3(\,b(x)\,) - 1$. Expand squares as
$(mx+c)^2 = m^2x^2 + 2mcx + c^2$ --- never as $m^2x^2 + c^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $(a \circ b)(x)$ as one simplified rule.
\step{A rule, not a number, is wanted, so the inner rule goes into the slot
where $a$'s variable stands and nothing is evaluated yet.}
\step{$a(b(x)) = 3(x^2+2) - 1 = 3x^2 + 6 - 1$.}
\[ \ans{3x^2 + 5} \quad\text{(check at } x=2:\; 12+5=17 \text{)} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $(b \circ a)(x)$ as one simplified rule.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9x^2 - 6x + 3}$}
\end{tryitbox}

\skillheading{3. Piecewise outer function: pick the piece}

\begin{formulabox}
A piecewise function tests \textbf{its own input}. In $w(g(x))$ that input is
$g(x)$, \emph{not} $x$ --- so compute the inner value first, compare \emph{that}
with the break point, and only then choose a rule. For a piecewise rule in $x$,
translate the break: $w$ breaks at $0$, so $w(a(x))$ breaks where $a(x) = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $w(a(0))$.
\step{The piece is chosen by what reaches $w$, so evaluate the inner function
before looking at either condition.}
\step{$a(0) = 3(0) - 1 = -1$. Since $-1 < 0$, $w$ uses the $-x$ piece, giving
$-(-1)$.}
\[ w(a(0)) = \ans{1} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $w(a(2))$. (Check the sign of $a(2)$ before choosing.)
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

\skillheading{4. Solving a composition equation}

\begin{formulabox}
Simplify the composition to one rule, set it equal to the target, and solve.
Quadratic compositions usually have \textbf{two} solutions --- take both roots.
\\[2pt]
With a piecewise outer function, solve \emph{each} piece and then test every
candidate against the condition that selects that piece; a candidate outside its
own region is rejected.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $(a \circ b)(x) = 32$.
\step{One rule for the composition already exists from section 2, so the
equation can be solved directly instead of unwinding two functions.}
\step{$3x^2 + 5 = 32$ gives $3x^2 = 27$, so $x^2 = 9$, which has two real
roots.}
\[ \ans{x = -3 \text{ or } x = 3} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $(b \circ a)(x) = 3$, using the rule from section 2.
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 0 \text{ or } x = \tfrac{2}{3}}$}
\end{tryitbox}
