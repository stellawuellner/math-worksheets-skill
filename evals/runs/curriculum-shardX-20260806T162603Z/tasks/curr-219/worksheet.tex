\wstitleblock{Slope and Rate of Change: Find the Error}{Grade 8 / Pre-Algebra}{}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Slope is $m = \dfrac{\text{change in } y}{\text{change in } x}$. Take
        both differences \emph{in the same order}, starting from the same point.
  \item In each ``find the error'' problem, work the slope out yourself first,
        then compare it with what the student wrote.
  \item When you name an error, say which quantity ended up in the wrong place
        --- ``upside down'' and ``sign flipped'' are different mistakes.
\end{itemize}

\vspace{0.1cm}

\problem[4.5cm]{Find the slope of the line through $A(2, 1)$ and $B(6, 9)$.}

\problem{Jonah found the slope of the line through $P(-3, 4)$ and $Q(1, -4)$.
He wrote $m = \dfrac{1 - (-3)}{-4 - 4} = -\dfrac{1}{2}$.
\begin{itemize}
  \item[(a)] Find the correct slope.
        \par\vspace*{2.6cm}\ansline
  \item[(b)] Name the error Jonah made.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A tank is draining. The table shows the gallons of water left after
$t$ hours. Ravi says the rate of change is $-\dfrac{6}{18}$ gallons per hour.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Hours} & 0 & 2 & 4 & 6 \\ \hline
\textbf{Gallons left} & 40 & 34 & 28 & 22 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] How many gallons does the tank lose between hour 0 and hour 6?
        \par\vspace*{1.9cm}\ansline
  \item[(b)] Find the rate of change in gallons per hour.
        \par\vspace*{1.9cm}\ansline
  \item[(c)] Name the error in Ravi's calculation.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}
