\sstitleblock{Rational and Irrational Numbers}

\begin{formulabox}
\textbf{One definition, one test.} A number is \textbf{rational} when it can be
written as $\dfrac{p}{q}$ with $p,q$ integers and $q \neq 0$; otherwise it is
\textbf{irrational}.\\[3pt]
\begin{tabular}{@{}p{5.3cm}p{9.3cm}@{}}
\textbf{Always rational} & integers, fractions, terminating decimals,
repeating decimals, roots of perfect squares \\[2pt]
\textbf{Irrational} & $\pi$, and $\sqrt{n}$ whenever $n$ is a whole number that
is not a perfect square \\
\end{tabular}\\[3pt]
The decimal is the giveaway: \emph{ends} or \emph{repeats} $\rightarrow$
rational; \emph{never ends and never repeats} $\rightarrow$ irrational.
\end{formulabox}

\skillheading{1. Roots of perfect squares}

\begin{formulabox}
$1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121, 144, 169, 196, 225$ are the perfect
squares up to $15^2$.\\[2pt]
If the radicand is in that list, the root is a whole number, so
$\sqrt{n} = \frac{\text{whole number}}{1}$ is \textbf{rational}. The same works
for fractions: $\sqrt{p/q} = \sqrt{p}/\sqrt{q}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Classify $\sqrt{81}$.
\step{The question is whether the root is exact, so check the radicand against
the perfect-square list rather than reaching for a calculator.}
\step{$81$ is in the list, so $\sqrt{81} = 9 = \tfrac{9}{1}$ --- a ratio of
integers.}
$\Rightarrow\quad \ans{9}$, and $\sqrt{81}$ is \textbf{rational}.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Classify $\sqrt{169}$ and give its exact value.\\[2pt]
\rotatebox{180}{\footnotesize check: rational, $\ans{13}$}
\end{tryitbox}

\skillheading{2. Roots that are not exact}

\begin{formulabox}
If the radicand sits \emph{between} two perfect squares, the root sits between
their roots and the decimal never ends or repeats --- it is
\textbf{irrational}. Bracket it first, then round:
\[ p^2 < n < q^2 \quad\Longrightarrow\quad p < \sqrt{n} < q \]
A rounded decimal is a \emph{stand-in}, never an equality.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Locate and classify $\sqrt{28}$.
\step{Since $28$ is not a perfect square, the useful move is to trap it between
the two nearest ones instead of hunting for an exact value.}
\step{$25 < 28 < 36$, so $5 < \sqrt{28} < 6$; a calculator gives
$5.2915\ldots$, which rounds to $\ans{5.29}$.}
$\Rightarrow\quad \sqrt{28}$ is \textbf{irrational}, and it sits well below the
midpoint of $5$ and $6$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Between which two whole numbers is $\sqrt{55}$, and what is it
to two decimal places?\\[2pt]
\rotatebox{180}{\footnotesize check: between 7 and 8, $\ans{7.42}$}
\end{tryitbox}

\skillheading{3. Decimals and fractions}

\begin{formulabox}
\textbf{Terminating:} read the decimal aloud and write what you hear, then
reduce. $0.7 = \frac{7}{10}$, \; $0.29 = \frac{29}{100}$, \;
$0.413 = \frac{413}{1000}$.\\[2pt]
\textbf{Repeating:} put the repeating block over as many $9$s as it has digits.
$0.\overline{4} = \frac{4}{9}$, \; $0.\overline{45} = \frac{45}{99}$. Either
way you end with a fraction, so both kinds are \textbf{rational}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Show that $0.125$ is rational.
\step{It terminates, so the definition can be met directly --- name the place
value and write the fraction it says.}
\step{``One hundred twenty-five thousandths'' is $\dfrac{125}{1000}$, and
dividing top and bottom by $125$ gives the lowest-terms form.}
$\Rightarrow\quad 0.125 = \ans{\tfrac{1}{8}}$, so it is \textbf{rational}.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $0.\overline{27}$ as a fraction in lowest terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{11}}$}
\end{tryitbox}

\skillheading{4. Comparing and ordering}

\begin{formulabox}
Put every number in the \emph{same} form --- decimals are easiest. Fractions:
divide. Roots: bracket with perfect squares, then refine by squaring your
guesses. Size, not notation, decides the order.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Order $2.5$, $\sqrt{7}$, $\tfrac{11}{4}$ from least to
greatest.
\step{The three are written in three different notations, so convert to a
common decimal form before comparing anything.}
\step{$\tfrac{11}{4} = 2.75$; and $4 < 7 < 9$ gives $2 < \sqrt{7} < 3$, with
$2.75^2 = 7.5625 > 7$, so $\sqrt{7} < 2.75$.}
$\Rightarrow\quad \ans{2.5 < \sqrt{7} < \tfrac{11}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Order $1.6$, $\sqrt{3}$, $\tfrac{7}{4}$ from least to
greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1.6 < \sqrt{3} < \tfrac{7}{4}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a radical sign does not make a number irrational
($\sqrt{9}$ is just $3$), and a calculator display is not proof of anything ---
it truncates every irrational number to a terminating, and therefore rational,
decimal.
\end{watchoutbox}
