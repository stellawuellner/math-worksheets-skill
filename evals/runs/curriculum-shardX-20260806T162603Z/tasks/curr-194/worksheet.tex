\wstitleblock{Around or Across? Radius, Diameter, Area, Perimeter}{Grades 6--7 Mathematics}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Before computing, decide two things: is the question asking for the
        distance \textbf{around} (perimeter, circumference) or the surface
        \textbf{covered} (area)? And if a circle is involved, were you given
        the \textbf{radius} or the \textbf{diameter}?
  \item Round answers with $\pi$ to two decimal places and include the unit on
        the answer line.
  \item Two problems show another student's work with a mistake. Find it, then
        fix it.
\end{itemize}

\vspace{0.3cm}

\problem{A circular clock face has a radius of $6$ cm. Find its circumference.
\par\vspace*{4cm}\answerline{cm}}

\problem{A round pizza has a \emph{diameter} of $10$ inches. Find its area.
\par\vspace*{4.5cm}\answerline{in$^2$}}

\problem{A rectangular garden measures $9$ m by $4$ m. How many meters of fence
are needed to enclose it?
\par\vspace*{4.5cm}\answerline{m}}

\problem{A rectangular rug is $7$ ft long and $5$ ft wide. How much floor does
it cover?
\par\vspace*{4.5cm}\answerline{ft$^2$}}

\problem{A circular pond has a \emph{diameter} of $14$ m. Leo walks once around
its edge. How far does he walk?
\par\vspace*{4.5cm}\answerline{m}}

\problem{Mara finds the area of a circle whose \emph{diameter} is $16$ cm. Her
work:
\begin{center}
$A = \pi r^2 = \pi \cdot 16^2 = 804.25$ square centimeters
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
  \item[(a)] Name the error in Mara's work. \par\vspace*{1.4cm}
  \item[(b)] Find the correct area.
\end{itemize}
\par\vspace*{2.8cm}\answerline{cm$^2$}}

\problem{Jon wants enough paint to cover a rectangular wall stripe $12$ ft long
and $3$ ft tall. His work:
\begin{center}
$2 \cdot (12 + 3) = 30$, so he buys paint for $30$ square feet
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
  \item[(a)] What went wrong? Explain in one sentence. \par\vspace*{1.4cm}
  \item[(b)] Find the amount of wall the paint must actually cover.
\end{itemize}
\par\vspace*{2.8cm}\answerline{ft$^2$}}

\problem{\textbf{Challenge.} A circular table top has a radius of $0.6$ m. A
metal trim runs around its edge, and a glass sheet covers its top.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
  \item[(a)] How long is the trim?
  \item[(b)] How large is the glass sheet?
\end{itemize}
\par\vspace*{3.2cm}\answerline{m}
\par\answerline{m$^2$}}
