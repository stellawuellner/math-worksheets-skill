\wstitleblock{Working with Signed Rational Numbers}{Grades 6--7 \quad The Number System}{}

\begin{center}
\textit{Adding and subtracting signed numbers, multiplying and dividing them,
ordering rationals written in different forms, and evaluating mixed expressions
in the right order}
\end{center}

\noindent\textbf{Before you start.} These problems are deliberately mixed, so
read each one before you reach for a rule.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Adding or subtracting?} Rewrite a subtraction as adding the
opposite, then think about which number is further from zero.
\item \textbf{Multiplying or dividing?} Count the negative signs: an even
number of them gives a positive result, an odd number gives a negative one.
\item \textbf{Several operations?} Brackets, then exponents, then
multiplication and division \emph{left to right}, then addition and
subtraction left to right.
\item \textbf{Comparing?} Put both numbers in the same form (decimals are
usually easiest), and remember that further left on the number line means
\emph{smaller}.
\end{itemize}

\problem[3.5cm]{Evaluate: \quad $(-4)(7)$}

\problem[3.5cm]{Order these numbers from \textbf{least to greatest}:
\begin{center}\large
$-1$, \qquad $-\frac{3}{4}$, \qquad $-0.6$, \qquad $\frac{1}{2}$
\end{center}}

\problem[4cm]{Evaluate: \quad $-3 + 4(-2)$ \\[2pt]
Which operation has to happen first, and why?}

\problem[4.5cm]{Evaluate and write the answer in lowest terms:
\begin{center}\large
$\left(-\dfrac{3}{4}\right)\left(\dfrac{8}{9}\right)$
\end{center}}

\problem[4cm]{Evaluate: \quad $2.5 + (-7.25)$}

\problem[4cm]{Write $<$, $>$ or $=$ between these two numbers, and say how you
decided:
\begin{center}\large
$-\dfrac{5}{6}$ \underline{\hspace{1.2cm}} $-0.8$
\end{center}}

\problem[4.5cm]{Evaluate: \quad $(-2)^3 + 12 \div (-4)$ \\[2pt]
Show the value of each piece before you combine them.}

\problem[5cm]{Priya evaluated $-24 \div (-6) \cdot (-2)$ and wrote $-2$. Find
her mistake, then evaluate the expression correctly.}

\problem[5cm]{Evaluate: \quad $-3(2 - 5) - \dfrac{4^2}{8}$ \\[2pt]
Write out each stage of your work.}

\problem[5.5cm]{\textbf{Challenge.} Order these four numbers from

mix of decimals and fractions.
\begin{center}\large
$-2.5$, \qquad $-\dfrac{11}{4}$, \qquad $-2.05$, \qquad $-\dfrac{9}{4}$
\end{center}}
