\sstitleblock{The Zero That Holds a Place}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Expanded form when a place is empty}

\begin{formulabox}
\textbf{Rule:} Write a plus part only for places that hold something.
An empty place (a zero digit) gets \emph{no} part in expanded form --- but
its zero must still appear in the standard form.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write 308 in expanded form.
\step{Read each place: three hundreds, an empty tens place, eight ones.}
\step{Write parts only for the places that hold something: 300 and 8.}
$\Rightarrow\quad 308 = \ans{300 + 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write 902 in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{900 + 2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Expanded form of 308 is \emph{not} $30 + 8$. The
zero keeps the 3 in the hundreds place, so its part is 300.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Standard form: keep the placeholder}

\begin{formulabox}
\textbf{Rule:} To pack $\text{hundreds} + \text{ones}$ back into one number,
give \emph{every} place a digit. A place with no part gets a \textbf{0}.
Three-digit numbers always show three digits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $700 + 6$ in standard form.
\step{Match parts to places: 700 fills the hundreds, 6 fills the ones, and nothing fills the tens.}
\step{Hold the empty tens place with a zero: seven, zero, six.}
$\Rightarrow\quad \ans{706}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $400 + 8$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{408}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Finding and fixing a placeholder error}

\begin{formulabox}
\textbf{Checklist for spotting the error:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Add up the wrong work --- what number does it \emph{really} make?
  \item Compare it with the number it was supposed to make.
  \item Find the digit that changed places, then rewrite it correctly.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Jade wrote $520 = 500 + 2$. Fix her expanded form.
\step{Test her work first: 500 + 2 makes only 502, not 520, so something moved.}
\step{In 520 the 2 sits in the tens place, so its part is 20, and the zero shows no ones.}
$\Rightarrow\quad 520 = \ans{500 + 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rex wrote $103 = 100 + 30$. Write the correct
expanded form of 103.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{100 + 3}$}
\end{tryitbox}
