\sstitleblock{More, Fewer, or the Same?}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Matching Pairs to Compare}

\begin{formulabox}
\textbf{Rule:} Draw a line from each object in one group to \textbf{one}
object in the other group. If a group has objects \emph{left over} with no
partner, that group has \textbf{more}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Row A has 6 dots and Row B has 4 dots. Pair them up
--- which row has more?
\step{Pair the rows dot for dot and look for leftovers --- the leftovers are the proof.}
\step{Four pairs form, and Row A still has two dots with no partner, so A has more: $\ans{6 > 4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Row A has 5 dots and Row B has 7 dots. After pairing,
one row has leftovers. Write the comparison with $<$, $>$ or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5 < 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- More or Fewer by Counting}

\begin{formulabox}
\textbf{Rule:} Count each group. The \textbf{bigger count} means more; the
\textbf{smaller count} means fewer. The count said later when counting up is
the bigger one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One basket holds 7 apples, another holds 9. Which
holds more?
\step{Compare the two counts: which number comes later when we count up?}
\step{``\ldots 7, 8, 9'' --- 9 comes later, so the basket of nine holds more: $\ans{7 < 9}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One jar holds 8 marbles, another holds 5. Write the
comparison with $<$, $>$ or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8 > 5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Spotting the Same Number}

\begin{formulabox}
\textbf{Rule:} Two groups have \textbf{the same} number when their counts
match --- even if one group is spread out, stacked, or arranged differently.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tight cluster counts to 6 and a long spread-out row
also counts to 6. More, fewer, or the same?
\step{Ignore how much space each group takes --- only the counts decide.}
\step{Both counts are six, so the groups are the same: $\ans{6 = 6}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A square of 4 dots and a row of 4 dots. Write the
comparison with $<$, $>$ or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4 = 4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A spread-out group \emph{looks} bigger but may not
\emph{be} bigger. Looks can trick you --- counts cannot.
\end{watchoutbox}
