\skillheading{Skill 1 --- Knowing a dot pattern at a glance}

\begin{formulabox}\textbf{Rule:}\ Some dot pictures have a shape you learn by
heart: a pair, a slant of three, a square of four, a square with a middle dot.
Say the number first; count only to check.
\end{formulabox}

\begin{examplebox}\textbf{Example:} \quad How many dots?
\ \begin{tikzpicture}[scale=0.34, baseline=(current bounding box.center)]
  \draw[rounded corners=4pt, thick] (0,0) rectangle (3,3);
  \fill (0.8,2.2) circle (0.3);
  \fill (2.2,2.2) circle (0.3);
  \fill (0.8,0.8) circle (0.3);
  \fill (2.2,0.8) circle (0.3);
\end{tikzpicture}
\step{The dots make a square with nothing in the middle --- a shape I already
know, so I name it instead of counting.}
\step{The square shape is worth $\ans{4}$.}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:} \quad How many dots?
\ \begin{tikzpicture}[scale=0.34, baseline=(current bounding box.center)]
  \draw[rounded corners=4pt, thick] (0,0) rectangle (3,3);
  \fill (0.8,2.2) circle (0.3);
  \fill (1.5,1.5) circle (0.3);
  \fill (2.2,0.8) circle (0.3);
\end{tikzpicture}
\quad\rotatebox{180}{\footnotesize check: $\ans{3}$}\end{tryitbox}

\skillheading{Skill 2 --- Reading a five-frame by its empty boxes}

\begin{formulabox}\textbf{Rule:}\ A five-frame is full at five, so read the
\emph{empty} boxes: \quad dots $=$ five $-$ empty boxes.
\end{formulabox}

\begin{examplebox}\textbf{Example:} \quad How many dots?
\ \begin{tikzpicture}[scale=0.36, baseline=(current bounding box.center)]
  \draw[thick] (0,0) grid (5,1);
  \fill (0.5,0.5) circle (0.3);
  \fill (1.5,0.5) circle (0.3);
  \fill (2.5,0.5) circle (0.3);
\end{tikzpicture}
\step{Two boxes are empty and a full frame is five --- so I subtract instead of
counting dots.}
\step{$5 - 2 = \ans{3}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:} \quad How many dots?
\ \begin{tikzpicture}[scale=0.36, baseline=(current bounding box.center)]
  \draw[thick] (0,0) grid (5,1);
  \fill (0.5,0.5) circle (0.3);
  \fill (1.5,0.5) circle (0.3);
  \fill (2.5,0.5) circle (0.3);
  \fill (3.5,0.5) circle (0.3);
\end{tikzpicture}
\quad\rotatebox{180}{\footnotesize check: $\ans{4}$}\end{tryitbox}

\skillheading{Skill 3 --- Putting two small groups together}

\begin{formulabox}\textbf{Rule:}\ Name each little group at a glance, hold the
bigger one in your head, and count on the smaller one:
group $+$ group $=$ total.
\end{formulabox}

\begin{examplebox}\textbf{Example:} \quad How many in all?
\ \begin{tikzpicture}[scale=0.32, baseline=(current bounding box.center)]
  \draw[rounded corners=4pt, thick] (0,0) rectangle (2.4,2.4);
  \fill (0.7,1.8) circle (0.3);
  \fill (1.2,1.2) circle (0.3);
  \fill (1.7,0.6) circle (0.3);
  \draw[rounded corners=4pt, thick] (3.4,0) rectangle (5.8,2.4);
  \fill (4.6,1.2) circle (0.3);
\end{tikzpicture}
\step{Each box is small enough to name on sight --- a slant of three and a
single --- so I add the names.}
\step{$3 + 1 = \ans{4}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:} \quad How many in all?
\ \begin{tikzpicture}[scale=0.32, baseline=(current bounding box.center)]
  \draw[rounded corners=4pt, thick] (0,0) rectangle (2.4,2.4);
  \fill (0.7,1.7) circle (0.3);
  \fill (1.7,0.7) circle (0.3);
  \draw[rounded corners=4pt, thick] (3.4,0) rectangle (5.8,2.4);
  \fill (4.1,1.8) circle (0.3);
  \fill (4.6,1.2) circle (0.3);
  \fill (5.1,0.6) circle (0.3);
\end{tikzpicture}
\quad\rotatebox{180}{\footnotesize check: $\ans{5}$}\end{tryitbox}

\skillheading{Skill 4 --- Deciding which picture shows more}

\begin{formulabox}\textbf{Rule:}\ Name \emph{both} pictures, then compare the
numbers. The symbol opens toward the larger one: $>$ is \emph{greater than},
$<$ is \emph{less than}. A wider card does not always hold more dots.
\end{formulabox}

\begin{examplebox}\textbf{Example:} \quad A slanted card of three, or a full
five-frame --- which is more?
\step{Neither picture needs counting: a slant is three and a full frame is five,
so I am just comparing two numbers I already know.}
\step{The smaller number is on the left, so the symbol opens toward the frame:
$3 \ans{<} 5$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:} \quad Left card: a single dot and a slant of
three. Right card: a slant of three. Which is more?\ %
\rotatebox{180}{\footnotesize check: $\ans{>}$}\end{tryitbox}
