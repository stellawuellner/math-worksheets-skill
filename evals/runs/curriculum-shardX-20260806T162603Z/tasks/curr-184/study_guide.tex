\sstitleblock{One-Variable Inequalities: The Sign-Flip Rule}

% =================== SKILL 1 ===================
\skillheading{Solving One-Step Inequalities}

\begin{formulabox}
\textbf{Rule:} solve an inequality with the same moves as an equation,
with \textbf{one exception} --- multiplying or dividing both sides by a
\textbf{negative} number \textbf{reverses} the inequality symbol
($<\;\leftrightarrow\;>$ and $\le\;\leftrightarrow\;\ge$).
Adding or subtracting never reverses it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $-6x < 24$
\step{The variable is multiplied by $-6$, so divide both sides by $-6$ --- and
because $-6$ is negative, reverse the symbol.}
\step{$x > \dfrac{24}{-6}$, so $x > -4$.}

$\Rightarrow\quad \ans{x > -4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-8x \ge 32$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \le -4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Solving Two-Step Inequalities}

\begin{formulabox}
\textbf{Order of moves:} undo the adding or subtracting first, then undo the
multiplying or dividing. Only the \emph{last} move can trigger the sign-flip
rule, and only when the number you divide by is negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $-4x + 3 > 15$
\step{Undo the $+3$ first (subtracting never flips the symbol), then divide by
the negative coefficient and reverse.}
\step{$-4x > 15 - 3$, so $-4x > 12$; dividing by $-4$ gives $x < -3$.}

$\Rightarrow\quad \ans{x < -3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-5x - 2 \le 13$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \ge -3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Catching a Missing Sign Flip}

\begin{formulabox}
\textbf{Checklist for grading your own work:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Did any step multiply or divide by a negative? Then the symbol must
        have reversed in that step.
  \item Test one easy number from your answer in the \emph{original}
        inequality --- if it makes the inequality false, the symbol is
        pointing the wrong way.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Jo says $-2x < 10$ means $x < -5$. Check Jo's claim.
\step{Jo divided by $-2$, a negative, so the symbol should have reversed ---
test a number from Jo's answer to confirm the claim fails.}
\step{Try $x = -6$ from Jo's set: $-2(-6) = 12$, and $12 < 10$ is false.
Solving correctly: divide $-2x < 10$ by $-2$ and reverse.}

$\Rightarrow\quad \ans{x > -5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Ali says $-3x \ge 21$ means $x \ge -7$. Fix it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \le -7}$}
\end{tryitbox}
