\sstitleblock{Signs and Conjugates with Complex Numbers}

\skillheading{Skill 1 --- Signs when you add, subtract, multiply}
\begin{formulabox}\textbf{Two rules that fix most sign errors:}\\[4pt]
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A minus in front of a complex number hits \emph{both} parts:
        $-(c - di) = -c + di$.
  \item $i^2 = -1$, so the last FOIL term always flips sign:
        $bi \cdot (-di) = -bd\,i^2 = +bd$.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Multiply $(3 + 4i)(1 - 2i)$.
\step{Both factors are binomials, so FOIL, then convert the $i^2$ term with $i^2 = -1$ before combining.}
\step{$3 - 6i + 4i - 8i^2 = 3 - 2i + 8$, so the product is $\ans{11 - 2i}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Multiply $(2 + 5i)(4 - 3i)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{23 + 14i}$}\end{tryitbox}

\skillheading{Skill 2 --- Dividing with the conjugate}
\begin{formulabox}\textbf{Rule:} the conjugate of $c + di$ is $c - di$ --- flip only the
sign in front of $i$. Multiplying a conjugate pair gives
$(c + di)(c - di) = c^2 + d^2$, a \textbf{sum} of squares (never $c^2 - d^2$), because
$-d^2i^2 = +d^2$. To divide, multiply the top \emph{and} the bottom by the conjugate
of the bottom.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Simplify $\dfrac{7 + i}{1 - 2i}$.
\step{The bottom is not real, so multiply top and bottom by its conjugate $1 + 2i$ to clear the $i$ downstairs.}
\step{Top: $(7 + i)(1 + 2i) = 7 + 14i + i - 2 = 5 + 15i$. Bottom: $1^2 + 2^2 = 5$. Dividing each part by 5 gives $\ans{1 + 3i}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Simplify $\dfrac{4 + 7i}{2 + i}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3 + 2i}$}\end{tryitbox}

\skillheading{Skill 3 --- Non-real solutions come in pairs}
\begin{formulabox}\textbf{Rule:} if a quadratic (or any polynomial) with real
coefficients has the non-real solution $a + bi$, then $a - bi$ is a solution too. A
negative discriminant is the signal: $\sqrt{-k} = i\sqrt{k}$, and the $\pm$ in the
quadratic formula produces the conjugate pair.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Solve $x^2 - 4x + 13 = 0$.
\step{The left side has no real factor pair, so use the quadratic formula and watch the discriminant's sign.}
\step{$16 - 52 = -36$, so $\sqrt{-36} = 6i$ and $x = \dfrac{4 \pm 6i}{2}$, giving $\ans{2 + 3i}$ and $\ans{2 - 3i}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $x^2 + 2x + 10 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1 + 3i}$ and $\ans{-1 - 3i}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} A conjugate changes ONLY the sign of the
imaginary part: the conjugate of $7 - 3i$ is $7 + 3i$, not $-7 - 3i$ and not $7 - 3i$
again. And $(c + di)(c - di) = c^2 + d^2$ --- if you ever write $c^2 - d^2$, you have
treated $i^2$ as $+1$.
\end{watchoutbox}
