\wstitleblock{Count-On Detectives}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item To count on, say the start number, then count your hops:
        for $5+2$, say ``5\,'' --- then hop ``6, 7\,''.
  \item Some kids on this page counted the start number as a hop.
        That lands one short!
  \item Be a detective: find each mistake, fix the sum, and tell what went
        wrong.
\end{itemize}

\problem[3cm]{Warm-up. Count on the right way: for $6+2$, say ``6\,'', then
hop two times. What do you land on?
\par\vspace{4pt}
$6\;+\;2\;=$~\ansblank}

\problem[3.5cm]{Jen added $5+3$. She counted: ``5, 6, 7\,'' --- and wrote
$5+3=7$. Write the correct sum, and tell why Jen's count went wrong.
\par\vspace{4pt}
$5\;+\;3\;=$~\ansblank}

\problem[3.5cm]{Kim added $6+3$. She counted: ``6, 7, 8\,'' --- and wrote
$6+3=8$. Find Kim's mistake. Write the correct sum, and show the count she
should have said.
\par\vspace{4pt}
$6\;+\;3\;=$~\ansblank}

\problem[3.5cm]{Raj added $8+2$. He counted: ``8, 9\,'' --- and wrote
$8+2=9$. Cross out the wrong sum, write the correct one, and tell how Raj
should count.
\par\vspace{4pt}
$8\;+\;2\;=$~\ansblank}

\problem[3.5cm]{Ava says $4+5=8$. Check her sum by counting on from 5.
Is Ava right? Write yes or no, then write the correct sum.
\par\vspace{4pt}
$4\;+\;5\;=$~\ansblank}

\problem[4cm]{Challenge! Leo says: ``I never make the count-on mistake,
because I say the start number first, and then I count my hops on my
fingers.'' Add $3+6$ Leo's way, starting from the bigger number. Then tell
how saying the start number first stops the mistake.
\par\vspace{4pt}
$3\;+\;6\;=$~\ansblank}
