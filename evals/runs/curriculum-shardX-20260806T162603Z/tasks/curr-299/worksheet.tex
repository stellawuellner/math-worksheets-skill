\wstitleblock{Boundary Conditions and Open or Closed Endpoints}{Algebra 1}{}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Exactly one rule may own a boundary value. The rule whose condition carries $\le$ or $\ge$ is the one that owns it; the other uses $<$ or $>$ and stops just short.
  \item If two conditions both claimed the same input, the definition would hand back two different outputs for one input --- and that is not a function at all.
  \item After solving an equation from one branch, \textbf{test each answer against that branch's condition}. A value can satisfy the formula and still not be a solution.
  \item Where a student's work is shown, it contains exactly one error --- and the arithmetic is usually fine.
\end{itemize}

\[ g(x) = \begin{cases} x^2 - 2, & x \le 4 \\ 3x + 1, & x > 4 \end{cases} \]

\par\medskip

\problem[6cm]{A piecewise function is defined by
\[ h(x) = \begin{cases} 3x + k, & x < 2 \\ x^2 + 1, & x \ge 2 \end{cases} \]
Find the value of $k$ that makes the two rules agree at the boundary, so the graph has no jump at $x = 2$.}

\problem[8cm]{A car park posts the price list
\[ C(t) = \begin{cases} 2t + 2, & 0 \le t \le 2 \\ 4t, & t > 2 \end{cases} \]
where $t$ is the stay in hours and $C$ is in dollars. Sofia copies the second condition as $t \ge 2$ instead of $t > 2$, and then cannot decide whether a two-hour stay costs $\$6$ or $\$8$.
\par\medskip
(a) Name the error in Sofia's version of the conditions, and explain why a price list written her way could not be a function.
\par\smallskip
(b) Using the list exactly as printed above, find the cost of a two-hour stay.
\par\smallskip
(c) Find the cost of a three-hour stay.}

\problem[6cm]{A piecewise function is defined by
\[ p(x) = \begin{cases} cx, & x < 5 \\ 2x + 5, & x \ge 5 \end{cases} \]
Find the value of $c$ for which the graph of $p$ has no jump at $x = 5$.}

\problem[8cm]{\textbf{Synthesis challenge.} A piecewise function is defined by
\[ q(x) = \begin{cases} x + a, & x \le -1 \\ x^2, & x > -1 \end{cases} \]
\par\smallskip
(a) Find the value of $a$ for which the two rules agree at $x = -1$.
\par\smallskip
(b) Using that value of $a$, find $q(-1)$ and $q(2)$.
\par\smallskip
(c) Explain which rule supplies $q(-1)$ and why, then say what would go wrong if both conditions were written with $\le$.}
