% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Comparing centre and spread}

\begin{formulabox}
\textbf{Centre --- where the data sit.} Mean $=$ total $\div$ how many. Median $=$ the
middle value once the data are in order.
\par\smallskip
\textbf{Spread --- how stretched the data are.} Range $=$ largest $-$ smallest.
\par\smallskip
\textbf{To compare two sets, quote one of each.} Equal centres do not make equal
distributions: the smaller range is the steadier set. One unusually large value drags the
mean but leaves the median alone, so report the median when a set has an outlier.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the mean and the range of $2$, $4$, $6$, $8$, $10$.
\step{A comparison needs one centre number and one spread number, so I compute the mean for centre and the range for spread.}
\step{The five values total thirty, and five shares of that total give the mean; the largest minus the smallest gives the range.}

Mean $= \ans{6}$, \quad range $= \ans{8}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the mean and the range of $3$, $4$, $5$, $6$, $7$.\\[2pt]
\rotatebox{180}{\footnotesize check: mean $\ans{5}$, range $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Reading a data display}

\begin{formulabox}
\textbf{Method:} read the scale before reading any bar. Then take the question one word at
a time: \emph{how many} for a single category, \emph{altogether} for the total,
\emph{how many more} for a difference, \emph{what fraction} for a category over the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tally of counters gives Red $7$, Blue $4$, Green $9$. Find the
total, and the number of blue counters.
\step{Two different questions on one display: a total needs every category, a single count needs only its own row.}
\step{Adding the three categories gives the total, while the blue row is read straight off the display.}

Total $= \ans{20}$, \quad blue $= \ans{4}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pet survey gives Cats $6$, Dogs $11$, Fish $3$. Find the total,
and the number of dogs.\\[2pt]
\rotatebox{180}{\footnotesize check: total $\ans{20}$, dogs $\ans{11}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Compound-event probability}

\begin{formulabox}
\textbf{Rule:} $P(\text{event}) = \dfrac{\text{favourable outcomes}}{\text{total
outcomes}}$, then reduce.
\par\smallskip
\textbf{For two things happening together,} the total is the first count times the second
count --- a table of all outcomes shows why. Count the favourable cells in that table.
When something is replaced, the second stage has the same counts as the first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A coin is flipped and a number cube is rolled. Find the
probability of tails and an even number.
\step{Two stages happening together, so the total outcomes multiply --- two coin results times six cube results.}
\step{Favourable cells are tails paired with each of the three even faces, and that count over the total reduces.}

$\Rightarrow\quad P = \dfrac{3}{12} = \ans{\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two number cubes are rolled. Find the probability that the total
is five. There are $4$ favourable pairs out of $36$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{9}}$}
\end{tryitbox}
