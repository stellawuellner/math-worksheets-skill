\wstitleblock{Expanded Form, Comparing, and Rounding}{Grade 4--5 \quad Multi-Digit Place Value and Rounding}{}

\begin{center}
\textit{Writing a number in expanded form, saying what each digit is worth,
comparing and ordering multi-digit numbers, and rounding to a named place}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A digit's value depends on where it sits. The same digit is worth ten
times as much in the place to its left.
\item To compare, start at the \emph{largest} place, not the largest digit.
Move right only when the digits match.
\item To round, look at the digit \emph{one place to the right} of the place
you are rounding to. Five or more rounds up; four or less stays.
\item Say the place out loud before you answer --- ones, tens, hundreds,
thousands, ten thousands. Naming the place is half of every problem here.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Write $6382$ in expanded form.}

\problem[5cm]{In the number $47308$, what is the value of the digit $7$?}

\problem[5cm]{Round $6382$ to the nearest hundred. Circle the digit you
looked at to decide.}

\problem[5cm]{Write $<$ or $>$ between $47308$ and $47380$. Name the
place that decided it.}

\problem[6cm]{A number is written in expanded form as
\[ 40000 + 3000 + 60 + 5. \]
Write the number in standard form. Be careful: one place has nothing in the
expanded form, so it needs a placeholder.}

\problem[5cm]{Round $47308$ to the nearest thousand. Name the digit you
looked at.}

\problem[6cm]{Kai wrote $8090 > 8900$. He said, ``the first one has a
$9$ in it and so does the second, but the first one also has an $8$ at the
front and a $9$, so it must be bigger.''

Write the correct comparison, and say in one sentence which place Kai should
have compared first.}

\problem[6cm]{Look at the number $55505$.

The $5$ in the ten-thousands place is worth $50000$. The $5$ in the
hundreds place is worth $500$. How many times greater is the first $5$ than
the second $5$?}

\problem[5cm]{Round $25649$ to the nearest thousand.}

\problem[7cm]{Put these three numbers in order from least to greatest:
\[ 72405 \qquad 72045 \qquad 70425 \]
All three use the same five digits, so compare place by place from the left.}

\problem[7cm]{Round $8950$ to the nearest hundred.

Something unusual happens here. After you round, explain in one sentence why
the thousands digit changed too.}

\problem[10cm]{\textbf{Challenge.} A stadium counted $38472$ people on
Friday and $38247$ people on Saturday.

\smallskip
(a) Write Friday's attendance in expanded form. \ansline

\smallskip
(b) Which day had more people? Write the comparison using $<$ or
$>$. \ansline

\smallskip
(c) Round each attendance to the nearest thousand. \ansline

\smallskip
(d) The two rounded numbers came out the same. Write one or two sentences
explaining why the rounded numbers cannot settle which day had more people,
and say which form you would use instead.}
