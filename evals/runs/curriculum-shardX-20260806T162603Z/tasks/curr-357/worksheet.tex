\wstitleblock{Polynomial Long Division}{Algebra 2}{}

\begin{center}
{\small\textit{Dividing polynomials by long division}}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item Set out every division in the long-division frame. Divide the leading
    terms, multiply back, subtract, bring down --- and subtract the whole line,
    not just its first term.
  \item Before you start, write the dividend with \emph{every} descending power
    present. A missing power gets a $0$ placeholder.
  \item Report the answer as \ quotient $+$ $\dfrac{\text{remainder}}{\text{divisor}}$,
    or as the quotient alone when the remainder is $0$.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{Divide: \quad $\dfrac{x^{2} + 7x + 12}{x + 3}$}

\problem[6cm]{Divide: \quad $\dfrac{x^{2} + 5x + 9}{x + 2}$}

\problem[7cm]{Divide: \quad $\dfrac{2x^{3} - 3x^{2} - 11x + 6}{x - 3}$}

\problem[7cm]{Divide: \quad $\dfrac{x^{3} - 8}{x - 2}$ \\[2pt]
{\small Write the dividend as $x^{3} + 0x^{2} + 0x - 8$ before you begin.}}

\problem[6cm]{Divide: \quad $\dfrac{3x^{2} - 5x + 4}{x - 1}$}

\problem[7cm]{Divide: \quad $\dfrac{x^{4} - 5x^{2} + 4}{x^{2} - 1}$}

\problem{Consider $\dfrac{x^{3} + 2x^{2} - 5x + 1}{x - 2}$.
\par\vspace{0.15cm}
(a) Divide, and give the quotient with its remainder.
\par\vspace*{6cm}\ansline
\par\vspace{0.15cm}
(b) Now substitute $x = 2$ into the dividend $x^{3} + 2x^{2} - 5x + 1$.
What value do you get?
\par\vspace*{2.6cm}\ansline}

\problem[7cm]{Divide: \quad $\dfrac{2x^{3} + x^{2} - 8x + 5}{x + 3}$}

\problem[7.5cm]{Divide: \quad $\dfrac{x^{3} + 4x^{2} - 3x + 2}{x^{2} + 1}$ \\[2pt]
{\small The divisor is a quadratic, so stop when what is left has degree below $2$.}}

\problem[7.5cm]{Divide: \quad $\dfrac{2x^{4} - 3x^{3} + x - 6}{x - 2}$}

\problem{Consider $\dfrac{x^{3} - 7x + 6}{x - 1}$.
\par\vspace{0.15cm}
(a) Divide. The remainder is $0$, so give the quotient.
\par\vspace*{5.5cm}\ansline
\par\vspace{0.15cm}
(b) Use part (a) to factor $x^{3} - 7x + 6$ completely.
\par\vspace*{2.8cm}\ansline}

\problem{The division $\dfrac{x^{3} + kx^{2} - 4x + 12}{x - 2}$ leaves no remainder.
\par\vspace{0.15cm}
(a) Find the value of $k$.
\par\vspace*{4.6cm}\ansline
\par\vspace{0.15cm}
(b) Explain how the remainder theorem lets you find $k$ without carrying out the
long division.
\par\vspace*{2.8cm}\ansline}
