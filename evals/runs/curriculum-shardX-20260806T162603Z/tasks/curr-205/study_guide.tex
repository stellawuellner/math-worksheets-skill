\sstitleblock{Ordering Rational and Irrational Numbers}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Putting every quantity into decimal form}

\begin{formulabox}
\textbf{Rule:} numbers can only be compared in the same form, and the decimal is the form
everything converts to.
\par\smallskip
\textbf{Fraction:} divide top by bottom --- the result terminates or repeats, exactly.
\textbf{Root:} squeeze between decimals whose squares trap the radicand, and round.
\textbf{Pi:} use three decimal places when a comparison is close.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $\dfrac{7}{8}$ and $\sqrt{13}$ as decimals.
\step{One is a fraction and one is a root, so they need different tools --- division for the fraction, a squeeze for the root.}
\step{Dividing seven by eight terminates after three places; for the root, the squares of three point six and three point seven straddle the radicand, so squeeze once more with hundredths.}

$\dfrac{7}{8} = \ans{0.875}$ \quad and \quad $\sqrt{13} \approx \ans{3.61}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $\dfrac{5}{16}$ and $\sqrt{23}$ as decimals.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.3125}$ and $\ans{4.80}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Comparing two quantities}

\begin{formulabox}
\textbf{Method:} convert both, then compare place by place from the left. Keep one more
decimal place than the point where they first differ.
\par\smallskip
\textbf{Negatives reverse:} further left on the number line is \emph{smaller}, so among
negatives the one with the bigger digits is the smaller number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$, $>$, or $=$ between $\dfrac{5}{7}$ and $0.72$.
\step{A fraction against a decimal cannot be read off as printed, so convert the fraction and line the two decimals up.}
\step{Dividing five by seven repeats, and it first differs from the given decimal in the hundredths place, where it is the smaller of the two.}

$\Rightarrow\quad \dfrac{5}{7} \ans{<} 0.72$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$ between $\sqrt{8}$ and $2.9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} do not judge size from how a number is printed. A short-looking
root can be larger than a long-looking decimal, and a bigger denominator does not mean a
bigger number. Convert first, every time.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Ordering a mixed list}

\begin{formulabox}
\textbf{Method:} convert every entry to the same number of decimal places, write the
decimals underneath one another, order those, then rewrite the answer using the
\emph{original} forms --- that is what the question asked for.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Order $\sqrt{7}$, $2.6$ and $\dfrac{8}{3}$, least to greatest.
\step{Three different forms, so nothing can be ordered until all three are decimals at the same precision.}
\step{The root squeezes to a little under two point sixty-five, the fraction repeats a little over two point sixty-six, and the plain decimal is the smallest of the three.}

$\Rightarrow\quad \ans{2.6} < \ans{\sqrt{7}} < \ans{\tfrac{8}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Order $\sqrt{3}$, $1.8$ and $\dfrac{7}{4}$, least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\sqrt{3}} < \ans{\tfrac{7}{4}} < \ans{1.8}$}
\end{tryitbox}
