\wstitleblock{Comparing Unlike Fractions}{Grade 4--5}{}
\begin{center}
\textit{Comparing with benchmarks, common denominators, and equivalent fractions}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Two tools are practiced here: comparing each fraction to the
        benchmark $\tfrac{1}{2}$, and rewriting both fractions with a common
        denominator.
  \item When a blank asks for a symbol, write $<$, $>$, or $=$.
\end{itemize}

\problem[3cm]{The strip below is cut into 8 equal parts, and half of it is
shaded. Use the strip to fill in the missing numerator:
$\tfrac{1}{2} = \tfrac{\square}{8}$
\begin{center}
\begin{tikzpicture}[scale=1]
  \fill[gray!45] (0,0) rectangle (4,0.7);
  \draw[thick] (0,0) rectangle (8,0.7);
  \foreach \i in {1,2,3,4,5,6,7} \draw (\i,0) -- (\i,0.7);
\end{tikzpicture}
\end{center}}

\problem[3cm]{Think about where $\tfrac{3}{8}$ sits on a number line from 0
to 1. Write the correct symbol:
$\tfrac{3}{8}$ \ansblank $\tfrac{1}{2}$}

\problem[4cm]{Compare $\tfrac{2}{3}$ and $\tfrac{3}{4}$ by rewriting both in
twelfths.
\par\vspace{2pt}
$\tfrac{2}{3} = \tfrac{\square}{12}$: \ansblank \qquad
$\tfrac{3}{4} = \tfrac{\square}{12}$: \ansblank \qquad
so \ $\tfrac{2}{3}$ \ansblank $\tfrac{3}{4}$}

\problem[3.5cm]{Lena's ribbon is $\tfrac{2}{5}$ m long. Her ruler is marked
in tenths of a meter. Write a fraction with denominator 10 that names the
same length.}

\problem[4cm]{Two juice jugs are the same size. Jug A is $\tfrac{5}{8}$ full
and Jug B is $\tfrac{3}{8}$ full. Compare each jug to the benchmark
$\tfrac{1}{2}$ to see which one is more than half full.
\par\vspace{2pt}
Jug A: \ $\tfrac{5}{8}$ \ansblank $\tfrac{1}{2}$ \qquad
Jug B: \ $\tfrac{3}{8}$ \ansblank $\tfrac{1}{2}$}

\problem[4.5cm]{Pedro ran $\tfrac{5}{6}$ mi and Amara ran $\tfrac{7}{9}$ mi.
Rewrite both distances in eighteenths of a mile to see who ran farther.
\par\vspace{2pt}
$\tfrac{5}{6} = \tfrac{\square}{18}$: \ansblank \qquad
$\tfrac{7}{9} = \tfrac{\square}{18}$: \ansblank \qquad
so \ $\tfrac{5}{6}$ \ansblank $\tfrac{7}{9}$}

\problem[4.5cm]{Compare $\tfrac{4}{9}$ and $\tfrac{6}{11}$ without finding a
common denominator: $\tfrac{4}{9}$ \ansblank $\tfrac{6}{11}$.
\par\vspace{2pt}
Then explain how comparing each fraction to $\tfrac{1}{2}$ tells you which
is greater.}

\problem[4cm]{Marco shades $\tfrac{2}{3}$ of one strip and $\tfrac{8}{12}$
of an identical strip.
\par\vspace{2pt}
(a) Fill in: $\tfrac{2}{3} = \tfrac{\square}{12}$: \ansblank
\par\vspace{2pt}
(b) Write the correct symbol: $\tfrac{2}{3}$ \ansblank $\tfrac{8}{12}$}

\problem[4.5cm]{A pizza pan of brownies: Jon has $\tfrac{7}{12}$ of a pan
left and Kayla has $\tfrac{5}{8}$ of a pan left. Rewrite both amounts in
twenty-fourths to compare them.
\par\vspace{2pt}
$\tfrac{7}{12} = \tfrac{\square}{24}$: \ansblank \qquad
$\tfrac{5}{8} = \tfrac{\square}{24}$: \ansblank \qquad
so \ $\tfrac{7}{12}$ \ansblank $\tfrac{5}{8}$}

\problem[5cm]{\textbf{Challenge.}
\par\vspace{2pt}
(a) Write the correct symbol: $\tfrac{3}{8}$ \ansblank $\tfrac{1}{2}$
\par\vspace{2pt}
(b) Find a fraction that is greater than $\tfrac{3}{8}$ but less than
$\tfrac{1}{2}$, and explain how you know it fits between them. \ansblank}
