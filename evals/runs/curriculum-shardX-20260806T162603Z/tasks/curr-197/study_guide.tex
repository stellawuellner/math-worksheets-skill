\sstitleblock{Computing and Interpreting Centre and Spread}

% =================== SKILL 1 ===================
\skillheading{Measures of Centre: Mean, Median, Mode}

\begin{formulabox}
\textbf{Mean} $=$ total $\div$ how many. \quad \textbf{Median} $=$ the middle
value once the list is in order (the mean of the middle two when there is an
even count). \quad \textbf{Mode} $=$ the value that appears most often.\par
Order the list first, every time. An unordered list makes the median a guess.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the mean and the median of $3,\ 9,\ 5,\ 10,\ 8$.
\step{The mean needs the total and the count; the median needs the list in
order, so do the ordering before anything else.}
\step{Total $= 35$, so the mean is $35 \div 5 = \ans{7}$; in order the list is
$3, 5, 8, 9, 10$, so the median is the third value, $\ans{8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the median of $4,\ 11,\ 6,\ 15,\ 9,\ 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Measures of Spread: Range and Interquartile Range}

\begin{formulabox}
\textbf{Range} $=$ largest $-$ smallest. It uses only the two ends, so a single
unusual value changes it completely.\par
\textbf{Interquartile range} $=$ upper quartile $-$ lower quartile. Order the
list, split it into a lower half and an upper half (leave the middle value out
if there is one), and take the median of each half. It measures the middle half,
so the ends cannot move it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the quartiles and the interquartile range of
$5,\ 7,\ 9,\ 12,\ 15,\ 18,\ 20,\ 24$.
\step{Eight values in order split evenly into a lower four and an upper four, so
take the median of each half.}
\step{Lower half $5, 7, 9, 12$ gives $\ans{8}$; upper half $15, 18, 20, 24$
gives $\ans{19}$; the interquartile range is $19 - 8 = \ans{11}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the interquartile range of
$1,\ 5,\ 9,\ 13,\ 17,\ 21,\ 25,\ 29$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Choosing the Pair that Tells the Truth}

\begin{formulabox}
\textbf{Rule:} report a centre \emph{and} a spread, and match them.\par
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Values fairly even, no extremes: \textbf{mean with range} is fine.
  \item One value far from the rest: \textbf{median with interquartile range},
        because neither is dragged by that one value.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Five prices, in dollars: $4,\ 5,\ 5,\ 6,\ 30$. Which
centre describes them?
\step{One value sits far above the others, so compute both the mean and the
median and see whether they agree.}
\step{The total is 50, so the mean is $50 \div 5 = \ans{10}$, while the median
is $\ans{5}$ --- and four of the five prices are under 7, so the median is the
honest one.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the median of $2,\ 3,\ 3,\ 4,\ 38$, and say whether
it or the mean describes these values better.\\[2pt]
\rotatebox{180}{\footnotesize check: median $\ans{3}$, and it is the better
description}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} you cannot average two means when the groups differ in
size. For 4 students at 80 and 6 at 90, recover the totals first: $320 + 540$
over 10 students gives 86, not 85.
\end{watchoutbox}
