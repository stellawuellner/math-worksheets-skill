\sstitleblock{Basic Matrix Operations}

\skillheading{Skill 1 --- Adding and subtracting matrices}

\begin{formulabox}
\textbf{Rule:} same shape, entry by entry. The entry in row $i$, column $j$ of $A + B$ is the entry of $A$ there plus the entry of $B$ there.\par
\textbf{Shapes must match exactly}, or the sum does not exist. Subtraction works the same way, one difference per position.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add the two tables of counts.
\step{Both matrices are $2 \times 2$, so the sum exists and each position is settled on its own --- no entry interacts with any other.}
\step{$\begin{bmatrix} 3 & 5 \\ 7 & 2 \end{bmatrix} + \begin{bmatrix} 4 & 1 \\ 8 & 6 \end{bmatrix}$: add $3 + 4$, $5 + 1$, $7 + 8$, $2 + 6$.}

\[ \begin{bmatrix} \ans{7} & \ans{6} \\ \ans{15} & \ans{8} \end{bmatrix} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\begin{bmatrix} 9 & 2 \\ 4 & 7 \end{bmatrix} + \begin{bmatrix} 3 & 6 \\ 5 & 1 \end{bmatrix}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\begin{bmatrix} \ans{12} & \ans{8} \\ \ans{9} & \ans{8} \end{bmatrix}$}
\end{tryitbox}

\skillheading{Skill 2 --- Multiplying a matrix by a number}

\begin{formulabox}
\textbf{Rule:} a scalar multiplies \emph{every} entry once. The shape never changes.\par
\textbf{In context:} raising every price by $20\%$ is the scalar $1.2$; combining, as in $3A - 2B$, means scale each matrix first, then add or subtract entry by entry.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $4 \begin{bmatrix} 2 & 5 \\ 3 & 1 \end{bmatrix}$.
\step{A scalar touches every position and nothing else changes, so the answer is the same shape with each entry multiplied once.}
\step{$4(2)$, $4(5)$, $4(3)$, $4(1)$ --- four separate products, no cross terms.}

\[ \begin{bmatrix} \ans{8} & \ans{20} \\ \ans{12} & \ans{4} \end{bmatrix} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $3 \begin{bmatrix} 4 & 1 \\ 2 & 6 \end{bmatrix}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\begin{bmatrix} \ans{12} & \ans{3} \\ \ans{6} & \ans{18} \end{bmatrix}$}
\end{tryitbox}

\skillheading{Skill 3 --- Multiplying two matrices}

\begin{formulabox}
\textbf{Shape rule:} an $m \times n$ times an $n \times p$ gives an $m \times p$. The two inner numbers must agree, or the product does not exist.\par
\textbf{Entry rule:} the entry in row $i$, column $j$ pairs row $i$ of the left matrix with column $j$ of the right one, multiplying matched positions and adding the results.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\begin{bmatrix} 2 & 3 \\ 1 & 4 \end{bmatrix}\begin{bmatrix} 5 & 1 \\ 2 & 6 \end{bmatrix}$.
\step{Inner dimensions agree ($2$ and $2$), so the product exists and is $2 \times 2$ --- four entries, each a row-times-column total.}
\step{Row $1$ with column $1$: $2(5) + 3(2)$. Row $1$ with column $2$: $2(1) + 3(6)$.}
\step{Row $2$ with column $1$: $1(5) + 4(2)$. Row $2$ with column $2$: $1(1) + 4(6)$.}

\[ \begin{bmatrix} \ans{16} & \ans{20} \\ \ans{13} & \ans{25} \end{bmatrix} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\begin{bmatrix} 3 & 2 \\ 5 & 1 \end{bmatrix}\begin{bmatrix} 4 \\ 7 \end{bmatrix}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\begin{bmatrix} \ans{26} \\ \ans{27} \end{bmatrix}$}
\end{tryitbox}

\skillheading{Skill 4 --- A system as a matrix equation}

\begin{formulabox}
\textbf{The translation:} coefficients go in the matrix, one row per equation and one column per unknown; the unknowns form a column; the right-hand sides form a column. That is $A\mathbf{x} = \mathbf{b}$.\par
\textbf{Then solve as usual} --- substitution or elimination. The matrix form is what makes the structure visible, not a new solving method.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $3x + 2y = 16$ and $x - y = 2$ as a matrix equation, and solve.
\step{Each equation becomes one row of coefficients, so the unknowns line up in columns and the constants form the right-hand column.}
\step{$\begin{bmatrix} 3 & 2 \\ 1 & -1 \end{bmatrix}\begin{bmatrix} x \\ y \end{bmatrix} = \begin{bmatrix} 16 \\ 2 \end{bmatrix}$}
\step{From the second row $x = y + 2$; substituting, $3(y + 2) + 2y = 16$, so $5y = 10$.}

\[ \ans{x = 4}, \qquad \ans{y = 2} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $2x + y = 11$ and $x + 3y = 18$ as a matrix equation, and solve.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$, $\ans{y = 5}$}
\end{tryitbox}
