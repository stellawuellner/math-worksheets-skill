\sstitleblock{Taking Away Objects and Writing Subtraction Equations}

\skillheading{Skill 1 --- Take Away Objects and Count What Is Left}

\begin{formulabox}
\textbf{The model:} start with a group, cross out the ones taken away, count what has \emph{no} X.\\[3pt]
\textbf{The words match the equation:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item how many you \textbf{start} with $\;\rightarrow\;$ the first number
  \item how many are \textbf{taken away} $\;\rightarrow\;$ the number after the minus sign
  \item how many are \textbf{left} $\;\rightarrow\;$ the number after the equals sign
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad There are 8 counters. Cross out 3 of them.
\step{Taking away means some counters leave the group, so this is a subtraction picture, not an adding one.}
\step{Cross out 3 counters, then touch and count only the ones with no X: 1, 2, 3, 4, 5. So $8 - 3 = \ans{5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad There are 6 counters. Cross out 4. How many are left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Write the Subtraction Equation for a Story}

\begin{formulabox}
\textbf{Rule:}\quad start $-$ taken away $=$ left.\\[3pt]
Words that mean \textbf{take away}: fly away, eat, give away, pop, leave, lose, break.
Write the number you \emph{started} with first --- it is always the biggest number in a take-away story.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 7 fish are in a tank. 4 fish are moved to another tank.
\step{``Moved away'' takes fish out of the tank, so the 7 is the whole group and the 4 is the part that leaves.}
\step{Write the whole first, then the part taken: $7 - 4$. Counting the fish left gives $7 - 4 = \ans{3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 9 balloons. 6 balloons pop. Write the equation and the answer.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Find the Missing Part}

\begin{formulabox}
\textbf{When the middle number is missing:}\quad start $-\;\underline{\hspace{0.7cm}}\; =$ left.\\[3pt]
Count \textbf{up} from the number left to the number you started with. Every count you say is one object
that was taken away. Then check by taking that many away.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 8 stickers. Some are used. 5 stickers are left. How many were used?
\step{The whole (8) and the leftover part (5) are known, so the missing number is the part that was taken away.}
\step{Count up from 5 to 8: ``6, 7, 8'' is 3 counts. Check: $8 - 3 = 5$ $\checkmark$, so the missing part is $\ans{3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 10 crackers. Some are eaten. 2 are left. How many were eaten?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} After you cross out, count the objects with \emph{no} X.
Counting the crossed-out ones answers a different question.
Also keep the bigger number first: a take-away story is never written $2 - 5$.
\end{watchoutbox}
