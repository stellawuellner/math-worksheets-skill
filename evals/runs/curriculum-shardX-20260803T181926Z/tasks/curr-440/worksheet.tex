\wstitleblock{Binomial Coefficients in Expansion and Probability}{Precalculus / Advanced Statistics \quad Discrete Mathematics}{}

\begin{center}
\textit{One set of numbers, two jobs: binomial coefficients as the terms of an
expansion and as the counts inside a probability}
\end{center}

\noindent\textbf{Directions.} The number $\binom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$
does two things on this sheet, and the point of mixing the problems is that you
have to notice which one is wanted.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=3pt]
\item \textbf{In an expansion:} \ $(a + b)^n = \sum \binom{n}{k} a^k b^{\,n-k}$.
The coefficient of a term is $\binom{n}{k}$ \emph{times} whatever powers of the
constants that term carries --- forgetting the second factor is the usual slip.
\item \textbf{In a count:} $\binom{n}{k}$ is the number of ways to choose $k$
objects from $n$ when order does not matter.
\item \textbf{In a probability:} the same count appears on top, on the bottom,
or both. With equally likely outcomes, use $\dfrac{\text{favourable}}{\text{total}}$;
for $n$ independent trials each succeeding with probability $p$, use
$\binom{n}{k}p^k(1-p)^{\,n-k}$.
\end{itemize}

\vspace{0.1cm}
\noindent Leave probabilities as exact fractions unless a problem says
otherwise. Show the binomial coefficient you used, not just the final number.

\problem[5cm]{Evaluate $\dbinom{7}{3}$ using the factorial formula. Show the
cancellation.}

\problem[5.5cm]{Expand and simplify: \quad $(x + 2)^3$}

\problem[6.5cm]{A club has $9$ members, $4$ of whom are seniors. A committee of
$3$ members is chosen at random.

\textbf{(a)} How many committees of $3$ can be formed altogether?

\textbf{(b)} How many of them consist entirely of seniors?

\textbf{(c)} What is the probability that a randomly chosen committee is all
seniors? Give an exact fraction.}

\problem[6cm]{\textbf{Pascal's rule.} Evaluate $\dbinom{9}{3}$, $\dbinom{9}{4}$
and $\dbinom{10}{4}$, then state the relationship between the three values and
explain where it appears in Pascal's triangle.}

\problem[7cm]{Expand and simplify: \quad $(2x - 1)^4$

\smallskip
\small Every term carries a power of $2$ as well as a binomial
coefficient.\normalsize}

\problem[6cm]{A fair coin is flipped $6$ times.

\textbf{(a)} In how many of the $64$ equally likely outcomes are exactly $4$ of
the flips heads?

\textbf{(b)} What is the probability of getting exactly $4$ heads? Give an exact
fraction.}

\problem[6.5cm]{Find the coefficient of $x^5$ in the expansion of
$(x + 3)^8$.

\smallskip
\small The binomial coefficient on its own is not the answer.\normalsize}

\problem[8cm]{A bag holds $5$ red marbles and $7$ blue marbles. Three marbles
are drawn at random, all at once.

\textbf{(a)} In how many ways can $2$ red marbles be chosen from the red ones,
and $1$ blue marble from the blue ones?

\textbf{(b)} How many three-marble draws are possible altogether?

\textbf{(c)} Find the probability that exactly $2$ of the three marbles are red.
Give an exact fraction in lowest terms.}

\problem[7.5cm]{A basketball player makes $70\%$ of her free throws, and the
throws are independent. She takes $5$ free throws.

Find the probability that she makes exactly $3$ of them. Give an exact fraction.}

\problem[8cm]{Find the constant term in the expansion of
$\left(2x + \dfrac{1}{x}\right)^{6}$.

\smallskip
\small Write the general term first and decide which value of $k$ makes the
power of $x$ equal to zero.\normalsize}

\problem[7.5cm]{Jo says: ``A fair coin is flipped $5$ times. The probability of
getting exactly $2$ heads is $\frac{2}{5}$, because $2$ of the $5$ flips have to
be heads.''

\textbf{(a)} How many of the $32$ equally likely outcomes have exactly $2$
heads?

\textbf{(b)} Find the correct probability as an exact fraction.

\textbf{(c)} Explain Jo's mistake in one sentence.}

\problem[9cm]{\textbf{Synthesis challenge.} An inspector tests $8$ components.
Each one is defective, independently of the others, with probability
$\frac{1}{4}$.

\textbf{(a)} In the expansion of $(q + p)^8$, where $p$ is the probability a
component is defective and $q = 1 - p$, one term gives the probability that
exactly $2$ components are defective. Write that term, including its binomial
coefficient.

\textbf{(b)} Evaluate that term as an exact fraction.

\textbf{(c)} Explain why the binomial coefficients appearing in this expansion
are exactly the counts of \emph{which} components are defective.}
