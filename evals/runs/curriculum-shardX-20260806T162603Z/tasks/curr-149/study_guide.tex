\sstitleblock{Finding Order and Scale Mistakes on Graphs}

\skillheading{Skill 1 --- The order inside an ordered pair}

\begin{formulabox}
\textbf{Rule:} $(x, y)$ means \textbf{across first, then up}. Always.\\[3pt]
A reversed pair is not the same place written differently --- it is a different place. $(6, 1)$ and $(1, 6)$ are built from the same digits and sit nowhere near each other.\\[3pt]
Two reversed pairs lie the same distance either side of the diagonal line where across equals up.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A point should be at $(6, 1)$ but was plotted at $(1, 6)$. Measured left to right, how far apart are the two marks?
\step{Left-to-right distance depends only on the across numbers, so I compare the first number of each pair and ignore the up numbers.}
\step{The across numbers are 6 and 1, so the gap is $6 - 1$.}
Distance across: $\ans{5}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A point should be at $(9, 2)$ but was plotted at $(2, 9)$. Measured left to right, how far apart are the marks?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$ units}
\end{tryitbox}

\skillheading{Skill 2 --- What one mark on an axis is worth}

\begin{formulabox}
\textbf{Rule:} Read the interval before you read the graph.\\[3pt]
value $=$ number of marks $\times$ value of one interval\\[3pt]
Find the interval by subtracting two labelled marks next to each other. Counting marks and reporting that count is the single most common graph-reading error, and it is always wrong by exactly the interval.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A vertical axis is labelled 0, 4, 8, 12, \ldots\ A bar reaches the 6th mark above zero. What value does it show?
\step{Two neighbouring labels differ by 4, so one mark is worth 4 --- I multiply the mark count by that instead of reading the count itself.}
\step{$6 \times 4 = 24$, which the labels confirm: the 6th mark is labelled 24.}
Value shown: $\ans{24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An axis is labelled 0, 25, 50, 75, \ldots\ A bar reaches the 3rd mark above zero. What value does it show?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\skillheading{Skill 3 --- Totals and means when the axis is scaled}

\begin{formulabox}
\textbf{Rule:} Convert marks to values \emph{first}, then total or average.\\[3pt]
A total built from mark counts is wrong by a factor of the interval --- and so is a mean built the same way. Multiplying at the end works too, because every bar shares the same interval, but converting first keeps the numbers meaningful.\\[3pt]
Changing the interval changes the picture, never the data.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad On a graph whose axis is marked every 10, four bars reach the 2nd, 5th, 3rd and 6th marks. What is the total?
\step{Mark counts are not values, so I add the counts and then convert the whole thing with the interval.}
\step{$2 + 5 + 3 + 6 = 16$ marks in all, and each mark is worth 10, so the total is $16 \times 10$.}
Total: $\ans{160}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On a graph whose axis is marked every 5, three bars reach the 4th, 1st and 6th marks. What is the total?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{55}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A taller bar does not always mean a bigger number. Before comparing two graphs, check that both axes use the same interval --- redrawing the same data on a finer scale makes every bar taller and changes nothing at all.
\end{watchoutbox}
