\sstitleblock{From Riemann Sums to the Fundamental Theorem}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Riemann Sums as Estimates}

\begin{formulabox}
\textbf{Recipe.} $\Delta x = \dfrac{b - a}{n}$, then add $f(x_k)\Delta x$ over
the chosen sample points: left endpoints, right endpoints, or midpoints.\par
\textbf{Which way it leans:} for an increasing $f$ a left sum underestimates and
a right sum overestimates; for a decreasing $f$ it is the other way round.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Right Riemann sum with $4$ subintervals for
$\displaystyle\int_0^4 x^2\,dx$.
\step{The interval is asked for in four equal pieces and the sample points are the right ends, so build the widths first and then add the heights.}
\step{$\Delta x = 1$ and the right endpoints are $1, 2, 3, 4$, so the sum is $1 + 4 + 9 + 16$.}

$\Rightarrow\quad \ans{30}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Right Riemann sum with $3$ subintervals for
$\displaystyle\int_0^3 x^2\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Accumulated Totals from a Rate}

\begin{formulabox}
\textbf{Rule.} If $r(t)$ is a rate, then $\displaystyle\int_a^b r(t)\,dt$ is the
\emph{total amount} accumulated between $a$ and $b$ --- evaluate an
antiderivative at the two limits and subtract.\par
Units multiply: a rate in litres per minute times minutes leaves litres. A
negative rate subtracts from the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\int_0^5 (2t + 1)\,dt$.
\step{The question asks how much accumulated in total, not how fast, so an antiderivative is needed and both limits get substituted.}
\step{$F(t) = t^2 + t$, so the value is $(25 + 5) - 0$.}

$\Rightarrow\quad \ans{30}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\int_0^4 (2t + 3)\,dt$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- What the FTC Lets You Conclude}

\begin{formulabox}
\textbf{Rule.} For $\displaystyle F(x) = \int_a^{u(x)} f(t)\,dt$,
\[ F'(x) = f\big(u(x)\big)\cdot u'(x). \]
So the \emph{sign of the integrand} decides where $F$ rises and falls, and a
sign change of $f$ marks a local extremum of $F$ --- all without ever computing
$F$ itself.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\displaystyle G(x) = \int_1^{x^2}(t + 2)\,dt$. Find
$G'(3)$.
\step{The upper limit is a function of $x$, so the theorem supplies the integrand at that limit and the chain rule supplies its derivative.}
\step{$G'(x) = \left(x^2 + 2\right)\cdot 2x$, so $G'(3) = 11\cdot 6$.}

$\Rightarrow\quad \ans{G'(3) = 66}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\displaystyle K(x) = \int_1^{x^2} 3t\,dt$. Find $K'(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{K'(2) = 48}$}
\end{tryitbox}
