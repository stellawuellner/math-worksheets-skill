\wstitleblock{Diagnosing Reversed and Non-Equivalent Ratios}{Grade 6}{}

\begin{center}
\textit{Find the mistake, name it, and fix it: reversed ratios, broken tables, and false equivalences}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Each problem shows work that might contain a ratio mistake. Decide
        what (if anything) went wrong, then fix it.
  \item Two classic errors to hunt for: writing the numbers in the
        \textbf{reversed order}, and \textbf{adding} to both parts of a ratio
        instead of multiplying.
\end{itemize}

\problem{A recipe uses 3 cups of oats and 5 cups of milk. Kim writes the ratio
of \textbf{milk to oats} as $3{:}5$. Write the correct ratio of milk to oats in
the blank, and say what Kim got mixed up.\par\vspace{4pt}
correct ratio of milk to oats: \ansblank
\par\vspace*{2.6cm}}

\problem{Jon says the ratios $4{:}6$ and $10{:}15$ are equivalent. Is he right?
Write $=$ or $\neq$ in the blank, and explain how you know by simplifying both
ratios in the work space.\par\vspace{4pt}
\begin{center}
$\dfrac{4}{6}$ \quad \ansblank \quad $\dfrac{10}{15}$
\end{center}
\par\vspace*{2.6cm}}

\problem{A car travels 120 miles in 3 hours. To find its speed, Leo computes
$3 \div 120 = 0.025$ and reports ``0.025 miles per hour.'' Find the correct
speed, and name the mistake Leo made.
\par\vspace*{3.2cm}\answerline{mph}}

\problem{The table is supposed to show rice and milk in the ratio $2{:}3$, but
one entry breaks the pattern.\par\vspace{6pt}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}
\hline
cups of rice & 2 & 4 & 6 & 8 \\
\hline
cups of milk & 3 & 6 & 10 & 12 \\
\hline
\end{tabular}
\end{center}
\vspace{4pt}
\begin{itemize}
  \item[(a)] Circle the entry that does not belong.
  \item[(b)] What should that entry be? \ansblank
\end{itemize}
\par\vspace*{2.4cm}}

\problem{A batch of 8 muffins needs 2 cups of flour. To find the flour for 12
muffins, Mia writes the proportion
\[ \frac{2}{8} = \frac{12}{x} \]
and gets $x = 48$ cups. Explain the error in Mia's proportion, then find the
correct number of cups for 12 muffins.
\par\vspace*{3.6cm}}

\problem{Three students each claim their ratio is equivalent to $3{:}4$. The
claims are $6{:}8$, \ $9{:}10$, and $15{:}20$ --- but one claim is false.
\par\vspace{4pt}
\begin{itemize}
  \item[(a)] Which ratio is \textbf{not} equivalent to $3{:}4$? \ansblank
  \item[(b)] Fix it: keep its first number and write what the second number
             should be. \ansblank
\end{itemize}
\par\vspace*{2.6cm}}

\problem{Punch A uses 3 cups of syrup for every 7 cups of water. Punch B uses 4
cups of syrup for every 9 cups of water. Raj writes $\frac{7}{3} > \frac{4}{9}$
and concludes, ``Punch A is much stronger.'' In the work space, say what Raj
mixed up. Then compare the punches fairly: write the correct comparison of
syrup-per-water fractions using $<$ or $>$.\par\vspace{4pt}
\begin{center}
$\dfrac{3}{7}$ \quad \ansblank \quad $\dfrac{4}{9}$
\end{center}
\par\vspace*{3.2cm}}

\problem{The table is supposed to show the ratio $5{:}2$, but \textbf{two} of
the bottom entries are wrong.\par\vspace{6pt}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}
\hline
top & 10 & 15 & 20 & 25 \\
\hline
bottom & 4 & 8 & 8 & 12 \\
\hline
\end{tabular}
\end{center}
\vspace{4pt}
\begin{itemize}
  \item[(a)] Corrected bottom entry under 15: \ansblank
  \item[(b)] Corrected bottom entry under 25: \ansblank
  \item[(c)] Explain the test you used on each column to decide whether it
             belongs.
\end{itemize}
