% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Related Rates: Differentiate First}

\begin{formulabox}
\textbf{Tell:} the problem gives a rate and asks for a rate.\par
Write the geometric relation, differentiate every variable with respect to $t$,
and substitute the instant's numbers only afterwards:
$\dfrac{d}{dt}\big(s^n\big) = ns^{n-1}\dfrac{ds}{dt}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A cube's edge grows at $2$ cm/s. Find $\dfrac{dV}{dt}$
when $s = 3$ cm.
\step{A rate is given and a rate is wanted, so differentiate the volume relation with respect to time rather than solving for a number first.}
\step{$V = s^3$, so $\dfrac{dV}{dt} = 3s^2\dfrac{ds}{dt} = 3(3)^2(2)$}

$\Rightarrow\quad \ans{\frac{dV}{dt} = 54}$ cubic centimetres per second.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same cube, same $2$ cm/s. Find $\dfrac{dV}{dt}$ when
$s = 5$ cm.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{150}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Optimization: Constraint and Objective}

\begin{formulabox}
\textbf{Tell:} the problem says largest, smallest, least, or most.\par
Name the objective (what is optimized) and the constraint (what is fixed), use
the constraint to reduce the objective to one variable, then set the
\emph{derivative} to zero --- never the objective itself.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rectangle's perimeter is $60$, so $A(x) = x(30 - x)$.
Maximize the area.
\step{The area is already in one variable, so the only remaining move is to differentiate it and set that derivative to zero.}
\step{$A(x) = 30x - x^2$, so $A'(x) = 30 - 2x$, and $30 - 2x = 0$ gives the critical value.}

$\Rightarrow\quad \ans{A'(x) = 30 - 2x}$, \quad maximum at $\ans{x = 15}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same setup with $A(x) = x(36 - x)$: find $A'(x)$ and the
maximizing $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{A'(x) = 36 - 2x}$, $\ans{x = 18}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Curve Analysis from f' and f''}

\begin{formulabox}
\textbf{Tell:} the problem asks where a graph rises, turns, or bends.\par
$f' = 0$ locates critical points; the sign of $f'$ says rising or falling.
$f'' = 0$ \emph{with a sign change} locates inflection points; the sign of
$f''$ says concave up or down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = x^3 - 3x^2 + 1$, find $f'(x)$ and the
critical values.
\step{Critical points are where the first derivative vanishes, so differentiate once and solve that equation.}
\step{$f'(x) = 3x^2 - 6x = 3x(x - 2)$, which is zero at each factor's root.}

$\Rightarrow\quad \ans{f'(x) = 3x^2 - 6x}$, \quad critical values $\ans{x = 0 \text{ or } 2}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = x^3 - 6x^2 + 2$, find $f'(x)$ and the
critical values.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f'(x) = 3x^2 - 12x}$, $\ans{x = 0 \text{ or } 4}$}
\end{tryitbox}
