\wstitleblock{Identities, Multiple Angles and Exact Solution Sets}{Precalculus}{}

\begin{center}\small\textit{Combining Pythagorean identities, sum and difference formulas, double angles, and exact solution sets on a restricted interval}\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Every answer is exact: keep radicals and fractions, and do not round.
  \item For an equation, give \emph{every} solution on the stated interval, in increasing order.
  \item The four tools --- Pythagorean identities, sum and difference formulas, double-angle formulas, and factoring to solve --- are mixed on purpose. Decide which one the problem needs before you start.
\end{itemize}
\vspace{0.15cm}

\problem[4.4cm]{Simplify $\dfrac{1 - \cos^2\theta}{\sin\theta}$ to a single trigonometric function.}

\problem[5cm]{Use a sum or difference formula to find the exact value of $\cos 15^\circ$.}

\problem[5.6cm]{An angle $\theta$ in the first quadrant has $\sin\theta = \dfrac{3}{5}$. Find the exact value of $\sin 2\theta$.}

\problem{Solve $2\cos\theta - \sqrt{3} = 0$ for $0^\circ \le \theta < 360^\circ$.
\par\vspace*{5cm}\answerline{degrees}}

\problem[6cm]{Use a difference formula to find the exact value of $\tan 15^\circ$. Rationalise the denominator so that no radical remains below the line.}

\problem{Solve $\sin 2\theta = \sin\theta$ for $0^\circ \le \theta < 360^\circ$. Give all four solutions.
\par\vspace*{6cm}\answerline{degrees}}

\problem[6cm]{Simplify $\dfrac{1}{1 + \cos\theta} + \dfrac{1}{1 - \cos\theta}$ to a single fraction with one trigonometric function in it.}

\problem{Solve $2\cos^2\theta + \cos\theta - 1 = 0$ for $0^\circ \le \theta < 360^\circ$.
\par\vspace*{6.4cm}\answerline{degrees}}

\problem{Solve $\sin 2\theta = \cos\theta$ for $0^\circ \le \theta < 360^\circ$. Give all four solutions.
\par\vspace*{6.4cm}\answerline{degrees}}

\problem[6cm]{Find the exact value of $\cos 105^\circ$ by writing $105^\circ$ as a sum of two special angles.}

\problem{Consider the identity $\dfrac{\sin\theta}{1 + \cos\theta} + \dfrac{1 + \cos\theta}{\sin\theta} = \dfrac{2}{\sin\theta}$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Combine the left-hand side over a common denominator and simplify it to a single term. \ansline
  \vspace*{3.4cm}
  \item[(b)] Prove the identity, working on the left-hand side only and naming the identity used at each step. \ansline
  \vspace*{4cm}
\end{itemize}}

\problem{Consider $\cos 2\theta + 3\sin\theta = 2$ on $0^\circ \le \theta < 360^\circ$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Use a double-angle formula to rewrite the equation, then factor the left-hand side of the resulting quadratic. \ansline
  \vspace*{3.4cm}
  \item[(b)] Give the exact solution set on the interval. \answerline{degrees}
  \vspace*{3.4cm}
  \item[(c)] Explain why this equation has exactly three solutions on the interval rather than four. \ansline
  \vspace*{2.8cm}
\end{itemize}}
