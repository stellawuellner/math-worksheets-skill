\wstitleblock{Completing Short Algebraic and Segment Proofs}{High-school geometry}{}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every line of a proof needs a reason. Use the properties of equality (addition, subtraction, multiplication, division, substitution, distributive), the definitions of midpoint and congruent segments, and the Segment Addition Postulate.
  \item All lengths are given with units. Answer with the unit shown on the answer line.
  \item Diagrams show labelling only; the measurements are in the problem text.
\end{itemize}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.4pt}

\problem{A shelf support of total length $22$ cm is built from three equal spacers of length $x$ cm and one $7$ cm bracket, so $3x + 7 = 22$.
\par\vspace{0.1cm}
(a) Find the spacer length $x$.
\par\vspace{0.15cm}
(b) Complete the reason column of the proof below.
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|p{5.2cm}|p{7.4cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $3x + 7 = 22$ & 1.\ Given \\ \hline
2.\ $3x = 15$ & 2.\ \rule{0pt}{0.95cm} \\ \hline
3.\ $x = 5$ & 3.\ \rule{0pt}{0.95cm} \\ \hline
\end{tabular}
\end{center}}
\par\vspace*{2.4cm}\answerline{cm}}

\problem{Three markers stand in a straight line along a trail, in the order $A$, $B$, $C$. The distances, in metres, are $AB = 2x$, $BC = x + 4$, and $AC = 25$.
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=1.0]
  \draw[thick] (0,0) -- (7,0);
  \fill (0,0) circle (1.6pt); \fill (4,0) circle (1.6pt); \fill (7,0) circle (1.6pt);
  \node[below] at (0,0) {$A$}; \node[below] at (4,0) {$B$}; \node[below] at (7,0) {$C$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
(a) Use the Segment Addition Postulate to write an equation, then find $x$.
\par\vspace*{3.4cm}\ansline
\par\vspace{0.15cm}
(b) Find the distance $BC$.
\par\vspace*{2.6cm}\answerline{m}}

\problem{A straight handrail $\overline{PQ}$ has its midpoint $M$ marked. Complete the reason column to prove that $PM$ is half of $PQ$.
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|p{5.6cm}|p{7.0cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $M$ is the midpoint of $\overline{PQ}$ & 1.\ Given \\ \hline
2.\ $\overline{PM} \cong \overline{MQ}$ & 2.\ \rule{0pt}{0.95cm} \\ \hline
3.\ $PM = MQ$ & 3.\ \rule{0pt}{0.95cm} \\ \hline
4.\ $PM + MQ = PQ$ & 4.\ \rule{0pt}{0.95cm} \\ \hline
5.\ $PM + PM = PQ$, so $PM = \frac{1}{2}PQ$ & 5.\ Substitution and division property \\ \hline
\end{tabular}
\end{center}}
\par\vspace*{2.8cm}\noansline}

\problem{Two cable runs on a bracket have lengths $5(x - 3)$ cm and $(2x + 9)$ cm, and they are equal.
\par\vspace{0.1cm}
(a) Find $x$.
\par\vspace{0.15cm}
(b) Write the reason for each line of the proof below.
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|p{5.2cm}|p{7.4cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $5(x - 3) = 2x + 9$ & 1.\ Given \\ \hline
2.\ $5x - 15 = 2x + 9$ & 2.\ \rule{0pt}{0.9cm} \\ \hline
3.\ $3x - 15 = 9$ & 3.\ \rule{0pt}{0.9cm} \\ \hline
4.\ $3x = 24$ & 4.\ \rule{0pt}{0.9cm} \\ \hline
5.\ $x = 8$ & 5.\ \rule{0pt}{0.9cm} \\ \hline
\end{tabular}
\end{center}}
\par\vspace*{2.8cm}\answerline{cm}}

\problem{Four survey markers lie in a straight line along a fence, in the order $A$, $B$, $C$, $D$. It is given that $AB = CD$.
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=1.0]
  \draw[thick] (0,0) -- (8,0);
  \fill (0,0) circle (1.6pt); \fill (1.6,0) circle (1.6pt);
  \fill (6.4,0) circle (1.6pt); \fill (8,0) circle (1.6pt);
  \node[below] at (0,0) {$A$}; \node[below] at (1.6,0) {$B$};
  \node[below] at (6.4,0) {$C$}; \node[below] at (8,0) {$D$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
(a) Prove that $AC = BD$. Write your argument in two or three sentences, naming a reason for each step.
\par\vspace*{4.6cm}
(b) The markers are then measured: $AB = CD = 3.5$ m and $BC = 12$ m. Find $AC$.
\par\vspace*{2.6cm}\answerline{m}}

\problem{The same four markers $A$, $B$, $C$, $D$ lie in a straight line, but now it is given that $AC = BD$. Complete the reason column to prove $AB = CD$.
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|p{5.6cm}|p{7.0cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $AC = BD$ & 1.\ \rule{0pt}{0.85cm} \\ \hline
2.\ $AB + BC = AC$ & 2.\ \rule{0pt}{0.85cm} \\ \hline
3.\ $BC + CD = BD$ & 3.\ \rule{0pt}{0.85cm} \\ \hline
4.\ $AB + BC = BC + CD$ & 4.\ \rule{0pt}{0.85cm} \\ \hline
5.\ $AB = CD$ & 5.\ \rule{0pt}{0.85cm} \\ \hline
\end{tabular}
\end{center}}
\par\vspace*{2.8cm}\noansline}

\problem{A rectangular garden bed has perimeter $P = 2l + 2w$. Its perimeter is $46$ m and its width is $8$ m.
\par\vspace{0.1cm}
(a) Find the length $l$.
\par\vspace*{4.2cm}\answerline{m}
\par\vspace{0.15cm}
(b) Name the property of equality that justifies each step of your work: replacing $P$ and $w$ by their measured values, removing the doubled width from both sides, and the final step.
\par\vspace*{3.6cm}}

\problem{A footbridge rests on four supports $W$, $X$, $Y$, $Z$ set in a straight line in that order. The two outer spans are equal, $WX = YZ$. The middle span is $XY = 6.4$ m and the whole bridge is $WZ = 21.0$ m.
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=1.0]
  \draw[thick] (0,0) -- (8,0);
  \fill (0,0) circle (1.6pt); \fill (2.4,0) circle (1.6pt);
  \fill (5.6,0) circle (1.6pt); \fill (8,0) circle (1.6pt);
  \node[below] at (0,0) {$W$}; \node[below] at (2.4,0) {$X$};
  \node[below] at (5.6,0) {$Y$}; \node[below] at (8,0) {$Z$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
(a) Prove that $WY = XZ$, naming a reason for each step.
\par\vspace*{4.8cm}
(b) Find the length of one outer span $WX$.
\par\vspace*{3.0cm}\answerline{m}}
