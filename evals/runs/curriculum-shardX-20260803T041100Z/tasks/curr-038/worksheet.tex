\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A missing addend is the \textbf{part you cannot see}. The big number is the \textbf{whole}.
  \item Count on from the part you know until you reach the whole, or take the known part away from the whole.
  \item Draw dots, cross out, or write your counting on the empty space. Then write your answer on the line.
\end{itemize}
\normalsize

\problem[3cm]{\textbf{Counters on a ten-frame.} Six counters are on the mat.
The mat holds \textbf{10} counters when it is full.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
\rule{0pt}{0.8cm}$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\rule{0pt}{0.8cm}$\bullet$ & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\end{tabular}
\end{center}
How many more counters make the mat full? \quad $6 + \underline{\hspace{1.2cm}} = 10$}

\problem[3cm]{\textbf{Counters on two ten-frames.} Eight counters are on the mats.
Ana wants \textbf{13} counters in all.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
\rule{0pt}{0.8cm}$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\rule{0pt}{0.8cm}$\bullet$ & $\bullet$ & $\bullet$ & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\end{tabular}\quad
\begin{tabular}{|c|c|c|c|c|}\hline
\rule{0pt}{0.8cm}\makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\rule{0pt}{0.8cm}\makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\end{tabular}
\end{center}
Draw the counters she still needs, then finish the sentence. \quad $8 + \underline{\hspace{1.2cm}} = 13$}

\problem[3.5cm]{\textbf{Number bond.} The whole is \textbf{15} apples.
One part is \textbf{9} apples. Write the missing part inside the empty circle,
then finish the equation.
\begin{center}\large
\begin{tabular}{c c c}
 & \fbox{\rule{0pt}{0.9cm}\makebox[1.1cm]{15}} & \\[4pt]
\fbox{\rule{0pt}{0.9cm}\makebox[1.1cm]{9}} & & \fbox{\rule{0pt}{0.9cm}\makebox[1.1cm]{}} \\
\end{tabular}
\end{center}
$9 + \underline{\hspace{1.2cm}} = 15$}

\problem[3.5cm]{\textbf{Use subtraction.} A box holds \textbf{16} crayons when it is full.
There are \textbf{7} crayons in the box now.
\begin{center}\large $7 + \underline{\hspace{1.2cm}} = 16$\end{center}
First write the subtraction sentence that finds the missing part.
Then solve it.}

\problem[3.5cm]{\textbf{Cube train.} The whole train is \textbf{12} cubes long.
The gray part at the end is \textbf{5} cubes. How long is the white part?
\begin{center}\large
\begin{tabular}{|c|c|}\hline
\rule{0pt}{0.9cm}\makebox[4.2cm]{white part} & \makebox[1.9cm]{5} \\\hline
\multicolumn{2}{|c|}{\rule{0pt}{0.7cm}\makebox[6.3cm]{whole train: 12}} \\\hline
\end{tabular}
\end{center}
$\underline{\hspace{1.2cm}} + 5 = 12$}

\problem{\textbf{Building a tower.} Sam is building a tower of \textbf{20} cubes.
He has already stacked \textbf{12} cubes. The rest of the cubes are blue.
How many blue cubes does he still need? Write the equation you used.\par\vspace*{3.5cm}
\answerline{cubes}}

\problem[4cm]{\textbf{Find the mistake.} Mia has \textbf{5} shells.
She wants \textbf{13} shells in all. She writes $5 + \underline{\hspace{1.2cm}} = 13$
and then says the missing number is \textbf{18} because $5 + 13 = 18$.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Explain in words what Mia did wrong.
  \item Write the correct missing number.
\end{itemize}}

\problem[4.5cm]{\textbf{Two missing numbers.} Solve both equations, then compare.
\begin{center}\large
$\underline{\hspace{1.2cm}} + 6 = 14$ \qquad and \qquad $9 + \underline{\hspace{1.2cm}} = 14$
\end{center}
Write $>$ or $<$ in the blank to compare the two missing numbers:
\begin{center}\large
first missing number \ \underline{\hspace{1.5cm}} \ second missing number
\end{center}}
