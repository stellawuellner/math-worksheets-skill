\wstitleblock{Reading and Sketching Slope Fields}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Each printed field draws the slope $\dfrac{dy}{dx}$ as a short segment
        at every lattice point. A solution curve is any curve that stays
        \emph{tangent} to those segments --- it never cuts across one.
  \item Sketch directly on the printed field. A dot marks the point your curve
        must pass through.
  \item Give exact values (fractions, $e$) unless a decimal is requested.
\end{itemize}
