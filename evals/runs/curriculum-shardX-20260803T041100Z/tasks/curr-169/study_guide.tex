\sstitleblock{Subtraction and Negative Signs: Diagnosing the Error}

\skillheading{Skill 1 --- Subtracting Signed Numbers}

\begin{formulabox}
\textbf{Rule --- add the opposite:} \quad $p - q = p + (-q)$
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item $p - (-q) = p + q$ \quad (subtracting a negative moves \emph{right})
  \item $p - q = p + (-q)$ \quad (subtracting a positive moves \emph{left})
  \item Same signs after rewriting: add the distances, keep that sign.
  \item Different signs after rewriting: subtract the distances, keep the sign of the farther-from-zero number.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Evaluate $-6 - (-2)$.
\step{The second number is negative, so ``add the opposite'' turns the subtraction into an addition of a positive.}
\step{$-6 - (-2) = -6 + 2$; the signs differ, so $6 - 2 = 4$ with the sign of $-6$, giving $\ans{-4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Evaluate $-9 - (-4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Compare Two Signed Expressions}

\begin{formulabox}
\textbf{Compare by rewriting, not by computing.} Turn each side into
``first number $+$ signed number,'' then read the two forms.\par
Two expressions are equal exactly when they add the same signed number to the same start.
On the number line, \emph{left is less}: $-9 < -5 < 5$, even though $9 > 5$ as digits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Put $<$, $>$, or $=$ between $-5 - (-2)$ and $-5 + 2$.
\step{Both sides start at $-5$, so rewriting the left side as an addition will make the comparison readable without arithmetic.}
\step{$-5 - (-2) = -5 + 2$, which is the right side character for character, so the relation is $\ans{=}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Put $<$, $>$, or $=$ between $3 - 8$ and $3 + 8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

\skillheading{Skill 3 --- Find and Fix a Sign Error}

\begin{formulabox}
\textbf{Diagnose from the answer, in this order:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Answer too far \emph{left}? The double negative was dropped: $-(-q)$ was read as $-q$.
  \item Answer too far \emph{right}? A subtraction of a positive was turned into an addition.
  \item Answer correct in size but positive? The two numbers were reversed to avoid a negative.
\end{itemize}
\textbf{Size check first:} subtracting a negative always increases the value; subtracting a positive always decreases it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A student writes $-1.5 - 3.5 = 2$. Find and fix.
\step{Subtracting a positive must move left, so an answer larger than $-1.5$ fails the size check --- the sign of the second term was flipped.}
\step{Correctly, $-1.5 - 3.5 = -1.5 + (-3.5)$; same signs, so the distances add to give $\ans{-5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A student writes $-2.75 - (-5) = -7.75$. Find and fix.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.25}$}
\end{tryitbox}
