\sstitleblock{Picking an Addition Strategy}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Count On (small second number)}

\begin{formulabox}
\textbf{When to use it:} one of the numbers is little (1, 2 or 3).\par
\textbf{Rule:} say the \textbf{bigger} number, then hop the little one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $7 + 2$
\step{The 2 is small, so counting on from the bigger number is the quick tool here.}
\step{Say 7, hop 8, 9 --- two hops for the 2. $\checkmark$}

$\Rightarrow\quad 7 + 2 = \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $1 + 8$ \ (start big!)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Doubles and Near Doubles}

\begin{formulabox}
\textbf{When to use it:} the numbers match, or almost match.\par
\textbf{Rule:} know the doubles by heart ($2+2$, $3+3$, $4+4$, $5+5$); for
neighbors, use the double and one more.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $2 + 3$
\step{2 and 3 are neighbors, so we use the double of the smaller number and one more.}
\step{$2+2=4$, and one more makes 5. $\checkmark$}

$\Rightarrow\quad 2 + 3 = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $4 + 5$ \ (use $4+4$ and one more)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Tens Partners}

\begin{formulabox}
\textbf{When to use it:} the two numbers fit together to make ten.\par
\textbf{Rule:} know the partner pairs --- 1 and 9, \ 2 and 8, \ 3 and 7, \ 4
and 6, \ 5 and 5 --- and answer 10 right away.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 + 2$
\step{8 and 2 are a partner pair, so no counting is needed --- partners always make ten.}
\step{$8 + 2$ fills the ten-frame exactly. $\checkmark$}

$\Rightarrow\quad 8 + 2 = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $9 + 1$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Counting on works for everything --- but it is slow
for big second numbers. If you catch yourself hopping 7 times, look again:
a double or a tens partner was probably hiding there!
\end{watchoutbox}
