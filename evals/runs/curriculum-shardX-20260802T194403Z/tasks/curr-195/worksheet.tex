\wstitleblock{Surface Area and Scale Design Workshop}{Grade 6--7 --- Geometry}{}

\noindent\small\textbf{Focus:} Prism surface area, cylinder surface area, composite plan areas,
and scale-factor design.\par\vspace{2pt}
\textbf{Directions:} Every problem takes more than one step. Write the formula you are using,
substitute the given measurements, and label your answer with its unit. Use $\pi \approx 3.14159$
and round money-free area answers to two decimal places only when the answer is not exact.
Diagrams show the shape and the letters used for its measurements; the numbers are in the problem.
\normalsize

\vspace{0.15cm}

\problem{A closed cardboard carton is a rectangular prism measuring
$\ell = 12$ cm long, $w = 8$ cm wide, and $h = 5$ cm high.
\begin{center}
\begin{tikzpicture}[scale=0.42]
  \draw[thick] (0,0) -- (4,0) -- (4,2.5) -- (0,2.5) -- cycle;
  \draw[thick] (4,0) -- (5.2,0.8) -- (5.2,3.3) -- (4,2.5);
  \draw[thick] (0,2.5) -- (1.2,3.3) -- (5.2,3.3);
  \draw[dashed] (0,0) -- (1.2,0.8) -- (5.2,0.8);
  \draw[dashed] (1.2,0.8) -- (1.2,3.3);
  \node[below] at (2,0) {$\ell$};
  \node[right] at (4.7,0.4) {$w$};
  \node[left] at (0,1.25) {$h$};
\end{tikzpicture}
\end{center}
Find the total surface area of the carton --- the cardboard needed for all six faces.
\par\vspace*{5cm}\answerline{cm$^2$}}

\problem{A floor plan is drawn to the scale 1 cm on the drawing $=$ 4 m in the building.
On the drawing, a rectangular room measures 6.5 cm by 3.25 cm.\\[3pt]
Find the \textbf{actual perimeter} of the room --- the length of skirting board needed to run
right around it.
\par\vspace*{5cm}\answerline{m}}

\problem{A closed tin can is a cylinder with radius $r = 3$ cm and height $h = 8$ cm.
\begin{center}
\begin{tikzpicture}[scale=0.42]
  \draw[thick] (0,3) ellipse (1.4 and 0.4);
  \draw[thick] (-1.4,3) -- (-1.4,0);
  \draw[thick] (1.4,3) -- (1.4,0);
  \draw[thick] (-1.4,0) arc (180:360:1.4 and 0.4);
  \draw[dashed] (1.4,0) arc (0:180:1.4 and 0.4);
  \node[right] at (0.05,1.5) {$h$};
  \draw (0,3) -- (1.4,3);
  \node[above] at (0.7,3.05) {$r$};
\end{tikzpicture}
\end{center}
Find the total surface area of the can: the curved side plus both circular ends.
\par\vspace*{5cm}\answerline{cm$^2$}}

\problem{The plan below shows the floor of a workshop. Each grid square is 1 metre by 1 metre.
\begin{center}
\begin{tikzpicture}[scale=0.4]
  \draw[step=1, gray!45, very thin] (0,0) grid (8,9);
  \draw[very thick] (0,0) -- (8,0) -- (8,5) -- (3,5) -- (3,9) -- (0,9) -- cycle;
\end{tikzpicture}
\end{center}
Split the plan into rectangles and find the total floor area.
\par\vspace*{4.5cm}\answerline{m$^2$}}

\problem{A garden shed is a rectangular prism 6 m long, 4 m wide and 3 m high. The four walls
and the flat roof will be painted; the concrete floor will \textbf{not} be painted.\\[3pt]
Find the area to be painted.
\par\vspace*{5.5cm}\answerline{m$^2$}}

\problem{On the same 1 cm $=$ 4 m drawing, a rectangular deck measures 5 cm by 3 cm.\\[3pt]
Find the \textbf{actual area} of the deck. (Careful: the scale factor for lengths is not the
scale factor for areas.)
\par\vspace*{5.5cm}\answerline{m$^2$}}

\problem{A paper label is wrapped around the curved side of a cylindrical can of radius 4 cm and
height 11 cm. The label is glued with an overlap, so its length is 1 cm longer than the
circumference of the can, and it is as tall as the can.\\[3pt]
Find the area of the paper label.
\par\vspace*{5.5cm}\answerline{cm$^2$}}

\problem{A shed design uses two identical L-shaped side panels, drawn below. Each panel is a
10 m by 4 m rectangle with a 6 m by 3 m rectangle attached along the top of its left-hand side.
\begin{center}
\begin{tikzpicture}[scale=0.35]
  \draw[very thick] (0,0) -- (10,0) -- (10,4) -- (6,4) -- (6,7) -- (0,7) -- cycle;
\end{tikzpicture}
\end{center}
Find the total area of material needed for \textbf{both} panels.
\par\vspace*{5cm}\answerline{m$^2$}}

\problem{A designer builds a model of a shipping crate at the scale 1 : 5, so every length on the
real crate is 5 times the matching length on the model. The model's total surface area is
148 cm$^2$.\\[3pt]
Find the total surface area of the real crate, in square centimetres.
\par\vspace*{5.5cm}\answerline{cm$^2$}}

\problem{A closed box has a square base measuring 8 cm by 8 cm, and its total surface area is
448 cm$^2$.\\[3pt]
Work backwards from the surface-area formula to find the height of the box.
\par\vspace*{6cm}\answerline{cm}}

\problem{\textbf{Design decision.} Two closed boxes hold the same volume of 216 cubic
centimetres. Box A is a cube measuring 6 cm by 6 cm by 6 cm. Box B measures 12 cm by 6 cm by
3 cm.\\[3pt]
Which box uses less cardboard? Support your answer with a full surface-area comparison for both
boxes, then give one practical reason a designer might still choose the other box.
\par\vspace*{7cm}\noansline}

\problem{\textbf{Synthesis.} A scale drawing of a cylindrical water tank uses the scale
1 cm $=$ 2.5 m. On the drawing the tank is 1.6 cm across (its diameter) and 3.2 cm tall.
The real tank will have its curved wall and its flat circular top painted; the base sits on the
ground and is not painted.\\[3pt]
Find the actual area to be painted.
\par\vspace*{6.5cm}\answerline{m$^2$}}
