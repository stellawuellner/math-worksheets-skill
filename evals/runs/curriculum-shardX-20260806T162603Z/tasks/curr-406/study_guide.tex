\sstitleblock{End Behavior and Multiplicity}

\skillheading{Skill 1 --- Zeros and what their multiplicity does}

\begin{formulabox}
\textbf{Rule.} A factor $(x - r)^{m}$ gives a zero at $x = r$ of multiplicity $m$.
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item $m$ odd: the curve \emph{crosses} the axis at $r$ (the product changes sign)
  \item $m$ even: the curve \emph{touches} and turns back (no sign change)
\end{itemize}
The multiplicities add up to the degree.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad List the zeros of $p(x) = (x - 4)(x + 1)^{2}$.
\step{Each factor contributes one zero, and its exponent is what says whether the curve crosses there --- so read the exponents, not just the roots.}
\step{$x - 4 = 0$ gives a crossing at $4$; $(x+1)^{2} = 0$ gives a touch at $-1$.}

$\Rightarrow\quad \ans{x = 4}$ \ and \ $\ans{x = -1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad List the zeros of $p(x) = (x + 5)(x - 2)^{2}$, and say which is a touch.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -5}$ crossing, $\ans{x = 2}$ touch}
\end{tryitbox}

\skillheading{Skill 2 --- End behavior from the leading term}

\begin{formulabox}
\textbf{Rule.} Only the degree and the sign of the leading coefficient matter.
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item degree \emph{even}: both ends go the same way --- up if the leading coefficient is positive, down if negative
  \item degree \emph{odd}: the ends go opposite ways --- up on the right if positive, down on the right if negative
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\lim_{x \to \infty} \left(-4x^{3} + x\right)$.
\step{Every other term is out-grown by the leading one, so discard them and read off the sign of $-4x^{3}$.}
\step{For large positive $x$, $x^{3}$ is large and positive, and multiplying by $-4$ makes it large and negative.}

$\Rightarrow\quad \ans{-\infty}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\lim_{x \to -\infty} \left(2x^{6} - 9x\right)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\infty}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the constant term never affects end behavior, and neither does any middle term. It shifts the curve but cannot change where the two ends go.
\end{watchoutbox}

\skillheading{Skill 3 --- Assembling the sketch}

\begin{formulabox}
\textbf{Four steps.} Plot the zeros and mark each as a crossing or a touch; find the $y$-intercept by evaluating at $0$; draw the two ends from the degree and the leading sign; join everything with one smooth curve that changes sign only where it should.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the $y$-intercept of $p(x) = (x - 3)(x + 1)^{2}$, then describe the sketch.
\step{The intercept is the one point not given by the factors, so evaluate at $0$ before drawing anything.}
\step{$p(0) = (0-3)(0+1)^{2} = -3$; the curve crosses at $3$, touches at $-1$, and being degree $3$ with a positive lead it falls left and rises right.}

$\Rightarrow\quad \ans{p(0) = -3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the $y$-intercept of $p(x) = (x - 2)(x + 4)^{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{p(0) = -32}$}
\end{tryitbox}
