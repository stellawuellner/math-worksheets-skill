\sstitleblock{Accumulation Functions and the Fundamental Theorem}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The FTC Derivative Rule}

\begin{formulabox}
\textbf{Rule.} If $\displaystyle G(x) = \int_a^x f(t)\,dt$ with $a$ constant,
then
\[ G'(x) = f(x). \]
Differentiating undoes the accumulation: you never need an antiderivative.
Read the units the same way --- if $f$ is litres per minute, $G$ is litres and
$G'$ is back to litres per minute.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\displaystyle G(x) = \int_1^x (t^2 + 3)\,dt$. Find $G'(4)$.
\step{The lower limit is constant and the upper limit is plain $x$, so the theorem applies directly and no antiderivative is needed.}
\step{$G'(x) = x^2 + 3$, so $G'(4) = 16 + 3$.}

$\Rightarrow\quad \ans{G'(4) = 19}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\displaystyle G(x) = \int_2^x (2t + 5)\,dt$. Find $G'(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{G'(3) = 11}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- A Variable Limit Needs the Chain Rule}

\begin{formulabox}
\textbf{Rule.} If the upper limit is a function $u(x)$, its derivative rides
along:
\[ \frac{d}{dx}\int_a^{u(x)} f(t)\,dt = f\big(u(x)\big)\cdot u'(x). \]
A variable in the \emph{lower} limit flips the sign, because reversing the
limits negates the integral.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\displaystyle H(x) = \int_0^{x^3}\sqrt{t}\,dt$. Find $H'(4)$.
\step{The upper limit is a function of $x$ rather than $x$ itself, so the theorem supplies the integrand and the chain rule supplies the extra factor.}
\step{$H'(x) = \sqrt{x^3}\cdot 3x^2$, so $H'(4) = 8\cdot 48$.}

$\Rightarrow\quad \ans{H'(4) = 384}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\displaystyle K(x) = \int_0^{x^2}(t + 1)\,dt$. Find $K'(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{K'(3) = 60}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Evaluating an Accumulated Amount}

\begin{formulabox}
\textbf{Rule.} To find how much accumulated --- rather than how fast it is
accumulating --- you \emph{do} need an antiderivative:
\[ \int_a^b f(t)\,dt = F(b) - F(a), \qquad F' = f. \]
A negative rate subtracts from the total, so a net accumulation can fall even
while the integral is still being taken.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle \int_0^4 (2t + 1)\,dt$.
\step{This asks for the total accumulated, not a rate, so find an antiderivative and subtract its values at the two limits.}
\step{$F(t) = t^2 + t$, so the integral is $(16 + 4) - (0 + 0)$.}

$\Rightarrow\quad \ans{20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle \int_0^3 (4t + 2)\,dt$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}
