\sstitleblock{Reading the Number in the Right Place}

\skillheading{Skill 1 --- Across first, then up}

\begin{formulabox}
\textbf{Rule:} in $(x,\,y)$ the \textbf{first} number counts across from the
corner and the \textbf{second} counts up.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Say it as you plot: ``along the hall, then up the stairs.''
  \item The same two numbers in the other order give a \emph{different} point,
        usually far away --- not a small slip.
  \item To count grid steps between two points: add the across difference and
        the up difference.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad How many grid steps apart are $(3,\,8)$ and $(8,\,3)$,
moving only along grid lines?
\step{The two pairs use the same numbers in opposite slots, so they are two
different points --- count the across move and the up move separately, then add
them.}
\step{Across: $3$ to $8$ is $5$ steps. Up: $8$ down to $3$ is $5$ steps. Total
$5 + 5 = \ans{10}$ steps.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad How many grid steps apart are $(2,\,5)$ and $(5,\,2)$,
moving only along grid lines?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Same two numbers'' never means ``same point''. Plot
both and look --- they sit on opposite sides of the diagonal.
\end{watchoutbox}

\skillheading{Skill 2 --- Counting marks on a line plot}

\begin{formulabox}
\textbf{Two numbers, two jobs.} The number \emph{under} the line says what was
measured. The marks \emph{above} it say how many.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item ``How many?'' $\rightarrow$ count marks. \ ``How long / how tall?''
        $\rightarrow$ read the number under the line.
  \item To find how many were measured altogether, add every stack.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Each $\times$ is one ribbon. How many ribbons are
$1\tfrac{1}{2}$ metres long?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}[scale=0.8]
  \draw[thick] (0.4,0) -- (3.8,0);
  \node at (1.0,0.32) {$\times$}; \node at (1.0,0.7) {$\times$};
  \node at (2.1,0.32) {$\times$}; \node at (2.1,0.7) {$\times$};
  \node at (2.1,1.08) {$\times$}; \node at (2.1,1.46) {$\times$};
  \node at (3.2,0.32) {$\times$}; \node at (3.2,0.7) {$\times$};
  \node at (3.2,1.08) {$\times$};
  \draw (1.0,0.09) -- (1.0,-0.09); \node[below] at (1.0,-0.09) {$1$};
  \draw (2.1,0.09) -- (2.1,-0.09); \node[below] at (2.1,-0.09) {$1\tfrac{1}{2}$};
  \draw (3.2,0.09) -- (3.2,-0.09); \node[below] at (3.2,-0.09) {$2$};
\end{tikzpicture}
\end{center}
\step{The question asks how many \emph{ribbons}, so the answer comes from
counting marks, not from the number printed under the line.}
\step{Find $1\tfrac{1}{2}$ on the line and count the marks stacked above it:
there are $\ans{4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad On another line plot, $2$ marks stand above the $5$, $5$
marks above the $6$ and $3$ marks above the $7$. How many things measured
$7$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Ordered pairs from two patterns}

\begin{formulabox}
\textbf{Rule:} whichever pattern is named \emph{first} supplies the
\emph{first} number of every pair.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Write both patterns out in a row before pairing them up, and match term
        $1$ with term $1$, term $2$ with term $2$, and so on.
  \item Getting both numbers right is only half the job --- the order carries
        the meaning.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Pattern P starts at $2$ and adds $4$. Pattern Q starts at
$1$ and adds $2$. Write the $3$rd pair $(P,\,Q)$.
\step{P is named first, so P's third term goes in the first slot and Q's third
term in the second --- list both patterns before pairing anything.}
\step{P: $2,\ 6,\ 10$, so the first number is $\ans{10}$. \ Q: $1,\ 3,\ 5$, so
the second number is $\ans{5}$. The pair is $(10,\,5)$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Pattern P starts at $5$ and adds $5$. Pattern Q starts at
$1$ and adds $3$. Write the $4$th pair $(P,\,Q)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$ then $\ans{10}$}
\end{tryitbox}
