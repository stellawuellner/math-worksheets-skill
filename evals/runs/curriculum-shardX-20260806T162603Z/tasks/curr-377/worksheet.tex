\wstitleblock{Product, Quotient, and Power Rules for Logarithms}{Algebra 2}{}

\begin{center}
\textit{The product rule, the quotient rule, the power rule, and combining all three to condense or expand a logarithm}
\end{center}

\noindent\textbf{The three rules} (for $b>0$, $b \ne 1$, and positive inputs)

\begin{center}
\begin{tabular}{|l|c|l|}
\hline
\textbf{Rule} & \textbf{Statement} & \textbf{Worked pattern} \\
\hline
Product & $\log_b(MN) = \log_b M + \log_b N$ & $\log_3 9 + \log_3 3 = \log_3 27 = 3$ \\
\hline
Quotient & $\log_b\!\dfrac{M}{N} = \log_b M - \log_b N$ & $\log_2 40 - \log_2 5 = \log_2 8 = 3$ \\
\hline
Power & $\log_b(M^{\,p}) = p\log_b M$ & $3\log_5 5 = \log_5 5^3 = 3$ \\
\hline
\end{tabular}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=3pt]
  \item Each rule reads in both directions: use it left to right to \emph{expand}, right to left to \emph{condense}.
  \item Work from the pattern in the table on problems 1--3, then keep going on your own.
  \item In problems where the base is written as $b$, the answer holds for every legal base --- leave $b$ in your answer.
  \item Where an exact value exists, give the exact value, not a decimal.
\end{itemize}
