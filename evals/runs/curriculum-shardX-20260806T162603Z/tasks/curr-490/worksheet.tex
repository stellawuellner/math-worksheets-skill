\begin{center}
\small Average value, accumulated change from a rate, and motion along a line
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The first three problems are a warm-up, one on each idea. After that the three ideas are mixed --- decide which one a problem is asking for before you write anything.
  \item Three questions to tell them apart: \emph{How much accumulated?} integrate the rate. \emph{What was it on average?} integrate, then divide by the length of the interval. \emph{How far did it actually travel?} integrate the absolute value of the velocity.
  \item Where an answer carries a unit, the answer line names it.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Find the average value of $f(x) = 3x^2$ on the interval $[0, 4]$.}

\problem{Water flows into a tank at the rate $r(t) = 4t + 2$ litres per minute, where $t$ is measured in minutes. How much water enters the tank during the first $6$ minutes?
\par\vspace*{5cm}\answerline{L}}

\problem{A particle moves along a straight line with velocity $v(t) = 3t^2 - 12$ metres per second. Find the particle's displacement between $t = 0$ and $t = 4$ seconds.
\par\vspace*{5cm}\answerline{m}}

\problem[5.5cm]{Find the average value of $g(x) = \sin x$ on the interval $[0, \pi]$. Give an exact answer and a decimal to three places.}

\problem{Sand is poured onto a pile at the rate $r(t) = 9 - t$ kilograms per minute, for $0 \le t \le 9$ minutes. How much sand is added to the pile over those $9$ minutes?
\par\vspace*{5cm}\answerline{kg}}

\problem{A particle moves along a straight line with velocity $v(t) = t^2 - 4t + 3$ metres per second, for $0 \le t \le 4$ seconds.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the times in $[0, 4]$ at which the particle changes direction. \ansline
  \item[(b)] Find the total distance the particle travels over $[0, 4]$.
\end{itemize}
\par\vspace*{6cm}\answerline{m}}

\problem[5.5cm]{The temperature inside a greenhouse is modelled by $T(t) = 18 + 6\sin\!\left(\dfrac{\pi t}{12}\right)$ degrees Celsius, where $t$ is the number of hours after midnight. Find the average temperature over the first $12$ hours. Give an exact answer and a decimal to three places.}

\problem{Water leaks from a storage tank at the rate $r(t) = 20e^{-0.1t}$ litres per hour, where $t$ is in hours. How much water leaks out during the first $10$ hours? Give an exact answer and a decimal to two places.
\par\vspace*{5.5cm}\answerline{L}}

\problem{A particle moves along the $x$-axis with velocity $v(t) = 4 - t$ metres per second, and its position at time $t = 0$ is $x = 3$ metres. Time is measured in seconds.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] At what time does the particle change direction? \ansline
  \item[(b)] Find the particle's position at $t = 6$.
\end{itemize}
\par\vspace*{5.5cm}\answerline{m}}

\problem{A particle moves along the $x$-axis with velocity $v(t) = t^2 - 6t + 8$ metres per second for $0 \le t \le 5$ seconds, starting at $x = 1$ metre.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the total distance the particle travels on $[0, 5]$. \answerline{m}
  \item[(b)] Find the particle's average velocity on $[0, 5]$. \answerline{m/s}
  \item[(c)] Explain why the average velocity is not the same as the average of the largest and smallest velocities on the interval.
\end{itemize}
