\problem{A machine shop cuts two congruent steel gusset plates, so that
$\triangle ABC \cong \triangle DEF$. The shop drawing lists $AB = (2x+3)$ cm and
$DE = 15$ cm. Find $x$.}

\noindent\textbf{Solution.} The statement pairs $A$ with $D$ and $B$ with $E$,
so $\overline{AB}$ corresponds to $\overline{DE}$ and the two lengths are equal:
$2x + 3 = 15$. Subtracting $3$ gives $2x = 12$, so \ans{x = 6}

\problem{The two spars of a kite frame form congruent triangles with
$\triangle PQR \cong \triangle STU$. The pattern lists
$m\angle P = (5x-10)^\circ$ and $m\angle S = 65^\circ$. Find $x$.}

\noindent\textbf{Solution.} $P$ is paired with $S$, so $\angle P$ and $\angle S$
are corresponding angles and have equal measures: $5x - 10 = 65$. Adding $10$
gives $5x = 75$, so \ans{x = 15}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{A footbridge is stiffened by two congruent gusset triangles with
$\triangle JKL \cong \triangle MNP$. The fabrication drawing gives
$JK = (3y+4)$ ft and $MN = (5y-12)$ ft. (a) Find $y$. (b) Find the length of
$\overline{MN}$, in feet.}

\noindent\textbf{Solution (a).} $J$ pairs with $M$ and $K$ pairs with $N$, so
$\overline{JK}$ corresponds to $\overline{MN}$:
\[ 3y + 4 = 5y - 12 .\]
Subtract $3y$ from both sides: $4 = 2y - 12$. Add $12$: $16 = 2y$, so \ans{y = 8}

\noindent\textbf{Solution (b).} Substitute into the expression for $MN$:
$MN = 5(8) - 12 = 40 - 12$, so \ans{MN = 28 \text{ ft}}

\noindent\textit{Check:} $JK = 3(8) + 4 = 28$ ft as well, as it must be.

\problem{A roof frame is built from two congruent trusses with
$\triangle ABC \cong \triangle XYZ$. The plan gives $m\angle A = 48^\circ$,
$m\angle B = (4x+6)^\circ$, and $m\angle Y = 74^\circ$. (a) Find $x$.
(b) Find $m\angle Z$, in degrees.}

\noindent\textbf{Solution (a).} $B$ pairs with $Y$, so $4x + 6 = 74$. Subtract
$6$: $4x = 68$, so \ans{x = 17}

\noindent\textbf{Solution (b).} $Z$ pairs with $C$, so $m\angle Z = m\angle C$.
The angles of $\triangle ABC$ sum to $180^\circ$, and $m\angle B = m\angle Y =
74^\circ$, so
\[ m\angle C = 180 - 48 - 74 .\]
Therefore \ans{m\angle Z = 58^\circ}

\problem{A sailmaker cuts two congruent triangular panels with
$\triangle RST \cong \triangle UVW$. The cutting list gives $RS = 12$ m,
$ST = (2n+5)$ m, $RT = 20$ m, and $VW = 17$ m. (a) Find $n$. (b) Find the
perimeter of $\triangle UVW$, in metres.}

\noindent\textbf{Solution (a).} $S$ pairs with $V$ and $T$ pairs with $W$, so
$\overline{ST}$ corresponds to $\overline{VW}$: $2n + 5 = 17$. Subtract $5$:
$2n = 12$, so \ans{n = 6}

\noindent\textbf{Solution (b).} Congruent triangles have equal perimeters, so
add the three sides of $\triangle RST$: $RS = 12$, $ST = 17$, $RT = 20$.
\[ 12 + 17 + 20 = 49 .\]
Therefore \ans{P = 49 \text{ m}}

\problem{A bicycle frame joint is reinforced with two congruent plates,
$\triangle GHJ \cong \triangle KLM$, with $m\angle G = (3x+12)^\circ$ and
$m\angle K = (5x-18)^\circ$. (a) Find $x$. (b) Find $m\angle G$, in degrees.
(c) Explain how you know that $m\angle J = m\angle M$ without measuring either
angle.}

\noindent\textbf{Solution (a).} $G$ pairs with $K$, so
\[ 3x + 12 = 5x - 18 .\]
Subtract $3x$: $12 = 2x - 18$. Add $18$: $30 = 2x$, so \ans{x = 15}

\noindent\textbf{Solution (b).} $m\angle G = 3(15) + 12 = 45 + 12$, so
\ans{m\angle G = 57^\circ}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response says
that the congruence statement pairs vertex $J$ with vertex $M$, so $\angle J$
and $\angle M$ are corresponding angles of congruent triangles; by CPCTC
corresponding parts of congruent triangles are congruent, so the two measures
are equal and neither has to be measured. Award full credit only when both the
correspondence and the CPCTC justification appear.

\problem{Two congruent triangular tent panels satisfy
$\triangle ABC \cong \triangle DEF$. The pattern lists $AB = (4x-7)$ cm,
$DE = (2x+9)$ cm, $BC = (3y+1)$ cm, and $EF = (y+15)$ cm. (a) Find $x$.
(b) Find $y$. (c) Find the length of $\overline{DE}$, in centimetres.}

\noindent\textbf{Solution (a).} $\overline{AB}$ corresponds to $\overline{DE}$:
\[ 4x - 7 = 2x + 9 .\]
Subtract $2x$: $2x - 7 = 9$. Add $7$: $2x = 16$, so \ans{x = 8}

\noindent\textbf{Solution (b).} $\overline{BC}$ corresponds to $\overline{EF}$:
\[ 3y + 1 = y + 15 .\]
Subtract $y$: $2y + 1 = 15$. Subtract $1$: $2y = 14$, so \ans{y = 7}

\noindent\textbf{Solution (c).} $DE = 2(8) + 9 = 16 + 9$, so
\ans{DE = 25 \text{ cm}}

\noindent\textit{Check:} $AB = 4(8) - 7 = 25$ cm, matching $DE$ as required.

\problem{A crane rests on two congruent support triangles with
$\triangle PQR \cong \triangle XYZ$. The specification gives $PQ = (2w+5)$ m,
$XY = 19$ m, $m\angle Q = (7k-4)^\circ$, and $m\angle Y = 66^\circ$.
(a) Find $w$. (b) Find $k$. (c) The engineer claims that knowing only
$PQ = XY$ and $m\angle Q = m\angle Y$ would already guarantee the two triangles
are congruent. Explain why that is not enough, and name one additional
measurement that would settle it.}

\noindent\textbf{Solution (a).} $\overline{PQ}$ corresponds to $\overline{XY}$,
so $2w + 5 = 19$. Subtract $5$: $2w = 14$, so \ans{w = 7}

\noindent\textbf{Solution (b).} $\angle Q$ corresponds to $\angle Y$, so
$7k - 4 = 66$. Add $4$: $7k = 70$, so \ans{k = 10}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response says
that one pair of congruent sides together with one pair of congruent angles is
not a triangle congruence criterion: infinitely many differently shaped
triangles share one side length and one angle measure. The response must then
name a sufficient addition --- a second pair of congruent angles (giving ASA or
AAS), or the second pair of sides that includes the given angle (giving SAS).
Award full credit only when a valid criterion is named.
