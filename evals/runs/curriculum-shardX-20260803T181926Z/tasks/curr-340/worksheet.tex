\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Most problems need \emph{two} theorems. Name each theorem you use beside the line where you use it.
  \item The diagrams show the configuration only --- every measurement you need is in the words of the problem, and the diagrams are not drawn to scale.
  \item Leave $\pi$ in your working and round only at the last step, to the precision the problem asks for.
\end{itemize}

\problem{In circle $O$, points $A$ and $B$ lie on the circle and the central angle
$\angle AOB$ measures 110$^\circ$. Point $C$ lies on the major arc $AB$. Find
$m\angle ACB$.
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.75]
  \draw[thick] (0,0) circle (1.8);
  \coordinate (O) at (0,0);
  \coordinate (A) at (160:1.8);
  \coordinate (B) at (20:1.8);
  \coordinate (C) at (270:1.8);
  \draw[thick] (A) -- (O) -- (B);
  \draw[thick] (A) -- (C) -- (B);
  \fill (O) circle (1.2pt);
  \node[below left] at (O) {$O$};
  \node[above left] at (A) {$A$};
  \node[above right] at (B) {$B$};
  \node[below] at (C) {$C$};
\end{tikzpicture}
\end{center}
