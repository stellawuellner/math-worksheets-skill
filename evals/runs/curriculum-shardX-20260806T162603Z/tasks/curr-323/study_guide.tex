\sstitleblock{Solving Right Triangles: Elevation and Depression}

\skillheading{Skill 1 --- Finding a side from an angle}

\begin{formulabox}
\textbf{Choose the ratio by what you have and what you want} (SOH-CAH-TOA), with
the angle measured from the horizontal:
\[ \sin A = \frac{\text{opp}}{\text{hyp}}, \qquad
   \cos A = \frac{\text{adj}}{\text{hyp}}, \qquad
   \tan A = \frac{\text{opp}}{\text{adj}}. \]
If the unknown is on top, multiply. If the unknown is underneath, divide.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A kite string $30$ m long makes an angle of $55^\circ$ with
level ground. How high is the kite?
\step{Want the height, which is opposite the angle, and I know the string, which
is the hypotenuse --- opposite over hypotenuse is sine.}
\step{$\text{height} = 30 \sin 55^\circ = 24.5745\ldots$}

$\text{height} \approx \ans{24.57}$ m
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} An $8$ m ladder makes an angle of $70^\circ$ with level ground.
How far up the wall does it reach?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7.52}$ m}
\end{tryitbox}

\skillheading{Skill 2 --- Finding an angle from two sides}

\begin{formulabox}
\textbf{Rule.} When two sides are known and the angle is not, use the inverse
function of whichever ratio those two sides form:
\[ A = \tan^{-1}\!\left(\frac{\text{opp}}{\text{adj}}\right), \qquad
   A = \sin^{-1}\!\left(\frac{\text{opp}}{\text{hyp}}\right), \qquad
   A = \cos^{-1}\!\left(\frac{\text{adj}}{\text{hyp}}\right). \]
Keep the calculator in degree mode.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A ramp rises $3$ m over a horizontal run of $20$ m. Find its
angle of elevation, to the nearest tenth of a degree.
\step{Want the angle and I know the rise (opposite) and the run (adjacent) ---
that pair is the tangent, so the angle is the inverse tangent.}
\step{$A = \tan^{-1}(3/20) = 8.5307\ldots^\circ$}

$A \approx \ans{8.5}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A ramp rises $4$ m over a horizontal run of $25$ m. Find its
angle of elevation, to the nearest tenth of a degree.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9.1}$ degrees}
\end{tryitbox}

\skillheading{Skill 3 --- Setting up a depression problem}

\begin{formulabox}
\textbf{Rule.} An angle of \emph{depression} is measured downwards from the
horizontal at the observer. The observer's horizontal line and the ground are
parallel, and the line of sight is a transversal, so
\[ \text{angle of depression at the top} \;=\; \text{angle of elevation at the bottom}. \]
Redraw the depression angle at the \emph{bottom} of the triangle and the problem
becomes an ordinary right triangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} From the top of a $40$ m cliff the angle of depression to a
buoy is $22^\circ$. How far is the buoy from the foot of the cliff?
\step{Move the $22^\circ$ down to the buoy, where it is the angle of elevation;
then the cliff height is opposite it and the distance I want is adjacent ---
tangent, with the unknown underneath, so divide.}
\step{$d = \dfrac{40}{\tan 22^\circ} = 99.0034\ldots$}

$d \approx \ans{99.00}$ m
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} From the top of a $60$ m cliff the angle of depression to a
buoy is $35^\circ$. How far is the buoy from the foot of the cliff?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{85.69}$ m}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the angle of depression is never the angle at the top of
the triangle between the vertical and the line of sight --- that angle is its
complement. Measure from the horizontal, every time.
\end{watchoutbox}
