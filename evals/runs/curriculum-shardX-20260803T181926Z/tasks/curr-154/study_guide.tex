\sstitleblock{Reversed and Non-Equivalent Ratios}

\skillheading{Skill 1 --- The order is part of the answer}
\begin{formulabox}\textbf{Rule:} ``the ratio of $A$ to $B$'' means $\dfrac{A}{B}$ ---
the quantity named FIRST goes on top, whatever order the numbers appeared in the
sentence.\par
\textbf{Sense check:} if the top quantity is the bigger one, the ratio must come out
greater than 1. A ratio below 1 when you expected ``more'' is a reversed ratio.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad A shelter has $10$ dogs and $15$ cats. Write
the ratio of cats to dogs.
\step{Cats are named first, so cats go on top even though dogs were counted first in the sentence.}
\step{$\dfrac{15}{10}$, and dividing both parts by 5 gives $\ans{\dfrac{3}{2}}$ --- greater than 1, as it should be with more cats.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad A box holds $9$ red pens and $12$ blue pens. Write
the ratio of blue to red in simplest form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{4}{3}}$}\end{tryitbox}

\skillheading{Skill 2 --- Scale both parts by the same number}
\begin{formulabox}\textbf{Rule:} to build an equivalent ratio, multiply (or divide)
\emph{both} parts by the same number. Find the factor from the part you know:
factor $=$ new value $\div$ old value.\par
\textbf{Adding does not work.} $2 : 3$ and $4 : 5$ are not equivalent, even though each
part grew by 2.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad A table starts at $3 : 5$. The first part
becomes $12$. What is the second part?
\step{Find the factor from the part that changed, then apply that same factor to the other part.}
\step{$12 \div 3 = 4$, so the second part is $5 \times 4 = \ans{20}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad A table starts at $4 : 7$. The first part becomes
$20$. What is the second part?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}\end{tryitbox}

\skillheading{Skill 3 --- Testing two ratios for equivalence}
\begin{formulabox}\textbf{Three tests, all equivalent:} simplify both and compare;
compare the unit rates (divide top by bottom); or cross multiply and see whether the two
products match. If any test says no, the ratios are not equivalent.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Are $6 : 9$ and $10 : 15$ equivalent?
\step{Both parts of each ratio share a factor, so simplifying is the quickest of the three tests here.}
\step{$\dfrac{6}{9} = \dfrac{2}{3}$ and $\dfrac{10}{15} = \dfrac{2}{3}$, so $\ans{=}$ --- they are equivalent.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $<$, $=$, or $>$ between $\dfrac{8}{12}$ and
$\dfrac{10}{16}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}\end{tryitbox}
