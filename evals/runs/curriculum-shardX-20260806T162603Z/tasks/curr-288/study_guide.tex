\sstitleblock{Differences of Squares and Perfect-Square Trinomials}

\skillheading{Skill 1 --- Factoring a difference of two squares}

\begin{formulabox}
\textbf{Rule:} \quad $a^2 - b^2 = (a-b)(a+b)$\par\smallskip
\textbf{How to spot it:} exactly \emph{two} terms, both perfect squares, joined by a \emph{minus} sign. Find $a$ and $b$ by taking the square root of each term: $\sqrt{9x^2} = 3x$, $\sqrt{49} = 7$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^2 - 81$.
\step{Two terms, both perfect squares, with a minus between them --- that is the difference-of-squares pattern, so no trinomial search is needed.}
\step{$a = \sqrt{x^2} = x$ and $b = \sqrt{81} = 9$, so the factors are $(a-b)$ and $(a+b)$.}
\par
$x^2 - 81 = $ $\ans{(x-9)(x+9)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $4x^2 - 25$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2x-5)(2x+5)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a \emph{sum} of two squares such as $x^2 + 16$ does \emph{not} factor over the whole numbers. The minus sign is part of the pattern.
\end{watchoutbox}

\skillheading{Skill 2 --- Factoring a perfect-square trinomial}

\begin{formulabox}
\textbf{Rules:} \quad $a^2 + 2ab + b^2 = (a+b)^2$ \qquad $a^2 - 2ab + b^2 = (a-b)^2$\par\smallskip
\textbf{How to spot it:} \emph{three} terms; the first and last are perfect squares; and the middle term equals $2ab$. The middle-term test is the one that decides it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^2 - 14x + 49$.
\step{Three terms with square ends suggests a perfect square, so name $a$ and $b$ and then test the middle term before committing.}
\step{$a = x$, $b = 7$, and $2ab = 2 \cdot x \cdot 7 = 14x$, which matches the middle term. The middle sign is negative, so use $(a-b)^2$.}
\par
$x^2 - 14x + 49 = $ $\ans{(x-7)^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $9x^2 + 30x + 25$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3x+5)^2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} square ends are not enough. $x^2 + 12x + 36$ is a perfect square because $2 \cdot x \cdot 6 = 12x$; $x^2 + 13x + 36$ is not. Always run the middle-term test.
\end{watchoutbox}

\skillheading{Skill 3 --- Common factor first, then the pattern}

\begin{formulabox}
\textbf{Rule:} \quad factor out the greatest common factor before looking for any special form, then factor what is left. ``Factor completely'' means keep going until nothing inside the parentheses factors again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $8x^2 - 18$ completely.
\step{Neither term is a perfect square on its own, but both are even --- take the common factor out first and look again.}
\step{$8x^2 - 18 = 2(4x^2 - 9)$, and inside the parentheses $4x^2 = (2x)^2$, $9 = 3^2$, a difference of squares.}
\par
$8x^2 - 18 = $ $\ans{2(2x-3)(2x+3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $5x^2 - 40x + 80$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5(x-4)^2}$}
\end{tryitbox}
