\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Every problem reads from the Beaker Plot below. Each $\times$ stands for one beaker.
  \item Write fractions in lowest terms, and write amounts greater than one as a mixed number.
  \item Show the addition or subtraction you did before writing the answer.
\end{itemize}

\problem[2.6cm]{How many beakers hold exactly $\dfrac{3}{8}$ of a litre?}

\problem[2.6cm]{How many beakers are recorded on the plot altogether?}

\problem[2.6cm]{Which amount was recorded more often than any other?}

\problem[2.8cm]{The two beakers at $\dfrac{5}{8}$ of a litre are emptied into one jug. How much water is in the jug?
\answerline{L}}

\problem[2.8cm]{What is the difference between the greatest amount and the least amount on the plot?
\answerline{L}}

\problem[3.2cm]{How much water is held altogether by the beakers whose amounts are less than $\dfrac{1}{2}$ of a litre?
\answerline{L}}

\problem[3.2cm]{The four beakers at $\dfrac{3}{8}$ of a litre are emptied into one jug and then shared equally between 3 empty beakers. How much water is in each of those 3 beakers?
\answerline{L}}

\problem[3.2cm]{What is the median of the 16 recorded amounts?
\answerline{L}}

\problem[4cm]{How much water is in the science cupboard altogether?
\answerline{L}}

\problem[3.4cm]{How much more water is held by all the beakers at $\dfrac{1}{2}$ of a litre together than by all the beakers at $\dfrac{1}{4}$ of a litre together?
\answerline{L}}

\problem[2.8cm]{How many beakers hold more than $\dfrac{1}{2}$ of a litre?}

\problem[3.4cm]{One more beaker, holding $\dfrac{7}{8}$ of a litre, is found and added to the plot.

  \item[(a)] What is the median of the 17 amounts now? \answerline{L}
  \item[(b)] Explain why adding the largest amount on the plot did not move the median very far.
