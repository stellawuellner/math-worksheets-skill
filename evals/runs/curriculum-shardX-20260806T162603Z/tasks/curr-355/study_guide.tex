% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Square Roots of Negative Numbers}

\begin{formulabox}
\textbf{Rule:} $\sqrt{-k} = i\sqrt{k}$ for $k > 0$, and a square has
\textit{two} roots: $x^{2} = -k$ gives $x = \pm i\sqrt{k}$.\\[3pt]
Isolate the square first; divide out any leading coefficient before taking
roots.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^{2} + 36 = 0$
\step{The square is already alone on one side after one move, so there is no need for the quadratic formula --- isolate and take both roots.}
\step{$x^{2} = -36$, and the two square roots of $-36$ are $6i$ and its negative.}

$\Rightarrow\quad \ans{x = 6i \text{ or } x = -6i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2x^{2} + 98 = 0$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7i \text{ or } x = -7i}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- When the Discriminant Is Negative}

\begin{formulabox}
\textbf{Formula:}
\[
  x = \frac{-b \pm \sqrt{b^{2} - 4ac}}{2a}
\]
When $b^{2} - 4ac < 0$ the two solutions are non-real conjugates. Divide
\textit{both} parts of the numerator by $2a$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^{2} - 6x + 25 = 0$
\step{There is no factor pair for $25$ that sums to $-6$, so go to the quadratic formula and read the discriminant first.}
\step{The discriminant is $36 - 100 = -64$, so the numerator is $6 \pm 8i$; dividing both parts by $2$ gives real part $3$ and imaginary part $4$.}

$\Rightarrow\quad \ans{x = 3 \pm 4i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^{2} + 2x + 26 = 0$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -1 \pm 5i}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- All Solutions of a Higher Degree}

\begin{formulabox}
\textbf{Rule:} an equation of degree $n$ has exactly $n$ solutions, counted with
multiplicity. Count what you have before you stop.\\[3pt]
Factor into quadratics, then solve each one --- a factor like $x^{2} + 4$
contributes a non-real pair, and it is the factor most often abandoned.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^{4} - 16 = 0$
\step{Degree four means four solutions, so factoring as far as possible is the way to find all of them.}
\step{$(x^{2} - 4)(x^{2} + 4) = 0$: the first factor gives the real pair, and the second gives $x^{2} = -4$, whose roots are imaginary.}

$\Rightarrow\quad \ans{x = 2,\ -2,\ 2i,\ -2i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^{4} - 5x^{2} - 36 = 0$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3,\ -3,\ 2i,\ -2i}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Checking and Reporting the Full Set}

\begin{formulabox}
\textbf{To check a solution:} substitute it in and simplify. If the expression
comes out $0$, the value is a solution.\\[3pt]
\textbf{Conjugate pairs:} when every coefficient is real, non-real solutions
come in pairs $a + bi$ and $a - bi$ --- so finding one hands you the other.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $x = 2 - i$ a solution of $x^{2} - 4x + 5 = 0$?
\step{Checking a proposed solution means substituting it and simplifying, not re-solving the equation.}
\step{$(2 - i)^{2} = 3 - 4i$ and $-4(2 - i) = -8 + 4i$; adding those with $5$, the imaginary parts cancel and the real parts give $3 - 8 + 5$.}

$\Rightarrow\quad \ans{0}$, so yes --- and $2 + i$ is the other solution.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Substitute $x = 3 + i$ into $x^{2} - 6x + 10$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0}$}
\end{tryitbox}
