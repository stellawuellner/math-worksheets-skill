\begin{center}
\textit{Adding, subtracting and multiplying polynomials, from a worked pattern to independent fluency}
\end{center}

\vspace{0.1cm}
\noindent\textbf{The pattern to copy}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Adding:} drop the brackets, then collect like terms.
        $(x^2 + 4x) + (3x^2 - x) = x^2 + 3x^2 + 4x - x = 4x^2 + 3x$.
  \item \textbf{Subtracting:} change \emph{every} sign in the second bracket first, then collect.
        $(x^2 + 4x) - (3x^2 - x) = x^2 + 4x - 3x^2 + x = -2x^2 + 5x$.
  \item \textbf{Multiplying:} every term of the first bracket meets every term of the second, then collect.
  \item Write each answer in standard form --- highest power first, no like terms left.
\end{itemize}

\problem[5cm]{Subtract $(2x^2 - 7x + 1)$ from $(6x^2 + x - 4)$.
Read the order carefully: the polynomial that comes after ``from'' is the one you start with.}

\problem[6cm]{Multiply and simplify: \quad $(x + 4)(x^2 - 3x + 2)$
\par\smallskip
Each of the two terms in the first bracket has to meet all three terms in the second, so expect six
products before you collect like terms.}

\problem[6cm]{Simplify: \quad $(2x + 3)^2 - (x - 1)(x + 1)$
\par\smallskip
Expand each product first, then subtract --- and remember that the whole second product is being
subtracted.}
