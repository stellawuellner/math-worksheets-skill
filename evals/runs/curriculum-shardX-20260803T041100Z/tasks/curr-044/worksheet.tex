\wstitleblock{Gaps and Overlaps When We Measure}{Kindergarten--Grade 1}{}

\begin{center}\small\textit{Laying units with no gaps or overlaps, spotting a gap or overlap
mistake, and fixing the count}\end{center}

\noindent\small\textbf{Directions:} Every child on this page measured with the
\textbf{same size cubes}.
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Some children laid their cubes the \textbf{good} way. Some left \textbf{gaps}.
        Some let the cubes \textbf{overlap}.
  \item Look at the picture first. Then answer with a number, a symbol, or a drawing.
  \item When a question says \emph{find the mistake}, tell what the child did wrong
        \emph{and} write the right answer.
\end{itemize}
\normalsize

\howtolay

\noindent\small\textit{How cubes are laid down. No numbers are shown here --- count the
cubes in each problem's own picture.}\normalsize

\problem{Ivy laid her cubes under the red ribbon. There are \textbf{no gaps} and
\textbf{no overlaps}. Count Ivy's cubes. How long is the ribbon?
\ivypic\par\vspace*{3cm}
\answerline{cubes}}

\problem[3cm]{Max measured the \textbf{same} red ribbon. He left gaps, so he counted only
6 cubes. Ivy measured it carefully and got 8 cubes.
\maxpic
Write $<$ or $>$ in the box to tell the truth about Max's number:\\[8pt]
\hspace{1.2cm}\large Max's count \symbolbox\ real length \noansline}

\newpage

\problem{\textbf{Find the mistake.} Rex measured the \textbf{same} red ribbon. He
let his cubes sit on top of each other, counted 10, and said ``the ribbon is 10 cubes
long.''
\rexpic
(a) Tell Rex what he did wrong. \quad (b) Write the ribbon's real length.\par\vspace*{3.5cm}
\answerline{cubes}}

\problem{Rex counted 10 cubes. Ivy counted 8 cubes. They measured the same ribbon.
How many \textbf{more} cubes did Rex count than Ivy?\par\vspace*{3.5cm}
\answerline{cubes}}

\newpage

\problem{\textbf{Find the mistake.} Sam measured a yellow straw. He left gaps
between his cubes and counted 6. A careful measure shows the straw is really 10 cubes
long.
\sampic
(a) Tell Sam what he did wrong. \quad (b) How many cubes did Sam's gaps hide?\par\vspace*{3.5cm}
\answerline{cubes}}

\problem[6cm]{Rex says: ``I counted 10 cubes and Ivy only counted 8, so my measuring is
better.''

\smallskip
Is Rex right? Draw a picture of the ribbon and the cubes, and tell how you know.
\noansline}
