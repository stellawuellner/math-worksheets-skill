\wstitleblock{Scatter Plots and Bivariate Association --- Direction, Form, and Strength}{Grade 8 / Pre-Algebra}{}

\begin{center}\small\textit{Describing a scatter plot in three words: which way it goes,
what shape it makes, and how tightly the points hug the trend}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Six scatter plots, \textbf{Plot A} to \textbf{Plot F}, are printed below. Every
        problem names the plot it is about.
  \item A complete description of a scatter plot has three parts:
        \textbf{direction} (positive, negative, or none), \textbf{form} (linear or
        nonlinear), and \textbf{strength} (how close the points sit to the trend).
  \item Each problem asks you to compute something first and describe second. The number
        is the evidence; the description is the answer.
  \item ``Residual'' means \emph{actual $y$ minus predicted $y$}. A residual close to zero
        means that point sits close to the line.
\end{itemize}
\normalsize

\problem[6cm]{\textbf{Plot A --- which way does it go?}

\smallskip
(a) Write the $y$-value of the leftmost point and the $y$-value of the rightmost point,
then put $>$, $<$, or $=$ between them.

\smallskip
leftmost~\ansblank \quad \underline{\hspace{1.2cm}} \quad rightmost~\ansblank

\smallskip
(b) State the direction of the association in Plot A, and say how your comparison in (a)
supports it.}

\problem[6cm]{\textbf{Plot B --- the other direction.}

\smallskip
(a) Do the same thing: leftmost $y$-value, rightmost $y$-value, and the correct symbol
between them.

\smallskip
leftmost~\ansblank \quad \underline{\hspace{1.2cm}} \quad rightmost~\ansblank

\smallskip
(b) State the direction, and complete this sentence: ``as $x$ increases, $y$ tends
to \underline{\hspace{2.5cm}}''.}

\problem[7cm]{\textbf{Plot D --- what shape do the points make?}

\smallskip
(a) Which $x$-value carries the \emph{lowest} $y$-value in Plot D?

\smallskip
(b) Follow the points from left to right. Describe in words what they do, then say whether
the form is \textbf{linear} or \textbf{nonlinear}.}

\problem[8cm]{\textbf{Plots A and E --- measuring how far a point sits from the trend.}
The residual of a point is $\text{actual } y - \text{predicted } y$.

\smallskip
(a) In Plot A the point at $x = 5$ has $y = 73$, and the trend line $y = 5x + 47$ predicts
$5(5) + 47$. Find the residual.

\smallskip
(b) In Plot E the point at $x = 5$ has $y = 44$, and the trend line $y = 5x + 25$ predicts
$5(5) + 25$. Find the residual.

\smallskip
Plot A residual~\ansblank \qquad Plot E residual~\ansblank

\smallskip
(c) Which of the two points sits closer to its trend line, and how do you know from the two
numbers?}

\problem[7cm]{\textbf{Plot C --- when there is no direction at all.}

\smallskip
(a) Which $x$-value carries the highest $y$-value? \quad (b) Which carries the lowest?

\smallskip
highest at $x = $~\ansblank \qquad lowest at $x = $~\ansblank

\smallskip
(c) On Plot A the highest and lowest $y$-values sat at opposite ends of the plot. Where do
they sit here? Use that to explain why Plot C has \textbf{no association}.}

\problem[8cm]{\textbf{Plot F --- one point that does not belong.}

\smallskip
(a) Which $x$-value carries the lowest $y$-value in Plot F? (Careful: the question asks
where the lowest \emph{height} is, not the smallest $x$.)

\smallskip
(b) Cover that point with your finger. Now describe the direction, form and strength of
what is left.

\smallskip
(c) A point like this is called an \textbf{outlier}. Explain in one sentence what it does
to your judgement of the strength of Plot F.}

\problem[9cm]{\textbf{Plots A and E --- putting a number on strength.} Here are the
residuals of all eight points on each plot, measured from the trend lines printed with the
figures.

\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|c|c|c|c|c|c|}
\hline
\textbf{Plot A residuals} & $0$ & $1$ & $-1$ & $0$ & $1$ & $-1$ & $0$ & $1$ \\ \hline
\textbf{Plot E residuals} & $5$ & $6$ & $-7$ & $7$ & $-6$ & $8$ & $-5$ & $6$ \\ \hline
\end{tabular}
\end{center}
\par\smallskip
(a) Find the range of each list of residuals (largest minus smallest).

\smallskip
Plot A range~\ansblank \qquad Plot E range~\ansblank

\smallskip
(b) Both plots have a positive linear association. Use your two ranges to say which one is
\textbf{stronger}, and explain what the range of the residuals is actually measuring.}

\problem[8cm]{\textbf{Plot D again --- why ``direction'' is the wrong question here.}

\smallskip
(a) What is the highest $y$-value anywhere on Plot D?

\smallskip
(b) A student argues: ``the last point is higher than the first point, so Plot D shows a
positive association.'' The student's evidence is true. Explain why the conclusion is
still wrong, using the word \textbf{form}.}

\problem[9cm]{\textbf{Plot F --- what one outlier does to a summary.}

\smallskip
(a) Find the mean of all eight $y$-values in Plot F: $20, 26, 31, 37, 42, 48, 10, 59$.

\smallskip
(b) Now remove the outlier and find the mean of the remaining seven $y$-values. Round to
the nearest hundredth.

\smallskip
with outlier~\ansblank \qquad without~\ansblank

\smallskip
(c) By how much did one point out of eight move the mean? Say what that shows about
trusting a summary from a small data set.}

\problem[10cm]{\textbf{Challenge.} Write two complete descriptions, one for \textbf{Plot A}
and one for \textbf{Plot E}. Each description must use all three words --- direction, form,
strength --- and must give the evidence for each.

\smallskip
Then judge this claim: \emph{``Plot E shows a strong link between the two quantities.''}
Say whether you agree, and support your answer with the residual numbers from problem 7
rather than with the slope of the line.}
