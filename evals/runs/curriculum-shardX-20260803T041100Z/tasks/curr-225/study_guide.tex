\sstitleblock{Linear Contexts, Tables, Equations, and Graphs}

\noindent\small One relationship, four faces. Whatever form you are handed,
you are always hunting the same two numbers: the \textbf{rate of change} and
the \textbf{initial value}.\normalsize

\skillheading{1. Reading values and changes off a table or graph}

\begin{formulabox}
\textbf{From a table:} each column is one $(x, y)$ pair. A relationship is
linear when equal steps in $x$ produce equal steps in $y$ --- check that
before you trust any prediction.\\[3pt]
\textbf{From a graph:} only read points where the line crosses a grid
\emph{intersection}; an estimated point produces a wrong slope that looks
plausible.\\[3pt]
\textbf{Change vs.\ value:} ``what is $y$'' asks for one entry; ``how much did
$y$ change'' asks for a subtraction between two entries.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A table gives $x$: $0$, $2$, $4$, $6$ \; and
$y$: $5$, $11$, $17$, $23$. \; How much does $y$ change from $x = 0$ to
$x = 4$?
\step{The question asks for a change, not a value, so read two entries and subtract instead of hunting for a single cell.}
\step{At $x = 4$ the table gives $y = 17$; at $x = 0$ it gives $y = 5$. The change is $17 - 5$.}
change in $y = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A table gives $x$: $0$, $3$, $6$, $9$ \; and
$y$: $40$, $34$, $28$, $22$. \; How much does $y$ change from $x = 0$ to
$x = 6$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-12}$}
\end{tryitbox}

\skillheading{2. Rate of change from any two points}

\begin{formulabox}
\textbf{Rule:} for any two points $(x_1, y_1)$ and $(x_2, y_2)$ of a linear
relationship,
\[ \text{rate of change} = m = \frac{y_2 - y_1}{x_2 - x_1} \]
\textbf{Sign check first:} rising line, positive rate; falling line, negative
rate. Do this before dividing --- it catches a swapped subtraction instantly.\\[3pt]
Keep both differences in the same order: top $y_2$ with bottom $x_2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A line passes through $(1, 4)$ and $(5, 16)$. Find its
rate of change.
\step{Two points of one linear relationship are all the slope formula needs, so no equation or graph is required here.}
\step{$\dfrac{16 - 4}{5 - 1} = \dfrac{12}{4}$, and the line rises, so a positive answer is expected.}
$m = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A line passes through $(2, 9)$ and $(6, 1)$. Find its
rate of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$}
\end{tryitbox}

\skillheading{3. Building the equation and using it both ways}

\begin{formulabox}
\textbf{Rule:} \quad $y = mx + b$, \; where $m$ is the rate of change and $b$
is the initial value (the $y$ when $x$ is zero).\\[3pt]
\textbf{In a story:} a one-time amount (joining fee, starting height) is $b$;
an ``each'' or ``per'' amount is $m$.\\[3pt]
\textbf{Forwards} --- given $x$, substitute and evaluate.\\
\textbf{Backwards} --- given $y$, set the expression equal to it and solve
for $x$.\\[3pt]
If the table has no $x = 0$ row, work back one step at a time to reach it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A pool holds water $30$ cm deep and is filling at
$4$ cm per hour. How deep is it after $7$ hours?
\step{The depth it already has is the initial value and the ``per hour'' amount is the rate, so the story hands over both numbers directly.}
\step{The equation is $h = 4t + 30$; substituting $t = 7$ gives $4(7) + 30 = 28 + 30$.}
$h = \ans{58}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A second pool holds water $25$ cm deep and fills at
$6$ cm per hour. How deep is it after $5$ hours?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{55}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} when a table steps by more than one, the change in $y$
is \emph{not} the rate of change --- you must still divide by the change in
$x$. And a rate read off a falling line must come out negative.
\end{watchoutbox}
