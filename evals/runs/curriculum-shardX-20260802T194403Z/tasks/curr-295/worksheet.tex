\wstitleblock{Solving Quadratic Equations: Choose a Method, Read the Roots}{Algebra 1}{}

\noindent\textbf{Focus:} Choosing among square roots, factoring, and the
quadratic formula, and interpreting roots in context.

\vspace{4pt}
\noindent\textbf{Before you solve, choose.} These problems are mixed on purpose.
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Square roots} --- when the variable appears only inside one
        squared quantity and there is no linear term: $ax^2 = k$ or $a(x-h)^2 = k$.
  \item \textbf{Factoring} --- when the quadratic factors over the integers
        (always check for a common factor first).
  \item \textbf{Quadratic formula} --- when it does not factor, or when you
        cannot tell quickly. It always works.
\end{itemize}
Write the method you picked on the \textbf{Method} line, then solve. In the word
problems, finish by saying what the roots mean for the situation.

\problem[4.6cm]{A ball is thrown straight up from ground level. Its height in
feet after $t$ seconds is $h = -16t^2 + 48t$. Solve $h = 0$ to find every time
the ball is at ground level, then say how long the ball is in the air.
\par Method: \underline{\hspace{5cm}}
\answerline{seconds}}

\par Method: \underline{\hspace{5cm}}
\answerline{m}}

\problem[5.2cm]{A ball leaves a player's hand $5$ feet above the ground moving
upward at $40$ feet per second. Its height in feet after $t$ seconds is
$h = -16t^2 + 40t + 5$. Find the time when the ball lands ($h = 0$), rounded to
the nearest hundredth of a second, and explain why the other root is thrown out.
\par Method: \underline{\hspace{5cm}}
\answerline{seconds}}

\problem{\textbf{Synthesis.} A club's profit in dollars from selling $n$
sweatshirts is $P = -n^2 + 30n - 189$. \par
Method: \underline{\hspace{5cm}}
\begin{enumerate}[label=(\alph*), leftmargin=1cm, itemsep=3cm, topsep=4pt]
  \item Break-even means $P = 0$. Solve $-n^2 + 30n - 189 = 0$ for $n$. \ansline
  \item The manager says selling $20$ sweatshirts breaks even. Use your roots to
        say whether she is right, and describe what happens to the profit for
        numbers of sweatshirts between the two break-even values. \ansline
  \vspace{2cm}
\end{enumerate}}
