\sstitleblock{Unit Rates with Whole Numbers}

% =================== SKILL 1 ===================
\skillheading{Finding a Unit Rate: Divide Down to One}

\begin{formulabox}
\textbf{Rule:} \quad unit rate $=$ total amount $\div$ number of units.\par
The word \textbf{per} names the unit you divide by: dollars \emph{per book}
means divide by books; miles \emph{per hour} means divide by hours.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 6 workbooks cost 42 dollars. Find the price per
workbook.
\step{``Per workbook'' says to divide the total cost by the number of
workbooks.}
\step{$42 \div 6 = 7$, so the unit price is $\ans{7}$ dollars per workbook.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 8 granola bars cost 48 dollars at a fancy shop. How many
dollars per bar?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Keep the order straight — miles per hour is miles
$\div$ hours. Dividing the other way gives hours per mile, a different (and
usually tiny) number.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Using a Unit Rate: Multiply Back Up}

\begin{formulabox}
\textbf{Rule:} Once you know the rate for \textbf{one} unit, any amount is one
multiplication: \quad total $=$ unit rate $\times$ number of units.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Stickers cost 9 cents each. How much do 6 stickers
cost?
\step{The unit rate (9 cents per sticker) is given, so multiply by the number
of stickers.}
\step{$9 \times 6 = 54$, so 6 stickers cost $\ans{54}$ cents.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A ferry ticket costs 7 dollars per person. How many
dollars do 8 people pay?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Comparing Two Rates Fairly}

\begin{formulabox}
\textbf{Rule:} To compare two deals or speeds with \textbf{different} sizes,
bring both to a unit rate first, then compare the unit rates. Smaller is better
for prices; bigger is better for speeds and distances.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Bag P holds 4 muffins for 24 dollars; bag Q holds 6
muffins for 30 dollars. Which costs less per muffin?
\step{Unit prices: $24 \div 4 = 6$ dollars per muffin, and $30 \div 6 = 5$
dollars per muffin.}
\step{Comparing them, bag Q is cheaper per muffin: $\ans{6 > 5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Runner X covers 18 km in 3 hours; runner Y covers 20 km
in 4 hours. Compare their km-per-hour fractions with $<$ or $>$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{18}{3} > \tfrac{20}{4}}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Unit rate:] the amount for exactly one unit
  \item[Unit price:] cost of one item — a unit rate for money
  \item[Per:] ``for each one''; names what you divide by
  \item[Rate table:] a ratio table with a column for one unit
\end{description}
\end{multicols}
