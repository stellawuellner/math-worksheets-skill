\wstitleblock{Take-From and Comparison Stories Within Ten}{Kindergarten--Grade 1}{}

\begin{center}\small\textit{Taking objects away, and comparing two groups to see
how many more or how many fewer}\end{center}

\noindent\textbf{Two kinds of story, one subtraction.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Take from:} you start with a group, some go away, and you count
        what is left. \ start $-$ went away $=$ left.
  \item \textbf{Compare:} two groups sit side by side and you match them up.
        The leftovers tell you how many \textbf{more} the big group has, and how
        many \textbf{fewer} the small group has. \ bigger $-$ smaller $=$ difference.
  \item Always say the unit out loud with your answer: not just ``four'', but
        ``four crayons''.
\end{itemize}
\noindent\rule{\linewidth}{0.4pt}

\problem[3cm]{\textbf{Take from.} You have \textbf{6} crayons. You give
\textbf{2} crayons to a friend. Cross out the crayons you gave away.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.6cm]\hline
\end{tabular}
\end{center}
How many crayons are left? \quad $6 - 2 = $~\ansblank\ crayons}

\problem[3cm]{\textbf{Take from.} There are \textbf{9} stickers on a page. Ana
uses \textbf{4} stickers. Cross out the stickers Ana uses.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|c|c|c|c|}\hline
$\star$ & $\star$ & $\star$ & $\star$ & $\star$ & $\star$ & $\star$ & $\star$ & $\star$ \\[0.6cm]\hline
\end{tabular}
\end{center}
How many stickers are left? \quad $9 - 4 = $~\ansblank\ stickers}

\problem[3.5cm]{\textbf{Compare.} The top row shows \textbf{7} red cubes. The

How many \textbf{more} red cubes than blue cubes are there? Write the
subtraction that answers the question.}

\problem[4cm]{\textbf{Take from.} A plate holds \textbf{8} grapes. Leo eats
\textbf{3} of the grapes.

\noindent Draw the grapes as dots. Cross out the ones Leo eats. Then write the
subtraction equation and answer: how many grapes are still on the plate?}

\problem[4cm]{\textbf{Compare.} At the pond there are \textbf{9} frogs and
\textbf{5} ducks.

\noindent Draw a row of frogs and a row of ducks, one under the other, and match
them up. How many \textbf{fewer} ducks than frogs are there? Write the
subtraction equation and the answer.}

\problem[3.5cm]{\textbf{Take from, missing part.} A jar holds \textbf{10}
buttons. Sam takes some buttons out. Now \textbf{6} buttons are in the jar.

\noindent How many buttons did Sam take out? \quad
$10 - $~\ansblank$\ = 6$ \ buttons}

\problem[3.5cm]{\textbf{Compare, missing part.} Sam has \textbf{9} shells. Sam
has \textbf{3} more shells than Tia.

\noindent How many shells does Tia have? Draw Sam's shells, circle the 3 extra
ones, and count what is left.}

\problem[4.5cm]{\textbf{Challenge --- two steps.} A bowl holds \textbf{8} apples.
The family eats \textbf{3} apples. There are also \textbf{2} pears in the bowl.

\noindent \textbf{(a)} How many apples are left in the bowl?

\noindent \textbf{(b)} Now how many \textbf{more} apples than pears are in the
bowl? Write both subtraction equations.}
