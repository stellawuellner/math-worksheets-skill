\wstitleblock{Integration by Parts: Choosing the Pieces}{AP Calculus AB/BC}{}

\begin{center}
\small Single and repeated integration by parts, and parts with logarithms and inverse trig
\end{center}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item $\displaystyle\int u\,dv = uv - \int v\,du$. Choose $u$ so that
        differentiating it makes the problem \emph{simpler}, and $dv$ so that
        integrating it does not make things worse.
  \item Write your $u$, $du$, $v$ and $dv$ before you start. No two problems
        below use the same skeleton.
  \item Indefinite answers need $+\,C$. Give exact values throughout.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Evaluate $\displaystyle\int x e^{x}\,dx$.}

\problem[5.5cm]{Evaluate $\displaystyle\int \ln x\,dx$. (There is only one
factor visible --- take $dv = dx$.)}

\problem[5.5cm]{Evaluate $\displaystyle\int x\sin x\,dx$.}

\problem[6cm]{Evaluate $\displaystyle\int x\ln x\,dx$.}

\problem[6.5cm]{Evaluate $\displaystyle\int_0^1 x e^{2x}\,dx$. Give an exact
value.}

\problem{Consider $\displaystyle\int x^2 e^{x}\,dx$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Evaluate it.
    \par\vspace*{4.2cm}\ansline
  \item[(b)] Explain why parts has to be applied twice here.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[6.5cm]{Evaluate $\displaystyle\int \arctan x\,dx$. (Again there is only
one factor --- take $dv = dx$.)}

\problem[7cm]{Evaluate $\displaystyle\int e^{x}\sin x\,dx$. Apply parts twice
and solve for the integral that comes back.}

\problem[6cm]{Evaluate $\displaystyle\int_0^{\pi} x\cos x\,dx$.}

\problem{Consider $\displaystyle\int_0^{b} x e^{-x}\,dx$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Evaluate it for $b = 3$. Give an exact value.
    \par\vspace*{3.6cm}\ansline
  \item[(b)] In general the integral equals $1 - (b + 1)e^{-b}$. Find its limit
    as $b \rightarrow \infty$.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Explain why taking $u = x$ rather than $u = e^{-x}$ is the
    strategic choice here.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}
