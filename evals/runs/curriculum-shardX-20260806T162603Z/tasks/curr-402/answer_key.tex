\problem{$f(x) = x^{2} + 1$, $g(x) = 3x - 2$. Find $(f \circ g)(4)$.}

\textbf{Solution:} Inner function first: $g(4) = 3(4) - 2 = 10$, then $f(10) = 10^{2} + 1 = 101$. \ans{(f \circ g)(4) = 101}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$p(x) = 2x + 5$, $q(x) = x^{2} - 3$. Write $(p \circ q)(x)$.}

\textbf{Solution:} Substitute $q(x)$ into every $x$ of $p$: $p(q(x)) = 2\left(x^{2} - 3\right) + 5 = 2x^{2} - 6 + 5$. \ans{(p \circ q)(x) = 2x^{2} - 1}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$u(x) = x - 4$, $v(x) = x^{2}$. Solve $(v \circ u)(x) = 49$.}

\textbf{Solution:} $(v \circ u)(x) = (x - 4)^{2}$, so
\begin{align*}
  (x - 4)^{2} &= 49 \\
  x - 4 &= \pm 7
\end{align*}
Both signs give a solution: $x = 11$ and $x = -3$.
\[ \boxed{x = -3 \quad \text{or} \quad x = 11} \]

\problem{$g(x) = x^{2} - 5$ and $f$ piecewise ($2x+1$ for $x<3$; $x^{2}-4$ for $x \geq 3$). \textbf{(a)} $(f \circ g)(4)$. \textbf{(b)} Which rule and why.}

\textbf{Solution (a):} $g(4) = 4^{2} - 5 = 11$. Since $11 \geq 3$, the second rule applies: $f(11) = 11^{2} - 4 = 117$. \ans{(f \circ g)(4) = 117}

\textbf{Solution (b):} \emph{Instructor-judged.} The response must name the inner value $11$ and compare it with the boundary $3$: because $11 \geq 3$, the rule $x^{2} - 4$ is the one that applies. Naming the rule without the comparison is only half the answer; using $2x + 1$ means the student tested the original input $4$ instead of the value actually being fed into $f$.

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \dfrac{2}{x - 1}$, $g(x) = \dfrac{1}{x}$. Write $(f \circ g)(x)$.}

\textbf{Solution:} Substitute and clear the compound fraction by multiplying top and bottom by $x$:
\[ f\!\left(\tfrac{1}{x}\right) = \frac{2}{\tfrac{1}{x} - 1} = \frac{2x}{1 - x} \]
\ans{(f \circ g)(x) = \dfrac{2x}{1 - x}}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \sqrt{x + 4}$, $g(x) = 2x - 1$. Solve $(f \circ g)(x) = 5$.}

\textbf{Solution:} $(f \circ g)(x) = \sqrt{(2x - 1) + 4} = \sqrt{2x + 3}$, so
\begin{align*}
  \sqrt{2x + 3} &= 5 \\
  2x + 3 &= 25 \\
  2x &= 22
\end{align*}
Check: $\sqrt{2(11) + 3} = \sqrt{25} = 5$ $\checkmark$. \ans{x = 11}

\problem{$f(x) = 3x - 2$ and $g$ piecewise ($-x$ for $x \leq 0$; $x^{2}$ for $x > 0$). \textbf{(a)} $(f \circ g)(-5)$. \textbf{(b)} $(g \circ f)(-5)$.}

\textbf{Solution (a):} $-5 \leq 0$, so $g(-5) = -(-5) = 5$; then $f(5) = 3(5) - 2 = 13$. \ans{(f \circ g)(-5) = 13}

\textbf{Solution (b):} Now $f$ runs first: $f(-5) = 3(-5) - 2 = -17$. Since $-17 \leq 0$, $g(-17) = -(-17) = 17$. The two answers differ, which is the point --- composition is not commutative. \ans{(g \circ f)(-5) = 17}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = x^{2} - 6x$, $g(x) = x + 3$. Write $(f \circ g)(x)$.}

\textbf{Solution:} Replace each $x$ in $f$ by $x + 3$:
\begin{align*}
  (x + 3)^{2} - 6(x + 3) &= x^{2} + 6x + 9 - 6x - 18
\end{align*}
The linear terms cancel. \ans{(f \circ g)(x) = x^{2} - 9}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = x^{2} + 2x$, $g(x) = x - 3$. Solve $(f \circ g)(x) = 0$.}

\textbf{Solution:} Do not expand --- the composed form factors immediately.
\begin{align*}
  (x - 3)^{2} + 2(x - 3) &= 0 \\
  (x - 3)\bigl[(x - 3) + 2\bigr] &= 0 \\
  (x - 3)(x - 1) &= 0
\end{align*}
\[ \boxed{x = 1 \quad \text{or} \quad x = 3} \]

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f$ piecewise ($x+7$ for $x<2$; $3x$ for $x \geq 2$). \textbf{(a)} $(f \circ f)(-1)$. \textbf{(b)} Why the two passes differ.}

\textbf{Solution (a):} First pass: $-1 < 2$, so $f(-1) = -1 + 7 = 6$. Second pass: the input is now $6$, and $6 \geq 2$, so $f(6) = 3(6) = 18$. \ans{(f \circ f)(-1) = 18}

\textbf{Solution (b):} \emph{Instructor-judged.} A correct response tracks the two passes separately and says that the branch is decided by whatever number is being fed in at that moment: $-1$ selects $x + 7$, and the output $6$ then selects $3x$. An answer that applies one rule twice --- $(-1 + 7) + 7 = 13$, or $3(-1)$ then $3(-3)$ --- has treated the branch as a property of the problem instead of a property of the input.

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \dfrac{x}{x + 1}$, $g(x) = \dfrac{1}{x - 2}$. Write $(f \circ g)(x)$.}

\textbf{Solution:} Substitute, then multiply numerator and denominator by $x - 2$:
\[ f\!\left(\tfrac{1}{x-2}\right) = \frac{\tfrac{1}{x-2}}{\tfrac{1}{x-2} + 1}
   = \frac{1}{1 + (x - 2)} \]
\ans{(f \circ g)(x) = \dfrac{1}{x - 1}}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \sqrt{x}$, $g(x) = x^{2} - 3x$. \textbf{(a)} Solve $(f \circ g)(x) = 2$. \textbf{(b)} Why neither answer is extraneous.}

\textbf{Solution (a):} $(f \circ g)(x) = \sqrt{x^{2} - 3x}$, so
\begin{align*}
  \sqrt{x^{2} - 3x} &= 2 \\
  x^{2} - 3x &= 4 \\
  x^{2} - 3x - 4 &= 0 \\
  (x - 4)(x + 1) &= 0
\end{align*}
\[ \boxed{x = -1 \quad \text{or} \quad x = 4} \]

\textbf{Solution (b):} \emph{Instructor-judged.} The test is the radicand, not the sign of $x$. At $x = -1$: $1 + 3 = 4 \geq 0$; at $x = 4$: $16 - 12 = 4 \geq 0$. Both inputs leave a non-negative quantity under the square root, and both give $\sqrt{4} = 2$, so neither is extraneous. Rejecting $-1$ because it is negative confuses the sign of the input with the sign of the radicand.
