\wstitleblock{Estimating Differences and Checking Your Answer}{Grades 2--3 \quad Number and Operations in Base Ten}{}

\begin{center}\small\textit{Estimating differences by rounding, judging whether an
answer is reasonable, and checking a difference by adding}\end{center}

\noindent\textbf{Three tools you will use today.}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item \textbf{Estimate.} Round \emph{each} number to the same place, then
        subtract the rounded numbers. Use $\approx$ (``is about'') instead of $=$.
  \item \textbf{Is it reasonable?} An exact answer should land close to the
        estimate. If it is far away, a trade was probably missed.
  \item \textbf{Check by adding.} Addition undoes subtraction. If
        $\text{start} - \text{taken away} = \text{answer}$, then
        $\text{answer} + \text{taken away}$ must give back the start.
\end{itemize}

\smallskip
\noindent\textbf{Worked model:} \quad A box holds $506$ crayons. $291$ crayons are
taken out. About how many are left?
\par
Round to the nearest hundred: $506 \approx 500$ and $291 \approx 300$, so
$500 - 300 = 200$ crayons, about $200$ crayons are left. The exact answer is
$215$ crayons. Because $215$ is close to $200$, the answer is reasonable. Check
it: $215 + 291 = 506$ crayons.

\noindent\textbf{Directions.} Read each story carefully. Show your rounding or
your trades in the work space. Write the number the question asks for on the
answer line, and label your answer with what it counts.

\problem[4.5cm]{The school library has $412$ books on the shelf. Students check
out $187$ books. Round \emph{each} number to the nearest hundred, then subtract
the rounded numbers to estimate how many books are still on the shelf.
\[ 412 - 187 \approx \underline{\hspace{1.6cm}} - \underline{\hspace{1.6cm}} \]}

\problem[4.5cm]{A ferry can carry $265$ passengers. On Monday $138$ passengers
rode it. Round \emph{each} number to the nearest ten, then subtract the rounded
numbers to estimate how many more passengers the ferry could have carried.
\[ 265 - 138 \approx \underline{\hspace{1.6cm}} - \underline{\hspace{1.6cm}} \]}

\problem[5cm]{Mr.~Diaz started with a crate of apples. He sold $165$ apples and
says $175$ apples are left. Check his subtraction by adding his answer back to
the apples he sold: find $175 + 165$. Write the sum on the answer line, then
write one sentence saying how many apples were in the crate at the start.}

\problem[4.5cm]{A stadium has $738$ seats. During a game $264$ seats are empty.
Round \emph{each} number to the nearest hundred, then subtract to estimate how
many seats are filled.
\[ 738 - 264 \approx \underline{\hspace{1.6cm}} - \underline{\hspace{1.6cm}} \]}

\problem[6cm]{A recycling club wants to collect $906$ cans. So far the club has
collected $497$ cans.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Estimate how many more cans are needed by rounding each number to
        the nearest hundred.
  \item[(b)] Now find the \emph{exact} number of cans still needed.
  \item[(c)] Is your exact answer close to your estimate? Write yes or no.
\end{itemize}
\par Write the estimate and then the exact number on the answer line.}

\problem[6cm]{A book has $623$ pages. Ben has read $175$ pages. Ben says
$552$ pages are left.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Check Ben's answer by adding: find $552 + 175$.
  \item[(b)] Does the check give back $623$ pages? If it does not, find the
        exact number of pages left.
\end{itemize}
\par Write the sum and then the exact number of pages on the answer line.}

\problem[4.5cm]{A farm packed $851$ eggs this week and shipped $349$ eggs to a
store. Round \emph{each} number to the nearest hundred, then subtract to
estimate how many eggs are still at the farm.
\[ 851 - 349 \approx \underline{\hspace{1.6cm}} - \underline{\hspace{1.6cm}} \]}

\problem[6cm]{A theater sold $604$ tickets for a show. Of those, $288$ tickets
were children's tickets and the rest were adult tickets.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Estimate the number of adult tickets by rounding each number to the
        nearest hundred.
  \item[(b)] Find the exact number of adult tickets.
\end{itemize}
\par Write the estimate and then the exact number on the answer line.}

\problem[6cm]{Rico had $703$ stickers and gave away $158$ stickers. He wrote
that $655$ stickers were left. Estimate the difference to the nearest hundred
and use your estimate to explain, in one sentence, why $655$ stickers cannot be
right. Then work $703 - 158$ correctly and write the exact number of stickers on
the answer line.}

\problem[5.5cm]{A food drive has a goal of $900$ cans. Volunteers collected
$348$ cans in week one and $275$ cans in week two. Round every number to the
nearest hundred, then estimate how many more cans are still needed to reach the
goal.
\[ 900 - 348 - 275 \approx \underline{\hspace{1.4cm}} - \underline{\hspace{1.4cm}} - \underline{\hspace{1.4cm}} \]}
