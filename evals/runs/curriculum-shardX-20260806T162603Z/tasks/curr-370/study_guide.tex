\skillheading{Skill 1 --- Simplify, and keep the domain}

\begin{formulabox}
\textbf{Rule:} factor top and bottom completely, then cancel whole shared
factors. The excluded values come from the \emph{original} denominator ---
cancelling changes the formula, never the domain.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{x^2-25}{x^2-3x-10}$.
\step{Nothing can cancel while the parts are sums, so factor both of them before
looking for a shared factor.}
\step{$\dfrac{(x-5)(x+5)}{(x-5)(x+2)}$: the factor $(x-5)$ is shared, and the
original denominator is zero at $x=5$ and $x=-2$, so both stay excluded.}

$\Rightarrow\quad \ans{\dfrac{x+5}{x+2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{x^2-36}{x^2+2x-24}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x-6}{x-4}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Intercepts}

\begin{formulabox}
\textbf{Rule:} an $x$-intercept is where the \emph{simplified} numerator is zero
(a zero that cancelled is a hole, not an intercept). The $y$-intercept is the
value at $x = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find both intercepts of $f(x) = \dfrac{2x-6}{x+4}$.
\step{A quotient is zero only when its top is zero, and the $y$-intercept is just
a substitution, so each intercept needs a different part of the formula.}
\step{$2x - 6 = 0$ gives $x = 3$; and $f(0) = \dfrac{-6}{4}$.}

$x$-intercept $\ans{x = 3}$, \quad $y$-intercept $\ans{-\tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find both intercepts of $g(x) = \dfrac{3x-12}{x+2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$ and $\ans{-6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Holes and vertical asymptotes}

\begin{formulabox}
\textbf{Rule:} a denominator factor that \emph{cancels} gives a hole, whose
height is the simplified value there; a factor that \emph{survives} gives a
vertical asymptote at its zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Locate the hole and the vertical asymptote of
$f(x) = \dfrac{x^2-1}{x^2-3x+2}$.
\step{Which zero is which is decided entirely by what cancels, so factor first
and then sort the two denominator zeros.}
\step{$\dfrac{(x-1)(x+1)}{(x-1)(x-2)} = \dfrac{x+1}{x-2}$: $(x-1)$ cancels, and at
$x=1$ the simplified form gives $\dfrac{2}{-1}$.}

hole at $\bigl(1,\ \ans{-2}\bigr)$; \quad asymptote $\ans{x = 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Locate the hole and the vertical asymptote of
$g(x) = \dfrac{x^2-4}{x^2-x-2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: hole at $\bigl(2, \ans{\tfrac{4}{3}}\bigr)$, asymptote $\ans{x = -1}$}
\end{tryitbox}

\skillheading{Skill 4 --- Horizontal asymptotes from the degrees}

\begin{formulabox}
\textbf{Rule:} compare the degree of the top with the degree of the bottom.
Smaller on top gives $y = 0$; equal degrees give $y = $ the ratio of the
\emph{leading} coefficients; larger on top gives no horizontal asymptote. The
constant terms never matter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the horizontal asymptote of
$f(x) = \dfrac{5x^2-2x}{2x^2+7}$.
\step{Only the highest-power terms survive when $x$ grows, so compare degrees
first and read the leading coefficients.}
\step{Both degrees are $2$, so the level is $\dfrac{5}{2}$.}

$\Rightarrow\quad \ans{y = \tfrac{5}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the horizontal asymptote of
$g(x) = \dfrac{3x+8}{4x-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = \tfrac{3}{4}}$}
\end{tryitbox}
