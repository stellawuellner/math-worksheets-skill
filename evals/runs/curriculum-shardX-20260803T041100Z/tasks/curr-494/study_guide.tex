\sstitleblock{Parameters, Speeds and Polar Limits Without the Classic Errors}

\skillheading{Skill 1 --- The parameter is not a coordinate}

\begin{formulabox}
$\dfrac{dy}{dx} = \dfrac{dy/dt}{dx/dt}$ \; (top over bottom, in that order),
evaluated at a value of $t$ --- never at an $x$- or $y$-value.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item To use a \emph{point}, first solve $x(t) = x_0$ for $t$, then keep only
        the roots that also give $y(t) = y_0$.
  \item Horizontal tangent: $dy/dt = 0$ with $dx/dt \neq 0$. Vertical tangent:
        the other way round.
  \item $dy/dx$ describes the curve; it is unchanged if the same path is traced
        at a different rate.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $x(t) = t^2 + t$ and $y(t) = t^3 + 2t$, find
$\dfrac{dy}{dx}$ at $t = 2$.
\step{A parameter value is given outright, so no point-solving is needed --- go
straight to the quotient, top over bottom.}
\step{$\dfrac{dy/dt}{dx/dt} = \dfrac{3t^2 + 2}{2t + 1}$, and at $t = 2$ that is
$\dfrac{14}{5}$.}

$\ans{\tfrac{14}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $x(t) = t^2 - t$ and $y(t) = t^3 + t$, find
$\dfrac{dy}{dx}$ at $t = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{28}{5}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Speed is a magnitude, not a sum}

\begin{formulabox}
Velocity $\mathbf{v}(t) = \langle x'(t),\, y'(t)\rangle$ (a vector). \quad
Speed $= |\mathbf{v}(t)| = \sqrt{x'(t)^2 + y'(t)^2}$ (a scalar, never negative).
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Distance travelled $= \displaystyle\int_a^b \sqrt{x'^2 + y'^2}\;dt$ ---
        the integral of speed, not of the components.
  \item Displacement uses the components and can cancel; distance never cancels.
  \item Bound check: the speed is at least the largest $|$component$|$ and at
        most their sum.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A particle has $x(t) = t^2$, $y(t) = t^3$. Find its speed
at $t = 1$.
\step{Speed is the magnitude of the velocity vector, so both derivatives get
squared before anything is added --- adding them first is the classic slip.}
\step{$x' = 2t$ and $y' = 3t^2$, so at $t = 1$ the speed is
$\sqrt{(2 \cdot 1)^2 + (3 \cdot 1^2)^2} = \sqrt{13}$.}

The speed is $\ans{3.61}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For the same path $x(t) = t^2$, $y(t) = t^3$, find the
speed at $t = 2$, to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12.65}$}
\end{tryitbox}

\skillheading{Skill 3 --- Choose polar limits from where $r$ vanishes}

\begin{formulabox}
Area swept by the radius: $A = \dfrac{1}{2}\displaystyle\int_{\alpha}^{\beta} r^2\,
d\theta$, where $[\alpha, \beta]$ traces the region \textbf{exactly once}.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Solve $r = 0$ first: two consecutive solutions bound one petal or loop.
  \item $r = a\cos(k\theta)$ has $k$ petals when $k$ is odd and $2k$ when $k$ is
        even --- and a full turn retraces an odd rose twice.
  \item Useful integral: $\cos^2 u = \dfrac{1 + \cos 2u}{2}$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the area of one petal of $r = 2\cos(2\theta)$.
\step{A petal runs between consecutive zeros of $r$, so solve $r = 0$ before
setting any limits: $\cos 2\theta = 0$ gives $\theta = \pm\frac{\pi}{4}$.}
\step{$A = \frac{1}{2}\displaystyle\int_{-\pi/4}^{\pi/4}(2\cos 2\theta)^2 d\theta
 = 2\displaystyle\int_{-\pi/4}^{\pi/4}\cos^2 2\theta\, d\theta$.}

One petal has area $\ans{\tfrac{\pi}{2}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the area of one petal of $r = 3\cos(2\theta)$, in
exact form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{9\pi}{8}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A negative speed, a slope evaluated at a coordinate, and
a polar area taken over $0$ to $2\pi$ by default are the three errors that
produce clean-looking numbers. Each is caught by one question: is this a
magnitude, is this a $t$, does this sweep the region once?
\end{watchoutbox}
