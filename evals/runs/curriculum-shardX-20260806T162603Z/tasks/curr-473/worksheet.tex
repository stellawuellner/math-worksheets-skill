\wstitleblock{Accumulation Functions and the Fundamental Theorem}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Each function below accumulates a measured \emph{rate}. The units of
        the accumulated amount are the rate's units times the units of the
        variable being integrated.
  \item You do not need to find an antiderivative to differentiate an
        accumulation function --- that is the point of the Fundamental Theorem.
  \item Give exact values (radicals, $e$) unless a decimal is requested.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Water enters a tank at $r(t) = 3t^2 + 4$ litres per minute. The
volume added during the first $t$ minutes is
$\displaystyle V(t) = \int_0^t \left(3u^2 + 4\right)\,du$ litres.
Find $V'(5)$, in litres per minute.}

\problem{Using the same tank and the same $V$, find the volume of water added
during the first $5$ minutes.
\par\vspace*{5cm}\answerline{L}}

\problem{A culture is counted by
$\displaystyle P(x) = 200 + \int_0^x \frac{12}{1 + t}\,dt$ thousand cells, where
$x$ is in days and $0 \le x \le 10$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $P'(3)$, in thousand cells per day.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] Explain why the culture never shrinks on this interval.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{A sensor accumulates charge
$\displaystyle Q(x) = \int_1^{x^2}\sqrt{1 + t^3}\,dt$ millicoulombs, where $x$
is in seconds.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the exact value of $Q'(2)$.
    \par\vspace*{3.0cm}\ansline
  \item[(b)] Give $Q'(2)$ to two decimal places.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A cooling tower still holds
$\displaystyle C(x) = \int_x^{12}\left(0.5t + 2\right)\,dt$ litres of coolant
after $x$ hours of operation, for $0 \le x \le 12$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $C'(4)$, in litres per hour.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Explain why $C'(x)$ is negative even though the integrand is positive.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{Water flows into and out of a holding pond at the net rate
$r(t) = 6 - t$ cubic metres per minute for $0 \le t \le 10$ (a negative rate
means the pond is losing water). The change in volume over the first $x$ minutes
is $\displaystyle V(x) = \int_0^x (6 - t)\,dt$ cubic metres.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $V'(8)$.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Find the value of $x$ at which $V$ is largest.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Justify that this value really gives the largest $V$ on $[0, 10]$.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{For the same pond and the same $V$:
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $V(6)$.
    \par\vspace*{2.8cm}\answerline{m$^3$}
  \item[(b)] Find $V(10)$.
    \par\vspace*{2.8cm}\answerline{m$^3$}
\end{itemize}}

\problem{A detector's reading is
$\displaystyle F(x) = \int_{2x}^{x^2} e^{-t^2}\,dt$, with $x$ in seconds.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the exact value of $F'(1)$.
    \par\vspace*{3.0cm}\ansline
  \item[(b)] Give $F'(1)$ to three decimal places.
    \par\vspace*{2.0cm}\ansline
  \item[(c)] Explain why $F'(x)$ is not simply $e^{-x^4} - e^{-4x^2}$.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}
