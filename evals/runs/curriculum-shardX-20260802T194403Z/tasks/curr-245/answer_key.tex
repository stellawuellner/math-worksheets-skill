\problem{\textbf{Ramp: rise $5$, run $12$, find the slanted surface.} The ramp
surface is the hypotenuse --- it is opposite the right angle formed by the
ground and the vertical rise.
\[ c^2 = 12^2 + 5^2 = 144 + 25 = 169, \qquad c = \sqrt{169} \]
$\ans{13\ \text{ft}}$}

\problem{\textbf{Ladder: hypotenuse $17$, one leg $8$, find the other leg.} The
ladder itself is the longest side, so $17$ goes in the $c$ slot and the height
$h$ is a leg.
\[ 8^2 + h^2 = 17^2 \;\Rightarrow\; h^2 = 289 - 64 = 225,\qquad h = \sqrt{225} \]
$\ans{15\ \text{ft}}$
\par\smallskip
\emph{Grading note:} a student who writes $\sqrt{17^2+8^2} \approx 18.79$ has put
the ladder in a leg slot. Ask which side is longest --- the answer can never
exceed the ladder's own length.}

\problem{\textbf{Library $(2,3)$ to pool $(7,15)$.} The horizontal gap is
$7 - 2 = 5$ and the vertical gap is $15 - 3 = 12$; these are the legs of a right
triangle whose hypotenuse is the straight-line distance.
\[ d^2 = 5^2 + 12^2 = 25 + 144 = 169, \qquad d = \sqrt{169} \]
$\ans{13\ \text{units}}$
\par\smallskip
\emph{Grading note:} $5 + 12 = 17$ is the walking-along-the-streets distance,
not the straight-line distance. Sketching the triangle prevents it.}

\problem{\textbf{Garden $24$ m by $10$ m: two sides vs.\ the diagonal.} Two
steps. First the diagonal:
$d = \sqrt{24^2 + 10^2} = \sqrt{576 + 100} = \sqrt{676} = 26$ m. Then compare
with the two-side walk: $24 + 10 = 34$ m. The saving is $34 - 26$.
$\ans{8\ \text{m}}$}

\problem{\textbf{Boats: $6$ km north and $9$ km east.} North and east are
perpendicular, so the two paths are the legs and the gap between the boats is
the hypotenuse.
\[ d = \sqrt{6^2 + 9^2} = \sqrt{36 + 81} = \sqrt{117} = 10.816\ldots \]
Rounded to the nearest tenth: $\ans{10.8\ \text{km}}$
\par\smallskip
\emph{Grading note:} $6 + 9 = 15$ km is the distance if the boats sailed one
after the other along the same bent path, not the direct gap.}

\problem{\textbf{Drone $(-4,-2)$ to $(5,10)$.} Horizontal change:
$5 - (-4) = 9$. Vertical change: $10 - (-2) = 12$. Subtracting a negative is
where most errors happen here --- count the units on a sketch to confirm.
\[ d = \sqrt{9^2 + 12^2} = \sqrt{81 + 144} = \sqrt{225} \]
$\ans{15\ \text{units}}$}

\problem{\textbf{Points on the $x$-axis exactly $10$ from $(6,8)$.} A point on
the trail is $(x, 0)$. Its distance to $(6,8)$ uses the same two legs as always:
$|x - 6|$ horizontally and $8$ vertically.
\begin{align*}
  (x - 6)^2 + (0 - 8)^2 &= 10^2 \\
  (x - 6)^2 + 64 &= 100 \\
  (x - 6)^2 &= 36 \\
  x - 6 = 6 \quad &\text{or} \quad x - 6 = -6
\end{align*}
Both answers are real places on the trail: one on each side of the point
directly below the station. $\ans{x = 0 \text{ or } x = 12}$}

\problem{\textbf{Kite: string $65$ ft, height $60$ ft.} The taut string is the
hypotenuse; the height is one leg and the ground distance $g$ is the other.
\[ g^2 = 65^2 - 60^2 = 4225 - 3600 = 625, \qquad g = \sqrt{625} \]
$\ans{25\ \text{ft}}$
\par\smallskip
\emph{Grading note:} $\sqrt{65^2 + 60^2} \approx 88.46$ means the string was
treated as a leg. The ground distance must be shorter than the string.}

\problem{\textbf{Ladder slipping from $12$ ft up to $5$ ft up.} The ladder's
length never changes, so run the theorem twice with $c = 13$.
\begin{align*}
  \text{before:} \quad b_1 &= \sqrt{13^2 - 12^2} = \sqrt{169 - 144} = \sqrt{25} = 5 \\
  \text{after:} \quad b_2 &= \sqrt{13^2 - 5^2} = \sqrt{169 - 25} = \sqrt{144} = 12
\end{align*}
The base moved $12 - 5$. $\ans{7\ \text{ft}}$
\par\smallskip
\emph{Note the symmetry:} the $5$--$12$--$13$ triangle simply swapped its legs,
which is why the numbers repeat.}

\problem{\textbf{Closed route $(1,2) \to (7,10) \to (7,2) \to (1,2)$.} Three
legs, and only the first one is slanted.
\begin{align*}
  (1,2) \to (7,10): \quad &\sqrt{(7-1)^2 + (10-2)^2} = \sqrt{36 + 64} = \sqrt{100} = 10 \\
  (7,10) \to (7,2): \quad &\text{vertical, } 10 - 2 = 8 \\
  (7,2) \to (1,2): \quad &\text{horizontal, } 7 - 1 = 6
\end{align*}
Total: $10 + 8 + 6$. $\ans{24\ \text{units}}$
\par\smallskip
\emph{Grading note:} vertical and horizontal legs need no square roots --- just
the difference of the changing coordinate. Using the distance formula there is
correct but slower.}

\problem{\textbf{Points on $x = 8$ exactly $13$ from $(3,-2)$.} A point on the
line is $(8, y)$, so the horizontal leg is fixed at $8 - 3 = 5$.
\begin{align*}
  5^2 + (y + 2)^2 &= 13^2 \\
  (y + 2)^2 &= 169 - 25 = 144 \\
  y + 2 = 12 \quad &\text{or} \quad y + 2 = -12
\end{align*}
$\ans{y = 10 \text{ or } y = -14}$
\par\smallskip
The two points are $(8, 10)$ and $(8, -14)$ --- one above the tower's height and
one below, symmetric about $y = -2$.}

\problem{\textbf{Crate $120 \times 90 \times 80$ cm: the longest pole.} Apply
the theorem twice, exactly as the hint says.
\begin{align*}
  \text{floor diagonal:} \quad f &= \sqrt{120^2 + 90^2} = \sqrt{14400 + 8100} = \sqrt{22500} = 150 \\
  \text{corner to corner:} \quad d &= \sqrt{f^2 + 80^2} = \sqrt{22500 + 6400} = \sqrt{28900}
\end{align*}
The floor diagonal and the vertical edge meet at a right angle, which is what
licenses the second use of the theorem.
$\ans{170\ \text{cm}}$
\par\smallskip
\emph{Grading note:} stopping at $150$ cm answers a different question --- the
longest pole that lies flat on the floor of the crate.}
