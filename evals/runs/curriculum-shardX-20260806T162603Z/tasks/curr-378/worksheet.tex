\wstitleblock{Solving Exponential and Logarithmic Equations}{Algebra 2}{}

\begin{center}
\textit{Matching bases, taking logarithms of both sides, converting a logarithmic equation to exponential form, and solving growth and decay models}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=3pt]
  \item If both sides can be written as powers of the \emph{same base}, set the exponents equal --- that gives an exact answer with no logarithm at all.
  \item If they cannot, take a logarithm of both sides and use $\log(M^{\,p}) = p\log M$ to bring the unknown down.
  \item A logarithmic equation is solved by rewriting it in exponential form. Every answer must keep each logarithm's input positive.
  \item Each modelling problem states its units. Give the answer in that unit, rounded as asked.
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\problem[3.8cm]{Solve for $x$: \quad $3^{\,x} = 81$}

\problem[4.0cm]{Solve for $x$: \quad $\log_4(x+3) = 2$}

\problem{A culture starts with $500$ bacteria and triples every hour, so the
count after $t$ hours is $N = 500 \cdot 3^{\,t}$. Find the number of bacteria
after $4$ hours.
\par\vspace*{4.2cm}\ansline}

\problem[4.4cm]{Solve $2^{\,x} = 20$, giving $x$ to two decimal places.}

\problem[4.6cm]{Solve for $x$: \quad $5^{\,2x-1} = 125$}

\problem{An account opens with $2000$ dollars and earns $6\%$ interest
compounded once a year, so its value after $t$ years is
$A = 2000(1.06)^{\,t}$ dollars. Find the value after $10$ years, rounded to the
nearest cent.
\par\vspace*{4.4cm}\ansline}

\problem[5.4cm]{Solve for $x$: \quad $\log_2 x + \log_2(x-2) = 3$}

\problem{A $100$ milligram sample of a tracer decays so that the amount left
after $t$ days is $A = 100(0.5)^{\,t/12}$ milligrams. Find the number of days
until $30$ milligrams remain, rounded to two decimal places.
\par\vspace*{4.6cm}\answerline{days}}

\problem[5.0cm]{Solve for $x$: \quad $9^{\,x+1} = 27^{\,x}$}
