\sstitleblock{Teen Numbers: One Ten and Some Ones}

\skillheading{Skill 1 --- Breaking a teen number into ten and ones}

\begin{formulabox}
\textbf{Rule:} Every teen number is \emph{one ten and some ones}.\par
To find the ones, take the ten out of the number: \quad ones $=$ number $-$ ten.\par
\textbf{Model:} fill one ten-frame first. Whatever is left over sits outside the frame, and that leftover is the ones.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $15$ is one ten and how many ones?
\step{I want the leftover after one ten is set aside, and the ten-frame sets exactly one ten aside, so I fill the frame first and count what stays outside.}
\step{Filling the frame uses ten of the counters, so the leftover is $15 - 10$, which is five.}

$\ans{5}$ ones
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $12$ is one ten and how many ones?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$ ones}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not count the ten again. Once the frame is full it stands for one whole ten, so only the counters \emph{outside} the frame are the ones.
\end{watchoutbox}

\skillheading{Skill 2 --- Building a teen number from ten and ones}

\begin{formulabox}
\textbf{Rule:} Going the other way, put the ten first and count the ones on: \quad number $=$ ten $+$ ones.\par
Start your count at ten --- never back at one --- and say one number word for each loose counter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One full ten-frame and $7$ loose counters. What number is that?
\step{The frame is already worth a whole ten, so instead of counting every counter I start the count at ten and count on the loose ones.}
\step{Counting on from ten --- eleven, twelve, thirteen, fourteen, fifteen, sixteen, seventeen --- lands on seventeen. In notation that is $10 + 7$.}

$\ans{17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One full ten-frame and $3$ loose counters. What number is that?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two teen numbers}

\begin{formulabox}
\textbf{Rule:} Both teen numbers hold exactly one ten, so the tens can never break the tie.\par
Compare the \emph{ones}: the teen number with more loose counters is the greater one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$, $>$, or $=$ between $16$ and $13$.
\step{Both are teen numbers, so each is built from one full frame plus some loose counters; the frames match, and that leaves only the loose counters to decide.}
\step{Set the two full frames side by side and they cancel. The first number has more counters left outside than the second, so the first is greater and the wide end faces it.}

$\ans{16 > 13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$ between $11$ and $15$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11 < 15}$}
\end{tryitbox}
