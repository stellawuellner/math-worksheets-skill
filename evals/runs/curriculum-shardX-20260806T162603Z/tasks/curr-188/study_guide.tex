\sstitleblock{Graphing Proportional Relationships}

% =================== SKILL 1 ===================
\skillheading{Reading the Unit Rate from a Graph}

\begin{formulabox}
\textbf{Rule:} a proportional relationship graphs as a straight line through
the origin $(0,0)$. Its \textbf{unit rate} is $k = \dfrac{y}{x}$ at any point
on the line --- and it is exactly the $y$-value sitting above $x = 1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line goes through $(0,0)$ and $(4, 14)$. Find the
unit rate.
\step{Any point on the line gives the rate --- divide $y$ by $x$ at the clear
point $(4, 14)$.}
\step{$k = \dfrac{14}{4} = 3.5$.}

$\Rightarrow\quad \ans{3.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line goes through $(0,0)$ and $(6, 15)$. Find the
unit rate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Writing the Equation from a Graph}

\begin{formulabox}
\textbf{Rule:} every proportional relationship has the equation $y = kx$,
where $k$ is the unit rate read from the graph. Once you know $k$, the
equation predicts $y$ for \emph{any} $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A proportional graph passes through $(5, 30)$. Write
its equation.
\step{Find $k$ first, then drop it into $y = kx$.}
\step{$k = \dfrac{30}{5} = 6$.}

$\Rightarrow\quad \ans{y = 6x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A proportional graph passes through $(8, 20)$. Write
its equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 2.5x}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Graphing from a Table}

\begin{formulabox}
\textbf{Steps:} (1) check the table is proportional: every row gives the same
ratio $\dfrac{y}{x}$; (2) plot each $(x, y)$ pair; (3) the points line up on a
straight line through the origin --- draw it. If the ratios differ, the points
will \emph{not} line up, and the relationship is not proportional.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table gives $(2, 7)$ and $(4, 14)$. Find the unit
rate before plotting.
\step{Check both rows give one rate --- that is what makes the line straight.}
\step{$\dfrac{7}{2} = 3.5$ and $\dfrac{14}{4} = 3.5$ --- same rate, so the
points lie on the line $y = 3.5x$.}

$\Rightarrow\quad \ans{3.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $(3, 12)$ and $(6, 24)$. Find the unit
rate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}
