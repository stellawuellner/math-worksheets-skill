\sstitleblock{Picture Graphs and Bar Graphs --- How to Read Them}

\skillheading{Skill 1 --- One picture stands for more than one}

\begin{formulabox}
\textbf{Always find the key first.} It is the little sentence under the graph that says
what one picture is worth.
\begin{itemize}[label={$\bullet$}, leftmargin=1.0cm, itemsep=0pt, topsep=1pt]
  \item Count the pictures in the row.
  \item Skip-count by the key number, once for each picture.
  \item Half a picture is worth half the key number.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many stickers did Sam get?
\begin{center}
\begin{tikzpicture}[x=1cm, y=1cm]
  \node[anchor=east, font=\small] at (1.1,1.55) {Sam};
  \foreach \i in {0,...,3}{\draw[fill=green!30, thick] (1.3+0.6*\i,1.4) rectangle ++(0.42,0.32);}
  \node[anchor=east, font=\small] at (1.1,0.95) {Tia};
  \foreach \i in {0,...,2}{\draw[fill=green!30, thick] (1.3+0.6*\i,0.8) rectangle ++(0.42,0.32);}
  \node[anchor=west, font=\small] at (0,0.3) {Key: one picture stands for 5 stickers};
\end{tikzpicture}
\end{center}
\step{The question asks for stickers, but the row shows pictures --- so read the key and
turn pictures into stickers before counting anything.}
\step{Sam has 4 pictures. Skip-count by fives: 5, 10, 15, 20.}

$\Rightarrow\quad \ans{20}$ stickers
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many stickers did Tia get? (Same key: one picture is 5.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$ stickers}
\end{tryitbox}

\skillheading{Skill 2 --- Bar graphs where the scale skips numbers}

\begin{formulabox}
\textbf{Look at the numbers up the side before you look at the bars.} If they go
$0, 2, 4, 6, \ldots$, then one step is worth $2$, not $1$.\par
\smallskip
Put a finger on the top of the bar and slide straight across to the scale. If the bar
stops \emph{between} two numbers, it is halfway --- take the number in the middle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many cars on Tuesday?
\begin{center}
\begin{tikzpicture}
\begin{axis}[ybar, width=6.6cm, height=3.6cm, bar width=11pt,
  ymin=0, ymax=10, ytick={0,2,4,6,8,10},
  symbolic x coords={Mon,Tue,Wed}, xtick=data,
  tick label style={font=\footnotesize},
  ymajorgrids=true, grid style={draw=gray!35, line width=0.2pt},
  axis lines=left, enlarge x limits=0.3]
  \addplot[fill=orange!45, draw=black] coordinates {(Mon,6) (Tue,9) (Wed,4)};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The scale counts by twos, so the height of a bar is not the number of steps --- read
the label the bar reaches, and check whether it stops on a line or between two.}
\step{Tuesday's bar stops between 8 and 10, exactly halfway, and halfway between 8 and 10
is 9.}

$\Rightarrow\quad \ans{9}$ cars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many cars on Monday? (Same graph, same scale.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$ cars}
\end{tryitbox}

\skillheading{Skill 3 --- How many more, and how many altogether}

\begin{formulabox}
\textbf{Read every bar you need first. Write the numbers down. Then do the arithmetic.}
\begin{itemize}[label={$\bullet$}, leftmargin=1.0cm, itemsep=0pt, topsep=1pt]
  \item ``How many \textbf{more}?'' $\rightarrow$ subtract the smaller from the bigger.
  \item ``How many \textbf{altogether}?'' $\rightarrow$ add all of them.
  \item Compare the \emph{amounts}, never the number of pictures or the number of steps.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Using the car graph above, how many more cars on Tuesday than on
Wednesday?
\step{``How many more'' means a subtraction, but the graph does not show the difference
--- so read both bars first and subtract the two numbers.}
\step{Tuesday is 9 cars and Wednesday is 4 cars, so the difference is $9 - 4$.}

$\Rightarrow\quad \ans{5}$ more cars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many cars altogether on Monday, Tuesday and Wednesday?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$ cars}
\end{tryitbox}
