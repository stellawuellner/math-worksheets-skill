\wstitleblock{Sort Shapes Like a Mathematician}{Grade 1}{}
\begin{center}
\textit{Sorting shapes by sides and corners, not by color or size}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Sort by things you can COUNT: sides and corners.
  \item Color, size, and tilt never decide a shape's group.
\end{itemize}
