\sstitleblock{Proportional Relationships}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- From a table to an equation}

\begin{formulabox}
\textbf{Rule:} a table shows a proportional relationship when $\dfrac{y}{x}$ is the
\emph{same} for every column. That common value is $k$, the constant of
proportionality, and the equation is $y = kx$.\par
\textbf{To find $k$:} pick any column and divide $y$ by $x$. Then check a second
column --- if it gives a different value, the relationship is not proportional.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A table pairs $x = 4$ with $y = 10$, and $x = 6$ with $y = 15$.
Find $k$ and write the equation.
\step{I want the constant of proportionality, and a table gives me matched pairs --- so I divide $y$ by $x$ in one column and check it against another.}
\step{$\dfrac{15}{6} = 2.5$, and the other column agrees: $\dfrac{10}{4} = 2.5$. So the equation is $y = 2.5x$.}
$\Rightarrow\quad \ans{k = 2.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A table pairs $x = 3$ with $y = 21$, and $x = 5$ with $y = 35$.
Find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k = 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- From a graph to a unit rate}

\begin{formulabox}
\textbf{How to spot one:} the graph of a proportional relationship is a
\emph{straight line through the origin}. A straight line that misses the origin is
not proportional.\par
\textbf{Unit rate $=$ slope:} pick a point $(x,y)$ where the line crosses a corner of
the grid and compute $\dfrac{y - 0}{x - 0}$. It is the value of $y$ when $x$ is $1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} The line below goes through the origin and through $(4, 10)$.
Find its unit rate.
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=5, ymin=0, ymax=12,
  xtick={0,1,...,5}, ytick={0,5,10}, grid=both,
  grid style={line width=0.15pt, draw=gray!30},
  tick label style={font=\tiny}, width=5.2cm, height=3.4cm, enlargelimits=false]
  \addplot[thick, blue, domain=0:4.6] {2.5*x};
  \fill[red] (axis cs:4,10) circle (1.6pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{The line passes through the origin, so the relationship is proportional and its unit rate is just the slope --- rise over run from the origin to a lattice point.}
\step{From $(0,0)$ to $(4,10)$ the rise is $10$ and the run is $4$: $\dfrac{10}{4} = 2.5$.}
$\Rightarrow\quad \ans{k = 2.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A line through the origin also passes through $(6, 9)$. What is its
unit rate?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k = 1.5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Using the equation $y = kx$}

\begin{formulabox}
\textbf{Given $x$, want $y$:} substitute and multiply --- $y = k \cdot x$.\par
\textbf{Given $y$, want $x$:} that is an equation to \emph{solve}, not a substitution:
divide, $x = y \div k$.\par
Knowing which of the two you are doing is most of the work: read the question for
which quantity is already known.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A relationship has $k = 8$. Find $y$ when $x = 7$.
\step{The constant and the $x$-value are both known and $y$ is wanted, so this is a substitution, not an equation to solve.}
\step{$y = 8 \times 7 = 56$.}
$\Rightarrow\quad \ans{y = 56}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A relationship has $k = 9$. Find $y$ when $x = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 54}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- From a story to a model}

\begin{formulabox}
\textbf{Steps:} name the two quantities with letters; find one matched pair in the
story; divide to get $k$; write $y = kx$; then use it.\par
\textbf{Careful:} a story with a fixed charge \emph{plus} a rate (a joining fee, a
monthly line rental) is \emph{not} proportional --- doubling the amount does not
double the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 3 pencils cost 2 dollars. At the same rate, what do 12 pencils cost?
\step{The cost rises in step with the number of pencils and one matched pair is given, so I find the cost of a single pencil and multiply.}
\step{$k = \dfrac{2}{3}$ dollars a pencil, so $12$ pencils cost $\dfrac{2}{3}\times 12 = 8$ dollars.}
$\Rightarrow\quad \ans{8}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 4 pens cost 6 dollars. At the same rate, what do 14 pens cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$ dollars}
\end{tryitbox}
