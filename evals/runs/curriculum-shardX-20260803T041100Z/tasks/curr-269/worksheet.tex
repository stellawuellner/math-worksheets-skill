\wstitleblock{Function Notation: When $f(x)$ Is Not Multiplication}{Algebra 1 --- Functions}{}

\noindent\textbf{Read the notation before you compute:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $f(x)$ is the \emph{output} of the function $f$ at the input $x$. The parentheses hold an input slot, not a factor.
  \item $f(3)$ means ``replace every $x$ in the rule with $3$.'' It does not mean $f$ times $3$.
  \item $f$ on its own is not a number, so nothing can be distributed across it.
  \item $2f(3)$ doubles the \emph{output}; $f(2 \cdot 3)$ doubles the \emph{input}. These are different instructions.
\end{itemize}

\noindent For every find-and-fix item, quote the exact step where the notation
was misread, then compute the correct value.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3.5cm]{Let $f(x) = x + 4$. Find $f(3)$.
\par\vspace{2pt}
Write the substitution step before the arithmetic.}

\problem[3.5cm]{Let $g(x) = x^2 - 2$. Find $g(5)$.}

\problem[4cm]{Let $f(x) = x^2 + 1$. Compute $2f(3)$.
\par\vspace{2pt}
Then state in one line how $2f(3)$ differs from $f(2 \cdot 3)$.}

\problem[4.5cm]{\textbf{Find and fix the mistake.} Leo is given $f(x) = 4x - 1$
and asked for $f(3)$. He writes $f(3) = 3(4x - 1)$ and reports $33$.
\par\vspace{2pt}
Name the misreading, then give the correct value of $f(3)$.}

\problem[4.5cm]{\textbf{Find and fix the mistake.} Sam is given
$f(x) = x^2 + 3x$ and asked for $f(5)$. He writes $f(2 + 3) = f(2) + f(3)$ and
reports $28$.
\par\vspace{2pt}
Explain which property Sam borrowed that functions do not obey in general, then
give the correct value of $f(5)$.}

\problem[4.5cm]{Let $f(x) = 5x - 12$. Find every input $x$ for which
$f(x) = 8$.
\par\vspace{2pt}
Say in one phrase why this is an equation to solve and not a product to divide.}

\problem[5cm]{Let $h(x) = x^2 - 4x$. Find every input $x$ for which $h(x) = 0$.
\par\vspace{2pt}
A classmate reads $h(x) = 0$ as ``$h$ times $x$ equals $0$'' and answers $x = 0$
only. Show what that reading misses.}

\problem[5cm]{\textbf{Explain.} Write a short paragraph a classmate could use to
tell the two readings of $f(x)$ apart: name what the parentheses hold, say why
$f$ by itself cannot be multiplied by anything, and give one concrete test that
exposes the multiplication misreading. \noansline}
