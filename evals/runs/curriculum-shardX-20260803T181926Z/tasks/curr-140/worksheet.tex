\wstitleblock{Estimating and Solving with Decimals}{Grades 4--5 number and operations in base ten}{}

\begin{center}
{\large Estimating first, then solving decimal problems exactly}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item When a problem says \emph{estimate}, round first and then compute --- do not
        work out the exact answer and round it at the end.
  \item When a problem asks for the exact answer, line up the decimal points before
        you add or subtract.
  \item Write the unit on the answer line whenever the question gives one.
\end{itemize}

\vspace{0.15cm}

\problem[4.5cm]{A notebook costs \$3.89 and a pen costs \$2.15. Estimate the total
cost by rounding each price to the nearest whole dollar, then adding.}

\problem{One week a runner ran 48.6 km. The next week she ran 27.3 km. Estimate her
total distance for the two weeks by rounding each distance to the nearest ten
kilometres, then adding.
\par\vspace*{4.5cm}\answerline{km}}

\problem[4.5cm]{Mia had \$50 in her account. She spent \$18.75 on a book and \$6.40
on lunch. Exactly how many dollars are left in her account?}

\problem{Each bag of rice has a mass of 2.4 kg. What is the exact mass of 6 bags?
\par\vspace*{4.5cm}\answerline{kg}}

\problem[4.5cm]{Four ribbons have lengths 0.9 m, 0.85 m, 0.905 m and 0.89 m. Write
the four lengths in order from shortest to longest.}

\problem{A recipe uses 1.25 L of milk and 0.4 L of cream.
\par\vspace{2pt}
(a) Exactly how much more milk than cream does the recipe use? \quad
(b) Exactly how much liquid is that altogether?
\par\vspace*{4cm}\answerline{L}\par\vspace{0.1cm}\answerline{L}}

\problem[4.5cm]{A school buys 6 folders that cost \$4.85 each. Estimate the total
cost by rounding the price of a folder to the nearest whole dollar, then multiplying.}

\problem{A ribbon 7.2 m long is cut into 4 pieces of equal length. Exactly how long
is each piece?
\par\vspace*{4.5cm}\answerline{m}}

\problem[5cm]{Sam worked out $4.6 \times 3$ and wrote 1.38 as his answer.
\par\vspace{2pt}
(a) Work out the exact product of 4.6 and 3. \quad
(b) Write $<$, $>$ or $=$ between your exact answer and Sam's answer of 1.38. \quad
(c) Which place value did Sam put his decimal point in by mistake?}

\problem{On a hike, Jo walked 3.75 km before lunch, 2.8 km after lunch and 4.05 km
in the evening.
\par\vspace{2pt}
(a) Exactly how far did Jo walk in total? \quad
(b) Her target was 10 km. Exactly how far past the target did she go?
\par\vspace*{4.5cm}\answerline{km}\par\vspace{0.1cm}\answerline{km}}

\problem{A tap fills a container with 0.35 L of water every second.
\par\vspace{2pt}
(a) Exactly how much water is in the container after 12 seconds? \quad
(b) That water is poured into jugs that each hold 0.6 L. Exactly how many jugs does
it fill?
\par\vspace*{4.5cm}\answerline{L}\par\vspace{0.1cm}\ansline}

\problem[7cm]{A class buys 4 packs of paper at \$6.75 each and 3 markers at \$2.40
each. The class has \$40 to spend.
\par\vspace{2pt}
(a) Estimate the total cost by rounding each price to the nearest whole dollar. \quad
(b) Now find the exact total cost. \quad
(c) Write $<$, $>$ or $=$ between your estimate and the exact total. \quad
(d) Exactly how many dollars are left from the \$40?}
