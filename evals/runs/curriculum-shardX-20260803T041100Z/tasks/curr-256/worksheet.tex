\wstitleblock{Explaining Legal Equation Transformations}{Algebra 1}{}

\noindent\emph{Naming the property behind each step, judging equivalence, finding the illegal step, and the inequality sign rule}

\vspace{0.15cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item A transformation is \textbf{legal} when the new equation has exactly the
        same solutions as the old one. Every line you write must be justified by
        one of the properties below.
  \item Fill in the \textbf{Reason} column of each table as you go, then finish
        the problem in the work space and write the final answer on the line.
  \item Where a problem shows someone else's work, name the first line that is
        \emph{not} justified before you fix anything.
\end{itemize}

\vspace{0.1cm}
\noindent\textbf{Reason bank (use these names, no values shown).}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Addition / Subtraction Property of Equality} --- add or subtract
        the same quantity on \emph{both} sides.
  \item \textbf{Multiplication / Division Property of Equality} --- multiply or
        divide \emph{both} sides by the same nonzero quantity.
  \item \textbf{Distributive Property} --- $a(b+c)=ab+ac$; rewrites one side
        without changing its value.
  \item \textbf{Combining like terms} --- rewrites one side without changing
        its value.
  \item \textbf{Not a legal step} --- the new equation does not have the same
        solutions.
\end{itemize}
