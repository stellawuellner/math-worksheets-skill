\wstitleblock{Scale Factor Confusion: Linear, Square, Cubic}{High School Geometry}{}

\begin{center}\small\textit{Scaling an area by the square of the factor, scaling a volume
by its cube, and recovering the linear factor from an area or volume ratio}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item For similar figures with linear scale factor $k$: every \emph{length} scales by
        $k$, every \emph{area} (including surface area) scales by $k^2$, and every
        \emph{volume} scales by $k^3$. The exponent counts the dimensions being scaled.
  \item Going backwards, take the matching root: $k = \sqrt{\text{area ratio}}$ and
        $k = \sqrt[3]{\text{volume ratio}}$.
  \item State which rule you are using before you compute. On a \textbf{find the
        mistake} problem, name the error in words \emph{and} show correct work.
\end{itemize}
\normalsize

\noindent\small\textit{Why the exponent matches the dimensions. No measurements are shown
here --- use the numbers given in each problem.}\normalsize

\problem{A polygon has an area of 12 cm$^2$. It is dilated by a scale factor of 3.
Find the area of the image.\par\vspace*{5cm}
\answerline{cm$^2$}}

\problem{Two cones are similar. The smaller one has a volume of 54 cm$^3$. Every
linear measurement of the larger cone is 2 times the matching measurement of the smaller.
Find the volume of the larger cone.\par\vspace*{6cm}
\answerline{cm$^3$}}

\problem{\textbf{Find the mistake.} Two triangles are similar with linear scale
factor $5 : 2$ (larger to smaller). The smaller triangle has area 18 cm$^2$. Rey writes:
\[ A_{\text{larger}} = 18 \cdot \frac{5}{2} = 45 \text{ cm}^2 \]
(a) Name Rey's error in words. \quad (b) Show correct work and give the larger area.\par\vspace*{8cm}
\answerline{cm$^2$}}

\problem[7cm]{Two similar pentagons have areas in the ratio $25 : 9$. The linear scale
factor $k$ therefore satisfies
\[ 9k^2 = 25 \]
(a) Solve the equation for \emph{every} value of $k$. \quad (b) Which value can be a
scale factor for a pair of similar figures, and why must the other be rejected?}

\problem[8cm]{\textbf{Find the mistake.} Two cylinders are similar. The smaller has volume
250 cm$^3$; the larger has volume 2000 cm$^3$. Two students disagree:

\smallskip
\emph{Nia:} ``The scale factor is $2000 / 250 = 8$.''\qquad
\emph{Kim:} ``No, it is $\sqrt{8} \approx 2.83$.''

\smallskip
\textbf{Both} are wrong. (a) Name the rule each student misapplied. \quad (b) Give the
correct linear scale factor.}

\problem[9cm]{A classmate claims: \emph{``If I double every edge of a rectangular storage
box, I will need twice as much paint for the outside, and it will hold twice as much
sand.''}

\smallskip
Using a labelled sketch and the scale-factor rules, explain what really happens to the
paint and to the sand, and say why the exponent matches the number of dimensions being
scaled.
