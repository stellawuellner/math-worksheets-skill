\wstitleblock{Composing Functions from Symbolic Rules}{Algebra 2 \quad Functions}{}

\begin{center}
\textit{Evaluating a composition at a number, building the composed rule from
symbolic definitions, deciding whether the order matters, and confirming
inverses by composition}
\end{center}

\noindent\textbf{The composition machine.} \ $(f \circ g)(x)$ is another way of
writing $f\big(g(x)\big)$. The output of the \emph{inner} rule becomes the
input of the \emph{outer} rule, so the inner rule always runs first. In the
picture below no numbers are shown --- it is only the wiring; the rules and
inputs come from each problem.

\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
  \node at (-0.2,0) {$x$};
  \draw[->, thick] (0.2,0) -- (1.3,0);
  \draw[thick] (1.3,-0.5) rectangle (2.8,0.5);
  \node at (2.05,0) {$g$};
  \draw[->, thick] (2.8,0) -- (4.4,0);
  \node[above] at (3.6,0.06) {$g(x)$};
  \draw[thick] (4.4,-0.5) rectangle (5.9,0.5);
  \node at (5.15,0) {$f$};
  \draw[->, thick] (5.9,0) -- (7.1,0);
  \node[right] at (7.1,0) {$f(g(x))$};
\end{tikzpicture}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Replace the outer rule's input variable with the \emph{entire} inner
rule, kept in parentheses, before you simplify anything.
\item Expand and combine like terms so the final rule is in simplest form.
\item Show the substitution step --- that step is what is being graded, not
just the last line.
\end{itemize}

\problem[5cm]{Let $f(x) = x + 4$ and $g(x) = 3x$. Send the input $2$ through
$g$ first, then send that output through $f$. Find $(f \circ g)(2)$.}

\problem[3cm]{Let $f(x) = 2x - 1$ and $g(x) = x^2$. Fill in the middle column
by running each input through $f$, then fill in the last column by running
that result through $g$. The last column is $(g \circ f)(x)$.

\problem[6cm]{Let $f(x) = x^2 + 1$ and $g(x) = x - 3$. Write the rule for
$(f \circ g)(x)$, expanded and simplified.}

\problem[5cm]{Keep the same two rules, $f(x) = x^2 + 1$ and $g(x) = x - 3$,
but reverse the order. Write the rule for $(g \circ f)(x)$, simplified.}

\problem[6cm]{Let $f(x) = x^2 - 2x$ and $g(x) = x + 5$. Write $(f \circ g)(x)$
as a polynomial in simplest form. (The inner rule goes into \emph{both} places
where $f$ shows its input.)}

\problem[6cm]{Let $f(x) = \dfrac{1}{x - 2}$ and $g(x) = x + 2$. Write
$f\big(g(x)\big)$ as a single fraction in simplest form.}

\problem[3cm]{Let $f(x) = 2x + 1$ and $g(x) = x^2$. Build both composed rules
in simplest form, then say in one phrase what the two answers show about
order.

\problem[6.5cm]{Let $f(x) = x + 3$ and $g(x) = 2x - 1$. First build the rule
for $(f \circ g)(x)$, then use that rule to find every $x$ with
$(f \circ g)(x) = 12$.}

\problem[8cm]{Let $f(x) = 3x + 7$ and $g(x) = \dfrac{x - 7}{3}$. Composition is
the test for inverses: show that $f\big(g(x)\big)$ simplifies to $x$
\emph{and} that $g\big(f(x)\big)$ simplifies to $x$. Both directions are
needed before you may call the two rules inverses.}

\problem[8cm]{Let $h(x) = \dfrac{1}{x^2 + 6}$. Find rules $f$ and $g$, neither
of them the identity rule $x$, so that $h(x) = f\big(g(x)\big)$. Then compose
your two rules back together and explain, in a sentence, how that composition
shows your choice works. \noansline}
