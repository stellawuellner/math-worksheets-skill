\sstitleblock{Fact Strategies: Turn It Around, Break It Apart}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Turn-around facts (commutative)}

\begin{formulabox}
\textbf{Rule:}\quad The order of the factors never changes the product.\par
An array turned on its side still holds the same number of things.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} You know $4 \times 9 = 36$. Find $9 \times 4$.
\step{The factors are the same two numbers in the other order, so the product cannot change.}
\step{$9 \times 4 = 36$, the same as $4 \times 9$.}

$\Rightarrow\quad\ans{36}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} You know $3 \times 8 = 24$. What is $8 \times 3$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Break-apart facts (distributive)}

\begin{formulabox}
\textbf{Rule:}\quad Split one factor into easy pieces, multiply each piece, add the parts.\par
Good splits use your anchor facts: fives, tens, and twos.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $7 \times 8$ by breaking the 8 apart.
\step{Split the 8 into $5 + 3$, two pieces with friendly facts.}
\step{$7 \times 5 = 35$ and $7 \times 3 = 21$, so $35 + 21 = 56$.}

$\Rightarrow\quad\ans{56}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $6 \times 12$ by splitting the 12 into $10 + 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{72}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Choosing the strategy}

\begin{formulabox}
\textbf{Rule:}\quad Ask two questions, in order:\par
Do I know the turn-around fact already? If yes, use it. If not, which factor splits into fives, tens, or twos most easily? Break that one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Choose a strategy for $8 \times 6$.
\step{Neither order is a fact I know cold, so break a factor: the 6 splits into $5 + 1$, both easy with 8.}
\step{$8 \times 5 = 40$ and $8 \times 1 = 8$, so $40 + 8 = 48$.}

$\Rightarrow\quad\ans{48}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Choose your own strategy for $7 \times 9$ and solve it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{63}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When you break a factor apart, multiply \emph{both} pieces --- then \emph{add} the two part-products. Forgetting one piece (or adding the pieces before multiplying) is the classic slip.
\end{watchoutbox}
