\wstitleblock{Slopes of Parallel and Perpendicular Lines}{High School Geometry}{}

\begin{center}
\small\itshape Using slopes to test whether segments are parallel or perpendicular, and to prove claims about figures on the coordinate plane
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Slope is the change in $y$ over the change in $x$ between two points on the line.
  \item Two non-vertical lines are \textbf{parallel} exactly when their slopes are equal, and \textbf{perpendicular} exactly when their slopes are opposite reciprocals --- that is, when the product of the slopes is $-1$.
  \item Show the slope calculation for every segment you compare. A conclusion with no slopes behind it earns no credit.
  \item Leave fractional slopes as fractions in lowest terms.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Find the slope of $\overline{AB}$ for $A(2, 1)$ and $B(6, 9)$.
Then state the slope of any line parallel to $\overline{AB}$.}

\problem[5cm]{Find the slope of $\overline{CD}$ for $C(-3, 5)$ and $D(1, -3)$.
Then state the slope of any line perpendicular to $\overline{CD}$.}

\problem[5.5cm]{$\overline{PQ}$ has endpoints $P(0, -2)$ and $Q(4, 4)$; $\overline{RS}$ has endpoints $R(-1, 3)$ and $S(3, 9)$.
Are the two segments parallel? Find both slopes and justify your answer.}

\problem[5.5cm]{$\overline{EF}$ has endpoints $E(1, 2)$ and $F(5, 4)$; $\overline{GH}$ has endpoints $G(0, 6)$ and $H(2, 2)$.
Are the two segments perpendicular? Find both slopes and justify your answer.}

\problem[5cm]{$\overline{JK}$ has endpoints $J(4, -1)$ and $K(4, 7)$.
Find the slope of $\overline{JK}$. Then state the slope of any line perpendicular to $\overline{JK}$ and explain why the opposite-reciprocal rule has to be read carefully here.}

\problem[6.5cm]{Quadrilateral $ABCD$ has vertices $A(-2, 1)$, $B(2, 3)$, $C(4, 7)$ and $D(0, 5)$.
Find the slopes of all four sides and use them to show that both pairs of opposite sides are parallel.}

\problem[6.5cm]{Triangle $PQR$ has vertices $P(-1, -1)$, $Q(3, 1)$ and $R(1, 5)$.
Use slopes to show that the angle at $Q$ is a right angle, then find the area of triangle $PQR$ in square units.}

\problem[7cm]{Quadrilateral $WXYZ$ has vertices $W(0, 0)$, $X(6, 2)$, $Y(8, 8)$ and $Z(2, 6)$.

\textbf{(a)} Find the slopes of $\overline{WX}$ and $\overline{XY}$ and decide whether $WXYZ$ is a rectangle.

\textbf{(b)} Find the midpoint of each diagonal and state what those midpoints show about the figure.}

\problem[7cm]{Quadrilateral $EFGH$ has vertices $E(1, 2)$, $F(4, 5)$, $G(7, 2)$ and $H(4, -1)$.
Find the slopes of $\overline{EF}$ and $\overline{FG}$, then find the lengths of those two sides to the nearest hundredth.
Explain what the four results together prove about $EFGH$.}

\problem[8cm]{Triangle $TUV$ has vertices $T(0, 0)$, $U(6, 2)$ and $V(2, 8)$.
Let $M$ be the midpoint of $\overline{TV}$ and let $N$ be the midpoint of $\overline{UV}$.

\textbf{(a)} Find the coordinates of $M$ and $N$.

\textbf{(b)} Find the slope of $\overline{MN}$ and the slope of $\overline{TU}$.

\textbf{(c)} Write a short coordinate proof that $\overline{MN}$ is parallel to $\overline{TU}$.}
