\sstitleblock{Interior and Exterior Polygon Angle Sums}

\skillheading{Skill 1 --- Interior angle sum of a convex polygon}

\begin{formulabox}
\textbf{Theorem:} \ For a convex polygon with $n$ sides,
\ $\text{sum of interior angles} = (n - 2)\,180^\circ$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \emph{Why:} diagonals from one vertex cut the polygon into exactly
        $n - 2$ triangles, and each triangle holds $180^\circ$.
  \item Running it backwards finds $n$: set $(n - 2)180$ equal to the given sum.
  \item If one angle is missing, add the known angles and subtract from the sum.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the interior angle sum of a convex heptagon.
\step{Only the number of sides is known and a total is wanted, so this is the
      interior angle sum theorem, not the exterior one.}
\step{$n = 7$, so the sum is $(7 - 2)180^\circ = 5 \cdot 180^\circ$.}
$\Rightarrow\quad \ans{900^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Three angles of a convex quadrilateral are $88^\circ$,
$154^\circ$, and $97^\circ$. Find the fourth, $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 21^\circ}$}
\end{tryitbox}

\skillheading{Skill 2 --- Exterior angles always total 360 degrees}

\begin{formulabox}
\textbf{Theorem:} \ Taking one exterior angle at each vertex,
\ $\text{sum of exterior angles} = 360^\circ$ --- for \emph{every} convex polygon,
whatever $n$ is.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item At each vertex the interior and exterior angles form a linear pair:
        $\text{interior} + \text{exterior} = 180^\circ$.
  \item Regular polygon: one exterior angle $= \dfrac{360^\circ}{n}$, so
        $n = \dfrac{360^\circ}{\text{one exterior angle}}$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find one exterior angle of a regular nonagon.
\step{The polygon is regular and an exterior angle is wanted, so share the fixed
      $360^\circ$ total equally among the vertices.}
\step{$n = 9$, so one exterior angle is $360^\circ \div 9$.}
$\Rightarrow\quad \ans{40^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Each exterior angle of a regular polygon is $45^\circ$.
How many sides has it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Angles of a regular polygon}

\begin{formulabox}
\textbf{Formulas:} \ In a \emph{regular} $n$-gon all angles are congruent, so
\[
  \text{one interior angle} = \frac{(n - 2)\,180^\circ}{n},
  \qquad
  \text{one exterior angle} = \frac{360^\circ}{n}.
\]
The two are supplementary, so either one gives the other instantly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find one interior angle of a regular pentagon.
\step{Regular means the interior total is shared equally, so divide the interior
      angle sum by the number of angles.}
\step{$\dfrac{(5 - 2)180^\circ}{5} = \dfrac{540^\circ}{5}$.}
$\Rightarrow\quad \ans{108^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Each interior angle of a regular polygon is $140^\circ$.
How many sides has it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The $360^\circ$ belongs to the \emph{exterior} angles only.
Dividing $360^\circ$ by $n$ gives an exterior angle, never an interior one --- and
$(n-2)180^\circ$ is a total for the whole polygon, not a single angle.
\end{watchoutbox}
