\sstitleblock{Random Samples and Simple Probability Models}

\noindent\small Two ideas run through everything here. A \emph{probability
model} says how likely an outcome is before you collect data; a \emph{random
sample} lets a small group speak for a large one. Both are fractions:
part over whole.\normalsize

\skillheading{Probability of one event}
\begin{formulabox}
\textbf{Rule:} When the outcomes are equally likely,
$P(\text{event}) = \dfrac{\text{favourable outcomes}}{\text{possible
outcomes}}$, written in lowest terms.\par
Every probability lands between $0$ and $1$. To predict a count in many trials,
multiply: $\text{predicted count} = P \times \text{number of trials}$. The
complement rule saves work: $P(\text{not } A) = 1 - P(A)$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A bag holds $8$ marbles, all the same size: $3$ are red.
One is drawn without looking. Find $P(\text{red})$.
\step{The marbles are identical apart from colour, so the $8$ outcomes are
equally likely and the fraction model applies.}
\step{Favourable is the $3$ red marbles, possible is all $8$ marbles:
$\frac{3}{8}$, which has no common factor to cancel.}

$P(\text{red}) = \ans{\frac{3}{8}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A spinner has $12$ equal sections and $5$ of them are
yellow. Find $P(\text{yellow})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{12}}$}
\end{tryitbox}

\skillheading{Estimating from a random sample}
\begin{formulabox}
\textbf{Rule:} In a random sample, the sample fraction estimates the population
fraction. So divide the count in the sample by the sample size, then multiply
by the population size: that product estimates the count in the whole
population.\par
This only works when the sample is \emph{random} --- every member of the
population having the same chance of being chosen. A sample gathered where the
answer is more common (a skate park, a team practice) is biased, and no
arithmetic can repair it.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad In a random sample of $25$ of a town's $400$ dogs, $9$
are brown. Estimate how many of the town's dogs are brown.
\step{The sample was random, so the brown fraction in the sample is the best
estimate of the brown fraction in the whole town --- scale it up.}
\step{$\frac{9}{25}$ of $400$ is $\frac{9}{25} \times 400 = 9 \times 16$.}

About $\ans{144}$ brown dogs
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad In a random sample of $20$ of a school's $300$ students,
$7$ ride the bus. Estimate how many students ride the bus.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{105}$}
\end{tryitbox}

\skillheading{Summarizing the sample you collected}
\begin{formulabox}
\textbf{Rule:} Order the values first. The \textbf{median} is the middle value
(the average of the two middle values when the count is even). The
\textbf{mean} is the total divided by how many values there are.\par
Split the ordered list at the median: $Q_1$ is the median of the lower half,
$Q_3$ the median of the upper half, and the \textbf{interquartile range} is
$Q_3 - Q_1$, the spread of the middle half. The \textbf{range}, largest minus
smallest, is a different number and uses only the two extremes.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Sample: $4$, $6$, $9$, $10$, $12$, $15$, $21$. Find the
median and the interquartile range.
\step{Both measures need the values in order and the list split at the middle,
so locate the median first and work outward.}
\step{Seven values, so the median is the fourth. Lower half $4$, $6$, $9$ gives
$Q_1 = 6$; upper half $12$, $15$, $21$ gives $Q_3 = 15$; then $15 - 6$.}

Median $\ans{10}$, IQR $\ans{9}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Sample: $2$, $5$, $7$, $8$, $11$, $13$, $16$. Find the
median and the interquartile range.\\[2pt]
\rotatebox{180}{\footnotesize check: median $\ans{8}$, IQR $\ans{8}$}
\end{tryitbox}
