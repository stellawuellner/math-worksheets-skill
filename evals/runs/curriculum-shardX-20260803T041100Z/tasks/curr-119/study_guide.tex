\sstitleblock{Diagnosing an Overgeneralized Number-Pattern Rule}

\skillheading{Skill 1 --- Test the rule on terms nobody checked}

\begin{formulabox}
\textbf{Rule:} a rule that fits the terms you used to \emph{make} it is not
evidence. Test it on a term you did not use.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Write the step between every pair of terms you can see, not just the first pair.
  \item If all the steps are the same number, the rule is ``add that step''.
  \item If each term is the one before times the same number, the rule is ``multiply''.
  \item ``Add'' and ``multiply'' rules often agree on the first two terms --- that
        agreement proves nothing.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $2$, $4$, $6$, $8$, \ldots\ Someone says ``double the term
before''. Test it, then find the $9$th term.
\step{Doubling and adding both explain $2 \rightarrow 4$, so test the next step
instead: doubling $4$ predicts $8$, but the third term is $6$ --- the doubling rule
is out.}
\step{Every step is $+2$, so the $n$th term is $2 \times n$, and the $9$th term is
$2 \times 9$.}
The $9$th term is $\ans{18}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $7$, $14$, $21$, $28$, \ldots\ Test ``double the term
before'', then find the $10$th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{70}$}
\end{tryitbox}

\skillheading{Skill 2 --- Beat a rule with one counterexample}

\begin{formulabox}
\textbf{Rule:} to disprove ``\emph{every} A is B'', you need one A that is not B.
One counterexample beats any number of supporting examples.\par
\smallskip
\textbf{Testing ``is it a multiple?'':} divide and look at the remainder.
A remainder of $0$ means yes; any other remainder means no --- and that number is
your counterexample.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad ``Every multiple of $3$ is a multiple of $9$.'' Test the
multiple of $3$ that the claim never mentions: $15$.
\step{To disprove a ``multiple of'' claim, divide the counterexample candidate by
$9$ and look at the remainder rather than hunting for more examples that work.}
\step{Nine goes into fifteen once: $15 - 9 \times 1$ is what is left over.}
The remainder is $\ans{6}$, so $15$ is not a multiple of $9$ and the claim is false.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad ``Every multiple of $5$ is a multiple of $10$.'' Divide $25$
by $10$ and write the remainder.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Repair the rule, then use it}

\begin{formulabox}
\textbf{How to repair an overgeneralized rule:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Split the claim into parts and test each part separately --- often one half
        survives and one half fails.
  \item Keep the part every term supports; cross out the part one term refutes.
  \item Write the repaired rule as a shortcut you can jump with:
        the multiples of $k$ have $n$th term $k \times n$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $6$, $12$, $18$, $24$, \ldots\ Claim: ``the terms are the
multiples of $6$, so every term is even and ends in $6$.'' Repair it, then find the
$11$th term.
\step{Test the claim one half at a time: ``multiples of $6$'' holds for every term,
but ``ends in $6$'' already fails at the second term, so only the first half is kept.}
\step{The repaired rule gives the $n$th term as $6 \times n$, so jump straight to
$6 \times 11$ instead of listing terms.}
The $11$th term is $\ans{66}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $9$, $18$, $27$, \ldots\ Repair the claim ``every term is a
multiple of $9$ and ends in an odd digit'', then find the $12$th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{108}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``It worked three times'' is the trap. Three cases can all
pass while the rule is still false --- and the cases people pick are usually the
ones they built the rule from.
\end{watchoutbox}
