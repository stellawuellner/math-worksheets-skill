\wstitleblock{Equivalent Expressions: Distribute, Combine, Factor}{Grade 6--7 Mathematics}{}
\begin{center}
\textit{distributing, combining like terms, and factoring out a common factor}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Show every step. An answer is an \emph{expression}, so keep the variable in it.
  \item In the story problems, the units and givens are stated in the problem --- use them.
\end{itemize}

\problem[3cm]{Use the distributive property to expand: \quad $4(x + 3)$}

\problem{A rectangular garden bed is 3 m deep. Its width is split into two sections:
$x$ m planted with tomatoes and 5 m planted with peppers.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Write the total area of the bed, in square meters, as a \emph{product}
  (depth times total width).\par\vspace*{1.5cm}\ansline
  \item[(b)] Distribute to write the same area as a \emph{sum} of the two section
  areas.\par\vspace*{1.5cm}\ansline
  \item[(c)] In one sentence, explain how the two-section garden shows that your
  product and your sum are equal.\par\vspace*{1.5cm}
\end{itemize}}

\problem[3.2cm]{Combine like terms: \quad $7x + 4 + 2x - 9$}

\problem[3.2cm]{Factor out the greatest common factor: \quad $6x + 15$}

\problem[3.2cm]{Expand: \quad $-2(4x - 7)$}

\problem{The three sides of a triangle measure $x$ cm, $(x + 4)$ cm, and $(2x - 1)$ cm.
Write an expression for the perimeter of the triangle, then combine like terms to
simplify it.\par\vspace*{3.8cm}\answerline{cm}}

\problem{4 friends visit a deli. Each friend buys a sandwich that costs $s$ dollars
and a juice that costs 2 dollars.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Write the total cost, in dollars, as a \emph{product} (number of friends
  times cost per friend).\par\vspace*{1.5cm}\ansline
  \item[(b)] Distribute to write the total cost as a \emph{sum}.\par\vspace*{1.5cm}\ansline
\end{itemize}}

\problem[3.4cm]{Factor out the greatest common factor: \quad $9x + 24$}

\problem[4.2cm]{Simplify completely: \quad $3(2x + 5) - 4x + 1$}

\problem[4.6cm]{\textbf{Challenge.} Simplify completely: \quad $5(2x + 3) - 2(x - 4)$}
