\wstitleblock{Second and Higher Derivatives}{AP Calculus AB/BC \quad Implicit, Inverse, and Higher-Order Differentiation}{}

\begin{center}\small\textit{Repeated power-rule differentiation, second
derivatives through the product and chain rules, implicit second derivatives, and
acceleration and concavity}\end{center}

\noindent\textbf{The derivative ladder.} A higher-order derivative is just
differentiation done again --- each rung is the derivative of the rung above it,
and each rung has its own meaning. Simplify before climbing to the next rung;
an unsimplified rung doubles the work on the one after it.

\begin{center}\small
\begin{tabular}{|>{\raggedright\arraybackslash}m{2.3cm}|>{\raggedright\arraybackslash}m{3.0cm}|>{\raggedright\arraybackslash}m{7.6cm}|}
\hline
\textbf{Rung} & \textbf{Notation} & \textbf{What it measures} \\ \hline
function & $f$, \ $y$, \ $s(t)$ & value, position \\ \hline
first & $f'$, \ $\frac{dy}{dx}$, \ $v(t)$ & rate of change, slope, velocity \\ \hline
second & $f''$, \ $\frac{d^2y}{dx^2}$, \ $a(t)$ & concavity, acceleration \\ \hline
third and up & $f'''$, \ $f^{(4)}$ & rate of change of the rung above \\ \hline
\end{tabular}
\end{center}
\noindent\small\textit{Notation used throughout this sheet. No values shown: use
the function given in each problem.}\normalsize

\noindent\textbf{Directions.}
\begin{itemize}[itemsep=2pt]
  \item Show each rung of the ladder, not just the last one --- write $f'$
        before $f''$.
  \item Simplify each derivative before differentiating again.
  \item Give exact values (no decimal approximations) unless a problem asks
        otherwise, and state units where the problem gives them.
\end{itemize}

\problem[6cm]{Find $f''(x)$ for $f(x) = x^5 - 3x^3 + 2x$.}

\problem[6cm]{Find the third derivative $g'''(x)$ for $g(x) = 2x^4 - x^2 + 7$.}

\problem[6cm]{Find $f''(x)$ for $f(x) = \sin(3x)$. The chain rule is needed on
both rungs.}

\problem[7cm]{Find $h''(x)$ for $h(x) = x^2 e^x$, and write your answer with
$e^x$ factored out.}

\problem[8cm]{A drone rises along a vertical track. Its height above the launch
pad, in metres, is $s(t) = t^3 - 6t^2 + 9t$, where $t$ is in seconds and
$0 \le t \le 5$.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Find the acceleration function $a(t) = s''(t)$, in metres per
        second squared.
  \item[(b)] Find the time $t$, in seconds, at which the acceleration is zero.
\end{itemize}
\par Write $a(t)$ and then the time on the answer line.}

\problem[7cm]{Let $f(x) = \ln x + x^3$ for $x > 0$.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Find $f''(x)$.
  \item[(b)] Evaluate $f''(2)$ and state what its sign says about the concavity
        of the graph of $f$ at $x = 2$.
\end{itemize}
\par Write $f''(x)$ and then the value of $f''(2)$ on the answer line.}

\problem[9cm]{The circle $x^2 + y^2 = 25$ defines $y$ implicitly as a function of
$x$ on the upper arc.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Differentiate implicitly to find $\dfrac{dy}{dx}$, then
        differentiate again to find $\dfrac{d^2y}{dx^2}$ in terms of $x$ and
        $y$. Substitute the original equation to write it in terms of $y$ alone.
  \item[(b)] Evaluate $\dfrac{d^2y}{dx^2}$ at the point $(3, 4)$.
\end{itemize}
\par Write the second derivative and then its value at $(3, 4)$ on the answer
line.}

\problem[9cm]{A particle moves along a line with position
$s(t) = t^4 - 4t^3 + 12$ metres at time $t$ seconds, $t \ge 0$.
\begin{itemize}[label={}, itemsep=2pt]
  \item[(a)] Find the acceleration $a(t) = s''(t)$, in metres per second squared.
  \item[(b)] Find every time $t \ge 0$ at which the acceleration is zero.
  \item[(c)] Using your answer to (b), say on which interval the position graph
        is concave down.
\end{itemize}
\par Write $a(t)$ and then the times on the answer line.}
