\sstitleblock{Naming Shapes Any Way They Turn}

% =================== SKILL 1 ===================
\skillheading{Count the sides and corners}

\begin{formulabox}
\textbf{Rule:} A shape's name comes from counting.\par
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item triangle: 3 straight sides \quad $\bullet$ \ square and rectangle: 4
  \item hexagon: 6 straight sides \quad $\bullet$ \ sides and corners always match!
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many sides and corners does a square have?
\step{Trace the shape with a finger --- count each straight side, then each
place where two sides meet.}
\step{A square has 4 straight sides, and the sides meet 4 times.}

$\Rightarrow\quad$ a square has $\ans{4}$ sides and 4 corners
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many corners does a triangle have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{A turned shape keeps its name}

\begin{formulabox}
\textbf{Rule:} Turning, tilting, or flipping a shape never changes its name.
The name lives in the sides and corners --- and turning changes neither.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A square is tilted to stand on one corner. How many
sides does it have now?
\step{Turning moves the shape but keeps every side and corner --- so count
just as before.}
\step{The four sides are all still there: count 4.}

$\Rightarrow\quad$ still $\ans{4}$ sides --- still a square!
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A triangle is flipped upside down. How many corners
does it have now?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ --- still a triangle}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A tilted square is NOT a new shape called a
``diamond.'' Before renaming a shape, count its sides and corners --- if
nothing changed, the name did not change either.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Compare shapes by counting}

\begin{formulabox}
\textbf{Rule:} To compare two shapes, count the same thing on both (sides or
corners), then subtract the counts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many more sides does a hexagon have than a
triangle?
\step{Count sides on each shape first, then find the difference.}
\step{Hexagon: 6 sides. Triangle: 3 sides. \ $6 - 3 = 3$.}

$\Rightarrow\quad$ the hexagon has $\ans{3}$ more sides
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many more sides does a square have than a
triangle?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}
