\wstitleblock{Random Samples and Probability Models}{Grades 6--7 Mathematics}{}

\noindent\small
Probability models for one event, estimates from a random sample, and summaries
of the sample itself. Every data set below comes from a stated sample, and
every count and unit is given in the problem.
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item For equally likely outcomes,
        $P(\text{event}) = \dfrac{\text{number of favourable outcomes}}{\text{number
        of possible outcomes}}$. Leave probabilities as fractions in lowest terms.
  \item A random sample stands in for its population: the fraction in the
        sample is your estimate of the fraction in the whole population.
  \item Show the computation behind each answer, not just the result.
\end{itemize}
\normalsize
\vspace{0.2cm}

\problem[4cm]{A raffle drum holds $20$ tickets, all the same size and shape,
and $8$ of them are blue. One ticket is drawn without looking. Find the
probability that the ticket drawn is blue. Give your answer as a fraction in
lowest terms.}

\problem[5cm]{A teacher picked a random sample of $8$ students from her class
roster and recorded how many minutes each spent on homework last night.
\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|}
\hline
$15$ & $20$ & $25$ & $30$ & $20$ & $35$ & $20$ & $15$ \\
\hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Find the mean number of minutes for this sample.
\textbf{(b)} Find the mode.}

\problem[6cm]{To estimate how many fish live in a pond, a biologist tags $12$
fish and releases them. A week later she nets a random sample of $40$ fish and
finds that $6$ of them are tagged. \textbf{(a)} What fraction of the sample was
tagged? Give it in lowest terms. \textbf{(b)} Assuming the same fraction of the
whole pond is tagged, estimate how many fish live in the pond.}

\problem[5cm]{A town surveyed a random sample of $12$ households and recorded
the number of pets in each.
\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|c|c|c|c|}
\hline
$0$ & $0$ & $1$ & $1$ & $1$ & $2$ & $2$ & $2$ & $3$ & $3$ & $4$ & $5$ \\
\hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Find the median number of pets per household in this sample.
\textbf{(b)} Find the total number of pets in the sampled households.}

\problem[6cm]{A middle school has $900$ students. The principal draws a random
sample of $50$ students from the enrolment list and finds that $14$ of them
walk to school. \textbf{(a)} What fraction of the sample walks to school, in
lowest terms? \textbf{(b)} Use that fraction to estimate how many students in
the whole school walk to school.}

\problem[6cm]{A fair number cube has faces numbered one through six.
\textbf{(a)} Find the probability of rolling a four on one roll.
\textbf{(b)} The cube will be rolled $300$ times. Use the probability model to
predict how many of those rolls land on four.}

\problem[6cm]{A weather club recorded the daily high temperature, in degrees
Fahrenheit, on seven randomly chosen days in April.
\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|}
\hline
$12$ & $15$ & $15$ & $18$ & $22$ & $27$ & $31$ \\
\hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Find the median of the sample. \textbf{(b)} Find the interquartile
range --- the spread of the middle half of the sample. A classmate answers
$19$ for part (b); explain what that number is instead.}

\problem[7cm]{A student wants to know what fraction of her school's $800$
students prefer skateboarding to biking. She asks $60$ students at the skate
park after school, and $45$ of them say skateboarding. \textbf{(a)} What
fraction of her sample prefers skateboarding, in lowest terms? \textbf{(b)}
What estimate for the whole school does that fraction give? \textbf{(c)}
Explain why this estimate should not be trusted, and describe a way of choosing
the sample that would make it trustworthy.}

\problem[8cm]{A cereal company puts one of four equally likely prize cards in
every box, and you want to know the chance of getting the card you want in your
next box. Design a simulation that models this situation. Say what random
device you would use, which outcomes stand for getting the card you want, what
counts as one trial, how many trials you would run, and how you would use the
results to estimate the probability.}
