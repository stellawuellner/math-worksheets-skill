\wstitleblock{Solving Linear Systems by Substitution}{Algebra 1}{}

\begin{center}\small\itshape
Substituting a given isolated variable, isolating first, special cases, and word models
\end{center}

\noindent\small\textbf{Directions:} Solve each system \textbf{by substitution}.
Show the substituted equation, the value you solve for first, and the
back-substitution. Write each answer as an ordered pair $(x, y)$ --- or, for a
word problem, as a labelled sentence. Check every answer in the equation you did
\emph{not} substitute into.\par\vspace{2pt}
\textbf{Worked pattern:} \; From $y = x + 1$ and $2x + y = 7$: replace $y$ in the
second equation to get $2x + (x + 1) = 7$, so $3x + 1 = 7$, $x = 2$. Back-substitute:
$y = 2 + 1 = 3$. Answer $(2, 3)$; check in $2x + y$: $4 + 3 = 7$ $\checkmark$
\normalsize

\problem[5cm]{$y = 2x$ \qquad and \qquad $3x + y = 15$}

\problem[5.5cm]{$y = x - 4$ \qquad and \qquad $2x + 3y = 3$}

\problem[6cm]{$2x + y = 11$ \qquad and \qquad $3x + 2y = 18$\\[2pt]
\small(Neither equation is solved for a variable yet --- isolate one first.)}

\problem[5.5cm]{$x = 3y + 1$ \qquad and \qquad $2x - 5y = 7$}

\problem[6cm]{$x + y = 7$ \qquad and \qquad $3x - y = 5$}

\problem[6.5cm]{A school play sold $12$ tickets to one family. Adult tickets cost
$8$ dollars each and child tickets cost $5$ dollars each, and the family paid
$81$ dollars in all. Let $x$ be the number of adult tickets and $y$ the number of
child tickets. Write the system, then solve it by substitution.}

\problem[6cm]{$y = 3x - 2$ \qquad and \qquad $2x + y = 5$\\[2pt]
\small(The solution is not a pair of whole numbers --- keep it in fraction form.)}

\problem[6cm]{$x - 3y = -1$ \qquad and \qquad $2x + y = 12$}

\problem[6cm]{$y = 2x + 3$ \qquad and \qquad $4x - 2y = 1$\\[2pt]
\small(Substitute as usual. If the variables disappear, say what the resulting
statement tells you about the system, and describe the graph.)}

\problem[6cm]{$y = 3x - 4$ \qquad and \qquad $6x - 2y = 8$\\[2pt]
\small(If the variables disappear here too, decide what \emph{this} statement
means, then give one specific pair that solves the system.)}

\problem[6.5cm]{The perimeter of a rectangle is $46$ m. Its length is $5$ m more
than twice its width. Let $x$ be the width and $y$ the length, both in metres.
Write the system, then solve it by substitution to find both dimensions.}

\problem[7cm]{A chemist mixes a $20\%$ acid solution with a $50\%$ acid solution
to make $12$ L of a $30\%$ acid solution. Let $x$ be the litres of $20\%$
solution and $y$ the litres of $50\%$ solution. The acid alone must balance:
$0.2x + 0.5y = 3.6$. Write the second equation, then solve the system by
substitution.}
