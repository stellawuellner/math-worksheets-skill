\problem{Solve $3^{\,x} = 81$.}

\textbf{Solution:} $81$ is a power of $3$: $81 = 3^4$. With equal bases the
exponents must be equal. \ans{x = 4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Solve $\log_4(x+3) = 2$.}

\textbf{Solution:} Rewrite in exponential form: $4^2 = x + 3$, so
$16 = x + 3$. The input at $x = 13$ is $16 > 0$, so the answer stands.
\ans{x = 13}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$N = 500 \cdot 3^{\,t}$ bacteria; find $N$ at $t = 4$ hours.}

\textbf{Solution:} Substitute $t = 4$ and evaluate the power first:
$3^4 = 81$, so $N = 500 \times 81$. \ans{40500}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Solve $2^{\,x} = 20$ to two decimal places.}

\textbf{Solution:} $20$ is not a power of $2$, so take logarithms of both sides
and bring the exponent down with the power rule:
\[ x\ln 2 = \ln 20 , \qquad x = \frac{\ln 20}{\ln 2} = \frac{2.99573\ldots}{0.69314\ldots}
   = 4.32192\ldots \]
\ans{x \approx 4.32}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Solve $5^{\,2x-1} = 125$.}

\textbf{Solution:} $125 = 5^3$, so $5^{2x-1} = 5^3$ and the exponents match:
$2x - 1 = 3$, giving $2x = 4$. \ans{x = 2}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$A = 2000(1.06)^{\,t}$ dollars; find $A$ at $t = 10$ years.}

\textbf{Solution:} Evaluate the growth factor first, then multiply:
\[ 1.06^{10} = 1.7908476\ldots , \qquad A = 2000 \times 1.7908476\ldots
   = 3581.6953\ldots \]
To the nearest cent, in dollars, \ans{3581.70}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Solve $\log_2 x + \log_2(x-2) = 3$.}

\textbf{Solution:} Condense with the product rule, then switch to exponential
form:
\begin{align*}
  \log_2\bigl(x(x-2)\bigr) &= 3 \\
  x(x-2) &= 2^3 = 8 \\
  x^2 - 2x - 8 &= 0 \\
  (x-4)(x+2) &= 0 .
\end{align*}
The candidates are $x = 4$ and $x = -2$. At $x = -2$ both inputs, $x$ and
$x - 2$, are negative, so neither logarithm exists --- that candidate is
discarded. At $x = 4$: $\log_2 4 + \log_2 2 = 2 + 1 = 3$ $\checkmark$
\ans{x = 4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$A = 100(0.5)^{\,t/12}$ mg; find $t$ when $A = 30$ mg.}

\textbf{Solution:} Divide by $100$ first so the power stands alone, then take
logarithms:
\begin{align*}
  (0.5)^{t/12} &= 0.3 \\
  \frac{t}{12}\ln 0.5 &= \ln 0.3 \\
  \frac{t}{12} &= \frac{\ln 0.3}{\ln 0.5} = 1.736965\ldots \\
  t &= 20.8435\ldots
\end{align*}
Rounded to two decimal places, \ans{20.84 \text{ days}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Solve $9^{\,x+1} = 27^{\,x}$.}

\textbf{Solution:} Both $9$ and $27$ are powers of $3$, so rewrite everything in
base $3$:
\[ \left(3^2\right)^{x+1} = \left(3^3\right)^{x}
   \quad\Longrightarrow\quad 3^{\,2x+2} = 3^{\,3x} . \]
Equal bases give $2x + 2 = 3x$. Check: $9^3 = 729 = 27^2$ $\checkmark$
\ans{x = 2}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Challenge: $\log_5(2x+3) - \log_5(x-1) = 1$. (a) Exact $x$.
(b) Decimal. (c) The domain condition.}

\textbf{Solution (a):} Condense with the quotient rule, then switch form:
\begin{align*}
  \log_5\!\frac{2x+3}{x-1} &= 1 \\
  \frac{2x+3}{x-1} &= 5^1 = 5 \\
  2x + 3 &= 5x - 5 \\
  3x &= 8 .
\end{align*}
\ans{x = \tfrac{8}{3}}

\textbf{Solution (b):} $\tfrac{8}{3} = 2.666\ldots$ \ans{2.67}

\textbf{Solution (c):} \emph{Instructor-judged.} Both inputs must be positive:
$2x + 3 > 0$ gives $x > -\tfrac{3}{2}$, and $x - 1 > 0$ gives $x > 1$. Together
the domain is $x > 1$. Since $\tfrac{8}{3} \approx 2.67 > 1$, the solution is
legitimate. The condensing step is only an identity on that domain, so a
candidate outside it would satisfy the equation condensing produced while making
the original expression undefined --- it would have to be discarded. Full credit
states $x > 1$ and confirms the solution meets it.
