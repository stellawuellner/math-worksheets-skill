\sstitleblock{Adding, Subtracting, and Multiplying Polynomials}

\noindent\small Adding and subtracting are about \emph{matching} terms;
multiplying is about \emph{pairing} them. Decide which of the two you are
doing before you write anything, because the rules for exponents are opposite:
like terms keep their exponent, multiplied terms add theirs.
\normalsize

\skillheading{1. Adding polynomials}
\begin{formulabox}
\textbf{Rule:} drop the parentheses and combine like terms.\\[3pt]
Like terms have \emph{identical} variable parts --- same letters, same
exponents. $4x^2$ and $9x^2$ are like; $4x^2$ and $4x$ are not; $x^2y$ and
$xy^2$ are not. Only the coefficients are added; the variable part never
changes.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $(2x^2 + 3x - 1) + (4x^2 - 5x + 6)$
\step{Addition changes no signs, so this is purely a sorting job --- group the
terms by degree.}
\step{$2x^2 + 4x^2 = 6x^2$; \ $3x - 5x = -2x$; \ $-1 + 6 = 5$.}
\par
$\ans{6x^2-2x+5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $(x^2 - 7x + 2) + (3x^2 + 7x - 9)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x^2-7}$}
\end{tryitbox}

\skillheading{2. Subtracting polynomials}
\begin{formulabox}
\textbf{Rule:} $A - B = A + (-B)$. Distribute the minus sign to
\emph{every} term of $B$, then add as usual.
\[ -(ax^2 + bx + c) = -ax^2 - bx - c \]
Writing the rewritten sum on its own line is what prevents the classic
one-term-only sign error.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $(5x^2 - 2x + 8) - (3x^2 + 4x - 1)$
\step{A subtraction is an addition in disguise, so negate the whole second
polynomial before combining anything.}
\step{$(5x^2 - 2x + 8) + (-3x^2 - 4x + 1)$, so
$2x^2$, \ $-6x$, \ $9$.}
\par
$\ans{2x^2-6x+9}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $(4x^2 + x - 3) - (x^2 - 5x + 7)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^2+6x-10}$}
\end{tryitbox}

\skillheading{3. Multiplying by a monomial}
\begin{formulabox}
\textbf{Rule:} distribute the monomial across every term, multiplying
coefficients and \emph{adding} exponents on each variable separately.
\[ x^m \cdot x^n = x^{m+n} \]
A negative monomial flips the sign of every term it touches --- all of them,
not just the first.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $3x^2(2x - 5)$
\step{One factor is a single term, so the distributive property applies
directly --- no FOIL needed.}
\step{$3x^2 \cdot 2x = 6x^3$ \ (exponents $2 + 1 = 3$); \
$3x^2 \cdot (-5) = -15x^2$.}
\par
$\ans{6x^3-15x^2}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $-2x(4x^2 - 3x + 1)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-8x^3+6x^2-2x}$}
\end{tryitbox}

\skillheading{4. Multiplying two binomials}
\begin{formulabox}
\textbf{Rule:} every term of the first factor times every term of the second
--- four products, then combine the two middle ones.
\[ (a+b)(c+d) = ac + ad + bc + bd \]
\textbf{(!)} $(x+k)^2$ is $(x+k)(x+k)$, which is $x^2 + 2kx + k^2$ --- never
$x^2 + k^2$. The middle term is the half that gets dropped.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $(x + 3)(x - 7)$
\step{Both factors have two terms, so there are four products to write before
anything can be combined.}
\step{$x^2 - 7x + 3x - 21$, and the middle terms give $-7x + 3x = -4x$.}
\par
$\ans{x^2-4x-21}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $(2x - 1)(x + 5)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x^2+9x-5}$}
\end{tryitbox}
