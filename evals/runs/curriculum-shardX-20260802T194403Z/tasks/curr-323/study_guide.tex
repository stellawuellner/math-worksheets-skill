\sstitleblock{Right Triangles in Elevation and Depression Problems}

\begin{formulabox}
\textbf{SOH-CAH-TOA, and how to pick one.} Label the triangle from the angle you
know or want: \textbf{opp} is across from it, \textbf{adj} is the leg touching
it, \textbf{hyp} is the line of sight.
\[
\sin\theta=\frac{\text{opp}}{\text{hyp}}\qquad
\cos\theta=\frac{\text{adj}}{\text{hyp}}\qquad
\tan\theta=\frac{\text{opp}}{\text{adj}}
\]
Pick the ratio that uses the side you \emph{have} and the side you \emph{want};
the third side never enters. Both angles are measured from the \textbf{horizontal}.
\end{formulabox}

\skillheading{1. Elevation, solving for a side (elevation-side)}

\begin{formulabox}
Looking \emph{up}. The angle sits at the observer, between the horizontal
ground and the line of sight.\\[2pt]
\textbf{Ground distance given} --- it is the \emph{adjacent} leg, so
height $=$ distance $\times \tan\theta$.\\[2pt]
\textbf{Slanted length given} (rope, string, wire) --- it is the
\emph{hypotenuse}, so height $=$ length $\times \sin\theta$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} From a point $60$ ft from the base of a mast, the angle of
elevation to the top is $25^\circ$. Find the height.
\step{The $60$ ft is measured along the ground, so it is adjacent to the angle
and the height is opposite it --- opposite over adjacent is tangent.}
\step{$h = 60\tan 25^\circ = 60(0.466307\ldots) = 27.9784\ldots$ ft}
$\Rightarrow\quad \ans{27.98\ \text{ft}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The angle of elevation to the top of a pole from a point $75$
ft away on level ground is $40^\circ$. Find the height, to the nearest
hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{62.93\ \text{ft}}$}
\end{tryitbox}

\skillheading{2. Elevation, solving for the angle (elevation-angle)}

\begin{formulabox}
Two sides known, angle wanted: write the ratio first, then undo it with the
inverse function.
\[
\tan\theta=\frac{\text{opp}}{\text{adj}}\quad\Longrightarrow\quad
\theta=\tan^{-1}\!\left(\frac{\text{opp}}{\text{adj}}\right)
\]
$\tan^{-1}$ (also written $\arctan$) is the \emph{inverse} tangent --- it is not
$1/\tan$. If a problem asks only for the ratio, leave it as a reduced fraction.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A path rises $5$ m over a horizontal run of $18$ m. Find its
angle of elevation.
\step{Both legs are known and the angle is the unknown, so form the
opposite-over-adjacent ratio and invert it rather than solving for a side.}
\step{$\tan\theta = \tfrac{5}{18} = 0.2777\ldots$, so
$\theta = \tan^{-1}(0.2777\ldots) = 15.5241\ldots^\circ$}
$\Rightarrow\quad \ans{15.5^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A ramp rises $7$ ft over a run of $24$ ft. Find its angle of
elevation, to the nearest tenth of a degree.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16.3^\circ}$}
\end{tryitbox}

\skillheading{3. Depression: move the angle into the triangle (depression-side)}

\begin{formulabox}
Looking \emph{down}. The angle of depression is measured at the observer from a
horizontal line, so it sits \emph{outside} the triangle. That horizontal is
parallel to the ground, so by alternate interior angles
\[
\text{angle of depression at the top} \;=\; \text{angle of elevation at the object.}
\]
Move it down to the object first --- then it is an ordinary elevation problem.
The height is opposite that angle and the ground distance is adjacent, so
$\text{distance} = \dfrac{\text{height}}{\tan\theta}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} From a tower $120$ m tall the angle of depression to a car is
$25^\circ$. How far is the car from the foot of the tower?
\step{The angle is outside the triangle, so transfer it to the car as an equal
angle of elevation before choosing any ratio.}
\step{The height is opposite and the distance adjacent, so
$\tan 25^\circ = \tfrac{120}{d}$ and
$d = \tfrac{120}{0.466307\ldots} = 257.3408\ldots$ m}
$\Rightarrow\quad \ans{257.34\ \text{m}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} From a cliff $90$ m high the angle of depression to a raft is
$18^\circ$. How far is the raft from the foot of the cliff?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{276.99\ \text{m}}$}
\end{tryitbox}

\skillheading{4. Two triangles sharing a leg (multi-triangle)}

\begin{formulabox}
Two objects sighted from one point make two right triangles with the \emph{same}
vertical leg. Solve each horizontal distance separately, then combine:
subtract for the gap between them, add if they are on opposite sides.
\textbf{Never subtract the angles.}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} From a cliff $40$ m high, the angles of depression to two
boats in line with the cliff are $45^\circ$ and $30^\circ$. How far apart are
they?
\step{One cliff, two sight lines, so treat it as two separate right triangles
that share the $40$ m leg rather than one triangle with a combined angle.}
\step{$d_{30} = \tfrac{40}{\tan 30^\circ} = 69.2820\ldots$ and
$d_{45} = \tfrac{40}{\tan 45^\circ} = 40$, so the gap is
$69.2820\ldots - 40 = 29.2820\ldots$ m}
$\Rightarrow\quad \ans{29.28\ \text{m}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} From a cliff $60$ m high the angles of depression to two boats
in line with the cliff are $50^\circ$ and $25^\circ$. Find the distance between
them.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{78.32\ \text{m}}$}
\end{tryitbox}
