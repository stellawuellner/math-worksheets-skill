\wstitleblock{Take Away and Write It}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Taking away means crossing out. Each crossed-out object is one taken
        away.
  \item Every take-away picture matches a subtraction equation:
        \ start $-$ taken away $=$ left.
  \item Count the objects that are \emph{not} crossed out to find what is left.
\end{itemize}

\problem[3cm]{The picture shows 6 apples. 2 are crossed out. How many apples
are left?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.8, line width=0.8pt]
  \foreach \x in {0,1,2,3,4,5} \fill (\x,0) circle (0.22);
  \foreach \x in {4,5} {
    \draw (\x-0.32,-0.32) -- (\x+0.32,0.32);
    \draw (\x-0.32,0.32) -- (\x+0.32,-0.32);
  }
\end{tikzpicture}
\end{center}
$6\;-\;2\;=$~\ansblank}

\problem[4cm]{Your turn to make the picture! Draw 7 circles. Cross out 3 of
them. How many circles are left?
\par\vspace{4pt}
$7\;-\;3\;=$~\ansblank}

\problem[3cm]{There are 8 counters. 3 are crossed out. Write the story
numbers in the equation, then write how many are left.
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.8, line width=0.8pt]
  \foreach \x in {0,1,2,3} \fill (\x,0.9) circle (0.22);
  \foreach \x in {0,1,2,3} \fill (\x,0) circle (0.22);
  \foreach \x in {1,2,3} {
    \draw (\x-0.32,0.58) -- (\x+0.32,1.22);
    \draw (\x-0.32,1.22) -- (\x+0.32,0.58);
  }
\end{tikzpicture}
\end{center}
\begin{center}
\scratchblank\ $-$ \scratchblank\ $=$~\ansblank
\end{center}}

\problem[3cm]{The ten-frame shows 9 dots. Cross out 4 dots. How many dots
are \emph{not} crossed out?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.55, line width=0.8pt]
  \draw (0,0) rectangle (5,2);
  \foreach \x in {1,...,4} \draw (\x,0) -- (\x,2);
  \draw (0,1) -- (5,1);
  \foreach \p in {(0.5,1.5),(1.5,1.5),(2.5,1.5),(3.5,1.5),(4.5,1.5),(0.5,0.5),(1.5,0.5),(2.5,0.5),(3.5,0.5)}
    \fill \p circle (0.17);
\end{tikzpicture}
\end{center}
$9\;-\;4\;=$~\ansblank}

\problem[4cm]{Challenge! The picture shows 10 balloons. Some popped, and now
7 balloons are left. Cross out balloons to show the ones that popped. How
many balloons popped?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.8, line width=0.8pt]
  \foreach \x in {0,1,2,3,4} \draw (\x,0.9) circle (0.24);
  \foreach \x in {0,1,2,3,4} \draw (\x,0) circle (0.24);
\end{tikzpicture}
\end{center}
$10\;-$~\ansblank~$=\;7$}
