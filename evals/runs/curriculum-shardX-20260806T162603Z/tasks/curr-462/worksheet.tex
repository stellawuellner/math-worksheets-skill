\begin{center}
Derivatives of inverse trigonometric functions and of inverse functions
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Inverse-trig rules with the chain rule built in:
  \[
    \frac{d}{dx}\arcsin u = \frac{u'}{\sqrt{1-u^2}}, \qquad
    \frac{d}{dx}\arccos u = \frac{-u'}{\sqrt{1-u^2}}, \qquad
    \frac{d}{dx}\arctan u = \frac{u'}{1+u^2}
  \]
  \item Inverse-function rule: if $f(a) = b$, then $\left(f^{-1}\right)'(b) = \dfrac{1}{f'(a)}$. Find $a$ first, then differentiate $f$, then take the reciprocal --- in that order.
  \item Leave answers exact. Simplify only what genuinely simplifies.
\end{itemize}
