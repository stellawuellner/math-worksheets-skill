\sstitleblock{Parametric, Polar, and Vector-Valued Calculus}

\noindent\small Every question here is one of four: a \emph{slope}, a
\emph{magnitude}, an \emph{intersection}, or an \emph{accumulation}. Name which
one before you write anything --- that choice, not the algebra, is what these
problems test.\normalsize

\skillheading{1. Parametric slope (parametric-derivative)}
\begin{formulabox}
\[ \frac{dy}{dx} = \frac{dy/dt}{dx/dt}, \qquad
   \frac{d^2y}{dx^2} = \frac{\frac{d}{dt}\!\left(\frac{dy}{dx}\right)}{dx/dt} \]
Horizontal tangent: $\frac{dy}{dt} = 0$ and $\frac{dx}{dt} \neq 0$.
Vertical tangent: $\frac{dx}{dt} = 0$ and $\frac{dy}{dt} \neq 0$. Both zero:
inconclusive, examine separately.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $x = t^2+t$, $y = t^3+2t$. Find $\frac{dy}{dx}$ at $t=1$.
\step{The curve is given through a parameter, so the slope is the ratio of the
two $t$-derivatives --- differentiate each separately first.}
\step{$\dfrac{dy}{dx} = \dfrac{3t^2+2}{2t+1}$, and at $t=1$ this is
$\dfrac{5}{3}$.}
$\ans{\frac{dy}{dx}\Big|_{t=1} = \tfrac{5}{3}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $x = 2t^2$, $y = t^3 - t$. Find $\frac{dy}{dx}$ at
$t = 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{11}{8}}$}
\end{tryitbox}

\skillheading{2. Vector motion: velocity, speed, acceleration}
\begin{formulabox}
$\mathbf{v}(t) = \langle x'(t),\, y'(t)\rangle$, \quad
speed $= |\mathbf{v}(t)| = \sqrt{x'(t)^2 + y'(t)^2}$, \quad
$\mathbf{a}(t) = \langle x''(t),\, y''(t)\rangle$.\\[3pt]
The particle is \textbf{speeding up} when $\mathbf{v}\cdot\mathbf{a} > 0$ and
slowing down when $\mathbf{v}\cdot\mathbf{a} < 0$ --- direction, not the sign of
$\mathbf{a}$ alone.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $\mathbf{r}(t) = \langle t^2 - t,\ t^3\rangle$. Find the
speed at $t=1$.
\step{Speed is a magnitude, so build the velocity vector first and then take its
length.}
\step{$\mathbf{v}(t) = \langle 2t-1,\ 3t^2\rangle$, so
$\mathbf{v}(1) = \langle 1, 3\rangle$ and
$|\mathbf{v}(1)| = \sqrt{1+9}$.}
$\ans{\sqrt{10} \approx 3.162}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $\mathbf{r}(t) = \langle 3t^2,\ t^3 - t\rangle$. Find the
speed at $t = 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\sqrt{265} \approx 16.279}$}
\end{tryitbox}

\skillheading{3. Where two polar curves meet (polar-intersection)}
\begin{formulabox}
Set $r_1(\theta) = r_2(\theta)$ and solve on $[0, 2\pi)$ --- that finds the
points reached at the \emph{same} angle.\\[3pt]
Then check the pole separately: if each curve hits $r=0$ at \emph{some} angle,
the curves also meet at the origin even though the angles differ.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Where do $r = 1+\cos\theta$ and $r = 3\cos\theta$ meet?
\step{Both radii are functions of $\theta$, so equal radius at equal angle is
the condition to solve.}
\step{$1+\cos\theta = 3\cos\theta \Rightarrow \cos\theta = \tfrac{1}{2}$.}
$\ans{\theta = \tfrac{\pi}{3},\ \tfrac{5\pi}{3}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find every $\theta$ in $[0, 2\pi)$ where $r = 2\sin\theta$
meets $r = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = \tfrac{\pi}{6},\ \tfrac{5\pi}{6}}$}
\end{tryitbox}

\skillheading{4. Accumulated quantities (accumulated-quantity)}
\begin{formulabox}
\[ \text{distance} = \int_a^b |\mathbf{v}(t)|\,dt, \qquad
   \text{polar area} = \frac{1}{2}\int_\alpha^\beta r^2\,d\theta, \qquad
   \frac{1}{2}\int_\alpha^\beta \left(r_{\text{out}}^2 - r_{\text{in}}^2\right) d\theta \]
Arc length of a parametric curve is the same integral as distance travelled.
Choose limits that trace the region or path exactly \emph{once}.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Area of one petal of $r = 2\sin(3\theta)$.
\step{This is an accumulation of area, so use $\frac{1}{2}\int r^2\,d\theta$
over one petal: $\theta$ from $0$ to $\frac{\pi}{3}$, where $r$ returns to $0$.}
\step{$\frac{1}{2}\int_0^{\pi/3} 4\sin^2(3\theta)\,d\theta
 = \int_0^{\pi/3}\left(1 - \cos(6\theta)\right) d\theta = \frac{\pi}{3}$.}
$\ans{\tfrac{\pi}{3} \approx 1.047}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A particle has $\mathbf{v}(t) = \langle 2t,\ 2t^2\rangle$.
Find the distance travelled on $0 \le t \le 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{2}{3}(10\sqrt{10}-1) \approx 20.415}$}
\end{tryitbox}
