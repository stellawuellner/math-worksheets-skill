\wstitleblock{Similar Triangles: Sides, Perimeter, and Area}{High-School Geometry \quad Similarity}{}

\begin{center}
\textit{Finding a scale factor and a missing side, scaling a perimeter by the
same factor, and scaling an area by the square of it}
\end{center}

\noindent\textbf{Before you start.} If two triangles are similar with scale
factor $k$ (large $\div$ small):
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item every pair of corresponding sides is in the ratio $k$;
\item the perimeters are in the ratio $k$ --- perimeter is built from sides;
\item the areas are in the ratio $k^2$ --- area is built from two dimensions.
\end{itemize}
Name the pair of corresponding parts you are using before you write a
proportion, and give every answer with its unit.

\vspace{0.2cm}

\problem[4cm]{$\triangle ABC \sim \triangle DEF$ with $AB = 6$ cm and
$DE = 15$ cm. Find the scale factor from $\triangle ABC$ to $\triangle DEF$.}

\problem{$\triangle ABC \sim \triangle PQR$, with $AB = 8$ cm, $BC = 12$ cm and
$PQ = 20$ cm. Find $QR$.
\par\vspace*{5cm}\answerline{cm}}

\problem{Two similar triangular garden beds have corresponding sides of $9$ cm
and $12$ cm on the plan. The smaller bed has a perimeter of $24$ cm on the
plan. Find the perimeter of the larger bed.
\par\vspace*{5cm}\answerline{cm}}

\problem{Two similar triangular tiles have corresponding sides of $10$ cm and
$15$ cm. The smaller tile has an area of $40$ cm$^2$. Find the area of the
larger tile.
\par\vspace*{5.5cm}\answerline{cm$^2$}}

\problem{In $\triangle ABC$, point $D$ lies on $\overline{AB}$ and point $E$
lies on $\overline{AC}$, with $\overline{DE} \parallel \overline{BC}$. Here
$AD = 6$ cm, $AB = 15$ cm and $BC = 20$ cm.

(a) Name the similarity criterion that makes
$\triangle ADE \sim \triangle ABC$, and say why it applies. \quad
(b) Find $DE$.
\par\vspace*{6cm}\answerline{cm}}

\problem{A student who is $6$ ft tall casts a shadow $4$ ft long. At the same
moment, a flagpole beside her casts a shadow $26$ ft long. The student, the
flagpole and their shadows form two similar right triangles. How tall is the
flagpole?
\par\vspace*{5cm}\answerline{ft}}

\problem{$\triangle ABC \sim \triangle XYZ$. The perimeter of $\triangle ABC$
is $45$ cm and the perimeter of $\triangle XYZ$ is $30$ cm. If $AB = 12$ cm,
find $XY$.
\par\vspace*{5cm}\answerline{cm}}

\problem{Two similar triangles have areas of $27$ cm$^2$ and $48$ cm$^2$. The
smaller triangle has a perimeter of $15$ cm. Find the perimeter of the larger
triangle.
\par\vspace*{6cm}\answerline{cm}}

\problem{Two similar triangles have perimeters of $30$ cm and $42$ cm. One side
of the smaller triangle measures $7$ cm. Find the length of the corresponding
side of the larger triangle.
\par\vspace*{5cm}\answerline{cm}}

\problem{\textbf{Challenge.} $\triangle ABC \sim \triangle DEF$. The perimeter
of $\triangle ABC$ is $36$ cm, the perimeter of $\triangle DEF$ is $54$ cm, and
the area of $\triangle ABC$ is $48$ cm$^2$.

(a) Find the scale factor from $\triangle ABC$ to $\triangle DEF$. \quad
(b) Find the area of $\triangle DEF$. \quad
(c) Explain why doubling every side of a triangle multiplies its area by four
rather than by two.
\par\vspace*{6.5cm}\answerline{cm$^2$}}
