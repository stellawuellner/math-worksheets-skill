\sstitleblock{Transformation Evidence for Congruence and Similarity}

\skillheading{Skill 1 --- Congruence evidence: equal lengths}

\begin{formulabox}
\textbf{Distance between two points:}
\[ d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2} \]
Translations, reflections and rotations are \textbf{rigid motions}: they never change a length.
So if a figure maps onto another by rigid motions, every pair of matching sides is equal and the
figures are congruent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the length of the segment from $(1, 2)$ to $(4, 6)$.
\step{Two coordinate points and a length wanted --- that is the distance formula.}
\step{Run $= 4 - 1 = 3$, rise $= 6 - 2 = 4$, so $d = \sqrt{9 + 16} = \sqrt{25}$.}
$\Rightarrow\quad d = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the length of the segment from $(0, 1)$ to $(6, 9)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Similarity evidence: one constant ratio}

\begin{formulabox}
\textbf{Rule:} \quad $k = \dfrac{\text{image length}}{\text{matching original length}}$\\[3pt]
Figures are \textbf{similar} when every pair of matching sides gives the \textit{same} $k$. They
are \textbf{congruent} only in the special case $k = 1$. One pair giving a different ratio ends
the argument: the figures are neither.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A side of $6$ matches a side of $9$. Find the ratio.
\step{Similarity is judged by a ratio, so divide the image length by the one it matches.}
\step{$k = \dfrac{9}{6}$, and dividing top and bottom by $3$ gives $\tfrac{3}{2}$.}
$\Rightarrow\quad k = \ans{\tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A side of $10$ matches a side of $4$. Find the ratio.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{2}{5}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Slope evidence: sides stay parallel}

\begin{formulabox}
\textbf{Slope:} \quad $m = \dfrac{y_2 - y_1}{x_2 - x_1}$\\[3pt]
A dilation multiplies every rise and every run by the same $k$, so slopes are unchanged and each
image side is parallel to the side it came from. Parallel matching sides mean equal matching
angles --- the angle half of the similarity argument.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the slope of the segment from $(1, 2)$ to $(5, 4)$.
\step{We want the direction of the segment, not its length, so use slope.}
\step{$m = \dfrac{4 - 2}{5 - 1} = \dfrac{2}{4}$, which reduces.}
$\Rightarrow\quad m = \ans{\tfrac{1}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope of the segment from $(2, 5)$ to $(6, 2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\tfrac{3}{4}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Naming the transformation}

\begin{formulabox}
\textbf{Coordinate rules:}\\[3pt]
reflection in the $x$-axis: $(x, y) \mapsto (x, -y)$ \quad$\bullet$\quad
reflection in the $y$-axis: $(x, y) \mapsto (-x, y)$\\[2pt]
rotation of $180^\circ$ about the origin: $(x, y) \mapsto (-x, -y)$ \quad$\bullet$\quad
translation: add the same pair to every point
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect $(3, 5)$ in the $x$-axis.
\step{Compare the wanted map with the rules above: reflecting in the $x$-axis keeps $x$ and flips $y$.}
\step{The $x$-coordinate stays $3$; the $y$-coordinate becomes $-5$.}
$\Rightarrow\quad \ans{(3, -5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rotate $(-2, 4)$ by $180^\circ$ about the origin.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2, -4)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} equal ratios prove \textit{similar}, not \textit{congruent}. Congruent is
the special case where the ratio is $1$ and the lengths themselves match.
\end{watchoutbox}
