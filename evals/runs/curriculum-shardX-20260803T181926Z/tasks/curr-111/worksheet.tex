\wstitleblock{Whole-Number Division: Quotients and Remainders}{Grades 4--5}{}

\begin{center}\small
Equal groups and arrays, remainders in context, estimating quotients, and checking with multiplication
\end{center}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Write every division answer in two parts: the \textbf{quotient} (how many
        full groups) and the \textbf{remainder} (what is left over).
  \item The remainder is always \textbf{smaller than the number you are dividing by}.
        If it is not, you can make one more group.
  \item Check any division by multiplying back:
        (number in each group) $\times$ (number of groups) $+$ (remainder).
  \item In a story, read the question again before you write the final answer ---
        sometimes the leftover means you need one more group, and sometimes it is
        just left over.
\end{itemize}
\normalsize

\problem[5cm]{\textbf{Circle equal groups.} The counters below show $27$.
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \foreach \r in {0,1,2}{\foreach \c in {0,...,8}{\fill[blue!55] (\c,\r) circle (0.2);}}
\end{tikzpicture}
\end{center}
Circle groups of $4$. How many full groups can you circle, and how many counters
are left outside the circles?\par
quotient \underline{\hspace{1.4cm}} \qquad remainder \underline{\hspace{1.4cm}}}

\problem[5cm]{\textbf{Build an array.} The tiles below show $38$.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,1,2,3}{\foreach \c in {0,...,8}{%
      \draw[thick, fill=orange!25] (\c*1.15,\r*1.15) rectangle ++(0.85,0.85);}}
  \foreach \c in {0,1}{\draw[thick, fill=orange!25] (\c*1.15,4.6) rectangle ++(0.85,0.85);}
\end{tikzpicture}
\end{center}
Rearrange them into rows of $5$. How many complete rows of $5$ can you make,
and how many tiles are left over?\par
quotient \underline{\hspace{1.4cm}} \qquad remainder \underline{\hspace{1.4cm}}}

\problem[4.5cm]{\textbf{Check with multiplication.} A baker has $45$ rolls and puts
$6$ rolls in each bag. After filling the bags there are $3$ rolls left on the tray.
\begin{center}\large
$6 \times \underline{\hspace{1.4cm}} \; + \; 3 \; = \; 45$
\end{center}
How many bags did the baker fill? Show the multiplication that proves it.}

\problem[4.5cm]{\textbf{Estimate first.} A club shares $632$ prize tickets equally
among $8$ members.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Round $632$ to the nearest hundred to get a friendly number.
  \item Divide your rounded number by $8$.
\end{itemize}
Write the rounded number and the estimated quotient.}

\problem{\textbf{Vans for the field trip.} $53$ students are going on a field trip.
Each van holds $8$ students.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Divide to find how many vans are completely full, and how many students are
        left without a seat.
  \item Then answer the question the trip organiser actually asked: how many vans
        must be booked?
\end{itemize}
\par\vspace*{5.5cm}\answerline{vans}}

\problem[5cm]{\textbf{Share the marbles.} The marbles below show $74$.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,...,9}{\foreach \c in {0,...,6}{\fill[violet!55] (\c*0.9,\r*0.55) circle (0.2);}}
  \foreach \c in {7,8,9,10}{\fill[violet!55] (\c*0.9,0) circle (0.2);}
\end{tikzpicture}
\end{center}
Nine friends share them equally. How many marbles does each friend get, and how
many are left over? Draw a ring around one friend's share.}

\problem[5cm]{\textbf{Partial quotients.} A collector puts $145$ stickers into an
album, $6$ stickers on each page. Fill in the table to find how many pages are
filled and how many stickers are left.
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Take away} & \textbf{Pages used} & \textbf{Stickers left} \\ \hline
\rule{0pt}{0.9cm} $6 \times 20 = 120$ & $20$ & \\ \hline
\rule{0pt}{0.9cm} & & \\ \hline
\rule{0pt}{0.9cm} & & \\ \hline
\end{tabular}
\end{center}
quotient \underline{\hspace{1.4cm}} \qquad remainder \underline{\hspace{1.4cm}}}

\problem[5cm]{\textbf{Find the mistake.} Nia divides $87$ by $4$ and writes
\begin{center}\large
$87 \div 4 = 20$ R $7$
\end{center}
Use the remainder rule to explain how you can tell she is wrong without dividing
again. Then write the correct quotient and remainder, and check them by
multiplying back.}

\problem[5cm]{\textbf{Estimate, then divide.} A library receives $1347$ new books
and puts $5$ on each shelf.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Round $1347$ to the nearest hundred, then divide to estimate the number of
        shelves.
  \item Now divide exactly. Write the quotient and the remainder.
\end{itemize}}

\problem{\textbf{Ribbon for bows.} A ribbon is $96$ centimetres long. Each bow uses
$7$ centimetres of ribbon.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item How many whole bows can be made, and how much ribbon is left over?
  \item The van problem needed one \emph{extra} van for the leftover students, but
        this problem does not need one extra bow. Explain in your own words why the
        two leftovers are treated differently.
\end{itemize}
\par\vspace*{6cm}\answerline{cm}}
