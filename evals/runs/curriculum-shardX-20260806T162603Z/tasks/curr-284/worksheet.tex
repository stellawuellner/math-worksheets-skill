\wstitleblock{Finding the Sign Errors in Parabola Transformations}{Algebra 1}{}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item In $y = a(x-h)^2 + k$ the vertex is $(h, k)$. The trap is the bracket: $x - 3$ moves the graph $3$ units \textbf{right}, and $x + 3$ moves it $3$ units \textbf{left} --- the opposite of how each one reads.
  \item The sign of $a$ decides the direction: $a < 0$ flips the parabola over and makes the vertex a maximum. The sign of $k$ raises or lowers it.
  \item Where a student's work is shown, it contains \textbf{exactly one} error. Find it before you compute anything.
  \item A single substitution checks a transformation: put one $x$-value into the equation and see whether the point you get matches the graph you expect.
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\problem[4.5cm]{The graph of $y = (x-5)^2$ touches the $x$-axis at exactly one point. Solve $(x-5)^2 = 0$ to find the $x$-coordinate of that point.}

\problem[5cm]{Find the $x$-intercepts of $y = (x+4)^2 - 1$ by solving $y = 0$.}

\problem[5cm]{For $y = -2(x-3)^2 + 5$, find the value of $y$ when $x = 1$.}

\problem[7cm]{Mia is asked for the value of $y = -(x+2)^2 + 7$ when $x = 1$. She writes:
\par\smallskip
\hspace{1cm} $y = (1+2)^2 + 7 = 9 + 7 = 16$
\par\medskip
(a) Name the sign error in Mia's work. Say which of her steps is the wrong one.
\par\smallskip
(b) Compute the correct value of $y$ when $x = 1$, and put it on the answer line.}

\problem[6cm]{Find the $x$-intercepts of $y = -x^2 + 6x - 5$ by solving $y = 0$.}

\problem[7.5cm]{Owen is asked for the $x$-intercepts of $y = 2(x+3)^2 - 8$. He writes:
\par\smallskip
\hspace{1cm} $2(x+3)^2 = 8$, \ so $(x+3)^2 = 4$, \ so $x + 3 = \pm 2$, \ so $x = 5$ and $x = 1$.
\par\medskip
(a) Name the sign error in Owen's last step. Everything before it is correct, so be precise about which step fails.
\par\smallskip
(b) Give the correct $x$-intercepts on the answer line.}

\problem[6cm]{The parent graph $y = x^2$ is reflected in the $x$-axis and then shifted $4$ units \emph{down}.
\par\medskip
(a) Write the equation of the new graph.
\par\smallskip
(b) Evaluate the new function at $x = 3$, and put that value on the answer line.}

\problem[8cm]{\textbf{Synthesis challenge.} \quad $y = -3(x+2)^2 + 12$
\par\medskip
(a) Write the vertex, and say whether the parabola opens upward or downward.
\par\smallskip
(b) Find the $x$-intercepts, and put them on the answer line.
\par\smallskip
(c) Explain what the sign of each of the three numbers $-3$, $+2$ and $+12$ does to the graph, and say which of those signs students most often reverse.}
