\wstitleblock{Handling the Ambiguous SSA Case Completely}{Precalculus}{}

\begin{center}\small\textit{Testing the height first, then reporting one triangle, two triangles, or none, and carrying both cases through applied problems}\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions.} Every problem gives two sides and an angle that is \emph{not} between them (SSA). Before solving, run the height test.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Side $a$ is opposite angle $A$, and $b$ is the other given side. The perpendicular from $C$ to the line through $A$ and $B$ has length $h = b\sin A$.
  \item $a < h$: no triangle. \ $a = h$: one right triangle. \ $h < a < b$: \textbf{two} triangles. \ $a \ge b$: one triangle.
  \item When two triangles exist, report \emph{both} answers. The law of sines gives the acute angle; its obtuse partner is $180^\circ$ minus it.
  \item Round lengths and angles to two decimal places, and state units.
\end{itemize}
\vspace{0.1cm}

\probfig{1}
\begin{itemize}[leftmargin=1.1cm, topsep=4pt, itemsep=0pt]

  \vspace*{2cm}
  \item[(b)] How many triangles fit the drawing? Justify your count by comparing $h$ with the given sides. \ansline

\end{itemize}}
