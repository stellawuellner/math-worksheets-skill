\vspace{0.1cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Each problem is missing one factor. Find the number that makes the equation true.
  \item Show how you found it: skip-count, draw equal groups, or write the multiplication fact you used.
  \item Every total in this worksheet is $100$ or less.
  \item Write your final answer on the answer line.
\end{itemize}

\problem[4cm]{A sticker sheet has $4$ equal rows. There are $24$ stickers on the sheet.
\par\vspace{0.12cm}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {1,...,4}{\foreach \j in {1,...,6}{\fill[blue!65] (0.36*\j,-0.36*\i) circle (2.6pt);}}
\end{tikzpicture}
\end{center}
\par How many stickers are in each row? \quad $4 \times \underline{\hspace{1.1cm}} = 24$}

\problem[4cm]{A garden has $5$ equal rows of seedlings. There are $35$ seedlings in the garden.
\par\vspace{0.12cm}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {1,...,5}{\foreach \j in {1,...,7}{\fill[green!55!black] (0.36*\j,-0.36*\i) circle (2.6pt);}}
\end{tikzpicture}
\end{center}
\par How many seedlings are in each row? \quad $5 \times \underline{\hspace{1.1cm}} = 35$}

\problem[4cm]{Tiles are laid in equal rows of $3$ tiles. Altogether there are $18$ tiles.
\par\vspace{0.12cm}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {1,...,6}{\foreach \j in {1,...,3}{\draw[fill=orange!35] (0.42*\j,-0.42*\i) rectangle ++(0.34,0.34);}}
\end{tikzpicture}
\end{center}
\par How many rows of tiles are there? \quad $\underline{\hspace{1.1cm}} \times 3 = 18$}
