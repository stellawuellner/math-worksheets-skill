\wstitleblock{Estimating Differences and Checking Answers}{Grades 2--3}{}

\begin{center}
\small estimating a difference by rounding to the nearest ten or the nearest hundred, and using the estimate to decide whether an exact answer is reasonable
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item To estimate a difference, round \emph{both} numbers first, then subtract the rounded numbers.
  \item Rounding to the nearest ten: look at the ones digit. Rounding to the nearest hundred: look at the tens digit. Below five rounds down; five or more rounds up.
  \item An estimate is a check, not the answer. If an exact answer lands far from the estimate, something has gone wrong.
\end{itemize}

\answerline{seats}}
