\wstitleblock{Solving Systems by Graphing}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Graph both lines, read the crossing point, and check it in both
equations}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Graph each line by plotting its $y$-intercept, then stepping by the
        slope. Draw both lines on the same grid.
  \item Read the coordinates of the crossing point, then substitute them into
        \emph{both} equations to check. Use the space under each grid for that
        check.
  \item If the two lines never cross, say so instead of guessing a point.
\end{itemize}
