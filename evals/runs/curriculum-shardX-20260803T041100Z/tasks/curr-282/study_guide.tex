\sstitleblock{Graphing Quadratics from Vertex or Standard Form}

\begin{formulabox}
\textbf{Vertex form} $y = a(x-h)^2 + k$ \quad vertex $(h,k)$, axis $x = h$.\\[2pt]
\textbf{Standard form} $y = ax^2 + bx + c$ \quad axis $x = -\frac{b}{2a}$,
$y$-intercept $c$; substitute the axis value to get the vertex height.\\[3pt]
\textbf{What $a$ does:} $a > 0$ opens up (vertex is a minimum), $a < 0$ opens
down (maximum); $|a| > 1$ narrower than $y = x^2$, $|a| < 1$ wider.\\[3pt]
\textbf{To sketch:} plot the vertex, draw the axis, plot the $y$-intercept,
mirror it across the axis, add the $x$-intercepts if there are any, and curve
through them.
\end{formulabox}

\skillheading{1. Graph from vertex form}

\begin{formulabox}
Read $h$ and $k$ straight off the equation --- but $(x+1)^2$ means
$(x-(-1))^2$, so $h = -1$. The sign inside the parentheses always flips. Then
$a$ tells you the direction and the width, and two symmetric points finish the
curve.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Sketch $y = (x+1)^2 - 9$ and state the vertex.
\step{The equation is already in vertex form, so no algebra is needed --- read
$h$ and $k$, then use symmetry for the rest of the curve.}
\step{$(x+1)^2 = (x-(-1))^2$ gives $h = -1$, and $k = -9$. Check:
$y(-1) = 0 - 9 = -9$. Since $a = 1 > 0$ it opens up, and setting $y = 0$ gives
$(x+1)^2 = 9$, so the $x$-intercepts are $-4$ and $2$.}
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=center, xmin=-6, xmax=5, ymin=-10, ymax=6,
  xtick={-6,-3,0,3}, ytick={-9,-6,-3,0,3,6},
  tick label style={font=\tiny}, width=6.2cm, height=5.2cm,
  enlargelimits=false, samples=60]
  \addplot[thick, blue, domain=-4.4:2.4] {(x+1)^2 - 9};
  \fill (axis cs:-1,-9) circle (1.7pt);
  \fill (axis cs:-4,0) circle (1.7pt);
  \fill (axis cs:2,0) circle (1.7pt);
\end{axis}
\end{tikzpicture}
\end{center}
Vertex height at $x = -1$: $\ans{-9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} For $y = -2(x-4)^2 + 3$, what is the $y$-value at the vertex,
and does the parabola open up or down?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ at $x = 4$; opens down}
\end{tryitbox}

\skillheading{2. Graph from standard form}

\begin{formulabox}
The axis of symmetry comes first: $x = -\frac{b}{2a}$. Keep both minus signs
--- the one in the formula and any minus on $b$ or on $a$. Substituting that
$x$ back into the equation gives the vertex height, and $c$ is the
$y$-intercept for free.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the vertex of $y = x^2 - 4x + 1$.
\step{Nothing factors nicely here, so go through the axis of symmetry rather
than hunting for intercepts.}
\step{$a = 1$, $b = -4$, so $x = -\dfrac{-4}{2(1)} = 2$. Substituting:
$y(2) = 4 - 8 + 1 = -3$.}
Vertex height at $x = 2$: $\ans{-3}$, so the vertex is $(2,-3)$ and the
$y$-intercept is $1$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the vertex of $y = x^2 + 6x + 5$. (Axis first, then
substitute.)\\[2pt]
\rotatebox{180}{\footnotesize check: axis $x = -3$, vertex height $\ans{-4}$}
\end{tryitbox}

\skillheading{3. Use the $x$-intercepts}

\begin{formulabox}
The $x$-intercepts are the solutions of $y = 0$. Factor when you can: from
$(x-p)(x-q)$ the intercepts are $p$ and $q$, and the axis of symmetry sits
exactly midway between them at $x = \frac{p+q}{2}$ --- an independent way to
find the vertex, and a free check on $-\frac{b}{2a}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the $x$-intercepts of $y = x^2 - 2x - 8$.
\step{The question asks where the graph meets the $x$-axis, which is the same
as solving $y = 0$ --- so factor rather than complete the square.}
\step{Two numbers with product $-8$ and sum $-2$ are $-4$ and $2$:
$x^2 - 2x - 8 = (x-4)(x+2)$, so $x - 4 = 0$ or $x + 2 = 0$.}
$x$-intercepts: $\ans{x = -2 \text{ and } x = 4}$ (so the axis is $x = 1$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the $x$-intercepts of $y = x^2 - x - 6$, then say where
the axis of symmetry is.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -2 \text{ and } x = 3}$, axis
$x = \frac{1}{2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $y = (x+5)^2$ has its vertex at $x = -5$, not $x = 5$.
And in $-\frac{b}{2a}$ the sign of $a$ counts too: for $y = -x^2 + 4x + 5$ the
axis is $x = -\frac{4}{-2} = 2$, not $-2$.
\end{watchoutbox}
