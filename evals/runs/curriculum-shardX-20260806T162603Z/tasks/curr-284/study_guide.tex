\sstitleblock{Getting Every Sign Right in a Transformed Parabola}

\skillheading{Skill 1 --- Which way does the bracket move it?}

\begin{formulabox}
\textbf{Rule:} \quad in $y = a(x-h)^2 + k$ the graph moves $h$ units sideways, and $h$ is the value that makes the bracket zero.\par\smallskip
$(x-3)$ gives $h = 3$: three units \emph{right}. $(x+3)$ gives $h = -3$: three units \emph{left}. The bracket always reads the opposite way round --- ask ``what makes it zero?'' instead of reading the sign.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the $x$-intercepts of $y = (x-4)^2 - 1$.
\step{The squared bracket is alone once the $1$ moves across, so take square roots --- and undo the bracket by doing the opposite of what it says.}
\step{$(x-4)^2 = 1$, so $x - 4 = \pm 1$, and undoing ``minus four'' means \emph{adding} four: $x = 4 \pm 1$.}
\par
$x = \ans{3}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the $x$-intercepts of $y = (x+1)^2 - 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the commonest error on this whole topic is answering $x = 1$ and $x = 5$ for $(x+3)^2 = 4$. You must \emph{subtract} $3$: the answers are $-1$ and $-5$.
\end{watchoutbox}

\skillheading{Skill 2 --- The minus in front, and the number outside}

\begin{formulabox}
\textbf{Order of operations decides the signs:} square the bracket \emph{first}, then multiply by $a$, then add $k$.\par\smallskip
A negative $a$ flips the parabola over, so the vertex becomes the highest point. A negative $k$ lowers the graph. Neither of them ever changes what is inside the bracket.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $y = -2(x-1)^2 + 6$, find $y$ when $x = 3$.
\step{This is a substitution, so nothing needs solving --- the only decision is the order, and the minus belongs to the $2$, which acts after the squaring.}
\step{$(3-1)^2 = 4$; then $-2 \cdot 4 = -8$; then $-8 + 6$.}
\par
$y = \ans{-2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $y = -(x+3)^2 + 10$, find $y$ when $x = -1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Checking a transformation with a point}

\begin{formulabox}
\textbf{The check:} once you have written a transformed equation, solve $y = 0$ or substitute one convenient $x$. If the answers do not sit symmetrically about the vertex you claimed, a sign is wrong.\par\smallskip
The two $x$-intercepts are always the same distance from the axis of symmetry --- that is the fastest way to catch a reversed bracket.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the $x$-intercepts of $y = -(x-3)^2 + 4$ and check them against the vertex.
\step{Set $y = 0$ and take roots; the point of the exercise is the check afterwards, so keep track of where the vertex is.}
\step{$(x-3)^2 = 4$ gives $x - 3 = \pm 2$, so $x = 3 \pm 2$. The vertex is at $x = 3$, and both answers are two units from it.}
\par
$x = \ans{1}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the $x$-intercepts of $y = -2(x+1)^2 + 8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{1}$}
\end{tryitbox}
