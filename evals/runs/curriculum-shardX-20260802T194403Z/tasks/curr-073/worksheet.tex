\wstitleblock{Unknown Factors and Related Division Facts}{Grade 2--3 --- Foundations of Division}{}

\noindent\small\textbf{Focus:} Arrays, unknown-factor equations, equal-group stories, and
related division facts.\par\vspace{2pt}
\textbf{Directions:} Every problem hides one factor. Write the multiplication equation with a
box or a question mark for the missing factor, think ``what times this gives the total?'', then
write the factor you found on the answer line. Draw or count if it helps.\normalsize

\vspace{0.2cm}

\problem[4cm]{Maya lines up her stickers in \textbf{4 equal rows}. She has \textbf{24} stickers in all.
\begin{center}\large
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$
\end{center}
Fill in the missing factor: \quad $4 \times \underline{\qquad} = 24$.\\[2pt]
How many stickers are in each row?}

\problem[4cm]{A gardener plants \textbf{28} seeds. Each row has \textbf{4} seeds, and every row is full.
\begin{center}\large
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\vdots$
\end{center}
\noindent(Only the first two rows are drawn.)\\[2pt]
Fill in the missing factor: \quad $\underline{\qquad} \times 4 = 28$.\\[2pt]
How many rows of seeds are there?}

\problem[4cm]{Find the unknown factor. Say the times table you are using out loud.
\begin{center}\Large $7 \times \underline{\qquad} = 56$\end{center}
What number belongs in the blank?}

\problem[4.5cm]{Nina has \textbf{45} stickers. She shares them equally among \textbf{5} friends, so
every friend gets the same number of stickers.\\[3pt]
Write the equation $5 \times \underline{\qquad} = 45$, then find the unknown factor.\\[2pt]
How many stickers does each friend get?}

\problem[4cm]{Sam already knows this multiplication fact: \quad $6 \times 8 = 48$.\\[3pt]
Use that fact --- do \textbf{not} start over --- to answer the division question below.
\begin{center}\Large $48 \div 8 = \underline{\qquad}$\end{center}
What is the quotient?}

\problem[4cm]{Find the unknown factor. The missing number is in front this time.
\begin{center}\Large $\underline{\qquad} \times 9 = 72$\end{center}
What number belongs in the blank?}

\problem[4.5cm]{A baking tray holds \textbf{42} muffin cups in equal rows. Each row has \textbf{6} cups.
The picture shows only the first two rows.
\begin{center}\large
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[4pt]
$\vdots$
\end{center}
Write $\underline{\qquad} \times 6 = 42$ and find the unknown factor.\\[2pt]
How many rows of muffin cups does the tray have?}

\problem[4.5cm]{A classroom has \textbf{6} tables, and every table has the same number of chairs.
Altogether there are \textbf{54} chairs.\\[3pt]
Write the equation $6 \times \underline{\qquad} = 54$, then solve it.\\[2pt]
How many chairs are at each table?}

\problem[5cm]{A mystery number times \textbf{7} equals \textbf{42}.\\[3pt]
\textbf{(Think first, then compute.)} Find the mystery number using the unknown-factor idea,
then multiply that same mystery number by \textbf{8}.\\[2pt]
What is the mystery number times 8?}

\problem[5.5cm]{A baker has \textbf{48} rolls. She puts \textbf{6} rolls in each bag, and then packs
\textbf{2} bags into each box.\\[3pt]
Work in two steps: first find how many bags she fills, then find how many boxes she fills.\\[2pt]
How many boxes does the baker fill?}
