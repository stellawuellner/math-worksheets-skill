\wstitleblock{Multiplication Three Ways}{Grade 3}{}
\begin{center}
\textit{moving among repeated addition, arrays, and products}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item The same multiplication idea can be written three ways: as \textbf{repeated addition}, as an \textbf{array}, and as a \textbf{product} (a multiplication fact).
  \item In each problem, move from the form you are given to the form the problem asks for, then find the total.
\end{itemize}

\problem[2.5cm]{Fill in the blanks: \quad $4 + 4 + 4 = {}$~\ansblank~${}\times 4 = {}$~\ansblank}

\problem[2.5cm]{Fill in the blanks: \quad $6 + 6 + 6 + 6 + 6 = {}$~\ansblank~${}\times 6 = {}$~\ansblank}

\problem[0pt]{Mia reads 8 pages of her book each day for 3 days. Write a multiplication fact for her reading, then find the total. \par\vspace{4pt}Multiplication fact: \ansline \par\vspace*{2.4cm} Total pages: \ansblank}

It has 3 rows with 6 dots in each row. Write a multiplication fact for the array, then count the dots. \par\vspace{4pt}Multiplication fact: \ansline \par\vspace*{2.4cm} Total dots: \ansblank}
