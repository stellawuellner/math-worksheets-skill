\sstitleblock{Area and Perimeter Without the Mix-Up}

% =================== SKILL 1: find-area ===================
\skillheading{Area: the Space Covered Inside}

\begin{formulabox}
\textbf{Rule:} Area of a rectangle $=$ length $\times$ width. Area is counted
in \emph{square} units: cm$^2$, m$^2$, square units.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle is 8 cm long and 3 cm wide. What is its area?
\step{The question asks how much space the rectangle covers inside, so this
is area: multiply length by width.}
\step{$8 \times 3 = 24$.}

Area: $\ans{24\ \text{cm}^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle is 9 cm long and 4 cm wide. What is its
area?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36\ \text{cm}^2}$}
\end{tryitbox}

% =================== SKILL 2: find-perimeter ===================
\skillheading{Perimeter: the Distance Around}

\begin{formulabox}
\textbf{Rule:} Perimeter of a rectangle $=$ add \emph{all four} sides:
$P = 2 \times (\text{length} + \text{width})$. Perimeter is counted in plain
length units: cm, m, ft.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle is 7 cm long and 2 cm wide. What is its
perimeter?
\step{The question asks for the distance around the edge, so add all four
sides --- two lengths and two widths.}
\step{$2 \times (7 + 2) = 2 \times 9 = 18$.}

Perimeter: $\ans{18\ \text{cm}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle is 5 m long and 4 m wide. What is its
perimeter?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18\ \text{m}}$}
\end{tryitbox}

% =================== SKILL 3: choose-area-or-perimeter ===================
\skillheading{Which One Does the Story Need?}

\begin{formulabox}
\textbf{Rule:} Ask: does the story \emph{cover} something (paint, carpet,
grass) --- that is \textbf{area} --- or \emph{go around} something (fence,
ribbon, frame) --- that is \textbf{perimeter}?
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Paint covers a wall that is 5 m long and 3 m tall. Area or
perimeter --- and how much?
\step{Paint covers the whole face of the wall, so this story needs area, not
the distance around.}
\step{$5 \times 3 = 15$.}

The paint covers $\ans{15\ \text{m}^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A string of lights goes around a window that is 4 m long and
2 m tall. Area or perimeter --- and how much?\\[2pt]
\rotatebox{180}{\footnotesize check: perimeter --- $\ans{12\ \text{m}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Check the unit of your answer! Area answers need
\emph{square} units (cm$^2$, m$^2$). Perimeter answers use plain length units
(cm, m). A square unit on a fence --- or a plain unit on a carpet --- means
the wrong tool was picked.
\end{watchoutbox}
