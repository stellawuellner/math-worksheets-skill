\begin{center}
\textit{Tax and tip added on, discounts and sale prices, and markups worked back
to the original price}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Every amount of money on this page is in \textbf{dollars}, and every
        percent is a percent \emph{of the price named in the same sentence} ---
        read which price that is before you multiply.
  \item Adding on (tax, tip, markup) multiplies by $1 + \dfrac{p}{100}$. Taking
        off (discount) multiplies by $1 - \dfrac{p}{100}$.
  \item Round money answers to the nearest cent and write the cents digits even
        when they are zeros.
\end{itemize}

\problem[5cm]{A book is priced at 24 dollars and the sales tax rate is
$5\%$.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the sales tax, in dollars: \ansblank
  \item[(b)] the total paid, in dollars: \ansblank
\end{itemize}}

\problem[5cm]{A meal costs 40 dollars before the tip, and the diners leave an
$18\%$ tip on that amount.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the tip, in dollars: \ansblank
  \item[(b)] the total paid, in dollars: \ansblank
\end{itemize}}

\problem[5cm]{A jacket is marked 60 dollars and the rack says $25\%$
off.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the discount, in dollars: \ansblank
  \item[(b)] the sale price, in dollars: \ansblank
\end{itemize}}

\problem[5cm]{A bicycle repair is billed at 132 dollars before tax, and the tax
rate is $6.5\%$. What is the total bill, in dollars?}

\problem[5cm]{A shop pays 35 dollars for a lamp and marks it up $40\%$ of what
it paid.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the markup, in dollars: \ansblank
  \item[(b)] the selling price, in dollars: \ansblank
\end{itemize}}

\problem[5.5cm]{A board game was priced at 48 dollars and is now selling for 36
dollars. What percent was taken off the original price?}

\problem[5cm]{A group leaves a $15\%$ tip on a restaurant bill of 86 dollars.
How much is the tip, in dollars?}

\problem[5.5cm]{A chair sells for 84 dollars, which is what the shop paid for it
plus a $20\%$ markup on that amount. How many dollars did the shop pay for the
chair?}

\problem[5.5cm]{A coat is on sale for 68 dollars after a $15\%$ discount. What
was the price, in dollars, before the discount?}

\problem[7cm]{A tent is marked 120 dollars. It is on sale at $30\%$ off, and a
sales tax of $8\%$ is charged on the sale price.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the sale price, in dollars: \ansblank
  \item[(b)] the sales tax, in dollars: \ansblank
  \item[(c)] the total paid, in dollars: \ansblank
  \item[(d)] Would charging the tax first and taking the discount second change
             the total? Explain in the work space.
