\wstitleblock{Adding Two-Digit Numbers with Place Value}{Grade 2--3 Arithmetic}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item Every problem comes with a picture. Use the picture to think, then
        write the matching number sentence underneath it.
  \item Add the \textbf{tens} with the tens and the \textbf{ones} with the ones.
        Never let a ten and a one get added together.
  \item Show your thinking in the space, then write the total on the
        answer line.
\end{itemize}

\vspace{0.15cm}
\noindent\textbf{How the pictures work.} A tall bar is one \emph{ten}. A small
square is one \emph{one}. An arrow on a number line is one \emph{jump}.

\problem[4.5cm]{\textbf{Base-ten blocks.} Count the tens and the ones in each
group, then add the two groups.
\par
\begin{center}
\begin{tikzpicture}[scale=0.8]
  \foreach \i in {0,1}{
    \draw[thick] (\i*0.5,0) rectangle (\i*0.5+0.3,2.0);
    \foreach \j in {1,...,9} {\draw (\i*0.5,\j*0.2) -- (\i*0.5+0.3,\j*0.2);}}
  \foreach \k in {0,1,2}{\draw[thick] (1.3+\k*0.3,0) rectangle (1.5+\k*0.3,0.2);}
  \node at (1.05,-0.5) {23};
  \node at (2.9,1.0) {$+$};
  \draw[thick] (3.5,0) rectangle (3.8,2.0);
  \foreach \j in {1,...,9} {\draw (3.5,\j*0.2) -- (3.8,\j*0.2);}
  \foreach \k in {0,...,4}{\draw[thick] (4.2+\k*0.3,0) rectangle (4.4+\k*0.3,0.2);}
  \node at (4.5,-0.5) {15};
\end{tikzpicture}
\end{center}
\par
Tens altogether: \underline{\hspace{1.2cm}} \qquad
Ones altogether: \underline{\hspace{1.2cm}}}

\problem[4.5cm]{\textbf{Base-ten blocks.} This one needs a trade. When ten ones
are together, circle them and swap them for one ten.
\par
\begin{center}
\begin{tikzpicture}[scale=0.8]
  \foreach \i in {0,1,2}{
    \draw[thick] (\i*0.5,0) rectangle (\i*0.5+0.3,2.0);
    \foreach \j in {1,...,9} {\draw (\i*0.5,\j*0.2) -- (\i*0.5+0.3,\j*0.2);}}
  \foreach \k in {0,...,7}{\draw[thick] (1.8+\k*0.3,0) rectangle (2.0+\k*0.3,0.2);}
  \node at (1.9,-0.5) {38};
  \node at (4.5,1.0) {$+$};
  \foreach \i in {0,1}{
    \draw[thick] (5.1+\i*0.5,0) rectangle (5.1+\i*0.5+0.3,2.0);
    \foreach \j in {1,...,9} {\draw (5.1+\i*0.5,\j*0.2) -- (5.1+\i*0.5+0.3,\j*0.2);}}
  \foreach \k in {0,...,6}{\draw[thick] (6.4+\k*0.3,0) rectangle (6.6+\k*0.3,0.2);}
  \node at (6.5,-0.5) {27};
\end{tikzpicture}
\end{center}
\par
Tens altogether: \underline{\hspace{1.2cm}} \qquad
Ones altogether: \underline{\hspace{1.2cm}}}

\problem[4.5cm]{\textbf{Make a friendly ten.} Rosa adds 39 and 24. She slides 1
over from the 24 so the first number becomes a ten. Finish her way and give
back what she borrowed at the end.
\par
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[thick,->] (0,0) -- (7.5,0);
  \foreach \x in {0,1,...,7}{\draw (\x,0.12) -- (\x,-0.12);}
  \node[below] at (0,-0.15) {39};
  \draw[thick,->] (0,0.15) to[out=60,in=120] (1,0.15);
  \node[above] at (0.5,0.4) {$+1$};
\end{tikzpicture}
\end{center}}

\problem[4cm]{\textbf{Estimate first.} Round each number to the nearest ten,
then add the two rounded numbers. Use the number line to decide which ten 58
is closer to.
\par
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[thick] (0,0) -- (7,0);
  \foreach \x in {0,...,7}{\draw (\x,0.1) -- (\x,-0.1);}
  \node[below] at (0,-0.15) {50};
  \node[below] at (7,-0.15) {60};
  \draw[thick,fill=black] (5.6,0) circle (0.09);
  \node[above] at (5.6,0.12) {58};
\end{tikzpicture}
\end{center}
\par
About how much is $58 + 34$?}
