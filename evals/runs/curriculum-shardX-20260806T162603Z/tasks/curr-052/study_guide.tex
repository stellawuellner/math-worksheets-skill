\sstitleblock{Numbers Three Ways}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Expanded form: stretch the number}

\begin{formulabox}
\textbf{Rule:} Write what each digit is \emph{worth}, joined by plus signs.
Hundreds digit $\times$ 100, tens digit $\times$ 10, ones digit as itself.
A place holding a zero is simply left out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write 352 in expanded form.
\step{Name what each digit is worth: three hundreds, five tens, two ones.}
\step{Three hundreds are worth 300, five tens are worth 50, two ones are worth 2.}
$\Rightarrow\quad 352 = \ans{300 + 50 + 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write 417 in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{400 + 10 + 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Word form: say it, then write it}

\begin{formulabox}
\textbf{Rule:} Say the hundreds first (``four hundred''), then the last two
digits together (``sixty-three''). Useful words:
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item twenty, thirty, forty, fifty, sixty, seventy, eighty, ninety
  \item Never write ``and'' inside a whole number.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write \emph{five hundred sixteen} in standard form.
\step{Turn each spoken part into a place: ``five hundred'' means 5 hundreds, ``sixteen'' means 1 ten and 6 ones.}
\step{Five hundreds, one ten, and six ones put a digit in every place.}
$\Rightarrow\quad \ans{516}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write \emph{seven hundred thirty-four} in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{734}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Standard form: pack the parts back up}

\begin{formulabox}
\textbf{Rule:} Sort the parts by size (hundreds, tens, ones), then write one
digit per place. If a place has no part, write \textbf{0} there to hold it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $600 + 40 + 8$ in standard form.
\step{Match each part to its place: 600 fills the hundreds, 40 fills the tens, 8 fills the ones.}
\step{Write the digits in place order: six, four, eight.}
$\Rightarrow\quad \ans{648}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $200 + 5$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{205}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $200 + 5$ is \emph{not} 25. The tens place is empty,
so a zero must hold it: 2 hundreds, 0 tens, 5 ones.
\end{watchoutbox}
