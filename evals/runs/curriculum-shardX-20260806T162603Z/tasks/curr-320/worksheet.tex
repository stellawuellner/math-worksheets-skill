{\small\textit{missing sides, scale factor, perimeter ratio and area ratio}}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item If two triangles are similar with scale factor $k$ (target length over
        starting length), then every side is multiplied by $k$, the perimeter is
        multiplied by $k$, and the area is multiplied by $k^2$.
  \item Decide which of those three relationships a problem needs before you
        compute --- the sheet mixes them deliberately.
  \item Give lengths and perimeters to one decimal place where needed, and
        include the unit on every measurement answer.
\end{itemize}

\vspace{0.1cm}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 5$ cm, $BC = 7$ cm and
$DE = 15$ cm. Find the length of $\overline{EF}$, in centimetres.
\par\vspace*{4.6cm}
\answerline{cm}}

\problem{$\triangle PQR \sim \triangle STU$ with $PQ = 12$ in, $PR = 21$ in and
$ST = 8$ in. Find the length of $\overline{SU}$, in inches.
\par\vspace*{4.6cm}
\answerline{in}}

\problem[5cm]{Two similar triangular road signs are cut from the same pattern.
The smaller sign has a base of $6$ cm and the larger sign has a base of $15$ cm.
Find the scale factor from the smaller sign to the larger sign.}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 8$ m and $DE = 12$ m. The
perimeter of $\triangle ABC$ is $30$ m. Find the perimeter of $\triangle DEF$,
in metres.
\par\vspace*{4.6cm}
\answerline{m}}

\problem{Two similar triangular floor tiles have corresponding sides of $6$ cm
and $10$ cm. The smaller tile has an area of $27$ cm$^2$. Find the area of the
larger tile, in square centimetres.
\par\vspace*{4.8cm}
\answerline{cm$^2$}}

\problem{$\triangle ABC \sim \triangle DEF$. The perimeter of $\triangle ABC$ is
$24$ cm and the perimeter of $\triangle DEF$ is $60$ cm. If $AB = 7$ cm, find
the length of $\overline{DE}$, in centimetres.
\par\vspace*{4.8cm}
\answerline{cm}}

\problem[5cm]{Two similar triangular garden plots have areas of $48$ m$^2$ and
$108$ m$^2$. Find the ratio of a side of the larger plot to the matching side of
the smaller plot.}

\problem{Two similar triangular sails have areas of $50$ cm$^2$ and $128$
cm$^2$. The smaller sail has a perimeter of $30$ cm.
\par\vspace{0.1cm}
(a) Find the perimeter of the larger sail, in centimetres. \answerline{cm}
\par\vspace*{3.0cm}
(b) Justify why the perimeter ratio is the square root of the area ratio rather
than the area ratio itself.
\par\vspace*{2.8cm}}

\problem{A triangular window is enlarged so that its area becomes $6.25$ times
the area of the original.
\par\vspace{0.1cm}
(a) Find the scale factor applied to each side. \ansline
\par\vspace*{2.8cm}
(b) The original window has a perimeter of $84$ cm. Find the perimeter of the
enlarged window, in centimetres. \answerline{cm}
\par\vspace*{2.8cm}}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 14$ cm and $DE = 21$ cm.
The area of $\triangle ABC$ is $56$ cm$^2$.
\par\vspace{0.1cm}
(a) Find the area of $\triangle DEF$, in square centimetres.
\answerline{cm$^2$}
\par\vspace*{3.0cm}
(b) Explain why multiplying every side of a triangle by the same factor
multiplies its area by the square of that factor.
\par\vspace*{2.8cm}}
