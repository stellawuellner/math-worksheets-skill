\wstitleblock{Division Three Ways}{Grade 3}{}
\begin{center}
\textit{matching division stories, drawings, and equations}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item The same division idea can appear as a \textbf{story}, as a \textbf{drawing} of equal groups, or as an \textbf{equation}.
  \item In each problem, write the division equation that matches what you are given, then solve it.
  \item Write equations with the division sign, like $10 \div 2 = 5$.
\end{itemize}
