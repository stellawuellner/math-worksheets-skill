\wstitleblock{Putting Three Amounts in Order}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \emph{Least} means the smallest amount. \emph{Greatest} means the biggest amount.
  \item Write the three numbers on the lines, smallest one first.
  \item Counting up helps: the number you say first is the least.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3.2cm]{Ana, Ben and Cara counted their stickers.
\par\vspace{0.2cm}
\begin{center}
\begin{tabular}{|l|c|}
\hline
\textbf{Child} & \textbf{Stickers} \\
\hline
Ana & $5$ \\ \hline
Ben & $8$ \\ \hline
Cara & $3$ \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.15cm}
Write the three sticker counts in order from least to greatest.
\par\vspace{0.3cm}
least \quad \ansblank \quad \ansblank \quad \ansblank \quad greatest}
