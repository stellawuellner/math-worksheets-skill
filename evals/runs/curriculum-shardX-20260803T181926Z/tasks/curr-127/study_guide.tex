\sstitleblock{Common Denominators for Unlike Fractions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Rewriting a Fraction with a New Denominator}

\begin{formulabox}
\textbf{Rule --- multiply top and bottom by the same number.} \ Whatever you multiply the denominator by, multiply the numerator by too. The value of the fraction does not change, only the size of the pieces. To find the factor, ask: \emph{what times the old denominator gives the new one?}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $\dfrac{2}{5}$ with a denominator of $20$.

\step{The denominator has to become $20$, so first find what the old denominator was multiplied by.}
\step{$20 \div 5 = 4$, so multiply the numerator by $4$ as well: $2 \times 4$.}

$\Rightarrow\quad$ new numerator $= \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $\dfrac{3}{7}$ with a denominator of $21$. What is the new numerator?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never \emph{add} to the denominator. Going from sevenths to twenty-firsts is multiplying by $3$, not adding $14$, and the numerator must be multiplied by the very same $3$.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Adding and Subtracting Unlike Fractions}

\begin{formulabox}
\textbf{Rule.} \ You can only add or subtract fractions whose pieces are the same size. Find a common denominator (any common multiple works; the \emph{least} one keeps the numbers small), rewrite both fractions with it, then add or subtract the numerators only --- the denominator stays put. Simplify at the end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $\dfrac{1}{3} + \dfrac{1}{4}$

\step{Thirds and quarters are different-sized pieces, so nothing can be added until both are written over one common denominator.}
\step{$3$ and $4$ share no factor, so use $12$: \ $\dfrac{1}{3} = \dfrac{4}{12}$ and $\dfrac{1}{4} = \dfrac{3}{12}$. Now add the numerators.}

$\Rightarrow\quad \ans{\tfrac{7}{12}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $\dfrac{3}{4} - \dfrac{2}{5}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{7}{20}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Mixed Numbers with Unlike Denominators}

\begin{formulabox}
\textbf{Rule.} \ Handle the whole numbers and the fraction parts separately. Give the fraction parts a common denominator first. When subtracting, if the top fraction is the smaller one, \emph{regroup}: borrow one whole and add it to the fraction part as a full set of those pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $1\,\dfrac{1}{4} + 2\,\dfrac{2}{3}$

\step{Whole numbers add straight away; the fraction parts are quarters and thirds, so they need a common denominator before they can be combined.}
\step{$1 + 2 = 3$, and with denominator $12$: \ $\dfrac{1}{4} = \dfrac{3}{12}$, $\dfrac{2}{3} = \dfrac{8}{12}$, so the fraction part is $\dfrac{11}{12}$.}

$\Rightarrow\quad \ans{3\,\tfrac{11}{12}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $3\,\dfrac{1}{2} - 1\,\dfrac{1}{3}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2\,\tfrac{1}{6}}$}
\end{tryitbox}
