\sstitleblock{Rounding Multi-Digit Numbers to Named Places}

\noindent\small Rounding is a choice between two numbers: the multiple of the
named place just below, and the one just above. The halfway number decides
which one wins. Numbers here are written without commas.
\normalsize

\skillheading{1. Rounding on a number line}
\begin{formulabox}
\textbf{Three steps:} find the benchmark below and the benchmark above, mark
the halfway number (lower benchmark $+$ half the place), then see which side
the number falls on. On halfway exactly, round up.
\begin{center}
\begin{tikzpicture}[x=1.1cm]
  \draw (0,0) -- (4,0);
  \foreach \x in {0,1,2,3,4} {\draw (\x,-0.1) -- (\x,0.1);}
  \node[below=1pt, font=\scriptsize] at (0,-0.1) {lower};
  \node[below=1pt, font=\scriptsize] at (2,-0.1) {halfway};
  \node[below=1pt, font=\scriptsize] at (4,-0.1) {upper};
\end{tikzpicture}
\end{center}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Round $6382$ to the nearest thousand.
\step{The named place is thousands, so the only two candidates are the
thousands on either side --- $6000$ and $7000$.}
\step{Halfway is $6500$, and $6382$ has not reached it, so the lower benchmark
wins.}
\par
$\ans{6000}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Round $2471$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2500}$}
\end{tryitbox}

\skillheading{2. Rounding to any named place}
\begin{formulabox}
\textbf{Digit rule (the number line, written short):} look at the digit
\emph{one place to the right} of the named place. If it is $5$ or more, round
up; if it is $4$ or less, round down. Every digit right of the named place
becomes $0$; digits to the left stay put unless a round-up carries into them.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Round $35728$ to the nearest thousand.
\step{The thousands digit is $5$, so look one place to its right --- the
hundreds digit --- because that is where the halfway test lives.}
\step{The hundreds digit is $7$, which is $5$ or more, so round the thousands
up from $35$ to $36$ and make the last three digits zeros.}
\par
$\ans{36000}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Round $9963$ to the nearest hundred. (Watch the
carry.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10000}$}
\end{tryitbox}

\skillheading{3. Estimating a sum or difference}
\begin{formulabox}
\textbf{Rule:} round \emph{each} number to the named place first, then add or
subtract the rounded numbers. Never round the answer you already computed --- an
estimate is meant to be quick and mental.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Estimate $587 + 319$ to the nearest hundred.
\step{The point is a mental sum, so round both numbers before adding anything.}
\step{$587$ is past $550$, so it becomes $600$; $319$ is before $350$, so it
becomes $300$; then $600 + 300$.}
\par
$\ans{900}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Estimate $4128 + 3675$ to the nearest thousand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8000}$}
\end{tryitbox}

\skillheading{4. Rounding numbers from real situations}
\begin{formulabox}
\textbf{Rule:} read which place the question names before you round anything ---
``to the nearest thousand'' and ``to the nearest ten thousand'' give different
answers for the same count. Keep the unit (people, metres, dollars) in the
answer, and remember a rounded count is a report, not the exact number.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A city library lent $18642$ books last year. Report that
to the nearest thousand.
\step{The question names thousands, so the candidates are $18000$ and $19000$
--- not ten thousands.}
\step{Halfway is $18500$ and $18642$ is past it, so round up.}
\par
$\ans{19000}$ books
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A stadium seats $7409$ people. Report that to the nearest
hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7400}$}
\end{tryitbox}
