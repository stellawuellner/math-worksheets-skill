\sstitleblock{Vertex, Axis, Intercepts, and Opening Direction}

\skillheading{Skill 1 --- The axis of symmetry and the vertex}

\begin{formulabox}
\textbf{For $f(x) = ax^2 + bx + c$:} the axis of symmetry is the vertical line $x = -\dfrac{b}{2a}$.\par
\textbf{The vertex sits on that line}, so its first coordinate is that value of $x$ and its second coordinate is $f$ evaluated there. Find the line first, then substitute --- never the other way round.\par
\textbf{Sign care:} $b$ carries its own sign into the formula, so a negative $b$ gives a positive axis.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the axis and the vertex of $f(x) = x^2 - 8x + 11$.
\step{The vertex is the one point on the axis of symmetry, so locating the axis is what makes the vertex a single substitution.}
\step{$a = 1$, $b = -8$, so the axis is $x = -\frac{-8}{2(1)} = \frac{8}{2}$.}
\step{Substitute: $4^2 - 8(4) + 11 = 16 - 32 + 11$.}

axis $x = \ans{4}$, \quad vertex $y$-value $= \ans{-5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the axis and the vertex $y$-value of $f(x) = x^2 + 6x + 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: axis $x = \ans{-3}$, vertex $y = \ans{-7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Both kinds of intercept}

\begin{formulabox}
\textbf{$y$-intercept:} the output at input $0$, which for $ax^2 + bx + c$ is simply $c$.\par
\textbf{$x$-intercepts:} the inputs that make the output $0$. Factor and use the zero-product property --- a product is zero only when one of its factors is.\par
\textbf{The sign flip:} a factor $(x + 6)$ gives the intercept $x = -6$, not $6$. Solve each bracket rather than copying its number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find both kinds of intercept for $f(x) = x^2 - x - 6$.
\step{Setting the output to zero is what the $x$-intercepts mean, and setting the input to zero is what the $y$-intercept means --- two different substitutions.}
\step{$x^2 - x - 6 = (x + 2)(x - 3)$, so each bracket is zero at $x = -2$ and $x = 3$.}
\step{At input $0$ the value is the constant term, $-6$.}

$x$-intercepts $\ans{-2}$ and $\ans{3}$; \quad $y$-intercept $\ans{-6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find both kinds of intercept for $f(x) = x^2 - 2x - 15$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ and $\ans{5}$; $y$-intercept $\ans{-15}$}
\end{tryitbox}

\skillheading{Skill 3 --- Which way it opens, and what that means}

\begin{formulabox}
\textbf{Rule:} $a > 0$ opens \emph{upward}, so the vertex is the \textbf{lowest} point --- a minimum. $a< 0$ opens \emph{downward}, so the vertex is the \textbf{highest} point --- a maximum.\par
\textbf{Why:} the squared term dominates for inputs far from the vertex, and its sign decides whether those far-away outputs run up or down.\par
\textbf{Use it as a check:} a downward parabola whose vertex you computed as a low point means a slip somewhere.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = -2x^2 + 4x + 1$, find the vertex and say what kind of point it is.
\step{The leading coefficient is negative, so the curve opens downward and the vertex will be the maximum --- worth knowing before computing it.}
\step{Axis: $x = -\frac{4}{2(-2)} = -\frac{4}{-4}$.}
\step{Value there: $-2(1)^2 + 4(1) + 1 = -2 + 4 + 1$.}

axis $x = \ans{1}$, \quad maximum value $\ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = -x^2 + 6x - 5$, find the axis and the maximum value.\\[2pt]
\rotatebox{180}{\footnotesize check: axis $x = \ans{3}$, maximum $\ans{4}$}
\end{tryitbox}
