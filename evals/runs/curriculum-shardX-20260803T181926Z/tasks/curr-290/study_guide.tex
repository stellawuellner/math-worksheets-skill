\sstitleblock{Choosing and Combining Factoring Methods}

\begin{formulabox}
\textbf{Run this checklist in order on every expression.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
\item \textbf{1. Common factor?} Pull out the GCF of all terms, variables
included. Then look at the bracket and start again from step 2.
\item \textbf{2. How many terms?} Two terms with a minus sign: difference of
squares, $A^2 - B^2 = (A - B)(A + B)$. Three terms: trinomial.
\item \textbf{3. Perfect square?} $A^2 \pm 2AB + B^2 = (A \pm B)^2$ --- square
ends, middle term equal to $2AB$.
\item \textbf{4. Leading coefficient $1$ or not?} If $1$, find a factor pair of
the constant adding to the middle coefficient. If not, group on the product of
the outer coefficients.
\item \textbf{5. Factor again.} Every bracket gets the checklist too. A sum of
squares, $A^2 + B^2$, is where you stop.
\end{itemize}
\end{formulabox}

\skillheading{1. Common factor first, then look again}

\begin{formulabox}
The GCF step is not optional politeness --- it is what makes the later pattern
visible. A coefficient that is not a perfect square ($2x^2$, $3x^2$) or a term
of degree three or higher is the usual sign that a common factor is hiding a
pattern underneath.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $2x^3 - 8x$ completely.
\step{Two terms and a minus sign look like a difference of squares, but $2x^3$
is not a square --- so run checklist step 1 first and take the common factor
out.}
\step{Both terms divide by $2$ and both carry an $x$, so the GCF is $2x$:
\ $2x^3 - 8x = 2x(x^2 - 4)$, and $x^2 - 4 = (x - 2)(x + 2)$.}
$2x^3 - 8x = \ans{2x(x-2)(x+2)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $3x^3 - 27x$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x(x-3)(x+3)}$}
\end{tryitbox}

\skillheading{2. Trinomials with leading coefficient 1}

\begin{formulabox}
$x^2 + bx + c = (x + p)(x + q)$ where $p + q = b$ and $pq = c$. Read the signs
before hunting: if $c > 0$ both numbers share the sign of $b$; if $c < 0$ they
have opposite signs, and the bigger one carries the sign of $b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $x^2 + 7x + 12$.
\step{No common factor and the leading coefficient is $1$, so this is the
factor-pair case --- no grouping needed.}
\step{Need a pair multiplying to $12$ and adding to $7$: \ $3$ and $4$, since
$3 \cdot 4 = 12$ and $3 + 4 = 7$.}
$x^2 + 7x + 12 = \ans{(x+3)(x+4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $x^2 - 2x - 15$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-5)(x+3)}$}
\end{tryitbox}

\skillheading{3. Trinomials with a leading coefficient that is not 1}

\begin{formulabox}
For $ax^2 + bx + c$ with $a \neq 1$: find a factor pair of the product $ac$ that
adds to $b$, split the middle term into those two pieces, then factor by
grouping in pairs. The two brackets you get must match, which is the built-in
check that the split was right.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $3x^2 + 11x + 6$.
\step{No common factor and the leading coefficient is $3$, so factor pairs of
the constant alone will not work --- group on the product $ac$ instead.}
\step{$ac = 3 \cdot 6 = 18$, and $9 + 2 = 11$, so split:
$3x^2 + 9x + 2x + 6 = 3x(x + 3) + 2(x + 3)$.}
$3x^2 + 11x + 6 = \ans{(3x+2)(x+3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $2x^2 - 5x - 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2x+1)(x-3)}$}
\end{tryitbox}

\skillheading{4. Special forms: squares you can spot}

\begin{formulabox}
$A^2 - B^2 = (A - B)(A + B)$ and $A^2 \pm 2AB + B^2 = (A \pm B)^2$. Squares to
know on sight: $4, 9, 16, 25, 36, 49, 64, 81, 100$; also $4x^2 = (2x)^2$,
$9x^2 = (3x)^2$, $x^4 = (x^2)^2$. Recognising a special form skips all trial
and error.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $4x^2 - 20x + 25$.
\step{Three terms whose first and last are perfect squares --- test the
perfect-square pattern before grouping, because it is far quicker when it
works.}
\step{$A = 2x$ and $B = 5$; the pattern needs a middle term of
$2AB = 2 \cdot 2x \cdot 5 = 20x$, and the trinomial has $-20x$, so the bracket
is a difference.}
$4x^2 - 20x + 25 = \ans{(2x-5)^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $9x^2 - 24x + 16$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3x-4)^2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A correct product is not always a complete
factorization. $(2x + 4)(x + 2)$ really does multiply back to $2x^2 + 8x + 8$,
but $2x + 4$ still hides a common factor of $2$ --- the complete answer is
$2(x + 2)^2$. Taking the GCF out first would have prevented it.
\end{watchoutbox}
