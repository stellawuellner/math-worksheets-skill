\sstitleblock{Computing Distance and Midpoint from Coordinates}

\skillheading{Skill 1 --- The Distance Formula}

\begin{formulabox}
\textbf{Formula:}\quad $d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$\\[3pt]
\textbf{Why:} the horizontal leg is $|x_2 - x_1|$, the vertical leg is $|y_2 - y_1|$,
and the segment is the hypotenuse --- so this \emph{is} the Pythagorean theorem written
in coordinates.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item Order does not matter: squaring removes the sign, so $A$ to $B$ equals $B$ to $A$.
  \item Leave $\sqrt{\ }$ exact unless a decimal is requested; simplify with the largest perfect-square factor.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the distance from $(2, 1)$ to $(10, 7)$.
\step{Two points and a length wanted, so build the right triangle: subtract the $x$'s for one leg and the $y$'s for the other.}
\step{Legs $10 - 2 = 8$ and $7 - 1 = 6$, so $d = \sqrt{8^2 + 6^2} = \sqrt{64 + 36} = \sqrt{100} = \ans{10}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the distance from $(-1, 3)$ to $(3, 6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 2 --- The Midpoint Formula}

\begin{formulabox}
\textbf{Formula:}\quad $M = \left( \dfrac{x_1 + x_2}{2},\ \dfrac{y_1 + y_2}{2} \right)$\\[3pt]
\textbf{Why:} a midpoint is not a length --- it is the \emph{average} of the endpoints,
one coordinate at a time. So you \textbf{add} here, where the distance formula subtracts.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item The answer is a \textbf{point} $(x, y)$, never a single number.
  \item Each coordinate of $M$ must land between the two given ones --- a fast sanity check.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the midpoint of $(-4, 7)$ and $(6, 1)$.
\step{The point halfway along a segment is the average of the endpoints, so average the $x$'s and the $y$'s separately.}
\step{$\dfrac{-4 + 6}{2} = 1$ and $\dfrac{7 + 1}{2} = 4$, and each lands between its endpoints, so $\ans{M = (1,\ 4)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the midpoint of $(3, -8)$ and $(9, 2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{M = (6,\ -3)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Using Midpoints in a Coordinate Proof}

\begin{formulabox}
\textbf{The move:} to prove two segments \textbf{bisect each other}, compute the midpoint
of each. If the two midpoints are the \emph{same point}, that point lies on both segments
and halves both, which is exactly what bisecting means.\par
Related coordinate claims: equal distances prove congruent sides (isosceles); a midpoint
plus a distance gives the length of a median.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Parallelogram $A(0,0)$, $B(8,2)$, $C(10,6)$, $D(2,4)$: do the
diagonals bisect each other?
\step{Bisecting each other means the two diagonals share one midpoint, so compute the midpoint of each diagonal and compare.}
\step{Diagonal $\overline{AC}$: $\left(\dfrac{0+10}{2},\ \dfrac{0+6}{2}\right) = \ans{(5,\ 3)}$, and diagonal $\overline{BD}$ gives the same point, so they bisect each other.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad One diagonal runs from $(-2, 1)$ to $(6, 9)$ and the other from
$(0, 7)$ to $(4, 3)$. Find the shared midpoint.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,\ 5)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not add the legs. $\sqrt{5^2 + 12^2} = 13$, not $5 + 12 = 17$.
And keep the two formulas apart: distance \emph{subtracts} then squares, midpoint
\emph{adds} then halves.
\end{watchoutbox}
