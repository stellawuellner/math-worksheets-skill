\sstitleblock{Identities, Multiple Angles, and Exact Solutions}

\noindent\small \textbf{Which job is it?} No equation to solve
$\Rightarrow$ simplify. A doubled angle $\Rightarrow$ convert it. An interval
given $\Rightarrow$ produce a complete exact solution set.\normalsize

\skillheading{1. Simplifying with the basic identities}

\begin{formulabox}
\textbf{Pythagorean:} \; $\sin^2\theta + \cos^2\theta = 1$, \;
$1 + \tan^2\theta = \sec^2\theta$, \; $1 + \cot^2\theta = \csc^2\theta$
(each also rearranges, e.g.\ $1 - \cos^2\theta = \sin^2\theta$).\\[3pt]
\textbf{Quotient / reciprocal:} \; $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$,
\; $\sec\theta = \dfrac{1}{\cos\theta}$, \; $\csc\theta = \dfrac{1}{\sin\theta}$.\\[3pt]
\textbf{Order of moves:} factor a common term first; look for a Pythagorean
form; only then rewrite everything in sines and cosines. Conjugate
denominators multiply to a difference of squares, which the Pythagorean
identity always collapses.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Simplify $\dfrac{\sec^2 x - 1}{\sec^2 x}$.
\step{The numerator is exactly one side of a rearranged Pythagorean identity, so substitute instead of expanding into sines and cosines straight away.}
\step{$\sec^2 x - 1 = \tan^2 x$, so the fraction is $\dfrac{\tan^2 x}{\sec^2 x} = \dfrac{\sin^2 x}{\cos^2 x}\cdot\cos^2 x$.}
$= \ans{\sin^2 x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Simplify $(1 - \cos^2 x)\csc x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\sin x}$}
\end{tryitbox}

\skillheading{2. Double-angle formulas: choosing the right form}

\begin{formulabox}
\[ \sin 2\theta = 2\sin\theta\cos\theta \qquad
   \tan 2\theta = \frac{2\tan\theta}{1 - \tan^2\theta} \]
\[ \cos 2\theta = \cos^2\theta - \sin^2\theta
   = 2\cos^2\theta - 1 = 1 - 2\sin^2\theta \]
\textbf{The cosine has three forms on purpose.} Pick the one written in the
function you already know, or the one that makes a neighbouring term collapse
($1 + \cos 2\theta = 2\cos^2\theta$ and $1 - \cos 2\theta = 2\sin^2\theta$).\\[3pt]
\textbf{Never} write $\sin 2\theta$ as $2\sin\theta$ --- that is not an
identity, and it can even leave the range of sine.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\sin\theta = \dfrac{5}{13}$ with $\theta$ in quadrant
I. Find $\cos 2\theta$.
\step{Only the sine is known, so choose the cosine form written purely in sine and no quadrant reasoning is needed --- the square removes the sign.}
\step{$\cos 2\theta = 1 - 2\sin^2\theta = 1 - 2\left(\dfrac{5}{13}\right)^{2} = 1 - \dfrac{50}{169}$.}
$\cos 2\theta = \ans{\,\dfrac{119}{169}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\cos\theta = \dfrac{3}{5}$. Find $\cos 2\theta$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\,\dfrac{-7}{25}}$}
\end{tryitbox}

\skillheading{3. Exact solution sets on an interval}

\begin{formulabox}
\textbf{The routine:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=1pt]
  \item \textbf{One function.} Use an identity to remove every other
        trigonometric function (and every multiple angle).
  \item \textbf{Factor, never divide.} Dividing by $\sin x$ throws away the
        solutions where $\sin x = 0$.
  \item \textbf{Reference angle, then quadrants.} Each factor gives a value;
        take its reference angle and place it in both quadrants where the sign
        is right.
  \item \textbf{Sweep the interval.} On $[0, 2\pi)$ list every angle, including
        $0$ itself, and keep them exact.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2\cos^2 x + \cos x - 1 = 0$ on $[0, 2\pi)$.
\step{One function appears but squared, so this is a quadratic in cosine --- factor it rather than isolating anything.}
\step{$(2\cos x - 1)(\cos x + 1) = 0$, so $\cos x = \dfrac{1}{2}$ (reference angle $\dfrac{\pi}{3}$, quadrants I and IV) or $\cos x = -1$.}
$\ans{x = \dfrac{\pi}{3}, \; \pi, \; \dfrac{5\pi}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2\sin^2 x + \sin x - 1 = 0$ on $[0, 2\pi)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \dfrac{\pi}{6}, \; \dfrac{5\pi}{6}, \; \dfrac{3\pi}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an interval question is not finished when you have one
angle --- each factor usually contributes two, and $0$ counts. And check that
every solution really lies in $[0, 2\pi)$: the right endpoint is excluded.
\end{watchoutbox}
