\sstitleblock{Completing Short Algebraic and Segment Proofs}

\skillheading{Skill 1 --- Naming the property behind each step}

\begin{formulabox}
\textbf{Properties of equality} --- one per line of an algebraic proof:\par
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=1pt]
  \item \textbf{Addition / Subtraction:} add or subtract the same quantity on both sides.
  \item \textbf{Multiplication / Division:} multiply or divide both sides by the same non-zero quantity.
  \item \textbf{Distributive:} $a(b + c) = ab + ac$ --- rewriting, not operating on both sides.
  \item \textbf{Substitution:} replace a quantity by one known to equal it.
\end{itemize}
Name the \emph{property}, not the arithmetic: ``divided by 3'' is what you did; ``division property of equality'' is why you were allowed to.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Prove $x = 8$ from $4x - 9 = 23$, naming a reason for each line.
\step{Each line of the proof undoes one operation, so I choose the property that names the operation I am undoing.}
\step{$4x - 9 = 23$ (given); $4x = 32$ (addition property of equality); then divide both sides by $4$ (division property of equality).}

$\Rightarrow\quad \ans{x = 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Prove a value of $x$ from $6x + 5 = 41$, naming a property for each line.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Segment addition as an equation}

\begin{formulabox}
\textbf{Segment Addition Postulate:} if $B$ is between $A$ and $C$, then $AB + BC = AC$.\par
\textbf{Definition of midpoint:} $M$ on $\overline{AB}$ with $\overline{AM} \cong \overline{MB}$.\par
\textbf{Definition of congruent segments:} $\overline{AM} \cong \overline{MB}$ means $AM = MB$ --- the step that turns a congruence mark into an equation you can solve.\par
The order matters in a proof: betweenness gives the sum; the definitions give the equalities.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $B$ is between $A$ and $C$, with $AB = 3x$ m, $BC = x + 2$ m and $AC = 26$ m. Find $x$.
\step{The stated betweenness is exactly the hypothesis of the Segment Addition Postulate, so the picture becomes the one equation $AB + BC = AC$.}
\step{$3x + (x + 2) = 26$, so $4x + 2 = 26$ and $4x = 24$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B$ is between $A$ and $C$, with $AB = 2x$ m, $BC = x + 5$ m, $AC = 29$ m. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Overlapping segments in a two-column proof}

\begin{formulabox}
For four points in order, the two \emph{overlapping} distances share a middle stretch.\par
\textbf{The move:} add the shared middle stretch to both sides of the given equality (addition property of equality), then rename each sum with the Segment Addition Postulate, then substitute.\par
Running it backwards --- \emph{subtracting} the shared stretch --- takes you from equal overlapping distances to equal outer stretches.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Points $A$, $B$, $C$, $D$ lie in order with the outer stretches equal, $AB = CD = a$ cm and the middle stretch $BC = b$ cm. With $a = 5$ and $b = 8$, find the overlapping distance $AC$.
\step{Adding the shared middle stretch to each equal outer stretch keeps the two sides equal, and the Segment Addition Postulate names those sums $AC$ and $BD$ --- so computing one of them computes both.}
\step{$AC = a + b = 5 + 8$, and $BD = b + a$ is the same total.}

$\Rightarrow\quad \ans{13}$ cm, and $BD$ equals it.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same picture with $a = 6.5$ cm and $b = 9$ cm. Find $AC$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15.5}$}
\end{tryitbox}
