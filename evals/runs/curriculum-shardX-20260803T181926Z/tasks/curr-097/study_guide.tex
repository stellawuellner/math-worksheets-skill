\sstitleblock{Reading and Making Line Plots}

\skillheading{Skill 1 --- Reading how many from a stack of X's}

\begin{formulabox}
\textbf{A line plot holds two kinds of number.} The numbers written \emph{along the bottom}
are the thing that was counted or measured. The \textsf{X}'s stacked \emph{above} a number
say how many times that thing happened. One \textsf{X} is always one thing.\par
\smallskip
So ``how many'' questions are answered by counting a stack, never by reading a label. A
number with an empty space above it is not a mistake: it means nothing landed there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many children own $2$ pets?
\begin{center}
\begin{tikzpicture}[x=0.62cm, y=0.30cm]
  \draw[thick] (0.4,0) -- (4.9,0);
  \foreach \v in {1,...,4} {\draw (\v,0.4) -- (\v,-0.4);
     \node[below, font=\small] at (\v,-0.4) {\v};}
  \foreach \v/\n in {1/3, 2/5, 3/2, 4/1} {%
     \foreach \i in {1,...,\n} {\node[font=\small] at (\v,\i) {\textsf{X}};}}
  \node[font=\small] at (2.65,-3.1) {Pets owned};
\end{tikzpicture}
\end{center}
\step{The question asks how many \emph{children}, so the answer is the size of a stack ---
find the $2$ along the bottom and count upwards, do not read the label.}
\step{Above the $2$: \textsf{X}, \textsf{X}, \textsf{X}, \textsf{X}, \textsf{X}.}

$\Rightarrow\quad \ans{5}$ children own $2$ pets.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same plot, how many children own $3$ pets?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$ children}
\end{tryitbox}

\skillheading{Skill 2 --- Making a line plot from a list}

\begin{formulabox}
\textbf{Three steps.} (1) Find the smallest and largest number on the list, and write every
whole number from one to the other along a line --- in order, evenly spaced, \emph{including
the ones that never appear}. (2) Go along the list and draw one \textsf{X} above each number,
crossing it off the list as you go. (3) Count the \textsf{X}'s: there must be exactly as many
as there were numbers on the list.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plot $3, 4, 5, 4, 3, 4, 6, 5$. How many \textsf{X}'s sit above $4$?
\begin{center}
\begin{tikzpicture}[x=0.62cm, y=0.30cm]
  \draw[thick] (2.4,0) -- (6.9,0);
  \foreach \v in {3,...,6} {\draw (\v,0.4) -- (\v,-0.4);
     \node[below, font=\small] at (\v,-0.4) {\v};}
  \foreach \v/\n in {3/2, 4/3, 5/2, 6/1} {%
     \foreach \i in {1,...,\n} {\node[font=\small] at (\v,\i) {\textsf{X}};}}
  \node[font=\small] at (4.65,-3.1) {Number rolled};
\end{tikzpicture}
\end{center}
\step{The list runs from $3$ to $6$, so the line needs $3, 4, 5, 6$ --- then one \textsf{X}
per number on the list, crossing each off as it is drawn.}
\step{Crossing off the $4$'s one at a time gives three of them, and the eight \textsf{X}'s on
the plot match the eight numbers on the list.}

$\Rightarrow\quad \ans{3}$ \textsf{X}'s sit above the $4$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same plot, how many \textsf{X}'s sit above $5$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$ \textsf{X}'s}
\end{tryitbox}

\skillheading{Skill 3 --- Totalling and comparing stacks}

\begin{formulabox}
\textbf{How many altogether?} Add the sizes of all the stacks --- that is the total number of
\textsf{X}'s, not the number of marks along the bottom.\par
\smallskip
\textbf{How many more?} Subtract one stack size from the other.\par
\smallskip
\textbf{How much bigger / longer?} That answer is about the thing measured, so subtract the
two \emph{labels} from the bottom of the line instead.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad On the pets plot ($3$, $5$, $2$ and $1$ \textsf{X}'s above $1$, $2$,
$3$ and $4$), how many children were asked altogether?
\step{Every child put exactly one \textsf{X} on the plot, so counting the children means
adding up all four stacks.}
\step{Add the four stack sizes: $3 + 5 + 2 + 1$.}

$\Rightarrow\quad \ans{11}$ children were asked.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same pets plot, how many more children own $2$ pets than own
$3$ pets?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ more children}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the commonest line-plot mistake is answering with a label instead of
a stack. Before writing anything down, ask yourself: \emph{is this answer counting things, or
measuring them?} Counting means a stack of \textsf{X}'s; measuring means a number from the
bottom of the line.
\end{watchoutbox}
