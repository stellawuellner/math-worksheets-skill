\sstitleblock{Teen Numerals and Number Words}

\skillheading{Skill 1 --- From a number word to a numeral}

\begin{formulabox}
\textbf{Rule:} A teen word has two pieces: the ones piece, then \emph{-teen} for the one ten.\par
Write the ten first, then count the ones on: \quad numeral $=$ ten $+$ ones.\par
\textbf{Word bank:} eleven, twelve, thirteen, fourteen, fifteen, sixteen, seventeen, eighteen, nineteen.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write the numeral for \emph{sixteen}.
\step{The word tells me the two parts of the number, so I split it into its ones piece and its teen piece before writing a single digit.}
\step{\emph{Six} is the ones and \emph{-teen} is the one ten, so the number is ten and six more.}

$10 + 6 = \ans{16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write the numeral for \emph{fourteen}.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The word says the ones piece first, but the numeral writes the ten first. \emph{Fifteen} is not $51$.
\end{watchoutbox}

\skillheading{Skill 2 --- From a numeral to a number word}

\begin{formulabox}
\textbf{Rule:} Read the ones digit, say its name, then add \emph{-teen} for the ten.\par
Three words change their sound and must simply be learned: eleven, twelve, and thirteen (not ``threeteen'').
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write the number word for the numeral $17$.
\step{The word is built from the ones digit, so before I can say anything I have to find how many ones the numeral holds.}
\step{Taking the one ten out leaves $17 - 10$ ones, and the name of that many ones is \emph{seven}. Adding the teen ending gives \emph{seventeen}.}

\emph{seventeen}, with ones part $\ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write the number word for the numeral $13$.\\[2pt]
\rotatebox{180}{\footnotesize check: thirteen, ones part $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the ones in a teen numeral}

\begin{formulabox}
\textbf{Rule:} In every teen numeral the left digit is the one ten and the right digit is the ones.\par
To check it, take the ten out: \quad ones $=$ numeral $-$ ten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $18$ is one ten and how many ones?
\step{I want the leftover once one whole ten is set aside, so I subtract that ten rather than counting all the way up from one.}
\step{Taking the ten out of the numeral leaves $18 - 10$, and the right-hand digit shows the same amount.}

$\ans{8}$ ones
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $11$ is one ten and how many ones?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$ one}
\end{tryitbox}

\skillheading{Skill 4 --- Comparing two teen numerals}

\begin{formulabox}
\textbf{Rule:} Every teen numeral holds exactly one ten, so the tens always match and cannot decide.\par
Compare the ones digits: the numeral with more ones is the greater number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$, $>$, or $=$ between $17$ and $12$.
\step{Both are teen numerals, so their tens are equal and comparing them tells me nothing; only the ones digits can settle it.}
\step{The ones digit of the first numeral is larger than the ones digit of the second, so the first number is greater and the wide end faces it.}

$\ans{17 > 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$ between $13$ and $18$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13 < 18}$}
\end{tryitbox}
