\sstitleblock{Subtracting Two-Digit Numbers with Place-Value Blocks}

\skillheading{Skill 1 --- Subtracting without opening a rod}

\begin{formulabox}
\textbf{Rule:} A tall rod is one ten; a small square is one.\par
Take the ones away first, then the tens. If the ones you have are enough, nothing needs opening --- cross out and count what is left.
\begin{center}
\begin{tikzpicture}[scale=0.22]
  \foreach \i in {0,...,2} {
    \draw[line width=0.6pt] (2*\i,0) rectangle (2*\i+1,10);
    \foreach \j in {1,...,9} { \draw[gray!65, very thin] (2*\i,\j) -- (2*\i+1,\j); }
  }
  \foreach \k in {0,...,3} { \draw[line width=0.6pt] (7,\k) rectangle (8,\k+1); }
\end{tikzpicture}
\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Work out $58 - 24$ with blocks.
\step{Before touching anything I check the ones: there are more ones here than I need to take away, so no rod has to be opened and I can just cross out.}
\step{Cross out four of the eight single squares, leaving four ones. Then cross out two of the five rods, leaving three rods. Three tens and four ones.}

$\ans{34}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Work out $76 - 43$ with blocks.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{33}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtracting when you must open a rod}

\begin{formulabox}
\textbf{Rule:} When there are not enough ones, \emph{open} one rod into ten ones. The amount does not change --- only how it is shown.\par
After opening: one rod fewer, ten ones more. Now take the ones away, then the tens.
\begin{center}
\begin{tikzpicture}[scale=0.22]
  \foreach \i in {0,...,2} {
    \draw[line width=0.6pt] (2*\i,0) rectangle (2*\i+1,10);
    \foreach \j in {1,...,9} { \draw[gray!65, very thin] (2*\i,\j) -- (2*\i+1,\j); }
  }
  \draw[line width=0.6pt, dashed] (6,0) rectangle (7,10);
  \foreach \k in {0,...,9} { \draw[line width=0.6pt] (9,\k) rectangle (10,\k+1); }
\end{tikzpicture}
\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Work out $45 - 18$ with blocks.
\step{The ones I have are fewer than the ones I must take away, so the first move is not subtraction at all --- it is opening a rod to get more ones.}
\step{Opening one rod leaves three rods and fifteen ones. Take eight ones away, leaving seven; take one rod away, leaving two rods. Two tens and seven ones.}

$\ans{27}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Work out $73 - 26$ with blocks.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{47}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimating a difference, then subtracting}

\begin{formulabox}
\textbf{Rule:} To estimate to the nearest ten, round \emph{both} numbers first, then subtract the rounded ones.\par
Look at the ones digit: below five rounds down, five or more rounds up.\par
The estimate is a check. If your exact answer lands far from it, something has gone wrong in one of the two.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $62 - 29$ to the nearest ten, then subtract exactly.
\step{An estimate is only worth doing if it is fast, so I round both numbers to tens and subtract those, keeping the exact work as a separate second pass.}
\step{Rounding gives sixty and thirty, and taking one from the other gives the estimate. Exactly: two ones cannot give up nine, so open a rod, leaving five rods and twelve ones; take nine ones, leaving three, then take two rods, leaving three.}

estimate $\ans{30}$ \qquad exact $\ans{33}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $85 - 38$ to the nearest ten, then subtract exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: estimate $\ans{50}$, exact $\ans{47}$}
\end{tryitbox}
