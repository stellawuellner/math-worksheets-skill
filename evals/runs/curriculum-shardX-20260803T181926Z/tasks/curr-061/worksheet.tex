\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A \textbf{ten rod} is a stick of ten little cubes. A \textbf{ones cube} is one little cube.
  \item Take away the tens first, then the ones. If there are not enough ones,
        \textbf{break one ten rod into ten ones} and try again.
  \item Show your thinking in the empty space --- draw, cross out, or write the numbers.
        Then write your answer on the line.
\end{itemize}
\normalsize

\problem[4.5cm]{\textbf{Take away with blocks.} The blocks below show $47$.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \foreach \i in {0,1,2,3}{%
    \draw[thick, fill=blue!8] (\i*0.45,0) rectangle ++(0.3,1.5);
    \foreach \j in {1,...,9}{\draw[gray!60] (\i*0.45,\j*0.15) -- ++(0.3,0);}}
  \foreach \k in {0,...,6}{%
    \draw[thick, fill=blue!8] (2.2+\k*0.35,0) rectangle ++(0.25,0.25);}
\end{tikzpicture}
\end{center}
Cross out $2$ ten rods and $3$ ones cubes to take away $23$.
How many blocks are left? \quad $47 - 23 = \underline{\hspace{1.2cm}}$}

\problem[4.5cm]{\textbf{Take away with blocks.} The blocks below show $68$.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \foreach \i in {0,1,2,3,4,5}{%
    \draw[thick, fill=blue!8] (\i*0.45,0) rectangle ++(0.3,1.5);
    \foreach \j in {1,...,9}{\draw[gray!60] (\i*0.45,\j*0.15) -- ++(0.3,0);}}
  \foreach \k in {0,...,7}{%
    \draw[thick, fill=blue!8] (3.1+\k*0.35,0) rectangle ++(0.25,0.25);}
\end{tikzpicture}
\end{center}
Cross out the blocks that show $35$. Write how many ten rods and how many ones
cubes are left, then write the number. \quad $68 - 35 = \underline{\hspace{1.2cm}}$}

\problem[4cm]{\textbf{Round to the nearest ten, then subtract.}
Estimate $61 - 28$.\par
Round each number to the nearest ten and write the two rounded numbers on the lines.
\begin{center}\large
\underline{\hspace{2cm}} $-$ \underline{\hspace{2cm}}
\end{center}
Now subtract the rounded numbers. About how big is the answer?}

\problem[4.5cm]{\textbf{Break a ten.} Find $52 - 27$ using expanded form.\par
Start with $52 = 50 + 2$ and $27 = 20 + 7$.
There are not enough ones to take $7$ from $2$, so break one ten:
$52 = 40 + 12$. Finish the subtraction and write the difference.}

\problem[4.5cm]{\textbf{Break a ten.} Find $83 - 46$ using expanded form.\par
Write $83$ and $46$ in expanded form, break one ten out of $83$ so there are
enough ones, then take away the tens and the ones.}

\problem[4.5cm]{\textbf{Estimate, then check.} Ravi is finding $85 - 47$.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item First estimate: round each number to the nearest ten and subtract.
  \item Then find the exact answer to $85 - 47$ by breaking a ten.
  \item Write both numbers, and say whether the exact answer is close to the estimate.
\end{itemize}}

\problem{\textbf{Sticker album.} Nadia had $74$ stickers. She glued $58$ of them
into her album. How many stickers does she have left?\par
Draw ten rods and ones cubes for $74$, cross out $58$, and write the number that is left.
\par\vspace*{5cm}\answerline{stickers}}

\problem[4.5cm]{\textbf{Find the mistake.} Theo works out $63 - 29$ like this:
\begin{center}\large
$60 - 20 = 40$ \qquad $9 - 3 = 6$ \qquad so $63 - 29 = 46$
\end{center}
Theo took the smaller ones digit away from the larger one instead of breaking a ten.
Explain what he should have done, then write the correct difference.}
