\sstitleblock{Integration by Parts: Choosing $u$ and $dv$ Strategically}

\noindent\small The formula is one line: $\int u\,dv = uv - \int v\,du$. All of
the skill is in the choice. A good choice makes $\int v\,du$ \emph{simpler} than
what you started with; a bad one makes it worse, and the fastest fix is to swap
$u$ and $dv$ and start again.\normalsize

\skillheading{1. Choosing $u$ and $dv$: one application}
\begin{formulabox}
\textbf{Formula:} $\displaystyle\int u\,dv = uv - \int v\,du$. \quad
\textbf{Tie-breaker (LIATE):} take $u$ from the earliest category present ---
\textbf{L}og, \textbf{I}nverse trig, \textbf{A}lgebraic, \textbf{T}rig,
\textbf{E}xponential --- and $dv$ from the latest.\\[3pt]
The reason behind the letters: $u$ should get \emph{simpler} when
differentiated, and $dv$ must be something you can actually integrate.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\int x e^{3x}\,dx$.
\step{A power times an exponential: differentiating $x$ removes it, and
$e^{3x}$ integrates easily, so $u = x$ and $dv = e^{3x}dx$.}
\step{$du = dx$ and $v = \frac{1}{3}e^{3x}$, so
$\int x e^{3x}dx = \frac{x}{3}e^{3x} - \frac{1}{3}\int e^{3x}dx$.}

$\ans{\dfrac{x}{3}e^{3x} - \dfrac{1}{9}e^{3x} + C}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\int x e^{-x}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-x e^{-x} - e^{-x} + C}$}
\end{tryitbox}

\skillheading{2. When a logarithm is present, it is always $u$}
\begin{formulabox}
\textbf{Rule:} $\ln x$ has no elementary antiderivative you can use as $v$, so
it can only be $u$ --- and that is lucky, because
$\frac{d}{dx}\ln x = \frac{1}{x}$ turns it into a power.\\[3pt]
If the logarithm stands alone, supply the missing factor: $\ln x = \ln x \cdot 1$,
so $dv = dx$ and $v = x$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\int x^{3}\ln x\,dx$.
\step{Taking $u = x^{3}$ would need $v = \int \ln x\,dx$, another parts problem;
taking $u = \ln x$ replaces it by $\frac{1}{x}$, so choose that.}
\step{$v = \frac{x^{4}}{4}$, and $\int \frac{x^{4}}{4}\cdot\frac{1}{x}dx
= \int \frac{x^{3}}{4}dx = \frac{x^{4}}{16}$.}

$\ans{\dfrac{x^{4}}{4}\ln x - \dfrac{x^{4}}{16} + C}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\int x^{2}\ln x\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x^{3}}{3}\ln x - \dfrac{x^{3}}{9} + C}$}
\end{tryitbox}

\skillheading{3. When parts has to run twice}
\begin{formulabox}
\textbf{Two ways it repeats.} If $u$ is a power, each round lowers the degree by
one, so $x^{2}$ needs two rounds and $x^{3}$ needs three --- keep the same style
of choice each time.\\[3pt]
\textbf{If neither factor simplifies} (exponential times sine or cosine),
nothing ever terminates: parts twice returns the original integral. Name it
$I$, apply parts twice, then solve the resulting equation for $I$ algebraically.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\int e^{x}\cos x\,dx$.
\step{Neither $e^{x}$ nor $\cos x$ gets simpler, so aim for the cycle rather
than for termination: set $I$ equal to the integral.}
\step{Parts twice (with $u$ the trig factor both times) gives
$I = e^{x}\cos x + e^{x}\sin x - I$, so $2I = e^{x}(\sin x + \cos x)$.}

$\ans{\dfrac{e^{x}(\sin x + \cos x)}{2} + C}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\int e^{2x}\sin x\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{e^{2x}(2\sin x - \cos x)}{5} + C}$}
\end{tryitbox}

\skillheading{4. Parts in definite and improper integrals}
\begin{formulabox}
\textbf{Definite:} $\displaystyle\int_{a}^{b} u\,dv = \Big[uv\Big]_{a}^{b} - \int_{a}^{b} v\,du$
--- the boundary term is evaluated, the leftover integral keeps the same limits.
No $+ C$.\\[3pt]
\textbf{Improper:} replace the infinite limit by $t$, integrate, and take
$\lim_{t \to \infty}$ at the end. Never substitute infinity into an
antiderivative. Useful fact: $t^{n}e^{-t} \to 0$ and $\frac{\ln t}{t} \to 0$,
because polynomials beat logarithms and exponentials beat polynomials.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\int_{0}^{\infty} x e^{-x}\,dx$.
\step{The upper limit is infinite, so integrate to a finite $t$ first; parts
applies exactly as usual with $u = x$, $dv = e^{-x}dx$.}
\step{$\int_{0}^{t} x e^{-x}dx = \left[-x e^{-x} - e^{-x}\right]_{0}^{t}
= -t e^{-t} - e^{-t} + 1$, and both $t$ terms tend to $0$.}

$\ans{1}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\int_{0}^{\infty} x e^{-2x}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{4}}$}
\end{tryitbox}
