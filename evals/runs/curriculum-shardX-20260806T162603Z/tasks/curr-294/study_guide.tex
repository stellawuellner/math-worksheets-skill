\sstitleblock{Complete Root Sets and Correct Signs}

\skillheading{Skill 1 --- Both square roots, every time}

\begin{formulabox}
\textbf{Rule:} \quad if $x^2 = k$ and $k > 0$, then $x = +\sqrt{k}$ \emph{and} $x = -\sqrt{k}$.\par\smallskip
Isolate $x^2$ first, then take the root of both sides and write the $\pm$ immediately --- writing it later is when it gets forgotten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 = 25$.
\step{The equation has no $x$ term, so factoring is unnecessary --- isolating $x^2$ and taking roots is the direct route, and it is the route where the negative root goes missing.}
\step{$x = \pm\sqrt{25} = \pm 5$. Check both: $(-5)^2 = 25$ and $5^2 = 25$.}
\par
$x = \ans{-5}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $3x^2 = 27$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{25}$ on its own means $5$, the positive root. The second solution comes from the $\pm$ you write when you take the root of \emph{both sides of an equation} --- it is not part of the square-root symbol.
\end{watchoutbox}

\skillheading{Skill 2 --- The zero-product property}

\begin{formulabox}
\textbf{Rule:} \quad if $A \cdot B = 0$ then $A = 0$ or $B = 0$. Move everything to one side so the other side is $0$, factor, then set each factor equal to $0$.\par\smallskip
\textbf{Never divide both sides by $x$.} That is division by something that may be zero, and it deletes the root $x = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 = 4x$.
\step{Both sides contain $x$, so the tempting move is to divide --- instead move everything to one side, because the zero-product property keeps every root.}
\step{$x^2 - 4x = 0$, so $x(x-4) = 0$, giving $x = 0$ or $x - 4 = 0$.}
\par
$x = \ans{0}$ \ or \ $x = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^2 + 3x = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{0}$}
\end{tryitbox}

\skillheading{Skill 3 --- Signs in the quadratic formula}

\begin{formulabox}
\textbf{Formula:} \quad $x = \dfrac{-b \pm \sqrt{b^2-4ac}}{2a}$\par\smallskip
Write $a$, $b$, $c$ \emph{with their signs} before substituting, then write $-b$ on its own line. The denominator is $2a$, not $2$: with $a = 3$ it is $6$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 - 5x + 6 = 0$.
\step{Three terms with a leading coefficient of $1$ could be factored, but the formula never fails, so use it here and practise the signs.}
\step{$a = 1$, $b = -5$, $c = 6$, so $-b = +5$ and $x = \dfrac{5 \pm \sqrt{25-24}}{2} = \dfrac{5 \pm 1}{2}$.}
\par
$x = \ans{2}$ \ or \ $x = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2x^2 + 3x - 2 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-2}$ or $x = \ans{\tfrac{1}{2}}$}
\end{tryitbox}
