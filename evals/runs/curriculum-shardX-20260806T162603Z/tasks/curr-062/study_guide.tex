\sstitleblock{Subtracting Across a Zero}

\skillheading{Skill 1 --- Borrowing when the tens are zero}

\begin{formulabox}
\textbf{Rule:} A column that cannot give enough borrows from the column on its left.\par
\textbf{A zero has nothing to lend.} Go one more column left: open a hundred into ten tens first, and only then lend a ten to the ones.\par
Every time a column lends, its own digit goes down by one. Cross it out and write the new digit above.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $405 - 132$.
\step{The ones are fine here, but the tens column is empty and still has to give three tens away, so the borrowing has to start one column further left than usual.}
\step{Open one hundred into ten tens: three hundreds are left and the tens column now holds ten. Ones: five take two leaves three. Tens: ten take three leaves seven. Hundreds: three take one leaves two.}

$\ans{273}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $807 - 264$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{543}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtracting from hundreds with two zeros}

\begin{formulabox}
\textbf{Rule:} When both the tens and the ones are zero, regroup once across the whole number before subtracting anything.\par
A round hundreds number becomes: one hundred fewer, nine tens, and ten ones. Then every column has something to give.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $300 - 148$.
\step{Neither the ones nor the tens can give anything away, so rather than borrow twice in the middle of the work I rewrite the whole number in a form where every column is stocked.}
\step{Three hundreds becomes two hundreds, nine tens and ten ones. Ones: ten take eight leaves two. Tens: nine take four leaves five. Hundreds: two take one leaves one.}

$\ans{152}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $700 - 235$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{465}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimating a difference first}

\begin{formulabox}
\textbf{Rule:} Round \emph{both} numbers to the nearest hundred, then subtract the rounded ones.\par
Look at the tens digit: below five rounds down, five or more rounds up.\par
The estimate is a safety net. A regrouping that goes missing usually throws the answer out by a whole hundred, which the estimate catches at a glance.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $604 - 287$ to the nearest hundred, then subtract exactly.
\step{Borrowing across a zero is where answers go wrong by a hundred, so I want a rough answer in hand before I start, to compare against afterwards.}
\step{Rounding gives six hundred and three hundred, and their difference is the estimate. Exactly: open a hundred, leaving five hundreds and ten tens, then nine tens and fourteen ones. Ones: fourteen take seven leaves seven. Tens: nine take eight leaves one. Hundreds: five take two leaves three.}

estimate $\ans{300}$ \qquad exact $\ans{317}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $902 - 458$ to the nearest hundred, then subtract exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: estimate $\ans{400}$, exact $\ans{444}$}
\end{tryitbox}

\skillheading{Skill 4 --- Finding a lost regrouping}

\begin{formulabox}
\textbf{Rule:} The two classic slips are taking the smaller digit from the larger one in a column, and opening a hundred but forgetting to lower the hundreds digit.\par
To find either, redo the subtraction column by column and stop at the first digit that disagrees.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Someone worked out $600 - 173$ and wrote $573$. What went wrong?
\step{An answer this much too big is the signature of subtraction done backwards inside a column, so I redo the columns and look for the first one where the bigger digit was put on top.}
\step{The correct regrouping is five hundreds, nine tens and ten ones. Ones: ten take three leaves seven, not the three that was written --- the digits were flipped. Tens: nine take seven leaves two. Hundreds: five take one leaves four.}

correct difference $\ans{427}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Someone worked out $500 - 264$ and wrote $364$. Find the correct difference.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{236}$}
\end{tryitbox}
