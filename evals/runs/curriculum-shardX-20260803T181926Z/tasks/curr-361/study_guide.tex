\sstitleblock{Factoring Cubes and Higher-Degree Expressions}

\skillheading{Skill 1 --- Sum and difference of cubes}

\begin{formulabox}
\textbf{Patterns:} $a^3 - b^3 = (a-b)(a^2+ab+b^2)$\par
\hspace*{4.05em} $a^3 + b^3 = (a+b)(a^2-ab+b^2)$\par
\textbf{Signs (SOAP):} the binomial keeps the \textbf{S}ame sign; the trinomial's
middle term takes the \textbf{O}pposite sign; its last term is
\textbf{A}lways \textbf{P}ositive.\par
Cubes on sight: $1$, $8$, $27$, $64$, $125$, $216$.
The trinomial never factors again over the integers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Factor $x^3 + 27$
\step{Two terms, both perfect cubes, so this is a cube pattern --- not something to group or treat as a quadratic.}
\step{Here $a = x$ and $b = 3$. The sum pattern gives the binomial $x+3$, then $x^2 - 3x + 9$.}

$\Rightarrow\quad \ans{(x+3)(x^2-3x+9)}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^3 - 64$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-4)(x^2+4x+16)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Factor by grouping}

\begin{formulabox}
\textbf{When:} four terms. Split them into two pairs, pull the GCF out of each pair,
and the two leftover binomials must \emph{match} --- that match is what you factor
out. If they differ only in sign, take $-1$ out of the second pair.\par
Then keep going: the remaining quadratic is often a difference of squares.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Factor $x^3 - 3x^2 - 4x + 12$
\step{Four terms with no factor common to all of them, so I try grouping in pairs.}
\step{$x^2(x-3) - 4(x-3) = (x-3)(x^2-4)$, and $x^2-4$ still splits.}

$\Rightarrow\quad \ans{(x-3)(x-2)(x+2)}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^3 + 5x^2 - x - 5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+5)(x-1)(x+1)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Quadratic in disguise}

\begin{formulabox}
\textbf{When:} the exponents come in the pattern $2n$ and $n$, as in
$x^4 - 10x^2 + 9$ (here $n = 2$). Substitute $u = x^n$, factor the ordinary
quadratic in $u$, then put $x^n$ back.\par
\textbf{You are not finished there:} each bracket in $x^2$ is usually a difference
of squares and has to be split too.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Factor $x^4 - 10x^2 + 9$
\step{The exponents are four and two, so $u = x^2$ turns this into a quadratic I already know how to factor.}
\step{$u^2 - 10u + 9 = (u-1)(u-9)$, so back in $x$ it is $(x^2-1)(x^2-9)$ --- and both brackets split.}

$\Rightarrow\quad \ans{(x-1)(x+1)(x-3)(x+3)}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^4 - 29x^2 + 100$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-2)(x+2)(x-5)(x+5)}$}
\end{tryitbox}

\skillheading{Skill 4 --- From factors to zeros and shape}

\begin{formulabox}
\textbf{Rule:} once a polynomial is a product of factors, set each factor equal to
zero --- the zero-product property turns one hard equation into several easy ones.\par
A factor of $x$ contributes the zero $x = 0$. A repeated factor makes the graph
\emph{touch} the axis; a single factor makes it \emph{cross}, so the sign flips there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Real zeros of $x^3 - x^2 - 6x$
\step{I want the zeros, and a factored form hands them over, so I factor completely first.}
\step{$x(x^2-x-6) = x(x-3)(x+2)$, so each of the three factors gives one zero.}

$\Rightarrow\quad \ans{x = -2,\ 0,\ 3}$

\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Real zeros of $x^3 - 9x$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -3,\ 0,\ 3}$}
\end{tryitbox}
