\sstitleblock{Boundary Conditions and Open or Closed Endpoints}

\skillheading{Skill 1 --- Who owns the boundary value?}

\begin{formulabox}
\textbf{Rule:} \quad the branch whose condition carries $\le$ or $\ge$ owns the boundary. The other one uses $<$ or $>$ and stops just short of it.\par\smallskip
Exactly one branch may own it. If two conditions both claimed the same input, the definition would produce two outputs for one input, and that is not a function.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 4x - 1$ when $x < 2$, and $10 - x$ when $x \ge 2$, find $f(2)$.
\step{The input is exactly the boundary, so the formulas are irrelevant until the conditions are read --- and only one of them includes $2$.}
\step{$x \ge 2$ includes $2$, so the second rule applies: $10 - 2$.}
\par
$f(2) = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For the same $f$, find $f(0)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $f(0) = \ans{-1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} at a boundary the arithmetic is almost never the problem. Read the two conditions first and decide who owns the input; then there is only one formula to use.
\end{watchoutbox}

\skillheading{Skill 2 --- Test each answer against the condition}

\begin{formulabox}
When you solve an equation using one branch's formula, the answers you get satisfy \emph{the formula}. They are solutions of the \emph{function} only if they also satisfy that branch's condition.\par\smallskip
So always finish with the test: substitute each answer into the condition and discard the ones that fail. A discarded value is not an arithmetic mistake --- it solves the formula --- it is simply outside where that formula applies.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $g(x) = x^2 + 1$ when $x \le 0$, solve $g(x) = 17$.
\step{Solve the formula first and expect two answers, then test both against the condition --- that test is the whole point of the exercise.}
\step{$x^2 + 1 = 17$ gives $x^2 = 16$, so the formula is solved by $-4$ and by $4$; only $-4$ satisfies $x \le 0$.}
\par
Formula solutions $x = \ans{-4}$ and $x = \ans{4}$; only $-4$ solves $g(x) = 17$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For the same $g$, solve $g(x) = 10$ and say which value survives.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{3}$; only $-3$ survives}
\end{tryitbox}

\skillheading{Skill 3 --- Making the two rules meet}

\begin{formulabox}
A graph has \textbf{no jump} at the boundary when the two formulas give the same value there. To find an unknown letter that makes this happen, set the two formulas equal \emph{at the boundary input} and solve for the letter.\par\smallskip
Substitute the boundary value into \emph{both} formulas in full --- dropping a constant from one side is the usual slip.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $h(x) = 2x + k$ when $x < 1$, and $x^2 + 3$ when $x \ge 1$, find $k$ so there is no jump at $x = 1$.
\step{No jump means the two formulas agree at the boundary, so substitute $x = 1$ into both and set them equal --- any other input would answer a different question.}
\step{$2(1) + k = 1^2 + 3$, so $2 + k = 4$.}
\par
$k = \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $h(x) = 2x + k$ when $x < 5$, and $x^2 - 10$ when $x \ge 5$, find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $k = \ans{5}$}
\end{tryitbox}
