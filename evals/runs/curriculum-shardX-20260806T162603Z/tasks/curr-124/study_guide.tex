\sstitleblock{Comparing Fractions Without Being Fooled}

\skillheading{Skill 1 --- Building an equivalent fraction}

\begin{formulabox}
\textbf{Rule:} Multiply (or divide) the top \emph{and} the bottom by the same number.
\[ \frac{a}{b} = \frac{a \times k}{b \times k} \]
Multiplying by $\frac{k}{k}$ is multiplying by 1, so the amount does not change --- the pieces are just re-cut. \textbf{Adding} the same number to top and bottom is a different fraction.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fill in: $\dfrac{2}{3} = \dfrac{\ \ }{15}$
\step{The denominators tell me the scale factor, so I find what 3 was multiplied by and apply the same factor to the top.}
\step{$3 \times 5 = 15$, so the numerator must also be multiplied by 5: $2 \times 5$.}
Missing numerator: $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Fill in: $\dfrac{3}{7} = \dfrac{\ \ }{28}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{Skill 2 --- Comparing over a common denominator}

\begin{formulabox}
\textbf{Rule:} Numerators can only be compared when the pieces are the same size.\\[3pt]
Rewrite both fractions over a common denominator, then compare the counts.\\[3pt]
\textbf{Same numerator?} The one with the \emph{smaller} denominator is larger --- fewer cuts means bigger pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$ or $>$ between $\dfrac{3}{4}$ and $\dfrac{5}{7}$.
\step{The pieces are different sizes, so comparing 3 against 5 would mean nothing --- I rewrite both over a shared denominator first.}
\step{$4 \times 7 = 28$: $\dfrac{3}{4} = \dfrac{21}{28}$ and $\dfrac{5}{7} = \dfrac{20}{28}$, and 21 twenty-eighths beats 20.}
$\dfrac{3}{4} \ans{>} \dfrac{5}{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$ or $>$ between $\dfrac{2}{9}$ and $\dfrac{3}{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\dfrac{2}{9} \ans{<} \dfrac{3}{8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Using one half as a benchmark}

\begin{formulabox}
\textbf{Rule:} A fraction is more than one half when its numerator is more than half of its denominator.\\[3pt]
If one fraction is below one half and the other is above it, the comparison is finished --- no common denominator needed. If both land on the same side of the benchmark, fall back to Skill 2.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$ or $>$ between $\dfrac{4}{9}$ and $\dfrac{5}{8}$.
\step{Before doing any arithmetic I check each fraction against one half, because a split verdict decides the comparison on its own.}
\step{Half of 9 is $4.5$ and 4 is below it, so the first is under a half; half of 8 is 4 and 5 is above it, so the second is over a half.}
$\dfrac{4}{9} \ans{<} \dfrac{5}{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$ or $>$ between $\dfrac{7}{15}$ and $\dfrac{9}{16}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\dfrac{7}{15} \ans{<} \dfrac{9}{16}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Two habits that look like reasoning and are not. Comparing only the tops (``7 beats 5, so seven twelfths beats five sixths'') ignores piece size. Comparing only the bottoms (``8 beats 5, so three eighths beats three fifths'') gets the order exactly backwards.
\end{watchoutbox}
