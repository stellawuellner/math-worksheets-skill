\wstitleblock{Using u-Substitution in Indefinite and Definite Integrals}{AP Calculus AB / BC \quad Integration Techniques}{}

\begin{center}
\textit{Recognising an inner function and its derivative, adjusting by a constant
factor, converting the limits of a definite integral to the new variable, and
evaluating an improper integral as a limit}
\end{center}

\noindent\textbf{The one idea behind every problem.} $u$-substitution is the
chain rule read backwards. If an integrand contains an inner function \emph{and}
a constant multiple of its derivative, naming that inner function $u$ turns the
whole integral into one you already know.

\begin{center}
\fbox{\parbox{0.92\linewidth}{\centering
\textbf{The substitution box --- fill one in before integrating}
\par\smallskip
\begin{tabular}{|l|l|}
\hline
\strut $u = $ \qquad\qquad & $du = $ \qquad\qquad $dx$ \\
\hline
\strut new lower limit & new upper limit \\
\hline
\end{tabular}
\par\smallskip
Indefinite integrals leave the limit row blank and back-substitute at the end.
Definite integrals fill it in and never return to $x$.}}

\smallskip
\small\textit{Reference layout. No values are filled in --- use the integrand
given in each problem.}\normalsize
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Write the substitution box first, every time. Most errors here are chosen,
not computed.
\item Include $+\,C$ on every indefinite integral.
\item Give exact answers. Write your final answer on the answer line.
\end{itemize}

\problem[7cm]{Evaluate $\displaystyle\int 2x\left(x^2+5\right)^3\,dx$.

\smallskip
\textbf{(a)} Fill in the substitution box: what is $u$, and what is $du$?

\smallskip
\textbf{(b)} Notice that $2x\,dx$ is exactly $du$ --- no adjusting factor is
needed. Rewrite the integral entirely in $u$.

\smallskip
\textbf{(c)} Integrate and back-substitute.}

\problem[7.5cm]{Evaluate $\displaystyle\int x\cos\!\left(x^2\right)dx$.

\smallskip
\textbf{(a)} Take $u = x^2$. Write $du$, then compare it with the $x\,dx$ that
actually appears in the integrand.

\smallskip
\textbf{(b)} What constant must multiply the integral to make up the difference?
Explain in one sentence why moving a constant across the integral sign is legal
but moving an $x$ is not.

\smallskip
\textbf{(c)} Finish the integration and back-substitute.}

\problem[7cm]{Evaluate $\displaystyle\int \frac{\left(\ln x\right)^2}{x}\,dx$.

\smallskip
\textbf{(a)} The integrand contains a function and its own derivative. Which
factor is which?

\smallskip
\textbf{(b)} Fill in the substitution box and rewrite the integral in $u$.

\smallskip
\textbf{(c)} Integrate and back-substitute.}

\problem[8cm]{Evaluate $\displaystyle\int_{0}^{2} x\,e^{x^2}\,dx$ exactly.

\smallskip
\textbf{(a)} With $u = x^2$, fill in \emph{all four} cells of the substitution
box, including the two new limits.

\smallskip
\textbf{(b)} Rewrite the definite integral in $u$ with its new limits. You should
not need to return to $x$ at any point.

\smallskip
\textbf{(c)} Evaluate. Leave your answer in terms of $e$.}

\problem[7.5cm]{Evaluate $\displaystyle\int \sin^3 x\,\cos x\,dx$.

\smallskip
\textbf{(a)} Two substitutions look plausible here: $u = \sin x$ and
$u = \cos x$. Try writing $du$ for each.

\smallskip
\textbf{(b)} Only one of them lets you replace \emph{every} $x$ in the
integrand. Say which, and why the other one strands a leftover factor.

\smallskip
\textbf{(c)} Carry out the substitution that works, integrate, and
back-substitute.}

\problem[8cm]{Evaluate $\displaystyle\int_{1}^{3} \frac{2x}{x^2+1}\,dx$ exactly.

\smallskip
\textbf{(a)} Fill in all four cells of the substitution box.

\smallskip
\textbf{(b)} Evaluate the integral in $u$. Your first answer will be a
difference of two logarithms.

\smallskip
\textbf{(c)} Simplify it to a single logarithm.}

\problem[8cm]{Evaluate $\displaystyle\int x^2\sqrt{x^3+1}\;dx$.

\smallskip
\textbf{(a)} Fill in the substitution box, then state the constant factor the
integral must carry.

\smallskip
\textbf{(b)} Rewrite $\sqrt{u}$ as a power before integrating.

\smallskip
\textbf{(c)} Integrate, simplify the two fractions into one coefficient, and
back-substitute.}

\problem[9cm]{\textbf{Challenge.} Consider the improper integral
$\displaystyle\int_{0}^{\infty} x\,e^{-x^2}\,dx$.

\smallskip
\textbf{(a)} An infinite limit is not a number, so rewrite the integral as a
limit of a proper definite integral on $[0, b]$.

\smallskip
\textbf{(b)} Use $u = -x^2$ or $u = x^2$ to find the antiderivative, evaluate it
on $[0, b]$, and take the limit as $b \to \infty$. What is the value of that
limit?

\smallskip
\textbf{(c)} State whether the integral converges or diverges, and give its
value. Write your part (b) limit and your part (c) value on the answer line.}
