\wstitleblock{Writing Piecewise Rules for Rates and Fees}{Algebra 1}{Writing the rule from a price list, using it to find a charge, and working back from a charge}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Real price lists change rate part-way through, which is exactly what a piecewise rule describes: one formula per band, each with the condition that says when it applies.
  \item Write every rule in the form
        $C(x) = \begin{cases} \text{formula},& \text{condition} \\ \text{formula},& \text{condition}\end{cases}$
        --- the conditions are part of the answer, not decoration.
  \item \textbf{The second band starts from what has already been charged.} If $\$13$ has been paid by $5$ miles, the rest of the ride costs $13 + 1.5(m-5)$, not $1.5m$.
  \item All money answers are in dollars; state the units of the input in your rule.
\end{itemize}

\problem[6cm]{A taxi charges a $\$3$ pick-up fee plus $\$2$ per mile for the first $5$ miles. After $5$ miles, the charge is $\$13$ so far plus $\$1.50$ for each additional mile. Let $m$ be the distance in miles.
\par\medskip
(a) Write $C(m)$, the fare in dollars, as a piecewise rule with both conditions.
\par\smallskip
(b) Find the fare for a $4$-mile ride, and put it on the answer line.}

\problem[5cm]{Use the taxi rule from Problem 1:
\[ C(m) = \begin{cases} 3 + 2m, & 0 \le m \le 5 \\ 13 + 1.5(m-5), & m > 5 \end{cases} \]
$C(2) = $~\ansblank \quad $C(9) = $~\ansblank}

\problem[6cm]{A swimming pool charges $\$5$ to enter, which covers a stay of up to $2$ hours. Each hour after the first $2$ costs an extra $\$3$. Let $t$ be the length of the stay in hours.
\par\medskip
(a) Write $C(t)$, the cost in dollars, as a piecewise rule with both conditions.
\par\smallskip
(b) Find the cost of a $5$-hour visit, and put it on the answer line.}

\problem[6cm]{A phone plan costs $\$25$ a month for up to $5$ gigabytes of data. Every gigabyte beyond the fifth costs an extra $\$10$. Let $g$ be the data used, in gigabytes.
\par\medskip
(a) Write $F(g)$, the monthly charge in dollars, as a piecewise rule with both conditions.
\par\smallskip
(b) Find the charge for a month in which $8$ gigabytes are used.}

\problem[6.5cm]{A taxi ride using the rule from Problem 1 cost $\$25$.
\par\medskip
(a) How many miles was the ride? Put the distance on the answer line.
\par\smallskip
(b) Explain how you knew, before solving anything, that the second branch of the rule was the one to use.}

\problem[6cm]{A library charges no fine for the first $3$ days a book is late. From the fourth day onwards the fine is $\$0.25$ for each day beyond the third. Let $d$ be the number of days late.
\par\medskip
(a) Write $F(d)$, the fine in dollars, as a piecewise rule with both conditions.
\par\smallskip
(b) Find the fine on a book returned $10$ days late.}

\problem[5.5cm]{An electricity supplier charges $\$0.12$ per kilowatt-hour for the first $300$ kilowatt-hours in a month. Beyond that, the bill is $\$36$ plus $\$0.18$ for each extra kilowatt-hour:
\[ E(k) = \begin{cases} 0.12k, & 0 \le k \le 300 \\ 36 + 0.18(k-300), & k > 300 \end{cases} \]
$E(250) = $~\ansblank \quad $E(500) = $~\ansblank}

\problem[6.5cm]{A worker is paid $\$18$ per hour for up to $40$ hours in a week. Every hour beyond $40$ is paid at $\$27$ per hour. Let $h$ be the hours worked in a week.
\par\medskip
(a) Write $P(h)$, the weekly pay in dollars, as a piecewise rule with both conditions.
\par\smallskip
(b) Find the pay for a $46$-hour week.}

\problem[6.5cm]{Using the pay rule from Problem 8, a worker earned $\$963$ in one week.
\par\medskip
(a) How many hours did the worker work? Put the number on the answer line.
\par\smallskip
(b) Explain how the amount $\$963$ told you which branch of the rule to use before you solved anything.}

\problem[8cm]{\textbf{Synthesis challenge.} Two phone plans charge for $n$ minutes of calls.
\par\smallskip
Plan A: \ $A(n) = 30 + 0.1n$ dollars, for every number of minutes.
\par\smallskip
Plan B: \ $\$20$ a month covering the first $200$ minutes, then $\$0.15$ for each minute beyond $200$.
\par\medskip
(a) Write $B(n)$ as a piecewise rule with both conditions.
\par\smallskip
(b) For calls longer than $200$ minutes, find the number of minutes at which the two plans cost the same.
\par\smallskip
(c) Explain which plan is cheaper below that number of minutes, and give evidence for your answer.}
