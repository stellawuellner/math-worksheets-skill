\wstitleblock{Ellipses and Hyperbolas in Real Structures}{Algebra 2}{}

\begin{center}
\small\textit{Reading vertices, axes, foci and asymptotes off ellipse and hyperbola models of real structures}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Every equation below models a real object, with the centre of the object at the origin and the units stated in the question.
  \item Ellipse $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$: vertices at $\pm a$ along the longer axis, foci at $\pm c$ with $c^2 = a^2 - b^2$ (subtract).
  \item Hyperbola $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$: vertices at $\pm a$, foci at $\pm c$ with $c^2 = a^2 + b^2$ (add), asymptote slopes $\pm b/a$.
  \item Give every length with its unit, and keep values exact.
\end{itemize}

\problem{The floor of a whispering gallery is an ellipse. With the centre of the room at the origin and both coordinates in metres, its edge satisfies $\dfrac{x^2}{169} + \dfrac{y^2}{144} = 1$.
\par\smallskip
(a)~Find the length of the major axis, that is, the greatest width of the room.
\par\vspace*{2.8cm}\answerline{m}
\par\smallskip
(b)~Find the length of the minor axis.
\par\vspace*{2.6cm}\answerline{m}}

\problem{The gallery of problem $1$ has two ``listening spots'', one at each focus of the ellipse. Using the same equation $\dfrac{x^2}{169} + \dfrac{y^2}{144} = 1$, with coordinates in metres, find the distance between the two listening spots.
\par\vspace*{5.4cm}\answerline{m}}

\problem{A semi-elliptical bridge arch spans $40$ feet at its base and rises $12$ feet at the centre. Taking the centre of the span as the origin, the arch satisfies $\dfrac{x^2}{400} + \dfrac{y^2}{144} = 1$ with $y \ge 0$, both coordinates in feet. Find the height of the arch at a point $10$ feet horizontally from the centre. Leave the answer exact.
\par\vspace*{5.6cm}\answerline{ft}}

\problem{The waist of a cooling tower is modelled by $\dfrac{x^2}{64} - \dfrac{y^2}{36} = 1$, with the narrowest ring of the tower at the origin and both coordinates in metres.
\par\smallskip
(a)~Find the distance across the tower at its narrowest point, that is, between the two vertices.
\par\vspace*{2.8cm}\answerline{m}
\par\smallskip
(b)~Find the distance from the centre to each focus.
\par\vspace*{2.6cm}\answerline{m}}

\problem{The same cooling tower, $\dfrac{x^2}{64} - \dfrac{y^2}{36} = 1$ with coordinates in metres, has two asymptotes.
\par\smallskip
(a)~Find the slope of the asymptote that rises from left to right.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~Explain what those asymptotes describe about the shape of the tower wall high above the waist.
\par\vspace*{2.6cm}\ansline}

\problem{A planetarium has an elliptical dome whose horizontal cross-section at floor level satisfies $\dfrac{x^2}{100} + \dfrac{y^2}{36} = 1$, with coordinates in metres.
\par\smallskip
(a)~Find the distance from the centre to each focus.
\par\vspace*{2.8cm}\answerline{m}
\par\smallskip
(b)~A cable is run from one focus to a point on the ellipse and back to the other focus. Find the total cable length, which is the same wherever that point is chosen.
\par\vspace*{2.8cm}\answerline{m}}

\problem{The cross-section of a hyperbolic mirror is modelled by $\dfrac{y^2}{9} - \dfrac{x^2}{16} = 1$, with coordinates in centimetres and the mirror's axis vertical.
\par\smallskip
(a)~Find the distance from the centre to each vertex.
\par\vspace*{2.8cm}\answerline{cm}
\par\smallskip
(b)~Find the distance from the centre to each focus.
\par\vspace*{2.6cm}\answerline{cm}}

\problem{A lamp throws a hyperbolic shadow line on a wall, modelled by $\dfrac{y^2}{25} - \dfrac{x^2}{144} = 1$ with coordinates in centimetres.
\par\smallskip
(a)~Find the slope of the asymptote that rises from left to right.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~Find the distance from the centre to each vertex.
\par\vspace*{2.6cm}\answerline{cm}}

\problem{An elliptical garden bed measures $30$ feet along its longer axis and $18$ feet along its shorter axis. Two sprinkler heads are to be placed at the foci of the ellipse. Find the distance between the two sprinkler heads.
\par\vspace*{5.6cm}\answerline{ft}}

\problem{A new whispering gallery is being designed. Its ceiling cross-section must be a semi-ellipse $40$ metres wide at floor level, and the two listening spots, at the foci, must sit $32$ metres apart.
\par\smallskip
(a)~Find the required height of the ceiling above the centre of the floor.
\par\vspace*{4cm}\answerline{m}
\par\smallskip
