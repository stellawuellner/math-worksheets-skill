\textit{missing-factor equations, division facts, and equal-group applications}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Every division question hides a multiplication fact: $18 \div 3$ asks ``$3 \times \textbf{?} = 18$.''
  \item Use a related multiplication fact you know to find each unknown factor.
  \item In the story problems, the givens are stated first and the question names the unit your answer needs.
\end{itemize}

\problem[2.5cm]{Find the missing factor: \quad $5 \times n = 20$, \quad $n = {}$~\ansblank}

\problem[3cm]{$18 \div 3 = {}$~\ansblank \quad Think: $3 \times \textbf{?} = 18$.}

\problem{Nadia has 24 stickers. She puts them equally onto 4 pages of her album. How many stickers go on each page? Think: $4 \times \textbf{?} = 24$.\par\vspace*{3cm}\answerline{stickers}}

\problem[3cm]{Find the missing factor: \quad $n \times 7 = 42$, \quad $n = {}$~\ansblank}

\problem[3cm]{$35 \div 5 = {}$~\ansblank \quad Think: $5 \times \textbf{?} = 35$.}

\problem{An art teacher has 30 markers. She packs 6 markers into each box. How many boxes does she fill? Think: $6 \times \textbf{?} = 30$.\par\vspace*{3cm}\answerline{boxes}}

\problem[3cm]{Find the missing factor: \quad $9 \times n = 54$, \quad $n = {}$~\ansblank}

\problem[3cm]{$48 \div 6 = {}$~\ansblank \quad Think: $6 \times \textbf{?} = 48$.}

\problem{Ms.\ Ortiz sets up 40 chairs for a concert. She puts 8 chairs in each row.
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] Write a multiplication fact with a missing factor that matches this story. \ansline
  \item[(b)] How many rows does she make?\par\vspace*{2.6cm}\answerline{rows}
\end{itemize}}

\problem{\textbf{Challenge.} There are 63 apples packed equally into 7 bags. Explain how the fact $7 \times 9 = 63$ helps you find how many apples are in each bag.\par\vspace{6pt}\noindent\underline{\hspace{12.5cm}}\par\vspace{14pt}\noindent\underline{\hspace{12.5cm}}\par\vspace*{1.8cm}\answerline{apples}}
