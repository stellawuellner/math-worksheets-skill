\wstitleblock{Adding, Subtracting, and Multiplying Polynomials}{Algebra 2}{}

\noindent\emph{Adding and subtracting by like terms, multiplying with an area box, and checking a product by substitution or by factoring back}

\vspace{0.15cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item Sums and differences come with a \textbf{like-terms table}: one column
        per power. Fill the rows, then combine down each column --- only terms
        in the same column may be combined.
  \item Products come with an \textbf{area box}: one row per term of the first
        factor, one column per term of the second. Every cell is one
        multiplication, and the answer is the sum of the cells.
  \item Write the final polynomial in descending order of degree on the answer
        line.
\end{itemize}
