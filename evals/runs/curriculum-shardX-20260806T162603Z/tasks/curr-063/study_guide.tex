\sstitleblock{Estimating Differences and Checking Answers}

\skillheading{Skill 1 --- Estimating to the nearest ten}

\begin{formulabox}
\textbf{Rule:} Round \emph{both} numbers to the nearest ten, then subtract the rounded numbers.\par
Look at the ones digit: below five rounds down, five or more rounds up.\par
\textbf{Never} round only one of the two --- an estimate built from one rounded number and one exact one is neither quick nor honest.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $83 - 46$ to the nearest ten.
\step{I want a quick answer I can carry in my head, and tens are the easiest numbers to subtract, so I move both numbers to the nearest ten before doing anything else.}
\step{The first number has three ones, so it rounds down to eighty. The second has six ones, so it rounds up to fifty. Eighty take away fifty is the estimate.}

$\ans{30}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $71 - 34$ to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40}$}
\end{tryitbox}

\skillheading{Skill 2 --- Estimating to the nearest hundred}

\begin{formulabox}
\textbf{Rule:} Same method, one place further left: round both numbers to the nearest hundred, then subtract.\par
Look at the \emph{tens} digit this time: below five rounds down, five or more rounds up.\par
Hundreds give a rougher estimate than tens, but the subtraction becomes one you can do in your head.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $573 - 241$ to the nearest hundred.
\step{These numbers are in the hundreds, so rounding to tens would still leave me subtracting awkward numbers; rounding to hundreds gives a subtraction I can say out loud.}
\step{The first number has seven tens, so it rounds up to six hundred. The second has four tens, so it rounds down to two hundred. Six hundred take away two hundred is the estimate.}

$\ans{400}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $418 - 262$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{100}$}
\end{tryitbox}

\skillheading{Skill 3 --- Checking an answer with an estimate}

\begin{formulabox}
\textbf{Rule:} Estimate first, subtract exactly second, then compare the two.\par
Rounding can move an answer by a few tens. It cannot move it by a whole hundred --- so a gap that big means the exact working has a mistake in it, not that the estimate was poor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $645 - 328$ to the nearest hundred, then subtract exactly and compare.
\step{The point of the estimate is to have something to test the exact answer against, so I work it out first and keep it beside me while I subtract.}
\step{Rounding gives six hundred take away three hundred. Exactly: regroup for the ones, leaving six hundreds, three tens and fifteen ones; fifteen take eight leaves seven, three take two leaves one, six take three leaves three. The two answers land within a few tens of each other, so nothing was dropped.}

estimate $\ans{300}$ \qquad exact $\ans{317}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $532 - 219$ to the nearest hundred, then subtract exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: estimate $\ans{300}$, exact $\ans{313}$}
\end{tryitbox}

\skillheading{Skill 4 --- Spotting an answer the estimate rules out}

\begin{formulabox}
\textbf{Rule:} If an exact answer sits more than about a hundred away from a nearest-hundred estimate, do not accept it --- go back and look for a lost regrouping.\par
Being close to the estimate does not prove an answer right, but being far from it does prove one wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Someone worked out $700 - 264$ and wrote $564$. Is that reasonable?
\step{Rather than redo the whole subtraction straight away, I estimate first: if the written answer is nowhere near the estimate, I already know it is wrong and where to look.}
\step{Rounding gives seven hundred take away three hundred, so about four hundred should be left. The written answer sits more than a hundred above that, which rounding cannot explain. Subtracting properly --- six hundreds, nine tens, ten ones --- gives the true difference.}

estimate $\ans{400}$ \qquad correct $\ans{436}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Someone worked out $805 - 397$ and wrote $508$. Estimate to the nearest hundred, then find the correct difference.\\[2pt]
\rotatebox{180}{\footnotesize check: estimate $\ans{400}$, correct $\ans{408}$}
\end{tryitbox}
