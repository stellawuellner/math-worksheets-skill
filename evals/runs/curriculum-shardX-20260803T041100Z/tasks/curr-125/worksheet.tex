\wstitleblock{Ordering Fractions and Mixed Numbers}{Grades 4--5}{}

\noindent\textbf{Rewrite into equivalent forms, compare a pair with benchmarks
or a common denominator, then order a whole set.}

\smallskip
\noindent\small The numbers below arrive in different forms --- proper
fractions, fractions greater than one, and mixed numbers. Before you can order
them you usually have to put them into \emph{one} form. Choose the quickest
route each time:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Benchmark} --- decide first whether each number is below a
        half, between a half and one, or above one. That alone settles many
        comparisons.
  \item \textbf{Same numerator or same denominator} --- with matching tops,
        the smaller bottom is the bigger number; with matching bottoms, compare
        the tops.
  \item \textbf{Common denominator} --- when the benchmarks tie, rewrite both
        numbers over one denominator and compare the numerators.
\end{itemize}
Show the rewriting you used, then write the answer on the answer line. Use
$<$, $>$, or $=$ where a comparison is asked for.\normalsize

\problem[4cm]{Rewrite $\tfrac{3}{4}$ as an equivalent fraction with denominator
$12$.}

\problem[4cm]{Write the mixed number $2\tfrac{1}{3}$ as a fraction greater than
one.}

\problem[4cm]{Which is greater, $\tfrac{3}{8}$ or $\tfrac{1}{2}$? Write the two
fractions with the correct symbol between them.}

\problem[5cm]{Put these three numbers in order from \emph{least to greatest}:
\[ \tfrac{1}{2}, \qquad \tfrac{1}{4}, \qquad \tfrac{3}{8} \]}

\problem[4.5cm]{Which is greater, $\tfrac{7}{8}$ or $\tfrac{5}{6}$? Write the
two fractions with the correct symbol between them.\par
\emph{Hint:} think about how far each one sits below one.}

\problem[4cm]{Write $\tfrac{11}{4}$ as a mixed number.}

\problem[5.5cm]{Put these three numbers in order from \emph{greatest to
least}:
\[ 2\tfrac{1}{2}, \qquad \tfrac{9}{4}, \qquad 2\tfrac{3}{8} \]}

\problem[5cm]{Which is greater, $\tfrac{5}{9}$ or $\tfrac{4}{7}$? Write the two
fractions with the correct symbol between them.\par
\emph{Careful:} the benchmark test does not settle this one.}

\problem[4cm]{Write $\tfrac{5}{6}$ as an equivalent fraction with denominator
$30$.}

\problem[6cm]{Put these four numbers in order from \emph{least to greatest}:
\[ \tfrac{3}{5}, \qquad \tfrac{1}{2}, \qquad \tfrac{7}{10}, \qquad
   \tfrac{13}{20} \]}

\problem[5cm]{Which is greater, $1\tfrac{3}{4}$ or $\tfrac{5}{3}$? Write the
two numbers with the correct symbol between them.}

\problem[7cm]{\textbf{Challenge.} These four numbers are written in three
different forms. Put them in order from \emph{least to greatest}, and show the
one common form you converted them all into.
\[ \tfrac{5}{3}, \qquad 1\tfrac{1}{2}, \qquad \tfrac{11}{8}, \qquad
   1\tfrac{5}{6} \]}
