\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Read each story. Every story starts with some things, and then more come.
  \item Show the story with a drawing, counters, or an equation.
  \item Write your answer on the answer line, with its label.
\end{itemize}

\problem{Leo has 3 crayons. Ana gives him 2 more crayons. Draw a circle for
each crayon. How many crayons does Leo have now?
\par\vspace*{4cm}\answerline{crayons}}
