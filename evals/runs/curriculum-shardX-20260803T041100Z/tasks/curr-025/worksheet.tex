\wstitleblock{Choosing a Fast Way to Add}{Kindergarten--Grade 1}{}

\noindent\textbf{Pick the fastest way: count on from the larger part, use a
double or near double, or make ten.}

\smallskip
\noindent\small Every problem below can be solved by counting every dot --- but
one of these three ways is quicker. For each problem, \textbf{circle} the way
you picked, then write the sum on the answer line.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Count on} --- hold the \emph{larger} part in your head and say
        the next numbers, one for each of the smaller part.
  \item \textbf{Double / near double} --- when the two parts are the same, use
        your double. When they are neighbours, use the double and one more.
  \item \textbf{Make ten} --- when the two parts fill a whole ten-frame
        together, the sum is ten.
\end{itemize}
\normalsize
