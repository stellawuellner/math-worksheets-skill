\wstitleblock{Taking Away Objects and Writing Subtraction Equations}{Kindergarten--Grade 1}{}

\noindent\textbf{How to work each problem:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Put your finger on the picture and count the objects you start with.
  \item Draw an \textbf{X} on each object that is taken away.
  \item Count only the objects with \textbf{no} X. That is how many are left.
  \item Then write the equation: \ start $-$ taken away $=$ left.
\end{itemize}
\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3cm]{You have \textbf{5} counters. Cross out \textbf{2} of them.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.7cm]\hline
\end{tabular}
\end{center}
How many counters are left? \quad $5 - 2 = $~\ansblank}

\problem[3cm]{Here are \textbf{7} stars. Cross out \textbf{3} of them.
\begin{center}\Large

How many stars are left? \quad $7 - 3 = $~\ansblank}

\problem[3.5cm]{A basket holds \textbf{8} apples. Ravi takes \textbf{5} apples out of the
basket. Cross out the apples Ravi takes.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.7cm]\hline
\end{tabular}
\end{center}
Write the whole subtraction equation for the picture, then answer:
how many apples are \textbf{still in the basket}?}

How many cubes are left? \quad $10 - 4 = $~\ansblank}

\problem[4cm]{\textbf{Story problem.} There are \textbf{6} birds sitting on a fence.
Then \textbf{2} birds fly away.

\noindent Draw 6 birds as dots. Cross out the ones that fly away.
Then write the subtraction equation and the answer.}

\problem[4cm]{Mia starts with \textbf{9} blocks. She gives some blocks away.
Now \textbf{4} blocks are left.

\noindent How many blocks did Mia give away? \quad
$9 - $~\ansblank$\ = 4$}

\problem[4cm]{A plate has \textbf{10} grapes. Some grapes are eaten.
Now \textbf{3} grapes are left.

\noindent How many grapes were eaten? \quad
$10 - $~\ansblank$\ = 3$}

\problem[4.5cm]{\textbf{Challenge.} Ten counters were on the desk. Someone took some
away. Now \textbf{6} counters are left.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.6cm]\hline
$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\[0.6cm]\hline
\end{tabular}
\end{center}
Cross out counters until only 6 are left. How many did you cross out?
Write the equation that tells the whole story.}
