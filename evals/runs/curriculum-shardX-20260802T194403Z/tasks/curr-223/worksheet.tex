\wstitleblock{Converting Between Standard Form and Slope-Intercept Form}{Grade 8 / Pre-Algebra --- Functions}{}

\noindent\textbf{Two forms of the same line.}\quad
Standard form is $Ax + By = C$; slope-intercept form is $y = mx + b$, where $m$ is the
slope and $b$ is the $y$-intercept. Converting means solving for $y$ (or clearing fractions
and collecting $x$ and $y$ on one side) --- the line never changes, only the way it is
written. Show every step, and write your final answer on the answer line.\par\vspace{0.15cm}
\noindent{\small\textit{This workshop covers: Standard form to slope-intercept form,
slope-intercept form to standard form, slope from a table or two points, and converting
inside a context.}}

\problem[5cm]{Write $2x + y = 7$ in slope-intercept form. Then state the slope and the
$y$-intercept.}

\problem[5cm]{Write $y = 3x - 5$ in standard form $Ax + By = C$, using integer coefficients
with $A > 0$.}

\problem[5.5cm]{Write $3x + 4y = 12$ in slope-intercept form. Give the slope as a fraction
in lowest terms.}

\problem{A pool is being drained at a steady rate. The table gives the water level
(in centimetres) after $x$ hours.
\begin{center}
\begin{tabular}{|c|c|c|c|c|}\hline
\textbf{$x$ (hours)} & $0$ & $2$ & $4$ & $6$\\\hline
\textbf{$y$ (cm)} & $30$ & $24$ & $18$ & $12$\\\hline
\end{tabular}
\end{center}
\begin{enumerate}[label=(\alph*), itemsep=2.7cm, leftmargin=1.2cm]
  \item Read the water level, in centimetres, after $4$ hours from the table. \ansline
  \item Use the table to find the rate of change and the starting level, then write the
        relationship in slope-intercept form. \ansline
  \item Rewrite that equation in standard form with integer coefficients and $A > 0$. \ansline
\end{enumerate}\vspace*{0.2cm}}

\problem[5.5cm]{Write $y = -\dfrac{2}{5}x + 3$ in standard form with integer coefficients
and $A > 0$. (Clear the fraction first.)}

\problem[5.5cm]{Write $-6x + 2y = 10$ in slope-intercept form. Watch the sign of the $x$
term when you divide.}

\problem{A line passes through the points $(2, 9)$ and $(6, 17)$.
\begin{enumerate}[label=(\alph*), itemsep=2.7cm, leftmargin=1.2cm]
  \item Find the slope of the line. \ansline
  \item Write the equation of the line in slope-intercept form. \ansline
  \item Rewrite that equation in standard form $Ax + By = C$ with integer coefficients. \ansline
\end{enumerate}\vspace*{0.2cm}}

\problem[5.5cm]{Write $y = \dfrac{1}{2}x - 4$ in standard form with integer coefficients and
$A > 0$.}

\problem[6cm]{A drama club sells child tickets for \$4 and adult tickets for \$6. The club
takes in \$120 in total, so the ticket numbers satisfy
\[ 4x + 6y = 120, \]
where $x$ is the number of child tickets and $y$ is the number of adult tickets sold.\par
\vspace{0.1cm}
Write this equation in slope-intercept form, and state the $y$-intercept.}

\problem{Sam earns \$15 for every lawn he mows and \$25 for every driveway he clears. He
wants to earn \$300 this month, so his jobs satisfy
\[ 15x + 25y = 300, \]
where $x$ is the number of lawns and $y$ is the number of driveways.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.2cm]
  \item Write the equation in slope-intercept form. \ansline
  \item The slope is a trade-off rate. If Sam mows $5$ more lawns, how many \emph{fewer}
        driveways does he need to clear to still reach \$300? \ansline
\end{enumerate}\vspace*{0.2cm}}
