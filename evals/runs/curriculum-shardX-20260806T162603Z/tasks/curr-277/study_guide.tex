\sstitleblock{Evaluating and Graphing Exponential Functions}

\skillheading{Skill 1 --- Evaluating a power rule}

\begin{formulabox}
\textbf{Order of work:} in $a \cdot b^{\,x}$ the exponent belongs to $b$ only. Compute $b^{\,x}$ first, then multiply by $a$.\par
\textbf{Two facts that carry most of the marks:} $b^{0} = 1$ for every nonzero base, so the graph always starts at $a$; and $b^{-n} = \frac{1}{b^{\,n}}$, so a negative exponent gives a small \emph{positive} value.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 4 \cdot 3^{\,x}$, find $f(2)$, $f(0)$ and $f(-1)$.
\step{The exponent sits on the $3$, so each answer is a power of three multiplied by four --- never four raised to anything.}
\step{$3^2 = 9$, so $4 \cdot 9$; and $3^0 = 1$, so $4 \cdot 1$.}
\step{$3^{-1} = \frac{1}{3}$, so $4 \cdot \frac{1}{3}$.}

$f(2) = \ans{36}$, \quad $f(0) = \ans{4}$, \quad $f(-1) = \ans{\tfrac{4}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = 6 \cdot 2^{\,x}$, find $g(3)$ and $g(-1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $g(3) = \ans{48}$, $g(-1) = \ans{3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Table to graph}

\begin{formulabox}
\textbf{Method:} choose inputs on both sides of $0$, evaluate, plot, then join with a \emph{smooth curve} --- exponential graphs are never made of straight segments.\par
\textbf{Shape facts:} the curve crosses the vertical axis at $a$; if $b > 1$ it climbs to the right, and if $0 < b < 1$ it falls. It approaches the horizontal axis but never reaches it, because a power of a positive base is never zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Build a table for $y = 3^{\,x}$ at $x = -1, 0, 1, 2$.
\step{Inputs either side of zero show both the flattening tail and the steep climb, which is what makes the curve recognisable.}
\step{$3^{-1} = \frac{1}{3}$ and $3^0 = 1$.}
\step{$3^1 = 3$ and $3^2 = 9$.}

$\ans{\tfrac{1}{3}}$, \quad $\ans{1}$, \quad $\ans{3}$, \quad $\ans{9}$ --- plot and join smoothly.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $y = 5 \cdot 2^{\,x}$, find the outputs at $x = -1$, $0$ and $2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.5}$, $\ans{5}$, $\ans{20}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $4 \cdot 2^{\,x}$ is not $8^{\,x}$, and $3^{-2}$ is not $-9$. The coefficient stays outside the power, and a negative exponent flips the power over rather than changing its sign.
\end{watchoutbox}

\skillheading{Skill 3 --- Solving by matching powers}

\begin{formulabox}
\textbf{Method:} to solve $a \cdot b^{\,x} = c$, divide both sides by $a$ \emph{first}, then rewrite the right-hand side as a power of $b$. Equal powers of the same base have equal exponents.\par
\textbf{Powers worth knowing:} powers of two run $2, 4, 8, 16, 32, 64$; powers of three run $3, 9, 27, 81$; powers of five run $5, 25, 125, 625$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3 \cdot 4^{\,x} = 192$.
\step{The coefficient blocks the power comparison, so clear it before rewriting anything as a power.}
\step{Divide by $3$: $4^{\,x} = 64$.}
\step{$64 = 4^3$, so the exponents match. Check: $3 \cdot 64 = 192$.}

$x = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2 \cdot 5^{\,x} = 1250$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{4}$}
\end{tryitbox}
