\wstitleblock{Which Quadrilateral Property Implies Which?}{High School Geometry}{}

\begin{center}
\small\textit{Find and fix the reasoning, then say which one-way implication was reversed}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Every square is a rectangle; not every rectangle is a square. Statements like this run in \emph{one} direction only --- watch for work that runs them backwards.
  \item Where a problem asks you to explain, write in full sentences and name the property you are using.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem{In parallelogram $ABCD$, $m\angle A = (5x + 15)^\circ$ and $m\angle B = (3x + 37)^\circ$. Kim reasons that ``a parallelogram has right angles'', writes $5x + 15 = 90$, and reports $x = 15$.
\par\vspace{0.15cm}
(a) Find the correct value of $x$. \quad \ansblank
\par\vspace*{2.7cm}
(b) Kim's arithmetic is fine; her assumption is not. Explain which one-way implication she reversed, and name the property she should have used.
\par\vspace*{3.2cm}}

\problem{In quadrilateral $JKLM$, $\overline{JK} \parallel \overline{LM}$ and $JK = LM$, with $JK = 3a + 5$ and $LM = 7a - 11$.
\par\vspace{0.15cm}
(a) Find $a$. \quad \ansblank
\par\vspace*{2.7cm}
(b) These conditions do force $JKLM$ to be a parallelogram. Explain why they do not force it to be a rectangle.
\par\vspace*{3.2cm}}

\problem{The diagonals of rhombus $ABCD$ meet at $E$, with $AE = 9$ and $BE = 12$.
\par\vspace{0.15cm}
(a) Find the side length $AB$. \quad \ansblank
\par\vspace*{2.7cm}
(b) A classmate says that since every side of $ABCD$ is now the same length, $ABCD$ must be a square. Explain which one-way implication she reversed.
\par\vspace*{3.2cm}}

\problem{Quadrilateral $WXYZ$ has diagonals that bisect each other and are congruent, with $WY = 6y - 4$ and $XZ = 2y + 16$. Ravi reads ``bisect each other'' as ``each diagonal is half of the other'', writes $6y - 4 = \frac{1}{2}(2y + 16)$, and reports $y = 2.4$.
\par\vspace{0.15cm}
(a) Find the correct value of $y$. \quad \ansblank
\par\vspace*{2.6cm}
(b) Find the length of each diagonal. \quad \ansblank
\par\vspace*{2.6cm}
(c) Explain why $WXYZ$ must be a rectangle but need not be a square.
\par\vspace*{3.0cm}}

\problem{Kite $PQRS$ has $PQ = QR$ and $PS = SR$, with $PQ = 4t - 1$, $QR = 2t + 7$ and $PS = 3t + 2$.
\par\vspace{0.15cm}
(a) Find $t$. \quad \ansblank
\par\vspace*{2.6cm}
(b) Find $PS$. \quad \ansblank
\par\vspace*{2.6cm}
(c) Every rhombus is a kite. Explain why this kite is not a rhombus.
\par\vspace*{3.0cm}}

\problem{Quadrilateral $ABCD$ has vertices $A(0, 0)$, $B(6, 0)$, $C(9, 4)$ and $D(3, 4)$. Its opposite sides are already known to be congruent: $AB = DC = 6$ and $AD = BC = 5$.
\par\vspace{0.15cm}
(a) Find the length of diagonal $AC$, to the nearest hundredth. \quad \ansblank
\par\vspace*{2.6cm}
(b) Find the length of diagonal $BD$. \quad \ansblank
\par\vspace*{2.6cm}
(c) Explain why $ABCD$ is a parallelogram but is not a rectangle. Use your two diagonal lengths.
\par\vspace*{3.0cm}}
