\wstitleblock{Modeling Loci and Intersections with Conic Sections}{Algebra 2 --- Conics and Modeling}{}

\noindent\small
Circles built from a distance condition, parabolas built from a focus and a
directrix, intersections of a line with a conic, and intersections of two
conics. Two ideas run through the whole sheet:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{A locus is a sentence about distance.} Translate the sentence
        into a distance equation, square both sides, and simplify. A circle is
        ``distance to one point is constant''; a parabola is ``distance to a
        point equals distance to a line''.
  \item \textbf{An intersection is a system.} Substitute one equation into the
        other, solve the single-variable equation that results, then substitute
        back for the partner coordinate. Report ordered pairs, not just
        $x$ values.
  \item Reference forms: $(x-h)^2 + (y-k)^2 = r^2$ for a circle with center
        $(h, k)$; $x^2 = 4py$ for a parabola with vertex at the origin, focus
        $(0, p)$ and directrix $y = -p$.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{A circle has center $(3, -2)$ and radius $5$.
\par\smallskip
Evaluate $(x-3)^2 + (y+2)^2$ at the point $(7, 1)$, and state whether that
point lies on the circle.}

\problem[5cm]{A circle is the set of all points $5$ units from the origin, so
its equation is $x^2 + y^2 = 25$.
\par\smallskip
Find both $y$ values on this circle when $x = 3$.}

\problem[5.5cm]{A parabola has focus $(0, 3)$ and directrix $y = -3$, so
$p = 3$ and its equation is $x^2 = 4py$.
\par\smallskip
Find the $y$ coordinate of the point on this parabola whose $x$ coordinate is
$6$.}

\problem[5cm]{For the same parabola from problem 3, $x^2 = 12y$.
\par\smallskip
Find both $x$ values on the parabola when $y = 12$.}

\problem[7cm]{Find every point where the line $y = x + 1$ meets the circle
$x^2 + y^2 = 25$. Give your answers as ordered pairs.}

\problem{Complete the square to rewrite
\[ x^2 + y^2 - 6x + 8y - 11 = 0 \]
in center-radius form, then state the center and the radius.
\par\vspace*{7.5cm}\ansline}

\problem[7.5cm]{Find every point where the circle $x^2 + y^2 = 25$ meets the
parabola $y = x^2 - 5$. Give your answers as ordered pairs.}

\problem[7cm]{Find every point where the line $y = 2x - 3$ meets the parabola
$y = x^2 - 2x + 1$, and say what the number of solutions tells you about how
the line sits against the curve.}

\problem{A point $P(x, y)$ moves so that it is always the same distance from
the point $(0, 5)$ as it is from the line $y = -5$.
\par\smallskip
(a) Starting from $\sqrt{x^2 + (y-5)^2} = y + 5$, derive the equation of the
locus and simplify it.
\par\vspace*{5cm}
(b) Using your equation, find both points on the locus with $y = 5$.
\par\vspace*{3.5cm}
\hfill (b) \underline{\hspace{3.4cm}}\noansline}

\problem[8.5cm]{Find every point where the circle $x^2 + y^2 = 25$ meets the
hyperbola $xy = 12$. Give your answers as ordered pairs.}

\problem{A point $P(x, y)$ moves so that its distance from $A(6, 0)$ is always
twice its distance from $B(0, 0)$.
\par\smallskip
Write $PA = 2 \cdot PB$ as a distance equation, square both sides, and simplify
until the locus is in center-radius form. State the center and the radius.
\par\vspace*{9cm}\ansline}

\problem{\textbf{Synthesis challenge.} The cross-section of a satellite dish
is the parabola $x^2 = 8y$, whose focus is $(0, 2)$ and whose directrix is
$y = -2$. A circular guard ring around it has equation $x^2 + y^2 = 20$.
\par\smallskip
(a) Find every point where the ring meets the dish.
\par\vspace*{4.5cm}
\hfill (a) \underline{\hspace{3.4cm}}
\par\smallskip
(b) Take the intersection point with positive $x$. Find its distance from the
focus $(0, 2)$, and separately find its distance from the directrix $y = -2$.
\par\vspace*{3.5cm}
\hfill (b) \underline{\hspace{3.4cm}}
\par\smallskip
(c) Explain what your two answers in part (b) say about every point of the
dish.
\par\vspace*{2.5cm}\noansline}
