\wstitleblock{Finding the Fault in a Proof}{High-school geometry}{}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A proof step is \textbf{circular} when its reason is the thing being proved, or something that depends on it. A step is \textbf{unsupported} when its reason is not a given, a definition, a postulate, or an earlier line --- ``it looks that way in the picture'' is not a reason.
  \item For each item: say \emph{which line} fails and \emph{why}, then repair it. Two of the items also ask you to redo the arithmetic correctly.
  \item Diagrams show labelling only; all measurements are in the problem text.
\end{itemize}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.4pt}

\problem{Two lines cross at $V$. A student writes the proof below.
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.35}
\begin{center}
\begin{tabular}{|p{6.0cm}|p{6.6cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $\overleftrightarrow{AC}$ and $\overleftrightarrow{BD}$ intersect at $V$ & 1.\ Given \\ \hline
2.\ $\angle AVB$ and $\angle CVD$ are a pair of opposite angles & 2.\ Definition of vertical angles \\ \hline
3.\ $\angle AVB \cong \angle CVD$ & 3.\ Because $\angle AVB$ and $\angle CVD$ are congruent \\ \hline
\end{tabular}
\end{center}}
\par\vspace{0.1cm}
(a) Name the line that fails, and say what is wrong with its reason.
\par\vspace*{3.4cm}
(b) Write a reason that would make the proof valid.
\par\vspace*{3.0cm}\noansline}

\problem{Ari is asked to solve $4x - 6 = 2x + 10$ and writes:
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.35}
\begin{center}
\begin{tabular}{|p{6.0cm}|p{6.6cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $4x - 6 = 2x + 10$ & 1.\ Given \\ \hline
2.\ $4x = 2x + 10$ & 2.\ Addition property of equality \\ \hline
3.\ $2x = 10$ & 3.\ Subtraction property of equality \\ \hline
4.\ $x = 5$ & 4.\ Division property of equality \\ \hline
\end{tabular}
\end{center}}
\par\vspace{0.1cm}
(a) Ari's answer is $x = 5$. Name the line where the mistake happens and say what rule was broken.
\par\vspace*{3.2cm}
(b) Redo the work correctly and give the true value of $x$.
\par\vspace*{3.2cm}\ansline}

\problem{$\overrightarrow{BD}$ is drawn inside $\angle ABC$. A student writes the proof below. The only given is that $\overrightarrow{BD}$ lies in the interior of $\angle ABC$.
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.9]
  \coordinate (B) at (0,0);
  \draw[thick] (B) -- (4,0);
  \draw[thick] (B) -- (3.2,2.1);
  \draw[thick] (B) -- (2.1,3.3);
  \fill (B) circle (1.6pt);
  \node[below right] at (4,0) {$C$}; \node[right] at (3.2,2.1) {$D$};
  \node[above] at (2.1,3.3) {$A$}; \node[below left] at (B) {$B$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
{\renewcommand{\arraystretch}{1.35}
\begin{center}
\begin{tabular}{|p{6.0cm}|p{6.6cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $\overrightarrow{BD}$ is interior to $\angle ABC$ & 1.\ Given \\ \hline
2.\ $\angle ABD \cong \angle DBC$ & 2.\ The two halves look equal in the figure \\ \hline
3.\ $m\angle ABD = m\angle DBC$ & 3.\ Definition of congruent angles \\ \hline
\end{tabular}
\end{center}}
\par\vspace{0.1cm}
(a) Which line is unsupported, and why does its reason not count?
\par\vspace*{3.2cm}
(b) State what would have to be added to the given information to make the proof valid, and name the reason line 2 would then use.
\par\vspace*{3.0cm}\noansline}

\problem{$M$ is the midpoint of $\overline{PQ}$, with $PM = 5x - 3$ and $MQ = 2x + 9$ in centimetres. Sam writes:
\par\vspace{0.15cm}
{\renewcommand{\arraystretch}{1.35}
\begin{center}
\begin{tabular}{|p{6.0cm}|p{6.6cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $M$ is the midpoint of $\overline{PQ}$ & 1.\ Given \\ \hline
2.\ $5x - 3 = 2x + 9$ & 2.\ Definition of midpoint \\ \hline
3.\ $3x = 6$ & 3.\ Subtraction property of equality \\ \hline
4.\ $x = 2$ & 4.\ Division property of equality \\ \hline
\end{tabular}
\end{center}}
\par\vspace{0.1cm}
(a) Lines 1 and 2 are correct. Name the first faulty line and describe the mistake exactly.
\par\vspace*{3.2cm}
(b) Redo the work correctly and give the true value of $x$.
\par\vspace*{3.2cm}\ansline}

\problem{A student is asked why two lines $j$ and $k$ cut by a transversal are parallel, and argues:
\par\vspace{0.15cm}
\begin{quote}
``The corresponding angles are congruent, so $j \parallel k$. And I know the corresponding angles are congruent because $j$ and $k$ are parallel.''
\end{quote}
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.9]
  \draw[thick, <->] (-3,1.2) -- (3,1.2) node[right] {$j$};
  \draw[thick, <->] (-3,-1.2) -- (3,-1.2) node[right] {$k$};
  \draw[thick, <->] (-2.0,-2.2) -- (2.0,2.2) node[above right] {$t$};
  \node[font=\small] at (0.4,1.75) {$\alpha$};
  \node[font=\small] at (-1.8,-0.72) {$\beta$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
(a) Explain why this argument proves nothing.
\par\vspace*{3.4cm}
(b) Rewrite it as a valid argument. Say what you are given at the start and which statement you are entitled to conclude.
\par\vspace*{3.4cm}\noansline}

\problem{$\angle GKH$ and $\angle HKJ$ form a linear pair. In degrees, $m\angle GKH = 3x + 20$ and $m\angle HKJ = 5x$. A student writes:
\par\vspace{0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.95]
  \draw[thick] (-3,0) -- (3,0);
  \draw[thick] (0,0) -- (1.5,2.0);
  \fill (0,0) circle (1.6pt);
  \node[below left] at (-3,0) {$G$}; \node[below right] at (3,0) {$J$};
  \node[above] at (1.5,2.0) {$H$}; \node[below] at (0,-0.05) {$K$};
\end{tikzpicture}
\end{center}
\par\vspace{0.1cm}
{\renewcommand{\arraystretch}{1.35}
\begin{center}
\begin{tabular}{|p{6.0cm}|p{6.6cm}|}\hline
\multicolumn{1}{|c|}{\textbf{Statement}} & \multicolumn{1}{c|}{\textbf{Reason}} \\ \hline
1.\ $\angle GKH$ and $\angle HKJ$ form a linear pair & 1.\ Given \\ \hline
2.\ $m\angle GKH = m\angle HKJ$ & 2.\ Vertical angles are congruent \\ \hline
3.\ $3x + 20 = 5x$, so $x = 10$ & 3.\ Properties of equality \\ \hline
\end{tabular}
\end{center}}
\par\vspace{0.1cm}
(a) Line 2 is unsupported. Say why the reason given does not apply here, and state the correct relationship and the postulate that supplies it.
\par\vspace*{3.4cm}
(b) Using the correct relationship, find the true value of $x$.
\par\vspace*{3.2cm}\ansline}
