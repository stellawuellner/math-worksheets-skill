\sstitleblock{Sorting Shapes by What Counts}

% =================== SKILL 1 ===================
\skillheading{Sort by the number of sides}

\begin{formulabox}
\textbf{Rule:} Count the straight sides of each shape, then group the equal
counts: 3 sides $\to$ triangles, 4 sides $\to$ squares and rectangles,
6 sides $\to$ hexagons.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A set has a triangle, a square, a tilted triangle,
and a rectangle. How many shapes go in the 3-sided group?
\step{Count sides one shape at a time --- tilt does not change a count.}
\step{Triangle: 3. Square: 4. Tilted triangle: 3. Rectangle: 4. Two shapes
have 3 sides.}

$\Rightarrow\quad$ the 3-sided group holds $\ans{2}$ shapes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A set has 8 shapes, and 5 of them have four sides.
How many shapes are NOT in the 4-sided group?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Sort by the number of corners}

\begin{formulabox}
\textbf{Rule:} Corners work just like sides: count where two sides meet.
A shape with 3 sides always has 3 corners. After sorting, you can compare
group sizes by subtracting.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad After a corner sort, the 3-corner group has 5 shapes
and the 4-corner group has 3 shapes. How many more shapes are in the
3-corner group?
\step{Comparing two group sizes means finding their difference --- subtract
the counts.}
\step{$5 - 3 = 2$.}

$\Rightarrow\quad$ the 3-corner group has $\ans{2}$ more shapes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One group has 7 shapes and another has 4. How many
more shapes are in the bigger group?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Defining attributes, not looks}

\begin{formulabox}
\textbf{Rule:} DEFINING attributes never change: number of sides, number of
corners, closed or open. Color, size, and tilt are just looks --- they can
change without changing the shape's name.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A big gray triangle and a tiny white triangle: how
many sides does each one have?
\step{Ignore the looks (big, small, gray, white) and count what defines the
shape.}
\step{Each one has exactly 3 straight sides --- so both are triangles, same
group.}

$\Rightarrow\quad$ each has $\ans{3}$ sides
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tilted square and an upright square: how many equal
sides does each have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$ --- same group!}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``It looks different'' is not a sorting reason. If
you cannot COUNT it, do not sort by it.
\end{watchoutbox}
