\wstitleblock{Measurement Review --- Volume, Conversions \& Angles}{Grade 5}{}
\begin{center}
\textit{Volume, unit conversion, and angle measure}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item The problems are mixed on purpose --- decide what each one is asking
        for (a volume? a converted unit? an angle?) before computing.
  \item Show your work, and write your answer on the labeled answer line.
\end{itemize}

\vspace{0.2cm}

\problem{Convert: \quad $3$ meters $=$ how many centimeters?
\par\vspace*{3cm}\answerline{cm}}

\problem{A tissue box is $4$ cm long, $2$ cm wide, and $3$ cm tall. What is
its volume?
\par\vspace*{3.5cm}\answerline{cm$^3$}}

\problem{Two angles together make a straight line. One measures $115^\circ$.
What is the measure of the other angle?
\par\vspace*{3.5cm}\answerline{degrees}}

\problem{A bag of rice weighs $2.5$ kilograms. How many grams is that?
\par\vspace*{3.5cm}\answerline{g}}

\problem{A right angle is split into two smaller angles. One measures
$35^\circ$. What is the measure of the other?
\par\vspace*{3.5cm}\answerline{degrees}}

\problem{A step stool is built from two boxes glued on top of each other.
Both boxes are $5$ cm long and $4$ cm wide; the bottom box is $3$ cm tall
and the top box is $2$ cm tall. What is the total volume of the stool?
\par\vspace*{5cm}\answerline{cm$^3$}}

\problem{A fun run is $3$ kilometers long. Mia has already run $1250$
meters. How many meters does she have left?
\par\vspace*{4.5cm}\answerline{m}}

\problem{The minute hand of a clock sweeps a full turn of $360^\circ$ in
$60$ minutes. How many degrees does it sweep in $20$ minutes?
\par\vspace*{4.5cm}\answerline{degrees}}

\problem{A movie lasts $2$ hours $15$ minutes. How many minutes is that in
all?
\par\vspace*{4cm}\answerline{min}}

\problem{A box has a volume of $72$ cubic centimeters. Its base is $4$ cm
long and $3$ cm wide. How tall is the box?
\par\vspace*{5cm}\answerline{cm}}

\problem{A juice jug holds $2$ liters. Every $1000$ milliliters is one
liter, and one paper cup holds $250$ milliliters. How many cups can be
filled from a full jug?
\par\vspace*{5cm}\ansline}

\problem{\textbf{Challenge.} A fish tank is $50$ cm long, $20$ cm wide, and
$30$ cm deep. Every $1000$ cubic centimeters of water is one liter.
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item[(a)] How much water fills the tank, in cubic centimeters?
        \par\vspace*{2.2cm}\answerline{cm$^3$}
  \item[(b)] How many liters is that?\par\vspace*{1.8cm}\answerline{L}
  \item[(c)] Explain why dividing by $1000$ turns cubic centimeters into
        liters.\par\vspace*{1.8cm}\ansline
\end{itemize}}
