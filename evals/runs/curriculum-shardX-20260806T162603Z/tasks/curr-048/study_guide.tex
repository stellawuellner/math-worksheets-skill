\sstitleblock{Position Words for Shapes}

% =================== SKILL 1 ===================
\skillheading{Count the shapes in one place}

\begin{formulabox}
\textbf{Rule:} First find the PLACE the question names (top shelf, bottom
row, between two shapes). Then count only the shapes in that place ---
nothing outside it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 5 shapes stand in a row. How many shapes are BETWEEN
the first one and the last one?
\step{``Between'' leaves out the two end shapes --- so count the row, then
take the two ends away.}
\step{$5 - 2 = 3$.}

$\Rightarrow\quad \ans{3}$ shapes are between the ends
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 6 shapes stand in a row. How many are between the
first and the last?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Name the shape at a position}

\begin{formulabox}
\textbf{Rule:} Position words point at ONE spot: below means straight down,
next to means touching distance, right and left follow YOUR hands. Find the
spot first, then name the shape by its sides and corners.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The shape below the circle has 6 straight sides.
How many corners does it have --- and what is it?
\step{Find the spot (straight down from the circle), then use the side
count: sides and corners always match.}
\step{6 sides means 6 corners --- that shape is a hexagon.}

$\Rightarrow\quad$ it has $\ans{6}$ corners: a hexagon
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The shape to the right of the door has 3 straight
sides. How many corners does it have, and what is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ corners --- a triangle}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Use position words yourself}

\begin{formulabox}
\textbf{Rule:} A good ``where'' sentence names the shape, a position word,
and a neighbor: \emph{``The star is ABOVE the box.''} Two position words
make the spot even clearer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The ball sits under the table. Finish: ``The ball is
\rule{1.2cm}{0.4pt} the table.''
\step{Ask: where is the ball when you look from the table? It is lower ---
straight down.}
\step{Lower and straight down is the word BELOW (or under).}

The sentence reads: the ball is \textbf{below} the table.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The lamp sits higher than the shelf, straight up.
Finish: ``The lamp is \rule{1.2cm}{0.4pt} the shelf.''\\[2pt]
\rotatebox{180}{\footnotesize check: above (or over)}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Left and right swap when a friend faces you! On
paper, use YOUR left hand and right hand --- the page always faces the same
way.
\end{watchoutbox}
