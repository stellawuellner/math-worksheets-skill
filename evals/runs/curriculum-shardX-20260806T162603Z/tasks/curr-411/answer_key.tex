\problem{Solve for $x$: \qquad $3^{\,x+2} = 3^{\,5}$}

\textbf{Solution:} Both sides are already powers of $3$. Because $b^{M} = b^{N}$ with $b > 0$, $b \neq 1$ forces $M = N$, set the exponents equal: $x + 2 = 5$, so \ans{x = 3}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $4^{\,x} = 64$}

\textbf{Solution:} Rewrite the right side with base $4$: $64 = 4^{3}$. The equation becomes $4^{\,x} = 4^{3}$, so $x = 3$. \ans{x = 3}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{The table gives values of $f(x) = 2^{\,x}$ for $x = 1, 2, 3, 4$. \textbf{(a)} Find $f(5)$. \textbf{(b)} Write $256$ as a power of $2$ and solve $2^{\,x} = 256$.}

\textbf{Solution (a):} Each row multiplies the previous value by $2$, so $f(5) = 16 \cdot 2 = 32$. Directly, $f(5) = 2^{5} = 32$. \ans{f(5) = 32}

\textbf{Solution (b):} Keep doubling past the table: $2^{5} = 32$, $2^{6} = 64$, $2^{7} = 128$, $2^{8} = 256$. So $2^{\,x} = 2^{8}$ and the exponents match. \ans{x = 8}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $5^{\,2x-1} = 5^{\,x+4}$}

\textbf{Solution:} Same base on both sides, so equate exponents:
\begin{align*}
  2x - 1 &= x + 4 \\
  x &= 5
\end{align*}
Check: $5^{2(5)-1} = 5^{9}$ and $5^{5+4} = 5^{9}$. \ans{x = 5}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $8^{\,x} = 32$}

\textbf{Solution:} Neither side is a power of the other, but both are powers of $2$: $8 = 2^{3}$ and $32 = 2^{5}$.
\begin{align*}
  \left(2^{3}\right)^{x} &= 2^{5} \\
  2^{3x} &= 2^{5} \\
  3x &= 5
\end{align*}
So \ans{x = \tfrac{5}{3}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$P(t) = 50 \cdot 2^{\,t}$ cells after $t$ hours. \textbf{(a)} Find $P(5)$. \textbf{(b)} Solve $50 \cdot 2^{\,t} = 3200$.}

\textbf{Solution (a):} $P(5) = 50 \cdot 2^{5} = 50 \cdot 32 = 1600$ cells. \ans{P(5) = 1600}

\textbf{Solution (b):} Divide by the initial amount first, so the equation becomes a pure power of $2$: $2^{\,t} = \dfrac{3200}{50} = 64 = 2^{6}$, hence $t = 6$ hours. \ans{t = 6}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $9^{\,x-1} = 27^{\,x}$}

\textbf{Solution:} Both $9$ and $27$ are powers of $3$: $9 = 3^{2}$, $27 = 3^{3}$.
\begin{align*}
  3^{\,2(x-1)} &= 3^{\,3x} \\
  2x - 2 &= 3x \\
  -2 &= x
\end{align*}
So \ans{x = -2}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $7^{\,x^{2}} = 7^{\,3x+4}$}

\textbf{Solution:} The bases already match, so the exponents are equal --- and this time the resulting equation is quadratic.
\begin{align*}
  x^{2} &= 3x + 4 \\
  x^{2} - 3x - 4 &= 0 \\
  (x - 4)(x + 1) &= 0
\end{align*}
Both roots are legitimate: an exponent may be any real number, so neither is extraneous.
\[ \boxed{x = -1 \quad \text{or} \quad x = 4} \]

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{The table gives values of $h(x) = 3^{\,x}$. \textbf{(a)} Find $h(-3)$. \textbf{(b)} Solve $3^{\,x} = \tfrac{1}{81}$.}

\textbf{Solution (a):} Moving left in the table divides by $3$ each step, and a negative exponent is a reciprocal: $h(-3) = 3^{-3} = \dfrac{1}{3^{3}} = \dfrac{1}{27}$. \ans{h(-3) = \tfrac{1}{27}}

\textbf{Solution (b):} Write the right side as a power of $3$: $81 = 3^{4}$, so $\dfrac{1}{81} = 3^{-4}$. Then $3^{\,x} = 3^{-4}$ and the exponents match. \ans{x = -4}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\left(\dfrac{1}{2}\right)^{\,x+1} = 8^{\,x-3}$}

\textbf{Solution:} A reciprocal base is a negative exponent: $\dfrac{1}{2} = 2^{-1}$, and $8 = 2^{3}$.
\begin{align*}
  \left(2^{-1}\right)^{x+1} &= \left(2^{3}\right)^{x-3} \\
  2^{\,-x-1} &= 2^{\,3x-9} \\
  -x - 1 &= 3x - 9 \\
  8 &= 4x
\end{align*}
So \ans{x = 2}
