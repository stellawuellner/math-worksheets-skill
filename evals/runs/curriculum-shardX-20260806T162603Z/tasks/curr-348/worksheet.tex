\wstitleblock{Dilations and Similar Figures in Practice}{High School Geometry}{}

\begin{center}
\small\textit{Scale drawings, models, shadows and enlargements --- every one built from a single scale factor}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Dilation about the origin by factor $k$: $(x, y) \to (kx,\ ky)$. About a centre $(h, j)$: $(x, y) \to \bigl(h + k(x - h),\ \ j + k(y - j)\bigr)$.
  \item A dilation multiplies every length by $k$ and leaves every angle alone, so the image is similar to the original.
  \item Units are stated in every problem --- give your answer in the unit asked for.
\end{itemize}

\problem[4.6cm]{A photograph $12$ cm wide is enlarged so that the copy is $30$ cm wide. Find the scale factor of the enlargement.}

\problem{A blueprint is drawn at a scale of $1$ cm to $40$ cm. A wall measures $8.5$ cm on the blueprint. Find the real length of that wall.

\par\vspace*{2.0cm}

\problem[5.0cm]{A model car is $18$ cm long. The real car it copies is $4.5$ m long, which is $450$ cm. Find how many times longer the real car is than the model.}

\problem{A student $1.6$ m tall casts a shadow $2.4$ m long. At that same moment a flagpole casts a shadow $13.5$ m long. The student and the flagpole form similar right triangles with their shadows. Find the height of the flagpole.
\par\vspace*{4.4cm}\answerline{m}}

\problem{An architect scales a courtyard plan by a factor of $2.5$. The original courtyard is a rectangle $12$ m by $8$ m.
\par\vspace{0.15cm}
(a) Find the perimeter of the scaled courtyard.
\par\vspace*{2.2cm}\answerline{m}
\par\vspace{0.25cm}

\par\vspace*{2.2cm}\answerline{m$^2$}
