\problem{School store: $2x + y = 11$, $x + 3y = 18$. \\
\textbf{Target.} The $y$ coefficients are $1$ and $3$, so scaling the first
equation by $3$ matches them --- cheaper than matching $x$.
\begin{align*}
  3(2x + y) &= 3(11) &&\Longrightarrow&& 6x + 3y = 33 \\
  \text{subtract } x + 3y &= 18 &&\Longrightarrow&& 5x = 15,\ \ x = 3
\end{align*}
Back-substitute into $2x + y = 11$: $6 + y = 11$, so $y = 5$.
Check in the second equation: $3 + 3(5) = 18$ $\checkmark$
\par A notebook costs \$$3$ and a binder costs \$$5$. \ans{(x, y) = (3, 5)}}

\problem{$4x - 3y = 5$, $x + y = 3$. \\
\textbf{Target.} Scale the second equation by $3$ so the $y$ terms are $-3y$ and
$+3y$.
\begin{align*}
  3(x + y) &= 3(3) &&\Longrightarrow&& 3x + 3y = 9 \\
  \text{add to } 4x - 3y &= 5 &&\Longrightarrow&& 7x = 14,\ \ x = 2
\end{align*}
Back-substitute into $x + y = 3$: $y = 1$. Check:
$4(2) - 3(1) = 5$ $\checkmark$ \ans{(x, y) = (2, 1)}}

\problem{Theater tickets: $a + c = 12$, $9a + 5c = 88$. \\
\textbf{Target.} Scale the count equation by $5$ to match the $5c$ term.
\begin{align*}
  5(a + c) &= 5(12) &&\Longrightarrow&& 5a + 5c = 60 \\
  \text{subtract from } 9a + 5c &= 88 &&\Longrightarrow&& 4a = 28,\ \ a = 7
\end{align*}
Then $c = 12 - 7 = 5$. Check the money equation:
$9(7) + 5(5) = 63 + 25 = 88$ $\checkmark$
\par The theater sold $7$ adult tickets and $5$ child tickets.
\ans{(a, c) = (7, 5)}}

\problem{$3x + 4y = 10$, $5x + 6y = 16$. \\
\textbf{Target.} No coefficient divides another, so scale \emph{both}. For $x$,
the least common multiple of $3$ and $5$ is $15$.
\begin{align*}
  5(3x + 4y) &= 5(10) &&\Longrightarrow&& 15x + 20y = 50 \\
  3(5x + 6y) &= 3(16) &&\Longrightarrow&& 15x + 18y = 48
\end{align*}
Subtract: $2y = 2$, so $y = 1$. Back-substitute into $3x + 4y = 10$:
$3x = 6$, $x = 2$. Check: $5(2) + 6(1) = 16$ $\checkmark$
\ans{(x, y) = (2, 1)}}

\problem{$5x + 2y = 4$, $3x + y = 3$. \\
\textbf{Target.} The $y$ coefficients are $2$ and $1$, so scale the second
equation by $2$.
\begin{align*}
  2(3x + y) &= 2(3) &&\Longrightarrow&& 6x + 2y = 6 \\
  \text{subtract } 5x + 2y &= 4 &&\Longrightarrow&& x = 2
\end{align*}
Back-substitute into $3x + y = 3$: $6 + y = 3$, so $y = -3$. Check the first
equation: $5(2) + 2(-3) = 10 - 6 = 4$ $\checkmark$ \ans{(x, y) = (2, -3)}}

\problem{Bakery: $m + s = 50$, $3m + 4s = 172$. \\
\textbf{Target.} Scale the count equation by $4$ to match the $4s$ term.
\begin{align*}
  4(m + s) &= 4(50) &&\Longrightarrow&& 4m + 4s = 200 \\
  \text{subtract from } 3m + 4s &= 172 &&\Longrightarrow&& -m = -28,\ \ m = 28
\end{align*}
Then $s = 50 - 28 = 22$. Check the money equation:
$3(28) + 4(22) = 84 + 88 = 172$ $\checkmark$
\par The bakery sold $28$ muffins and $22$ scones. \ans{(m, s) = (28, 22)}}

\problem{$4x + 5y = 7$, $6x + 7y = 11$. \\
\textbf{Target.} Both need scaling. Eliminating $x$ is cheaper: the least common
multiple of $4$ and $6$ is $12$, while $y$ would need $35$.
\begin{align*}
  3(4x + 5y) &= 3(7) &&\Longrightarrow&& 12x + 15y = 21 \\
  2(6x + 7y) &= 2(11) &&\Longrightarrow&& 12x + 14y = 22
\end{align*}
Subtract: $y = -1$. Back-substitute into $4x + 5y = 7$: $4x - 5 = 7$, so
$x = 3$. Check: $6(3) + 7(-1) = 18 - 7 = 11$ $\checkmark$
\ans{(x, y) = (3, -1)}}

\problem{Canoe: $2b + 2c = 24$, $3b - 3c = 24$. \\
\textbf{Target.} The $c$ coefficients are $2$ and $-3$; their least common
multiple is $6$, so scale both.
\begin{align*}
  3(2b + 2c) &= 3(24) &&\Longrightarrow&& 6b + 6c = 72 \\
  2(3b - 3c) &= 2(24) &&\Longrightarrow&& 6b - 6c = 48
\end{align*}
Add: $12b = 120$, so $b = 10$. Back-substitute into $2b + 2c = 24$:
$20 + 2c = 24$, so $c = 2$. Check the upstream equation:
$3(10) - 3(2) = 24$ $\checkmark$
\par The canoe moves at $10$ kilometres per hour in still water and the current
runs at $2$ kilometres per hour. \ans{(b, c) = (10, 2)}}

\problem{Maya's mistake on $x + 4y = 14$, $3x - 2y = 14$. \\
\textbf{(a) What is wrong.} Maya multiplied only the left side by $3$ and left
the right side at $14$. That is a different equation, not the same one rewritten:
$3x + 12y = 14$ has different solutions from $x + 4y = 14$. Multiplying
\emph{every} term --- both sides --- by $3$ is allowed because it scales both
sides of a true statement equally, so exactly the same pairs $(x, y)$ still
satisfy it. \emph{This part is flagged for manual review; a student answer that
says ``she has to multiply the $14$ too, or the equation is no longer
equivalent'' earns full credit.}
\par \textbf{(b) The correct work.}
\begin{align*}
  3(x + 4y) &= 3(14) &&\Longrightarrow&& 3x + 12y = 42 \\
  \text{subtract } 3x - 2y &= 14 &&\Longrightarrow&& 14y = 28,\ \ y = 2
\end{align*}
Back-substitute into $x + 4y = 14$: $x + 8 = 14$, so $x = 6$. Check:
$3(6) - 2(2) = 18 - 4 = 14$ $\checkmark$ \ans{(x, y) = (6, 2)}}

\problem{Salt mixture: $x + y = 20$, $30x + 50y = 45 \cdot 20$. \\
\textbf{Setting up the second equation.} Percent times litres is the salt in
each part, and the salt in the two parts must equal the salt in the mixture:
$45 \cdot 20 = 900$.
\par \textbf{Target.} Scale the volume equation by $30$ to match the $30x$ term.
\begin{align*}
  30(x + y) &= 30(20) &&\Longrightarrow&& 30x + 30y = 600 \\
  \text{subtract from } 30x + 50y &= 900 &&\Longrightarrow&& 20y = 300,\ \ y = 15
\end{align*}
Then $x = 20 - 15 = 5$. Check the salt equation:
$30(5) + 50(15) = 150 + 750 = 900$ $\checkmark$
\par The technician uses $5$ litres of the $30\%$ solution and $15$ litres of the
$50\%$ solution. \ans{(x, y) = (5, 15)}}
