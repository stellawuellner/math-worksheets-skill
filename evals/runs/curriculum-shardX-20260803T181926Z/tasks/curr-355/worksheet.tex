\wstitleblock{Complex Numbers: Complete Solution Sets}{Algebra 2 \quad Complex Numbers}{}

\begin{center}
\textit{Simplifying complex products and powers of the imaginary unit, solving
quadratics with complex roots, testing a proposed complex solution, and
reporting the complete solution set of a polynomial}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Write every complex number in the standard form $a + bi$, with the real
part first.
\item ``Report the complete solution set'' means list \emph{every} solution,
real and non-real. A degree-$n$ polynomial equation has $n$ solutions counted
with multiplicity.
\item Remember $i^2 = -1$. That single replacement is what turns an imaginary
product back into a real number.
\item The problems mix methods on purpose: decide first whether the equation
wants square roots, factoring, or the quadratic formula.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Multiply and write the result in the form $a + bi$:
\quad $(2 + 3i)(4 - i)$}

\problem[4.5cm]{Simplify $i^{23}$. Write the answer as a single term.}

\problem[5cm]{Solve and report the complete solution set:
\quad $x^2 + 49 = 0$}

\problem[5cm]{Solve and report the complete solution set:
\quad $(x - 3)^2 + 16 = 0$}

\problem[6cm]{Show by substitution that $x = 1 + 4i$ is a solution of
$x^2 - 2x + 17 = 0$. Simplify all the way to a single number; state what that
number has to be for the check to succeed.}

\problem[6cm]{Find every zero of $p(x) = x^4 - 16$, real and non-real, and
list the complete zero set.}

\problem[6cm]{Solve and report the complete solution set:
\quad $x^2 + 4x + 13 = 0$}

\problem[6cm]{Find the complete zero set of $p(x) = x^3 + 4x$. State how many
zeros the degree of the polynomial promises, and check that you have that
many.}

\problem[6.5cm]{Ben substituted $x = -3 - 4i$ into $x^2 + 6x + 25$, got the
value $32$, and concluded that $-3 - 4i$ is not a solution of
$x^2 + 6x + 25 = 0$. Redo the substitution correctly, report the value you
get, and state in one sentence what Ben did wrong.}

\problem[6.5cm]{Find the complete solution set of $x^4 + 3x^2 - 4 = 0$.
(Hint: the equation is quadratic in $x^2$.)}

\problem[6.5cm]{Solve and report the complete solution set:
\quad $4x^2 - 4x + 5 = 0$}

\problem[7cm]{\textbf{Synthesis.} A classmate tells you that $x = 3i$ is one
zero of $p(x) = x^4 - 2x^3 + 10x^2 - 18x + 9$. Use that fact --- and what it
forces about a second zero --- to find the complete solution set of
$p(x) = 0$. Say which solution is repeated.}
