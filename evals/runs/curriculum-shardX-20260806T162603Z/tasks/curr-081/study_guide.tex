\sstitleblock{Unit Fractions in Area Models}

\skillheading{Skill 1 --- Naming the Unit Fraction in a Model}

\begin{formulabox}
\textbf{Rule:} a \textbf{unit fraction} is one equal piece of a whole. Its top number is always $1$;
its bottom number is how many equal pieces make the whole.\par
one piece out of $d$ equal pieces $= \frac{1}{d}$. \quad The shape does not matter --- bar, circle or
grid --- only the count of equal pieces does.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One piece of this bar is shaded. Name the unit fraction.
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}
  \fill[blue!30] (0,0) rectangle (0.7,0.55);
  \draw[thick] (0,0) rectangle (3.5,0.55);
  \foreach \x in {0.7,1.4,2.1,2.8}{\draw[thick] (\x,0) -- (\x,0.55);}
\end{tikzpicture}
\end{center}
\step{The question asks for one piece of a whole, so I need a unit fraction --- the top number is fixed at one and I only have to count the equal pieces.}
\step{Counting the pieces across the bar gives $5$ equal pieces, so the bottom number is $5$.}

one shaded piece $= \ans{\tfrac{1}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pizza is cut into $4$ equal slices. What unit fraction is one slice?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{4}}$}
\end{tryitbox}

\skillheading{Skill 2 --- The Pieces Must Be Equal}

\begin{formulabox}
\textbf{Watch the word ``equal.''} Counting pieces is not enough. A shape cut into three pieces only
shows \emph{thirds} when all three pieces are the same size.\par
Test it: could the pieces be swapped and still cover the same amount? If not, they are not equal
parts and no piece is a unit fraction of that whole.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Does the shaded piece show one third?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}
  \fill[red!25] (0,0) rectangle (0.5,0.55);
  \draw[thick] (0,0) rectangle (3,0.55);
  \draw[thick] (0.5,0) -- (0.5,0.55);
  \draw[thick] (1.2,0) -- (1.2,0.55);
\end{tikzpicture}
\end{center}
\step{Before naming any fraction I check the equal-parts condition, because a unit fraction is only defined when every piece is the same size.}
\step{The three pieces here are clearly different widths --- the shaded one is the narrowest --- so this model does not show thirds at all, and the shaded piece has no unit-fraction name.}

Answer: no. Three pieces, but not three \emph{equal} pieces.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A strip is cut into two pieces, one much longer than the other. Does the short
piece show one half? Say why.\\[2pt]
\rotatebox{180}{\footnotesize check: no --- halves must be two pieces of the same size}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing Unit Fractions}

\begin{formulabox}
\textbf{Rule:} when two wholes are the \emph{same size}, more pieces means each piece is
\emph{smaller}.\par
So for unit fractions, the bigger bottom number names the smaller piece:
$\frac{1}{2} > \frac{1}{3} > \frac{1}{4} > \frac{1}{6} > \frac{1}{8}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two same-size bars: one cut into $4$ pieces, one cut into $8$. Which shaded
piece is bigger?
\step{Both wholes are the same size, so I can compare the pieces just by comparing how many pieces each whole was cut into.}
\step{Cutting the same bar into $8$ pieces makes each piece smaller than cutting it into $4$, so the fourth is the bigger piece.}

$\ans{\tfrac{1}{4} > \tfrac{1}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same-size bars, one cut into $2$ pieces and one into $6$. Write $>$, $<$ or $=$
between $\frac{1}{2}$ and $\frac{1}{6}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2} > \tfrac{1}{6}}$}
\end{tryitbox}
