\sstitleblock{How to Count On Without Losing a Hop}

\skillheading{Skill 1 --- Say the first number, then start hopping}

\begin{formulabox}
\textbf{The rule:} the number you start from is \textbf{free}. It costs no hop.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Say the first number quietly, then begin counting with the
        \textbf{next} number.
  \item Take exactly one hop for each thing you are adding. Count the hops on
        your fingers so you can see them.
  \item If your answer is one too small, you counted the start number as a hop.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $7 + 2$
\step{Hold the first number in your head and remember it is free, so the very
first thing you say out loud is already hop one.}
\step{Seven \ldots\ then eight (hop one), nine (hop two)}

The total is $\ans{9}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $5 + 3$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Start from the bigger number}

\begin{formulabox}
\textbf{The rule:} you may add in either order, so start from the
\textbf{bigger} number and hop the smaller one.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Fewer hops means fewer chances to lose one.
  \item The total is the same both ways, so turning the problem around is
        always allowed.
  \item Look for the bigger number \emph{before} you start counting, not after.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $2 + 6$
\step{The second number is bigger, so turn the problem around first and hop the
small one to keep the hop count tiny.}
\step{Six \ldots\ then seven (hop one), eight (hop two)}

The total is $\ans{8}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $3 + 6$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 3 --- Count on to find the missing part}

\begin{formulabox}
\textbf{The rule:} to find how many \textbf{more} you need, count on from what
you have up to the total --- and the answer is the number of \textbf{hops}, not
the number of words you say.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Put up one finger for each hop. At the end, count your fingers.
  \item Check your answer by adding it back on: the two parts must make the
        whole.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad I have $6$. How many more do I need to have $10$?
\step{Count on from what I have up to the total, and put up a finger for each
hop so the hops can be counted instead of the words.}
\step{Six \ldots\ then seven, eight, nine, ten --- that is four fingers up}

I need $\ans{4}$ more.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad I have $4$. How many more do I need to have $7$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An answer that is exactly one too small almost always
means the first number was counted as a hop. Say it quietly, then start hopping
on the \emph{next} number.
\end{watchoutbox}
