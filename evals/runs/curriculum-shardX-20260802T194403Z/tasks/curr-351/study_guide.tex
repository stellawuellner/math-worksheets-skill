\sstitleblock{Interpreting Powers of $i$ and the Complex Plane}

\skillheading{Skill 1 --- The cycle of powers of $i$}
\begin{formulabox}\textbf{$i^{2} = -1$, and everything else follows.}\\[3pt]
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|c|}\hline
$i^{1}$ & $i^{2}$ & $i^{3}$ & $i^{4}$ & $i^{5}$ & $i^{6}$\\\hline
$i$ & $-1$ & $-i$ & $1$ & $i$ & $-1$\\\hline
\end{tabular}
\end{center}
Because $i^{4} = 1$, the powers repeat in blocks of four. \textbf{Divide the
exponent by 4 and keep only the remainder:} $0 \rightarrow 1$,
$1 \rightarrow i$, $2 \rightarrow -1$, $3 \rightarrow -i$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Simplify $i^{34}$.
\step{The exponent is large, so instead of multiplying it out, locate where 34
lands in the four-step cycle --- only the remainder matters.}
\step{$34 \div 4$ is 8 remainder 2, so $i^{34} = (i^{4})^{8}\cdot i^{2}
= 1 \cdot i^{2} = \ans{-1}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Simplify $i^{41}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{i}$}\end{tryitbox}

\skillheading{Skill 2 --- Plotting and adding on the plane}
\begin{formulabox}\textbf{The dictionary:} \ $a + bi \ \longleftrightarrow \
\text{the point } (a, b)$, with the real part on the horizontal \textbf{Re}
axis and the imaginary part on the vertical \textbf{Im} axis.\\[3pt]
Draw each number as an arrow from the origin. Then \textbf{adding is tip-to-tail
arrow addition}, which in symbols is just adding the parts separately:
$(a + bi) + (c + di) = (a + c) + (b + d)i$. The distance from the origin is
$|a + bi| = \sqrt{a^{2} + b^{2}}$ --- the Pythagorean theorem on the two legs.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $(5 - 2i) + (-3 + 6i)$
\step{Addition is the tip-to-tail move on the grid, and that move adds the
horizontal steps to each other and the vertical steps to each other.}
\step{Real parts: $5 + (-3) = 2$. Imaginary parts: $-2 + 6 = 4$. The tip lands
at $(2, 4)$, which is $\ans{2 + 4i}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $(4 + i) + (-6 + 2i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2 + 3i}$}\end{tryitbox}

\skillheading{Skill 3 --- Multiplying by $i$ is a quarter turn}
\begin{formulabox}\textbf{Multiplying by $i$ rotates the arrow $90^\circ$
counterclockwise about the origin and leaves its length alone.}\\[3pt]
In symbols: $i(a + bi) = ai + bi^{2} = -b + ai$, so the point $(a, b)$ moves to
$(-b, a)$ --- exactly a quarter turn.\\[2pt]
This is \emph{why} the powers of $i$ cycle: four quarter turns is one full turn,
so $i^{4} = 1$ brings every point home.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $i(2 - 5i)$
\step{Multiplying by $i$ should turn the arrow to $(2, -5)$ a quarter turn, so
expect the answer in the next quadrant counterclockwise --- up and to the right.}
\step{$i(2 - 5i) = 2i - 5i^{2} = 2i + 5 = \ans{5 + 2i}$, and indeed $(2, -5)$
has moved to $(5, 2)$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $i(-1 + 3i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3 - i}$}\end{tryitbox}

\skillheading{Skill 4 --- Non-real solutions and conjugates}
\begin{formulabox}\textbf{Rule:} a quadratic with real coefficients and a
negative discriminant has \emph{no} real solutions but exactly two complex ones,
and they are always a \textbf{conjugate pair}: $p + qi$ and $p - qi$, mirror
images across the real axis.\\[2pt]
Use $\sqrt{-k} = i\sqrt{k}$ for $k > 0$. On the graph, ``no real solutions''
just means the parabola misses the $x$-axis --- the solutions live off the real
line, not nowhere.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Solve $x^{2} - 6x + 25 = 0$.
\step{The discriminant $36 - 100$ is negative, so there are no real solutions
and the answers will be a conjugate pair --- complete the square to see them.}
\step{$(x - 3)^{2} + 16 = 0$, so $(x - 3)^{2} = -16$ and $x - 3 = \pm 4i$,
giving $\ans{3 + 4i}$ and $\ans{3 - 4i}$: the points $(3, 4)$ and $(3, -4)$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $x^{2} - 2x + 10 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1 + 3i}$ and $\ans{1 - 3i}$}\end{tryitbox}
