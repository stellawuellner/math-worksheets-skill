\wstitleblock{Halves and Fourths --- Are the Parts Really Equal?}{Kindergarten--Grade 1}{}

\noindent\small Some friends cut shapes into parts and gave the parts names.
A part is a \textbf{half} only when the two parts hold the \textbf{same} number
of squares. A part is a \textbf{fourth} only when all four parts hold the
\textbf{same} number of squares.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Count the little squares inside each part.
  \item If the counts are not the same, the parts are not equal shares.
  \item Then fix what the friend did wrong.
\end{itemize}
\normalsize

\vspace{0.15cm}

\problem[3.5cm]{Warm up. Rosa cut this rectangle into two parts. Count the squares
in each part.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[step=1, gray!55, thin] (0,0) grid (4,3);
  \draw[thick] (0,0) rectangle (4,3);
  \draw[line width=1.6pt] (2,0) -- (2,3);
  \node[font=\large] at (1,-0.5) {A};
  \node[font=\large] at (3,-0.5) {B};
\end{tikzpicture}
\end{center}
Are Part A and Part B the same size? Write $=$ if they are the same, or write
$<$ or $>$ to show which one is bigger.}

\problem[3.5cm]{\textbf{Find and fix.} Kim cut this rectangle and said,
``Part A is one half.'' Part A covers $3$ squares. Part B covers $9$ squares.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[step=1, gray!55, thin] (0,0) grid (4,3);
  \draw[thick] (0,0) rectangle (4,3);
  \draw[line width=1.6pt] (1,0) -- (1,3);
  \node[font=\large] at (0.5,-0.5) {A};
  \node[font=\large] at (2.5,-0.5) {B};
\end{tikzpicture}
\end{center}
Write $=$, $<$, or $>$ between the two counts to show what is really true.
Then tell a friend why Part A is not one half.}

\problem[4cm]{\textbf{Find and fix.} Ben cut this rectangle into two parts.
Part A has $4$ squares and Part B has $8$ squares. Ben wants both parts to be
halves, so he says, ``Move $4$ squares from Part B over to Part A.''
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[step=1, gray!55, thin] (0,0) grid (6,2);
  \draw[thick] (0,0) rectangle (6,2);
  \draw[line width=1.6pt] (2,0) -- (2,2);
  \node[font=\large] at (1,-0.5) {A};
  \node[font=\large] at (4,-0.5) {B};
\end{tikzpicture}
\end{center}
Ben's move does not work. How many squares should really move from Part B to
Part A so that the two parts are halves?}

\problem[4cm]{Kai cut this square into four pieces and called them fourths.
Piece A covers $2$ squares, Piece B covers $6$ squares, Piece C covers $6$
squares, and Piece D covers $2$ squares.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[step=1, gray!55, thin] (0,0) grid (4,4);
  \draw[thick] (0,0) rectangle (4,4);
  \draw[line width=1.6pt] (2,0) -- (2,4);
  \draw[line width=1.6pt] (0,3) -- (2,3);
  \draw[line width=1.6pt] (2,1) -- (4,1);
  \node[font=\large] at (1,3.5) {A};
  \node[font=\large] at (1,1.5) {B};
  \node[font=\large] at (3,2.5) {C};
  \node[font=\large] at (3,0.5) {D};
\end{tikzpicture}
\end{center}
Write the four counts in order, smallest first.}

\problem[4cm]{\textbf{Find and fix.} Ivy's rectangle has $20$ little squares.
She wants to cut it into $4$ equal fourths. Ivy folded the rectangle one time,
counted one of the two parts, and wrote $10$ squares for a fourth.
\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[step=1, gray!55, thin] (0,0) grid (5,4);
  \draw[thick] (0,0) rectangle (5,4);
\end{tikzpicture}
\end{center}
Ivy's number is too big. How many squares should each fourth really have?}
