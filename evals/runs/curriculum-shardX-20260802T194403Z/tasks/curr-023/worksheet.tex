\wstitleblock{Add-To Stories Within 10}{Kindergarten--Grade 1}{}

\noindent\small\textbf{How to work:} Each story starts with some things, then
\textbf{more} come. Draw or count to find how many there are \textbf{in all}.
Write a number sentence like $3 + 2 = 5$, then write the total on the answer
line.\normalsize

\problem[3cm]{\large There are \textbf{3} red counters on the ten-frame.
Then \textbf{2} more counters are put on the frame.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
\rule{0pt}{0.85cm}$\bullet$ & $\bullet$ & $\bullet$ & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\rule{0pt}{0.85cm}\makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\end{tabular}
\end{center}
How many counters are on the frame \textbf{in all}?}

\problem[3cm]{\large Maya has \textbf{4} stickers. Her teacher gives her
\textbf{3} more stickers. How many stickers does Maya have \textbf{in all}?}

\problem[3.5cm]{\large \textbf{4} birds are sitting on a branch. Then \textbf{5}
more birds land on the same branch.\\[6pt]
Finish the number sentence, then write how many birds are on the branch in all.
\begin{center}\Large
$4 \;+\; 5 \;=\;$ \makebox[2cm]{\hrulefill}
\end{center}}

\problem[3cm]{\large A ten-frame has \textbf{6} counters on it.
Then \textbf{2} more counters are put on the frame.
\begin{center}\large
\begin{tabular}{|c|c|c|c|c|}\hline
\rule{0pt}{0.85cm}$\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ & $\bullet$ \\\hline
\rule{0pt}{0.85cm}$\bullet$ & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} & \makebox[0.8cm]{} \\\hline
\end{tabular}
\end{center}
How many counters are on the frame \textbf{in all}?}

\problem[3cm]{\large \textbf{2} fish are in the tank. Then \textbf{6} more fish
are put in the tank.\\[4pt]
Start at the \textbf{bigger} number and count on. How many fish are in the tank
\textbf{in all}?}

\problem[3.5cm]{\large Sam builds a tower with \textbf{5} blocks. Then he adds
\textbf{3} more blocks to the same tower.\\[6pt]
Write the number sentence for this story, then write the total.
\begin{center}\Large
\makebox[1.4cm]{\hrulefill} $+$ \makebox[1.4cm]{\hrulefill} $=$ \makebox[1.4cm]{\hrulefill}
\end{center}}

\problem[3.5cm]{\large Ana has \textbf{3} crayons. Her mom gives her \textbf{2}
crayons. Then her brother gives her \textbf{4} crayons.\\[4pt]
How many crayons does Ana have \textbf{in all}?}

\problem[4cm]{\large Ben has \textbf{4} toy cars. His sister gives him
\textbf{3} more cars, and his dad gives him \textbf{2} more cars.\\[4pt]
Ben says, ``Now I have \textbf{8} cars.''\\[4pt]
Count carefully. Is Ben right? Write the correct total on the answer line.}
