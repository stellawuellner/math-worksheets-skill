\wstitleblock{Diagnosing Transformation-Order and Inverse-Domain Errors}{Precalculus}{}

\begin{center}
\small\textit{Telling inside changes from outside changes, ordering transformations correctly, and choosing the right inverse branch}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Each problem has two jobs: compute the correct value, and say in words what went wrong. Both are graded.
  \item A diagnosis names the rule that was broken. ``They got a different number'' is a check, not a diagnosis.
  \item Show the substitution you used --- one well-chosen input settles most of these arguments.
\end{itemize}
\vspace{0.15cm}

\problem[0pt]{Let $f(x) = x^{2}$. A student claims that $g(x) = f(x) + 5$ and $h(x) = f(x + 5)$ have the same graph.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Compute $g(2)$ \ansblank\ and $h(2)$ \ansblank
  \par\vspace*{2.6cm}
  \item[(b)] Name the error in the student's claim, saying what each $5$ actually does.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{To graph $y = 2(x - 3)^{2} + 1$ starting from $y = x^{2}$, a student shifted up $1$ first, then stretched vertically by $2$, then shifted right $3$, and reported a vertex height of $2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Evaluate the printed rule $y = 2(x - 3)^{2} + 1$ at $x = 3$ to find the true vertex height.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Name the error in the student's order of operations.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{A student claims that the inverse of $f(x) = x^{2}$ is $f^{-1}(x) = \sqrt{x}$, with no restriction placed on $f$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Compute $f(-4)$ \ansblank\ and then the square root of that result \ansblank
  \par\vspace*{2.6cm}
  \item[(b)] Say what restriction $f$ must carry before $\sqrt{x}$ can be its inverse.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Maya was asked for the rule that stretches $y = |x|$ vertically by $3$ and then shifts it down $4$. She wrote $g(x) = 3\bigl(|x| - 4\bigr)$ and reported $g(2) = -6$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Write the correct rule and use it to find the correct $g(2)$.
  \par\vspace*{2.8cm}\ansline
  \item[(b)] Name her error, saying which number was multiplied when it should not have been.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Let $f(x) = \sqrt{x - 2}$. A student wrote $f^{-1}(x) = x^{2} + 2$ and said its domain is all real numbers.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Using the student's rule, compute $f\bigl(f^{-1}(-3)\bigr)$.
  \par\vspace*{2.8cm}\ansline
  \item[(b)] Explain why that result shows the domain claim is wrong, and give the restriction $f^{-1}$ needs.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Let $f(x) = \sqrt{x}$. A student says the graph of $y = f(2x)$ is the graph of $f$ stretched horizontally by a factor of $2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Solve $f(2x) = 4$ \ansblank\ and solve $f(x) = 4$ \ansblank
  \par\vspace*{2.8cm}
  \item[(b)] Say whether the factor $2$ stretches or compresses the graph horizontally, using your two answers as the evidence.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{A student described $y = -(x + 1)^{2} + 5$ as: reflect $y = x^{2}$ in the $x$-axis, shift left $1$, shift up $5$ --- but carried out the reflection last, and reported a vertex height of $-5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Evaluate the printed rule at $x = -1$ to find the true vertex height.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Name the error, saying which step must come before which.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Let $f(x) = x^{2} - 4x + 7$ with the domain restricted to $x \leq 2$. A student wrote $f^{-1}(x) = 2 + \sqrt{x - 3}$ and so reported $f^{-1}(12) = 5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Complete the square, then find the correct value of $f^{-1}(12)$.
  \par\vspace*{3.4cm}\ansline
  \item[(b)] Explain which branch of the parabola the restriction keeps, and why that fixes the sign of the square root.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}
