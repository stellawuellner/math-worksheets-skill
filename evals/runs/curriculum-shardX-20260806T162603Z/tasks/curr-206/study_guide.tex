\sstitleblock{Product and Quotient Laws for Exponents}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The expanded-form model}

\begin{formulabox}
\textbf{What an exponent means:} $x^4 = x \cdot x \cdot x \cdot x$ --- the exponent is a
\emph{count of factors}, nothing more.
\par\smallskip
\textbf{Why the laws work:} multiplying two powers writes the two groups of factors end to
end, so the counts add. Dividing cancels matching factors top and bottom, so the counts
subtract. Expand whenever a law feels uncertain --- the model never lies.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Use expanded form on $2^2 \cdot 2^3$.
\step{The claim is about how many factors there are, so I write them all out and count rather than reaching for a rule.}
\step{Two factors followed by three more makes a run of five factors of the base, and multiplying that run out gives the value.}

Exponent $= \ans{5}$, \quad value $= \ans{32}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Use expanded form on $3^2 \cdot 3^2$: give the exponent and the
value.\\[2pt]
\rotatebox{180}{\footnotesize check: exponent $\ans{4}$, value $\ans{81}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The product law}

\begin{formulabox}
\textbf{Law:} \quad $x^m \cdot x^n = x^{m+n}$ \quad --- same base, add the exponents.
\par\smallskip
It holds for negative exponents too: adding a negative count simply cancels factors.
When numbers ride in front, multiply those separately: $(3x^2)(4x^5) = 12x^7$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $x^6 \cdot x^2$.
\step{Same base and a multiplication, so the product law applies and the exponents add.}
\step{Six factors followed by two more is a single run of eight factors of the same base.}

$\Rightarrow\quad x^6 \cdot x^2 = \ans{x^8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $w^3 \cdot w^7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{w^{10}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} multiplying powers adds the exponents; it never multiplies them.
Test any rule you are unsure of on small numbers you can check by hand.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The quotient law}

\begin{formulabox}
\textbf{Law:} \quad $\dfrac{x^m}{x^n} = x^{m-n}$ \quad --- same base, subtract the bottom
exponent from the top one.
\par\smallskip
If the bottom exponent is larger, the result has a negative exponent; rewrite it as one
over the positive power. Numbers in front divide separately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{k^9}{k^4}$.
\step{Same base and a division, so the quotient law applies and the exponents subtract.}
\step{Four factors on the bottom cancel four of the nine on top, leaving the difference as the count.}

$\Rightarrow\quad \dfrac{k^9}{k^4} = \ans{k^5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{p^5}{p^8}$, answer with a positive exponent.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{p^3}}$}
\end{tryitbox}
