% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Square Units Measure Flat Space}

\begin{formulabox}
\textbf{Rule:} \quad length $\to$ cm \quad\textbullet\quad flat surface
(area) $\to$ cm$^2$ \quad\textbullet\quad filled space (volume) $\to$
cm$^3$.\par
Area of a rectangle: $A = \ell \times w$ --- it counts how many unit
\emph{squares} cover the surface, so its unit is always \emph{squared}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the area of a $9$ cm by $5$ cm card.
\step{A card's face is a flat surface, so we count unit squares --- multiply
the two side lengths and report square centimeters.}
\step{$A = 9 \times 5 = 45$ unit squares.}

$\Rightarrow\quad A = \ans{45\ \text{cm}^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the area of an $8$ cm by $3$ cm bookmark.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24\ \text{cm}^2}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Cubic Units Fill Solid Space}

\begin{formulabox}
\textbf{Rule:} \quad volume of a box: $V = \ell \times w \times h$ --- it
counts how many unit \emph{cubes} fill the space, so its unit is always
\emph{cubed}. Three lengths multiplied $\to$ the unit is used three times.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the volume of a $4 \times 3 \times 2$ cm box.
\step{The box holds stacked layers of unit cubes, so multiply all three
dimensions and report cubic centimeters.}
\step{Base layer: $4 \times 3 = 12$ cubes; $2$ layers make $24$ cubes.}

$\Rightarrow\quad V = \ans{24\ \text{cm}^3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the volume of a $5 \times 2 \times 2$ cm box.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20\ \text{cm}^3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Converting Squared Units}

\begin{formulabox}
\textbf{Rule:} \quad squared units use the conversion factor \emph{twice}.
Each meter is $100$ cm, so
\[ 1\ \text{m}^2 = 100 \times 100\ \text{cm}^2 . \]
(Cubed units would use it three times.)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert $3$ m$^2$ to square centimeters.
\step{A square meter is a $100$ cm by $100$ cm square, so the factor $100$
appears twice --- once for each dimension.}
\step{$3 \times 100 \times 100 = 3 \times 10000 = 30000$.}

$\Rightarrow\quad 3\ \text{m}^2 = \ans{30000\ \text{cm}^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $5$ m$^2$ to square centimeters.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{50000\ \text{cm}^2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a correct \emph{number} with the wrong \emph{unit}
is a wrong answer. Before writing the unit, ask: am I measuring a length, a
flat surface, or a filled space?
\end{watchoutbox}
