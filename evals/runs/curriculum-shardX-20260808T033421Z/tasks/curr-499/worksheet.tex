\wstitleblock{Find and Fix: Inconclusive Tests and Forgotten Endpoints}{AP Calculus BC}{}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every problem shows student work that is wrong in one specific way. Repair the computation, then name the rule that was broken.
  \item Two errors dominate this topic: treating an \emph{inconclusive} test as if it had reached a conclusion, and stopping at the open interval a ratio test produces without testing the endpoints.
  \item Part (a) always wants a value; put it on the Answer line. The remaining parts want a sentence or two, written in the work space.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{A student writes: ``For $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n}$, the terms satisfy $\lim_{n\to\infty}\frac{1}{n} = 0$, so by the nth-term test the series converges.''
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Confirm the limit the student computed.
  \item[(b)] Name the error, and state what the nth-term test does and does not prove.
\end{itemize}}

\problem[6.5cm]{For $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n\,2^{n}}$ a student computes the radius correctly and then writes: ``so the interval of convergence is $(-2, 2)$.''
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Confirm the radius of convergence, using the coefficient ratio $\dfrac{2(n+1)}{n}$.
  \item[(b)] Write the exact sum of the series at the left endpoint $x = -2$.
  \item[(c)] Name the error and give the correct interval of convergence.
\end{itemize}}

\problem[6.5cm]{For $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n^{2}}$ a student estimates the sum with the first $8$ terms and reports the error bound as $\dfrac{1}{64} \approx 0.0156$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Give the correct error bound as an exact fraction.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[6.5cm]{A student computes $\displaystyle\sum_{n=2}^{\infty}\left(\frac{1}{4}\right)^{n}$ as $\dfrac{1}{1 - \frac{1}{4}} = \dfrac{4}{3} \approx 1.333$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Give the correct sum as an exact fraction.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[7cm]{A student writes: ``For $\displaystyle\sum_{n=1}^{\infty}\frac{n^{2}}{n^{3}+1}$ the ratio test gives a limit of $1$, so the series converges.''
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Confirm the ratio-test limit. \ansline
  \item[(b)] Find $\displaystyle\lim_{n\to\infty}\frac{n^{2}/(n^{3}+1)}{1/n}$, the limit comparison against the harmonic series. \ansline
  \item[(c)] Name the error and give the correct conclusion.
\end{itemize}}

\problem[6.5cm]{For $\displaystyle\sum_{n=1}^{\infty}\frac{(2x-6)^{n}}{n}$ a student reports a radius of $1$ and an interval of $(2, 4)$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Rewrite $(2x-6)^{n}$ as $2^{n}(x-3)^{n}$, so the coefficients are $c_{n} = 2^{n}/n$ with ratio $\dfrac{n+1}{2n}$, and give the correct radius.
  \item[(b)] Name both faults in the student's answer and give the correct interval of convergence.
\end{itemize}}

\problem[7cm]{A student writes: ``$\displaystyle\int_{1}^{\infty} x^{-3/2}\,dx = \left[-\tfrac{2}{5}x^{-5/2}\right]_{1}^{\infty} = \tfrac{2}{5}$, so by the integral test $\sum_{n=1}^{\infty} n^{-3/2}$ converges and its sum is $\tfrac{2}{5} = 0.4$.''
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Evaluate the integral correctly.
  \item[(b)] Name \emph{both} errors in the student's sentence.
\end{itemize}}

\problem[7cm]{Here is a student's complete answer for $\displaystyle\sum_{n=1}^{\infty}\frac{(x+2)^{n}}{n\,3^{n}}$: ``The ratio test gives $\frac{|x+2|}{3} < 1$, so $-5 < x < 1$ and the interval of convergence is $(-5, 1)$.''
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Substitute the left endpoint $x = -5$ and write the exact sum of the resulting series. \ansline
  \item[(b)] Substitute the right endpoint $x = 1$ and give $\displaystyle\lim_{n\to\infty}$ of the resulting terms. \ansline
  \item[(c)] Name the error and give the correct interval of convergence.
\end{itemize}}
