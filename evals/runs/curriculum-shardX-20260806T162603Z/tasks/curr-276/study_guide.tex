\sstitleblock{Telling Exponential Change from Linear Change}

\skillheading{Skill 1 --- The differences-and-ratios test}

\begin{formulabox}
\textbf{Test a table over equal input steps:}\par
\emph{Linear} --- successive outputs have a constant \textbf{difference} $d$; rule $y = a + dx$.\par
\emph{Exponential} --- successive outputs have a constant \textbf{ratio} $r$; rule $y = a \cdot r^{\,x}$.\par
\textbf{Do both tests, not one.} A table can look ``steadily growing'' either way; only the constant one tells you which.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test the table $2,\ 6,\ 18,\ 54$ at inputs $0, 1, 2, 3$.
\step{The inputs step by one each time, so both tests are fair --- run the ratio test and the difference test and see which comes out constant.}
\step{Ratios: $6 \div 2 = 3$, and $18 \div 6 = 3$ as well, so the ratio holds.}
\step{Differences: $6 - 2 = 4$, then $18 - 6 = 12$ --- not constant.}

ratio $= \ans{3}$, differences $\ans{4}$ then $\ans{12}$: exponential, $y = 2 \cdot 3^{\,x}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Test $7,\ 11,\ 15,\ 19$. Find the first difference and the first ratio.\\[2pt]
\rotatebox{180}{\footnotesize check: difference $\ans{4}$ (constant, so linear), ratio $\ans{\tfrac{11}{7}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Evaluating each kind of rule}

\begin{formulabox}
\textbf{Linear:} $a + dx$ --- multiply the step by the input, then add the start.\par
\textbf{Exponential:} $a \cdot r^{\,x}$ --- raise the multiplier to the input \emph{first}, then multiply by the start. The exponent applies to $r$ alone, never to $a$.\par
\textbf{Percentages:} a rise of $p\%$ makes $r = 1 + \frac{p}{100}$; a fall of $p\%$ makes $r = 1 - \frac{p}{100}$. So $15\%$ growth is $r = 1.15$ and $12\%$ decay is $r = 0.88$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $L(x) = 12 + 5x$ and $E(x) = 12 \cdot 1.5^{\,x}$. Find both at $x = 4$.
\step{One rule adds a fixed amount each step and the other multiplies by a fixed factor, so the arithmetic order differs: multiply-then-add against power-then-multiply.}
\step{$L(4) = 12 + 5(4) = 12 + 20$.}
\step{$E(4)$: first $1.5^4 = 5.0625$, then $12 \times 5.0625$.}

$L(4) = \ans{32}$ \quad and \quad $E(4) = \ans{60.75}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $L(x) = 20 + 7x$ and $E(x) = 20 \cdot 1.2^{\,x}$. Find both at $x = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $L(3) = \ans{41}$, $E(3) = \ans{34.56}$}
\end{tryitbox}

\skillheading{Skill 3 --- Which model wins in the long run}

\begin{formulabox}
\textbf{The headline fact:} a growing exponential eventually passes \emph{any} linear rule, however big the linear step and however small the start. Early values prove nothing --- the linear rule is often ahead for the first few inputs.\par
\textbf{Why:} the linear rule adds the same amount every step; the exponential rule adds a percentage of a total that keeps getting larger, so its own step grows.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 30 + 10x$ and $g(x) = 3 \cdot 2^{\,x}$. Compare them at $x = 4$ and at $x = 6$.
\step{One comparison cannot show a crossover, so test an early input and a later one and watch the order swap.}
\step{At $x = 4$: $30 + 40$ against $3 \cdot 16$ --- the linear rule leads.}
\step{At $x = 6$: $30 + 60$ against $3 \cdot 64$ --- the exponential rule has taken over.}

$f(4) = \ans{70}$, $g(4) = \ans{48}$; \quad $f(6) = \ans{90}$, $g(6) = \ans{192}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 50 + 5x$ and $g(x) = 2 \cdot 3^{\,x}$. Compare them at $x = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $f(5) = \ans{75}$, $g(5) = \ans{486}$}
\end{tryitbox}
