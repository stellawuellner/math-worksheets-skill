\begin{center}
{\small\textit{slopes from coordinates, parallel and perpendicular tests, and classifying figures}}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Slope from two points: $m = \dfrac{y_2-y_1}{x_2-x_1}$ --- rise over run,
        with both differences taken in the \emph{same} order.
  \item Two lines are \textbf{parallel} exactly when their slopes are equal.
        They are \textbf{perpendicular} exactly when the product of their slopes
        is $-1$, that is, when each slope is the negative reciprocal of the
        other.
  \item Give every slope as an integer or a fraction in lowest terms, not as a
        decimal. A vertical line has no slope --- write "undefined".
\end{itemize}

\problem[4.4cm]{Find the slope of the line through $(-4,5)$ and $(6,-1)$.}

\problem[4.4cm]{Find the slope of the line through $(5,-2)$ and $(5,9)$.}

\problem{Line $j$ passes through $(-2,1)$ and $(2,9)$. Line $k$ passes through
$(0,-5)$ and $(3,1)$.
\par\vspace{0.1cm}
(a) Find the slope of line $j$. \ansline
\par\vspace*{2.2cm}
(b) Find the slope of line $k$. \ansline
\par\vspace*{2.2cm}
(c) State whether the two lines are parallel, perpendicular or neither, and
justify your answer using the slopes.
\par\vspace*{2.2cm}}

\problem{Line $p$ passes through $(-1,4)$ and $(3,6)$. Line $q$ passes through

\par\vspace{0.1cm}

\par\vspace*{2.2cm}
(b) Find the slope of line $q$. \ansline
\par\vspace*{2.2cm}
(c) State whether the two lines are parallel, perpendicular or neither, and
justify your answer using the slopes.
\par\vspace*{2.2cm}}

\problem[4.8cm]{Find the slope of any line parallel to the line through
$(-6,2)$ and $(2,-4)$.}

\problem[5.2cm]{Find the slope of any line perpendicular to the line through

\problem{Quadrilateral $ABCD$ has vertices $A(-1,0)$, $B(5,2)$, $C(7,8)$ and
$D(1,6)$.
\par\vspace{0.1cm}
(a) Find the slope of $\overline{AB}$. \ansline
\par\vspace*{2.0cm}
(b) Find the slope of $\overline{DC}$. \ansline
\par\vspace*{2.0cm}
(c) Find the slope of $\overline{AD}$. \ansline
\par\vspace*{2.0cm}
(d) Explain why $ABCD$ is a parallelogram. (You will need the slope of
$\overline{BC}$ as well.)
\par\vspace*{2.4cm}}

\problem{Triangle $PQR$ has vertices $P(-2,1)$, $Q(4,3)$ and $R(6,-3)$.
\par\vspace{0.1cm}
(a) Find the slope of $\overline{PQ}$. \ansline
\par\vspace*{2.2cm}
(b) Find the slope of $\overline{QR}$. \ansline
\par\vspace*{2.2cm}
(c) Is the triangle right-angled at $Q$? Justify your answer using the slopes.
\par\vspace*{2.4cm}}

\problem{Quadrilateral $EFGH$ has vertices $E(-3,-1)$, $F(1,-3)$, $G(5,5)$ and
$H(1,7)$.
\par\vspace{0.1cm}
(a) Find the slope of $\overline{EF}$. \ansline
\par\vspace*{2.0cm}
(b) Find the slope of $\overline{HG}$. \ansline
\par\vspace*{2.0cm}
(c) Find the slope of $\overline{FG}$. \ansline
\par\vspace*{2.0cm}
(d) Using both the parallel and the perpendicular tests, explain what kind of

well.)
