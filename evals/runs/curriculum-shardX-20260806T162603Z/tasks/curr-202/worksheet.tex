\textit{Perfect square roots, perfect cube roots, and roots inside larger expressions}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Every root on this sheet comes out exact --- no calculator, no decimals.
  \item $\sqrt{\phantom{x}}$ asks: what number times itself gives the number inside?
        $\sqrt[3]{\phantom{x}}$ asks: what number used as a factor three times gives it?
  \item Write the multiplication that proves your root, then put the value on the answer line.
\end{itemize}
