\sstitleblock{Limits at the Edges}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading a Limit From One Side}

\begin{formulabox}
\textbf{Rule:} $x \to k^-$ means $x < k$ and $x \to k^+$ means $x > k$. On one
side, an absolute value or a piecewise rule becomes a single ordinary formula ---
rewrite it that way \emph{first}, then take the limit. If the two one-sided
values disagree, the two-sided limit does not exist.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to 1^-} \frac{|x-1|}{x-1}$
\step{Approaching from the left means $x < 1$, so $x - 1$ is negative and the absolute value must be rewritten as $-(x-1)$ before anything else.}
\step{For $x < 1$ the expression is $\dfrac{-(x-1)}{x-1} = -1$, a constant, so the limit is that constant.}

$\Rightarrow\quad \lim_{x \to 1^-} \dfrac{|x-1|}{x-1} = $ $\ans{-1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to -2^+} \frac{|x+2|}{x+2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Hole or Vertical Asymptote?}

\begin{formulabox}
\textbf{Rule:} at a zero of the denominator, factor and cancel. If the factor
cancels, the point is a \textbf{hole} and the limit is a finite number ---
substitute into the reduced form. If it does not cancel, the point is a
\textbf{vertical asymptote}: the limit is $+\infty$ or $-\infty$, and the sign
comes from the sign of the surviving factor on the side you approach from.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle F(x) = \frac{x^2-9}{x^2-2x-3}$ find
$\displaystyle\lim_{x \to 3} F(x)$ and $\displaystyle\lim_{x \to -1^+} F(x)$.
\step{Both $3$ and $-1$ zero the denominator, so factor first --- only the factor that cancels gives a hole.}
\step{$F(x) = \dfrac{(x-3)(x+3)}{(x-3)(x+1)} = \dfrac{x+3}{x+1}$. At $x=3$ that is $\dfrac{6}{4}$; at $x \to -1^+$ the top tends to $2$ and $x+1$ is a small positive.}

$\Rightarrow\quad$ hole: $\ans{\tfrac{3}{2}}$ \quad asymptote: $\ans{+\infty}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $G(x) = \dfrac{x^2-1}{x^2-3x+2}$ find
$\displaystyle\lim_{x\to1} G(x)$ and $\displaystyle\lim_{x\to2^+} G(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2}$ \ and \ $\ans{+\infty}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- End Behavior as x Grows Large}

\begin{formulabox}
\textbf{Rule:} divide every term by the highest power of $x$ in the
\emph{denominator}, then use $\frac{c}{x^n} \to 0$. Shortcut for polynomials:
top degree $<$ bottom degree gives $0$; equal degrees give the ratio of leading
coefficients; top degree $>$ bottom degree gives $\pm\infty$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to \infty} \frac{4x^3-x}{2x^3+5}$
\step{Top and bottom are both cubic, so divide through by $x^3$ and let every leftover fraction die.}
\step{$\dfrac{4 - 1/x^2}{2 + 5/x^3} \to \dfrac{4-0}{2+0} = 2$, which is just the ratio of the leading coefficients.}

$\Rightarrow\quad \lim_{x \to \infty} \dfrac{4x^3-x}{2x^3+5} = $ $\ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to \infty} \frac{7x^3+2x}{2-x^3}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{x^2} = |x|$, not $x$. Heading to $-\infty$ that
minus sign is real, so an expression like $\sqrt{9x^2+4}\,/\,(3x-1)$ tends to
$1$ at the right end and $-1$ at the left end.
\end{watchoutbox}
