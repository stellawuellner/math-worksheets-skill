% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Inscribed and central angles}

\begin{formulabox}
\textbf{Inscribed angle:} an inscribed angle is \emph{half} the central angle
standing on the same arc: $m\angle ACB = \tfrac{1}{2}\, m\angle AOB$.\par
\textbf{Angle in a semicircle:} an angle inscribed on a diameter is a right angle.\par
\textbf{Cyclic quadrilateral:} opposite angles are supplementary (they add to $180^\circ$).\par
\textbf{Arc measure} equals the measure of its central angle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} In circle $O$, central $\angle AOB = 100^\circ$ and $C$ lies on the
major arc. Find $m\angle ACB$.
\step{The angle I want is inscribed and the angle I know is central, and both stand on the same arc $AB$ --- that is exactly the inscribed-angle theorem.}
\step{Inscribed $=$ half of central: $m\angle ACB = \tfrac{1}{2}(100^\circ)$.}
$\Rightarrow\quad \ans{50}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} In circle $O$, central $\angle AOB = 130^\circ$ and $C$ lies on the
major arc. Find $m\angle ACB$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{65}$ degrees}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Arc length and sector area}

\begin{formulabox}
A central angle of $\theta$ degrees cuts off $\dfrac{\theta}{360}$ of the circle,
so it takes that same fraction of the circumference and of the area:\par
\textbf{Arc length} $= \dfrac{\theta}{360}\cdot 2\pi r$ \qquad
\textbf{Sector area} $= \dfrac{\theta}{360}\cdot \pi r^2$\par
\textbf{Segment area} $=$ sector area $-$ area of the triangle formed by the two radii and the chord.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A circle has radius 6 cm and a central angle of $120^\circ$. Find
the arc length to the nearest tenth of a centimetre.
\step{I want a length along the curve, not an area, so I take the angle's fraction of the circumference.}
\step{$\dfrac{120}{360}\cdot 2\pi(6) = \tfrac{1}{3}(12\pi) = 4\pi \approx 12.566$ cm.}
$\Rightarrow\quad \ans{12.6}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A circle has radius 9 cm and a central angle of $60^\circ$. Find the
arc length to the nearest tenth of a centimetre.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9.4}$ cm}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Chords, tangents and secants}

\begin{formulabox}
\textbf{Two chords meeting inside} at $P$: \ $AP \cdot PB = CP \cdot PD$ --- the two
products of the pieces are equal.\par
\textbf{Tangent and secant from an outside point} $P$: \ $PT^2 = PA \cdot PB$, where
$PB$ is the \emph{whole} secant, not just the far piece.\par
\textbf{Two secants from an outside point:} \ $PA \cdot PB = PC \cdot PD$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Chords $AB$ and $CD$ meet inside a circle at $P$ with $AP = 4$,
$PB = 9$, $CP = 3$ and $PD = x$. Find $x$.
\step{The intersection point is inside the circle and I know three of the four pieces, so the intersecting-chords theorem gives an equation in $x$.}
\step{$3x = 4 \cdot 9 = 36$, so $x = 36 \div 3$.}
$\Rightarrow\quad \ans{x = 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Chords $AB$ and $CD$ meet inside a circle at $P$ with $AP = 5$,
$PB = 8$, $CP = 4$ and $PD = x$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 10}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- The equation of a circle}

\begin{formulabox}
\textbf{Centre-radius form:} $(x-h)^2 + (y-k)^2 = r^2$ has centre $(h,k)$ and radius $r$
--- read the centre with the signs \emph{flipped}.\par
\textbf{To complete the square:} halve the coefficient of $x$, square it, and add it to
both sides; do the same for $y$. Then the radius is the square root of the right-hand side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the centre and radius of $x^2 + y^2 - 4x + 6y - 12 = 0$.
\step{The equation has $x$ and $y$ terms but no squared binomials, so I complete the square in each variable to reach centre-radius form.}
\step{Half of $4$ is $2$ and half of $6$ is $3$: $(x-2)^2 + (y+3)^2 = 12 + 4 + 9 = 25$, and $r = \sqrt{25}$.}
$\Rightarrow\quad$ centre $\ans{(2,-3)}$, radius $\ans{r = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the centre and radius of $x^2 + y^2 - 8x + 2y - 8 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: centre $\ans{(4,-1)}$, radius $\ans{r = 5}$}
\end{tryitbox}
