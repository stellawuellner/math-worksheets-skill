\sstitleblock{Explaining Legal Moves in Equations and Inequalities}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Naming the property behind a move}

\begin{formulabox}
\textbf{Think of a balance.} Whatever you do to one pan you must do to the other, or it tips.
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=2pt]
  \item Add the same amount to both sides --- \emph{addition property of equality}.
  \item Subtract the same amount from both sides --- \emph{subtraction property}.
  \item Multiply or divide both sides by the same non-zero number --- \emph{multiplication} or
        \emph{division property}.
  \item Rewrite ONE side into an equal expression (distribute, collect like terms) --- that is the
        distributive property or plain arithmetic, not a property of equality. It is still legal.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2x - 9 = 5$ and name each move.
\step{The variable is tied up by a subtraction and then a multiplication, so I undo them in the reverse order, saying what each move is as I go.}
\step{Add $9$ to both sides (addition property): $2x = 14$.}
\step{Divide both sides by $2$ (division property): $x = 7$.}

$\Rightarrow\quad \ans{x = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $3x - 4 = 11$, naming each move as you go.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Solving when the variable is on both sides}

\begin{formulabox}
\textbf{Order of work:} clear brackets and fractions first (one-side rewrites), then collect the
variable on one side with a both-sides addition or subtraction, then finish with a both-sides
division. Every line you write must be true whenever the line above it is true --- that is the whole
test of a legal move.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $5(x + 2) = 3x + 16$.
\step{There is a bracket on one side and an $x$ on both, so I clear the bracket first and only then move the variable across.}
\step{Distribute on the left (one-side rewrite): $5x + 10 = 3x + 16$.}
\step{Subtract $3x$ from both sides: $2x + 10 = 16$. Subtract $10$ from both sides: $2x = 6$.}
\step{Divide both sides by $2$.}

$\Rightarrow\quad \ans{x = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $4(x - 1) = 2x + 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The one move that flips an inequality}

\begin{formulabox}
\textbf{Rule:} adding or subtracting the same amount on both sides never changes the direction of an
inequality. Multiplying or dividing both sides by a \emph{negative} number does: the symbol must be
reversed, because a negative multiplier swaps which side is larger. Multiplying by a positive number
changes nothing. Always check one number from your answer set at the end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $-2x + 1 \ge 9$.
\step{The variable has a negative coefficient, so I expect exactly one flip --- at the division step, not before it.}
\step{Subtract $1$ from both sides (no flip): $-2x \ge 8$.}
\step{Divide both sides by $-2$, and reverse the symbol: $x \le -4$. Check $x = -5$: $10 + 1 = 11 \ge 9$, true.}

$\Rightarrow\quad \ans{x \le -4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-4x \le 24$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \ge -6}$}
\end{tryitbox}
