\sstitleblock{Reading and Comparing Multi-Digit Whole Numbers}

\skillheading{1. What a digit is worth (place-value-meaning)}

\begin{formulabox}
\textbf{Rule:} a digit's \emph{value} is the digit times the value of its place.\\[3pt]
\begin{tabular}{@{}cccccc@{}}
hundred thousands & ten thousands & thousands & hundreds & tens & ones\\
$100{,}000$ & $10{,}000$ & $1{,}000$ & $100$ & $10$ & $1$
\end{tabular}\\[3pt]
Each place is \textbf{ten times} the place on its right, so moving one column left multiplies
the worth of a digit by 10.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In $82{,}417$, what is the value of the digit 2?
\step{The question asks what the 2 is \emph{worth}, so first find which column it is in, then
multiply by that column's value.}
\step{Counting from the right: 7 ones, 1 ten, 4 hundreds, 2 thousands. So the 2 is
$2 \times 1{,}000 = \ans{2{,}000}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In $59{,}304$, what is the value of the digit 9?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9{,}000}$}
\end{tryitbox}

\skillheading{2. Expanded form and word form (expanded-and-word-form)}

\begin{formulabox}
\textbf{Three ways to write one number:}\\[3pt]
\begin{tabular}{@{}ll@{}}
standard form & $43{,}206$\\
expanded form & $40{,}000 + 3{,}000 + 200 + 6$\\
word form & forty-three thousand, two hundred six
\end{tabular}\\[3pt]
\textbf{Zeros hold places.} A place with nothing in it still needs a 0 in standard form, or
every digit to its right slides into the wrong column.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $40{,}000 + 3{,}000 + 200 + 6$ in standard form.
\step{Each addend names one place, so write the digits into their columns and fill empty
columns with 0.}
\step{4 ten thousands, 3 thousands, 2 hundreds, \textbf{no} tens (write 0), 6 ones:
$\ans{43{,}206}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $700{,}000 + 20{,}000 + 50 + 1$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{720{,}051}$}
\end{tryitbox}

\skillheading{3. Comparing and ordering (compare-order)}

\begin{formulabox}
\textbf{Rule:} more digits means a bigger whole number. With the \emph{same} number of
digits, line the numbers up and compare from the \textbf{left}; the first place where the
digits differ decides it --- everything to the right of it is irrelevant.\\[3pt]
$<$ means \emph{is less than}, \quad $>$ means \emph{is greater than}. The open side always
faces the larger number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $26{,}458$ and $26{,}485$.
\step{Both have five digits, so a digit count decides nothing --- scan from the left for the
first column where they disagree.}
\step{Ten thousands $2 = 2$, thousands $6 = 6$, hundreds $4 = 4$, tens $5$ against $8$:
first difference, and $5 < 8$, so $26{,}458 \; \ans{<} \; 26{,}485$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$ or $>$: \quad $71{,}309$ \underline{\hspace{0.9cm}} $71{,}093$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\skillheading{4. Rounding to estimate (round-estimate)}

\begin{formulabox}
\textbf{Rule:} to round to a place, look at the digit \emph{one place to the right}.\\[3pt]
0--4: keep the rounding digit (round down). \quad 5--9: add one to it (round up).\\
Every digit to the right of the rounding place becomes 0.\\[3pt]
To \textbf{estimate a sum}, round each number first, then add the rounded numbers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $63{,}472$ to the nearest thousand.
\step{Rounding to thousands, so the thousands digit is the one that may change and the
hundreds digit is what decides.}
\step{Thousands digit 3, hundreds digit 4. Since $4 < 5$, keep the 3 and zero the rest:
$\ans{63{,}000}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $28{,}715$ to the nearest thousand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{29{,}000}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Rounding is not comparing. $34{,}650$ and $34{,}509$ both round to
$35{,}000$, but they are not equal --- compare the \emph{original} numbers when a problem
asks which is greater.
\end{watchoutbox}
