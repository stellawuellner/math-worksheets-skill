\sstitleblock{Multi-Step Linear Equations with Rational Coefficients}

\skillheading{1. Clear the fractions with the LCD}

\begin{formulabox}
\textbf{Rule.} Multiplying both sides of an equation by the same nonzero
number gives an \emph{equivalent} equation --- one with exactly the same
solution. So pick the \textbf{LCD} of every denominator and multiply
\emph{each term on both sides} by it. Every fraction disappears at once.\\[3pt]
\textbf{Order of moves:} clear fractions $\rightarrow$ distribute
$\rightarrow$ combine like terms $\rightarrow$ undo $+$ and $-$ $\rightarrow$
undo $\times$ and $\div$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{2} + \dfrac{x}{5} = 7$.
\step{Two different denominators, so clear them both at once instead of
combining fractions: the LCD of $2$ and $5$ is $10$.}
\step{Multiply every term by $10$: \ $5x + 2x = 70$, so $7x = 70$.}

$\Rightarrow\quad \ans{x = 10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{3} + \dfrac{x}{4} = 14$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 24}$}
\end{tryitbox}

\skillheading{2. Variable on both sides}

\begin{formulabox}
\textbf{Rule.} Clear the fractions first --- multiply \emph{all four} terms by
the LCD, the ones without fractions too. Then move every variable term to one
side and every constant to the other, and finish with one division.\\[3pt]
A term such as $\dfrac{x+2}{3}$ has the whole numerator over $3$, so
multiplying by the LCD keeps it together: $6 \cdot \dfrac{x+2}{3} = 2(x+2)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{3}{5}x - 2 = \dfrac{x}{10} + 1$.
\step{Variables sit on both sides and both sides carry fractions, so clear
with the LCD of $5$ and $10$, which is $10$, before collecting anything.}
\step{Multiply every term by $10$: \ $6x - 20 = x + 10$. Subtract $x$, add
$20$: \ $5x = 30$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{5}{6}x - 1 = \dfrac{x}{3} + 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 10}$}
\end{tryitbox}

\skillheading{3. Inequalities with rational coefficients}

\begin{formulabox}
\textbf{Rule.} Solve an inequality exactly like an equation, with one extra
rule: multiplying or dividing both sides by a \textbf{negative} number
\emph{reverses} the symbol ($>$ becomes $<$, $\ge$ becomes $\le$). Adding or
subtracting never reverses it, and multiplying by a positive number never
does either.\\[3pt]
\textbf{Check:} test one number from your answer and one number outside it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $-\dfrac{3}{4}x + 2 \ge 8$.
\step{The coefficient of $x$ is negative, so plan on turning the symbol at the
final step --- everything before that step is ordinary equation work.}
\step{Subtract $2$: \ $-\dfrac{3}{4}x \ge 6$. Multiply by $-\dfrac{4}{3}$ and
\emph{reverse} the symbol.}

$\Rightarrow\quad \ans{x \le -8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-\dfrac{2}{5}x - 1 > 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x < -10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} when you multiply by the LCD, multiply \emph{every}
term, including the plain numbers --- forgetting the constant is the most
common slip. And reverse the inequality symbol only when the number you
multiply or divide by is negative, never because the answer itself is
negative.
\end{watchoutbox}
