\wstitleblock{Direction, Form, and Strength in Scatter Plots}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Reading a scatter plot, then naming its direction, its form, and how strong it is}
\end{center}

\vspace{0.05cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Direction:} positive (both go up together), negative (one goes up as the other goes down), or none.
  \item \textbf{Form:} linear if a single straight line would follow the points, non-linear if the pattern bends.
  \item \textbf{Strength:} strong if the points hug the pattern closely, weak if they are spread widely around it.
  \item Answer the number question first, then write the description in full sentences.
\end{itemize}
