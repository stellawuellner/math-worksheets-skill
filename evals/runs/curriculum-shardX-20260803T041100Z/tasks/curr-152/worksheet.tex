\wstitleblock{Building Equivalent Ratio Tables}{Grade 6--7 Mathematics}{}

\noindent\small
Completing ratio tables, finding the scale factor between rows, adding a
unit-rate column, and deciding whether two tables are equivalent. Work with
the same habit every time:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A ratio table stays equivalent when you multiply (or divide)
        \emph{both} entries of a column by the same number.
  \item Find that number first --- ``what times the old entry gives the new
        one?'' --- then apply it to the partner entry.
  \item Adding the same amount to both entries does \textbf{not} keep a ratio
        equivalent. $2:5$ and $4:7$ are different ratios.
  \item The unit rate is the column whose first entry is $1$; it is the fastest
        way to compare two tables.
\end{itemize}
\normalsize

\vspace{0.15cm}

\problem[4cm]{A muffin recipe uses $2$ cups of flour for every $5$ cups of
milk. Complete the ratio table.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|c|c|}
\hline
\textbf{Flour (cups)} & $2$ & $4$ & $6$ & $10$ \\ \hline
\textbf{Milk (cups)}  & $5$ & \hspace{1.6cm} & \hspace{1.6cm} & \hspace{1.6cm} \\ \hline
\end{tabular}
\end{center}}

\problem[4.5cm]{One row of a ratio table reads $3$ red beads to $8$ white
beads. Another row of the same table has $21$ red beads. By what number was
the first row multiplied, and how many white beads are in that row?
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Red beads}   & $3$ & $21$ \\ \hline
\textbf{White beads} & $8$ & \hspace{1.6cm} \\ \hline
\end{tabular}
\end{center}}

\problem[4.5cm]{A school keeps a ratio of $5$ teachers for every $90$
students. Complete the table by scaling from the \emph{student} row.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Teachers} & $5$  & \hspace{1.6cm} & \hspace{1.6cm} \\ \hline
\textbf{Students} & $90$ & $180$ & $270$ \\ \hline
\end{tabular}
\end{center}}

\problem[4.5cm]{A bottling machine fills $6$ bottles every $4$ seconds. Add
the unit-rate column to the table: how many bottles does it fill in $1$
second?
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Bottles} & $6$ & \hspace{1.6cm} \\ \hline
\textbf{Seconds} & $4$ & $1$ \\ \hline
\end{tabular}
\end{center}}

\problem[5cm]{Ana's table pairs $4$ green tiles with $6$ blue tiles. Ben's
table pairs $14$ green tiles with $21$ blue tiles. Are the two tables
equivalent? Write the two ratios as fractions in lowest terms and join them
with $<$, $>$, or $=$.}

\problem[5.5cm]{A ratio table scales $8$ minutes of video to $20$ megabytes of
storage. The next column shows $12$ minutes. Find the scale factor from the
first column to the second, then find the missing storage entry.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Minutes}   & $8$  & $12$ \\ \hline
\textbf{Megabytes} & $20$ & \hspace{1.6cm} \\ \hline
\end{tabular}
\end{center}}

\problem[5.5cm]{A paint mix uses $3$ parts blue to $5$ parts yellow. Fill in
both missing entries. Notice that one column scales \emph{up} from the first
column and the other does not use a whole-number scale factor.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Blue (mL)}   & $3$ & \hspace{1.6cm} & $15$ \\ \hline
\textbf{Yellow (mL)} & $5$ & $20$ & \hspace{1.6cm} \\ \hline
\end{tabular}
\end{center}}

\problem[5cm]{A train covers $150$ miles in $4$ hours at a steady speed. Add
the unit-rate column to the table, then state the speed.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
\textbf{Miles} & $150$ & \hspace{1.6cm} \\ \hline
\textbf{Hours} & $4$   & $1$ \\ \hline
\end{tabular}
\end{center}
\par
\answerline{mph}}

\problem[5.5cm]{Two lemonade recipes are given as ratio tables of scoops of
mix to cups of water. Recipe P uses $3$ scoops to $5$ cups; recipe Q uses $5$
scoops to $8$ cups. Which recipe tastes stronger, that is, which has more mix
per cup of water? Compare the two ratios with $<$, $>$, or $=$ and say which
recipe wins.}

\problem[6cm]{A ratio table is built from $5:3$. One column of the table reads
$x$ to $42$. Write and solve an equation to find $x$.}

\problem[6cm]{A ratio table starts with the column $9$ to $12$. A later column
of the same table has $84$ in the second row. Let $k$ be the scale factor.
Write and solve an equation for $k$, then find the missing first-row entry.}

\problem[6.5cm]{\textbf{Synthesis challenge.} A paint mix keeps blue to yellow
in the ratio $4:7$. One batch uses $66$ ounces of paint in total. Build a ratio
table with a ``total'' row, find how many equal parts the $66$ ounces splits
into, and state how many ounces of blue and how many ounces of yellow the
batch contains.}
