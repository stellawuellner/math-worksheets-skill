\wstitleblock{Converting Between Rectangular and Polar Coordinates}{Precalculus \quad Polar and Parametric Representations}{}

\begin{center}
\textit{Polar to rectangular, rectangular to polar, and choosing the angle that
matches the quadrant}
\end{center}

\noindent\textbf{Directions.} Every problem has a polar grid beside it. Plot the
point first --- the picture tells you which quadrant the answer must land in,
and that is the step a calculator cannot do for you.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Polar $\rightarrow$ rectangular:} $x = r\cos\theta$, \;
$y = r\sin\theta$. Substitute and simplify; leave exact radicals unless a
decimal is asked for.
\item \textbf{Rectangular $\rightarrow$ polar:} $r = \sqrt{x^2 + y^2}$ (a
distance, so never negative), and $\tan\theta = \dfrac{y}{x}$.
\item \textbf{The angle needs a decision.} $\tan\theta = \dfrac{y}{x}$ has two
solutions on $[0^\circ, 360^\circ)$, half a turn apart, and the arctangent key
only ever returns one of them. Plot the point, see the quadrant, and pick the
solution that lands there.
\end{itemize}
