\sstitleblock{Solving One-Step Equations with Rational Coefficients}

\begin{formulabox}
\textbf{One step in, one step out.} An equation $cx = k$ says ``$x$ was
multiplied by $c$, and the result is $k$.'' Undo that one multiplication on both
sides.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item $c$ is a \textbf{fraction} $\to$ multiply both sides by its
\textbf{reciprocal} (flip it). The reciprocal of $\frac{a}{b}$ is $\frac{b}{a}$,
and a negative coefficient keeps its sign when flipped.
\item $c$ is a \textbf{decimal} $\to$ divide both sides by $c$.
\item If $c$ is less than 1, the solution is \emph{bigger} than $k$; if $c$ is
bigger than 1, the solution is \emph{smaller} than $k$. Predict this first --- it
catches most mistakes before they happen.
\end{itemize}
\end{formulabox}

\skillheading{1. Unit-fraction coefficient: multiply by the denominator}

\begin{formulabox}
$\dfrac{x}{b} = k$ means the bar $x$ was cut into $b$ equal parts and one part is
$k$. The reciprocal of $\frac{1}{b}$ is just $b$, so multiply both sides by $b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $\dfrac{x}{5} = 8$.
\step{One fifth of the bar is 8, so the whole bar is five of those parts ---
multiply both sides by 5 rather than guessing.}
\step{$5 \cdot \dfrac{x}{5} = 5 \cdot 8$, and the 5s on the left cancel.}
\[ x = \ans{40} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $\dfrac{x}{6} = 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{42}$}
\end{tryitbox}

\skillheading{2. Decimal coefficient: divide both sides}

\begin{formulabox}
$0.6x = k$ was built by multiplying, so divide. If the division is awkward,
multiply the numerator and denominator by 10 or 100 to clear the decimal:
$\dfrac{9}{0.6} = \dfrac{90}{6}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $0.6x = 9$.
\step{The coefficient is less than 1, so predict a solution larger than 9, then
divide to find it.}
\step{$x = \dfrac{9}{0.6} = \dfrac{90}{6}$, which is 15 --- larger than 9, as
predicted.}
\[ x = \ans{15} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $0.25x = 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{3. Non-unit fraction: multiply by the reciprocal}

\begin{formulabox}
$\dfrac{a}{b}x = k$ means $a$ of the $b$ equal parts are worth $k$. Multiplying
by $\dfrac{b}{a}$ turns the coefficient into 1 in a single step --- it is the
same as dividing by $a$ and then multiplying by $b$, done at once.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $\dfrac{3}{4}x = 15$.
\step{Three of four parts are 15, so flip the coefficient and multiply both
sides by the reciprocal.}
\step{$\dfrac{4}{3}\cdot\dfrac{3}{4}x = \dfrac{4}{3}\cdot 15 = \dfrac{60}{3}$.}
\[ x = \ans{20} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $\dfrac{2}{5}x = 14$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

\skillheading{4. Check by substituting, not by re-solving}

\begin{formulabox}
To check a solution, put the value back into the \emph{original} equation and
evaluate the left side. If it equals the right side, the value is a solution. Do
not use the reciprocal here --- that is the solving move, not the checking move.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Is $x = 20$ a solution of $\dfrac{3}{4}x = 15$?
\step{Checking means substituting into the equation exactly as written, so use
the coefficient $\frac{3}{4}$, not its reciprocal.}
\step{$\dfrac{3}{4}\cdot 20 = \dfrac{60}{4}$, which is 15 --- matching the right
side, so 20 is a solution.}
\[ \dfrac{3}{4}\cdot 20 = \ans{15} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Check $x = 35$ in $\dfrac{2}{5}x = 14$ by substituting.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} multiplying by the coefficient instead of its reciprocal
is the single most common error --- $\frac{2}{3}x = 18$ does \emph{not} give
$x = \frac{2}{3}\cdot 18$. And keep the sign with the reciprocal: the undo of
$-\frac{3}{5}$ is $-\frac{5}{3}$.
\end{watchoutbox}
