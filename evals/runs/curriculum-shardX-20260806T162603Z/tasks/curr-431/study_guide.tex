\sstitleblock{Converting Between Rectangular and Polar Coordinates}

\skillheading{Skill 1 --- Polar to rectangular}

\begin{formulabox}
$x = r\cos\theta$, \qquad $y = r\sin\theta$.\par
Read them off the right triangle with legs $x$ and $y$ and hypotenuse $r$: the leg beside $\theta$ is $r\cos\theta$, the leg opposite it is $r\sin\theta$.\par
This direction is always safe --- no quadrant decision is needed, because the signs of $\cos\theta$ and $\sin\theta$ already carry it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert the polar point $(10, 30^\circ)$ to rectangular form.
\step{Going from polar to rectangular means substituting into the two formulas --- nothing to decide, so this direction is the easy one.}
\step{$x = 10\cos 30^\circ = 10\left(\dfrac{\sqrt{3}}{2}\right)$, \quad $y = 10\sin 30^\circ = 10\left(\dfrac{1}{2}\right)$}
$\Rightarrow\quad (x, y) = (\ans{5\sqrt{3}},\ \ans{5})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $(12, 150^\circ)$ to rectangular form.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-6\sqrt{3}}$, $y = \ans{6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Rectangular to polar}

\begin{formulabox}
$r = \sqrt{x^2 + y^2}$ \quad (always taken positive here), \qquad $\tan\theta = \dfrac{y}{x}$.\par
The $r$ formula is the Pythagorean theorem on the same triangle. Sketch the point first: knowing its quadrant before you touch the calculator is what makes the angle step reliable.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert the rectangular point $(4, 4\sqrt{3})$ to polar form.
\step{Both coordinates are positive, so the point is in quadrant I and the calculator's inverse tangent will need no correction.}
\step{$r = \sqrt{16 + 48} = \sqrt{64}$, \quad $\tan\theta = \dfrac{4\sqrt{3}}{4} = \sqrt{3}$}
$\Rightarrow\quad r = \ans{8}$, \ $\theta = \ans{60}^\circ$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $(5, 5)$ to polar form with $r > 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{5\sqrt{2}}$, $\theta = \ans{45}^\circ$}
\end{tryitbox}

\skillheading{Skill 3 --- Choosing the angle by quadrant}

\begin{formulabox}
$\tan^{-1}$ only ever returns an angle between $-90^\circ$ and $90^\circ$, so it can only land in quadrant I or IV. Tangent repeats every $180^\circ$, so opposite directions share the same ratio.\par
\textbf{Rule:} if $x < 0$, add $180^\circ$ to the calculator's answer. If $x > 0$ and $y < 0$, add $360^\circ$ to bring it into $[0^\circ, 360^\circ)$. If $x = 0$ the formula does not apply --- read the angle off the axis.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Convert $(-3, -3\sqrt{3})$ to polar form with $0^\circ \le \theta < 360^\circ$.
\step{Both coordinates are negative, so the point is in quadrant III, and the inverse tangent cannot report an angle there --- expect to correct it.}
\step{$r = \sqrt{9 + 27} = 6$; $\tan^{-1}\!\left(\dfrac{-3\sqrt{3}}{-3}\right) = 60^\circ$, and $x < 0$, so add $180^\circ$.}
$\Rightarrow\quad r = \ans{6}$, \ $\theta = \ans{240}^\circ$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Convert $(-4, 4)$ to polar form with $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{4\sqrt{2}}$, $\theta = \ans{135}^\circ$}
\end{tryitbox}

\skillheading{Skill 4 --- One point, many names}

\begin{formulabox}
A polar name is not unique. The same point is $(r, \theta)$, $(r, \theta + 2\pi)$, and also $(-r, \theta + \pi)$.\par
\textbf{Why the last one works:} a negative first coordinate means face the given direction and step \emph{backwards}, which lands you where facing the opposite direction and stepping forwards would.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rename the polar point $\left(4, \dfrac{\pi}{6}\right)$ so that its first coordinate is $-4$.
\step{Flipping the sign of the distance reverses the direction, so the angle must turn by a straight angle to compensate.}
\step{$\theta' = \dfrac{\pi}{6} + \pi = \dfrac{\pi}{6} + \dfrac{6\pi}{6}$}
$\Rightarrow\quad \theta' = \ans{\tfrac{7\pi}{6}}$, so the point is also $\left(-4, \dfrac{7\pi}{6}\right)$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rename $\left(3, \dfrac{3\pi}{4}\right)$ so that its first coordinate is $-3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta' = \ans{\tfrac{7\pi}{4}}$}
\end{tryitbox}
