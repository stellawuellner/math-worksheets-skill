\wstitleblock{Evaluating Functions from Rules, Tables, and Graphs}{Grade 8 / Pre-Algebra}{}

\noindent\small
Evaluating a function from its rule, from a table of values, and from a graph,
then comparing two function values at the same input.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=3pt]
  \item $f(3)$ means ``the output of $f$ when the input is $3$'' --- it is never
        $f$ times $3$.
  \item Substitute inside parentheses first, then simplify.
  \item Problems 5, 7, 8, 11, and 12 all read from \textbf{Graph A} below. Every
        other problem carries its own rule or its own table.
\end{itemize}
\normalsize

\problem[4cm]{The rule for a function is $f(x) = 3x + 4$. \par Find $f(2)$.}

\problem[4.5cm]{The rule for a function is $f(x) = x^2 - 5$. \par Find $f(-3)$.
Be careful with the sign before you square.}

\problem[5cm]{The table gives every value of the function $h$ that was recorded.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}\hline
$x$ & 1 & 2 & 3 & 4 & 5 \\\hline
$h(x)$ & 7 & 10 & 13 & 16 & 19 \\\hline
\end{tabular}
\end{center}
\par Find $h(4)$.}

\problem[5cm]{The rule for a function is $g(x) = \dfrac{x + 7}{2}$. \par Find $g(5)$.}

\problem[5cm]{Use \textbf{Graph A}. Find $f(3)$ by locating the input on the
horizontal axis and reading the height of the solid line above it.}

\problem[5.5cm]{The table gives every value of the function $p$ that was recorded.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}\hline
$x$ & 0 & 1 & 2 & 3 & 4 \\\hline
$p(x)$ & 8 & 5 & 4 & 5 & 8 \\\hline
\end{tabular}
\end{center}
\par For which input is the output of $p$ the smallest? Write that input value.}

\problem[5cm]{Use \textbf{Graph A}. Find $g(1)$ --- read the dashed line, not the
solid one.}

\problem[6cm]{Use \textbf{Graph A}, whose rules are $f(x) = 2x - 1$ and
$g(x) = -x + 5$. \par Evaluate both functions at the input $1$, then write $<$ or
$>$ so the sentence is true:
\par \hspace*{1cm}$f(1)$ \ \underline{\hspace{1.2cm}} \ $g(1)$}

\problem[5cm]{The rule for a function is $f(x) = 5 - 2x$. \par Find $f(-4)$.}

\problem[6cm]{A club charges a joining fee plus a monthly fee. The table shows the
total cost $C$, in dollars, after $n$ months.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|}\hline
$n$ & 1 & 2 & 3 & 4 & 5 & 6 \\\hline
$C(n)$ & 12 & 19 & 26 & 33 & 40 & 47 \\\hline
\end{tabular}
\end{center}
\par How many dollars more is $C(6)$ than $C(2)$?}

\problem[6cm]{Use \textbf{Graph A}, whose rules are $f(x) = 2x - 1$ and
$g(x) = -x + 5$. \par Evaluate both functions at the input $4$, then write $<$ or
$>$ so the sentence is true:
\par \hspace*{1cm}$f(4)$ \ \underline{\hspace{1.2cm}} \ $g(4)$}

\problem[7cm]{Look again at \textbf{Graph A}, where $f(x) = 2x - 1$ and
$g(x) = -x + 5$. The two lines cross above the input $2$.
\par Explain, in two or three sentences, why both rules give the same output at
that input, and what the crossing point tells you about comparing $f$ and $g$ on
either side of it. Support your explanation with values from the graph or the rules.
\noansline}
