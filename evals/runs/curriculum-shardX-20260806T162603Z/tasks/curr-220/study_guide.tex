\sstitleblock{Rates of Change: Table, Graph, Equation}

\skillheading{Skill 1 --- Rate of change from a table}

\begin{formulabox}
\textbf{Rule:} pick two rows. Rate $=$ (change in the output row) $\div$
(change in the input row), taking both differences in the same direction.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the rate of change for the table
$x = 0, 3$ with $y = 4, 19$.
\step{Two rows are enough for a linear table, so use the pair given and put
the output change on top.}
\step{Output change $19 - 4 = 15$; input change $3 - 0 = 3$; the rate is
$15/3$.}

Each unit of $x$ adds this much $y$. \quad $\ans{m = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $x = 0, 4$ with $y = 50, 26$. Find the
rate of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = -6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Rate of change from an equation}

\begin{formulabox}
\textbf{Rule:} in $y = mx + b$ the rate of change is $m$, the number
multiplying the input. The constant $b$ is the starting value and never
changes the rate.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the rate of change of $y = 6x + 20$.
\step{Nothing needs solving --- the rate is what one extra unit of $x$ adds,
so compare the value at $x$ with the value at $x + 1$.}
\step{$\bigl(6(x + 1) + 20\bigr) - (6x + 20) = 6$, which is exactly the number
multiplying $x$.}

The 20 cancels, as the rule promises. \quad $\ans{m = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the rate of change of $y = 100 - 8x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = -8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Rate of change from a graph}

\begin{formulabox}
\textbf{Rule:} choose two points where the line crosses grid corners, then
count the rise and the run between them. Rising left to right means a positive
rate; falling means a negative one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the rate of change of the line drawn below.
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=4, ymin=0, ymax=12,
  xtick={0,1,2,3,4}, ytick={0,3,6,9,12},
  grid=both, grid style={line width=0.25pt, draw=gray!55},
  major grid style={line width=0.4pt, draw=gray!70},
  tick label style={font=\tiny}, xlabel={$x$}, ylabel={$y$},
  width=6.4cm, height=4.4cm, enlargelimits=false]
  \addplot[thick, blue, domain=0:3, samples=2] {3*x + 3};
  \fill[blue] (axis cs:0,3) circle (1.7pt);
  \fill[blue] (axis cs:2,9) circle (1.7pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{Two marked lattice points are enough, so read them off and count rise
over run rather than guessing the steepness.}
\step{From $(0, 3)$ to $(2, 9)$ the rise is $9 - 3 = 6$ and the run is
$2 - 0 = 2$, giving $6/2$.}

Rising left to right, so positive. \quad $\ans{m = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line passes through the marked points $(1, 2)$ and
$(4, 8)$. Find its rate of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = 2}$}
\end{tryitbox}

\skillheading{Skill 4 --- Comparing two rates}

\begin{formulabox}
\textbf{Rule:} convert both relationships to a rate in the \emph{same units}
first, then compare the two numbers. On a graph the larger rate is the steeper
line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table gives 5 dollars per hour; a job pays
$y = 6x$ dollars for $x$ hours. Which rate is greater?
\step{The two relationships arrive in different forms, so pull a rate out of
each before comparing anything.}
\step{Table rate 5 dollars per hour; equation rate is the number multiplying
$x$, so 6 dollars per hour.}

Table rate first, then equation rate. \quad $\ans{5 < 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph rises 9 metres per second; another runner
follows $d = 4t$. Compare the graph's rate with the runner's.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9 > 4}$}
\end{tryitbox}
