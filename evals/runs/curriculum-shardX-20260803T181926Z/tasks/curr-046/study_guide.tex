\sstitleblock{Shape Names --- Turning Does Not Change Them}

\skillheading{Skill 1 --- Count the sides to name it}

\begin{formulabox}
\textbf{Count the straight sides. The number tells you the name.}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $3$ straight sides $=$ \textbf{triangle}
  \item $4$ straight sides, all the same length $=$ \textbf{square}
  \item $4$ straight sides, two long and two short $=$ \textbf{rectangle}
  \item $6$ straight sides $=$ \textbf{hexagon}
  \item no straight sides, one curved edge $=$ \textbf{circle}
\end{itemize}
It does not matter which way the shape is turned. Touch each side as you count.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is this shape called?
\begin{center}
\begin{tikzpicture}[thick, scale=0.8]
  \draw (0,1.2) -- (1.6,1.2) -- (0.8,0) -- cycle;
\end{tikzpicture}
\end{center}
\step{It is upside down, so do not go by how it looks --- count the straight sides
instead, because that is what the name is made of.}
\step{Touch the top edge, then the right edge, then the left edge: that is one, two,
three straight sides.}

$\Rightarrow\quad \ans{3}$ sides, so it is a \textbf{triangle}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many straight sides does this shape have?
\begin{center}
\begin{tikzpicture}[thick, scale=0.75, rotate=17]
  \draw (0:0.8) -- (60:0.8) -- (120:0.8) -- (180:0.8) -- (240:0.8) -- (300:0.8) -- cycle;
\end{tikzpicture}
\end{center}
\rotatebox{180}{\footnotesize check: $\ans{6}$ sides, a hexagon}
\end{tryitbox}

\skillheading{Skill 2 --- Turning changes nothing}

\begin{formulabox}
\textbf{Turn a shape and nothing countable changes.} Same number of sides, same number of
corners, same name.\par
\smallskip
A square balanced on its corner is still a \textbf{square}. ``Diamond'' says which way it
is turned; it does not say what the shape is.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Count the corners on each. Is one bigger than the other?
\begin{center}
\begin{tikzpicture}[thick, scale=0.7]
  \draw (0,0) rectangle (1.4,1.4);
  \draw (3.7,0) -- (4.7,1.0) -- (3.7,2.0) -- (2.7,1.0) -- cycle;
\end{tikzpicture}
\end{center}
\step{These are the same square, and one of them has been turned --- so instead of
guessing, count the corners on each one and compare the two numbers.}
\step{Flat one: one, two, three, four corners. Turned one: one, two, three, four corners.}

$\Rightarrow\quad \ans{4 = 4}$, so both are squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Count the corners on each triangle and compare them.
\begin{center}
\begin{tikzpicture}[thick, scale=0.7]
  \draw (0,0) -- (1.4,0) -- (0.7,1.2) -- cycle;
  \draw (3.0,0.2) -- (4.4,0.9) -- (3.2,1.7) -- cycle;
\end{tikzpicture}
\end{center}
\rotatebox{180}{\footnotesize check: $\ans{3 = 3}$, both are triangles}
\end{tryitbox}

\skillheading{Skill 3 --- Sort, count, then compare}

\begin{formulabox}
\textbf{Sort first, then count, then compare.} Name every shape in the group before you
count, so a turned shape lands in the right pile.\par
\smallskip
Then put the two numbers side by side: $>$ means the first is bigger, $<$ means the second
is bigger, $=$ means they are the same.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which group has more shapes?
\begin{center}
\begin{tikzpicture}[thick, scale=0.55]
  \draw (0,0) circle (0.5); \draw (1.4,0) circle (0.5); \draw (2.8,0) circle (0.5);
  \draw[dashed, gray] (3.9,-0.9) -- (3.9,0.9);
  \draw (4.7,-0.5) -- (5.7,-0.5) -- (5.2,0.5) -- cycle;
  \draw (6.2,0.5) -- (7.2,0.5) -- (6.7,-0.5) -- cycle;
  \draw (7.7,-0.5) -- (8.7,0.0) -- (7.9,0.6) -- cycle;
  \draw (9.2,-0.5) -- (10.2,-0.5) -- (9.7,0.5) -- cycle;
  \draw (10.7,0.5) -- (11.7,0.5) -- (11.2,-0.5) -- cycle;
\end{tikzpicture}
\end{center}
\step{Count each group on its own before comparing anything, because the eye is easily
fooled when one group's shapes are bigger than the other's.}
\step{Left group: three circles. Right group: five triangles, some of them turned.}

$\Rightarrow\quad \ans{3 < 5}$, so the right group has more
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which group has more shapes?
\begin{center}
\begin{tikzpicture}[thick, scale=0.55]
  \draw (0,-0.5) rectangle (1.0,0.5);
  \draw (1.9,0) -- (2.6,0.7) -- (3.3,0) -- (2.6,-0.7) -- cycle;
  \draw (4.0,-0.5) rectangle (5.0,0.5);
  \draw (5.9,0) -- (6.6,0.7) -- (7.3,0) -- (6.6,-0.7) -- cycle;
  \draw[dashed, gray] (7.9,-0.9) -- (7.9,0.9);
  \draw (9.3,0) -- ++(60:0.6) -- ++(120:0.6) -- ++(180:0.6) -- ++(240:0.6) -- ++(300:0.6) -- cycle;
  \draw (11.6,0) -- ++(60:0.6) -- ++(120:0.6) -- ++(180:0.6) -- ++(240:0.6) -- ++(300:0.6) -- cycle;
\end{tikzpicture}
\end{center}
\rotatebox{180}{\footnotesize check: $\ans{4 > 2}$, the left group has more}
\end{tryitbox}
