\wstitleblock{Related Rates and Optimization: Diagnosing the Setup}{AP Calculus AB / BC --- Applications of Derivatives}{}

\noindent\textbf{Setup discipline --- the four errors this sheet targets:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Substituting too early.} Instantaneous values go in \emph{after} differentiating. A number written into the relation first is treated as a constant and its rate disappears.
  \item \textbf{Leaving a second variable in.} Use the geometry to eliminate variables until one remains, then differentiate.
  \item \textbf{Inverting a ratio.} Solve the differentiated equation symbolically for the rate you want before any arithmetic.
  \item \textbf{Writing the wrong constraint.} An optimization constraint must describe the actual situation, not the familiar formula that resembles it.
\end{itemize}

\noindent For every find-and-fix item, quote the specific line of the setup that
is wrong, then redo the problem from that line on.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4cm]{\textbf{Setup step.} A circle's area is $A = \pi r^2$. Differentiate
with respect to $r$ to get the factor that will multiply $\dfrac{dr}{dt}$ when the
chain rule is applied.}

\problem[4.5cm]{\textbf{Setup step --- eliminate first.} A cone's volume is
$V = \tfrac{1}{3}\pi r^2 h$, and for this cone the height is always twice the
radius. Substitute the constraint so $V$ depends on $r$ alone, then differentiate
with respect to $r$.}

\problem[5cm]{A spherical balloon is inflated at a constant $100$ cubic
centimetres per second. Find $\dfrac{dr}{dt}$ at the instant when the radius is
$5$ centimetres. Round to three decimal places.
\par\vspace{2pt}
Write the differentiated relation before you substitute anything.}

\problem[5.5cm]{A $10$-foot ladder leans against a vertical wall. Its base slides
away from the wall at $2$ feet per second. Find the rate at which the top of the
ladder slides down at the instant the base is $6$ feet from the wall. Round to
three decimal places.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A conical tank has radius
$3$ metres at the top and depth $6$ metres, and water drains out at
$12$ cubic metres per minute. A student writes
$\dfrac{dV}{dt} = \tfrac{1}{3}\pi r^2 \dfrac{dh}{dt}$, treating the radius as the
fixed value $3$.
\par\vspace{2pt}
Name what that step assumes about the water's surface, then find
$\dfrac{dh}{dt}$ correctly when the depth is $4$ metres. Round to three decimal
places.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A rectangular pen is built
against a barn wall, so fencing is needed on only three sides. There are
$200$ feet of fencing. A student writes the constraint $2x + 2y = 200$.
\par\vspace{2pt}
Correct the constraint, write the area as a function of $x$ alone, differentiate
it, and find the $x$ that maximises the area.}

\problem[6cm]{\textbf{Open-box optimisation.} An open-top box is made from a
square sheet measuring $24$ inches by $24$ inches by cutting a square of side $x$
from each corner and folding up the sides.
\par\vspace{2pt}
Write the volume as a function of $x$, differentiate it, and find every critical
value. State which critical value must be rejected and why.}

\problem[5cm]{\textbf{Explain.} A classmate substitutes the given instantaneous
radius into the volume formula before differentiating, gets a constant, and
concludes the rate is zero. Write a short paragraph explaining what went wrong,
distinguishing a quantity that varies from one that is genuinely constant, and
stating exactly when substituting a given value is legal. \noansline}
