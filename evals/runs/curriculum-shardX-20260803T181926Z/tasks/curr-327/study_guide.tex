\sstitleblock{Using Slopes to Prove Lines Parallel or Perpendicular}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Parallel Lines Have Equal Slopes}

\begin{formulabox}
\textbf{Slope formula:} \ for points $(x_1, y_1)$ and $(x_2, y_2)$, \ $m = \dfrac{y_2 - y_1}{x_2 - x_1}$ --- the rise over the run.\par
\textbf{Parallel test:} two distinct non-vertical lines are parallel \emph{exactly when} $m_1 = m_2$. Equal slopes mean equal direction, so the lines never meet. Two vertical lines are parallel as well, even though neither has a slope.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Line $\ell$ passes through $A(1, 1)$ and $B(5, 3)$. What is the slope of every line parallel to $\ell$?

\step{Parallel lines carry the same slope, so the whole question reduces to finding the slope of $\ell$ itself.}
\step{$m = \dfrac{3 - 1}{5 - 1} = \dfrac{2}{4}$, reduced to lowest terms.}

$\Rightarrow\quad \ans{m = \tfrac{1}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Line $n$ passes through $C(-2, -1)$ and $D(2, 5)$. What is the slope of every line parallel to $n$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = \tfrac{3}{2}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Slopes of Perpendicular Lines}

\begin{formulabox}
\textbf{Perpendicular test:} two non-vertical lines are perpendicular \emph{exactly when} $m_1 m_2 = -1$; equivalently, each slope is the \textbf{opposite reciprocal} of the other --- flip the fraction and change the sign.\par
\textbf{The exception:} a vertical line has no slope at all, so the product rule cannot be used. A line perpendicular to a vertical line is horizontal, and its slope is $0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the slope of a line perpendicular to the line through $E(-1, 2)$ and $F(3, 4)$.

\step{Perpendicular means opposite reciprocal, so find the slope of $\overline{EF}$ first and then transform it.}
\step{$m_{EF} = \dfrac{4 - 2}{3 - (-1)} = \dfrac{2}{4}$, which is $\ans{m_{EF} = \tfrac{1}{2}}$; flipping it gives $2$ and changing the sign gives the perpendicular slope. Check: $\left(\tfrac{1}{2}\right)(-2) = -1$ $\checkmark$}

$\Rightarrow\quad \ans{m = -2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope of a line perpendicular to the line through $G(0, 5)$ and $H(2, -1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m_{GH} = -3}$, so the perpendicular slope is $\ans{m = \tfrac{1}{3}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the opposite reciprocal changes \emph{both} things. The perpendicular partner of $\tfrac{2}{3}$ is $-\tfrac{3}{2}$ --- not $\tfrac{3}{2}$ (sign forgotten) and not $-\tfrac{2}{3}$ (fraction not flipped).
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Coordinate Proofs Using Slopes}

\begin{formulabox}
\textbf{How a coordinate proof works.} \ Turn the claim into slopes, compute, compare, then say what the comparison proves:
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item both pairs of opposite sides parallel $\Rightarrow$ parallelogram
  \item parallelogram with one pair of adjacent sides perpendicular $\Rightarrow$ rectangle
  \item midpoints come from $\left(\frac{x_1 + x_2}{2}, \frac{y_1 + y_2}{2}\right)$; lengths from the distance formula
\end{itemize}
Finish with a sentence naming the theorem you used --- the arithmetic alone is not a proof.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A triangle has third side $\overline{TU}$ from $T(0, 0)$ to $U(8, 4)$, and the midpoints of the other two sides are $M(1, 3)$ and $N(5, 5)$. Show $\overline{MN} \parallel \overline{TU}$.

\step{Parallel is a statement about direction, so compare the slope of the midsegment with the slope of the third side.}
\step{$m_{MN} = \dfrac{5 - 3}{5 - 1} = \dfrac{2}{4}$, and $m_{TU} = \dfrac{4 - 0}{8 - 0} = \dfrac{4}{8}$ --- the same number, so the segments are parallel.}

$\Rightarrow\quad \ans{m = \tfrac{1}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The midpoints of two sides of a triangle are $(2, 4)$ and $(7, 6)$. Find the slope of the midsegment joining them.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = \tfrac{2}{5}}$}
\end{tryitbox}
