\wstitleblock{Factoring Special Forms: Squares}{Algebra 1 \quad Polynomials}{}

\begin{center}
\textit{Factoring a difference of two squares, factoring a perfect-square
trinomial, and taking out a common factor before the special form appears}
\end{center}

\noindent\textbf{The two patterns.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Difference of squares:} $A^2 - B^2 = (A - B)(A + B)$. Two terms,
a minus sign between them, and both terms perfect squares.
\item \textbf{Perfect-square trinomial:} $A^2 \pm 2AB + B^2 = (A \pm B)^2$.
Three terms; the first and last are squares and the middle is twice the product
of their roots.
\item \textbf{Always look for a common factor first.} $2x^2 - 50$ is not a
difference of squares as written --- it becomes one after the $2$ comes out.
\end{itemize}

\vspace{0.1cm}

\problem[5cm]{A square patio has side length $x$ metres. A square hot tub with
side length $4$ metres is removed from one corner, so the paved area left over
is $x^2 - 16$ square metres.

Write the paved area in factored form.}

\problem[5cm]{Factor completely: \quad $2x^2 - 50$

\smallskip
\small Two terms and a minus sign, but $2x^2$ is not a perfect square. Deal
with that first.\normalsize}

\problem[5cm]{A square mosaic has an area of $x^2 + 14x + 49$ square
centimetres, and its side length is a whole expression in $x$.

Factor the area to find the side length of the mosaic.}

\problem[5.5cm]{A student factored $x^2 - 9$ and wrote $(x - 3)^2$.

Expand $(x - 3)^2$ to show what that answer really equals, explain in one
sentence which pattern the student confused, then write the correct
factorization of $x^2 - 9$.}

\problem[5.5cm]{Factor completely: \quad $18x^2 - 32$

\smallskip
\small Neither term is a perfect square as written.\normalsize}

\problem[5.5cm]{Factor completely: \quad $x^4 - 81$

\smallskip
\small One of the factors is itself a difference of squares. Keep factoring
until no factor can be split over the integers.\normalsize}
