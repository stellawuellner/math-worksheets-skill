\wstitleblock{Conditional and Independent-Event Probability}{Precalculus / Statistics}{}

\begin{center}
\small Two-way tables, conditional probability, testing independence, and the multiplication rule
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Give every probability as an exact fraction in lowest terms unless the problem asks for a decimal.
  \item Problems 1--4 and 10 use the survey table below. All counts are numbers of students.
  \item Write the probability statement you are computing (for example $P(A \mid B)$) before you compute it.
\end{itemize}

\vspace{0.2cm}
\begin{center}
\textbf{Survey of 120 students at one school (counts of students)}\\[4pt]
\begin{tabular}{lccc}
\toprule
 & \textbf{Plays a sport} & \textbf{No sport} & \textbf{Total} \\
\midrule
Plays an instrument      & 18 & 22 & 40  \\
Plays no instrument      & 42 & 38 & 80  \\
\midrule
\textbf{Total}           & 60 & 60 & 120 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}

\problem[4cm]{One of the 120 surveyed students is chosen at random. Find the probability
that the student plays an instrument.}

\problem[4.5cm]{A student is chosen at random \emph{from those who play an instrument}.
Find the probability that this student also plays a sport.}

\problem[4.5cm]{A student is chosen at random from those who play a sport. Find the
probability that this student plays an instrument.}

\problem[3.5cm]{Test whether ``plays a sport'' and ``plays an instrument'' are independent
for these 120 students.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $P(\text{sport}) \cdot P(\text{instrument})$. \quad \ansblank
  \item[(b)] Compute $P(\text{sport and instrument})$. \quad \ansblank
  \item[(c)] Explain what the comparison of (a) and (b) tells you about independence.
\end{itemize}
\vspace*{2.4cm}}

\problem[5cm]{Two cards are dealt from a standard 52-card deck without replacement.
Find the probability that both cards are hearts. (There are 13 hearts in the deck.)}

\problem[4.5cm]{A machine produces parts independently, and each part is defective with
probability $0.04$. Two parts are selected. Find the probability that both are defective.}

\problem[5cm]{On the same production line, each part is good with probability $0.96$,
independently of the others. Five parts are selected. Find the probability that at least one
of the five is defective. Round to four decimal places.}

\problem[3.5cm]{A trial is run twice, independently. Each trial succeeds with probability $p$
and fails with probability $q$, where $p + q = 1$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Expand $(p + q)^2$. \quad \ansblank
  \item[(b)] Explain which two-trial outcome each term of your expansion counts, and why one term carries a factor of $2$.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{In a screening programme, $5\%$ of people have a certain condition. The test
returns a positive result for $90\%$ of people who have it, and for $8\%$ of people who do not.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the probability that a randomly chosen person tests positive. Round to three decimal places. \quad \ansblank
  \item[(b)] Given that a person tested positive, find the probability that they have the condition. Round to three decimal places. \quad \ansblank
\end{itemize}
\vspace*{3cm}}

\problem[3.5cm]{Return to the survey table of 120 students.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the probability that a randomly chosen student plays neither a sport nor an instrument. \quad \ansblank
  \item[(b)] Find the probability that a student plays no instrument, given that the student plays no sport. \quad \ansblank
  \item[(c)] Explain whether knowing that a student plays no sport changes the chance that they play no instrument.
\end{itemize}
