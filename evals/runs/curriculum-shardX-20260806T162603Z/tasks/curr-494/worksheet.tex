\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Each problem shows work that is wrong in one specific way. Repair it, then say what rule was broken.
  \item Part (a) always wants a corrected value; put it on the Answer line. Part (b) wants a sentence or two, written in the work space.
  \item Angles are in radians throughout.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{A curve is given by $x = t^2$, $y = t^3 - 3t$. A student is asked for $\dfrac{dy}{dx}$ at $t = 2$ and reports $9$, having computed $\dfrac{dy}{dt} = 3t^2 - 3$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct value of $\dfrac{dy}{dx}$ at $t = 2$.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[6cm]{A particle in the plane has velocity components $x'(t) = 3t^2$ and $y'(t) = 4t$. Asked for its speed at $t = 1$, a student reports $7$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct speed at $t = 1$.
  \item[(b)] Name the error, and say why the components cannot simply be added.
\end{itemize}}

\problem[6cm]{To find the area enclosed by the polar curve $r = 4\sin\theta$, a student writes $A = \frac{1}{2}\displaystyle\int_0^{2\pi} 16\sin^2\theta\,d\theta$ and obtains $8\pi \approx 25.133$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct area. Give an exact answer and a decimal to three places.
  \item[(b)] Explain what went wrong with the limits, and describe how to check the correct interval.
\end{itemize}}

\problem[6.5cm]{For the curve $x = t^2$, $y = t^4$ (with $t > 0$) a student correctly finds $\dfrac{dy}{dx} = 2t^2$, then differentiates that with respect to $t$ and reports $\dfrac{d^2y}{dx^2} = 12$ at $t = 3$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Compute $\dfrac{d}{dt}\!\left(\dfrac{dy}{dx}\right)$, then find the correct value of $\dfrac{d^2y}{dx^2}$ at $t = 3$.
  \item[(b)] Name the step the student omitted.
\end{itemize}}

\problem[6.5cm]{Asked for the area of one petal of the polar rose $r = 3\cos(3\theta)$, a student integrates over a full turn and reports $\frac{9\pi}{2} \approx 14.137$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct area of one petal. Give an exact answer and a decimal to three places.
  \item[(b)] Explain how the correct limits are found from the equation itself.
\end{itemize}}

\problem[6.5cm]{A particle has velocity components $x'(t) = 2t$ and $y'(t) = 3t^2$ for $0 \le t \le 2$. Asked for the total distance travelled, a student computes $\displaystyle\int_0^2 2t\,dt + \int_0^2 3t^2\,dt = 4 + 8 = 12$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct total distance, to three decimal places.
  \item[(b)] Name the error the student made.
\end{itemize}}

\problem[7cm]{For the polar curve $r = 1 + \cos\theta$, a student is asked for the slope of the tangent line at $\theta = \frac{\pi}{2}$. They compute $\dfrac{dr}{d\theta} = -\sin\theta$, evaluate it at $\frac{\pi}{2}$, and report a slope of $-1$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct slope $\dfrac{dy}{dx}$ at $\theta = \frac{\pi}{2}$.
  \item[(b)] Explain why $\dfrac{dr}{d\theta}$ is not the slope of the tangent line.
\end{itemize}}

\problem[7cm]{A particle has position $x(t) = t^2 - 4t$, $y(t) = \dfrac{t^3}{3} - t$. Asked for the speed at $t = 3$, a student evaluates the position at $t = 3$, gets $(-3, 6)$, and reports $\sqrt{45} \approx 6.708$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the correct speed at $t = 3$, to three decimal places.
  \item[(b)] Name the error, then decide whether the particle is speeding up or slowing down at $t = 3$ and justify your answer.
\end{itemize}}
