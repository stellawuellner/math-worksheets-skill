\sstitleblock{Solving Logarithmic Equations with Domain Checks}

\skillheading{Skill 1 --- Converting a logarithm to exponential form}

\begin{formulabox}
\textbf{Definition.} For $b > 0$, $b \neq 1$: \quad
$\log_{b} A = c$ \ \ means \ \ $A = b^{\,c}$.\par
\textbf{Domain rule.} The argument $A$ must be strictly positive. Write that condition down \emph{before} solving --- it is what tells you which answers to keep.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_{2}(x - 5) = 4$.
\step{One logarithm, a number on the other side --- convert straight to exponential form, then check the argument stays positive.}
\step{$x - 5 = 2^{4} = 16$, so $x = 21$; then $21 - 5 = 16 > 0$.}

$\Rightarrow\quad \ans{x = 21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_{3}(x + 2) = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 25}$}
\end{tryitbox}

\skillheading{Skill 2 --- Equating arguments of matching logarithms}

\begin{formulabox}
\textbf{One-to-one property.} \quad $\log_{b} A = \log_{b} B$ \ \ means \ \ $A = B$, \ provided $A > 0$ and $B > 0$.\par
A candidate that makes either argument negative is extraneous, even though it satisfies $A = B$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_{7}(4x - 1) = \log_{7}(x + 8)$.
\step{The bases match and each side is a single logarithm, so the one-to-one property lets me drop both logs at once.}
\step{$4x - 1 = x + 8$, so $3x = 9$; at $x = 3$ the arguments are $11$ and $11$, both positive.}

$\Rightarrow\quad \ans{x = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_{2}(3x + 4) = \log_{2}(x + 12)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Condensing two logarithms, then solving}

\begin{formulabox}
\textbf{Product:} $\log_{b} M + \log_{b} N = \log_{b}(MN)$. \quad
\textbf{Quotient:} $\log_{b} M - \log_{b} N = \log_{b}\!\left(\frac{M}{N}\right)$.\par
Condensing usually produces a quadratic, so expect two candidates and expect the domain to reject one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_{2} x + \log_{2}(x - 2) = 3$.
\step{Two logarithms with the same base added together --- condense with the product rule first, since the domain needs $x > 2$.}
\step{$x(x - 2) = 2^{3} = 8$, so $x^{2} - 2x - 8 = 0$ and $(x - 4)(x + 2) = 0$; $x = -2$ fails $x > 2$.}

$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_{5} x + \log_{5}(x - 4) = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}
