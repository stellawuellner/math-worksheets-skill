\sstitleblock{Expanded Form, Comparing, and Rounding}

\skillheading{1. Expanded form: what each digit is worth}

\begin{formulabox}
\textbf{Every place is ten times the one to its right.}
\par\medskip
\begin{center}
{\small\begin{tabular}{|P|P|P|P|P|}
\hline
\textbf{Ten thousands} & \textbf{Thousands} & \textbf{Hundreds} &
\textbf{Tens} & \textbf{Ones} \\ \hline
\end{tabular}}
\end{center}
\par\medskip
\textbf{Expanded form} writes the number as the sum of those values. A place
with no digit still gets a zero in standard form --- the placeholder is what
keeps every other digit where it belongs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $4907$ in expanded form.
\step{Expanded form asks for each digit's value, so name the place of every
digit before writing anything.}
\step{$4$ thousands, $9$ hundreds, $0$ tens, $7$ ones, so the values are
$4000$, $900$, nothing, and $7$.}

$\Rightarrow\quad \ans{4907 = 4000 + 900 + 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $5263$ in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5263 = 5000 + 200 + 60 + 3}$}
\end{tryitbox}

\skillheading{2. The value of one digit}

\begin{formulabox}
\textbf{Digit or value?} The \emph{digit} is the symbol you see; the
\emph{value} is the digit times its place. In $3600$ the digit $6$ has the
value $600$.\\[3pt]
\textbf{Same digit, two places:} each step to the left multiplies the value by
$10$. Two places apart is $100$ times; three places apart is $1000$ times.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} In $2640$, what is the value of the $6$?
\step{The question asks for a value, not a digit, so find which place the $6$
sits in and multiply.}
\step{Counting from the right --- ones, tens, hundreds --- the $6$ is in the
hundreds place, so its value is $6 \times 100$.}

$\Rightarrow\quad \ans{600}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} In $19342$, what is the value of the $9$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9000}$}
\end{tryitbox}

\skillheading{3. Comparing: biggest place, not biggest digit}

\begin{formulabox}
\textbf{Rule:} line the numbers up by place and start at the \emph{left}. Move
right only while the digits are equal. The first place where they differ
decides the whole comparison --- nothing further right can change it.\\[3pt]
The small point of $<$ and $>$ aims at the smaller number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $<$ or $>$ between $52861$ and $52816$.
\step{Comparing starts at the largest place, so walk left to right until the
digits stop matching.}
\step{Ten-thousands, thousands and hundreds all match ($5$, $2$, $8$). The
tens place is the first difference: $6$ against $1$.}

$\Rightarrow\quad \ans{52861 > 52816}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $<$ or $>$ between $30472$ and $30742$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30472 < 30742}$}
\end{tryitbox}

\skillheading{4. Rounding to a named place}

\begin{formulabox}
\textbf{Three steps:} find the place you are rounding to; look at the digit
\emph{one place to its right}; five or more rounds up, four or less stays.
Then every digit to the right becomes zero.\\[3pt]
\textbf{Watch the carry:} if the rounding place holds a $9$ and it rounds up,
the extra one moves into the next place left --- ten hundreds is one thousand.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Round $6473$ to the nearest hundred.
\step{Rounding to hundreds is a choice between two hundreds, so the digit that
decides is the one just to the right --- the tens digit.}
\step{The tens digit is $7$, which is five or more, so the hundreds digit goes
from $4$ up to $5$ and the rest become zero.}

$\Rightarrow\quad \ans{6500}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Round $24681$ to the nearest thousand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25000}$}
\end{tryitbox}
