\sstitleblock{Solving and Graphing One-Step Inequalities}

\begin{formulabox}
\textbf{Solve it like an equation, with one extra rule.} Undo the single
operation on both sides. Adding or subtracting never changes the sign;
\textbf{multiplying or dividing by a negative reverses it} ($<$ becomes $>$,
$\le$ becomes $\ge$).\\[3pt]
\textbf{Graph it.} Open circle $\circ$ for $<$ or $>$ (boundary not included),
solid dot $\bullet$ for $\le$ or $\ge$ (boundary included). Then shade the ray
that holds the numbers that work.\\[3pt]
\textbf{Always test one number} from the shaded side back in the original
inequality. One test catches a missed flip every time.
\end{formulabox}

\skillheading{1. Undo an addition or a subtraction}

\begin{formulabox}
If a number is added to the variable, subtract it from both sides; if it is
subtracted, add it. The inequality sign never moves for these two operations,
even when the answer comes out negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve and graph $x + 3 > 10$.
\step{Only one thing is being done to $x$ --- $3$ is added --- so undo it by
subtracting $3$, and leave the sign alone.}
\step{$x + 3 - 3 > 10 - 3$ gives $x > 7$. Strict $>$, so an open circle at $7$
shaded right.}
\begin{center}
\begin{tikzpicture}[x=0.5cm,y=1cm]
  \draw[thick,<->] (-1.6,0) -- (11.6,0);
  \foreach \i in {0,2,...,10} {\draw (\i,0.1) -- (\i,-0.1);
     \node[below=2pt,font=\scriptsize] at (\i,0) {$\i$};}
  \draw[very thick,->] (7,0) -- (11.3,0);
  \draw[thick,fill=white] (7,0) circle (2.4pt);
\end{tikzpicture}
\end{center}
Solution: $\ans{x > 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve and graph $k - 4 \le 1$. Which kind of circle goes at
the boundary?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k \le 5}$ (solid dot, shaded left)}
\end{tryitbox}

\skillheading{2. Undo a multiplication or a division}

\begin{formulabox}
If the variable is multiplied by a positive number, divide both sides by it; if
it is divided, multiply. Positive multiplier or divisor means the sign stays
put --- a negative number in the \emph{answer} is not a reason to flip.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve and graph $5x \le 35$.
\step{The variable is multiplied by $5$, so divide both sides by $5$; $5$ is
positive, so the sign does not move.}
\step{$\dfrac{5x}{5} \le \dfrac{35}{5}$ gives $x \le 7$. The $\le$ includes the
boundary, so a solid dot at $7$ shaded left.}
\begin{center}
\begin{tikzpicture}[x=0.5cm,y=1cm]
  \draw[thick,<->] (-1.6,0) -- (11.6,0);
  \foreach \i in {0,2,...,10} {\draw (\i,0.1) -- (\i,-0.1);
     \node[below=2pt,font=\scriptsize] at (\i,0) {$\i$};}
  \draw[very thick,->] (7,0) -- (-1.3,0);
  \fill (7,0) circle (2.4pt);
\end{tikzpicture}
\end{center}
Solution: $\ans{x \le 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve and graph $\dfrac{p}{2} > 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{p > 6}$ (open circle, shaded right)}
\end{tryitbox}

\skillheading{3. Flip when you multiply or divide by a negative}

\begin{formulabox}
Multiplying by a negative reflects every number across $0$, so the larger one
lands on the smaller side: $3 > 1$, but $-3 < -1$. That is why the sign must
reverse the moment you divide or multiply by a negative --- and only then.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve and graph $-3x < 12$.
\step{The coefficient is negative, so the step that isolates $x$ is a division
by $-3$ --- the one move that reverses the sign.}
\step{$\dfrac{-3x}{-3}$ and $\dfrac{12}{-3}$ give $x$ and $-4$, and $<$ turns
into $>$: $x > -4$. Test $x = 0$: $-3(0) = 0 < 12$ is true, and $0$ is in the
shaded part.}
\begin{center}
\begin{tikzpicture}[x=0.5cm,y=1cm]
  \draw[thick,<->] (-9.6,0) -- (3.6,0);
  \foreach \i in {-8,-6,...,2} {\draw (\i,0.1) -- (\i,-0.1);
     \node[below=2pt,font=\scriptsize] at (\i,0) {$\i$};}
  \draw[very thick,->] (-4,0) -- (3.3,0);
  \draw[thick,fill=white] (-4,0) circle (2.4pt);
\end{tikzpicture}
\end{center}
Solution: $\ans{x > -4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve and graph $-w \ge 5$. (The coefficient is $-1$.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{w \le -5}$ (solid dot, shaded left)}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the flip rule keys on what you multiply or divide
\emph{by}, never on a minus sign that merely appears in the problem. In
$6w \le -18$ the divisor is $6$, so nothing flips; in $-2x > 8$ the divisor is
$-2$, so it does.
\end{watchoutbox}
