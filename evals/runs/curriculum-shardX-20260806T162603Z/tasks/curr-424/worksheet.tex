\wstitleblock{Find and Fix: Illegal Cancellation and Lost Solutions}{Precalculus}{}

\vspace{0.15cm}
\noindent\textbf{Directions.} Each problem shows work a student handed in. For every one:
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item produce the complete, correct solution set on the stated interval, in increasing order;
  \item then name the error in a sentence or two --- say which step is illegal and \emph{which} solutions it destroyed or invented.
\end{itemize}
\vspace{0.15cm}

\problem{A student solves $2\sin\theta\cos\theta = \sin\theta$ on $0^\circ \le \theta < 360^\circ$ by dividing both sides by $\sin\theta$, getting $2\cos\theta = 1$, and reports $\theta = 60^\circ, 300^\circ$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the complete solution set. \answerline{degrees}
  \vspace*{3cm}
  \item[(b)] Name the error. \ansline
  \vspace*{2.4cm}
\end{itemize}}
