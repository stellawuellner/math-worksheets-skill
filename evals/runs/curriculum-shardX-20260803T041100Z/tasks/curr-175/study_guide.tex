\sstitleblock{Equivalent Expressions: Expanded and Factored Forms}

\begin{formulabox}
\textbf{One property, two directions.}
\[
a(b+c) \;=\; ab + ac
\]
Left to right it \textbf{expands}; right to left it \textbf{factors}. Two
expressions are \textbf{equivalent} when they agree for \emph{every} value of
the variable --- so one disagreeing substitution disproves equivalence, while
one agreeing substitution proves nothing.
\end{formulabox}

\skillheading{1. Expanding with the distributive property}

\begin{formulabox}
The multiplier outside touches \textbf{every} term inside --- one product per
term. Write the inside as a sum of signed terms ($x - 7$ is $x + (-7)$) so the
sign rules do the work: negative times negative is positive.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Expand $4(x+3)$.
\step{The parentheses hold two terms, so there will be two products --- write
them before simplifying anything.}
\step{$4 \cdot x = 4x$ and $4 \cdot 3 = 12$, so the sum of the products is
$4x + 12$.}
\[ 4(x+3) = \ans{4x+12} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Expand $5(2x+7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10x+35}$}
\end{tryitbox}

\skillheading{2. Factoring out the greatest common factor}

\begin{formulabox}
Break each term into prime factors, take everything the terms \emph{share}, and
write it outside. Divide each original term by that factor to get what stays
inside.\\[2pt]
\textbf{Completely} means the terms left inside share nothing: if they still do,
you pulled out a common factor, not the \emph{greatest} one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $6x + 9$ completely.
\step{Factoring is expanding read backwards, so start from what the two terms
have in common rather than from the expression as a whole.}
\step{$6x = 2 \cdot 3 \cdot x$ and $9 = 3 \cdot 3$ share only 3, and dividing
gives $2x$ and 3 inside.}
\[ 6x + 9 = \ans{3(2x+3)} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $14x - 21$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7(2x-3)}$}
\end{tryitbox}

\skillheading{3. Collecting like terms}

\begin{formulabox}
Like terms have the \textbf{identical variable part}: $5x$ and $2x$ are like,
$5x$ and $5y$ are not, and $5x$ and 9 are not. Add or subtract only the
coefficients; the variable part never changes.\\[2pt]
If parentheses are present, expand first --- nothing can be collected through a
set of parentheses.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $5x + 2x + 9$ with as few terms as possible.
\step{Sort the terms by their variable part first, so that only genuinely like
terms get added.}
\step{$5x$ and $2x$ are like and give $7x$; the 9 has no $x$, so it is carried
along unchanged.}
\[ 5x + 2x + 9 = \ans{7x+9} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $3(x+2) + 2x$ with as few terms as possible.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x+6}$}
\end{tryitbox}

\skillheading{4. Testing equivalence by substitution}

\begin{formulabox}
Pick a value, substitute it into \textbf{both} expressions, and compare.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item Different values $\Rightarrow$ \textbf{not equivalent}. One counterexample
is a complete disproof.
\item Same value $\Rightarrow$ evidence only. Two different expressions can
still cross at one point.
\end{itemize}
Avoid $x = 0$ and $x = 1$ as test values --- too many wrong expressions agree
there by accident.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the value of $3(x+5)$ at $x = 4$, to test it against a
claimed expansion.
\step{Evaluate the correct expression first, so there is a reference value to
compare a claim against.}
\step{Substituting gives $3(4+5) = 3 \cdot 9$, and that product is 27.}
\[ 3(x+5) \text{ at } x=4 \;=\; \ans{27} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the value of $2(x-3)$ at $x = 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} distribute to \emph{every} term, not just the first ---
$3(x+4)$ is $3x+12$, never $3x+4$. When the multiplier is negative, both signs
inside change. And after factoring, check that the terms inside the parentheses
no longer share a factor.
\end{watchoutbox}
