\sstitleblock{Solving Systems by Substitution}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- When one equation is already solved}

\begin{formulabox}
\textbf{The move:} if one equation says $y = (\text{something in } x)$, replace $y$ in the
\emph{other} equation by that something. The result is one equation in one variable.
\textbf{Two habits that prevent most errors:} wrap the substituted expression in brackets, and put
it into the equation it did \emph{not} come from --- substituting back into its own equation only
ever gives $0 = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $y = 3x + 1$ and $2x + y = 11$.
\step{The first equation already hands me $y$ as an expression in $x$, so substitution turns the second equation into one I can solve straight away.}
\step{$2x + (3x + 1) = 11$, so $5x + 1 = 11$, $5x = 10$, $x = 2$.}
\step{Back-substitute: $y = 3(2) + 1 = 7$. Check in the second equation: $4 + 7 = 11$.}

$\Rightarrow\quad \ans{(2,\,7)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = 2x + 1$ and $x + y = 10$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\,7)}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Isolating a variable first}

\begin{formulabox}
\textbf{When neither equation is ready:} pick the variable whose coefficient is $1$ or $-1$ and
isolate it. That choice is what keeps fractions out of the substitution.
Isolating a variable is itself a legal move --- add or subtract the same amount on both sides, then
divide --- so the new equation still says exactly what the old one said.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x + 2y = 11$ and $3x - y = 5$.
\step{Nothing is solved for a variable yet, so I isolate the one with coefficient 1 --- the $x$ in the first equation --- to avoid fractions.}
\step{$x = 11 - 2y$. Substituting into the second: $3(11 - 2y) - y = 5$.}
\step{$33 - 6y - y = 5$, so $-7y = -28$ and $y = 4$; then $x = 11 - 8 = 3$.}

$\Rightarrow\quad \ans{(3,\,4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x + 3y = 13$ and $2x - y = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(4,\,3)}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- When the variable disappears}

\begin{formulabox}
\textbf{Two endings that look like failures and are not:} sometimes the substitution cancels every
$x$ and leaves a statement with no variable in it.
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=2pt]
  \item A \emph{false} statement such as $-4 = 0$: no solution. The lines are parallel.
  \item A \emph{true} statement such as $8 = 8$: infinitely many solutions. The two equations
        describe the same line.
\end{itemize}
Write which of the two it is; do not try to solve the leftover statement.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $y = 2x + 5$ and $4x - 2y + 6 = 0$.
\step{The first equation is ready to substitute, so I do that and watch what happens to the variable rather than assuming a number will come out.}
\step{$4x - 2(2x + 5) + 6 = 0$, so $4x - 4x - 10 + 6 = 0$, which is $-4 = 0$.}
\step{Every $x$ has cancelled and the statement left behind is false, so nothing can satisfy both.}

$\Rightarrow\quad \ans{\text{no solution}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = 3x + 2$ and $6x - 2y + 1 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{no solution}}$}
\end{tryitbox}
