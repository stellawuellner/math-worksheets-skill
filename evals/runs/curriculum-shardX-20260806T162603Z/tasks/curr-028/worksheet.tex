\wstitleblock{Take-Away and Comparing Stories}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Some stories \textbf{take away}: things pop, fly off, or get eaten.
  \item Some stories \textbf{compare}: who has more, and how many more?
  \item Draw counters or cross things out any time it helps, and write your
        answer with its label.
\end{itemize}

\problem{Jon has 6 balloons. 2 balloons pop. How many balloons does Jon have
left?
\par\vspace*{3.5cm}\answerline{balloons}}

\problem{There are 8 cookies on a plate. Max eats 3 cookies. Draw the
cookies and cross out the eaten ones if it helps. How many cookies are left?
\par\vspace*{3.5cm}\answerline{cookies}}

\problem{9 kites are flying. 4 kites come down to land. Write the story
numbers in the equation, then write how many kites are still flying.
\par\vspace{6pt}
\begin{center}
\scratchblank\ $-$ \scratchblank\ $=$~\ansblank
\end{center}
\par\vspace*{2.5cm}}

\problem{Lena has 7 stickers. Ben has 5 stickers. How many more stickers
does Lena have than Ben?
\par\vspace*{3cm}\answerline{stickers}}

\problem{Row A shows 6 counters. Row B shows 4 counters. Draw a line from
each counter in Row A to a partner in Row B. How many counters in Row A have
no partner?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.8, line width=0.8pt]
  \node[left] at (-0.5,1.1) {Row A};
  \foreach \x in {0,1,2,3,4,5} \fill (\x,1.1) circle (0.2);
  \node[left] at (-0.5,0) {Row B};
  \foreach \x in {0,1,2,3} \draw (\x,0) circle (0.2);
\end{tikzpicture}
\end{center}
\par\vspace*{2.5cm}\answerline{counters}}

\problem{Zoe has 10 grapes. She eats 4 grapes, and then she eats 2 more.
How many grapes does Zoe have left?
\par\vspace*{3.5cm}\answerline{grapes}}

\problem{Sam has 9 marbles. Tia has 6 marbles. How many fewer marbles does
Tia have than Sam?
\par\vspace*{3cm}\answerline{marbles}}

\problem{Challenge! Ava picks 10 flowers. She gives 3 flowers to her mom.
Kim has 5 flowers.
\begin{itemize}[leftmargin=1.2cm, itemsep=10pt, topsep=8pt]
  \item How many flowers does Ava have left?~\ansblank
  \item Who has more flowers now? Write her name:~\ansblank
  \item How many more?~\ansblank
\end{itemize}
