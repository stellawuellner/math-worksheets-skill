\sstitleblock{One, No, or Infinitely Many Solutions}

\begin{formulabox}
\textbf{The whole method in one move:} simplify each side, then get the
variable terms together. What is left decides everything.\par
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item A variable term survives $\Rightarrow$ \textbf{one solution}.
  \item The variable cancels and the leftover statement is \textbf{false}
        $\Rightarrow$ \textbf{no solution}.
  \item The variable cancels and the leftover statement is \textbf{true}
        $\Rightarrow$ \textbf{infinitely many solutions}.
\end{itemize}
\end{formulabox}

\skillheading{Skill 1 --- Equations with exactly one solution}

\begin{formulabox}
\textbf{Test:} after simplifying, the two sides have \emph{different} numbers
of the variable. Then one subtraction leaves a single variable term, and
dividing gives the one value that works.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and classify $5x + 2 = 3x + 10$.
\step{The sides carry different amounts of $x$ (five of them against three),
so the variable cannot cancel --- expect exactly one solution.}
\step{Subtract $3x$: $2x + 2 = 10$. Subtract 2: $2x = 8$. Divide by 2:}

$x = 4$, one solution. \quad $\ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve and classify $7x + 1 = 2x + 16$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Equations with no solution}

\begin{formulabox}
\textbf{Test:} the variable terms match exactly, but the constants do not.
Subtracting the variable term leaves a false number sentence, and no value of
the variable can repair it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and classify $4x + 9 = 4x + 2$.
\step{Both sides carry the same amount of $x$, so the variable will cancel ---
the constants alone will decide the answer.}
\step{Subtract $4x$ from both sides: $9 = 2$, and the two constants differ by
7, so the sentence is false.}

No value of $x$ works. \quad $\ans{7 \neq 0}$, no solution
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve and classify $6x + 5 = 6x - 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6 \neq 0}$, no solution}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``No solution'' is not the same as ``$x = 0$''.
$x = 0$ means zero works; no solution means nothing works. Write the words.
\end{watchoutbox}

\skillheading{Skill 3 --- Equations true for every value}

\begin{formulabox}
\textbf{Test:} after distributing, the two sides are the \emph{same
expression}. Subtracting leaves a true sentence such as $0 = 0$, so every
number is a solution.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and classify $3(2x + 4) = 6x + 12$.
\step{The left side is not simplified yet, so distribute first and then
compare the two sides term by term.}
\step{Distribute: $3(2x + 4) = 6x + 12$, which is exactly the right side.}

Every value of $x$ works. \quad $\ans{6x + 12 = 6x + 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve and classify $5(x - 2) = 5x - 10$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x - 10 = 5x - 10}$, infinitely many}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} distribute to \emph{every} term inside the
parentheses. Forgetting the second one turns an ``infinitely many'' equation
into a fake one-solution answer.
\end{watchoutbox}

\vspace{0.2cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Solution:] a value that makes the equation a true statement.
  \item[Identity:] an equation true for every value of the variable.
  \item[Like terms:] terms with the same variable part, which can be combined.
  \item[Distribute:] multiply the outside factor by each term inside.
\end{description}
\end{multicols}
