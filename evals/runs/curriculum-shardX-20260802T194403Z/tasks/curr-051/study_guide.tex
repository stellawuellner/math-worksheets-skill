\sstitleblock{Reading Base-Ten Models: Hundreds, Tens, and Ones}

\skillheading{1. Reading a model (read-base-ten-model)}

\begin{formulabox}
\textbf{The three blocks:} \quad
\btflat~$=$ one \textbf{hundred} \quad$\cdot$\quad
\btrod~$=$ one \textbf{ten} \quad$\cdot$\quad
\btone~$=$ one \textbf{one}\\[4pt]
\textbf{Rule:} count flats, then rods, then units, and write one digit in each place:
\quad $(\text{flats}) \times 100 + (\text{rods}) \times 10 + (\text{units}) \times 1$.
An empty group still needs a digit --- write $0$ to hold its place.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A model shows 3 flats, 5 rods, and 4 units. What number is it?
\step{Each kind of block has one job: flats count hundreds, rods count tens, units count ones --- so I count each group and put its count in that place.}
\step{$3 \times 100 = 300$, \ $5 \times 10 = 50$, \ $4 \times 1 = 4$, \ and $300 + 50 + 4 = \ans{354}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A model shows 2 flats, no rods, and 7 units. What number is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{207}$}
\end{tryitbox}

\skillheading{2. Expanded form (expanded-form-from-model)}

\begin{formulabox}
\textbf{Expanded form} writes what each group of blocks is worth, then adds:\\[3pt]
$347 = 300 + 40 + 7$ \quad (3 flats $+$ 4 rods $+$ 7 units).\par
The digit tells \emph{how many}; the place tells \emph{how much each one is worth}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write 6 flats, 1 rod, and 8 units in expanded form.
\step{Expanded form is just the block count with each group's value written out, so I turn each group into its value before adding.}
\step{$6$ flats $= 600$, \ $1$ rod $= 10$, \ $8$ units $= 8$, so the expanded form is $600 + 10 + 8 = \ans{618}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write 4 flats, 5 rods, and 3 units in expanded form and as a number.\\[2pt]
\rotatebox{180}{\footnotesize check: $400+50+3 = \ans{453}$}
\end{tryitbox}

\skillheading{3. Trading rods for flats (regroup-tens-into-hundreds)}

\begin{formulabox}
\textbf{Rule:} a place can only hold the digits $0$--$9$. If a model shows ten or more
rods, trade: \quad $10$ rods $=1$ flat. \quad Trading never changes the number, only
how it is shown.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A model shows 1 flat, 12 rods, and 4 units.
\step{Twelve rods will not fit in the tens place, so I trade ten of them for one flat before reading any digits.}
\step{$12$ rods $= 1$ flat $+ 2$ rods, so the model is $2$ flats, $2$ rods, $4$ units: $200 + 20 + 4 = \ans{224}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A model shows 3 flats, 11 rods, and 2 units. What number is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{412}$}
\end{tryitbox}

\skillheading{4. Comparing two models (compare-base-ten-models)}

\begin{formulabox}
\textbf{Rule:} read both numbers first, then compare \textbf{hundreds first}. Only if
the hundreds tie do the tens decide; only if the tens tie do the ones decide.\par
$<$ means \emph{is less than}; \ $>$ means \emph{is greater than}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Model A: 2 flats, 5 rods, 1 unit. Model B: 2 flats, 3 rods, 9 units.
\step{More single blocks does not mean a bigger number, so I read each model and then compare one place at a time from the left.}
\step{A $= 200 + 50 + 1 = 251$, \ B $= 200 + 30 + 9 = 239$; hundreds tie, and $5$ tens beat $3$ tens, so A $\ans{>}$ B}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Model A: 1 flat, 8 rods, 4 units. Model B: 2 flats, no rods, 3 units.\\[2pt]
\rotatebox{180}{\footnotesize check: A $\ans{<}$ B}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an empty group needs a $0$. Three flats and six units is
$306$, never $36$ --- the $0$ is what keeps the $3$ in the hundreds place.
\end{watchoutbox}
