\wstitleblock{Adding Three-Digit Numbers with Regrouping}{Grades 2--3}{}

\begin{center}
\small adding three-digit numbers with one regroup and with two, estimating a sum by rounding to the nearest hundred, and finding a regrouping mistake
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Add one column at a time, starting with the ones.
  \item When a column adds to ten or more, write the ones digit and carry the ten into the next column.
  \item Write the small carry digit above the column it goes into, so nothing gets lost.
\end{itemize}
