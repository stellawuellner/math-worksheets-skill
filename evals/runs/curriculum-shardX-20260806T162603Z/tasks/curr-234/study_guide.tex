\sstitleblock{Input, Output, and Function Notation}

\skillheading{Skill 1 --- Telling the input from the output}

\begin{formulabox}
\textbf{Rule:} in $f(a) = b$, the number in the brackets is the \textbf{input} and the number
after the equals sign is the \textbf{output}.\\[3pt]
``Find $f(4)$'' hands you the input and asks for the output. ``Find $x$ when $f(x) = 15$'' hands
you the output and asks for the input. Read the question for which one is missing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 5x + 1$. \quad Find $f(4)$.
\step{The $4$ sits in the brackets, so it is the input --- substitute it and simplify.}
\step{$f(4) = 5(4) + 1 = 20 + 1$}
$\Rightarrow\quad f(4) = \ans{21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 5x + 1$. \quad Find $f(-1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Working backwards from an output}

\begin{formulabox}
\textbf{Rule:} when the output is given, set the rule \textit{equal} to it and solve for the
input. Do not substitute the output for $x$.\\[3pt]
$f(x) = \text{given output} \quad\Longrightarrow\quad$ solve the equation.
Check your answer by substituting it back into the rule.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 2x + 7$. \quad Find $x$ when $f(x) = 15$.
\step{$15$ is an output, so it belongs on the right of an equation --- solve, do not substitute.}
\step{$2x + 7 = 15$, so $2x = 8$ and $x = 8 \div 2$. Check: $2(4) + 7 = 15$ $\checkmark$}
$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 2x + 7$. \quad Find $x$ when $f(x) = 21$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}

\skillheading{Skill 3 --- Spotting a swapped substitution}

\begin{formulabox}
\textbf{Rule:} the input replaces $x$ and \textit{only} $x$. Every other number in the rule stays
exactly where it was written.\\[3pt]
To check someone's work, rewrite the rule with brackets where $x$ was --- $4(\;\;) - 9$ --- and
drop the input into the gap. If their line does not match that shape, the numbers were swapped.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $h(x) = 4x - 9$. A student wrote $h(-2) = 4(-9) - 2 = -38$. Correct it.
\step{Compare the student's line with the shape $4(\;\;) - 9$: the input and the constant have changed places.}
\step{Put the input in the gap: $h(-2) = 4(-2) - 9 = -8 - 9$}
$\Rightarrow\quad h(-2) = \ans{-17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $h(x) = 4x - 9$. A student wrote $h(3) = 4(-9) - 3 = -39$. Correct it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an output can look like an input. In $p(5)$ the $5$ is an input; in
$p(x) = 5$ the same $5$ is an output. Check which side of the equals sign it is on.
\end{watchoutbox}
