\sstitleblock{Product, Quotient, and Power Properties of Logarithms}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Expanding a Logarithm into Terms}

\begin{formulabox}
\textbf{The three properties} (same base $b$ throughout, all arguments positive):
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item \textbf{Product:} $\log_b(MN) = \log_b M + \log_b N$
  \item \textbf{Quotient:} $\log_b(M/N) = \log_b M - \log_b N$
  \item \textbf{Power:} $\log_b(M^p) = p\log_b M$
\end{itemize}
Expanding runs left to right: multiplication becomes addition, division becomes subtraction, an exponent becomes a coefficient.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $\ln(16x)$, writing the constant as a multiple of $\ln 2$.

\step{The argument is a product, so the product property splits it; the numeric factor is a power of $2$, so the power property finishes it.}
\step{$\ln(16x) = \ln 16 + \ln x$, and $16 = 2^4$, so $\ln 16 = 4\ln 2$.}

$\Rightarrow\quad \ans{4\ln 2 + \ln x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $\log_5\!\left(\dfrac{x}{125}\right)$ into two terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\log_5 x - 3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Condensing to a Single Logarithm}

\begin{formulabox}
\textbf{Rule --- read the same three properties right to left.} \ Deal with coefficients \emph{first}: a number in front of a log is an exponent inside it. Then everything added goes into the numerator and everything subtracted goes into the denominator of one logarithm. Condensing is the move that makes an equation solvable, because $\log_b M = \log_b N$ forces $M = N$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $2\log_7 3 + \log_7 5$ as a single logarithm.

\step{There is a coefficient in front of the first term, so the power property must be applied before the two logs can be merged.}
\step{$2\log_7 3 = \log_7(3^2) = \log_7 9$, and the sum then condenses by the product property to $\log_7(9 \cdot 5)$.}

$\Rightarrow\quad \ans{\log_7 45}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $\ln 24 - \ln 3$ as a single logarithm and simplify it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\ln 8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\log_b M + \log_b N$ is $\log_b(MN)$, never $\log_b(M + N)$; and a coefficient becomes an exponent, so $2\log_b 3 = \log_b 9$, not $\log_b 6$.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Evaluating and Solving with the Properties}

\begin{formulabox}
\textbf{Rule.} \ To \emph{evaluate}, condense to one logarithm and then ask what power of the base gives that argument. To \emph{solve}, condense each side to a single logarithm, drop the logs, and solve the equation that remains --- then check that every original argument is positive, since a value that makes one negative is extraneous.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\log_4 8 + \log_4 2$ without a calculator.

\step{Neither term is a whole number on its own, but their product is a power of the base, so condense before evaluating.}
\step{$\log_4 8 + \log_4 2 = \log_4(8 \cdot 2) = \log_4 16$, and $4^2 = 16$.}

$\Rightarrow\quad \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_2 x + \log_2 5 = \log_2 30$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}
