\sstitleblock{Tape Diagrams and Double Number Lines for Ratios}

\noindent\small Both diagrams do the same job: they keep two quantities
scaling together. Find the value of \emph{one} box, or the value at \emph{one}
unit, and the rest is multiplication.
\normalsize

\skillheading{1. Tape diagrams: find one box first}
\begin{formulabox}
\textbf{Rule:} draw one box per part. Every box is worth the same amount, so
value of one box $=$ known amount $\div$ number of boxes it covers. Then
multiply for the other quantity or for the total.
\begin{center}
\begin{tikzpicture}[scale=0.75]
  \node[left, font=\scriptsize] at (-0.1,0.25) {red};
  \foreach \i in {0,1,2} {\draw (\i*0.7,0) rectangle (\i*0.7+0.7,0.5);}
  \node[left, font=\scriptsize] at (-0.1,-0.75) {blue};
  \foreach \i in {0,1,2,3,4} {\draw (\i*0.7,-1) rectangle (\i*0.7+0.7,-0.5);}
\end{tikzpicture}
\end{center}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Red to blue is $3 : 5$. There are $12$ red counters.
How many blue?
\step{The known amount covers the red tape only, so start by splitting it over
the $3$ red boxes rather than over all the boxes.}
\step{One box is $12 \div 3 = 4$ counters; blue is $5$ boxes, so $5 \times 4$.}
\par
$\ans{20}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Cats to dogs is $7 : 4$. There are $21$ cats. How many
dogs?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{2. Double number lines: move both lines together}
\begin{formulabox}
\textbf{Rule:} matching values sit above each other. Multiply or divide
\emph{both} lines by the same number to reach a new pair --- never one line
alone.
\begin{center}
\begin{tikzpicture}[x=0.85cm]
  \draw[->] (-0.2,0) -- (3.4,0);
  \draw[->] (-0.2,-0.8) -- (3.4,-0.8);
  \foreach \x in {0,1,2,3} {\draw (\x,-0.08) -- (\x,0.08); \draw (\x,-0.88) -- (\x,-0.72);}
  \node[right, font=\scriptsize] at (3.4,0) {kilograms};
  \node[right, font=\scriptsize] at (3.4,-0.8) {dollars};
\end{tikzpicture}
\end{center}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $4$ kilograms of flour cost $10$ dollars. What do $6$
kilograms cost?
\step{Six is not a whole number of steps from four, so go through one kilogram
--- the tick that makes every other value one multiplication.}
\step{One kilogram is $10 \div 4 = 2.5$ dollars, so six kilograms is
$6 \times 2.5$.}
\par
$\ans{15}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $6$ tickets cost $18$ dollars. What do $10$ tickets
cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$}
\end{tryitbox}

\skillheading{3. Reading the unit rate off the line}
\begin{formulabox}
\textbf{Rule:} the unit rate is the number sitting opposite $1$. Divide the
known pair by the first quantity to get there. Once it is written on the line,
every later question is ``multiply by the unit rate''.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $5$ notebooks cost $45$ dollars. Find the cost of one.
\step{The question asks for the value opposite $1$ notebook, so divide both
lines by $5$.}
\step{$45 \div 5 = 9$, and $5 \times 9 = 45$ checks it.}
\par
$\ans{9}$ dollars per notebook
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $8$ pens cost $56$ dollars. Find the cost of one pen.
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{4. Comparing two ratios}
\begin{formulabox}
\textbf{Rule:} two ratios can only be compared at the same place --- the same
number of one quantity. Use the unit rate, or write both as fractions with a
common denominator. Bigger boxes do not mean a bigger ratio.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Which is greater, $\frac{3}{4}$ or $\frac{5}{8}$?
\step{The denominators differ, so nothing can be compared until both parts are
measured against the same whole.}
\step{$\frac{3}{4} = \frac{6}{8}$, and $6$ eighths is more than $5$ eighths.}
\par
$\ans{\frac{3}{4} > \frac{5}{8}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare $\frac{2}{3}$ and $\frac{5}{6}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{2}{3} < \frac{5}{6}}$}
\end{tryitbox}
