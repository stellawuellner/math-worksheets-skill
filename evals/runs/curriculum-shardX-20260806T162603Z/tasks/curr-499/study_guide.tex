\sstitleblock{Two Ways a Series Argument Goes Wrong}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- When a test reaches no conclusion}

\begin{formulabox}
\textbf{Two tests that stay silent, and when.} The nth-term test proves divergence when the terms do not approach zero; if they do approach zero it proves \emph{nothing}. The ratio test decides when its limit is not $1$; a limit of exactly $1$ is no answer at all --- and for terms that are a ratio of polynomials it is guaranteed to come out $1$, so start with limit comparison instead.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The ratio test returns $1$ for $\displaystyle\sum_{n=1}^{\infty}\frac{n}{2n^{2}+1}$. Settle it another way.
\step{Ratio-of-polynomial terms always give a ratio limit of one, so switch to limit comparison and pick the partner from the leading powers, $n/n^{2} = 1/n$.}
\step{$\dfrac{n/(2n^{2}+1)}{1/n} = \dfrac{n^{2}}{2n^{2}+1}$, whose limit is finite and positive.}

$\Rightarrow\quad L = \ans{\frac{1}{2}}$, and since $\sum \frac{1}{n}$ diverges, so does this series
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $L$ for $\displaystyle\sum_{n=1}^{\infty}\frac{n}{3n^{2}+2}$ against $\displaystyle\sum\frac{1}{n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{1}{3}}$, so it diverges}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Endpoints are never free}

\begin{formulabox}
\textbf{The ratio test settles the open interval only.} At an endpoint its limit equals $1$ by construction, so it is silent exactly where you still need an answer. Substitute each endpoint back into the original series and test the ordinary series you get. The two most common results: $\sum \frac1n$ diverges, so that endpoint is excluded; $\sum \frac{(-1)^{n}}{n}$ converges, so that one is included.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n\,5^{n}}$ (centre $0$, radius $5$), test the endpoint $x = -5$.
\step{The ratio test cannot decide an endpoint, so substitute the value and identify the ordinary series that results.}
\step{At $x = -5$ the terms become $\dfrac{(-5)^{n}}{n\,5^{n}} = \dfrac{(-1)^{n}}{n}$, the alternating harmonic series, which converges.}

$\Rightarrow\quad \ans{-\ln 2 \approx -0.693}$, so this endpoint is included
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n^{2}\,4^{n}}$ (radius $4$), test the endpoint $x = 4$ and give the sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{\pi^{2}}{6}}$, so this endpoint is included}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Reading a formula off the wrong term}

\begin{formulabox}
\textbf{Formulas read the series as written.} A geometric sum is (\emph{first term actually written}) divided by $1-r$, so a sum starting at $n=3$ has first term $ar^{3}$, not $a$. An alternating remainder bound is the size of the \emph{first term omitted}, $a_{N+1}$, not the last one kept. And an integral test decides convergence without giving the sum. Check which index a formula is quoting before you use it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\sum_{n=3}^{\infty}\left(\frac{1}{2}\right)^{n}$.
\step{The ratio is one half, so the geometric formula applies --- but it needs the first term this sum actually contains, which is the one at $n = 3$.}
\step{First term $\left(\frac12\right)^{3} = \frac18$, ratio $\frac12$, so the sum is $\dfrac{1/8}{1 - 1/2}$.}

$\Rightarrow\quad \ans{\frac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\sum_{n=3}^{\infty}\left(\frac{1}{3}\right)^{n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{1}{18}}$}
\end{tryitbox}
