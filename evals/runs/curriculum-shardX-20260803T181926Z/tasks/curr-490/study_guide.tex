\skillheading{1. Average value of a function}

\begin{formulabox}
\textbf{Rule:} the average value of $f$ on $[a, b]$ is
$f_{\text{avg}} = \frac{1}{b-a}\int_a^b f(x)\,dx$. Ask ``what is the typical
height?'' --- the integral gives area, and dividing by the width $b - a$ turns
that area back into a height. The answer carries the units of $f$, never the
units of the integral. Warning sign that you have forgotten the division: an
answer that grows when the interval gets longer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the average value of $f(x) = 6x$ on $[1, 4]$.
\step{The question asks for a typical height rather than a total, so this is
the average-value formula: integrate, then divide by the interval width.}
\step{Width $= 4 - 1 = 3$, so compute $\int_1^4 \frac{6x}{3}\,dx
= \int_1^4 2x\,dx = \left[x^2\right]_1^4 = 16 - 1$.}
$f_{\text{avg}} = \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the average value of $f(x) = 4x$ on $[0, 5]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{2. Accumulated change from a rate}

\begin{formulabox}
\textbf{Rule:} if $r(t)$ is a rate of change, then $\int_a^b r(t)\,dt$ is the
\emph{change} in the quantity between $t = a$ and $t = b$, and the amount at the
end is $Q(b) = Q(a) + \int_a^b r(t)\,dt$. The integral alone is never the final
amount unless the starting amount is zero. When two rates act at once, integrate
the net rate: in minus out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Water enters a tank at $r(t) = 2t + 1$ litres per minute. How
much enters during the first $4$ minutes?
\step{The given function is a rate and the question asks for an amount, so
integrate the rate over the time interval --- no division here, unlike an
average value.}
\step{$\int_0^4 (2t + 1)\,dt = \left[t^2 + t\right]_0^4 = 16 + 4$.}
Water added $= \ans{20}$ litres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} How much enters in the first $4$ minutes at
$r(t) = 3t + 2$ litres per minute?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32}$}
\end{tryitbox}

\skillheading{3. Net displacement from velocity}

\begin{formulabox}
\textbf{Rule:} for a particle with velocity $v(t)$, the net displacement on
$[a, b]$ is $\int_a^b v(t)\,dt$, and the position is
$s(b) = s(a) + \int_a^b v(t)\,dt$. The integral is signed: an interval where
$v < 0$ subtracts. A negative answer is not an error --- it means the particle
finished behind where it started.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A particle has $v(t) = 2t - 5$ metres per second on
$[0, 4]$. Find its net displacement.
\step{``Displacement'' asks where the particle ended relative to where it
started, so integrate the velocity itself and keep the signs.}
\step{$\int_0^4 (2t - 5)\,dt = \left[t^2 - 5t\right]_0^4 = 16 - 20$.}
Displacement $= \ans{-4}$ metres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the net displacement for $v(t) = 2t - 8$ on
$[0, 5]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-15}$}
\end{tryitbox}

\skillheading{4. Total distance travelled}

\begin{formulabox}
\textbf{Rule:} total distance on $[a, b]$ is $\int_a^b |v(t)|\,dt$. Absolute
values are not integrated directly: solve $v(t) = 0$ first, split the interval
at every sign change inside it, integrate each piece, then add the absolute
values of the pieces. Total distance is always $\ge$ the size of the
displacement, and the two agree exactly when $v$ never changes sign.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the total distance for $v(t) = t - 3$ on $[0, 5]$.
\step{``Total distance'' means integrate speed, so find the sign change first:
$v = 0$ at $t = 3$, which is inside the interval, so the trip splits there.}
\step{$\int_0^3 (t - 3)\,dt = -4.5$ and $\int_3^5 (t - 3)\,dt = 2$; add the
sizes: $4.5 + 2$.}
Total distance $= \ans{6.5}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the total distance for $v(t) = t - 2$ on
$[0, 6]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Average \emph{velocity} is displacement over elapsed
time; average \emph{speed} is total distance over elapsed time. They are equal
only when the velocity never changes sign. Read the question for the word
``total'' --- it is the single word that decides whether an absolute value
belongs in your integral.
\end{watchoutbox}
