\wstitleblock{Adding and Subtracting Unlike Fractions}{Grade 5 Math}{}

\begin{center}
\small\itshape Building common denominators, then adding and subtracting unlike fractions and mixed numbers
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Fractions can only be added or subtracted when the pieces are the same size --- that is, when the denominators match.
  \item Find a common denominator first, rewrite each fraction with it, and only then add or subtract the numerators.
  \item Show the rewritten fractions in your work, not just the final answer.
  \item Give every answer in lowest terms, and write mixed numbers instead of top-heavy fractions.
\end{itemize}

\vspace{0.2cm}

\problem[4cm]{Rewrite $\dfrac{1}{2}$ as an equivalent fraction with a denominator of $8$.
What is the missing numerator?}

\problem[4cm]{Fill in the missing numerator: \qquad $\dfrac{3}{4} = \dfrac{\rule{0.7cm}{0.4pt}}{12}$}

\problem[5.5cm]{Rewrite $\dfrac{5}{6}$ and $\dfrac{3}{8}$ as equivalent fractions that both have the common denominator $24$.}

\problem[5cm]{Add. Give the answer in lowest terms. \qquad $\dfrac{1}{2} + \dfrac{1}{4}$}

\problem[5cm]{Add. Give the answer in lowest terms. \qquad $\dfrac{2}{3} + \dfrac{1}{6}$}

\problem[5cm]{Subtract. Give the answer in lowest terms. \qquad $\dfrac{3}{4} - \dfrac{1}{6}$}

\problem[5.5cm]{Find the least common denominator of $\dfrac{3}{8}$ and $\dfrac{5}{12}$, then rewrite each fraction with that denominator.}

\problem[5cm]{Add. Give the answer in lowest terms. \qquad $\dfrac{5}{8} + \dfrac{1}{3}$}

\problem[5cm]{Subtract. Give the answer in lowest terms. \qquad $\dfrac{5}{6} - \dfrac{3}{10}$}

\problem[6cm]{Add the mixed numbers. Give the answer as a mixed number in lowest terms.
\qquad $2\,\dfrac{1}{2} + 1\,\dfrac{1}{3}$}

\problem[6cm]{Subtract the mixed numbers. You will need to regroup. Give the answer as a mixed number in lowest terms.
\qquad $4\,\dfrac{3}{4} - 1\,\dfrac{5}{6}$}

\problem[6cm]{A muffin recipe uses $\dfrac{2}{3}$ of a cup of flour and $\dfrac{3}{4}$ of a cup of oats.
How many cups of dry ingredients does the recipe use in all? Give the answer as a mixed number in lowest terms.}
