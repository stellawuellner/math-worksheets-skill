\wstitleblock{Modeling Periodic Behavior from Graphs}{Precalculus}{}

\begin{center}\small\textit{Reading amplitude and midline from a graph, recovering the period, writing the model, and solving with it}\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Every model here has the form $y = A\sin(Bx) + D$ or $y = A\cos(Bx) + D$, possibly with a horizontal shift.
  \item Read maxima, minima and periods off the graph; then compute. Exact values (fractions, multiples of $\pi$) unless a unit is asked for.
  \item Amplitude is measured from the \emph{midline}, never from the horizontal axis.
\end{itemize}
\vspace{0.1cm}
