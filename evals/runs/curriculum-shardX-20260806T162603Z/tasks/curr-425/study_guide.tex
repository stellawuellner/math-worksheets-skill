\skillheading{Skill 1 --- Pythagorean identities}

\begin{formulabox}
$\sin^2\theta + \cos^2\theta = 1$, \quad so \quad $\sin^2\theta = 1 - \cos^2\theta$ \ and \ $\cos^2\theta = 1 - \sin^2\theta$.\par
Also $1 + \tan^2\theta = \sec^2\theta$ \ and \ $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.\par
Use these to turn a mixed expression into one written in a single ratio.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{1 - \sin^2\theta}{\cos\theta}$.
\step{The numerator is one of the Pythagorean rearrangements, so replacing it first turns the fraction into a cancellation.}
\step{$1 - \sin^2\theta = \cos^2\theta$, so the fraction is $\dfrac{\cos^2\theta}{\cos\theta}$.}
$\Rightarrow\quad \ans{\cos\theta}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\tan\theta\,\cos\theta$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\sin\theta}$}
\end{tryitbox}

\skillheading{Skill 2 --- Sum and difference formulas}

\begin{formulabox}
$\sin(A \pm B) = \sin A\cos B \pm \cos A\sin B$ \qquad $\cos(A \pm B) = \cos A\cos B \mp \sin A\sin B$\par
Watch the sign on cosine: it \emph{flips}. Build an unfamiliar angle out of $30^\circ$, $45^\circ$, $60^\circ$ and $90^\circ$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the exact value of $\sin 75^\circ$.
\step{Seventy-five is not a special angle, but it is a sum of two that are, so a sum formula turns it into known values.}
\step{$\sin(45^\circ + 30^\circ) = \dfrac{\sqrt{2}}{2}\cdot\dfrac{\sqrt{3}}{2} + \dfrac{\sqrt{2}}{2}\cdot\dfrac{1}{2}$}
$\Rightarrow\quad \sin 75^\circ = \ans{\dfrac{\sqrt{6} + \sqrt{2}}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the exact value of $\cos 75^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{\sqrt{6} - \sqrt{2}}{4}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Double-angle formulas}

\begin{formulabox}
$\sin 2\theta = 2\sin\theta\cos\theta$ \qquad $\cos 2\theta = \cos^2\theta - \sin^2\theta = 1 - 2\sin^2\theta = 2\cos^2\theta - 1$\par
Choose the version of $\cos 2\theta$ written in the ratio the rest of the problem already uses --- that is what leaves a quadratic in one function.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad If $\sin\theta = \dfrac{1}{3}$, find $\cos 2\theta$.
\step{Only the sine is known, so pick the form of the double angle written in sine alone and no cosine is needed at all.}
\step{$\cos 2\theta = 1 - 2\sin^2\theta = 1 - 2\left(\dfrac{1}{9}\right)$}
$\Rightarrow\quad \ans{\tfrac{7}{9}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad If $\sin\theta = \dfrac{2}{5}$, find $\cos 2\theta$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{17}{25}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Exact solution sets}

\begin{formulabox}
\textbf{Method:} reduce to one ratio, factor, set each factor to zero, and place each value in every quadrant its sign allows.\par
\textbf{Count check:} a value strictly between $-1$ and $1$ gives \emph{two} angles per turn; the extremes $1$ and $-1$ give exactly \emph{one}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $(2\sin\theta - 1)(\sin\theta + 1) = 0$ for $0^\circ \le \theta < 360^\circ$.
\step{A product is zero when a factor is zero, so this is two small equations, and the count check tells me what to expect from each.}
\step{$\sin\theta = \dfrac{1}{2}$ gives two angles; $\sin\theta = -1$ is the minimum and gives one.}
$\Rightarrow\quad \theta = \ans{30}^\circ, \ans{150}^\circ, \ans{270}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $(2\cos\theta + 1)(\cos\theta - 1) = 0$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{0}^\circ, \ans{120}^\circ, \ans{240}^\circ$}
\end{tryitbox}
