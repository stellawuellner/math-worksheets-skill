\sstitleblock{Estimating and Solving with Decimals}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Estimating by rounding first}

\begin{formulabox}
\textbf{Rule:} to estimate, round \emph{every} number first and then compute. Do not
work out the exact answer and round it at the end --- that is checking, not estimating.\par
\textbf{How to round:} look at the digit one place to the right of the place you are
rounding to. If it is $5$ or more, round up; if it is $4$ or less, round down.\par
\textbf{Why bother:} the estimate tells you roughly where the exact answer must land,
so a misplaced decimal point stands out at once.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Estimate $27.4 + 41.8$ by rounding each number to the nearest ten.
\step{The question says estimate to the nearest ten, so I round both numbers to the nearest ten before I add anything.}
\step{$27.4$ is nearer $30$ than $20$, and $41.8$ is nearer $40$ than $50$: $30 + 40 = 70$.}
$\Rightarrow\quad \ans{70}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Estimate $38.2 + 24.9$ by rounding each number to the nearest
ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{60}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Adding and subtracting decimals}

\begin{formulabox}
\textbf{Rule:} line up the decimal points, one under the other, and fill any short
number with zeros on the end --- $0.4$ and $0.40$ are the same amount.\par
\textbf{Two-step money problems:} add everything that was spent first, then subtract
that one total from the money you started with. Write a whole dollar amount as
$50.00$ so the columns line up.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Ben has \$20. He spends \$7.45 and \$3.80. How much is left?
\step{"How much is left" means subtracting, but two things were bought --- so I add the two prices into one total first, then take that away from \$20.}
\step{$7.45 + 3.80 = 11.25$, and then $20.00 - 11.25 = 8.75$.}
$\Rightarrow\quad \ans{8.75}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Ana has \$30. She spends \$12.60 and \$4.75. How much is left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12.65}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiplying and dividing decimals}

\begin{formulabox}
\textbf{Multiplying:} ignore the decimal points and multiply the digits, then count
how many digits sat after the decimal points altogether and put that many after the
point in your answer.\par
\textbf{Dividing by a whole number:} divide as usual and keep the decimal point in
line.\par
\textbf{Dividing by a decimal:} make both numbers ten (or a hundred) times bigger
first --- $4.2 \div 0.6$ is the same as $42 \div 6$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $4 \times 3.25$, and then $9.6 \div 4$.
\step{Both are exact calculations, so I multiply the digits and count decimal places for the first, and share into equal parts for the second.}
\step{$4 \times 325 = 1300$ with two decimal places gives $13.00$; and $96 \div 4 = 24$, so $9.6 \div 4 = 2.4$.}
$\Rightarrow\quad \ans{13}$ and $\ans{2.4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $5 \times 1.8$, and then $7.5 \div 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$ and $\ans{1.5}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Comparing and ordering decimals}

\begin{formulabox}
\textbf{Rule:} give every number the same number of decimal places by writing zeros
on the end, then compare them place by place from the left.\par
\textbf{Careful:} a longer decimal is not automatically bigger --- $0.58$ has more
digits than $0.6$ but is smaller, because $0.6 = 0.60$ and $60 > 58$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Which is greater, $0.6$ or $0.58$?
\step{The two numbers have different numbers of decimal places, so before comparing anything I give them the same number by writing a zero on the end.}
\step{$0.6$ becomes $0.60$; comparing hundredths, $60$ beats $58$.}
$\Rightarrow\quad \ans{0.6 > 0.58}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $<$ or $>$ between $2.05$ and $2.5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.05 < 2.5}$}
\end{tryitbox}
