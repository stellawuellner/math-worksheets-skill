\sstitleblock{Separable Equations: the three places it goes wrong}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Separating before integrating}

\begin{formulabox}
\textbf{Two questions, in this order.} (1) Does the right side factor as (something in $x$) times (something in $y$)? A sum is not a veto: $x + xy = x(1+y)$ separates, $x + y$ does not. (2) Only after every $y$ sits with $dy$ may you integrate --- and then each side is integrated in \emph{its own} variable, with the power rule dividing by the new exponent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A student integrates $\displaystyle\int 6x\,dx$ as $6x^2$. Repair the general solution of $\dfrac{dy}{dx} = 6xy$.
\step{The separation is fine, so the only thing to check is the antiderivative: the power rule raises the exponent and then divides by the new one, which the student skipped.}
\step{$\displaystyle\int \frac{dy}{y} = \int 6x\,dx$ gives $\ln|y|$ on the left and $\frac{6}{2}x^2 = 3x^2$ on the right.}

$\Rightarrow\quad \ans{\ln|y| = 3x^2 + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the general solution of $\dfrac{dy}{dx} = 10xy$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\ln|y| = 5x^2 + C}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Where the constant belongs}

\begin{formulabox}
\textbf{The constant moves when you exponentiate.} From $\ln|y| = F(x) + C$ you get $y = e^{F(x)}e^{C} = Ae^{F(x)}$: the constant \emph{multiplies}. Writing $y = e^{F(x)} + C$ is the single most common slip in this topic. For a positive starting value, $A$ is just $y$ at the starting point --- no extra algebra needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{dy}{dt} = 0.02\,y$ with $y(0) = 500$, then find $y(50)$.
\step{The equation is exponential growth, so the answer has the form $Ae^{kt}$; the job is to place the constant as a multiplier and read $A$ off the starting value.}
\step{$\ln|y| = 0.02t + C$ gives $y = Ae^{0.02t}$, and $y(0) = 500$ forces $A = 500$, so $y = 500e^{0.02t}$ and $y(50) = 500e^{1}$.}

$\Rightarrow\quad \ans{y(50) \approx 1359.141}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{dy}{dt} = 0.03\,y$ with $y(0) = 200$, then find $y(100)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y(100) \approx 4017.107}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Substituting the starting values}

\begin{formulabox}
\textbf{Substitute into the line you actually wrote.} Put the starting pair into the \emph{implicit} equation, exactly as it stands, before rearranging. If that line reads $\tfrac{1}{2}y^2 = \ldots + C$, the constant comes from $\tfrac{1}{2}y_0^2$, not from $y_0$ or $y_0^2$. Swapping the two coordinates, or reading the constant off a numerator, are the two ways this line fails.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{dy}{dx} = \dfrac{6x}{y}$ with $y(0) = 4$ and $y > 0$; find $y(3)$.
\step{The implicit form will be quadratic in $y$, so substitute the starting pair into that quadratic form and solve for the constant before taking any square root.}
\step{$y\,dy = 6x\,dx$ gives $\tfrac{1}{2}y^2 = 3x^2 + C$; at $(0, 4)$ this reads $8 = C$, so $y^2 = 6x^2 + 16$ and $y(3) = \sqrt{54 + 16}$.}

$\Rightarrow\quad \ans{y(3) \approx 8.367}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{dy}{dx} = \dfrac{8x}{y}$ with $y(0) = 3$ and $y > 0$; find $y(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y(2) \approx 6.403}$}
\end{tryitbox}
