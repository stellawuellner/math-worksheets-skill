\sstitleblock{Rewriting Circles and Parabolas in Standard Form}

\skillheading{Skill 1 --- Completing the square on a parabola}

\begin{formulabox}
\textbf{The move:} to complete $x^2 + px$, add $(p/2)^2$. Since you may not change the
equation, subtract the same amount straight back: $x^2 + px + c = (x + p/2)^2 - (p/2)^2 + c$.\par
\textbf{Target:} $y = a(x-h)^2 + k$, whose vertex is $(h, k)$ --- the whole reason for the
rewrite. Note the sign: $(x+3)^2$ means $h = -3$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $y = x^2 + 8x + 10$ in vertex form.
\step{Vertex form needs a perfect square, and the only thing standing in the way is the constant --- so build the square the $x$ terms want and repay it immediately.}
\step{Half of $8$ is $4$, so add and subtract $16$: \ $y = (x^2 + 8x + 16) - 16 + 10$.}

$\Rightarrow\quad \ans{y = (x+4)^2 - 6}$ \quad (vertex $(-4, -6)$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $y = x^2 - 6x + 2$ in vertex form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = (x-3)^2 - 7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Completing the square on a circle}

\begin{formulabox}
\textbf{Two squares, one equation.} Group the $x$ terms and the $y$ terms, complete each
square separately, and add \emph{both} completing constants to the other side.\par
\textbf{Target:} $(x-h)^2 + (y-k)^2 = r^2$, centre $(h, k)$, radius $r$. The right-hand side
is $r^2$, so square-root it before reporting a radius.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $x^2 + y^2 - 2x + 6y - 6 = 0$ in standard form.
\step{Neither variable is in a square yet, so complete both --- and move the loose constant across first so the bookkeeping stays visible.}
\step{Half of $-2$ is $-1$ (add $1$); half of $6$ is $3$ (add $9$): \ $(x^2 - 2x + 1) + (y^2 + 6y + 9) = 6 + 1 + 9$.}

$\Rightarrow\quad \ans{(x-1)^2 + (y+3)^2 = 16}$ \quad (centre $(1,-3)$, radius $4$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $x^2 + y^2 + 4x - 8y + 4 = 0$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+2)^2 + (y-4)^2 = 16}$}
\end{tryitbox}

\skillheading{Skill 3 --- Harder squares, and the feature you read off}

\begin{formulabox}
\textbf{Factor before you complete.} For $y = ax^2 + bx + c$, pull $a$ out of the terms
containing $x$ only: $y = a(x^2 + \tfrac{b}{a}x) + c$. Complete the square inside the bracket,
then multiply the correction back out by $a$ as it leaves.\par
\textbf{For a circle} the leading coefficients on $x^2$ and $y^2$ are equal, so divide the
whole equation by that number first and the problem becomes Skill 2.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $y = 2x^2 - 8x + 3$ in vertex form, and give the vertex height.
\step{The square cannot be completed while a $2$ multiplies $x^2$, so factor it out of the $x$ terms first and leave the constant outside untouched.}
\step{$y = 2(x^2 - 4x) + 3 = 2\bigl((x-2)^2 - 4\bigr) + 3 = 2(x-2)^2 - 8 + 3$}
\step{Vertex height: at $x = 2$, $\ y = 2(4) - 8(2) + 3 = 8 - 16 + 3$.}

$\Rightarrow\quad \ans{y = 2(x-2)^2 - 5}, \qquad$ vertex height $\ans{-5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $y = 3x^2 - 6x + 4$ in vertex form and give the vertex height.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 3(x-1)^2 + 1}$, height $\ans{1}$}
\end{tryitbox}

\skillheading{Skill 4 --- Building the equation from features}

\begin{formulabox}
\textbf{Standard form is a template.} Given centre $(h,k)$ and radius $r$, write
$(x-h)^2 + (y-k)^2 = r^2$ --- watch the sign flip, a centre of $-3$ gives $(x+3)$.\par
\textbf{General form} comes from multiplying out and collecting:
$x^2 + y^2 + Dx + Ey + F = 0$. Expanding is the reverse of completing the square, so it is
also the fastest way to check your work.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write the circle with centre $(2,-3)$ and radius $5$ in general form.
\step{Fill the template first, then expand --- going straight to general form invites a sign slip on the centre.}
\step{$(x-2)^2 + (y+3)^2 = 25$, and expanding gives $x^2 - 4x + 4 + y^2 + 6y + 9 - 25 = 0$.}

$\Rightarrow\quad \ans{x^2 + y^2 - 4x + 6y - 12 = 0}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write the circle with centre $(-1, 4)$ and radius $3$ in general form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + y^2 + 2x - 8y + 8 = 0}$}
\end{tryitbox}
