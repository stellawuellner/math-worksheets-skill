{\small\textit{Factoring cubes and higher-degree expressions}}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Three patterns to recognise}

\vspace{0.1cm}
\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{What you see} & \textbf{Pattern} & \textbf{Factored form} \\ \hline
Two terms, both cubes
  & $a^{3} - b^{3}$
  & $(a - b)(a^{2} + ab + b^{2})$ \\ \hline
Two terms, both cubes
  & $a^{3} + b^{3}$
  & $(a + b)(a^{2} - ab + b^{2})$ \\ \hline
Four terms
  & group in pairs
  & pull a common factor from each pair \\ \hline
$x^{4}, x^{2}$, constant
  & quadratic in $x^{2}$
  & factor as if $x^{2}$ were a single letter \\ \hline
\end{tabular}
\end{center}

\vspace{0.1cm}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item Take out a common factor \emph{first}, every time.
  \item \emph{Factor completely}: keep going until no factor can be broken down
    further over the real numbers.
  \item The quadratic left over from a cube pattern
    ($a^{2} \pm ab + b^{2}$) does not factor again --- that is expected, not a
    dead end.
\end{itemize}
