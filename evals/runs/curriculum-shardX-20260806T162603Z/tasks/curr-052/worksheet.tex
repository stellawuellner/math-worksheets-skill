\wstitleblock{Numbers Three Ways}{Grade 2 Math}{}
\begin{center}
\textit{standard form, word form, and expanded form}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item A number can be written three ways. Here is 234 written every way:
  \item \textbf{Standard form:} 234 \quad
        \textbf{Expanded form:} $234 = 200 + 30 + 4$ \quad
        \textbf{Word form:} two hundred thirty-four
  \item Follow the pattern. Each problem asks for one of the three forms.
\end{itemize}

\problem[2.6cm]{Write \textbf{46} in expanded form.}

\problem[2.6cm]{Write the number in standard form: \quad $300 + 20 + 9$}

\problem[2.6cm]{Write \textbf{573} in expanded form.}

\problem[2.5cm]{Write \textbf{285} in word form.\par\vspace{10pt}
\noindent\underline{\hspace{0.95\linewidth}}\noansline}

\problem[2.6cm]{Write the number in standard form: \quad \textbf{six hundred forty-one}}

\problem[3cm]{Write \textbf{705} in expanded form. Be careful --- one place is empty!}

\problem[2.5cm]{Write \textbf{460} in word form.\par\vspace{10pt}
\noindent\underline{\hspace{0.95\linewidth}}\noansline}

\problem[3cm]{Write the number in standard form: \quad $9 + 800 + 30$\par
\emph{Hint: the parts are not in place-value order.}}

\problem[3cm]{Write the number in standard form: \quad \textbf{nine hundred seven}}

\problem{\textbf{Challenge.} Fill in the missing part of the expanded form:
\par\vspace{8pt}
\hspace*{1.5cm} $800 \;+\;$ \ansblank\ $\;+\; 2 \;=\; 862$
\par\vspace{8pt}
Then, in your work space, explain how you know which place the missing part
comes from.\par\vspace*{3cm}}
