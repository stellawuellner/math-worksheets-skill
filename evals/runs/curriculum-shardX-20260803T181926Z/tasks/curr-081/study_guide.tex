\sstitleblock{Unit Fractions in Area Models}

\skillheading{1. Count the equal parts, then name the shaded one}

\begin{formulabox}
\textbf{Bottom number:} how many \emph{equal parts} the whole shape is cut
into. \quad \textbf{Top number:} how many parts are shaded.\\[3pt]
When exactly one part is shaded, the top number is $1$ and the fraction is
called a \textbf{unit fraction}: $\frac{1}{2}$, $\frac{1}{3}$, $\frac{1}{4}$,
$\frac{1}{5}$, \ldots\ Count the \emph{parts}, never the cut lines --- three
lines across a bar make four parts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Name the shaded part.
\begin{center}
\begin{tikzpicture}[x=0.8cm,y=0.8cm]
  \fill[blue!25] (0,0) rectangle (1,1.1);
  \draw[thick] (0,0) rectangle (5,1.1);
  \foreach \i in {1,2,3,4} \draw[thick] (\i,0) -- (\i,1.1);
\end{tikzpicture}
\end{center}
\step{One part is shaded, so this will be a unit fraction --- the only thing to
work out is the bottom number.}
\step{Touch and count the equal parts across the bar: one, two, three, four,
five.}

One of five equal parts: $\ans{\tfrac{1}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A circle is cut into $6$ equal slices and one slice is
shaded. What fraction is the shaded slice?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{6}}$}
\end{tryitbox}

\skillheading{2. First check: are the parts really the same size?}

\begin{formulabox}
A fraction name only fits when the whole is cut into parts of the \emph{same
size}. Pieces that are all different are still pieces of the whole --- they
just have no fraction name yet.\\[3pt]
So look before you count: \emph{equal parts?} then \emph{how many?}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which bar's shaded piece can be named?
\begin{center}
\begin{tikzpicture}[x=0.8cm,y=0.8cm]
  \fill[blue!25] (0,0) rectangle (1.25,1);
  \draw[thick] (0,0) rectangle (5,1);
  \foreach \i in {1.25,2.5,3.75} \draw[thick] (\i,0) -- (\i,1);
  \begin{scope}[yshift=-1.6cm]
  \fill[blue!25] (0,0) rectangle (0.6,1);
  \draw[thick] (0,0) rectangle (5,1);
  \foreach \i in {0.6,2.3,3.1} \draw[thick] (\i,0) -- (\i,1);
  \end{scope}
\end{tikzpicture}
\end{center}
\step{Compare the pieces inside each bar first --- only the bar with equal
pieces can carry a fraction name.}
\step{The top bar has four pieces all the same width; the bottom bar's four
pieces are all different, so it is out.}

Top bar, one of four equal parts: $\ans{\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A square is cut into $3$ strips of exactly the same size
and one strip is shaded. What fraction is shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$}
\end{tryitbox}

\skillheading{3. More cuts make smaller pieces}

\begin{formulabox}
Two unit fractions of the \emph{same whole}: the one with the bigger bottom
number is the \emph{smaller} piece, because the whole was chopped into more
pieces.\\[2pt]
$>$ means \emph{is greater than} \qquad $<$ means \emph{is less than}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which shaded piece is bigger?
\begin{center}
\begin{tikzpicture}[x=0.8cm,y=0.8cm]
  \fill[blue!25] (0,0) rectangle (2.5,1);
  \draw[thick] (0,0) rectangle (5,1);
  \draw[thick] (2.5,0) -- (2.5,1);
  \begin{scope}[yshift=-1.6cm]
  \fill[blue!25] (0,0) rectangle (1,1);
  \draw[thick] (0,0) rectangle (5,1);
  \foreach \i in {1,2,3,4} \draw[thick] (\i,0) -- (\i,1);
  \end{scope}
\end{tikzpicture}
\end{center}
\step{Both bars are the same length and both have one part shaded, so only the
size of one part decides this.}
\step{Two cuts of the whole give a piece half the bar; five cuts give a much
narrower piece.}

The half is bigger: $\frac{1}{2}$ $\ans{>}$ $\frac{1}{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two same-size pizzas: one cut into $6$ equal slices, one
cut into $3$. Which sign makes $\frac{1}{6}$ \underline{\hspace{1cm}}
$\frac{1}{3}$ true?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} count the \emph{pieces}, not the cut lines --- one line
down the middle of a bar makes two parts, not one. And a bigger bottom number
never means a bigger piece: $\frac{1}{8}$ is smaller than $\frac{1}{2}$, even
though $8$ is bigger than $2$.
\end{watchoutbox}
