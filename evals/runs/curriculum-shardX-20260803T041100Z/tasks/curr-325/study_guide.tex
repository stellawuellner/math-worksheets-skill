\sstitleblock{Choosing a Right-Triangle Method}

\noindent\small \textbf{The decision rule.} Look at what is GIVEN, not at what
is asked. Angle given or wanted? If no, Pythagoras. If yes and the angle is
$30^\circ$, $45^\circ$ or $60^\circ$, a special triangle. Otherwise a trig
ratio.\normalsize

\skillheading{1. Pythagorean theorem: two sides, no angle}

\begin{formulabox}
\textbf{Rule:} in a right triangle with the right angle at $C$,
\[ a^2 + b^2 = c^2 \qquad (c \text{ is always the hypotenuse}) \]
\textbf{Use it when} two sides are known and the third is wanted, and no angle
appears anywhere. \textbf{Missing leg:} rearrange to $b^2 = c^2 - a^2$ ---
subtract, never add.\\[3pt]
\textbf{Sanity check:} the hypotenuse must come out largest. Worth memorising:
$3$-$4$-$5$, $5$-$12$-$13$, $8$-$15$-$17$ and their multiples.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The legs of a right triangle are $9$ and $12$. Find the
hypotenuse.
\step{Two sides are given and no angle appears, so a trig ratio has nothing to work with --- Pythagoras is the only tool that fits.}
\step{Both given sides are legs, so add: $9^2 + 12^2 = 81 + 144 = 225$, and the hypotenuse is $\sqrt{225}$.}
$c = \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A right triangle has hypotenuse $20$ and one leg $16$.
Find the other leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{2. Special triangles: 45-45-90 and 30-60-90}

\begin{formulabox}
\textbf{Two ratios worth knowing exactly:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=1pt]
  \item $45^\circ$-$45^\circ$-$90^\circ$: \; leg $:$ leg $:$ hypotenuse
        $= 1 : 1 : \sqrt{2}$. \; Hypotenuse $=$ leg $\times \sqrt{2}$.
  \item $30^\circ$-$60^\circ$-$90^\circ$: \; short leg $:$ long leg $:$
        hypotenuse $= 1 : \sqrt{3} : 2$. The short leg is \emph{opposite} the
        $30^\circ$ angle and is half the hypotenuse.
\end{itemize}
\textbf{Use it when} an acute angle is $30^\circ$, $45^\circ$ or $60^\circ$ ---
you get an exact answer with no calculator. A square's diagonal and an
equilateral triangle's altitude always create these.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad An isosceles right triangle has legs of $8$. Find the
hypotenuse.
\step{Isosceles and right forces the two acute angles to be equal, which is the special ratio case --- exact, no calculator work needed.}
\step{Hypotenuse $=$ leg $\times \sqrt{2} = 8\sqrt{2}$, which is exact; as a decimal it is $8 \times 1.41$.}
$c = 8\sqrt{2} \approx \ans{11.31}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A right triangle has hypotenuse $14$ and an acute
angle of $30^\circ$. Find the side opposite that angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{3. Trig ratios: when an angle is in play}

\begin{formulabox}
\textbf{SOH-CAH-TOA}, always taken \emph{relative to the angle you are using}:
\[ \sin A = \frac{\text{opp}}{\text{hyp}}, \qquad
   \cos A = \frac{\text{adj}}{\text{hyp}}, \qquad
   \tan A = \frac{\text{opp}}{\text{adj}} \]
\textbf{Side missing:} pick the ratio containing the two sides you care about
(one known, one wanted), then multiply or divide.\\[3pt]
\textbf{Angle missing:} form the same ratio from the two known sides, then
apply the inverse. Example: adjacent $6$, hypotenuse $10$ gives
$\cos A = 0.6$, so $A = \cos^{-1}(0.6) \approx 53.13^\circ$.\\[3pt]
Choosing the ratio is the whole skill: the one you want is the one whose two
sides are the two you have.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A right triangle has hypotenuse $17$ and an acute angle
of $28^\circ$. Find the side opposite that angle.
\step{The angle is not one of the special ones, so the exact ratios are out; the wanted side is opposite the known angle and the known side is the hypotenuse, and that pairing is sine.}
\step{$\sin 28^\circ = \dfrac{\text{opp}}{17}$, so the opposite side is $17\sin 28^\circ$.}
opp $= \ans{7.98}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A right triangle has hypotenuse $24$ and an acute angle
of $41^\circ$. Find the side \emph{adjacent} to that angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18.11}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``opposite'' and ``adjacent'' are named relative to the
angle you are using, so they swap when you switch angles --- the hypotenuse
never does. And if the missing quantity is an angle, the answer comes out of an
inverse ratio; forgetting the inverse returns a ratio between $-1$ and $1$
instead of a number of degrees.
\end{watchoutbox}
