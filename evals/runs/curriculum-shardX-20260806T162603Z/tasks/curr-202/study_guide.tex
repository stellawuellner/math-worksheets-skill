% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Evaluating perfect square roots}

\begin{formulabox}
\textbf{Rule:} $\sqrt{N} = s$ exactly when $s \times s = N$ (and $s$ is not negative).
\par\smallskip
\textbf{Perfect squares worth knowing by sight:}
\[ 1,\; 4,\; 9,\; 16,\; 25,\; 36,\; 49,\; 64,\; 81,\; 100,\; 121,\; 144,\; 169,\; 196,\; 225 \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\sqrt{196}$.
\step{The radicand is a perfect square, so I hunt for the factor that repeats twice --- no estimating needed.}
\step{$10 \times 10 = 100$ is too small and $20 \times 20 = 400$ is too big, so try the teens: $14 \times 14 = 196$.}

$\Rightarrow\quad \sqrt{196} = \ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\sqrt{121}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Evaluating perfect cube roots}

\begin{formulabox}
\textbf{Rule:} $\sqrt[3]{N} = s$ exactly when $s \times s \times s = N$.
\par\smallskip
\textbf{Perfect cubes worth knowing by sight:}
$1, 8, 27, 64, 125, 216, 343, 512, 729, 1000$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\sqrt[3]{343}$.
\step{A cube root asks for the factor used three times, so I test cubes rather than squares.}
\step{$6 \times 6 \times 6 = 216$ is too small; $7 \times 7 \times 7 = 343$ lands exactly.}

$\Rightarrow\quad \sqrt[3]{343} = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\sqrt[3]{1000}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{64} = 8$ but $\sqrt[3]{64} = 4$. The same radicand has a
different square root and cube root --- always read the little index first.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Roots inside a larger expression}

\begin{formulabox}
\textbf{Rule:} a root is a grouping symbol. Evaluate every root first, then do the
multiplying, then the adding and subtracting. A number written in front of a root
multiplies the \emph{value} of the root --- it never moves inside.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $2\sqrt{36} + \sqrt[3]{27}$.
\step{Two roots and a coefficient: order of operations says roots first, then the multiplication, then the sum.}
\step{$\sqrt{36} = 6$ and $2 \times 6 = 12$; \ $\sqrt[3]{27} = 3$; \ $12 + 3 = 15$.}

$\Rightarrow\quad 2\sqrt{36} + \sqrt[3]{27} = \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\sqrt{400} - 3\sqrt[3]{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}
