\sstitleblock{Rounding to Named Places}

\skillheading{Skill 1 --- Rounding on a Number Line}

\begin{formulabox}
\textbf{Model:} draw the two nearest ``round'' numbers at the ends, mark the
\textbf{midpoint}, and place your number. Whichever end it is closer to is the
rounded value. (Exactly at the midpoint: round up.)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round 276 to the nearest hundred.
\step{276 sits between 200 and 300, so mark the midpoint 250 and see which side
276 is on.}
\step{$276 > 250$, so 276 is closer to 300.}
$\Rightarrow\quad \ans{300}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Use a number line to round 43 to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40}$}
\end{tryitbox}

\skillheading{Skill 2 --- The Digit Rule}

\begin{formulabox}
\textbf{Rule:} underline the digit in the named place, then look \textbf{one
place to the right}: 0--4 keeps the underlined digit, 5--9 rounds it up.
Everything to the right of the named place becomes zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $8{,}624$ to the nearest thousand.
\step{Underline the thousands digit 8, then check the digit to its right: the
hundreds digit is 6.}
\step{$6 \geq 5$, so the 8 rounds up to 9 and the rest zero out:
$8{,}624 \rightarrow 9{,}000$}
$\Rightarrow\quad \ans{9{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $8{,}624$ to the nearest \emph{hundred}.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8{,}600}$}
\end{tryitbox}

\skillheading{Skill 3 --- Rounding in Real Contexts}

\begin{formulabox}
\textbf{When rounding shows up in life:} reported attendance, poster totals,
news headlines. The context names the place (``to the nearest thousand'') ---
find that place, apply the digit rule, and keep the units in your answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A city counts $31{,}865$ bus riders. A report gives the
count to the nearest thousand. What number is reported?
\step{The named place is thousands: underline the 1; the digit to its right is
8.}
\step{$8 \geq 5$, so 31 thousands become 32 thousands:
$31{,}865 \rightarrow 32{,}000$ riders}
$\Rightarrow\quad \ans{32{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A school raises $4{,}712$ dollars. What is the total to
the nearest hundred dollars?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4{,}700}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} rounding can carry. Rounding $3{,}997$ to the nearest
ten gives $4{,}000$ --- the 99 tens become 100 tens, and the carry changes
three digits at once. Never round just the underlined digit by itself.
\end{watchoutbox}
