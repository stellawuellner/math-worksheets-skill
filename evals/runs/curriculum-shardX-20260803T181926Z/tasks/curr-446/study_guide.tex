\sstitleblock{Center, Spread, and Shape}

\skillheading{Skill 1 --- Center: mean or median?}

\begin{formulabox}
\textbf{Mean} $=$ total $\div$ count. It is a balance point, so every observation pulls on
it by its distance from the centre --- one extreme value moves it.\par
\smallskip
\textbf{Median} $=$ the middle of the ordered list (average the middle two when the count is
even). It is a counting point, so it does not care how far away the extremes are.\par
\smallskip
\textbf{Rule:} symmetric data $\Rightarrow$ report the mean. Skewed data or an outlier
$\Rightarrow$ report the median, which is \emph{resistant}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Report the centre of $3,\ 5,\ 6,\ 8,\ 9,\ 11,\ 40$.
\step{One value sits far above the rest, so the data is not symmetric --- that makes the
median the summary to report, not the mean.}
\step{Seven values, so the middle is the 4th: $3, 5, 6, \mathbf{8}, 9, 11, 40$. (The mean
would be $11.71$, higher than six of the seven values.)}

$\Rightarrow\quad$ median $= \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Report the centre of $12,\ 14,\ 15,\ 17,\ 20,\ 22,\ 60$.\\[2pt]
\rotatebox{180}{\footnotesize check: median $= \ans{17}$}
\end{tryitbox}

\skillheading{Skill 2 --- Spread: range or IQR?}

\begin{formulabox}
\textbf{Range} $=$ max $-$ min. Only two observations, both extreme, so a single outlier can
double it.\par
\smallskip
\textbf{IQR} $= Q_3 - Q_1$, the width of the middle half. Split the ordered data at the
median; $Q_1$ is the median of the lower half and $Q_3$ the median of the upper half.\par
\smallskip
\textbf{Rule:} the IQR travels with the median and is resistant; the range and the standard
deviation travel with the mean and are not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the IQR of $4,\ 6,\ 7,\ 9,\ 10,\ 12,\ 13,\ 30$.
\step{Spread beside a resistant centre means the IQR, so split the eight ordered values into
two halves of four and take the median of each half.}
\step{Lower half $4, 6, 7, 9$ gives $Q_1 = 6.5$; upper half $10, 12, 13, 30$ gives
$Q_3 = 12.5$; so $Q_3 - Q_1 = 12.5 - 6.5$.}

$\Rightarrow\quad$ IQR $= \ans{6}$ \quad (the range is $26$ --- more than four times as big)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the IQR of $10,\ 12,\ 14,\ 18,\ 20,\ 24,\ 26,\ 70$.\\[2pt]
\rotatebox{180}{\footnotesize check: IQR $= \ans{12}$}
\end{tryitbox}

\skillheading{Skill 3 --- Shape: read it off the two centres}

\begin{formulabox}
\textbf{Shape} is \textbf{symmetric}, \textbf{skewed right} (long tail toward the high
values), or \textbf{skewed left} (long tail toward the low values).\par
\smallskip
\textbf{The tail pulls the mean and leaves the median:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item mean $>$ median $\Rightarrow$ skewed right
  \item mean $<$ median $\Rightarrow$ skewed left
  \item mean $=$ median $\Rightarrow$ symmetric
\end{itemize}
On a histogram, name the side the \emph{tail} is on, not the side the peak is on.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A data set has mean $14$ and median $9$. Name the shape.
\step{The two centres respond differently to a tail, so compare them rather than trying to
picture the data.}
\step{The mean sits above the median, and only far-off high values can lift a balance point
away from a counting point.}

$\Rightarrow\quad \ans{14 > 9}$, so the distribution is \textbf{skewed right}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A data set has mean $31$ and median $40$. Name the shape.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{31 < 40}$, skewed left}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} shape is decided \emph{first}. Centre and spread are a matched pair
chosen from it --- mean with standard deviation, or median with IQR. Reporting a mean
alongside an IQR, or quoting a mean for visibly skewed data, is the error that costs marks
even when every calculation is right.
\end{watchoutbox}
