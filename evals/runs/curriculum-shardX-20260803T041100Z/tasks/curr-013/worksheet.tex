\wstitleblock{Ordering Three Amounts from Least to Greatest}{Kindergarten--Grade 1}{}

\noindent\small\textbf{Directions}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item First find all three amounts. Count the picture or work out the number, and write each one down in the work space.
  \item Then write the three numbers in the blanks: the \textbf{least} (smallest) amount first, the \textbf{greatest} (largest) amount last.
\end{itemize}
\normalsize
