\wstitleblock{Diagnosing Index-Shift and Convergence-Condition Errors}{Precalculus}{}

\begin{center}
\small Find the mistake, compute the correct value, and name what went wrong.
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Each problem shows one student's work with exactly one thing wrong in it.
  \item Part (a) asks for the correct value. Part (b) asks you to name the error in a sentence or two.
  \item Give exact values --- fractions, not decimals --- unless a problem says otherwise.
\end{itemize}

\vspace{0.15cm}

\problem[3.5cm]{A student evaluates $\displaystyle\sum_{k=2}^{6} k$ by using the formula
$\dfrac{n(n+1)}{2}$ with $n = 6$ and writes $21$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct value of the sum. \quad \ansblank
  \item[(b)] Name the error in the student's method.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{A student rewrites $\displaystyle\sum_{n=1}^{4} 3^{\,n}$ as
$\displaystyle\sum_{m=0}^{4} 3^{\,m+1}$ and evaluates the new sum as $363$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct value of $\displaystyle\sum_{n=1}^{4} 3^{\,n}$. \quad \ansblank
  \item[(b)] Name the error in the rewriting.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{A student applies $\dfrac{a}{1-r}$ to
$\displaystyle\sum_{n=0}^{\infty} \left(\tfrac{3}{2}\right)^{n}$ and reports a sum of $-2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Evaluate $\displaystyle\sum_{n=0}^{\infty} \left(\tfrac{2}{3}\right)^{n}$, which the formula \emph{does} apply to. \quad \ansblank
  \item[(b)] Explain why the same formula cannot be used on the student's series.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{A student says
$\displaystyle\lim_{n \to \infty} \frac{3n + 2}{n - 5} = 1$, reasoning that ``the $2$ and the
$-5$ disappear and the $n$'s cancel''.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct limit. \quad \ansblank
  \item[(b)] Name the error in the reasoning.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{To rewrite $\displaystyle\sum_{k=3}^{7} (k-2)^2$ with a new index
$j = k - 2$, a student writes $\displaystyle\sum_{j=1}^{7} j^{\,2}$ and gets $140$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct value of $\displaystyle\sum_{k=3}^{7} (k-2)^2$. \quad \ansblank
  \item[(b)] Name the error, and say how many terms the sum should contain.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{A student evaluates
$\displaystyle\sum_{n=1}^{\infty} 5\left(\tfrac{1}{4}\right)^{n-1}$ as
$\dfrac{5/4}{1 - 1/4} \approx 1.67$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct sum as an exact fraction. \quad \ansblank
  \item[(b)] Name the error in the value used for $a$.
\end{itemize}
\vspace*{2.4cm}}

\problem[3.5cm]{A student claims the sequence $a_n = (-1)^n \dfrac{n}{n+1}$ converges to $1$,
because $\dfrac{n}{n+1} \to 1$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $a_{10}$ and $a_{11}$ as exact fractions.
        \quad $a_{10} = $~\ansblank \quad $a_{11} = $~\ansblank
  \item[(b)] Use those two values to name the error in the student's claim.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{A student evaluates
$\displaystyle\sum_{n=2}^{\infty} 3\left(\tfrac{1}{2}\right)^{n}$ as
$\dfrac{3}{1 - 1/2} = 6$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the correct sum as an exact fraction. \quad \ansblank
  \item[(b)] Name the error, and state what $a$ should have been.
\end{itemize}
