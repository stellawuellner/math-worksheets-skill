% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Adding and Subtracting Polynomials}

\begin{formulabox}
\textbf{Rule:} only \textit{like} terms combine --- same variable, same
exponent. $ax^{n} \pm bx^{n} = (a \pm b)x^{n}$, and $x^{3}$ never combines with
$x^{2}$.\\[3pt]
\textbf{Subtraction:} rewrite $-(\,\cdots)$ by changing the sign of
\textit{every} term inside, then add. Answers go in standard form, highest power
first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(4x^{3} - x + 2) + (2x^{3} + 5x^{2} - 6)$
\step{This is an addition, so no signs change --- the whole job is sorting terms by degree and combining the matching ones.}
\step{Cubes: $4 + 2$ gives $6x^{3}$. There is only one square term, $5x^{2}$, and only one linear term, $-x$. Constants: $2 - 6$.}

$\Rightarrow\quad \ans{6x^3 + 5x^2 - x - 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(7x^{2} - 3x) - (2x^{2} + 4x - 5)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x^2 - 7x + 5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The Box Model for a Product}

\begin{formulabox}
\textbf{Model:} a product is the area of a rectangle whose sides are split into
the terms. One row per term of the left factor, one column per term of the top
factor; each cell is one partial product, and the answer is the sum of the
cells.\\[3pt]
\textbf{Count check:} $m$ terms times $n$ terms means exactly $mn$ cells. If you
have fewer, one is missing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(x + 5)(x^{2} - 2x + 3)$
\step{Two terms times three terms, so the box has six cells --- fill all six before collecting anything.}
\step{From $x$: $x^{3}$, $-2x^{2}$, $3x$. From $5$: $5x^{2}$, $-10x$, $15$. Now the squares give $-2 + 5$ and the linear terms give $3 - 10$.}

$\Rightarrow\quad \ans{x^3 + 3x^2 - 7x + 15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(x - 3)(x^{2} + 4x - 1)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^3 + x^2 - 13x + 3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiply, Then Check by Substituting}

\begin{formulabox}
\textbf{Degree check:} the degree of a product is the \textit{sum} of the
degrees of the factors, and the leading coefficient is the product of the
leading coefficients. Read those off first and see whether your answer matches.\\[3pt]
\textbf{Number check:} pick a value of $x$, put it into the original factors and
into your expanded answer. Two different numbers means an error.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(2x - 1)(3x^{2} + x - 4)$
\step{Degree $1$ times degree $2$ must give degree $3$, with leading coefficient $2 \cdot 3$ --- so expect the answer to start with $6x^{3}$, and use that as the first check.}
\step{From $2x$: $6x^{3}$, $2x^{2}$, $-8x$. From $-1$: $-3x^{2}$, $-x$, $4$. Collecting, the squares give $2 - 3$ and the linear terms give $-8 - 1$.}

$\Rightarrow\quad \ans{6x^3 - x^2 - 9x + 4}$, and at $x = 1$ both sides give $0$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(3x + 2)(x^{2} - x + 5)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^3 - x^2 + 13x + 10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the most common slip is a missing cell --- a three-term
factor times a two-term factor needs six partial products, not five. The second
most common is a subtraction sign that reached only the first term inside the
bracket.
\end{watchoutbox}
