\wstitleblock{Groups, Arrays, and Products}{Grades 2--3 \quad Operations and Algebraic Thinking}{}

\begin{center}
\textit{Turning repeated addition into a multiplication, reading an array as
rows of equal size, writing a product back as repeated addition, and finding a
missing factor}
\end{center}

\noindent\textbf{Three ways to show the same amount.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Equal groups added again and again:} $5 + 5 + 5$.
\item \textbf{An array} --- rows that all hold the same number of things.
\item \textbf{A multiplication:} $3 \times 5$ means \emph{3 groups of 5}.
\end{itemize}
They all give the same total. Show how you found each answer.

\vspace{0.2cm}

\problem[3.5cm]{Ana counts her stickers by fives:
\begin{center}\Large $5 + 5 + 5 + 5$\end{center}
Write this as a multiplication, then find the total.}

matches it, then give the total.

\problem[3.5cm]{Write $7 + 7 + 7$ as a multiplication, then find the total.}

\begin{center}\Large

Write the multiplication for the rows, and then the multiplication you get by
counting the columns instead. Give the total.}

\problem[4cm]{Marco has $6$ bags with $3$ marbles in each bag. Write $6 \times 3$
as a repeated addition, then find how many marbles he has.}

\problem[3.5cm]{Fill in the missing factor, then say how you know.
\begin{center}\Large $4 \times \underline{\hspace{1.2cm}} = 24$\end{center}}

$\circ$ \; $\circ$ \; $\circ$ \; $\circ$ \\[0.25cm]

$\circ$ \; $\circ$ \; $\circ$ \; $\circ$ \\[0.25cm]
$\circ$ \; $\circ$ \; $\circ$ \; $\circ$

Write the multiplication and find how many muffins are on the tray.}

\problem[4cm]{Nia jumps along a number line, $6$ at a time:
\begin{center}\Large $6 + 6 + 6 + 6 + 6$\end{center}
Write the multiplication that matches her jumps, and find where she lands.}

\problem[3.5cm]{Fill in the missing factor.
\begin{center}\Large $\underline{\hspace{1.2cm}} \times 5 = 35$\end{center}
Then finish this sentence: that many groups of $5$ makes $35$.}

\problem[4.5cm]{Ben was asked for $4 \times 6$. He wrote $4 + 6 = 10$.
Explain what Ben mixed up, write $4 \times 6$ as a repeated addition, and give
the correct product.}

\problem[5cm]{An array has $4$ rows with $6$ tiles in each row. Sam adds
\emph{one more row} of $6$ tiles.

(a) How many rows are there now? \quad (b) How many tiles are there
altogether? Show the multiplication you used.}

\problem[6cm]{\textbf{Challenge.} Kai has $32$ tiles. He puts them in an array
with $8$ tiles in each row.

(a) How many rows does he make? \quad (b) Write the repeated addition that
matches that array. \quad (c) Kai then rebuilds the array with only $4$ tiles
in each row. How many rows does he need now?}
