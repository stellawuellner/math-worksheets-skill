\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]

  \item Answer the number part first, then write the explanation in full sentences.
\end{itemize}

\problem[3cm]{Ravi was asked to plot the point $(2, 5)$ on the Grid. His mark landed on point $B$ instead.
\begin{itemize}
  \item[(a)] Write the ordered pair for point $B$: \quad $($~\ansblank~$,$~\ansblank~$)$
  \item[(b)] Explain the mistake Ravi made.
\end{itemize}

\problem[3cm]{Use the Book Graph. Leo says Thursday shows 7 books, because the bar reaches the 7th mark above zero. How many books does Thursday actually show?}

\problem[3cm]{Use the Book Graph. Sam counted the marks on the Wednesday bar and reported 2 books.
\begin{itemize}
  \item[(a)] How many books does Wednesday actually show? \ansline
  \item[(b)] Explain Sam's mistake.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]

\problem[3cm]{On the Grid, points $C$ and $D$ are a reversed pair: one of them is $(4, 7)$ and the other is $(7, 4)$. Measured left to right, how far apart are they?
\answerline{units}}

\problem[3cm]{Use the Book Graph. Mia counted the marks on each bar and added those counts, reporting 21 books for the week. How many books were actually read that week?}

\problem[3.2cm]{The same book data is redrawn on a new graph whose vertical axis is marked every 2 books instead of every 5. A student looks at the new Friday bar and says ``it is taller, so more books were read on Friday''.
\begin{itemize}
  \item[(a)] On the new axis, how many marks tall is the Friday bar? \ansline
  \item[(b)] Were more books read on Friday? Explain.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3.4cm]{Use the Book Graph. A student found the mean by averaging the mark counts and reported 4.2 books per day.
\begin{itemize}
  \item[(a)] Write the actual mean number of books per day. \ansline
  \item[(b)] Write the range of the five daily totals. \ansline
\end{itemize}}

\problem[3.6cm]{A student is told to plot $(3, 30)$ on a graph whose horizontal axis is marked every 1 unit and whose vertical axis is marked every 5 units. The student counts 3 marks across and 30 marks up.
\begin{itemize}
  \item[(a)] How many marks up should the student have counted? \ansline
  \item[(b)] Counting 30 marks up, what $y$-value did the mark really land on? \ansline
  \item[(c)] Explain the two separate things that went wrong.
\end{itemize}
