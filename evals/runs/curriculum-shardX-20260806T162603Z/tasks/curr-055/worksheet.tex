\wstitleblock{Rounding and Ordering on the Number Line}{Grade 3 Math}{}
\begin{center}
\textit{rounding to the nearest ten, rounding to the nearest hundred, and
ordering on a number line}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item To round, find your number on the line and ask: which labeled end is
        it closer to? The taller middle mark shows the halfway point ---
        halfway rounds \emph{up}.
  \item Some number lines are left blank so you can mark numbers yourself.
\end{itemize}
