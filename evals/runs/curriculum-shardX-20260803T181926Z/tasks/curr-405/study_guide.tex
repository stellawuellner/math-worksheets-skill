\sstitleblock{Transformations, Compositions and Inverses Together}

\begin{formulabox}
\textbf{One idea, three faces.} A transformation is $f$ composed with a simple
linear map; an inverse is the composition that undoes one.\\[3pt]
\textbf{Inside changes $x$, and lies about direction:} $f(x-h)$ moves right by
$h$; $f(kx)$ compresses horizontally by $1/k$; $f(-x)$ reflects in the
$y$-axis.\\[3pt]
\textbf{Outside changes $y$, and means what it says:} $af(x)+c$ stretches
vertically by $a$ and shifts up by $c$; $-f(x)$ reflects in the $x$-axis.\\[3pt]
\textbf{Order matters} when a stretch and a vertical shift meet: stretch first,
then shift.
\end{formulabox}

\skillheading{1. Transform a rule}

\begin{formulabox}
Substitute the inside change for every $x$ in the rule, then apply the outside
changes to the whole result. Read the direction from the position, never from
the sign alone: $f(x+1)$ moves \emph{left} $1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x)=x^2$. Write $g(x)=f(x+1)-4$ as an expanded polynomial.
\step{The $+1$ sits inside $f$, so it is a horizontal move (left $1$), and the
$-4$ sits outside, so it lowers the whole graph by $4$.}
\step{Substituting: $g(x)=(x+1)^2-4=x^2+2x+1-4$.}

Collecting, $g(x)=\ans{x^2+2x-3}$ --- the parabola moved left $1$ and down $4$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x)=x^2$. Write $h(x)=2f(x-3)$ as an expanded
polynomial.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x^2-12x+18}$}
\end{tryitbox}

\skillheading{2. Compose two functions}

\begin{formulabox}
$(f\circ g)(x)=f(g(x))$: evaluate the \emph{inner} function first, then feed its
whole output into the outer one. The order is not interchangeable ---
$f\circ g$ and $g\circ f$ are usually different functions. Any input the inner
function rejects, or that produces a value the outer function rejects, is
outside the domain of the composition.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x)=3x-1$ and $g(x)=x^2+2$. Find $(f\circ g)(2)$.
\step{Composition runs inside out, so $g$ must be evaluated before $f$ ever
sees the number.}
\step{$g(2)=2^2+2=6$, and that whole value becomes the input of $f$:
$f(6)=3(6)-1$.}

So $(f\circ g)(2)=\ans{17}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} With the same $f$ and $g$, find $(g\circ f)(2)$. Does it match
the example?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{3. Invert a function}

\begin{formulabox}
Write $y=f(x)$, swap $x$ and $y$, solve for $y$. That is $f^{-1}$.
Check it by composing: $f\bigl(f^{-1}(x)\bigr)=x$. Domain and range trade
places, and the graph of $f^{-1}$ is the graph of $f$ reflected across $y=x$.
An inverse exists only when no output repeats --- otherwise restrict the domain
until none does.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x)=4x+5$. Find $f^{-1}$ and use it to compute
$f^{-1}(13)$.
\step{An inverse undoes the operations in reverse order, and swapping the
variables is the mechanical way to arrange that.}
\step{$y=4x+5$ swaps to $x=4y+5$; subtracting $5$ and dividing by $4$ gives
$f^{-1}(x)=\dfrac{x-5}{4}$, so $f^{-1}(13)=\dfrac{13-5}{4}$.}

The value is $\ans{2}$, and indeed $f(2)=13$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x)=\dfrac{x}{2}-6$. Find $f^{-1}$ and compute
$f^{-1}(-4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f^{-1}$ is not $\dfrac{1}{f}$ --- the $-1$ marks an
inverse function, not a reciprocal. And undoing a composition reverses the
order: $(f\circ g)^{-1}=g^{-1}\circ f^{-1}$, socks on before shoes, shoes off
before socks.
\end{watchoutbox}
