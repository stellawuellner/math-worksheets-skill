\sstitleblock{Taylor Polynomials and Remainder Bounds}

\noindent\small A Taylor polynomial answers ``what value?''; a remainder
bound answers ``how wrong could that be?'' --- and the second is what an exam grades.
\normalsize

\skillheading{1. Building a Taylor polynomial from derivative values}
\begin{formulabox}
\textbf{Rule:}
$T_n(x) = f(c) + f'(c)(x-c) + \dfrac{f''(c)}{2!}(x-c)^2 + \cdots
+ \dfrac{f^{(n)}(c)}{n!}(x-c)^n$.\\[3pt]
The $k$th derivative \emph{value} is the numerator, $k!$ the denominator, and
the power of $(x-c)$ matches $k$; Maclaurin is the case $c = 0$.\\[3pt]
\textbf{(!)} The factorial is what forces $T_n^{(k)}(c) = f^{(k)}(c)$ --- drop
it and the polynomial no longer matches the function at all.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} $f(1) = 2$, $f'(1) = 6$, $f''(1) = -4$. Approximate $f(1.5)$
with $T_2$ about $c = 1$.
\step{Only three derivative values are given, so $T_2$ is the best available
and every term is built on the increment $x - 1$.}
\step{$T_2(x) = 2 + 6(x-1) - \dfrac{4}{2}(x-1)^2$, and at $x = 1.5$ the
increment is $0.5$: $\;2 + 3 - 2(0.25)$.}
\par
$\ans{4.5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} $g(3) = 1$, $g'(3) = -2$, $g''(3) = 6$. Approximate $g(3.5)$
with $T_2$ about $c = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.75}$}
\end{tryitbox}

\skillheading{2. Building one from a known Maclaurin series}
\begin{formulabox}
\textbf{Memorise four:} $e^u = \sum \dfrac{u^n}{n!}$; \;
$\sin u = u - \dfrac{u^3}{3!} + \dfrac{u^5}{5!} - \cdots$; \;
$\cos u = 1 - \dfrac{u^2}{2!} + \dfrac{u^4}{4!} - \cdots$; \;
$\ln(1+u) = u - \dfrac{u^2}{2} + \dfrac{u^3}{3} - \cdots$\\[3pt]
\textbf{Then substitute} $u = -x^2$, $u = 2x$, $u = x^3$, whatever the problem
needs, and collect up to the degree asked for --- differentiating
$e^{-x^2}$ six times gives the same answer for ten times the work.\\[3pt]
\textbf{(!)} Substitution changes which \emph{degree} each term reaches: with
$u = -x^2$, the $n = 3$ term of $e^u$ is already degree $6$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Use the degree-4 Maclaurin polynomial for $\cos x$ to
approximate $\cos\!\left(\tfrac12\right)$.
\step{Cosine's series is known, so no differentiating is needed --- just stop
it at degree 4 and substitute.}
\step{$1 - \dfrac{x^2}{2} + \dfrac{x^4}{24}$ at $x = \tfrac12$ is
$1 - \dfrac{1}{8} + \dfrac{1}{384} = \dfrac{384 - 48 + 1}{384}$.}
\par
$\ans{\dfrac{337}{384}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Use the degree-4 Maclaurin polynomial for $e^{-x}$ to
approximate $e^{-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{3}{8}}$}
\end{tryitbox}

\skillheading{3. Bounding the remainder}
\begin{formulabox}
\textbf{Lagrange:} $|R_n(x)| \le \dfrac{M\,|x-c|^{\,n+1}}{(n+1)!}$, where
$\left|f^{(n+1)}(z)\right| \le M$ for \emph{every} $z$ between $c$ and $x$.
Both the exponent and the factorial use $n+1$ --- using $n$ is the most common
error in the topic.\\[3pt]
\textbf{Alternating series:} if the series alternates with terms decreasing to
$0$, the error is smaller than the \textbf{first omitted term} --- and its
sign says whether the estimate is high or low.\\[3pt]
\textbf{Finding $M$:} bound the $(n+1)$st derivative on the interval --- take
$M = e^{b}$ or larger for $e^x$ on $[0,b]$, and $M = 1$ for $\sin$ or $\cos$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} $\left|f'''(z)\right| \le 6$ on $[0, 0.3]$. Bound the error
when $T_2$ about $c = 0$ approximates $f(0.3)$.
\step{The polynomial has degree $n = 2$, so the bound needs the third
derivative and the third power --- that is what $n+1$ means.}
\step{$\dfrac{6 \cdot |0.3|^{3}}{3!} = \dfrac{6 \cdot 27/1000}{6}$.}
\par
$\ans{\dfrac{27}{1000}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} $\left|f^{(4)}(z)\right| \le 24$ on $[1, 1.25]$. Bound the
error when $T_3$ about $c = 1$ approximates $f(1.25)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{256}}$}
\end{tryitbox}

\skillheading{4. Using a polynomial to settle a limit}
\begin{formulabox}
Replace each function by its Maclaurin series, cancel what the numerator
subtracts away, and divide term by term: \textbf{the first surviving term is
the limit}.\\[3pt]
\textbf{(!)} Keep enough terms --- truncating $e^x$ at $1+x$ makes
$e^x - 1 - x$ look like exactly $0$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Evaluate
$\displaystyle\lim_{x \to 0} \frac{\sin x - x}{x^3}$ using series.
\step{Subtracting $x$ kills the leading term of $\sin x$, so the limit is
decided by the first term that survives.}
\step{$\sin x - x = -\dfrac{x^3}{6} + \dfrac{x^5}{120} - \cdots$, so dividing
by $x^3$ leaves $-\dfrac{1}{6} + \dfrac{x^2}{120} - \cdots$}
\par
$\ans{-\dfrac{1}{6}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Evaluate
$\displaystyle\lim_{x \to 0} \frac{1 - \cos x}{x^2}$ using series.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{2}}$}
\end{tryitbox}
