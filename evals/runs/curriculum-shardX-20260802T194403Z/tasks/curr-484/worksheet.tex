\wstitleblock{Diagnosing Separation and Initial-Condition Errors}{AP Calculus AB/BC $\bullet$ Error Analysis}{}

\noindent\small\textbf{Directions.} Separable equations go wrong in three
predictable places: integrating before the variables are actually separated,
losing the constant of integration, and applying the initial condition at the
wrong moment. Where a student's work is shown, \textbf{(i)} name the first wrong
line, \textbf{(ii)} say what rule it breaks, \textbf{(iii)} solve correctly.
Round decimals to two places and check each solution by substituting it back
into the differential equation.\normalsize

\problem[4cm]{Find the general solution of $\dfrac{dy}{dx} = 6x^{2} - 4x$.
Include the constant of integration, and say in one phrase what it represents on
a slope field.}

\problem[4.5cm]{Same equation, $\dfrac{dy}{dx} = 6x^{2} - 4x$, now with
$y(1) = 5$. Find the particular solution and evaluate $y(2)$.}

\problem[4.5cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = xy$ with
$y(0) = 2$; find $y(2)$. Wren's attempt:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
\rule{0pt}{2.6ex}$\displaystyle\int dy = \int xy\,dx \ \Rightarrow \ y = \frac{x^{2}}{2} + C$;
\quad $y(0) = 2 \Rightarrow C = 2$; \quad $y(2) = 2 + 2 = 4$ \\[2pt]\hline
\end{tabular}
\end{center}
\noindent Name the first wrong line, say what Wren did to the $y$ on the right
side, and give the correct $y(2)$.}

\problem[4.5cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = 3y$ with
$y(0) = 4$; find $y(1)$. Dev's attempt:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
\rule{0pt}{2.6ex}$\ln|y| = 3x + C \ \Rightarrow \ y = e^{3x} + C$;
\quad $y(0) = 4 \Rightarrow C = 3$; \quad $y(1) = e^{3} + 3 \approx 23.09$ \\[2pt]\hline
\end{tabular}
\end{center}
\noindent Dev's first line is correct. Name the line that is not, explain what
exponentiating does to a sum, and find $y(1)$ correctly.}

\problem[4.5cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = \dfrac{x}{y}$
with $y(0) = 3$; find $y(4)$. Rosa separates correctly to
$\dfrac{y^{2}}{2} = \dfrac{x^{2}}{2} + C$, then writes:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
\rule{0pt}{2.6ex}``Square-root both sides'': $y = x + C$;
\quad $y(0) = 3 \Rightarrow C = 3$; \quad $y(4) = 4 + 3 = 7$ \\[2pt]\hline
\end{tabular}
\end{center}
\noindent Two things went wrong in that first line. Name both, then solve
correctly for $y(4)$.}

\problem[4.5cm]{\textbf{Find and fix.} Solve $\dfrac{dy}{dx} = y^{2}$ with
$y(0) = 1$; find $y(0.5)$. Sam's attempt:
\begin{center}\footnotesize
\begin{tabular}{|l|}\hline
\rule{0pt}{2.6ex}$\displaystyle\int y^{-2}dy = \int dx \ \Rightarrow \ \frac{1}{y} = x + C$;
\quad $y(0) = 1 \Rightarrow C = 1$; \quad $y(0.5) \approx 0.67$ \\[2pt]\hline
\end{tabular}
\end{center}
\noindent One sign is missing. Fix it and give $y(0.5)$. Then say what the
corrected solution does as $x$ approaches $1$, and why that behaviour is
invisible in Sam's version.}

\problem[4.5cm]{Solve $\dfrac{dy}{dx} = \dfrac{2x}{x^{2}+1}$ with $y(0) = 1$ and
evaluate $y(2)$ to two decimal places. Then state the step at which the initial
condition must be applied, and why applying it before integrating would be
meaningless.}

\problem[5cm]{A classmate argues: ``As long as you add a constant somewhere, it
does not matter whether you add it before or after exponentiating.'' Using
$\dfrac{dy}{dx} = 3y$ from problem 4, show this is false: differentiate
$y = e^{3x} + 3$, compare the result with $3y$, and explain what the constant
becomes when it is carried through the exponential correctly.
