% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Dividing by a Pure Imaginary Number}

\begin{formulabox}
\textbf{Conjugate:} the conjugate of $ci$ is $-ci$, and $(ci)(-ci) = c^2$, a real
number.\\[3pt]
\textbf{Method:} multiply the top and the bottom by that conjugate, replace every
$i^2$ with $-1$, then divide each part of the numerator by the real denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{12 + 8i}{4i}$
\step{The denominator is a single imaginary term, so the fastest conjugate to use is $-4i$ --- it clears the $i$ from the bottom in one step.}
\step{Top: $(12 + 8i)(-4i) = -48i + 32$. Bottom: $(4i)(-4i) = 16$. Dividing both parts by $16$ gives the real part $2$ and the imaginary part $-3$.}

$\Rightarrow\quad \ans{2 - 3i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{15 - 10i}{5i}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2 - 3i}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Dividing by a Complex Binomial}

\begin{formulabox}
\textbf{Conjugate:} the conjugate of $c + di$ is $c - di$, and their product is
always real:
\[
  (c + di)(c - di) = c^2 + d^2
\]
That real value is the new denominator, which is why the conjugate --- and not
the divisor itself --- is the thing to multiply by.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{34}{5 - 3i}$
\step{An $i$ in the denominator means the quotient is not in standard form yet, so multiply top and bottom by the conjugate $5 + 3i$.}
\step{Bottom: $25 + 9 = 34$. Top: $34(5 + 3i)$. The $34$ cancels, leaving the real part $5$ and the imaginary part $3$.}

$\Rightarrow\quad \ans{5 + 3i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{20 + 10i}{1 + 2i}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8 - 6i}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Quotients in an Applied Setting}

\begin{formulabox}
\textbf{Where this shows up:} in an alternating-current circuit, voltage,
current and impedance are complex, and \textit{current} $=$ \textit{voltage}
$\div$ \textit{impedance}. Divide exactly as above; the real part of the answer
is the in-phase amount and the imaginary part records the phase shift. Keep the
unit that belongs to the quotient --- volts $\div$ ohms gives amps.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A voltage of $15 + 5i$ volts drives an impedance of
$1 + 2i$ ohms. Find the current in amps.
\step{Current is voltage divided by impedance, so this is a quotient of two complex numbers --- multiply top and bottom by the conjugate $1 - 2i$.}
\step{Bottom: $1 + 4 = 5$. Top: $(15 + 5i)(1 - 2i) = 25 - 25i$. Dividing both parts by $5$ gives the real part $5$ and the imaginary part $-5$.}

$\Rightarrow\quad \ans{5 - 5i}$ amps
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A voltage of $22 + 4i$ volts drives an impedance of
$1 + i$ ohms. Find the current in amps.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13 - 9i}$ amps}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} multiply by the \textit{conjugate}, not by the
denominator itself --- multiplying $c + di$ by $c + di$ leaves an $i$ downstairs.
And $(c + di)(c - di) = c^2 + d^2$, a \textit{sum}: the minus sign turns into a
plus when $i^2$ is replaced by $-1$.
\end{watchoutbox}
