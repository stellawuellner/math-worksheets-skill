\sstitleblock{Area Models for Two-Digit Multiplication}

\skillheading{Skill 1 --- Build the Model (Split the Factors)}

\begin{formulabox}
\textbf{Rule:} split each factor into tens $+$ ones and put one split along
each side of the rectangle. The splits must add back to the original factors.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Split 28 for an area model: $28 = 20 + \,?$
\step{A two-digit number splits into its tens and its ones; 28 has 2 tens
already written as 20.}
\step{$28 - 20 = 8$, so the ones part is 8.}

$\Rightarrow\quad \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $63 = 60 + \,?$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Find Each Partial Product}

\begin{formulabox}
\textbf{Rule:} each cell multiplies one part of each factor --- tens
$\times$ tens, tens $\times$ ones, ones $\times$ tens, ones $\times$ ones.
Use the basic fact, then attach a zero for every ten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In $34 \times 12$, what is the largest cell?
\step{The largest cell multiplies the two tens parts: 30 and 10.}
\step{$30 \times 10 = 300$ (fact $3 \times 1$, plus two zeros).}

$\Rightarrow\quad \ans{300}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In $26 \times 13$, what is the largest cell?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{200}$}
\end{tryitbox}

\skillheading{Skill 3 --- Add the Cells to Get the Total}

\begin{formulabox}
\textbf{Rule:} the product is the \textbf{sum of all four cells} --- the whole
rectangle. Leaving out a cell is the classic area-model mistake.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Finish $21 \times 34$ from its model.
\step{Cells: $20 \times 30 = 600$, \ $20 \times 4 = 80$, \ $1 \times 30 = 30$,
\ $1 \times 4 = 4$ --- all four must be added.}
\step{$600 + 80 + 30 + 4 = 714$}

$\Rightarrow\quad \ans{714}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Use a model to find $12 \times 13$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{156}$}
\end{tryitbox}

\skillheading{Skill 4 --- Estimate to Check}

\begin{formulabox}
\textbf{Rule:} round each factor to the nearest ten and multiply the rounded
numbers. If the exact answer is far from the estimate, hunt for a missing
cell.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $29 \times 41$.
\step{Round each factor to the nearest ten: 29 goes to 30 and 41 goes to 40.}
\step{$30 \times 40 = 1{,}200$}

$\Rightarrow\quad \ans{1{,}200}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $52 \times 38$ the same way.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2{,}000}$}
\end{tryitbox}
