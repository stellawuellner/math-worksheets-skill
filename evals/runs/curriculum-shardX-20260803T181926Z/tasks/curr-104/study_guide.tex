\sstitleblock{Rounding Boundaries and Place-Holding Zeros}

\skillheading{Skill 1 --- Round to a place you are given}
\begin{formulabox}\textbf{Three steps:}\par
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Underline the digit in the place you are rounding to.
  \item Look at the \textbf{one digit to its right}. That digit decides, and no other digit does.
  \item $5$ or more: the underlined digit goes up by one. Less than $5$: it stays. Everything to the right becomes $0$.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Round $6{,}382$ to the nearest hundred.
\step{The hundreds digit is 3, so the digit that decides is the tens digit just to its right.}
\step{The tens digit is 8, which is 5 or more, so the hundreds go up: $\ans{6{,}400}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Round $7{,}418$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7{,}400}$}\end{tryitbox}

\skillheading{Skill 2 --- The exact halfway case}
\begin{formulabox}\textbf{Rule:} when the deciding digit is exactly $5$, round
\textbf{up}. ``$5$ or more'' includes the $5$ itself, so $3{,}650$ rounds to $3{,}700$,
not $3{,}600$.\par
\textbf{And round the original number only.} Rounding to the ten first and then to the
hundred can push an answer past the halfway point that never should have crossed it.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Round $3{,}650$ to the nearest hundred.
\step{The deciding digit is the tens digit, and it is exactly 5 --- the halfway case, which the rule sends up.}
\step{The hundreds go from 6 to 7 and the smaller places become zeros: $\ans{3{,}700}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Round $2{,}450$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2{,}500}$}\end{tryitbox}

\skillheading{Skill 3 --- Zeros hold the places open}
\begin{formulabox}\textbf{Rule:} a rounded number has the same number of places it
started with. $5{,}048$ to the nearest hundred is $5{,}000$, never $50$ --- the zeros
keep the $5$ in the thousands place.\par
The same idea handles a rollover: when a place fills up to $10$, it becomes $0$ and one
unit carries into the place on its left.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad What is $10$ more than $3{,}995$?
\step{Adding a ten fills the tens place past 9, so expect a carry rather than a single changed digit.}
\step{$9$ tens $+ 1$ ten $= 10$ tens $= 1$ hundred, and $9$ hundreds $+1$ hundred $=1$ thousand, giving $\ans{4{,}005}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad What is $100$ more than $4{,}950$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5{,}050}$}\end{tryitbox}
