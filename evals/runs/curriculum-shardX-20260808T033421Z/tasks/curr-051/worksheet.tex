\wstitleblock{Place Value with Base-Ten Models}{Grade 2 Math}{}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item In these models, a big square (flat) shows one hundred, a tall bar (rod) shows one ten, and a small square (cube) shows one.
  \item Count the blocks carefully, then write your answer on the blank.
\end{itemize}

  \draw[thick] (0,0) rectangle (1.6,1.6);
  \foreach \i in {0,1,2} {
    \draw[thick] ({2.2+0.55*\i},0) rectangle ({2.5+0.55*\i},1.6);

  }
  \foreach \k in {0,...,4} \draw[thick] ({4.3+0.45*\k},0) rectangle ({4.6+0.45*\k},0.3);
\end{tikzpicture}
\end{center}}
