\wstitleblock{Whole-Number Division: Up to Four Digits by One-Digit Divisors}{Grades 4--5 $\bullet$ Fluency Practice}{}

\begin{center}\small
\textit{three-digit quotients, four-digit dividends, remainders, and estimating to check}
\end{center}

\noindent\small\textbf{Directions.} Divide using the long-division cycle:
\textbf{divide, multiply, subtract, bring down}, and repeat until there are no
digits left to bring down. Show all of your work in the space under each
problem, then write your final answer on the answer line. If a step leaves a
place empty, write a $0$ in the quotient --- do not skip the place. You can
always check a quotient by multiplying it back by the divisor.\normalsize

\problem[4.5cm]{Divide. \qquad $168 \div 4$}

\problem[4.5cm]{Divide. \qquad $375 \div 5$}

\problem[4.5cm]{Divide. \qquad $618 \div 6$
\smallskip\noindent\small Watch the tens place carefully.\normalsize}

\problem[4.5cm]{Nadia wants to estimate $2{,}835 \div 7$ before she divides.
She replaces the dividend with the nearby \emph{compatible number} $2{,}800$,
because $7$ divides it evenly. What estimate does Nadia get?}

\problem[5cm]{Now find the exact quotient. \qquad $2{,}835 \div 7$
\smallskip\noindent\small Then compare it with the estimate you found above ---
were you close?\normalsize}

\problem[5cm]{Divide. Write your answer as a quotient with a remainder.
\qquad $517 \div 4$}

\problem[5cm]{Divide. \qquad $891 \div 3$}

\problem[5cm]{A number was divided by $6$ and the quotient came out to exactly
$143$ with nothing left over. What was the number that got divided?
\smallskip\noindent\small Multiplication undoes division --- use that.\normalsize}

\problem[5.5cm]{Divide. \qquad $4{,}872 \div 8$}

\problem[5.5cm]{A factory has $3{,}275$ pencils. The pencils are packed into
boxes, and each full box holds $6$ pencils. How many \emph{full} boxes can the
factory pack?}

\problem[5.5cm]{The same factory packs the same $3{,}275$ pencils into full
boxes of $6$. After every full box is packed, how many pencils are left over?}

\problem[6cm]{A school printed $4{,}158$ raffle tickets and shared them equally
among $7$ classes. Each class then split its tickets equally among $9$ students.
How many raffle tickets did each student receive?
\smallskip\noindent\small This takes two divisions. Estimate first so you know
about how big each answer should be.\normalsize}
