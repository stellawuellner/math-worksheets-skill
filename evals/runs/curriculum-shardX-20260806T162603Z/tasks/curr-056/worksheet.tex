\wstitleblock{Adding Two-Digit Numbers with Place-Value Strategies}{Grade 2 Math}{}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item Break each number into \textbf{tens} and \textbf{ones}, add the pieces,
        then put them back together.
  \item Fill in every blank --- the small blanks show your thinking, and they
        lead you to the sum.
\end{itemize}

\problem{Add $34 + 25$ by breaking both numbers apart.\par\vspace{6pt}
\hspace*{1cm} tens: \ $30 + 20 =$~\ansblank \qquad
ones: \ $4 + 5 =$~\ansblank\par\vspace{6pt}
\hspace*{1cm} put the pieces together: \ $34 + 25 =$~\ansblank
\par\vspace*{2.6cm}}

\problem{Add $41 + 37$ by breaking both numbers apart.\par\vspace{6pt}
\hspace*{1cm} tens: \ $40 + 30 =$~\ansblank \qquad
ones: \ $1 + 7 =$~\ansblank\par\vspace{6pt}
\hspace*{1cm} put the pieces together: \ $41 + 37 =$~\ansblank
\par\vspace*{2.6cm}}

\problem{Add $26 + 48$. This time the ones make more than a ten!\par\vspace{6pt}
\hspace*{1cm} tens: \ $20 + 40 =$~\ansblank \qquad
ones: \ $6 + 8 =$~\ansblank\par\vspace{6pt}
\hspace*{1cm} put the pieces together: \ $60 + 14 =$~\ansblank
\par\vspace*{2.6cm}}

\problem{Dara shows $37 + 25$ with base-ten blocks: 3 rods and 7 cubes for the
first number, 2 rods and 5 cubes for the second.\par\vspace{6pt}
(a) How many \emph{cubes} (ones) are there in all? \quad \ansblank\par\vspace{6pt}
(b) Trade ten cubes for one rod. What is the sum? \quad \ansblank
\par\vspace*{2.8cm}}

\problem{Hop along the number line to add $46 + 38$: first one big jump of
$30$, then a jump of $8$.\par\vspace{6pt}
\hspace*{1cm} start at 46, jump $+30$: land on \ansblank \qquad
then jump $+8$: land on \ansblank
\par\vspace{4pt}
\begin{center}\begin{tikzpicture}
  \draw[<->, thick] (-0.5,0) -- (9.5,0);
  \foreach \i in {0,...,12} \draw ({0.75*\i},-0.1) -- ({0.75*\i},0.1);

\par\vspace*{2.5cm}}

\problem{Mia adds $39 + 24$ with a friendly number: she adds $40 + 24$
first, then takes away one.\par\vspace{6pt}
\hspace*{1cm} $40 + 24 =$~\ansblank \qquad
so \ $39 + 24 =$~\ansblank\par\vspace{6pt}
Explain why Mia takes away one at the end.
\par\vspace*{2.8cm}}

\problem{Use the place-value chart to add $57 + 36$.\par\vspace{4pt}
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
 & \textbf{tens} & \textbf{ones} \\ \hline
first number & 50 & 7 \\ \hline
second number & 30 & 6 \\ \hline
\end{tabular}
\end{center}
\vspace{4pt}
\hspace*{1cm} tens: \ $50 + 30 =$~\ansblank \qquad
ones: \ $7 + 6 =$~\ansblank\par\vspace{6pt}
\hspace*{1cm} sum: \ $57 + 36 =$~\ansblank
\par\vspace*{2.6cm}}
