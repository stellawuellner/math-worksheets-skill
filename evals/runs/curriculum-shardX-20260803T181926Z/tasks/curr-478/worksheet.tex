\wstitleblock{Integrating Rational Functions With Partial Fractions}{AP Calculus AB / BC \quad Integration Techniques}{}

\begin{center}\small\textit{Splitting a rational function into distinct linear
pieces, evaluating a definite integral after the split, and taking an improper
integral to a limit}\end{center}

\noindent\textbf{The one move behind every problem.} A rational function whose
denominator factors into \emph{distinct linear} factors is a sum of constants
over those factors, and each piece integrates to a logarithm:
$\dfrac{N(x)}{(x-r_1)(x-r_2)} = \dfrac{A}{x-r_1} + \dfrac{B}{x-r_2}$.

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.0cm, itemsep=1pt, topsep=2pt]
\item Routine: factor the denominator, write a constant over each factor, clear
denominators, then substitute each root in turn --- every term but one vanishes.
\item Check the degree first: if the numerator's degree is not lower than the
denominator's, divide before you split.
\item Show the decomposition before the integration, keep absolute values
inside every logarithm, and include $+\,C$ on indefinite integrals.
\item Give exact answers, and combine a difference of logarithms into one
logarithm.
\end{itemize}
\noindent\rule{\linewidth}{0.4pt}

\problem[7cm]{$\displaystyle\int \frac{5}{(x-2)(x+3)}\,dx$.

\textbf{(a)} Write the decomposition, clear denominators, and find $A$ and $B$
by substituting the two roots. \quad \textbf{(b)} Integrate.}

\problem[7cm]{$\displaystyle\int \frac{5x+1}{(x-1)(x+2)}\,dx$.

\textbf{(a)} The numerator is linear now. Say in one sentence why the same
two-constant form still works, then find $A$ and $B$. \quad
\textbf{(b)} Integrate.}

\problem[7.5cm]{\textbf{Marginal cost.} A workshop's marginal cost at production
level $x$ (hundreds of units) is
$C'(x) = \dfrac{24}{(x+1)(x+3)}$ dollars per unit.

\textbf{(a)} Decompose $C'(x)$. \quad \textbf{(b)} The added cost of going from
$x = 0$ to $x = 1$ is $\displaystyle\int_{0}^{1} C'(x)\,dx$; evaluate it exactly
and combine into one logarithm. \quad \textbf{(c)} State the units and say what
the number measures.}

\problem[8cm]{$\displaystyle\int \frac{2}{x(x-1)(x+1)}\,dx$.

\textbf{(a)} Three distinct factors need three constants. Clear denominators and
substitute $x = 0$, $x = 1$, $x = -1$. \quad \textbf{(b)} Integrate, and say why
the root substitution made one constant negative.}

\problem[8cm]{\textbf{Filtration.} Water passes through a filter at the rate
$r(t) = \dfrac{12}{(t+1)(t+3)}$ litres per minute, $t$ minutes after start-up.

\textbf{(a)} Decompose $r(t)$. \quad \textbf{(b)} Evaluate
$\displaystyle\int_{0}^{3} r(t)\,dt$ exactly and combine into one logarithm.
\quad \textbf{(c)} Is the rate increasing or decreasing on this interval?
Justify from the formula, not from a graph.}

\problem[8.5cm]{\textbf{Leak.} A tank leaks a chemical at the rate
$r(t) = \dfrac{2}{t(t+2)}$ kilograms per hour, for $t \ge 1$.

\textbf{(a)} Decompose $r(t)$ and write the antiderivative as a single
logarithm. \quad \textbf{(b)} Evaluate it on $[1, b]$ and take the limit as
$b \to \infty$. \quad \textbf{(c)} Does
$\displaystyle\int_{1}^{\infty} r(t)\,dt$ converge or diverge? Give its value,
and put your part (b) limit and part (c) value on the answer line.}

\problem[8.5cm]{\textbf{Logistic growth.} A population $p$ (thousands) satisfies
$\dfrac{dp}{dt} = \dfrac{p(10-p)}{10}$, so the time to grow from $p = 2$ to
$p = 4$ is $\displaystyle\int_{2}^{4} \frac{10}{p(10-p)}\,dp$ years.

\textbf{(a)} Decompose the integrand. Both constants are positive --- say why
the $10-p$ factor does not make its own constant negative. \quad
\textbf{(b)} Evaluate the integral exactly, as one logarithm. \quad
\textbf{(c)} Without computing it, explain why growing from $p = 4$ to $p = 6$
takes less time.}

\problem[9cm]{\textbf{Challenge --- total emission.} A process releases a
pollutant at the rate $r(t) = \dfrac{6}{(t+1)(t+4)}$ kilograms per year,
$t \ge 0$.

\textbf{(a)} Decompose $r(t)$ and write the antiderivative as a single
logarithm. \quad \textbf{(b)} Rewrite the total emission as
$\displaystyle\lim_{b\to\infty}\int_{0}^{b} r(t)\,dt$, evaluate on $[0, b]$, and
take the limit. \quad \textbf{(c)} Does it converge? Give the exact total, then
explain why $r(t) \to 0$ is not on its own enough --- name a rate that decays to
0 and still gives an infinite total.}
