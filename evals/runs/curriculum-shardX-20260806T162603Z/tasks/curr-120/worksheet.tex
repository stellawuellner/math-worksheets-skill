\wstitleblock{Factor Structure Puzzles}{Grades 4--5 Mathematics}{}

\begin{center}
{\small Factor pairs, divisibility tests, common multiples, and prime factor structure}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--4 warm up on factor pairs. After that the four kinds are mixed, so decide which tool each puzzle needs before you start.
  \item Show the factors or multiples you listed. A puzzle solved by guessing is not solved.
\end{itemize}

\vspace{0.2cm}

\problem[3cm]{Fill in the missing factor: \quad $7 \times$~\ansblank~$= \ 91$}

\problem[3.2cm]{A rectangle is made from 84 unit squares. One side is 12 units long. How long is the other side?
\answerline{units}}

\problem[3.2cm]{Kai lists every factor pair of 54 and picks the pair whose two numbers are closest together. The larger number of that pair is 9. Write the smaller number of that pair.}

\problem[3.2cm]{A teacher shares 108 stickers into equal packs with none left over. Each pack must hold more than ten stickers and fewer than fifteen stickers. How many stickers are in each pack?}

\problem[3.5cm]{Ivy wants to know whether $4236$ is divisible by 6.
\begin{itemize}
  \item[(a)] Write the sum of the digits 4, 2, 3 and 6. \ansline
  \item[(b)] Is the number divisible by 6? Explain using the test for 2 and the test for 3.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3.2cm]{A red lamp flashes every 6 seconds and a blue lamp flashes every 10 seconds. They flash together right now. How many seconds pass before they next flash together?
\answerline{seconds}}

\problem[3.5cm]{Write 126 as a product of prime numbers, then write its largest prime factor on the answer line.}

\problem[3.2cm]{In the three-digit number $4\,?\,7$, the $?$ stands for one missing digit. Which digit makes the whole number divisible by 9?}

\problem[3.2cm]{One ferris wheel finishes a turn every 8 minutes and another finishes a turn every 20 minutes. They start together. How many minutes pass before they next start a turn together?
\answerline{minutes}}

\problem[3.2cm]{Mina arranges 96 tiles into a rectangle with equal rows. The number of rows is a factor of 96 that is greater than five and less than eight. How many tiles are in each row?}

\problem[3.5cm]{The prime factorisation of a mystery number is $2 \times 2 \times 3 \times 5 \times 5$.
\begin{itemize}
  \item[(a)] What is the mystery number? \ansline
  \item[(b)] Is the mystery number divisible by 15? Explain how the list of prime factors tells you, without dividing.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[4cm]{A bakery packs cookies into boxes of 12 or into trays of 18. Some totals fill whole boxes exactly \emph{and} fill whole trays exactly.
\begin{itemize}
  \item[(a)] What is the smallest total greater than zero that works? \ansline
  \item[(b)] The bakery has 540 cookies. How many of those smallest totals fit into 540? \ansline
  \item[(c)] Explain how the prime factors of 12 and 18 show that 540 works.
\end{itemize}
