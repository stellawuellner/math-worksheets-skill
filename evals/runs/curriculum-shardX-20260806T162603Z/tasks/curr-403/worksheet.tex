\wstitleblock{Finding Inverses with Domain Restrictions}{Precalculus}{}

\begin{center}
\small\textit{Inverting a measured model, restricting a domain before inverting, and checking that two rules undo each other}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item To invert a rule, swap the roles of input and output and solve for the new output.
  \item Every applied problem states its units. Answer in the unit printed on the answer line.
  \item A rule that is not one-to-one has no inverse until its domain is cut down --- say what the cut is before you take a square root.
\end{itemize}
\vspace{0.15cm}

\problem{A phone plan charges $C(m) = 0.05m + 20$, where $C$ is the monthly cost in dollars and $m$ is the number of call minutes used. A bill came to $45$ dollars. Use the inverse of $C$ to find the number of minutes used.
\par\vspace*{4.5cm}\answerline{min}}

\problem[5cm]{Let $f(x) = 4x + 9$ and $g(x) = \dfrac{x - 9}{4}$. Compute $(f \circ g)(x)$ and simplify it to show that $g$ undoes $f$.}

\problem{The area of a circle is $A(r) = \pi r^{2}$, where $r$ is the radius in centimetres and $A$ is the area in square centimetres. Only $r > 0$ is meaningful.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Write the domain of $A^{-1}$ as an inequality in $A$.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] A circle has area $64\pi$ square centimetres. Use $A^{-1}$ to find its radius.
  \par\vspace*{3.0cm}\answerline{cm}
\end{itemize}}

\problem{A spring hangs from a hook. With a load of $w$ kilograms attached, its length is $L(w) = 12 + 0.4w$ centimetres. Find the load that stretches the spring to a length of $22$ centimetres.
\par\vspace*{4.5cm}\answerline{kg}}

\problem[6cm]{Let $f(x) = \dfrac{2x + 1}{x - 3}$ and $g(x) = \dfrac{3x + 1}{x - 2}$. Compute $(f \circ g)(x)$ and simplify it to show that $g$ undoes $f$.}

\problem{A stone dropped from rest falls $d(t) = 5t^{2}$ metres in $t$ seconds, for $t \geq 0$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] The stone has fallen $80$ metres. Use the inverse of $d$ to find the time taken.
  \par\vspace*{2.8cm}\answerline{s}
  \item[(b)] Explain why the restriction $t \geq 0$ is what allows $d$ to have an inverse at all.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{Temperature converts by $F(c) = \dfrac{9}{5}c + 32$, where $c$ is in degrees Celsius and $F$ is in degrees Fahrenheit. Find the Celsius temperature that corresponds to $5$ degrees Fahrenheit.
\par\vspace*{4.5cm}\answerline{degrees Celsius}}

\problem{Let $f(x) = x^{2} - 6x + 5$ with the domain restricted to $x \geq 3$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Complete the square, then write the domain of $f^{-1}$ as an inequality in $x$.
  \par\vspace*{3.0cm}\ansline
  \item[(b)] Find $f^{-1}(21)$.
  \par\vspace*{3.0cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = \dfrac{1}{x - 4}$ and $g(x) = \dfrac{1}{x} + 4$. Compute $(f \circ g)(x)$ and simplify it to show that $g$ undoes $f$.}

\problem{A ball is thrown straight down from a $45$ metre ledge. Its height is $h(t) = 45 - 5t^{2}$ metres, for $0 \leq t \leq 3$ seconds.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find the time at which the ball is $25$ metres above the ground.
  \par\vspace*{3.0cm}\answerline{s}
  \item[(b)] Explain why the restriction $0 \leq t \leq 3$ has to be stated before this model can be inverted.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}
