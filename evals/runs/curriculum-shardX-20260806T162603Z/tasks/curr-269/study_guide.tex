\sstitleblock{Function Notation Is Not Multiplication}

\skillheading{Skill 1 --- Reading the notation as an instruction}

\begin{formulabox}
\textbf{Rule:} in $f(x)$, the letter $f$ is the function's \emph{name} and the parentheses hold its \emph{input}. There is no multiplication anywhere in the symbol.\par
\textbf{So $f(4)$ says:} wherever the rule shows the input variable, write $4$ instead; then simplify. The name $f$ never appears in the arithmetic.\par
\textbf{Test yourself:} if your work still contains the letter $f$, you multiplied instead of substituting.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 5x - 3$, find $f(4)$.
\step{The $4$ sits inside the parentheses, so it is the input --- this is a substitution, not a product.}
\step{Replace $x$ by $4$: $5(4) - 3 = 20 - 3$.}

$f(4) = \ans{17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 6x - 1$, find $f(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $f(5) = \ans{29}$}
\end{tryitbox}

\skillheading{Skill 2 --- Substituting a negative input}

\begin{formulabox}
\textbf{Rule:} substitute the input \emph{in parentheses}, then apply the exponents and signs. Writing $-2$ without parentheses is what turns a correct substitution back into a sign error.\par
\textbf{Watch the square:} $(-2)^2 = 4$, while $-2^2 = -4$. The first squares the input; the second squares $2$ and then negates.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $h(x) = x^2 - 3x$, find $h(-2)$.
\step{The input is negative, so wrap it in parentheses before doing anything --- that keeps the square and the subtraction honest.}
\step{$(-2)^2 - 3(-2) = 4 + 6$.}

$h(-2) = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $h(x) = x^2 - 3x$, find $h(-4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $h(-4) = \ans{28}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f(a + b)$ is \emph{not} $f(a) + f(b)$. The parentheses hold one input, so the whole sum goes into the rule at once. Check it on $f(x) = x^2$: $f(1+2) = 9$, but $f(1) + f(2) = 5$.
\end{watchoutbox}

\skillheading{Skill 3 --- Solving for the input}

\begin{formulabox}
\textbf{Two questions that look alike:} ``find $f(5)$'' gives you the input and asks for the output --- substitute. ``Solve $f(x) = 5$'' gives you the output and asks for the input --- set the rule equal to $5$ and solve.\par
\textbf{Never divide by the function's name.} There is no factor to divide away; undo the rule one operation at a time instead.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 3x + 7$, solve $f(x) = 25$.
\step{The output is given and the input is unknown, so this is an equation to solve, not a substitution.}
\step{$3x + 7 = 25$; subtract $7$ to get $3x = 18$, then divide by $3$.}

$x = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 4x - 5$, solve $f(x) = 23$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{7}$}
\end{tryitbox}
