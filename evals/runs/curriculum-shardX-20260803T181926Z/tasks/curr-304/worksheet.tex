\wstitleblock{Circular Reasoning and Unsupported Statements}{High School Geometry}{}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A reason must come from the \textbf{givens}, a \textbf{definition}, a \textbf{postulate}, or a theorem already proved. Anything else is not yet available to you.
  \item A step is \textbf{circular} when its reason is the very statement being proved (or a theorem whose proof depends on it).
  \item A step is \textbf{unsupported} when its reason describes a property nobody gave you --- ``the lines are parallel'', ``$M$ is a midpoint'', ``those are vertical angles''.
  \item Name the faulty step first, then repair it. Show the algebra for any measure you are asked to find.
\end{itemize}
\normalsize

\problem[7cm]{\textbf{Find the circular step.} A student is asked to prove the Vertical
Angles Theorem and writes the proof below.\par
\smallskip
\textbf{Given:} $\angle AQC$ and $\angle BQD$ are vertical angles. \quad
\textbf{Prove:} $\angle AQC \cong \angle BQD$.\par
\smallskip
{\renewcommand{\arraystretch}{1.3}
\noindent\begin{tabular}{|p{0.44\textwidth}|p{0.44\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
$\angle AQC$ and $\angle BQD$ are vertical angles & Given \\ \hline
$m\angle AQC = m\angle BQD$ & Vertical angles are congruent \\ \hline
$\angle AQC \cong \angle BQD$ & Definition of congruent angles \\ \hline
\end{tabular}}\par
\smallskip
Which step is circular, and why? Then name the two facts about linear pairs that a
correct proof would use instead.}

\problem[6cm]{\textbf{Which reason works?} Point $M$ lies between $A$ and $C$ on
$\overline{AC}$. A proof needs the statement $AM + MC = AC$. Three reasons are offered:
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(i)] Segment Addition Postulate
  \item[(ii)] Because $AM + MC = AC$
  \item[(iii)] Because $M$ is the midpoint of $\overline{AC}$
\end{itemize}
Choose the reason that legitimately supports the statement, and say what is wrong with
each of the other two --- one of them is circular, the other assumes something the
givens never state.}

\problem{\textbf{Find the mistake.} Points $A$, $Q$, and $C$ are collinear and ray $QB$
is drawn, so $\angle AQB$ and $\angle BQC$ form a linear pair. It is given that
$m\angle AQB = 3x + 10$ and $m\angle BQC = 5x - 30$.\par
\smallskip
Dana writes ``$m\angle AQB = m\angle BQC$, because corresponding angles are
congruent'', solves $3x + 10 = 5x - 30$, and reports $x = 20$ and
$m\angle AQB = 70$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] What does Dana's reason assume that the givens never state?
  \item[(b)] Use the linear pair to find $x$, then find $m\angle AQB$.
\end{itemize}\par\vspace*{6cm}\answerline{degrees}}

\problem[7cm]{\textbf{Find the mistake.} Point $M$ lies between $A$ and $B$. It is given
that $AM = 2x + 3$, $MB = 4x - 7$, and $AB = 20$.\par
\smallskip
Ben writes ``$AM = MB$, because $M$ is the midpoint of $\overline{AB}$'', solves
$2x + 3 = 4x - 7$, and reports $x = 5$ and $AM = 13$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Which given would Ben need before he could write $AM = MB$? Is it there?
  \item[(b)] Use the Segment Addition Postulate to find $x$, then find $AM$.
\end{itemize}}

\problem[8cm]{\textbf{Repair the argument.} In $\triangle ABC$ it is given that
$\angle A \cong \angle B$. A student argues:\par
\smallskip
\noindent\emph{``$\overline{CA} \cong \overline{CB}$ because the triangle is isosceles,
and the triangle is isosceles because its base angles $\angle A$ and $\angle B$ are
congruent.''}\par
\smallskip
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain exactly why this argument is circular.
  \item[(b)] Write a correct two-column argument (statements and reasons) that reaches
    $\overline{CA} \cong \overline{CB}$ from the given.
\end{itemize}}
