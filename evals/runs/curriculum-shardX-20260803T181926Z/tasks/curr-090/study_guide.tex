% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Two-step problems about time}

\begin{formulabox}
\textbf{The plan:} step 1, put together the times that go together (add them).
Step 2, do the second thing the story asks --- add on one more time, or take the
time used away from the time you started with.\par
\textbf{Words that tell you to add:} \emph{then}, \emph{after that}, \emph{in all}.
\textbf{Words that tell you to subtract:} \emph{left}, \emph{still}, \emph{how much more}.\par
$1$ hour $= 60$ minutes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Tom reads for 30 minutes, then plays for 15 minutes, then reads
for 20 more minutes. How many minutes is that in all?
\step{Everything happens one after the other and the question asks for the total, so I add --- but two numbers at a time.}
\step{$30 + 15 = 45$ minutes, and then $45 + 20 = 65$ minutes.}
$\Rightarrow\quad \ans{65}$ minutes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Ana swims for 25 minutes, rests for 15 minutes, then swims for
30 more minutes. How many minutes in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{70}$ minutes}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Two-step problems about money}

\begin{formulabox}
\textbf{The plan:} step 1, find the total spent (add the prices, or multiply when
several things cost the same). Step 2, subtract that total from the money you
started with to find the change.\par
\textbf{Line up the dots} when you add or subtract dollars and cents, so the dollars
sit under the dollars and the cents under the cents.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Jo has \$15. She buys a game for \$7.50 and a card for \$2.25.
How much money does she have left?
\step{The question asks what is left, so I must first know the whole amount she spent --- add the two prices, then subtract.}
\step{$\$7.50 + \$2.25 = \$9.75$, and then $\$15.00 - \$9.75 = \$5.25$.}
$\Rightarrow\quad \ans{5.25}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Raj has \$20. He buys a ball for \$6.50 and a cap for \$9.25.
How much money does he have left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4.25}$ dollars}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Two-step problems about mass}

\begin{formulabox}
\textbf{Same units first!} $1$ kg $= 1000$ g. If one mass is in kilograms and
another is in grams, change the kilograms to grams before you add or subtract.\par
\textbf{The plan:} step 1, change the units. Step 2, add the masses that go into
the container, or subtract the mass that is taken out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A melon has a mass of 1 kg. A bag of grapes has a mass of 350 g.
What is the total mass in grams?
\step{The two masses use different units, so before adding anything I change the kilograms into grams.}
\step{$1$ kg $= 1 \times 1000 = 1000$ g, and then $1000 + 350 = 1350$ g.}
$\Rightarrow\quad \ans{1350}$ g
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A box has a mass of 2 kg and a book has a mass of 250 g. What is
the total mass in grams?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2250}$ g}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Two-step problems about liquid volume}

\begin{formulabox}
\textbf{Same units first!} $1$ L $= 1000$ mL. Change the litres to millilitres
before you take anything away.\par
\textbf{The plan:} step 1, work out how much was poured out --- multiply when
every glass is the same size. Step 2, subtract it from what the container held.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A bottle holds 1 L. You pour out 2 cups, and each cup holds
150 mL. How many millilitres are left in the bottle?
\step{The bottle is in litres and the cups are in millilitres, so I change the litres first, then work out how much left the bottle.}
\step{$1$ L $= 1000$ mL; poured out: $2 \times 150 = 300$ mL; left: $1000 - 300 = 700$ mL.}
$\Rightarrow\quad \ans{700}$ mL
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A jug holds 2 L. You pour out 3 glasses, and each glass holds
250 mL. How many millilitres are left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1250}$ mL}
\end{tryitbox}
