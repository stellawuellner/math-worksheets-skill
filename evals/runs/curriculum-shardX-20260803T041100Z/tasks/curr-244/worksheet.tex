\wstitleblock{Pythagorean Theorem: Which Side Goes Where?}{Grade 8 / Pre-Algebra}{}

\begin{center}\small\textit{Finding a hypotenuse, finding a missing leg, and finding
coordinate distance without dropping the hypotenuse into a leg slot}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $a^2 + b^2 = c^2$ holds only when $c$ is the \textbf{hypotenuse} --- the side
        across from the right angle, and always the longest side.
  \item Before you compute, write down which given side is the hypotenuse. If the
        hypotenuse is known and a leg is missing, you \emph{subtract}:
        $b^2 = c^2 - a^2$.
  \item Sanity-check every answer: a leg must come out \emph{shorter} than the
        hypotenuse.
  \item On a \textbf{find the mistake} problem, name the mistake in words \emph{and}
        show correct work. Round decimals to the nearest hundredth.
\end{itemize}
\normalsize

\refrt

\problem{A right triangle has legs of 6 cm and 8 cm. Find the length of its
hypotenuse.\par\vspace*{5cm}
\answerline{cm}}

\problem{A right triangle has a hypotenuse of 13 cm and one leg of 5 cm. Find the
length of the other leg.\par\vspace*{6cm}
\answerline{cm}}

\problem{\textbf{Find the mistake.} A 17-foot ladder leans against a wall with its
foot 8 feet from the wall. Jae wants the height $h$ the ladder reaches and writes:
\[ 17^2 + 8^2 = h^2 \qquad h \approx 18.79 \text{ feet} \]
(a) Explain how you can tell Jae's answer is impossible \emph{before} checking any
arithmetic. \quad (b) Name the substitution mistake, then find the correct height.\par\vspace*{8cm}
\answerline{ft}}

\problem{Find the distance between $A(2, 3)$ and $B(7, 15)$ on the coordinate plane.
Sketch the right triangle whose hypotenuse is $\overline{AB}$.\par\vspace*{6cm}
\answerline{units}}

\problem{\textbf{Find the mistake.} A right triangle has legs of 9 cm and 40 cm. Sam
writes: \emph{``40 is the longest side I was given, so it must be the hypotenuse,''} and
computes
\[ \sqrt{40^2 - 9^2} \approx 38.97 \text{ cm} \]
(a) Explain why 40 cannot be the hypotenuse here. \quad (b) Find the correct hypotenuse.\par\vspace*{8cm}
\answerline{cm}}

\problem{\textbf{Find the mistake.} Kim finds the distance from $P(-3, 2)$ to
$Q(5, 8)$. She writes: \emph{``The horizontal difference is 8 and the vertical difference
is 6. The 8 is bigger, so it is the hypotenuse,''} and computes
\[ \sqrt{8^2 - 6^2} \approx 5.29 \text{ units} \]
(a) Name Kim's mistake --- which segment really is the hypotenuse? \quad (b) Find the
correct distance $PQ$.\par\vspace*{8cm}
\answerline{units}}

\problem[7cm]{A television screen is a rectangle whose diagonal measures 65 inches and
whose height is 33 inches. Its width $x$ satisfies
\[ x^2 + 33^2 = 65^2 \]
(a) Solve the equation for \emph{every} value of $x$. \quad (b) Which solution is the
screen's width, and why must the other one be thrown out?}

\problem{A rectangular field measures 80 m by 60 m. Rosa walks from one corner to
the opposite corner along two sides of the field. Tam walks the straight diagonal between
the same two corners.

\smallskip
How many meters shorter is Tam's walk than Rosa's?\par\vspace*{8cm}
\answerline{m}}
