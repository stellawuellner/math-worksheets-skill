\wstitleblock{Three Views of a Limit}{Precalculus / Introduction to Limits}{}

\begin{center}
\textit{reading a limit from a table, from a graph, from algebra, and as the
limit of partial sums}
\end{center}

\noindent A limit describes where a function is \emph{headed} as the input
approaches a value --- never what happens \emph{at} that value. This sheet asks
the same question through four lenses: a table of values, a graph, algebra, and
the partial sums of a series. After the warm-up the four are interleaved, so
decide which tool the problem is handing you before you start computing.

\problem[4cm]{The table below has already been computed for
$f(x) = \dfrac{x^2-9}{x-3}$.

\begin{center}\small
\begin{tabular}{|c|c|c|c|c|c|c|c|}\hline
$x$ & $2.9$ & $2.99$ & $2.999$ & $3$ & $3.001$ & $3.01$ & $3.1$ \\\hline
$f(x)$ & $5.9$ & $5.99$ & $5.999$ & undefined & $6.001$ & $6.01$ & $6.1$ \\\hline
\end{tabular}
\end{center}

\noindent What does the table say $\displaystyle\lim_{x\to 3} f(x)$ is? Say in
one sentence why ``undefined at $x=3$'' does not stop the limit from existing.}

\problem[5cm]{Evaluate algebraically:
$\displaystyle\lim_{x\to 4}\frac{x^2-16}{x-4}$. Direct substitution gives
$\tfrac{0}{0}$, so rewrite before you substitute.}

\problem[4.5cm]{The graph of $g(x) = \dfrac{x^2-4}{x-2}$ is shown. The open
circle marks the one input where $g$ is not defined.

\begin{center}
\begin{tikzpicture}
\begin{axis}[width=8.2cm, height=5.4cm, axis lines=middle,
  xmin=-1.4, xmax=5.4, ymin=-0.8, ymax=7.6,

  yticklabel style={fill=white, inner sep=1pt}]

\end{axis}
\end{tikzpicture}\\[-2pt]
{\small Graph of $g$ for this problem.}
\end{center}

\noindent Read $\displaystyle\lim_{x\to 2} g(x)$ from the graph, and say in one
sentence why the open circle does not change the answer.}

\problem[5cm]{Write out and evaluate the partial sum
$\displaystyle\sum_{n=1}^{5}(2n-1)$.}

\problem[5.5cm]{Let $f(x) = \dfrac{\sqrt{x+1}-1}{x}$. Complete the table,
rounding each value to four decimal places.

\begin{center}\small
\begin{tabular}{|c|>{\centering\arraybackslash}p{2.2cm}|c|>{\centering\arraybackslash}p{2.2cm}|}\hline
$x$ & $-0.01$ & $0$ & $0.01$ \\\hline
$f(x)$ & & undefined & \\[0.9cm]\hline
\end{tabular}
\end{center}

\noindent Then state the exact value of $\displaystyle\lim_{x\to 0}f(x)$ as a
fraction. Write the two table entries and the limit on your answer line.}

\problem[5.5cm]{Evaluate algebraically:
$\displaystyle\lim_{x\to 2}\frac{x^3-8}{x^2-4}$. \; (Hint:
$a^3-b^3=(a-b)(a^2+ab+b^2)$ and $a^2-b^2=(a-b)(a+b)$.)}

\problem[5cm]{The graph of a piecewise function $h$ is shown. The solid dot
marks the value of $h(1)$; the open circle marks a point the graph approaches

\noindent \textbf{(a)} $\displaystyle\lim_{x\to 1^-}h(x)$ \quad
\textbf{(b)} $\displaystyle\lim_{x\to 1^+}h(x)$ \quad
\textbf{(c)} Does $\displaystyle\lim_{x\to 1}h(x)$ exist? Justify your answer
using your values from (a) and (b).}

\problem[5cm]{Evaluate the infinite geometric series
$\displaystyle\sum_{n=0}^{\infty}3\left(\tfrac{1}{2}\right)^{n}$, and state the
partial-sum limit it represents.}

\problem[5.5cm]{Evaluate algebraically:
$\displaystyle\lim_{x\to 0}\frac{\sqrt{x+9}-3}{x}$ \; by multiplying by the
conjugate of the numerator.}

\problem[5.5cm]{Let $r(x) = \dfrac{2x+3}{x-1}$. Complete the table, rounding

$r(x)$ & & \\[0.9cm]\hline

\noindent Then state $\displaystyle\lim_{x\to\infty}r(x)$. Write both table

\problem[5cm]{The graph of $p(x) = \dfrac{3x^2+1}{x^2+2}$ is shown with its

\begin{center}

\noindent Read $\displaystyle\lim_{x\to\infty}p(x)$ from the graph, then confirm
it algebraically by dividing the numerator and denominator by $x^2$.}

\problem[7cm]{\textbf{Synthesis.} Consider
$\displaystyle\sum_{n=1}^{\infty}\left(\tfrac{1}{2}\right)^{n}$.

\textbf{(a)} Compute the fifth partial sum
$S_5=\displaystyle\sum_{n=1}^{5}\left(\tfrac{1}{2}\right)^{n}$ as an exact
fraction.

\textbf{(b)} Find the sum of the infinite series.

\textbf{(c)} A classmate says ``an infinite sum just means you keep adding
forever''. Using your answers to (a) and (b), write two or three sentences
explaining what the sum of an infinite series really is, and how the numerical,
graphical and algebraic views of this series agree.}
