\sstitleblock{Zeros That Hold a Place}

\skillheading{Skill 1 --- What a zero does in a number}
\begin{formulabox}\textbf{The model:} every number has three places.\par
\begin{center}\small
\begin{tabular}{|c|c|c|}\hline
\textbf{hundreds} & \textbf{tens} & \textbf{ones} \\\hline
$100$s & $10$s & $1$s \\\hline
\end{tabular}
\end{center}
A $0$ means \emph{none of that size} --- and it holds the place open so the other
digits do not slide over.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad 5 hundreds, 0 tens, 3 ones.
\step{Give each place its own piece, including the empty one, so nothing slides.}
\step{$500 + 0 + 3$, and the pieces make $\ans{503}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad 8 hundreds, 0 tens, 6 ones. What number is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{806}$}\end{tryitbox}

\skillheading{Skill 2 --- Fixing a missing zero}
\begin{formulabox}\textbf{Check your work like this:} add up the pieces of your
expanded form. If the total is not the number you started with, a digit has slid into
the wrong place. Read the number again, one place at a time, and give the empty place
its $0$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Sam writes $604 = 600 + 40$. Fix it.
\step{Add Sam's pieces: $600 + 40 = 640$, which is not 604, so a digit slid --- check each place.}
\step{604 has 6 hundreds, 0 tens, 4 ones, so it is $600 + 0 + 4 = \ans{604}$. The 4 is ones, not tens.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Fix this: $209 = 200 + 90$. What should the pieces add up to?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{209}$}\end{tryitbox}

\skillheading{Skill 3 --- Same digits, different value}
\begin{formulabox}\textbf{Rule:} to compare, look at the hundreds place first. If the
hundreds match, compare the tens. If the tens match too, compare the ones. A $0$ in a
place means that place is worth nothing --- so it is the smallest a place can be.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Compare $507$ and $570$.
\step{Both have 5 hundreds, so the hundreds do not decide it --- move to the tens place.}
\step{$507$ has 0 tens and $570$ has 7 tens, so $\ans{507 < 570}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $<$ or $>$ between $830$ and $803$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{830 > 803}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Do not skip the empty place when you write
expanded form. $308$ is $300 + 0 + 8$. If you write $300 + 80$ you have moved the 8 into
the tens place and made a different number, $380$.
\end{watchoutbox}
