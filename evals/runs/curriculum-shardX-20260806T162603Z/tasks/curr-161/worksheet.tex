\wstitleblock{Finding a Percent of a Quantity}{Grade 6}{}

\begin{center}
\textit{Hundredths grids, ten-percent tape strips, and decimal multipliers for
finding a percent of a quantity}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Every problem here asks the same question in a different costume:
        \emph{how much is this percent of this amount?}
  \item Each problem names a model --- a hundredths grid, a ten-percent tape
        strip, or the decimal form of the percent. Show that model in your work
        space, then write the number it gives you on the answer line.
  \item Percent means ``per one hundred.'' Keep that sentence in view.
\end{itemize}

\problem[5cm]{A hundredths grid is one square cut into 100 equal parts, so each
part is one percent of the whole. Ms.~Ruiz shades 30 parts of a grid to stand
for the tomato plants in her garden of 200 plants. How many plants do the
shaded parts stand for?}

\problem[5cm]{On another hundredths grid, 45 of the 100 parts are shaded to
stand for the novels on a shelf of 60 books. How many books do the shaded parts
stand for?}

\problem[5cm]{A tape diagram splits the 350 seats of a theatre into 10 equal
strips, so one strip is ten percent of the seats. Fill in the tape table, then
record the two amounts.\par\vspace{4pt}
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
Percent of the seats & $10\%$ & $30\%$ \\ \hline
Number of seats & (a) & (b) \\[0.45cm] \hline
\end{tabular}
\end{center}\par\vspace{4pt}
\begin{itemize}
  \item[(a)] seats in one ten-percent strip: \ansblank
  \item[(b)] seats in three of those strips: \ansblank
\end{itemize}}

\problem[5cm]{A percent becomes a decimal when you divide it by 100. A rack
holds 84 jackets, and $25\%$ of them are blue.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] $25\%$ written as a decimal: \ansblank
  \item[(b)] blue jackets on the rack: \ansblank
\end{itemize}}

\problem[5cm]{A pizza night cost 60 dollars. Build the tip out of ten-percent
strips instead of multiplying: draw the strip for $10\%$, halve it for $5\%$,
then put the pieces together.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] $10\%$ of 60 dollars: \ansblank
  \item[(b)] $5\%$ of 60 dollars: \ansblank
  \item[(c)] $15\%$ of 60 dollars: \ansblank
\end{itemize}}

\problem[5cm]{A school has 250 students, and $8\%$ of them were absent today.
Write the percent as a decimal, multiply, and give the number of absent
students.}

\problem[5cm]{A hundredths grid stands for a 400 gram bag of trail mix, so the
100 equal parts share the bag evenly. The 35 shaded parts are the
raisins.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] grams of trail mix in one part of the grid: \ansblank
  \item[(b)] grams of raisins in the bag: \ansblank
\end{itemize}}

\problem[5cm]{A hiking club has 45 members. Use ten-percent strips of the

\begin{itemize}
  \item[(a)] $10\%$ of the members: \ansblank
  \item[(b)] $60\%$ of the members: \ansblank
\end{itemize}}

\problem[5.5cm]{A repair bill of 320 dollars carries a $7.5\%$ service charge.
Use the decimal form of the percent.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] $7.5\%$ written as a decimal: \ansblank
  \item[(b)] the service charge, in dollars: \ansblank
\end{itemize}}

\problem[6.5cm]{Maya says that $20\%$ of 150 dollars must be more than $25\%$ of
100 dollars, because 150 dollars is the larger amount to start
with.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] $20\%$ of 150 dollars: \ansblank
  \item[(b)] $25\%$ of 100 dollars: \ansblank
  \item[(c)] Use your two amounts to explain whether Maya's reasoning is
             correct. Write your answer in the work space.
\end{itemize}}
