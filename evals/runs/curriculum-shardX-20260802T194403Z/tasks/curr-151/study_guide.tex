\sstitleblock{Ratios and Equivalent Ratio Tables}

\skillheading{Skill 1 --- Reading a ratio off a model}

\begin{formulabox}
\textbf{Three ways to write one comparison:}\\[4pt]
$a$ to $b$ \qquad $a:b$ \qquad $\dfrac{a}{b}$\par
\smallskip
\textbf{Order carries meaning.} Whatever the question names first is written
first. Count the model, write the pair, then simplify by dividing both numbers
by a common factor --- the model itself does not change, only the notation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A strip shows 8 star counters and 12 moon counters.
Write stars to moons in simplest form.
\step{The question names stars first, so stars is the first number --- this is
a part compared with another part, read straight off the model.}
\step{Stars to moons is 8 to 12, or $8:12$, or $\frac{8}{12}$; dividing both by
4 gives $\ans{\frac{2}{3}}$ --- for every 2 stars there are 3 moons.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A strip shows 9 triangle counters and 6 square counters.
Write triangles to squares in simplest form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{3}{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Part-to-part or part-to-whole?}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
If two parts are in the ratio $a:b$, the whole is $a + b$ equal shares.\par
\smallskip
part to part $= \dfrac{a}{b}$ \qquad part to whole $= \dfrac{a}{a+b}$\par
\smallskip
Only the \textbf{part-to-whole} form may be read as ``that fraction of the
group.'' The denominator you need is usually a number the question never
printed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A jar holds green marbles and blue marbles in the ratio
2 to 7. What fraction of the marbles is green?
\step{The question asks for a fraction \emph{of the whole jar}, so the 7 is the
wrong denominator --- build the whole first.}
\step{The whole is $2 + 7 = 9$ shares and green is 2 of them, so the fraction
is $\ans{\frac{2}{9}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A squad has forwards and defenders in the ratio 5 to 3.
What fraction of the squad is forwards?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{8}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Equivalent ratio tables}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
Multiply or divide \textbf{both} quantities by the same number to get an
equivalent ratio. Every column of a ratio table names the same relationship, so
any two columns may be set equal as a proportion and cross-multiplied.\par
\smallskip
Dividing down to a small, friendly column first usually makes the scale-up
easy.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 6 pens cost 15 dollars. What do 8 pens cost?
\step{The price per pen is steady, so the two columns are equivalent ratios ---
write a proportion rather than adding on 2 more dollars.}
\step{$\frac{x}{8} = \frac{15}{6}$, so $6x = 120$ and the cost is $\ans{20}$
dollars. Check: 2 pens cost 5 dollars, and $4 \times 5 = 20$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 4 notebooks cost 10 dollars. What do 14 notebooks
cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

\skillheading{Skill 4 --- The value of a ratio}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
The \textbf{value} of the ratio $a:b$ is the quotient $a \div b$ --- a single
number, the unit rate ``$a$-units for every 1 $b$-unit.''\par
\smallskip
Two ratios are equivalent exactly when their values are equal, so comparing
values is how you decide which of two ratios is larger.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A typist types 96 words in 8 minutes. Write the ratio of
words to minutes and find its value.
\step{``Per minute'' means the minutes must become 1, so divide the first
quantity by the second.}
\step{The ratio is $\frac{96}{8}$, and its value is $96 \div 8 = \ans{12}$
words per minute.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A craft kit puts 105 beads on 7 strings, the same number
on each. Find the value of the ratio of beads to strings.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $a:b$ and $\frac{a}{b}$ are the same ratio, but
``$\frac{a}{b}$ of the group'' is a different claim. Before writing a fraction,
ask: compared with the \emph{other part}, or with the \emph{whole}? If it is
the whole, add the parts first.
\end{watchoutbox}
