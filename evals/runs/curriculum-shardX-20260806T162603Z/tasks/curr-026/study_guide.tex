\sstitleblock{Taking Away and Writing It}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Cross Out to Subtract}

\begin{formulabox}
\textbf{Rule:} To take away, cross out one object for each one taken. What
is left are the objects with \emph{no} cross. Count only those.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 5 stars are drawn. 2 are crossed out. How many stars
are left?
\step{Crossed-out stars are gone, so we count only the stars without a cross.}
\step{Counting the plain stars: 1, 2, 3 --- and the equation agrees: $5-2=3$. $\checkmark$}

$\Rightarrow\quad 5 - 2 = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Draw 6 dots and cross out 4. How many dots are
left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Write the Subtraction Equation}

\begin{formulabox}
\textbf{Rule:} The story numbers go into the equation in order:\par
start number $-$ number taken away $=$ number left.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 7 fish are in a tank. 3 are scooped out. Write the
equation and solve it.
\step{The start number is 7 and the taken-away number is 3, so the equation is $7-3$.}
\step{Count back from 7 three times: 6, 5, 4. $\checkmark$}

$\Rightarrow\quad 7 - 3 = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 8 bees are on a flower. 5 fly off. Write the equation
and solve it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Take-Away Facts with Ten}

\begin{formulabox}
\textbf{Rule:} Taking a number from 10 leaves its \textbf{partner}: think
``what goes with it to make 10?'' instead of counting back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $10 - 8$
\step{Instead of counting back 8 times, we ask which partner goes with 8 to make ten.}
\step{$8 + 2 = 10$, so taking 8 away leaves 2. $\checkmark$}

$\Rightarrow\quad 10 - 8 = \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $10 - 3$ \ (think: 3 and what make ten?)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Count what is \emph{left}, not what you crossed out!
In a picture of $6-2$, the answer is the 4 plain objects --- not the 2 with
crosses.
\end{watchoutbox}
