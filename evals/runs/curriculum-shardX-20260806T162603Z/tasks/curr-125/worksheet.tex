\wstitleblock{Ordering Fractions and Mixed Numbers}{Grade 4--5}{}
\begin{center}
\textit{Ordering with benchmarks, common denominators, and mixed numbers}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Choose your tool for each problem: a benchmark like $\tfrac{1}{2}$
        or 1, a common denominator, or rewriting a mixed number.
  \item When a blank asks for a symbol, write $<$, $>$, or $=$.
\end{itemize}

\problem[3cm]{Write the correct symbol:
$\tfrac{5}{8}$ \ansblank $\tfrac{1}{2}$}

\problem[3cm]{Write $\tfrac{7}{4}$ as a mixed number.}

\problem[4cm]{Order from least to greatest by rewriting everything in
sixths: $\tfrac{2}{3}$, \ $\tfrac{1}{2}$, \ $\tfrac{5}{6}$
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[4cm]{Use the benchmark $\tfrac{1}{2}$ to compare $\tfrac{3}{8}$ and
$\tfrac{5}{9}$.
\par\vspace{2pt}
$\tfrac{3}{8}$ \ansblank $\tfrac{1}{2}$ \qquad
$\tfrac{5}{9}$ \ansblank $\tfrac{1}{2}$ \qquad
so \ $\tfrac{3}{8}$ \ansblank $\tfrac{5}{9}$}

\problem[3.5cm]{On a number line, point $P$ sits at $2\tfrac{1}{3}$. Write
the location of $P$ as an improper fraction in thirds.}

\problem[4cm]{Order from least to greatest (twenty-fourths make a good
common denominator): $\tfrac{7}{8}$, \ $\tfrac{3}{4}$, \ $\tfrac{5}{6}$
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[4.5cm]{Order from least to greatest. Rewrite each as eighths first:
$1\tfrac{3}{4}$, \ $\tfrac{15}{8}$, \ $1\tfrac{1}{2}$
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[4.5cm]{Three hiking trails: Pine is $\tfrac{5}{12}$ mi, Oak is
$\tfrac{4}{6}$ mi, and Elm is $\tfrac{1}{2}$ mi. Order the trail lengths
from least to greatest. \emph{Hint: compare each to $\tfrac{1}{2}$.}
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[4.5cm]{Ben says: ``$2\tfrac{1}{4} < \tfrac{9}{4}$, because 2 is
less than 9.''
\par\vspace{2pt}
(a) Write $2\tfrac{1}{4}$ in fourths: \ansblank
\par\vspace{2pt}
(b) Write the correct symbol: $2\tfrac{1}{4}$ \ansblank $\tfrac{9}{4}$
\par\vspace{2pt}
(c) Explain what Ben overlooked.}

\problem[4.5cm]{Order from least to greatest (thirty-sixths work for all
three): $\tfrac{5}{6}$, \ $\tfrac{7}{9}$, \ $\tfrac{8}{12}$
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[4.5cm]{Order from least to greatest using the benchmarks 1 and
$1\tfrac{1}{2}$: $\tfrac{13}{8}$, \ $1\tfrac{1}{4}$, \ $\tfrac{3}{2}$
\par\vspace{2pt}
\ansblank \ $<$ \ \ansblank \ $<$ \ \ansblank}

\problem[5cm]{\textbf{Challenge.} Three ribbons: ribbon A is $1\tfrac{5}{6}$
m, ribbon B is $\tfrac{11}{6}$ m, and ribbon C is $1\tfrac{2}{3}$ m.
\par\vspace{2pt}
(a) Compare A and B: $1\tfrac{5}{6}$ \ansblank $\tfrac{11}{6}$
\par\vspace{2pt}
(b) Compare C and B: $1\tfrac{2}{3}$ \ansblank $\tfrac{11}{6}$
\par\vspace{2pt}
(c) Which ribbon is the shortest? Say why two of the ribbons are a tie.}
