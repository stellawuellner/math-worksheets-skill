\sstitleblock{Binomial Coefficients in Probability and Expansion}

\skillheading{Skill 1 --- Binomial coefficients and Pascal's rows}

\begin{formulabox}
\textbf{Formula:}\quad $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$ --- the number of ways to
choose $k$ things from $n$. \quad \textbf{Addition rule:}
$\dbinom{n}{k} = \dbinom{n-1}{k-1} + \dbinom{n-1}{k}$, which is how each Pascal entry is the
sum of the two above it. Also $\dbinom{n}{k} = \dbinom{n}{n-k}$, so every row is symmetric.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\dbinom{7}{3}$.
\step{This is a straight count of choices, so apply the factorial formula and cancel before multiplying.}
\step{$\dfrac{7!}{3!\,4!} = \dfrac{7 \cdot 6 \cdot 5}{3 \cdot 2 \cdot 1}$.}

$\Rightarrow\quad \dbinom{7}{3} = \ans{35}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\dbinom{8}{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{Skill 2 --- Expanding a binomial power}

\begin{formulabox}
\textbf{Binomial theorem:}
\[ (a + b)^n = \sum_{k=0}^{n} \binom{n}{k} a^{\,n-k} b^{\,k} \]
The exponents on $a$ count down while those on $b$ count up, and every term's exponents add
to $n$. The coefficients are row $n$ of Pascal's triangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $(x + 1)^3$.
\step{A whole power of a binomial with a small exponent --- read the coefficients straight off Pascal's row 3 rather than multiplying out.}
\step{Row $3$ is $1, 3, 3, 1$, so the terms are $x^3$, $3x^2$, $3x$, $1$.}

$\Rightarrow\quad (x+1)^3 = \ans{x^3 + 3x^2 + 3x + 1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $(x - 2)^3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^3 - 6x^2 + 12x - 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Picking out one term}

\begin{formulabox}
\textbf{Rule:} the term of $(a+b)^n$ containing $b^k$ is $\dbinom{n}{k}a^{\,n-k}b^{\,k}$.
Choose $k$ from the power you want, then let $n - k$ fill the other exponent --- never expand
the whole thing when one term is asked for.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the coefficient of $x^2$ in $(x + 2)^4$.
\step{Only one term carries $x^2$, so match exponents first: $x^2$ needs the $2$ used twice, giving $k = 2$.}
\step{The term is $\dbinom{4}{2}x^2 \cdot 2^2 = 6 \cdot x^2 \cdot 4$.}

$\Rightarrow\quad$ coefficient $= \ans{24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the coefficient of $x^3$ in $(x + 1)^6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

\skillheading{Skill 4 --- Binomial probability}

\begin{formulabox}
\textbf{Rule:} for $n$ independent trials each succeeding with probability $p$,
\[ P(\text{exactly } k \text{ successes}) = \binom{n}{k}p^{\,k}(1-p)^{\,n-k}. \]
The power gives the chance of one particular ordering; the coefficient counts how many
orderings there are. For ``at least $k$'', add the separate terms.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A fair coin is flipped $4$ times ($16$ equally likely sequences, $6$ of them with exactly two heads). Find $P(\text{exactly two heads})$.
\step{All sequences are equally likely, so the probability is a plain count of favourable sequences over all sequences.}
\step{$P = \dfrac{6}{16}$, which reduces by $2$.}

$\Rightarrow\quad P = \ans{\tfrac{3}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $4$ flips; $4$ of the $16$ sequences have exactly one head.
Find that probability.\\[2pt]
\rotatebox{180}{\footnotesize check: $P = \ans{\tfrac{1}{4}}$}
\end{tryitbox}
