\wstitleblock{A Fraction of a Whole Number or Set}{Grade 4--5 \quad Multiplying and Dividing Fractions}{}

\begin{center}
\textit{Finding one equal group as a unit fraction of a set, finding several
equal groups as a non unit fraction, and working backwards from a part to the
whole}
\end{center}

\noindent\textbf{The two moves, every time.}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
\item \textbf{Divide by the bottom number.} It says how many \emph{equal
groups} the whole set is broken into. One group is the unit fraction.
\item \textbf{Multiply by the top number.} It says how many of those groups to
take.
\item So $\frac{2}{5}$ of $20$ means: make $5$ equal groups of $4$, then take
$2$ of them. Divide first, multiply second.
\item Draw or circle the groups before you write any numbers --- the picture is
what makes the two moves obvious.
\end{itemize}

\problem[4.5cm]{A box holds $24$ crayons. Ravi takes $\frac{1}{4}$ of them.
Circle the crayons in the picture to show the equal groups, then write how many
crayons Ravi takes.
\begin{center}
\begin{tikzpicture}[x=0.62cm, y=0.62cm]
  \foreach \r in {0,1,2} \foreach \c in {0,...,7}
    \fill[blue!30] (\c,-\r) circle (0.2);
\end{tikzpicture}
\end{center}}

\problem[4.5cm]{Now work backwards. Leo ate $\frac{1}{5}$ of a bag of grapes,
