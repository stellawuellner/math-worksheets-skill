\wstitleblock{Coordinates, Line Plots and Patterns --- Find the Mistake, Then Fix It}{Grades 4--5}{}

\noindent\small Every mistake on this sheet comes from reading a number in the
wrong place. Three rules stop all of them.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Across first, then up.} In $(x,\,y)$ the first number counts
        across from the corner and the second counts up. $(3,\,8)$ and
        $(8,\,3)$ are two different places.
  \item \textbf{On a line plot, count the marks.} The number written under the
        line says \emph{what} was measured. The marks say \emph{how many}.
  \item \textbf{In a pattern pair, keep the patterns in order.} The first
        pattern's term always goes in the first slot.
\end{itemize}
\normalsize

\problem[5cm]{\textbf{Warm-up. Grid.} Point C is at $(5,\,4)$ and point D is at
$(4,\,5)$. They are built from the same two numbers but they are not the same
place. Moving only along grid lines, how many steps take you from C to D?}

\problem[5cm]{\textbf{Find and fix. Grid.} Nia wants to plot $(2,\,7)$. She
counts $7$ across and $2$ up instead, and lands on a point that is already
labelled.
\par\smallskip
Which labelled point did Nia land on, and which one did she mean? Then, moving
only along grid lines, how many steps apart are the two points?}

\problem[5cm]{\textbf{Find and fix. Patterns.} Pattern X starts at $0$ and
adds $3$ each time. Pattern Y starts at $0$ and adds $1$ each time. Matching
terms are written as ordered pairs $(X,\,Y)$.
\par\smallskip
Ravi is asked how far \emph{up} the pair with an across-value of $12$ sits, and
he answers $12$. Ravi read the pair the wrong way round. What is the correct
up-value?}

\problem[5cm]{\textbf{Find and fix. Line plot.} Omar says, ``$3$ plants are
$3$ inches tall --- I read the $3$ under the marks.''
\par\smallskip
Omar copied the height instead of counting the marks above it. How many plants
are exactly $3$ inches tall?}

\problem[5cm]{\textbf{Line plot.} Which height happened most often? Careful:
the answer is the height written \emph{under} the tallest stack, not how many
marks are in it. Give your answer in inches.}

\problem[5cm]{\textbf{Find and fix. Line plot.} Kaya says, ``$5$ plants were
measured, because there are $5$ different heights marked on the line.''
\par\smallskip
Kaya counted heights instead of plants. How many plants were measured
altogether?}

\problem[7cm]{\textbf{Find and fix. Patterns.} Pattern A starts at $1$ and adds
$2$: \ $1,\ 3,\ 5,\ 7,\ 9$. \ Pattern B starts at $3$ and adds $6$: \ $3,\ 9,\
15,\ 21,\ 27$. Matching terms make the ordered pairs $(A,\,B)$.
\par\smallskip
Sam writes the $4$th ordered pair as $(21,\,7)$.
\par\smallskip
\textbf{(a)} What is the $4$th term of Pattern A --- the number that belongs
first?
\quad \textbf{(b)} What is the $4$th term of Pattern B --- the number that
belongs second?
\quad \textbf{(c)} Write the $4$th ordered pair correctly and say what Sam did.}

\problem{\textbf{Show what you know.}
\par\smallskip
\textbf{(a)} On the grid below, plot and label the points $(6,\,2)$ and
$(2,\,6)$.
\par
\textbf{(b)} Write one or two sentences explaining why they are different places
even though they are made from the same two numbers.
\par
\textbf{(c)} Write down a way to remember which number goes first, in words a
classmate could use.
