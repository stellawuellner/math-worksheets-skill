\sstitleblock{Proportional Relationships: Four Representations}

% =================== SKILL 1 ===================
\skillheading{Graph to Equation}

\begin{formulabox}
\textbf{Rule:} pick any clearly-marked point $(x, y)$ on the line, compute
$k = \dfrac{y}{x}$, and write $y = kx$. The line must pass through the origin,
or the relationship is not proportional.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A proportional graph passes through $(6, 9)$. Write its
equation.
\step{Divide $y$ by $x$ at the marked point to get $k$, then write $y = kx$.}
\step{$k = \dfrac{9}{6} = 1.5$.}

$\Rightarrow\quad \ans{y = 1.5x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A proportional graph passes through $(4, 22)$. Write its
equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 5.5x}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Table to Equation}

\begin{formulabox}
\textbf{Rule:} compute $\dfrac{y}{x}$ for every row. If all rows agree, that
shared ratio is $k$ and the table's equation is $y = kx$. If the rows disagree,
the table is not proportional and has no such equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table gives $(2, 4.5)$ and $(4, 9)$. Write its
equation.
\step{Check each row's ratio --- they must match before you may write $y = kx$.}
\step{$\dfrac{4.5}{2} = 2.25$ and $\dfrac{9}{4} = 2.25$ --- same ratio.}

$\Rightarrow\quad \ans{y = 2.25x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $(3, 4.5)$ and $(6, 9)$. Write its
equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 1.5x}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Equation to Context}

\begin{formulabox}
\textbf{Rule:} in $y = kx$, the number $k$ is the \textbf{unit rate} --- what
one unit of $x$ is worth in the story (price per kg, km per hour). To answer a
story question, substitute the given $x$ and compute.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $y = 2.5x$ gives the cost in dollars of $x$ notebooks.
Find the cost of $7$ notebooks.
\step{The question hands you $x$ --- substitute it into the equation and
multiply.}
\step{$y = 2.5 \cdot 7 = 17.5$.}

$\Rightarrow\quad \ans{17.5}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $y = 4.5x$ gives kilometers walked in $x$ hours. Find
$y$ when $x = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$ km}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Context to Graph}

\begin{formulabox}
\textbf{Rule:} a steady rate in a story graphs as the line through the origin
that rises by the rate for each unit step. Compute two or three points with
$y = \text{rate} \times x$, plot them, and draw the line through $(0, 0)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A cyclist rides $8$ km each hour. Find the point on the
graph at $3$ hours.
\step{Multiply the rate by the time to get the height of the graph there.}
\step{$y = 8 \cdot 3 = 24$, so the graph passes through the point $(3, 24)$.}

$\Rightarrow\quad \ans{24}$ km at $3$ hours
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A kayaker paddles $7$ km each hour. Find $y$ on the
graph at $4$ hours.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$ km}
\end{tryitbox}
