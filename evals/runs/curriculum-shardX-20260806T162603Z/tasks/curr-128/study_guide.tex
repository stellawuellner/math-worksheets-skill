\sstitleblock{Adding and Subtracting Mixed Numbers}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Adding mixed numbers}

\begin{formulabox}
\textbf{Plan:} add the wholes, add the fraction parts, then put them back
together. If the fraction parts add to more than one whole, move that extra
whole over to the whole-number part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $1\tfrac{2}{5} + 2\tfrac{1}{5}$.
\step{Same-size parts (fifths), so add wholes and fifths separately, then recombine.}
\step{Wholes: $1 + 2 = 3$; fifths: $\tfrac{2}{5} + \tfrac{1}{5} = \tfrac{3}{5}$; together $\ans{3\tfrac{3}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $2\tfrac{1}{3} + 1\tfrac{1}{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3\tfrac{2}{3}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Subtracting mixed numbers}

\begin{formulabox}
\textbf{Plan:} subtract the wholes, subtract the fraction parts, put them
back together. Always check first whether the top fraction part is big
enough --- if not, you will need Skill 3.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $3\tfrac{4}{5} - 1\tfrac{1}{5}$.
\step{$\tfrac{4}{5}$ is bigger than $\tfrac{1}{5}$, so no regrouping is needed --- subtract wholes and fifths separately.}
\step{Wholes: $3 - 1 = 2$; fifths: $\tfrac{4}{5} - \tfrac{1}{5} = \tfrac{3}{5}$; together $\ans{2\tfrac{3}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $4\tfrac{5}{8} - 2\tfrac{1}{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2\tfrac{1}{2}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Regrouping to subtract}

\begin{formulabox}
\textbf{Rule:} when the fraction you are taking away is bigger than the one
you have, trade one whole for a fraction:
$4\tfrac{1}{5} = 3\tfrac{6}{5}$. The value does not change --- one whole is
just rewritten as $\tfrac{5}{5}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Subtract $4\tfrac{1}{5} - 1\tfrac{3}{5}$.
\step{$\tfrac{3}{5}$ is bigger than $\tfrac{1}{5}$, so trade one whole: $4\tfrac{1}{5}$ becomes $3\tfrac{6}{5}$.}
\step{Wholes: $3 - 1 = 2$; fifths: $\tfrac{6}{5} - \tfrac{3}{5} = \tfrac{3}{5}$; together $\ans{2\tfrac{3}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Subtract $3\tfrac{1}{4} - 1\tfrac{3}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1\tfrac{1}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} in a word problem, write the unit with your answer.
An answer of $2\tfrac{1}{2}$ by itself does not say whether it means meters,
kilograms, or cups.
\end{watchoutbox}
