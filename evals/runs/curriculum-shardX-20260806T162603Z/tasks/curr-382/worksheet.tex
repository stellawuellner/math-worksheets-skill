\wstitleblock{Finding and Verifying Inverse Functions}{Algebra 2}{}

\begin{center}
\small\textit{Finding inverses of linear, rational, radical and power rules, then checking them by composition}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item To find an inverse rule, write $x = f(y)$ and then solve for $y$; what you get is $f^{-1}(x)$.
  \item Leave every answer as an exact expression --- no decimals, no rounding.
  \item Write the algebra that produced the answer, not just the answer.
\end{itemize}

\vspace{0.1cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[5cm]{Find the inverse rule for $f(x) = 5x - 8$.}

\problem[5cm]{Find the inverse rule for $g(x) = \dfrac{x}{4} + 3$.}

\problem{Let $p(x) = 2x + 9$ and $q(x) = \dfrac{x - 9}{2}$.
\par\smallskip
(a)~Simplify $p(q(x))$ completely.
\par\vspace*{3cm}\ansline
\par\smallskip
(b)~Explain what the part (a) result tells you about how $p$ and $q$ are related.
\par\vspace*{2.6cm}\ansline}

\problem[5.5cm]{Find the inverse rule for $r(x) = \dfrac{2x + 1}{x - 3}$.}

\problem[5.5cm]{Find the inverse rule for $s(x) = \sqrt{x + 7} - 2$, whose inputs satisfy $x \ge -7$.}

\problem[5cm]{For $m(x) = 4x + 3$, find $m^{-1}(19)$.}

\problem[5.5cm]{Find the inverse rule for $t(x) = \dfrac{3 - 2x}{5}$.}

\problem[6cm]{Find the inverse rule for $u(x) = \dfrac{4}{x + 6} + 1$.}

\problem[6cm]{Let $w(x) = (x - 4)^3 + 2$. A classmate claims that $w^{-1}(x) = \sqrt[3]{x - 2} + 4$. Test the claim by simplifying $w\bigl(w^{-1}(x)\bigr)$, and write the simplified result.}

\problem{Let $n(x) = 7 - 3x$.
\par\smallskip
(a)~Find $n^{-1}(x)$.
\par\vspace*{3cm}\ansline
\par\smallskip
(b)~Simplify $n\bigl(n^{-1}(x)\bigr)$ to confirm your rule.
\par\vspace*{2.6cm}\ansline}

\problem{Let $v(x) = x^2 - 6x + 5$, with inputs restricted to $x \ge 3$.
\par\smallskip
(a)~Confirm the claim $v^{-1}(x) = 3 + \sqrt{x + 4}$ by simplifying $v\bigl(v^{-1}(x)\bigr)$.
\par\vspace*{3.4cm}\ansline
\par\smallskip
(b)~Explain why the restriction $x \ge 3$ has to be stated for $v$ to have an inverse at all.
\par\vspace*{2.8cm}\ansline}

\problem{Let $f(x) = \dfrac{x - 3}{2x + 1}$.
\par\smallskip
(a)~Find $f^{-1}(x)$.
\par\vspace*{3.6cm}\ansline
\par\smallskip
(b)~One input value has to be excluded from the domain of $f^{-1}$. Name it, and explain what goes wrong there.
\par\vspace*{2.8cm}\ansline}
