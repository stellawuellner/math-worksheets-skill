\sstitleblock{Differentiating Accumulation Functions with the FTC}

\skillheading{1. The basic FTC derivative (ftc-part2-basic)}

\begin{formulabox}
If $f$ is continuous and $\displaystyle g(x) = \int_a^{x} f(t)\,dt$ with $a$ constant, then
\[ g'(x) = f(x). \]
The integrand \textbf{is} the derivative --- do not evaluate the integral first. Two variants
to recognise:\\[2pt]
$\displaystyle \int_{x}^{b} f(t)\,dt$ has derivative $-f(x)$ (flip the limits, pick up the
minus sign), and $\displaystyle \int_a^{x} f(t)\,dt$ always satisfies $g(a) = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $g(x) = \int_2^{x}\left(t^2 + 1\right)dt$. Find $g'(3)$.
\step{The lower limit is constant and the upper limit is $x$ alone, so this is the plain FTC
form and the derivative is just the integrand.}
\step{$g'(x) = x^2 + 1$, so $g'(3) = 3^2 + 1 = \ans{10}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $g(x) = \int_5^{x}\left(2t - 1\right)dt$. Find $g'(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{2. The chain-rule form (ftc-chain)}

\begin{formulabox}
If $\displaystyle g(x) = \int_a^{u(x)} f(t)\,dt$, then
\[ g'(x) = f\!\left(u(x)\right) \cdot u'(x). \]
Substitute the whole upper limit into $f$, \textbf{then} multiply by its derivative. Missing
the factor $u'(x)$ is the standard error --- if the upper limit is anything other than a bare
$x$, the factor is there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $g(x) = \int_0^{x^2}\left(t + 1\right)dt$. Find $g'(3)$.
\step{The upper limit is $u(x) = x^2$, not $x$, so the chain-rule form applies: evaluate the
integrand at $u(x)$ and multiply by $u'(x) = 2x$.}
\step{$g'(x) = \left(x^2 + 1\right)\cdot 2x$, so $g'(3) = (9 + 1)(6) = \ans{60}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $g(x) = \int_1^{x^3} 2t\,dt$. Find $g'(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{192}$}
\end{tryitbox}

\skillheading{3. Evaluating accumulated change (accumulation-value)}

\begin{formulabox}
When the question asks \emph{how much} has accumulated (a volume, a distance, a total), you
do evaluate --- FTC Part 2:
\[ \int_a^{b} f(t)\,dt = F(b) - F(a), \qquad F' = f. \]
Rate question $\rightarrow$ differentiate (Skill 1). Total question $\rightarrow$ integrate.
Units follow the same rule: (units of $f$) $\times$ (units of $t$).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Water enters at $r(t) = 4t + 2$ L/min. How much enters in the first
$3$ minutes?
\step{The question asks for a total, not a rate, so evaluate the accumulation
$\int_0^{3} r(t)\,dt$ rather than differentiating it.}
\step{$F(t) = 2t^2 + 2t$, so the total is $F(3) - F(0) = (18 + 6) - 0 = \ans{24}$ litres.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\int_0^{4}\left(3t + 1\right)dt$ (litres, if $3t+1$ is
in L/min).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{4. Reading $g$ from $f$ (accumulation-analysis)}

\begin{formulabox}
Because $g' = f$, every question about the \emph{shape} of $g$ is a sign question about $f$:\\[2pt]
$f > 0 \Rightarrow g$ increasing \quad $\cdot$ \quad $f < 0 \Rightarrow g$ decreasing \quad
$\cdot$ \quad $f$ changes $+$ to $-$ at $c \Rightarrow g$ has a maximum at $c$\\[2pt]
$f$ increasing $\Rightarrow g$ concave up. Justify with ``$g' = f$ by the FTC'', then argue
from the sign of $f$ --- the sign change, not the zero, is what creates the extremum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(t) = t - 4$ and $g(x) = \int_0^{x} f(t)\,dt$ on
$[0,7]$. Where is $g$ smallest?
\step{A minimum of $g$ needs $g'$ to change from negative to positive, and $g' = f$ by the
FTC, so find where $f$ changes sign.}
\step{$f(t) = 0$ at $t = \ans{4}$; $f < 0$ before it and $f > 0$ after, so $g$ decreases then
increases and its minimum is at $x = 4$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(t) = 6 - 2t$, $g(x) = \int_0^{x} f(t)\,dt$ on
$[0,5]$. Where is $g$ largest?\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $g(x) = \int_a^x f(t)\,dt$ is a function of $x$ only --- $t$ is a
dummy variable and can never appear in the answer. And a variable \emph{lower} limit flips
the sign: $\frac{d}{dx}\int_x^b f = -f(x)$.
\end{watchoutbox}
