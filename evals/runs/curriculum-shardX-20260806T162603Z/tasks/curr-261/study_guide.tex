\sstitleblock{Interpreting the Solution of a System on a Graph}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading the solution off a graph}

\begin{formulabox}
\textbf{What a solution is:} an ordered pair $(x, y)$ that makes \emph{both} equations true. On a
graph that is the point where the two lines cross --- the only point lying on both. A point on just
one line solves that equation alone, not the system. Always write the answer as a pair
$(x, y)$ and check it in \emph{both} equations.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Read the solution of $y = x + 2$ and $y = -2x + 8$ from their graph.
\step{A solution has to sit on both lines, so I look for the single crossing point rather than solving anything.}
\step{The lines cross at $x = 2$; substituting gives $y = 2 + 2 = 4$ on the first and $y = -4 + 8 = 4$ on the second, so both agree.}

$\Rightarrow\quad \ans{(2,\,4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Where do $y = x + 1$ and $y = -x + 7$ cross?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\,4)}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- One solution, none, or infinitely many}

\begin{formulabox}
\textbf{Read it off the slopes and intercepts:}
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=2pt]
  \item Different slopes: the lines cross once --- \emph{exactly one solution}.
  \item Same slope, different $y$-intercepts: parallel, never meeting --- \emph{no solution}.
  \item Same slope \emph{and} same intercept: one line drawn twice --- \emph{infinitely many}.
\end{itemize}
Put both equations in $y = mx + b$ form before comparing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many solutions has $y = 3x + 2$ and $y = 3x - 4$?
\step{Both are already in slope-intercept form, so I compare slopes first --- that is what decides the number of crossings.}
\step{The slopes are both $3$, so the lines are parallel; the intercepts $2$ and $-4$ differ, so they are not the same line.}
\step{Parallel and distinct lines never meet.}

$\Rightarrow\quad \ans{\text{no solution}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many solutions has $y = -x + 5$ and $y = -x + 1$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{no solution}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Graphing a system to solve it}

\begin{formulabox}
\textbf{How:} rewrite each equation as $y = mx + b$, plot the $y$-intercept, count the slope as
rise over run to a second point, and draw the line. Do that for both, then read the crossing point.
A graphical reading is exact only when the crossing lands on a lattice point --- otherwise use it
as an estimate and confirm by substitution.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $y = -x + 6$ and $2y = x$ by graphing.
\step{Neither equation is in a form I can plot directly from a second point, so I put both into slope-intercept form first.}
\step{The first is already $y = -x + 6$: intercept $6$, slope $-1$. The second becomes $y = \frac{1}{2}x$: through the origin, slope $\frac{1}{2}$.}
\step{The lines meet where $-x + 6 = \frac{1}{2}x$, so $x = 4$ and $y = 2$ --- a lattice point, so the graph can be read exactly.}

$\Rightarrow\quad \ans{(4,\,2)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = -x + 5$ and $y = 2x - 1$ by graphing.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,\,3)}$}
\end{tryitbox}
