\sstitleblock{Rigid Motions on the Coordinate Plane}

\begin{formulabox}
\textbf{A rigid motion moves the plane without changing any distance.} Translations, reflections and rotations are all rigid motions, so an image is always congruent to its original --- lengths and angles survive, only position changes.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Translations}

\begin{formulabox}
\textbf{Rule:} $(x, y) \to (x + a,\ y + b)$. Positive $a$ moves right, negative $a$ moves left; positive $b$ moves up, negative $b$ moves down.\par
The two coordinates never interact --- work on each one separately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Translate $P(3, -4)$ by $(x, y) \to (x - 5,\ y + 2)$.
\step{A translation adds a fixed amount to each coordinate on its own, so this is two small sums rather than one combined step.}
\step{$x' = 3 - 5$ \quad and \quad $y' = -4 + 2$}

$\Rightarrow\quad P'\,($ $\ans{-2}$ $,$ $\ans{-2}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Translate $(-1, 3)$ by $(x, y) \to (x + 7,\ y - 9)$ and give the $x$-coordinate of the image.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Reflections}

\begin{formulabox}
\textbf{The five you need:} in the $x$-axis $(x, -y)$; in the $y$-axis $(-x, y)$; in $y = x$ $(y, x)$; in $y = -x$ $(-y, -x)$; in the vertical line $x = c$ $\ (2c - x,\ y)$.\par
\textbf{Check:} the mirror line always sits exactly halfway between a point and its image, so the point and its image are the same distance from it on opposite sides.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect $(12, 7)$ in the vertical line $x = 5$.
\step{The mirror is vertical, so only the horizontal position moves --- the $y$-coordinate is untouched and the rule to use is the $x = c$ one.}
\step{$x' = 2(5) - 12 = 10 - 12$}

$\Rightarrow\quad x' = $ $\ans{-2}$, and the $y$-coordinate stays $7$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Reflect $(10, -6)$ in the line $x = 3$ and give the $x$-coordinate of the image.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-4}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Rotations}

\begin{formulabox}
\textbf{About the origin:} quarter turn counterclockwise $(x, y) \to (-y,\ x)$; half turn $(x, y) \to (-x,\ -y)$; quarter turn clockwise $(x, y) \to (y,\ -x)$.\par
\textbf{About any centre $(h, k)$, quarter turn counterclockwise:} $(x, y) \to (h - (y - k),\ k + (x - h))$. Measure the point's offset from the centre, turn that offset, and add it back on.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Turn $(4, 6)$ a quarter turn counterclockwise about $(2, 1)$.
\step{The centre is not the origin, so the origin rules do not apply directly --- use the centre form, which turns the point's offset from the centre.}
\step{$x' = 2 - (6 - 1) = 2 - 5$ \quad and \quad $y' = 1 + (4 - 2) = 1 + 2$}

$\Rightarrow\quad$ image $= ($ $\ans{-3}$ $,$ $\ans{3}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Turn $(5, 8)$ a quarter turn counterclockwise about the origin and give the $x$-coordinate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} reflecting in $y = x$ swaps the coordinates but changes no signs; reflecting in $y = -x$ swaps them \emph{and} changes both signs. And a rotation about a point other than the origin is never just a sign change --- go through the offset form.
\end{watchoutbox}
