\problem{Simplify $\dfrac{x^2-4}{x^2+5x+6}$.}

\textbf{Solution:} $\dfrac{(x-2)(x+2)}{(x+2)(x+3)}$; the shared factor $(x+2)$
cancels. \ans{\dfrac{x-2}{x+3}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$f(x) = \dfrac{x-4}{x+1}$. (a) $x$-intercept. (b) $y$-intercept.}

\textbf{Solution (a):} A quotient is zero only when its numerator is zero (and
the denominator is not): $x - 4 = 0$. \ans{x = 4}

\textbf{Solution (b):} $f(0) = \dfrac{0-4}{0+1}$. \ans{-4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$g(x) = \dfrac{x^2-x-6}{x^2-9}$. (a) Simplify. (b) Vertical asymptote.}

\textbf{Solution (a):} $\dfrac{(x-3)(x+2)}{(x-3)(x+3)}$, and $(x-3)$ cancels.
\ans{\dfrac{x+2}{x+3}}

\textbf{Solution (b):} Only the surviving denominator factor can produce an
asymptote: $x + 3 = 0$. The cancelled value $x = 3$ is a hole, not an asymptote.
\ans{x = -3}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Horizontal asymptote of $h(x) = \dfrac{3x^2-x}{x^2+4}$.}

\textbf{Solution:} Numerator and denominator both have degree $2$, so divide
every term by $x^2$:
\[ \frac{3x^2-x}{x^2+4} = \frac{3 - \frac{1}{x}}{1 + \frac{4}{x^2}}
   \longrightarrow \frac{3}{1} \quad\text{as } |x| \text{ grows.} \]
\ans{y = 3}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$u(x) = \dfrac{x^2-9}{x^2-x-12}$. (a) $x$-intercepts. (b) $y$-intercept.}

\textbf{Solution (a):} $\dfrac{(x-3)(x+3)}{(x-4)(x+3)} = \dfrac{x-3}{x-4}$. The
factor $(x+3)$ cancelled, so $x = -3$ is a hole, not an intercept --- the graph
has no point there at all. The only $x$-intercept comes from $x - 3 = 0$.
\ans{x = 3}

\textbf{Solution (b):} $u(0) = \dfrac{0-9}{0-0-12} = \dfrac{-9}{-12}$.
\ans{\tfrac{3}{4}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$v(x) = \dfrac{x^3-9x}{x^2+2x-15}$. (a) Simplify. (b) Excluded values.}

\textbf{Solution (a):} $\dfrac{x(x-3)(x+3)}{(x+5)(x-3)}$; cancel $(x-3)$.
\ans{\dfrac{x(x+3)}{x+5}}

\textbf{Solution (b):} The domain is set by the original denominator:
$(x+5)(x-3) = 0$. \ans{x = -5 \text{ or } x = 3}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$w(x) = \dfrac{x^2-5x+6}{x^2-4}$. (a) Height of the hole.
(b) Vertical asymptote.}

\textbf{Solution (a):} $\dfrac{(x-2)(x-3)}{(x-2)(x+2)} = \dfrac{x-3}{x+2}$, so the
hole sits at $x = 2$. Its height is the value of the cancelled form there:
$\dfrac{2-3}{2+2}$. \ans{-\tfrac{1}{4}}

\textbf{Solution (b):} The surviving factor is $(x+2)$. \ans{x = -2}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Horizontal asymptote of $r(x) = \dfrac{2x+7}{5x-1}$.}

\textbf{Solution:} Equal degrees (both $1$), so the ratio of the leading
coefficients decides the level: $\dfrac{2}{5}$. \ans{y = \tfrac{2}{5}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$s(x) = \dfrac{2x^2-8}{x^2+1}$. (a) $x$-intercepts. (b) $y$-intercept.}

\textbf{Solution (a):} $2x^2 - 8 = 2(x-2)(x+2) = 0$, and the denominator
$x^2+1$ is never zero, so nothing cancels. \ans{x = -2 \text{ or } x = 2}

\textbf{Solution (b):} $s(0) = \dfrac{-8}{1}$. \ans{-8}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$t(x) = \dfrac{x^2-16}{x^3-4x^2}$. (a) Simplify. (b) Excluded values.}

\textbf{Solution (a):} $\dfrac{(x-4)(x+4)}{x^2(x-4)}$; cancel $(x-4)$.
\ans{\dfrac{x+4}{x^2}}

\textbf{Solution (b):} $x^3 - 4x^2 = x^2(x-4) = 0$ gives
\ans{x = 0 \text{ or } x = 4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{$k(x) = \dfrac{4x^2+1}{2x^2-3x}$. (a) Horizontal asymptote.
(b) Vertical asymptotes. (c) What the degrees tell you.}

\textbf{Solution (a):} Equal degrees, so the level is the ratio of the leading
coefficients, $\dfrac{4}{2}$. \ans{y = 2}

\textbf{Solution (b):} $2x^2 - 3x = x(2x-3) = 0$, and neither factor cancels with
$4x^2+1$. \ans{x = 0 \text{ or } x = \tfrac{3}{2}}

\textbf{Solution (c):} \emph{Instructor-judged.} Both degrees are $2$. Equal
degrees mean the graph levels off, and the height it levels off at is the ratio
of the \emph{leading} coefficients, $4/2 = 2$ --- the constant terms play no part.
(If the top degree were smaller the level would be $y=0$; if larger there would
be no horizontal asymptote.) Full credit needs the equal-degree observation and
the leading-coefficient ratio.

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.2cm}

\problem{Synthesis: $F(x) = \dfrac{x^3-x^2-6x}{x^2-9}$. (a) Simplify.
(b) $x$-intercepts. (c) Height of the hole. (d) Vertical asymptote.
(e) Why $x=3$ is not an asymptote.}

\textbf{Solution (a):} Factor both parts:
\[ \frac{x(x-3)(x+2)}{(x-3)(x+3)} , \]
and cancel $(x-3)$. \ans{\dfrac{x(x+2)}{x+3}}

\textbf{Solution (b):} Set the simplified numerator to zero: $x(x+2) = 0$.
\ans{x = 0 \text{ or } x = -2}

\textbf{Solution (c):} The cancelled factor puts the hole at $x = 3$, at height
$\dfrac{3(3+2)}{3+3} = \dfrac{15}{6}$. \ans{\tfrac{5}{2}}

\textbf{Solution (d):} The surviving denominator factor is $(x+3)$.
\ans{x = -3}

\textbf{Solution (e):} \emph{Instructor-judged.} The factor $(x-3)$ divides out
of both the numerator and the denominator. After cancelling, nothing in the
denominator vanishes at $x = 3$, so the values stay finite as $x$ approaches $3$
--- the graph simply has one point missing there, at $\bigl(3, \tfrac{5}{2}\bigr)$.
Only $(x+3)$, which survives the cancelling, sends the graph off to infinity.
