\sstitleblock{Division: Group Size or Group Count?}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Sharing equally (find the group size)}

\begin{formulabox}
\textbf{Rule:}\quad total $\div$ number of groups $=$ number \textbf{in each} group.\par
The story gives you \emph{how many groups}; your answer counts what lands in each one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 21 grapes are shared equally onto 3 plates. How many grapes on each plate?
\step{The 3 counts plates (groups), so divide the total by it to get the size of each share.}
\step{$21 \div 3 = 7$, because $3 \times 7 = 21$.}

$\Rightarrow\quad\ans{7}$ grapes on each plate
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 32 crackers are shared equally onto 4 plates. How many crackers on each plate?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Making groups (find the group count)}

\begin{formulabox}
\textbf{Rule:}\quad total $\div$ number in each group $=$ number \textbf{of} groups.\par
The story gives you the \emph{size} of a group; your answer counts how many groups you can make.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 24 eggs are packed 6 to a carton. How many cartons are filled?
\step{The 6 is the size of each group, so divide the total by it to count the groups.}
\step{$24 \div 6 = 4$, because $6 \times 4 = 24$.}

$\Rightarrow\quad\ans{4}$ cartons
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 45 bouncy balls are packed 9 to a bin. How many bins are filled?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Checking what your answer counts}

\begin{formulabox}
\textbf{Check yourself:}\quad After dividing, ask: \emph{does my number count groups, or things in a group?}\par
If your answer is a number the story already gave you, you probably repeated a given instead of dividing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 36 pears are shared equally into 4 baskets. A student answers: ``4 pears in each basket.'' Fix it.
\step{The 4 is the number of baskets --- a given, not the answer. The question asks for the size of each share, so divide.}
\step{$36 \div 4 = 9$, because $4 \times 9 = 36$.}

$\Rightarrow\quad\ans{9}$ pears in each basket, not 4
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 40 stickers are shared equally onto 5 pages. A student answers ``5 stickers per page.'' What is the correct number per page?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}
