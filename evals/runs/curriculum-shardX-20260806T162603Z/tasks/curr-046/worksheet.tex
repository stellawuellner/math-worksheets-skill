\wstitleblock{Shape Detective: Any Way It Turns}{Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item A shape keeps its name no matter how it is turned or tilted.
  \item Count the straight sides and corners first --- they tell you the name.
\end{itemize}

\problem[2.5cm]{Look at this shape:\par
\begin{center}\begin{tikzpicture}[scale=0.9]
\draw[thick] (0,0) -- (2,0) -- (1,1.4) -- cycle;
\end{tikzpicture}\end{center}
(a) It has \ansblank\ straight sides.\par
(b) Its name is \ansblank.}

\problem[2.5cm]{This shape is tilted so it stands on a corner:\par
\begin{center}\begin{tikzpicture}[scale=0.8]
\draw[thick] (0,0) -- (1.1,1.1) -- (0,2.2) -- (-1.1,1.1) -- cycle;
\end{tikzpicture}\end{center}
(a) It has \ansblank\ sides, all the same length, and square corners.\par
(b) Even tilted, its name is still \ansblank.}

\problem[2.5cm]{This shape is tipped over:\par
\begin{center}\begin{tikzpicture}[scale=0.8]
\draw[thick, rotate=30] (0,0) rectangle (2.4,1.2);
\end{tikzpicture}\end{center}
(a) It has \ansblank\ sides.\qquad
(b) It has \ansblank\ corners.\par
(c) Its name is \ansblank.}

\problem[2.5cm]{This shape points down instead of up:\par
\begin{center}\begin{tikzpicture}[scale=0.9]
\draw[thick] (0,1.4) -- (2,1.4) -- (1,0) -- cycle;
\end{tikzpicture}\end{center}
(a) It has \ansblank\ corners.\par
(b) Its name is \ansblank.}

\problem[3cm]{The hexagon has 6 sides. The square has 4 sides.\par
\begin{center}\begin{tikzpicture}[scale=0.7]
\draw[thick, rotate=15] (0:1) -- (60:1) -- (120:1) -- (180:1) -- (240:1) -- (300:1) -- cycle;
\begin{scope}[xshift=3cm]
\draw[thick] (0,-0.7) rectangle (1.4,0.7);
\end{scope}
\end{tikzpicture}\end{center}
How many more sides does the hexagon have than the square? \ansblank}

\problem[3.5cm]{Ben looks at this tilted shape and says, ``That is a diamond,
not a square!''\par
\begin{center}\begin{tikzpicture}[scale=0.8]
\draw[thick, rotate=25] (0,0) rectangle (1.5,1.5);
\end{tikzpicture}\end{center}
(a) Count its sides: \ansblank\par
(b) Is Ben right? Say why or why not.}

\problem[3cm]{Kim drew a shape with 3 straight sides. She made it very long,
very skinny, and very tilted.\par
(a) How many corners must her shape have? \ansblank\par
(b) What is the name of her shape? \ansblank}

\problem[4cm]{Challenge!\par
\begin{center}\begin{tikzpicture}[scale=0.7]
\draw[thick, rotate=35] (0,0) rectangle (1.4,1.4);
\begin{scope}[xshift=3cm]
\draw[thick] (0,0) -- (1.8,0.4) -- (0.4,1.5) -- cycle;
\end{scope}
\end{tikzpicture}\end{center}
The square has 4 corners and the triangle has 3 corners.\par
(a) How many corners do the two shapes have all together? \ansblank\par
(b) Does turning a shape change its name? Say why or why not.}
