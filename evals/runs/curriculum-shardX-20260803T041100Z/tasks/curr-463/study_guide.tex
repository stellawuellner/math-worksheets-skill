\sstitleblock{Second and Higher Derivatives}

\skillheading{1. Climbing the ladder: repeated differentiation}

\begin{formulabox}
\textbf{Definition:} $f'' = (f')'$, \ $f''' = (f'')'$, \ $f^{(n)} = (f^{(n-1)})'$
--- each rung is the derivative of the one above.\\[3pt]
\textbf{Polynomials:} each rung lowers the degree by one, so a degree-$n$
polynomial has $f^{(n)}$ constant and $f^{(n+1)} = 0$.\\[3pt]
\textbf{Simplify between rungs.} An unsimplified derivative costs you twice on
the next one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $f''(x)$ for $f(x) = x^4 - 5x^2$.
\step{Only the power rule is needed, so take one rung at a time and simplify
before differentiating again.}
\step{$f'(x) = 4x^3 - 10x$, then differentiate that.}
\step{$f''(x) = \ans{12x^2 - 10}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $g'''(x)$ for $g(x) = x^5 + 2x^3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{60x^2 + 12}$}
\end{tryitbox}

\skillheading{2. Second derivatives with the product and chain rules}

\begin{formulabox}
\textbf{The rules do not stop at the first rung.} Whatever rule built $f'$,
$f'$ is a new function and needs its own rule.\\[3pt]
\textbf{Chain rule twice:} for $f = \sin(kx)$, each differentiation contributes
a factor of $k$, so $f'' = -k^2\sin(kx)$.\\[3pt]
\textbf{Product rule:} factor the common exponential or trig piece out of $f'$
first --- it turns two product-rule applications into one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $f''(x)$ for $f(x) = xe^x$.
\step{$f$ is a product, so $f'$ needs the product rule --- and $f'$ is another
product, so the second rung needs it again.}
\step{$f'(x) = e^x + xe^x = e^x(x+1)$, so
$f''(x) = e^x(x+1) + e^x(1) = e^x(x+2)$.}
\step{$f''(x) = \ans{(x+2)e^x}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $f''(x)$ for $f(x) = \cos(4x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-16\cos(4x)}$}
\end{tryitbox}

\skillheading{3. Implicit second derivatives}

\begin{formulabox}
\textbf{Two rungs, same rule.} Differentiate the equation implicitly to get
$\frac{dy}{dx}$, then differentiate \emph{that} --- and every $y$ inside it is
still a function of $x$, so it carries another $\frac{dy}{dx}$.\\[3pt]
\textbf{Finish by substituting:} replace $\frac{dy}{dx}$ with the expression
from the first rung, then use the original equation to simplify.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $x^2 + y^2 = 9$ on the upper arc, find
$\frac{d^2y}{dx^2}$ in terms of $x$.
\step{The equation is not solved for $y$, so differentiate implicitly, then
differentiate the result and substitute back.}
\step{$2x + 2y\frac{dy}{dx} = 0 \Rightarrow \frac{dy}{dx} = -\frac{x}{y}$;
differentiating again gives $-\frac{y^2 + x^2}{y^3} = -\frac{9}{y^3}$.}
\step{On the upper arc $y = \sqrt{9-x^2}$, so
$\frac{d^2y}{dx^2} = \ans{\dfrac{-9}{(9-x^2)^{3/2}}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $x^2 + y^2 = 16$ on the upper arc, find
$\frac{d^2y}{dx^2}$ in terms of $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{-16}{(16-x^2)^{3/2}}}$}
\end{tryitbox}

\skillheading{4. What the second derivative means}

\begin{formulabox}
\textbf{Motion:} $s(t)$ position, $s'(t) = v(t)$ velocity,
$s''(t) = a(t)$ acceleration.\\[3pt]
\textbf{Shape:} $f'' > 0$ concave up, $f'' < 0$ concave down. A sign change in
$f''$ is an inflection point.\\[3pt]
Setting $s''(t) = 0$ finds the instants where acceleration vanishes --- the
candidates for an inflection point on the position graph.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $s(t) = t^3 - 3t^2$ metres, find when the
acceleration is zero.
\step{Acceleration is the second rung, so differentiate twice and set that
equal to zero.}
\step{$s'(t) = 3t^2 - 6t$, so $a(t) = s''(t) = 6t - 6$; solving $6t - 6 = 0$
gives the time.}
\step{$\ans{t = 1}$ second, where the position graph changes concavity.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $s(t) = 2t^3 - 12t^2 + 5t$ metres, find when the
acceleration is zero.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 2}$}
\end{tryitbox}
