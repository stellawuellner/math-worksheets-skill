\sstitleblock{Postulates, Theorems, Converses, and Biconditionals}

\skillheading{Skill 1 --- Postulate or theorem, and using it}

\begin{formulabox}
\textbf{Postulate:} accepted without proof --- where the system starts (two points determine a line; the Segment and Angle Addition Postulates; the Linear Pair Postulate).\par
\textbf{Theorem:} proved from postulates, definitions and earlier theorems (vertical angles are congruent; supplements of the same angle are congruent).\par
\textbf{The test:} could someone write a proof of it? Then it is a theorem. Either way, both are legal reasons in a proof --- the difference is where they came from.\par
\textbf{Addition postulates:} $AB + BC = AC$ when $B$ is between $A$ and $C$; \ $m\angle ABD + m\angle DBC = m\angle ABC$ when $\overrightarrow{BD}$ is interior.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $B$ is between $A$ and $C$ with $AB = 3x + 2$, $BC = 2x - 4$, and $AC = 23$. Find $x$.
\step{Betweenness is exactly the hypothesis of the Segment Addition Postulate, so the postulate converts the picture into the single equation $AB + BC = AC$.}
\step{$(3x + 2) + (2x - 4) = 23$, so $5x - 2 = 23$ and $5x = 25$.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B$ is between $A$ and $C$ with $AB = 4x + 1$, $BC = x + 4$, $AC = 35$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Writing and testing a converse}

\begin{formulabox}
``If $P$ then $Q$'' \ $\longrightarrow$ \ converse ``if $Q$ then $P$''. Swap the two halves; change nothing else.\par
\textbf{A converse must be judged on its own.} A true statement can have a false converse (linear pair $\Rightarrow$ supplementary is true; supplementary $\Rightarrow$ linear pair is false). To disprove one, give a case meeting the new hypothesis that breaks the new conclusion.\par
\textbf{Converses do real work:} the forward theorem tells you what parallel lines \emph{give}; the converse is what lets you \emph{conclude} lines are parallel.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Corresponding angles measure $(3x + 10)^\circ$ and $(5x - 20)^\circ$. Find $x$ so the lines are parallel.
\step{To \emph{conclude} parallel lines I need the converse of the Corresponding Angles Postulate, whose hypothesis is that the two corresponding angles are congruent --- so I set the measures equal.}
\step{$3x + 10 = 5x - 20$, so $30 = 2x$.}

$\Rightarrow\quad \ans{x = 15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Corresponding angles measure $(2x + 25)^\circ$ and $(4x - 15)^\circ$. Find $x$ so the lines are parallel.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 20}$}
\end{tryitbox}

\skillheading{Skill 3 --- When a biconditional is allowed}

\begin{formulabox}
``$P$ if and only if $Q$'' asserts \textbf{both} ``if $P$ then $Q$'' and ``if $Q$ then $P$''. Write one only when you have checked both directions --- which is why \emph{definitions} are biconditional and most theorems are not.\par
Reading it in a problem: a biconditional may be used in either direction, so it both supplies a fact and licenses a conclusion.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An angle measures $(8x - 6)^\circ$. Find $x$ that makes it a right angle.
\step{``Right angle if and only if the measure is $90^\circ$'' runs backwards as well as forwards, so I may set the given measure equal to $90$ and conclude the angle is right.}
\step{$8x - 6 = 90$, so $8x = 96$.}

$\Rightarrow\quad \ans{x = 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An angle measures $(5x + 15)^\circ$. Find $x$ that makes it a right angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a true statement does not license its converse. Before writing ``if and only if'', test the backwards direction --- one counterexample kills it.
\end{watchoutbox}
