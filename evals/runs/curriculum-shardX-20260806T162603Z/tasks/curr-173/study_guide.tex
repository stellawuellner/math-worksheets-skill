\sstitleblock{Distribute, Combine, Factor}

\skillheading{Skill 1 --- The distributive property}

\begin{formulabox}
\textbf{Rule:} \quad $a(b + c) = ab + ac$ \quad --- multiply the outside factor by
\emph{every} term inside. Picture a rectangle of width $b + c$: the two smaller
rectangles $ab$ and $ac$ tile the whole area.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $5(x + 4)$
\step{The factor outside the parentheses must multiply every term inside, so distribute it to each term one at a time.}
\step{$5 \cdot x = 5x$ and $5 \cdot 4 = 20$, so the sum is $5x + 20$.}

$\Rightarrow\quad 5(x + 4) = \ans{5x + 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $7(x + 2)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7x + 14}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} with a negative factor, the sign distributes too ---
every term inside flips its sign.
\end{watchoutbox}

\skillheading{Skill 2 --- Combining like terms}

\begin{formulabox}
\textbf{Rule:} like terms have the \emph{same variable part}. Add or subtract their
coefficients and keep the variable part unchanged. A constant only combines with
another constant.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $6x + 3 + 2x - 8$
\step{Sort the terms into like groups first: the terms with $x$, and the plain numbers, each combine only with their own kind.}
\step{$x$-terms: $6x + 2x = 8x$. \ Constants: $3 - 8 = -5$.}

$\Rightarrow\quad 6x + 3 + 2x - 8 = \ans{8x - 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $9x + 2 - 4x + 6$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5x + 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Factoring out the GCF}

\begin{formulabox}
\textbf{Rule:} factoring \emph{undoes} distributing. Find the greatest common factor
of the terms, then write each term as GCF times something:
$ab + ac = a(b + c)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $8x + 20$
\step{Look for the largest number that divides both coefficients evenly; that number comes outside the parentheses.}
\step{The GCF of $8$ and $20$ is $4$: \ $8x = 4 \cdot 2x$ and $20 = 4 \cdot 5$.}

$\Rightarrow\quad 8x + 20 = \ans{4(2x + 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $10x + 15$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5(2x + 3)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} check your factoring by distributing back --- you should
land exactly on the expression you started with.
\end{watchoutbox}
