\sstitleblock{Quadratics: Choosing a Method and Reading the Roots}

\begin{formulabox}
\textbf{Decide before you solve.} Put the equation in the form
$ax^2 + bx + c = 0$ (or leave it as a perfect square), then:\par
\begin{itemize}[leftmargin=0.6cm, itemsep=0pt, topsep=2pt]
  \item no $x$ term, or the variable sits inside one square $\Rightarrow$
        \textbf{square roots}
  \item factors over the integers (check a common factor first) $\Rightarrow$
        \textbf{factoring}
  \item anything else, or you cannot see it quickly $\Rightarrow$
        \textbf{quadratic formula} --- it always works
\end{itemize}
Discriminant test: $\Delta = b^2 - 4ac$. A perfect-square $\Delta$ means it
factors; $\Delta > 0$ two real roots, $\Delta = 0$ one, $\Delta < 0$ none.
\end{formulabox}

\skillheading{1. Square roots (square-root-method)}

\begin{formulabox}
$(x - h)^2 = k \ \Rightarrow\ x - h = \pm\sqrt{k} \ \Rightarrow\ x = h \pm \sqrt{k}$.
\quad Always write \textbf{both} signs. Divide off the outside coefficient first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $(x - 1)^2 = 25$.
\step{The variable appears only inside one square and there is no separate $x$ term, so square roots is the shortest route.}
\step{$x - 1 = \pm 5$, so $x = 1 + 5$ or $x = 1 - 5$: \ $\ans{x = 6 \text{ or } x = -4}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2(x + 4)^2 = 50$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 1 \text{ or } x = -9}$}
\end{tryitbox}

\skillheading{2. Factoring (factoring-method)}

\begin{formulabox}
Move everything to one side, factor, then use the \textbf{zero-product
property}: if $AB = 0$ then $A = 0$ or $B = 0$.\par
For $x^2 + bx + c$: find two numbers with product $c$ and sum $b$.
Never divide both sides by $x$ --- that deletes the root $x = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 + 2x - 15 = 0$.
\step{It is already equal to zero and monic with small numbers, so hunt for a factor pair rather than reaching for the formula.}
\step{Product $-15$, sum $2$: use $5$ and $-3$, so $(x+5)(x-3) = 0$ and $\ans{x = 3 \text{ or } x = -5}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^2 - 7x + 12 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3 \text{ or } x = 4}$}
\end{tryitbox}

\skillheading{3. Quadratic formula (quadratic-formula)}

\begin{formulabox}
\[ x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a} \]
Works on every quadratic. Use it the moment the factor hunt stalls --- and
simplify the radical instead of rounding, unless the problem asks for a decimal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 - 6x + 4 = 0$.
\step{No integer pair multiplies to $4$ and adds to $-6$, so factoring is a dead end and the formula is the efficient choice.}
\step{$\Delta = 36 - 16 = 20$, so $x = \dfrac{6 \pm \sqrt{20}}{2} = \dfrac{6 \pm 2\sqrt{5}}{2}$, giving $\ans{x = 3 + \sqrt{5} \text{ or } x = 3 - \sqrt{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^2 + 2x - 4 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -1 + \sqrt{5} \text{ or } x = -1 - \sqrt{5}}$}
\end{tryitbox}

\skillheading{4. Reading roots in context (interpret-roots-in-context)}

\begin{formulabox}
Solve the equation first, then ask what each root \emph{means}. A root that
solves the equation can still be impossible in the story --- a negative length,
a negative time, a fraction of a person. Say which root answers the question and
why the other is rejected, and give the answer with its unit.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rectangle is $3$ m longer than it is wide and has area
$40$ square metres. Find its width $w$.
\step{Area $= w(w+3) = 40$ becomes $w^2 + 3w - 40 = 0$; the numbers are small, so factor, then test each root against the picture.}
\step{$(w+8)(w-5) = 0$ gives $\ans{w = 5 \text{ or } w = -8}$; a width cannot be negative, so the rectangle is $\ans{5 \text{ m}}$ wide}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A ball thrown up from the ground has height
$h = -16t^2 + 64t$ feet after $t$ seconds. How long is it in the air?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 4 \text{ seconds}}$}
\end{tryitbox}
