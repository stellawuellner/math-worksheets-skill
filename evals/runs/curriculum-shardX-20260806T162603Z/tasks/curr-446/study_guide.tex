\sstitleblock{Summarizing Distributions with Center, Spread and Shape}

\skillheading{Skill 1 --- Measures of center}

\begin{formulabox}
\textbf{Mean} $= \dfrac{\text{sum of the values}}{\text{how many values}}$ --- the balance point,
which every value pulls on. \quad \textbf{Median} $=$ the middle value of the sorted list
(the average of the two middle values when the count is even) --- a position, not a total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the mean of $3, 5, 8, 10, 14$.
\step{The mean shares the total equally, so add everything and divide by how many values there are.}
\step{$\dfrac{3+5+8+10+14}{5} = \dfrac{40}{5}$.}

$\Rightarrow\quad$ mean $= \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the mean of $2, 4, 9, 11, 14$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Measures of spread}

\begin{formulabox}
\textbf{Range} $=$ largest $-$ smallest: one number, easily wrecked by a single extreme value.
\quad \textbf{IQR} $= Q_3 - Q_1$, the width of the middle half. Split the sorted list at the
median, then take the median of each half. \quad \textbf{Standard deviation} is the typical
distance of a value from the mean.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the IQR of $4, 6, 8, 11, 15, 18, 20, 22$.
\step{The IQR describes the middle half, so split the sorted list in two and find the median of each half.}
\step{Lower half $4, 6, 8, 11$ has median $7$; upper half $15, 18, 20, 22$ has median $19$; the difference is $19 - 7$.}

$\Rightarrow\quad \text{IQR} = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the IQR of $1, 3, 6, 8, 12, 14, 17, 19$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

\skillheading{Skill 3 --- Shape, outliers, and which center to trust}

\begin{formulabox}
\textbf{Rule:} compare the mean with the median. Mean above median means a right tail
(right-skewed); mean below means a left tail; roughly equal means roughly symmetric.
An outlier is a value beyond $Q_1 - 1.5\,\text{IQR}$ or $Q_3 + 1.5\,\text{IQR}$. The median and
the IQR are \emph{resistant} --- an extreme value barely moves them --- so report those when a
distribution is skewed or has outliers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the median of $5, 6, 7, 8, 40$.
\step{The $40$ will drag the mean upward, so the median is the honest center here --- take the middle value of the sorted list.}
\step{Five values, so the middle one is the third: $5, 6, \underline{7}, 8, 40$. (The mean, by contrast, is $13.2$.)}

$\Rightarrow\quad$ median $= \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the median of $2, 3, 4, 5, 36$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 4 --- Reading a display}

\begin{formulabox}
\textbf{Rule:} a frequency table, bar chart or dot plot stores counts, not the original list.
The total is the sum of the counts; the tallest bar names the most common category. Translate
the display into the notation before computing: ``how many altogether'' is a sum, ``how many
more'' is a difference, ``what fraction'' is a count over the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tally of favourite colours reads Red $8$, Blue $14$, Green $6$. How many were surveyed?
\step{Each person is counted in exactly one category, so the number surveyed is the sum of the counts.}
\step{$8 + 14 + 6$.}

$\Rightarrow\quad$ total surveyed $= \ans{28}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pet tally reads Cats $9$, Dogs $17$, Fish $4$. How many pets altogether?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$}
\end{tryitbox}
