\wstitleblock{Signed Rational Numbers: Mixed Operations Review}{Grade 6--7 Mathematics}{}
\begin{center}
\textit{adding, subtracting, multiplying, and dividing signed rational numbers, with mixed operations}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item After the warm-up, the operations are mixed --- decide which sign rule applies before you compute.
  \item Show your work; write each answer in simplest form.
\end{itemize}

\problem[2.6cm]{Evaluate: \quad $-9 + 4$}

\problem[2.6cm]{Evaluate: \quad $4 - 11$}

\problem[2.6cm]{Evaluate: \quad $(-4)(-7)$}

\problem[3.2cm]{Evaluate: \quad $-\tfrac{2}{5} + \left(-\tfrac{1}{5}\right)$}

\problem[3.2cm]{Evaluate: \quad $-7 - (-9)$}

\problem[3.2cm]{Evaluate, writing the answer as a fraction in simplest form: \quad $-36 \div 8$}

\problem[3.6cm]{Evaluate: \quad $-\tfrac{5}{6} + \tfrac{1}{4}$}

\problem[3.2cm]{Evaluate: \quad $1.5 - (-2.25)$}

\problem[3.6cm]{Evaluate: \quad $\left(-\tfrac{2}{3}\right)\!\left(\tfrac{9}{8}\right)$}

\problem[3cm]{Write $<$ or $>$ in the blank to make a true statement:
\[ -\tfrac{7}{8} \;\; \raisebox{-2pt}{\ansblank} \;\; -0.9 \]}

\problem[4cm]{Evaluate: \quad $-2\left(\tfrac{3}{4} - \tfrac{5}{4}\right)$}

\problem[4.5cm]{\textbf{Challenge.} Evaluate: \quad
$\left(-\tfrac{3}{4} + \tfrac{1}{2}\right) \div \left(-\tfrac{5}{8}\right)$}
