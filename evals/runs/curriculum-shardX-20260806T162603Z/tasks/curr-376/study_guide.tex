\sstitleblock{Exponential Form and Logarithmic Form}

\skillheading{Skill 1 --- Exponential statement to logarithm}

\begin{formulabox}
\textbf{The one fact, two ways:} \quad
$b^{\,y} = x \iff \log_b x = y$ \quad (with $b > 0$, $b \ne 1$, $x > 0$).
The base $b$ never moves. The \emph{exponent} becomes the value of the logarithm,
and the \emph{result} becomes its input.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $3^4 = 81$ in logarithmic form.
\step{The statement already names a base, an exponent and a result, so the switch
is just a matter of putting each in its new place.}
\step{Base $3$ stays the base; the exponent $4$ becomes the value; the result
$81$ becomes the input.}

$\Rightarrow\quad \log_3 81 = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $2^6 = 64$ in logarithmic form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\log_2 64 = \ans{6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Logarithm back to a power}

\begin{formulabox}
\textbf{Method:} read $\log_b x = y$ out loud as ``$b$ to the power $y$ gives
$x$'', then write $b^{\,y} = x$ and evaluate. This is how you release an unknown
that is trapped inside a logarithm.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_4 x = 3$.
\step{The unknown is the \emph{input} of the logarithm, so switching to
exponential form puts it alone on one side.}
\step{$\log_4 x = 3$ becomes $4^3 = x$, and $4^3 = 64$.}

$\Rightarrow\quad \ans{x = 64}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_3 x = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 81}$}
\end{tryitbox}

\skillheading{Skill 3 --- Evaluating a logarithm}

\begin{formulabox}
\textbf{Ask the question the notation asks:} $\log_b x$ means ``what power of $b$
gives $x$?'' Build up the powers of $b$ until you reach $x$. The answer can be a
fraction (a root) or negative (a reciprocal).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\log_5 125$.
\step{Evaluating a logarithm is a search through the powers of its base, so list
them until the input appears.}
\step{$5^1 = 5$, $5^2 = 25$, $5^3 = 125$, so the exponent that works is $3$.}

$\Rightarrow\quad \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\log_8 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Solving by switching form}

\begin{formulabox}
\textbf{Method:} when the unknown is inside a logarithm, switch to exponential
form first, then solve the ordinary equation left behind. Check afterwards that
the logarithm's input really is positive at your answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_3(x+2) = 2$.
\step{The unknown sits inside the logarithm, so nothing algebraic can reach it
until the statement is written as a power.}
\step{$3^2 = x + 2$, so $9 = x + 2$. The input at $x=7$ is $9 > 0$, so the
answer stands.}

$\Rightarrow\quad \ans{x = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_5(2x-1) = 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 13}$}
\end{tryitbox}
