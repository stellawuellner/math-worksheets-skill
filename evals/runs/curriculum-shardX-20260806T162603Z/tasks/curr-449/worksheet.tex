\wstitleblock{Diagnosing Causation Claims and Biased Study Designs}{Precalculus / Statistics}{}

\begin{center}
\small Check the arithmetic first, then say what is wrong with the claim.
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Each problem reports a real-sounding claim. In every case the numbers are correct and the conclusion is not.
  \item Part (a) asks you to compute the figure the claim rests on. The later part asks you to name the flaw in a sentence or two.
  \item Give exact fractions in lowest terms unless a decimal is requested.
\end{itemize}

\vspace{0.15cm}

\problem[3.5cm]{A press release says: ``Sales of our new flavour \emph{doubled} this month ---
up $100\%$!'' The shop sold $4$ units last month and $8$ this month, out of $5000$ units of all
flavours sold this month.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What fraction of this month's total sales was the new flavour? \quad \ansblank
  \item[(b)] Explain why the headline is misleading even though the arithmetic is right.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{A club with $3000$ members posts a survey on its website. Of the $240$ members
who replied, $210$ said they like the new logo. The club announces: ``Seven-eighths of our
members like the new logo.''
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What proportion of the \emph{respondents} said they like it? \quad \ansblank
  \item[(b)] Explain why that proportion cannot be reported as a fact about all $3000$ members.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{Across the summer months, a town's weekly ice cream sales and its weekly count
of swimming-pool rescues have correlation $r = 0.90$. A blogger concludes that eating ice cream
makes people need rescuing.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute $r^2$ for these data. \quad \ansblank
  \item[(b)] Name the flaw in the blogger's causal claim.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{A reporter stands outside a gym and asks people leaving it whether they
exercise daily. Of the $60$ people asked, $45$ say yes. The story runs as ``Three-quarters of
townspeople exercise every day''.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What proportion of the people asked said yes? \quad \ansblank
  \item[(b)] Name the bias in this sample and say which way it pushes the result.
\end{itemize}
\vspace*{2.6cm}}

\problem[3.5cm]{Five students who chose to take a test-prep course scored $78, 80, 82, 84, 86$.
Five who did not take it scored $70, 72, 74, 76, 78$. The course advertises: ``Our students
score $8$ points higher.''
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find the mean score of the course group. \quad \ansblank
  \item[(b)] Find the mean score of the other group. \quad \ansblank
  \item[(c)] Explain why this $8$-point gap does not show that the course works, and describe the study that would.
\end{itemize}
\vspace*{2.8cm}}

\problem[3.5cm]{A drug trial reports: ``Our drug cuts the risk of the disease by $50\%$.'' In
the trial the risk fell from $2$ cases per $1000$ people to $1$ case per $1000$ people.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Compute the absolute change in risk, as an exact fraction. \quad \ansblank
  \item[(b)] Explain why the $50\%$ headline gives a misleading impression of the benefit.
\end{itemize}
\vspace*{2.8cm}}

\problem[3.5cm]{A gardener tests a fertilizer on a field of $400$ plants. She treats the $200$
plants on the sunny south half and leaves the $200$ on the shaded north half untreated. Of the
treated plants $150$ thrived; of the untreated plants $100$ thrived. She concludes the
fertilizer works.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] What proportion of the treated plants thrived? \quad \ansblank
  \item[(b)] Name the flaw in this design and describe how to fix it.
\end{itemize}
\vspace*{2.8cm}}

\problem[3.5cm]{\textbf{Synthesis.} A city reports hospital admissions of Nov $120$, Dec $210$,
Jan $260$, Feb $190$. Flu-shot campaigns run over the same four months, and a commentator
writes: ``Admissions rise exactly when flu shots are given --- the shots are sending people to
hospital.''
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find the total admissions over the four months. \quad \ansblank
  \item[(b)] Find the admissions in the busiest month. \quad \ansblank
  \item[(c)] Explain what is wrong with the commentator's reasoning, and state what evidence would be needed to settle the question.
\end{itemize}
