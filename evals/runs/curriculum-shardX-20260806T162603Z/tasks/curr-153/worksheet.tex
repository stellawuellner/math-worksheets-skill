\wstitleblock{Ratios with Tape Diagrams and Double Number Lines}{Grade 6}{}

\begin{center}
\textit{Tape diagrams for sharing and differences, and double number lines for ratio problems}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item A \textbf{tape diagram} shows a ratio as rows of equal boxes. Find what
        one box is worth first, then count boxes.
  \item A \textbf{double number line} shows two quantities that grow together.
        Matching marks always show the same ratio.
  \item Show your steps in the work space, and include units in your thinking.
\end{itemize}

\problem{The tape diagram shows the juice and water in a fruit punch. Every box
holds the same amount, and each box stands for 4 liters. How many liters of
juice are in the punch?\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.75]
  \node[left] at (-0.2,1.45) {\small juice};
  \fill[black!25] (0,1.1) rectangle (2,1.8);
  \foreach \i in {0,1} \draw[thick] (\i,1.1) rectangle (\i+1,1.8);
  \node[left] at (-0.2,0.35) {\small water};
  \foreach \i in {0,1,2} \draw[thick] (\i,0) rectangle (\i+1,0.7);
  \node[right] at (3.3,0.35) {\small each box $= 4$ liters};
\end{tikzpicture}
\end{center}
\par\vspace*{2.8cm}\answerline{L}}
