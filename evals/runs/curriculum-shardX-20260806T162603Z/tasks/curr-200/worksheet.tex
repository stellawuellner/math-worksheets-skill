\wstitleblock{Comparing Distributions and Compound Events}{Grades 6--7 Mathematics}{}

\begin{center}
\textit{Comparing centre and spread, reading a data display, and finding compound-event
probabilities}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Centre} says where a data set sits: the mean is the fair-share value, the
        median is the middle value once the data are in order.
  \item \textbf{Spread} says how stretched it is: the range is the largest value minus the
        smallest. Two sets can share a centre and still look nothing alike.
  \item \textbf{Compound events} happen together. Count the favourable outcomes and the
        total outcomes, then write the probability as a fraction in lowest terms.
  \item Leave probabilities as exact fractions, never as decimals.
\end{itemize}
