\wstitleblock{Subtracting Across a Zero}{Grades 2--3}{}

\begin{center}
\small subtracting across a zero in the tens place, subtracting from a number that ends in two zeros, estimating a difference to the nearest hundred, and finding a regrouping mistake
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Work right to left, ones first. A column that cannot give enough has to borrow from the column on its left.
  \item A zero has nothing to lend. Go one more column left, open a hundred into ten tens, and \emph{then} lend a ten to the ones.
  \item Whenever a column lends, cross out its digit and write the smaller number above it --- both times.
\end{itemize}
