\wstitleblock{Choosing and Applying a Convergence Test}{AP Calculus BC}{}

\begin{center}
\small Comparison, integral, ratio, and alternating-series tests
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Each problem names the test to use, or the structure of the terms makes it obvious. Do the computation the test asks for, then write the conclusion it licenses --- and only that conclusion.
  \item Factorials and fixed powers point to the ratio test; a rational expression in $n$ points to limit comparison; a term you can integrate points to the integral test; alternating signs point to the alternating series test and its remainder bound.
  \item Give exact answers unless a decimal is asked for.
\end{itemize}

\vspace{0.2cm}

\problem[5.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}+3}$, to be tested by limit comparison with $\displaystyle\sum \frac{1}{n^{2}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty}\frac{1/(n^{2}+3)}{1/n^{2}}$.
  \item[(b)] State the conclusion the limit comparison test licenses, and explain why the limit being positive matters.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty} n e^{-n^{2}}$, to be tested by the integral test with $f(x) = xe^{-x^{2}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Evaluate $\displaystyle\int_{1}^{\infty} xe^{-x^{2}}\,dx$ exactly.
  \item[(b)] State the conditions you checked before applying the test and the conclusion you reach. Explain why the value of the integral is not the sum of the series.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=0}^{\infty}\frac{2^{n}}{n!}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] The ratio of consecutive terms simplifies to $\dfrac{2}{n+1}$. Find its limit as $n \to \infty$.
  \item[(b)] This series has a familiar exact sum. Write it.
\end{itemize}}

\problem[6.5cm]{Consider the alternating harmonic series $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}{n}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty}\frac{1}{n}$, the limit of the term sizes.
  \item[(b)] Using the first $100$ terms as an estimate of the sum, give the exact error bound the alternating series remainder guarantees.
  \item[(c)] Justify both parts: name the two conditions the test requires and state the remainder bound you used.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{3n+1}{n^{3}+5}$, to be tested by limit comparison with $\displaystyle\sum\frac{1}{n^{2}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty}\frac{(3n+1)/(n^{3}+5)}{1/n^{2}}$.
  \item[(b)] State the conclusion, and explain how the comparison series was chosen.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{n!}{3^{n}}$. The ratio of consecutive terms simplifies to $\dfrac{n+1}{3}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the limit of that ratio as $n \to \infty$.
  \item[(b)] State the conclusion, and explain what the size of the terms themselves shows independently.
\end{itemize}}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=2}^{\infty}\frac{1}{n(\ln n)^{2}}$, to be tested by the integral test.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Evaluate $\displaystyle\int_{2}^{\infty}\frac{dx}{x(\ln x)^{2}}$ exactly, then give a decimal to three places.
  \item[(b)] Name the substitution you used, state the conditions you checked, and give the conclusion.
\end{itemize}}

\problem[6cm]{Consider $\displaystyle\sum_{n=0}^{\infty}\frac{(-1)^{n}}{2n+1}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] This series has a well-known exact sum. Write it.
  \item[(b)] Give the exact remainder bound guaranteed if the first five terms ($n = 0$ through $n = 4$) are used as an estimate.
\end{itemize}}

\problem[6.5cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{n^{2}\,3^{n}}{n!}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] The ratio of consecutive terms simplifies to $\dfrac{3(n+1)}{n^{2}}$. Find its limit.
  \item[(b)] Show the cancellation that produces that ratio, and state the conclusion.
\end{itemize}}

\problem[7cm]{Consider $\displaystyle\sum_{n=1}^{\infty}\frac{n+2}{n^{3}-n+4}$. A student begins with the ratio test and finds the limit of the ratio is $1$, which settles nothing.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\displaystyle\lim_{n\to\infty}\frac{n+2}{n^{3}-n+4}\cdot n^{2}$.
  \item[(b)] Write the exact sum of the comparison series $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}}$.
  \item[(c)] Explain why the ratio test could never have settled this series, and give the conclusion limit comparison reaches.
\end{itemize}}
