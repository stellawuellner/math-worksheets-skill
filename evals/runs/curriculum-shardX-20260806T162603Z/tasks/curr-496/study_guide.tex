\sstitleblock{Geometric Series, p-Series, and the nth-Term Test}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Geometric series}

\begin{formulabox}
\textbf{Test:} consecutive terms have a constant ratio $r$. \textbf{Verdict:} converges exactly when $|r| < 1$. \textbf{Value:} $\dfrac{\text{first term actually written}}{1 - r}$. The starting index matters only through that first term --- $\sum_{n=0}$ and $\sum_{n=1}$ of the same expression have different first terms and therefore different sums.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\sum_{n=1}^{\infty} 3\left(\frac{1}{4}\right)^{n}$.
\step{Consecutive terms differ by the constant factor one quarter, so this is geometric and the closed form applies; the only care needed is which term is first.}
\step{The sum starts at $n = 1$, so the first term is $3\cdot\frac{1}{4} = \frac{3}{4}$ and $r = \frac{1}{4}$, giving $\dfrac{3/4}{1 - 1/4}$.}

$\Rightarrow\quad \ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\sum_{n=1}^{\infty} 5\left(\frac{1}{3}\right)^{n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{2}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- p-series}

\begin{formulabox}
\textbf{Test:} the terms are $\dfrac{1}{n^{p}}$ for a constant $p$. \textbf{Verdict:} converges exactly when $p > 1$; diverges when $p \le 1$. The boundary case $p = 1$ is the harmonic series, which diverges even though its terms go to zero --- the standing reminder that shrinking terms are not enough. Values worth knowing: $p = 2$ gives $\frac{\pi^{2}}{6}$, $p = 4$ gives $\frac{\pi^{4}}{90}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Does $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{6}}$ converge, and to what?
\step{The terms are one over a fixed power of $n$, so this is a $p$-series and the exponent alone decides convergence.}
\step{Here $p = 6 > 1$, so it converges; its value is the standard even-exponent result $\frac{\pi^{6}}{945}$.}

$\Rightarrow\quad \ans{\frac{\pi^{6}}{945}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Does $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^{2}}$ converge, and to what?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{\pi^{2}}{6}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The nth-term test}

\begin{formulabox}
\textbf{One direction only.} If $\displaystyle\lim_{n\to\infty} a_{n} \neq 0$, then $\sum a_{n}$ diverges. If the limit \emph{is} zero, the test says nothing --- the series may converge or diverge, and the harmonic series shows both outcomes are possible. So use it as a cheap first screen: take the limit, and if it is not zero you are finished.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What does the nth-term test say about $\displaystyle\sum_{n=1}^{\infty}\frac{4n+3}{2n+1}$?
\step{Before reaching for any comparison, check the cheapest thing: whether the individual terms even shrink to zero.}
\step{Dividing top and bottom by $n$ gives $\dfrac{4 + 3/n}{2 + 1/n}$, whose limit is $\dfrac{4}{2}$ --- not zero, so the series diverges.}

$\Rightarrow\quad \ans{2}$, so the series diverges
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\lim_{n\to\infty}\frac{7n-1}{3n+5}$ and say what it tells you about the series.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{7}{3}}$, so it diverges}
\end{tryitbox}
