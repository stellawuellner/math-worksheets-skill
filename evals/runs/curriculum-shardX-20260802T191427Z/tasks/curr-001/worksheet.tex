\end{tabular}
\end{center}}

\problem[2.5cm]{Start at \textbf{56}. Count forward \textbf{4} more.
\begin{center}\Large
\begin{tabular}{|c|c|c|c|c|}\hline

\end{tabular}
\end{center}

\problem[2.5cm]{Start at \textbf{68} and count forward \textbf{3}. Say the numbers
