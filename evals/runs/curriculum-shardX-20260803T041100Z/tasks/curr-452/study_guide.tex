\sstitleblock{One-Sided, Infinite, and End-Behavior Limits}

\noindent\small Three different questions wear the same notation. Read the
arrow first: a superscript $\pm$ means \emph{pick a side}, a finite target with
a zero denominator means \emph{find a sign}, and $x \to \pm\infty$ means
\emph{compare degrees}.
\normalsize

\skillheading{1. One-sided limits}
\begin{formulabox}
\textbf{Rule:} $x \to c^{-}$ uses only values just below $c$; $x \to c^{+}$
only values just above. The two-sided limit exists exactly when both exist and
agree.\\[3pt]
This is how absolute values and piecewise rules get resolved: the side tells
you which branch, or which sign, is in force.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $\displaystyle \lim_{x \to 3^{-}} \frac{|x-3|}{x-3}$
\step{An absolute value cannot be simplified without knowing a sign, and the
$-$ superscript supplies it.}
\step{For $x < 3$, $x-3 < 0$, so $|x-3| = -(x-3)$ and the quotient is $-1$ for
every such $x$.}
\par
$\ans{-1}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $\displaystyle \lim_{x \to 4^{+}} \frac{|x-4|}{x-4}$
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

\skillheading{2. Infinite limits at a vertical asymptote}
\begin{formulabox}
\textbf{Rule:} nonzero numerator over a denominator going to $0$ is
$\pm\infty$. Get the sign from the denominator on the side you are on:
\[ \frac{(+)}{(0^{+})} \to \infty, \qquad \frac{(+)}{(0^{-})} \to -\infty \]
Test a nearby value such as $c + 0.01$ if the sign is not obvious. Never
report a number.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $\displaystyle \lim_{x \to 2^{+}} \frac{5}{x-2}$
\step{The numerator is a nonzero constant while the denominator collapses to
zero, so the answer is infinite and only the sign is in doubt.}
\step{Just above $2$, $x-2$ is a small positive number, and $5$ over a small
positive number is large and positive.}
\par
$\ans{\infty}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $\displaystyle \lim_{x \to 1^{-}} \frac{3}{x-1}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\infty}$}
\end{tryitbox}

\skillheading{3. End behavior as x goes to infinity}
\begin{formulabox}
\textbf{Rule for a rational function:} compare degrees.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item top degree $<$ bottom degree: limit is $0$
  \item degrees equal: limit is the ratio of leading coefficients
  \item top degree $>$ bottom degree: no finite limit ($\pm\infty$)
\end{itemize}
The safe method underneath all three: divide every term by the highest power
of $x$ in the denominator. Inside a square root that means dividing by $x^2$,
and $\sqrt{x^2} = -x$ when $x \to -\infty$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $\displaystyle \lim_{x \to \infty} \frac{5x^2-x}{2x^2+3}$
\step{Both parts have degree $2$, so the tails grow at the same rate and the
limit is the ratio of the leading coefficients.}
\step{Dividing through by $x^2$ gives
$\dfrac{5 - 1/x}{2 + 3/x^2} \to \dfrac{5}{2}$.}
\par
$\ans{\dfrac{5}{2}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad
$\displaystyle \lim_{x \to -\infty} \frac{6x^3+x}{3x^3-2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{4. Making a piecewise function continuous}
\begin{formulabox}
\textbf{Rule:} $f$ is continuous at $x = c$ when
\[ \lim_{x \to c^{-}} f(x) = \lim_{x \to c^{+}} f(x) = f(c) \]
To find an unknown constant, evaluate each branch at $c$, set the two
expressions equal, and solve. One equation, one unknown.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Find $a$ so that $f(x) = 3x + a$ for $x < 1$ and
$f(x) = x^2 + 5$ for $x \ge 1$ is continuous at $x = 1$.
\step{Continuity is exactly the statement that the two one-sided limits agree,
which turns the question into one equation.}
\step{Left: $3 + a$. \ Right: $1 + 5 = 6$. \ So $3 + a = 6$.}
\par
$\ans{a = 3}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find $c$ so that $f(x) = 2x + c$ for $x < 4$ and
$f(x) = x^2 - 3$ for $x \ge 4$ is continuous at $x = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c = 5}$}
\end{tryitbox}
