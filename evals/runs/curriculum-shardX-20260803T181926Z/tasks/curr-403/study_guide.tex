\sstitleblock{Finding Inverses with Domain Restrictions}

\begin{formulabox}
\textbf{When an inverse exists.} $f$ has an inverse function exactly when $f$
is \textbf{one-to-one} (each output comes from one input --- the horizontal
line test). If it is not, restrict the domain to a piece on which it is, and
say which piece you kept.\\[3pt]
\textbf{The swap that never changes:} domain of $f^{-1}$ $=$ range of $f$, and
range of $f^{-1}$ $=$ (restricted) domain of $f$. An inverse written without
its domain is only half an answer.\\[3pt]
\textbf{The procedure:} write $y = f(x)$, swap $x$ and $y$, solve for $y$, then
choose the branch or sign that the restriction forces.
\end{formulabox}

\skillheading{1. Invert a one-to-one function and state its domain}

\begin{formulabox}
Linear, radical and simple rational functions are already one-to-one, so no
restriction is needed --- but the domain still has to be read off the
\emph{range} of $f$. Undo the operations in reverse order: the last thing $f$
does is the first thing $f^{-1}$ undoes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x) = 2x + 9$. Find $f^{-1}$ and then $f^{-1}(15)$.
\step{A non-constant linear function is one-to-one on all of the reals, so
there is nothing to restrict --- go straight to the swap.}
\step{$y = 2x + 9$; swap and solve: $x = 2y + 9$, so
$f^{-1}(x) = \dfrac{x - 9}{2}$, defined for every real $x$. Then
$f^{-1}(15) = \dfrac{15 - 9}{2}$.}

$\Rightarrow\quad f^{-1}(15) = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x) = 5x - 2$. Find $f^{-1}$, then compute
$f^{-1}(18)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{2. Restrict a quadratic, then pick the branch}

\begin{formulabox}
A parabola fails the horizontal line test, so restrict at the \textbf{vertex}:
$x \geq h$ keeps the right branch, $x \leq h$ the left one. Complete the square
to find $h$.\\[3pt]
Solving for $y$ produces $\pm\sqrt{\ }$. The kept branch chooses the sign: the
outputs of $f^{-1}$ must lie in the restricted domain of $f$, so the right
branch takes $+$ and the left branch takes $-$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x) = (x - 1)^2$ restricted to $x \geq 1$. Find $f^{-1}$
and $f^{-1}(9)$.
\step{The vertex sits at $x = 1$ and the restriction keeps the right branch, so
the inverse's outputs must all be at least $1$ --- that will fix the sign.}
\step{$y = (x - 1)^2$; swap and solve: $y = 1 \pm \sqrt{x}$, and the plus sign
is the one with outputs above $1$. So $f^{-1}(x) = 1 + \sqrt{x}$ with domain
$x \geq 0$, and $f^{-1}(9) = 1 + 3$.}

$\Rightarrow\quad f^{-1}(9) = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x) = (x + 3)^2$ restricted to $x \geq -3$. Compute
$f^{-1}(25)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{3. Verify an inverse pair by composition}

\begin{formulabox}
Two functions are inverses only if \emph{both} compositions collapse:
$f(g(x)) = x$ and $g(f(x)) = x$, each on the appropriate domain. One direction
alone is not enough when a domain has been restricted.\\[3pt]
A single test value is a fast sanity check; the algebraic composition is the
proof.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Show $g(x) = \dfrac{x + 4}{3}$ is the inverse of
$f(x) = 3x - 4$, then compute $g(14)$.
\step{Composition is the definition, so substitute $f$ into $g$ and simplify
rather than re-deriving anything.}
\step{$g(f(x)) = \dfrac{(3x - 4) + 4}{3} = \dfrac{3x}{3} = x$, so the pair is
inverse. Then $g(14) = \dfrac{14 + 4}{3}$.}

$\Rightarrow\quad g(14) = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Check that $g(x) = \dfrac{x + 6}{2}$ inverts
$f(x) = 2x - 6$, then compute $g(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f^{-1}$ never means $\dfrac{1}{f}$. And an inverse
formula without its domain is a different function: $x^2 + 4$ is defined
everywhere, but the inverse of $\sqrt{x - 4}$ is $x^2 + 4$ \emph{only for}
$x \geq 0$.
\end{watchoutbox}
