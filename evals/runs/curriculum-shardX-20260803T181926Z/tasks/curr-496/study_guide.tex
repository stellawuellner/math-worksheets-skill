\sstitleblock{Geometric Series, $p$-Series, and the $n$th-Term Test}

\skillheading{Skill 1 --- Geometric series: classify by $r$, then sum}

\begin{formulabox}
\textbf{Shape:} $\sum a r^{\,k}$ --- every term is a fixed multiple $r$ of the one before
it. Find $r$ by dividing any term by its predecessor.\par
\smallskip
\textbf{Test:} converges exactly when $|r| < 1$; diverges when $|r| \geq 1$. The sign of
$r$ only makes the terms alternate.\par
\smallskip
\textbf{Sum (only when $|r| < 1$):} $S = \dfrac{a}{1 - r}$, where $a$ is the
\emph{first term actually written}, not the coefficient. Substitute the starting index to
find it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\sum_{n=0}^{\infty} 5\left(\tfrac{1}{3}\right)^{n}$.
\step{Consecutive terms differ by the fixed factor $\tfrac{1}{3}$, so this is geometric
and the sum formula applies once $|r| < 1$ is confirmed.}
\step{$a = 5\left(\tfrac{1}{3}\right)^{0} = 5$ and $r = \tfrac{1}{3}$, so
$S = \dfrac{5}{1 - \tfrac{1}{3}} = \dfrac{5}{\tfrac{2}{3}}$}

$\Rightarrow\quad S = \ans{15/2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\sum_{n=1}^{\infty} 8\left(\tfrac{1}{2}\right)^{n-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $S = \ans{16}$}
\end{tryitbox}

\skillheading{Skill 2 --- $p$-series: classify by $p$ alone}

\begin{formulabox}
\textbf{Shape:} $\sum \dfrac{1}{n^{p}}$ with $p > 0$ constant. Rewrite roots as powers
first: $\sqrt{n} = n^{1/2}$, $\sqrt[3]{n^{2}} = n^{2/3}$.\par
\smallskip
\textbf{Test:} converges when $p > 1$; diverges when $p \leq 1$. That single comparison is
the whole test.\par
\smallskip
\textbf{The borderline case is $p = 1$}, the harmonic series $\sum \tfrac{1}{n}$, and it
\emph{diverges}. The test gives no sum --- only a verdict.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Classify $\displaystyle\sum_{n=1}^{\infty} \frac{1}{n^{4/3}}$.
\step{The terms are a constant power of $n$ in the denominator, so this is a $p$-series and
the only thing to decide is where $p$ sits relative to $1$.}
\step{Here $p = \tfrac{4}{3}$, and $\tfrac{4}{3} = 1.333\ldots$ is greater than $1$}

$\Rightarrow\quad \ans{4/3 > 1}$, so the series converges
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Classify $\displaystyle\sum_{n=1}^{\infty} \frac{1}{n^{2/5}}$ ---
write the comparison of $p$ with $1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2/5 < 1}$, diverges}
\end{tryitbox}

\skillheading{Skill 3 --- The $n$th-term test: a divergence detector only}

\begin{formulabox}
\textbf{Statement:} if $\displaystyle\lim_{n \to \infty} a_n \neq 0$ (or the limit fails to
exist), then $\sum a_n$ \textbf{diverges}.\par
\smallskip
\textbf{What it never does:} a limit of $0$ proves nothing at all. Say ``inconclusive'' and
reach for another test.\par
\smallskip
\textbf{Doing the limit:} for a ratio of polynomials, compare degrees --- equal degrees give
the ratio of leading coefficients (nonzero, so the series diverges), and a bigger
denominator degree gives $0$ (inconclusive).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Apply the $n$th-term test to
$\displaystyle\sum_{n=1}^{\infty} \frac{4n + 3}{n + 7}$.
\step{Nothing here is geometric or a $p$-series, so start with the cheapest test --- look
at what the terms themselves approach.}
\step{Numerator and denominator are both degree $1$, so divide by $n$:
$\lim\limits_{n\to\infty} \dfrac{4 + 3/n}{1 + 7/n} = \dfrac{4}{1}$}

$\Rightarrow\quad \lim a_n = \ans{4} \neq 0$, so the series diverges
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Apply the $n$th-term test to
$\displaystyle\sum_{n=1}^{\infty} \frac{5n^{2} - 2}{2n^{2} + n}$ --- find $\lim a_n$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\lim a_n = \ans{5/2}$, diverges}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``the terms go to zero, so it converges'' is the single most
expensive error in this unit. It is the $n$th-term test read backwards. Both
$\sum \tfrac{1}{n}$ (divergent) and $\sum \left(\tfrac{1}{2}\right)^{n}$ (convergent) have
terms going to zero --- the limit alone cannot tell them apart.
\end{watchoutbox}
