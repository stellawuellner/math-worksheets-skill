\skillheading{Skill 1 --- Evaluating a binomial coefficient}

\begin{formulabox}
\textbf{Definition:}\quad
$\dbinom{n}{r} = \dfrac{n!}{r!\,(n-r)!}$\par
\smallskip
It counts \textbf{unordered} selections of $r$ things from $n$. Cancel the
larger factorial in the denominator against the tail of $n!$ before
multiplying, and remember the symmetry $\binom{n}{r} = \binom{n}{n-r}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Evaluate $\dbinom{8}{3}$.
\step{Order does not matter in a binomial coefficient, so use the factorial
definition and cancel before multiplying.}
\step{$\dfrac{8!}{3!\,5!} = \dfrac{8 \cdot 7 \cdot 6}{3 \cdot 2 \cdot 1}
= \dfrac{336}{6} = \ans{56}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Evaluate $\dbinom{10}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{210}$}
\end{tryitbox}

\skillheading{Skill 2 --- Expanding a binomial power in full}

\begin{formulabox}
\textbf{Binomial theorem:}
\[ (p+q)^{n} = \sum_{r=0}^{n} \binom{n}{r} p^{\,n-r} q^{\,r} \]
The two exponents always add to $n$. Whatever stands in the $p$ or $q$ slot
goes there \textbf{whole} --- a coefficient or a minus sign inside the bracket
is raised to the power too.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Expand $(x+2)^{4}$.
\step{A full expansion of a small power is fastest straight from the row of
coefficients, taken with descending powers of $x$.}
\step{Row 4 is $1, 4, 6, 4, 1$, and the powers of 2 climb as the powers of $x$
fall: $x^{4}$, $4x^{3}(2)$, $6x^{2}(4)$, $4x(8)$, $16$.}
\par
$\Rightarrow\quad \ans{x^{4}+8x^{3}+24x^{2}+32x+16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Expand $(y-3)^{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y^{3}-9y^{2}+27y-27}$}
\end{tryitbox}

\skillheading{Skill 3 --- Extracting one specific term}

\begin{formulabox}
\textbf{General term:}\quad
$T_{r+1} = \dbinom{n}{r} p^{\,n-r} q^{\,r}$\par
\smallskip
\textbf{Method:} match the exponent you were asked for to $r$ (or to $n-r$),
then evaluate \emph{both} pieces --- the binomial coefficient and the power of
the constant --- and multiply. Never expand the whole binomial for one term.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the coefficient of $x^{3}$ in $(x+2)^{5}$.
\step{Only one term is wanted, so match the exponent to $r$ instead of
expanding.}
\step{$x^{5-r} = x^{3}$ gives $r = 2$; then $\dbinom{5}{2} = 10$ and
$2^{2} = 4$, so the coefficient is $10 \cdot 4 = \ans{40}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the coefficient of $x^{2}$ in $(x+3)^{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{54}$}
\end{tryitbox}

\skillheading{Skill 4 --- Reading the structure of a row}

\begin{formulabox}
\textbf{Two substitutions worth memorising:}\par
\smallskip
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item $p = q = 1$: \ $\displaystyle\sum_{r=0}^{n}\binom{n}{r} = 2^{n}$ ---
        each row of Pascal's triangle sums to a power of two.
  \item $x = 1$: the sum of the coefficients of any polynomial is its value at
        $x = 1$, so $(1+cx)^{n}$ has coefficient sum $(1+c)^{n}$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the sum of the entries in row 5 of Pascal's
triangle.
\step{A whole row summed is the $p = q = 1$ case of the binomial theorem, so no
entry has to be listed.}
\step{$\sum_{r=0}^{5}\binom{5}{r} = 2^{5} = \ans{32}$; the entries
$1+5+10+10+5+1$ confirm it.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the sum of the coefficients of $(1+3x)^{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{256}$}
\end{tryitbox}
