\sstitleblock{Interpreting Remainders in Real Situations}

\skillheading{Skill 1 --- Counting only the groups that fill up}

\begin{formulabox}
\textbf{Rule:} Divide, then \emph{drop} the remainder.\\[3pt]
Use this when the question says \emph{completely full}, \emph{whole}, or \emph{how many can you make}. A leftover is not enough to make one more group, so it is not counted.\\[3pt]
Total $\div$ group size $=$ quotient R remainder \quad $\Rightarrow$ \quad answer $=$ quotient
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad There are 87 crayons. Each cup holds 6 crayons. How many cups can be filled completely?
\step{The question asks for cups that are completely full, so any leftover crayons do not count --- this is a drop-the-remainder question.}
\step{$6 \times 14 = 84$, and $87 - 84 = 3$, so $87 \div 6$ is 14 R 3. The 3 leftover crayons cannot fill a 15th cup.}
Cups filled completely: $\ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad There are 71 beads. Each bracelet uses 8 beads. How many complete bracelets can be made?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Counting every group you must provide}

\begin{formulabox}
\textbf{Rule:} Divide, then \emph{add one} whenever there is a remainder.\\[3pt]
Use this when the question says \emph{needed}, \emph{must order}, or \emph{so everyone fits}. The leftovers are real things that still need a place, so they need one more group --- even a group that is not full.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad There are 119 campers. Each canoe holds 6 campers. How many canoes are needed?
\step{Every camper must get into a canoe, so leftover campers still need one --- this is a round-the-quotient-up question.}
\step{$6 \times 19 = 114$, and $119 - 114 = 5$, so $119 \div 6$ is 19 R 5. Those 5 campers need a canoe of their own.}
Canoes needed: $\ans{20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A van carries 7 people. How many vans are needed for 45 people?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{Skill 3 --- When the leftover is the answer}

\begin{formulabox}
\textbf{Rule:} The remainder itself is the answer.\\[3pt]
Use this when the question says \emph{left over}, \emph{remaining}, or \emph{how much is not used}.\\[3pt]
remainder $=$ total $-$ (group size $\times$ quotient)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A florist has 100 roses and makes bunches of 12. How many roses are left after every full bunch is made?
\step{The question asks what is not used, so the answer is the remainder itself, not the number of bunches.}
\step{$12 \times 8 = 96$, and $100 - 96 = 4$, so $100 \div 12$ is 8 R 4.}
Roses left over: $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad There are 62 stickers, put onto sheets of 9. How many stickers are left over?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 4 --- Estimating the quotient first}

\begin{formulabox}
\textbf{Rule:} Round each number to the nearest ten to get numbers that divide easily, then divide.\\[3pt]
An estimate tells you roughly how big the answer should be, so a wildly wrong exact answer is easy to spot. An estimate is a check, never the final answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $178 \div 21$ by rounding each number to the nearest ten.
\step{Exact division by 21 is awkward, so round both numbers to friendly tens first and divide those instead.}
\step{$178$ rounds to $180$ and $21$ rounds to $20$; then $180 \div 20 = 9$.}
Estimate: $\ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $243 \div 31$ by rounding each number to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}
