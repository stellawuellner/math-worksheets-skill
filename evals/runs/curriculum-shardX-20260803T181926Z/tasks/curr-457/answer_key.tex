\problem{$f(x) = 4x^5 - 3x^2 + 7$

\textbf{Step 1.} Differentiate term by term with the power rule
$\frac{d}{dx}x^n = nx^{n-1}$: the exponent multiplies out front and drops by
one.

\textbf{Step 2.} $4x^5 \rightarrow 20x^4$, \ $-3x^2 \rightarrow -6x$, and the
constant $7$ has slope zero.

$\ans{f'(x) = 20x^4 - 6x}$}

\problem{$g(x) = 6\sqrt{x} + \dfrac{5}{x^2}$

\textbf{Step 1.} The power rule needs powers, so rewrite:
$g(x) = 6x^{1/2} + 5x^{-2}$.

\textbf{Step 2.} Differentiate: $6 \cdot \tfrac{1}{2}x^{-1/2} = 3x^{-1/2}$ and
$5 \cdot (-2)x^{-3} = -10x^{-3}$.

\textbf{Step 3.} Rewrite with positive exponents.

$\ans{g'(x) = \dfrac{3}{\sqrt{x}} - \dfrac{10}{x^3}}$}

\problem{Limit definition for $f(x) = x^3$.

\textbf{Step 1.} Expand the binomial in the numerator:
$(x + h)^3 = x^3 + 3x^2h + 3xh^2 + h^3$, so the numerator is
$3x^2h + 3xh^2 + h^3$.

\textbf{Step 2.} Every surviving term contains $h$, so factor it out and
cancel --- this is the step the definition is really testing:
\[ \frac{h\left(3x^2 + 3xh + h^2\right)}{h} = 3x^2 + 3xh + h^2 . \]

\textbf{Step 3.} Now the limit is a substitution: let $h \to 0$ and the last
two terms vanish.

\textbf{Agreement.} The power rule gives $\frac{d}{dx}x^3 = 3x^2$, the same
result --- which is exactly why the rule is allowed to replace this work.

$\ans{f'(x) = 3x^2}$}

\problem{$y = (x^2 + 3)(2x - 5)$

\textbf{Step 1.} Product rule with $u = x^2 + 3$, $v = 2x - 5$, so $u' = 2x$
and $v' = 2$.

\textbf{Step 2.} $y' = u'v + uv' = 2x(2x - 5) + (x^2 + 3)(2)$.

\textbf{Step 3.} Expand and collect: $4x^2 - 10x + 2x^2 + 6$.

\textit{Check:} expanding first gives $y = 2x^3 - 5x^2 + 6x - 15$, whose
derivative is the same.

$\ans{y' = 6x^2 - 10x + 6}$}

\problem{$f(x) = \dfrac{3x - 1}{x + 4}$

\textbf{Step 1.} Quotient rule with $u = 3x - 1$, $v = x + 4$, so $u' = 3$ and
$v' = 1$.

\textbf{Step 2.} $f'(x) = \dfrac{u'v - uv'}{v^2}
= \dfrac{3(x + 4) - (3x - 1)}{(x + 4)^2}$.

\textbf{Step 3.} Simplify the numerator: $3x + 12 - 3x + 1 = 13$. The
$x$-terms cancel, which is why this derivative is never zero.

$\ans{f'(x) = \dfrac{13}{(x + 4)^2}}$}

\problem{$h(x) = x^2\cos x$

\textbf{Step 1.} Product rule with $u = x^2$, $v = \cos x$, so $u' = 2x$ and
$v' = -\sin x$.

\textbf{Step 2.} $h'(x) = 2x\cos x + x^2(-\sin x)$. The minus sign comes from
the derivative of cosine, not from the rule.

$\ans{h'(x) = 2x\cos x - x^2\sin x}$}

\problem{$f(x) = \dfrac{x^2 + 1}{x^2 - 1}$

\textbf{Step 1.} Quotient rule with $u = x^2 + 1$, $v = x^2 - 1$, so
$u' = v' = 2x$.

\textbf{Step 2.} $f'(x) = \dfrac{2x(x^2 - 1) - (x^2 + 1)(2x)}{(x^2 - 1)^2}$.

\textbf{Step 3.} The numerator is $2x^3 - 2x - 2x^3 - 2x$; the cubic terms
cancel and $-2x - 2x$ remains.

$\ans{f'(x) = \dfrac{-4x}{(x^2 - 1)^2}}$}

\problem{$f(x) = (2x + 1)\sqrt{x}$

\textbf{(a) Step 1.} Product rule with $u = 2x + 1$ and $v = x^{1/2}$, so
$u' = 2$ and $v' = \tfrac{1}{2}x^{-1/2}$.

\textbf{Step 2.} $f'(x) = 2\sqrt{x} + (2x + 1) \cdot \dfrac{1}{2\sqrt{x}}$.

\textbf{Step 3.} Split the second term:
$\dfrac{2x}{2\sqrt{x}} + \dfrac{1}{2\sqrt{x}} = \sqrt{x} + \dfrac{1}{2\sqrt{x}}$,
so the two radical terms combine to $3\sqrt{x}$.

\textbf{(b)} Substitute $x = 4$: $3\sqrt{4} = 6$ and
$\dfrac{1}{2\sqrt{4}} = \dfrac{1}{4}$, giving a tangent slope of $6.25$.

$\ans{f'(x) = 3\sqrt{x} + \dfrac{1}{2\sqrt{x}}, \quad f'(4) = 6.25}$}

\problem{$f(x) = \dfrac{x^2}{2x - 3}$

\textbf{(a) Step 1.} Quotient rule with $u = x^2$, $v = 2x - 3$, so $u' = 2x$
and $v' = 2$.

\textbf{Step 2.} $f'(x) = \dfrac{2x(2x - 3) - x^2(2)}{(2x - 3)^2}$.

\textbf{Step 3.} Numerator: $4x^2 - 6x - 2x^2$, which collapses to
$2x^2 - 6x$.

\textbf{(b)} At $x = 1$ the numerator is $2 - 6 = -4$ and the denominator is
$(2 - 3)^2 = 1$.

$\ans{f'(x) = \dfrac{2x^2 - 6x}{(2x - 3)^2}, \quad f'(1) = -4}$}

\problem{$f(x) = \dfrac{x\sin x}{x + 1}$

\textbf{Step 1.} The outermost structure is a quotient, so start there:
$u = x\sin x$ and $v = x + 1$, with $v' = 1$.

\textbf{Step 2.} $u$ is itself a product, so the product rule gives
$u' = \sin x + x\cos x$.

\textbf{Step 3.} Substitute into $\dfrac{u'v - uv'}{v^2}$.

$\ans{f'(x) = \dfrac{(\sin x + x\cos x)(x + 1) - x\sin x}{(x + 1)^2}}$}
