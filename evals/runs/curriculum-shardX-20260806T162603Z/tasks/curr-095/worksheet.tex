\wstitleblock{Composite Area and Perimeter Challenges}{Grade 3}{}

\begin{center}
\textit{Rectangle and composite-shape area and perimeter challenges}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item \emph{Area} covers the inside (square units). \emph{Perimeter} goes
        around the edge (plain length units).
  \item For a composite shape, split it into rectangles for area --- and walk
        all the way around, side by side, for perimeter.
  \item Show your work, and write the unit with every answer.
\end{itemize}

% ---------------------------------------------------------------- Problem 1
\problem{A rectangle is 6 cm long and 4 cm wide. What is its \emph{area}?
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (6,4);
  \node[below, font=\small] at (3,0) {6 cm};
  \node[right, font=\small] at (6,2) {4 cm};
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{cm$^2$}}

% ---------------------------------------------------------------- Problem 2
\problem{A rectangular pen is 9 m long and 3 m wide. What is its
\emph{perimeter}?
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (9,3);
  \node[below, font=\small] at (4.5,0) {9 m};
  \node[right, font=\small] at (9,1.5) {3 m};
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{m}}

% ---------------------------------------------------------------- Problem 3
\problem{Each little square on the grid is one square unit. Find the
\emph{area} of the L-shape by counting squares or by splitting it into two
rectangles.
\begin{center}

  \fill[blue!15] (0,0) -- (4,0) -- (4,1) -- (2,1) -- (2,3) -- (0,3) -- cycle;
  \draw[gray!55, thin] (1,0) -- (1,3);
  \draw[gray!55, thin] (2,0) -- (2,3);
  \draw[gray!55, thin] (3,0) -- (3,1);
  \draw[gray!55, thin] (0,1) -- (4,1);
  \draw[gray!55, thin] (0,2) -- (2,2);
  \draw[very thick] (0,0) -- (4,0) -- (4,1) -- (2,1) -- (2,3) -- (0,3) -- cycle;
