\sstitleblock{Transformations of Absolute-Value Functions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading the Vertex and the Transformations}

\begin{formulabox}
\textbf{Vertex form:} $g(x) = a\,|x - h| + k$. \ \textbf{Vertex} $(h, k)$; \ \textbf{axis of symmetry} $x = h$.\par
\textbf{What each letter does to $y = |x|$:} $h$ shifts horizontally (right when $x - h$, left when $x + h$); $k$ shifts vertically; $a < 0$ flips the V so it opens down; $|a| > 1$ makes it narrower, $0 < |a| < 1$ makes it wider.\par
\textbf{Fast way to find $h$:} set the inside of the bars equal to $0$ and solve.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the vertex of $g(x) = -2\,|x + 5| + 3$ and describe the graph.
\step{The vertex is the one point where the distance inside the bars is zero, so set the inside to zero and read $k$ off the outside.}
\step{$x + 5 = 0 \Rightarrow x = -5$; \ then $g(-5) = -2\,|{-5} + 5| + 3 = 3$. Since $a = -2$, the V opens down and is narrower than $y = |x|$.}

$\Rightarrow\quad \text{vertex } \ans{(-5,\,3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the vertex of $g(x) = 4\,|x - 6| - 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(6,\,-1)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $|x + 3|$ shifts LEFT, not right --- rewrite it as $|x - (-3)|$ and the sign stops fooling you. And $a$ multiplies only the bars, so in $3\,|x + 2| - 7$ the $3$ never touches the $-7$.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Sketching the Graph from Vertex Form}

\begin{formulabox}
\textbf{Three steps:} (1) plot the vertex $(h, k)$; (2) from the vertex go right $1$ and up $a$ to get a point on the right arm, then left $1$ and up $a$ for the left arm; (3) draw two straight rays --- an absolute-value graph is never curved.\par
\textbf{$x$-intercepts:} solve $a\,|x - h| + k = 0$. They are always the same distance left and right of $x = h$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Sketch $y = |x - 1| - 4$ and give its $x$-intercepts.
\step{Vertex form is already given, so plot the vertex first and then find where the two arms cross the axis.}
\step{Vertex: $x - 1 = 0 \Rightarrow x = 1$, and $y = -4$ there. Intercepts: $|x - 1| = 4 \Rightarrow x - 1 = \pm 4$.}

$\Rightarrow\quad \text{vertex } \ans{(1,\,-4)}, \quad x = \ans{-3} \text{ or } \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Give the vertex and both $x$-intercepts of $y = |x + 2| - 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(-2,\,-3)}$, $x = \ans{-5}$ or $\ans{1}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Solving with the Graph in Mind}

\begin{formulabox}
\textbf{Why there are two answers:} a horizontal line above the vertex of an upward V cuts it twice, and the two crossings sit the same distance on either side of $x = h$.\par
\textbf{Method:} isolate the bars, then split. \ $a\,|x - h| + k = c \ \Rightarrow \ |x - h| = \dfrac{c - k}{a} \ \Rightarrow \ x = h \pm \dfrac{c - k}{a}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3\,|x - 2| - 5 = 7$.
\step{The bars are not alone yet, so isolate them first; only then can the V be split into its two arms.}
\step{$3\,|x - 2| = 12 \Rightarrow |x - 2| = 4 \Rightarrow x = 2 \pm 4$.}

$\Rightarrow\quad x = \ans{-2} \text{ or } x = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\,|x + 1| - 3 = 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-7}$ or $x = \ans{5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} divide by $a$ BEFORE splitting the bars. Splitting $3\,|x - 2| = 12$ into $3x - 2 = \pm 12$ loses the bars and both answers.
\end{watchoutbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Writing the Equation from the Graph}

\begin{formulabox}
\textbf{Read the vertex, then one more point.} The vertex hands you $h$ and $k$ at a glance. Substitute any second point $(x_1, y_1)$ into $y = a\,|x - h| + k$ and solve the resulting one-variable equation for $a$: \ $a = \dfrac{y_1 - k}{|x_1 - h|}$.\par
\textbf{Sanity check:} if the second point is BELOW the vertex, $a$ must come out negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A V has vertex $(2, 1)$ and passes through $(5, 7)$. Write its equation.
\step{The vertex gives two of the three unknowns for free, so the extra point is only needed to pin down the steepness $a$.}
\step{$h = 2$, $k = 1$, so $7 = a\,|5 - 2| + 1 \Rightarrow 3a + 1 = 7 \Rightarrow a = 2$.}

$\Rightarrow\quad y = \ans{2}\,|x - 2| + \ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A V has vertex $(-4, 6)$ and passes through $(-1, 0)$. Write its equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{-2}\,|x + 4| + \ans{6}$}
\end{tryitbox}
