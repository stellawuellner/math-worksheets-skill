% =================== SKILL 1 ===================
\skillheading{Skill 1 --- From repeated addition to a product}

\begin{formulabox}
\textbf{Rule:}\quad Count how many times the same number repeats.\par
that count $\times$ the repeated number $=$ the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $7 + 7 + 7$ as a multiplication fact and find the total.
\step{The 7 repeats 3 times, so the fact is $3 \times 7$.}
\step{$7 + 7 + 7 = 21$, so $3 \times 7 = 21$.}

$\Rightarrow\quad\ans{21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $5 + 5 + 5 + 5$ as a multiplication fact and find the total.\\[2pt]
\rotatebox{180}{\footnotesize check: $4 \times 5 = \ans{20}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- From an array to a product}

\begin{formulabox}
\textbf{Rule:}\quad rows $\times$ number in each row $=$ total.\par
An array is repeated addition you can \emph{see}: every row adds the same number again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An array has 4 rows with 9 dots in each row. Write its fact and total.
\step{Each of the 4 rows holds 9 dots, so the fact is $4 \times 9$.}
\step{$4 \times 9 = 36$.}

$\Rightarrow\quad\ans{36}$ dots
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} An array has 6 rows with 3 dots in each row. What is its fact and total?\\[2pt]
\rotatebox{180}{\footnotesize check: $6 \times 3 = \ans{18}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- From a product back to repeated addition}

\begin{formulabox}
\textbf{Rule:}\quad $3 \times 8$ means ``3 groups of 8'' --- write one 8 for each group.\par
Turning the fact around (8 groups of 3) changes the picture, never the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $3 \times 8$ as repeated addition and find the total.
\step{Three groups of 8 means the 8 is added 3 times: $8 + 8 + 8$.}
\step{$8 + 8 = 16$, and $16 + 8 = 24$.}

$\Rightarrow\quad\ans{24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $2 \times 9$ as repeated addition and find the total.\\[2pt]
\rotatebox{180}{\footnotesize check: $9 + 9 = \ans{18}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The multiplication sign never means ``add the two numbers.'' A fact like $6 \times 9$ counts 6 whole groups of 9 --- the answer is much bigger than $6 + 9$.
\end{watchoutbox}
