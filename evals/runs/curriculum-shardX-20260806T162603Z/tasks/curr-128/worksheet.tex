\wstitleblock{Mixed Numbers in Real Life}{Grade 4--5}{}
\begin{center}
\textit{Adding, subtracting, and regrouping mixed numbers in real contexts}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Every problem is a real situation. All the numbers you need are given
        in the problem; write your answer with its unit.
  \item Give every answer as a mixed number in simplest form.
\end{itemize}

\problem{On Saturday, Priya hikes the Fern Trail, which is $1\tfrac{1}{4}$ mi
long, and then the Creek Trail, which is $2\tfrac{1}{4}$ mi long. How many
miles does she hike in all?
\par\vspace*{4cm}\answerline{mi}}

\problem{A board is $4\tfrac{3}{8}$ ft long. A carpenter saws off a piece
$2\tfrac{1}{8}$ ft long. How many feet of board are left?
\par\vspace*{4cm}\answerline{ft}}

\problem{A muffin recipe uses $2\tfrac{2}{3}$ cups of flour and a bread
recipe uses $1\tfrac{2}{3}$ cups. How many cups of flour do the two recipes
need altogether?
\par\vspace*{4.5cm}\answerline{cups}}

\problem{A pitcher holds $3\tfrac{5}{6}$ L of lemonade. Guests drink
$1\tfrac{1}{6}$ L. How many liters of lemonade are left?
\par\vspace*{4.5cm}\answerline{L}}

\problem{Two watermelons are weighed at a market stand. The small one weighs
$1\tfrac{3}{5}$ kg and the large one weighs $2\tfrac{4}{5}$ kg. What is their
total weight, in kilograms?
\par\vspace*{4.5cm}\answerline{kg}}

\problem{A spool holds $5\tfrac{1}{4}$ m of rope. Deshi uses $2\tfrac{3}{4}$ m
to hang a swing. How many meters of rope are left on the spool?
\emph{Hint: you will need to regroup.}
\par\vspace*{5cm}\answerline{m}}

\problem{At the fair, Jo's pumpkin weighs $6\tfrac{1}{8}$ kg and Ben's
pumpkin weighs $3\tfrac{5}{8}$ kg. How many kilograms heavier is Jo's
pumpkin?
\par\vspace*{5cm}\answerline{kg}}

\problem{Mira does homework for $1\tfrac{1}{2}$ hours and then practices
violin for $2\tfrac{1}{4}$ hours. How many hours is that in all?
\par\vspace*{5cm}\answerline{hours}}

\problem{Zoe needs $4\tfrac{1}{3}$ m of fabric for a costume. She already has
$1\tfrac{2}{3}$ m. How many more meters of fabric does she need to buy?
\par\vspace*{5cm}\answerline{m}}

\problem{\textbf{Challenge.} A punch bowl holds 6 cups. Miguel pours in
$2\tfrac{3}{4}$ cups of orange juice and $1\tfrac{3}{4}$ cups of pineapple
juice. How many more cups of juice would fill the bowl to the top?
\par\vspace*{5.5cm}\answerline{cups}}
