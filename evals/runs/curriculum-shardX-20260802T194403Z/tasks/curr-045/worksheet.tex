\wstitleblock{Measuring and Picture Graphs}{Kindergarten--Grade 1}{}

\begin{center}\small\textit{Measuring with same-size units, reading a picture graph, and comparing from data}\end{center}

\noindent\small\textbf{Directions:} Some questions use the \textbf{Sticker Chart}. Some use
the \textbf{Measuring Strip}. Every page shows you both. Read each question and decide
\emph{which} picture holds the answer. Count carefully, then write your answer on the
line.\normalsize

\datapanel

\problem[3cm]{Look at the \textbf{Sticker Chart}. How many stickers does Ana have?}

\problem[3cm]{Use the \textbf{Measuring Strip}. The pencil is 7 paper clips long. The
crayon is 4 paper clips long. Write $>$ or $<$ in the box to make it true:\\[8pt]
\hspace*{1.2cm}\large pencil \symbolbox\ crayon \noansline}

\problem[3cm]{Look at the \textbf{Sticker Chart}. Who has the \textbf{most} stickers?}

\newpage
\datapanel

\problem[3cm]{Use the \textbf{Sticker Chart}. How many \textbf{more} stickers does Ana
have than Leo?}

\problem[3cm]{Use the \textbf{Measuring Strip}. How many paper clips \textbf{longer} is
the straw than the marker?}

\problem[3cm]{Use the \textbf{Sticker Chart}. How many stickers do all four children have
\textbf{in all}?}

\problem[3cm]{Use the \textbf{Sticker Chart}. Who has the \textbf{fewest} stickers?}

\newpage
\datapanel

\problem[3.5cm]{Use the \textbf{Measuring Strip}. Write \emph{crayon}, \emph{marker}, and
\emph{straw} in order from \textbf{shortest to longest}.\\[8pt]
\quad shortest \ordblanks\ longest \noansline}

\problem[3cm]{Mia says: \emph{``Leo and Sam together have more stickers than Ana and I
do.''} Use the \textbf{Sticker Chart}. Find both totals, then write $>$ or $<$ to show
which side really is bigger:\\[8pt]
\hspace*{1cm}\large Leo $+$ Sam \symbolbox\ Ana $+$ Mia \noansline}

\problem[5cm]{Rosa measured the \textbf{same} straw two ways. With paper clips she counted
9. With blocks she counted only 5.\\[4pt]
Which is longer, one paper clip or one block? Draw a picture and tell how you know.}
