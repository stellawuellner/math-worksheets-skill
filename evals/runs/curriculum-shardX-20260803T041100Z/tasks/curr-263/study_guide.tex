\sstitleblock{Solving Systems by Elimination with Scaling}

\skillheading{1. Scaling one equation}

\begin{formulabox}
\textbf{The move:} multiplying \emph{every term on both sides} of an equation by
a nonzero constant gives an equivalent equation --- same solutions, new
coefficients. That is what makes elimination legal.\\[3pt]
\textbf{When one equation is enough:} one coefficient of the target variable
divides the other. Scale the smaller one up to match, then add or subtract so
the target drops out.\\[3pt]
\textbf{Add or subtract?} Opposite signs $\Rightarrow$ add. Same sign
$\Rightarrow$ subtract.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $x + 2y = 11$ and $3x - y = 5$.
\step{The $y$ coefficients are $2$ and $-1$, and $1$ divides $2$, so scaling the
second equation alone will match them --- no need to touch the first.}
\step{$2(3x - y) = 2(5) \Rightarrow 6x - 2y = 10$. Add to $x + 2y = 11$:
$7x = 21$, so $x = 3$.}
\step{Back-substitute: $3 + 2y = 11 \Rightarrow y = 4$. Check in $3x - y$:
$9 - 4 = 5$ $\checkmark$ \quad $\ans{(x, y) = (3, 4)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2x + y = 7$ and $x + 3y = 11$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x, y) = (2, 3)}$}
\end{tryitbox}

\skillheading{2. Scaling both equations}

\begin{formulabox}
\textbf{When neither coefficient divides the other,} scale both. Take the least
common multiple of the two coefficients of the target variable, and multiply
each equation by what it needs to reach it.\\[3pt]
\textbf{Choose the cheaper variable first:} compare the two LCMs and eliminate
the variable with the smaller one. Smaller multipliers mean smaller numbers and
fewer arithmetic slips.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2x + 3y = 12$ and $5x + 4y = 23$.
\step{Neither $x$ pair nor $y$ pair divides evenly, so scale both; for $x$ the
LCM of $2$ and $5$ is $10$, cheaper than the LCM of $3$ and $4$.}
\step{$5(2x + 3y) = 5(12) \Rightarrow 10x + 15y = 60$ and
$2(5x + 4y) = 2(23) \Rightarrow 10x + 8y = 46$. Subtract: $7y = 14$, so $y = 2$.}
\step{Back-substitute: $2x + 6 = 12 \Rightarrow x = 3$. Check in $5x + 4y$:
$15 + 8 = 23$ $\checkmark$ \quad $\ans{(x, y) = (3, 2)}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $3x + 2y = 16$ and $4x + 5y = 26$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x, y) = (4, 2)}$}
\end{tryitbox}

\skillheading{3. Building the system from a situation}

\begin{formulabox}
\textbf{Two unknowns need two equations.} Name each unknown with a letter and a
unit before writing anything else.\\[3pt]
\textbf{Where the two equations come from:} usually one counts things (items,
litres, hours) and one measures value (dollars, salt, distance).\\[3pt]
\textbf{Then eliminate as usual}, and finish by translating the ordered pair back
into the words of the problem.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Two adult and three child tickets cost \$$37$; one adult
and four child tickets cost \$$31$. Find each price in dollars.
\step{Two unknown prices, two purchase facts, so name $a$ and $c$ in dollars and
write one equation per purchase: $2a + 3c = 37$ and $a + 4c = 31$.}
\step{The $a$ coefficients are $2$ and $1$, so scale the second by $2$:
$2a + 8c = 62$. Subtract the first: $5c = 25$, so $c = 5$.}
\step{Back-substitute: $a + 20 = 31 \Rightarrow a = 11$. Check:
$2(11) + 3(5) = 37$ $\checkmark$}
$\Rightarrow\quad \ans{(a, c) = (11, 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Three notebooks and two pens cost \$$23$; one notebook and
four pens cost \$$21$. Find the price of a notebook, $x$, and of a pen, $y$, in
dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x, y) = (5, 4)}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Scaling means multiplying \emph{both sides}. Turning
$x + 4y = 14$ into $3x + 12y = 14$ changes the equation and loses the solution;
the right side must become $42$.
\end{watchoutbox}
