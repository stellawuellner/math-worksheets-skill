\sstitleblock{Using Twos, Fives, and Tens as Anchor Facts}

\begin{formulabox}
\textbf{Three facts you can always count, and never have to memorise.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \textbf{twos} --- every other number: 2, 4, 6, 8, \dots
\item \textbf{fives} --- the nickel count: 5, 10, 15, 20, \dots
\item \textbf{tens} --- the number with a zero after it: 10, 20, 30, \dots
\end{itemize}
Every other fact is one small step away from one of these three. Find the
nearest anchor, take the step, and the fact is yours.
\end{formulabox}

\skillheading{1. Double a two-fact to get a four-fact}

\begin{formulabox}
4 is two 2s, so six groups of 4 is six groups of 2 counted \emph{twice}. Count
by twos to get the two-fact, then double the answer. The same move takes a
four-fact to an eight-fact.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $9 \times 4$.
\step{Nine fours is hard to count, but nine twos is not --- so start at the
two-fact and double it.}
\step{$9 \times 2 = 18$, and doubling gives $18 + 18$.}
\[ 9 \times 4 = \ans{36} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Use the two-fact $8 \times 2$ to find $8 \times 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32}$}
\end{tryitbox}

\skillheading{2. Take one group off a ten-fact to get a nine-fact}

\begin{formulabox}
9 is one less than 10, so a nine-fact is the ten-fact minus \textbf{one whole
group}. Write the ten-fact first, then subtract the group size --- not 1.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $6 \times 9$.
\step{Nine groups is one group short of ten groups, so start high at the
ten-anchor and remove a group.}
\step{$6 \times 10 = 60$, and one group is worth 6, so take 6 away.}
\[ 6 \times 9 = \ans{54} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Use the ten-fact $4 \times 10$ to find $4 \times 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

\skillheading{3. Add one more group to a five-fact}

\begin{formulabox}
A six-fact is the five-fact plus \textbf{one more group}. A seven-fact is the
five-fact plus \textbf{two more groups} --- which is the same as the five-fact
plus the two-fact, because $5 + 2 = 7$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $7 \times 6$.
\step{Six groups is one group past the five-anchor, so build up from the
five-fact instead of counting sixes.}
\step{$7 \times 5 = 35$, and one more group of 7 is $35 + 7$.}
\[ 7 \times 6 = \ans{42} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Use the five-fact $6 \times 5$ to find $6 \times 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

\skillheading{4. Divide by counting up on an anchor}

\begin{formulabox}
Division asks \textbf{how many jumps} of one size fill the total. On a two,
five, or ten anchor you can simply count up and keep score with your fingers ---
the number of jumps is the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $35 \div 5$.
\step{Counting up on the five-anchor is easier than sharing 35 out one at a
time, so count fives and track the jumps.}
\step{5, 10, 15, 20, 25, 30, 35 --- that is 7 jumps, so $5 \times 7 = 35$.}
\[ 35 \div 5 = \ans{7} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} How many jumps of 10 make 80? Write the division fact.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the step from an anchor is always a whole \emph{group},
never a single counter --- $7 \times 9$ is $70 - 7$, not $70 - 1$. And say which
anchor you used out loud; an answer with no anchor behind it is a guess.
\end{watchoutbox}
