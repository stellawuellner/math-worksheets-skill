\problem[5cm]{Name the relationship between each pair of angles. Choose from:
corresponding, alternate interior, alternate exterior, same-side (co-interior),
vertical, linear pair.
\begin{itemize}[leftmargin=1.2cm, itemsep=3pt, topsep=3pt]
  \item $\angle 1$ and $\angle 5$ \hfill \underline{\hspace{5cm}}
  \item $\angle 4$ and $\angle 6$ \hfill \underline{\hspace{5cm}}
  \item $\angle 2$ and $\angle 8$ \hfill \underline{\hspace{5cm}}
\end{itemize}
\noansline}

\problem[5.5cm]{Suppose $m\angle 1 = 112^\circ$. Find $m\angle 5$ and then
$m\angle 6$. Name the pair you used for each step.}

\problem[5.5cm]{Suppose $m\angle 4 = 65^\circ$. Find $m\angle 6$, then find
$m\angle 5$. One of these two angles is congruent to $\angle 4$ and the other is
supplementary to it --- the names tell you which is which.}

\problem[6cm]{$\angle 2$ and $\angle 6$ are corresponding angles, with
$m\angle 2 = (3x + 10)^\circ$ and $m\angle 6 = (5x - 24)^\circ$.
Write the equation the angle pair justifies, solve for $x$, and then give
$m\angle 2$.}

\problem[6cm]{Suppose $m\angle 7 = 74^\circ$. Find $m\angle 1$, then find
$m\angle 4$. State the pair name used at each step.}

\problem[6cm]{$\angle 3$ and $\angle 6$ are same-side interior angles, with
$m\angle 3 = (2x + 15)^\circ$ and $m\angle 6 = (3x + 40)^\circ$. Same-side
interior angles are supplementary, not congruent --- write the correct equation,
solve for $x$, and give $m\angle 3$.}

\problem[6.5cm]{$\angle 2$ and $\angle 8$ are alternate exterior angles, with
$m\angle 2 = (4x - 9)^\circ$ and $m\angle 8 = (3x + 21)^\circ$. Find $x$ and
$m\angle 2$. Then check your answer a second way: find $m\angle 7$ and confirm
that $\angle 7$ and $\angle 8$ form a linear pair.}

\problem[7cm]{Explain why $\angle 3$ and $\angle 6$ must be supplementary
whenever $\ell_1 \parallel \ell_2$. Use at least one angle-pair name and one
other angle in the diagram as a stepping stone. Two or three sentences.
\noansline}
