\problem{Convert $3$ m to centimeters}

\textbf{Solution:} Each meter is $100$ centimeters, so $3$ meters is
$3 \times 100$. \ans{300 \text{ cm}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Tissue box $4 \times 2 \times 3$ cm}

\textbf{Solution:} $V = \ell \times w \times h = 4 \times 2 \times 3$. The
base layer holds $4 \times 2 = 8$ centimeter cubes and there are $3$ layers.
\ans{24 \text{ cm}^3}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Straight-line pair with $115^\circ$}

\textbf{Solution:} Angles on a straight line total $180^\circ$, so the other
angle is $180 - 115 = 65$. \ans{65^\circ}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Rice: $2.5$ kg in grams}

\textbf{Solution:} Each kilogram is $1000$ grams:
$2.5 \times 1000 = 2500$. \ans{2500 \text{ g}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Right angle split, one part $35^\circ$}

\textbf{Solution:} A right angle totals $90^\circ$, so the other part is
$90 - 35 = 55$. \ans{55^\circ}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Step stool: two boxes, $5 \times 4$ base, heights $3$ and $2$}

\textbf{Solution:} Both boxes share the $5 \times 4$ base, so stack the
heights first: $3 + 2 = 5$ cm tall in all. Then
$V = 5 \times 4 \times 5 = 100$. (Or add the two volumes: $60 + 40$.)
\ans{100 \text{ cm}^3}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Fun run: $3$ km, $1250$ m already run}

\textbf{Solution:} Convert first so both distances use the same unit:
$3$ km $= 3 \times 1000 = 3000$ m. Then subtract: $3000 - 1250 = 1750$.
\ans{1750 \text{ m}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Minute hand: $360^\circ$ in $60$ minutes; $20$ minutes}

\textbf{Solution:} Each minute the hand sweeps $360 \div 60 = 6$ degrees.
In $20$ minutes: $6 \times 20 = 120$. \ans{120^\circ}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Movie: $2$ h $15$ min in minutes}

\textbf{Solution:} Each hour is $60$ minutes: $2 \times 60 = 120$, plus the
extra $15$ gives $135$. \ans{135 \text{ min}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Box: volume $72$ cm$^3$, base $4 \times 3$ cm --- height?}

\textbf{Solution:} The base area is $4 \times 3 = 12$, so each layer of
cubes holds $12$. The number of layers is $72 \div 12 = 6$.
\ans{6 \text{ cm}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Jug: $2$ L, cups of $250$ mL}

\textbf{Solution:} Convert the jug to milliliters: $2 \times 1000 = 2000$
mL. Then $2000 \div 250 = 8$. The jug fills \ans{8} cups.

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}\vspace{0.1cm}

\problem{Challenge: fish tank $50 \times 20 \times 30$ cm}

\textbf{Solution:}
(a) $V = 50 \times 20 \times 30$: base $50 \times 20 = 1000$, times $30$
layers. \ans{30000 \text{ cm}^3}

(b) $30000 \div 1000 = 30$. \ans{30 \text{ L}}

(c) Every $1000$ cubic centimeters makes one liter, so dividing the
cubic-centimeter count by $1000$ counts how many liter-sized groups fit ---
that is exactly what converting to liters means. A correct response must
connect the division to regrouping $1000$ cm$^3$ per liter.
