\wstitleblock{Finding the Whole or the Percent Rate}{Grade 7}{}

\begin{center}
\textit{Rates from a part and a whole, wholes from a part and a rate, and
solving the percent equation}
\end{center}

\noindent\textbf{Directions:} every percent problem holds three numbers ---
a \textbf{part}, a \textbf{whole}, and a \textbf{rate}. Two are given; find the
third.
\begin{itemize}
  \item \textbf{Worked pattern (rate).} $9$ of the $30$ chairs are taken.
        $\dfrac{\text{part}}{\text{whole}} = \dfrac{9}{30} = 0.3$, and
        $0.3 \times 100 = 30$, so $30\%$ are taken.
  \item \textbf{Worked pattern (whole).} $9$ chairs is $30\%$ of the room.
        $\dfrac{30}{100}\,x = 9$, so $x = 9 \div 0.30 = 30$ chairs.
  \item Show the division or the equation you used --- an answer with no setup
        earns no credit. Percents may come out as decimals; that is fine.
\end{itemize}

\problem[4.5cm]{A minibus has 40 seats and 14 of them are taken. What percent of
the seats are taken?}

\problem[4.5cm]{A quiz has 24 questions. Dev answered 18 of them correctly. What
percent did he answer correctly?}

\problem[5cm]{A theatre has already sold 12 tickets, and that is $25\%$ of all
the tickets it printed. How many tickets did it print?}

\problem[5cm]{Of the 45 seedlings in a tray, 27 have flowered. What percent have
flowered?}

\problem[5cm]{A hiker has walked 63 kilometres, which is $70\%$ of the whole
trail. How many kilometres long is the trail?}

\problem[5.5cm]{In a class survey, 15 students chose soccer, and that is $60\%$
of the class. Write the percent equation $\dfrac{60}{100}\,x = 15$ and solve it
for the class size $x$.}

\problem[5cm]{A jar holds 56 marbles and 7 of them are green. What percent of
the marbles are green?}

\problem[5.5cm]{A cart holds several books; 45 of them are non-fiction, and that
is $12.5\%$ of the cart. How many books are on the cart?}

\problem[5.5cm]{A shop sold 91 shirts last month, which was $35\%$ of the shirts
it had in stock. Write the percent equation $\dfrac{35}{100}\,x = 91$ and solve
it for the size $x$ of the stock.}

\problem[5.5cm]{A bird count recorded 60 birds, and 17 of them were finches.
What percent of the birds counted were finches? Round to the nearest tenth of a
percent.}

\problem[5.5cm]{A deposit of 84 dollars is $20\%$ of the price of a
bicycle.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] the full price of the bicycle, in dollars: \ansblank
  \item[(b)] the amount still owed after the deposit, in dollars: \ansblank
\end{itemize}}

\problem[6.5cm]{A survey reported that 45 students walk to school, and that is
$30\%$ of the students who were asked.\par\vspace{4pt}
\begin{itemize}
  \item[(a)] Write the percent equation $\dfrac{30}{100}\,x = 45$ and solve it
             for the number of students asked: \ansblank
  \item[(b)] Explain why dividing 45 by $0.30$ gives the same number as solving
             that equation. Write your explanation in the work space.
\end{itemize}}
