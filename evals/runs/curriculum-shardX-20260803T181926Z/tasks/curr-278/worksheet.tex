\wstitleblock{Writing Growth and Decay Models From Data}{Algebra 1 \quad Exponential Functions}{}

\begin{center}\small\textit{Reading a growth or decay factor out of a table,
turning a percent rate into a factor, and recovering the factor from two data
points}\end{center}

\noindent\textbf{The model you are writing every time.}
\[ y = a\cdot b^{\,t} \]
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $a$ is the amount when $t = 0$; $b$ is the factor the amount is
        multiplied by in one unit of time.
  \item $b > 1$ is growth, $0 < b < 1$ is decay. A rate of $r$ per period gives
        $b = 1 + r$ for growth and $b = 1 - r$ for decay.
  \item Always say what $t$ counts (hours, years, weeks) and what $y$ measures,
        with its unit, before you use the model.
  \item Round only at the last step. Give money to the cent and other measures
        to two decimal places unless the problem says otherwise.
\end{itemize}
\noindent\rule{\linewidth}{0.4pt}

\problem[6cm]{\textbf{Bacteria.} A biology class counts the cells in a culture
once an hour. The count is in thousands of cells.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
Time $t$ (hours) & 0 & 1 & 2 & 3 \\ \hline
Cells (thousands) & 50 & 150 & 450 & 1350 \\ \hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Divide each count by the one before it. What growth factor $b$ do
you get, and is it the same every hour?

\textbf{(b)} Write the model $y = a\cdot b^{\,t}$ for this culture, saying what
$a$ and $b$ mean here.

\textbf{(c)} Use your model to predict the count, in thousands of cells, at
$t = 5$ hours.}

\problem{\textbf{Medicine.} A nurse measures how much of a 80 mg dose is
still in the bloodstream each hour.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
Time $t$ (hours) & 0 & 1 & 2 & 3 \\ \hline
Amount (mg) & 80 & 40 & 20 & 10 \\ \hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Find the factor $b$ from the table. Is this growth or decay, and how
does the value of $b$ tell you?

\textbf{(b)} Write the model for the amount in the bloodstream after $t$ hours.

\textbf{(c)} How many milligrams are left after 6 hours?
\par\vspace*{6cm}\answerline{mg}}

\problem[6cm]{\textbf{Population.} A town has 8000 residents, and planners
expect the population to grow by 3 percent each year.

\textbf{(a)} A 3 percent yearly increase multiplies the population by what
factor $b$? Check your factor by computing one year of growth from 8000.

\textbf{(b)} Write the model $P = a\cdot b^{\,t}$, where $t$ is years from now
and $P$ is the number of residents.

\textbf{(c)} Predict the population 10 years from now, to the nearest resident.}

\problem[6cm]{\textbf{Depreciation.} A laptop is bought for 1200 dollars and
loses 15 percent of its value every year.

\textbf{(a)} Losing 15 percent a year means keeping what percent, and so
multiplying by what factor $b$? Check the factor by finding the value after one
year.

\textbf{(b)} Write the model $V = a\cdot b^{\,t}$ for the value in dollars after
$t$ years.

\textbf{(c)} What is the laptop worth after 4 years, in dollars, to the nearest
cent?}

\problem{\textbf{Bouncing ball.} A ball is dropped from 120 cm and its
rebound height is measured after each bounce.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
Bounce $n$ & 0 (drop) & 1 & 2 & 3 \\ \hline
Height (cm) & 120 & 84 & 58.8 & 41.16 \\ \hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Find the rebound factor $b$ by dividing the first bounce height by
the drop height.

\textbf{(b)} Write the model for the height, in centimetres, after $n$ bounces.

\textbf{(c)} Predict the height of the sixth bounce, in centimetres.
\par\vspace*{6.5cm}\answerline{cm}}

\problem[6.5cm]{\textbf{Savings.} An account is opened with 2000 dollars. No
money is added or taken out. Two years later the balance is 2205 dollars.

\textbf{(a)} Two years of growth multiply the balance by $b$ twice. Write the
equation this gives, and solve it for $b$.

\textbf{(b)} Write the model for the balance in dollars after $t$ years, and
state the annual interest rate as a percent.

\textbf{(c)} What will the balance be after 5 years, in dollars, to the nearest
cent?}

\problem{\textbf{Algae treatment.} A pond has 400 square metres of
surface covered by algae. A weekly treatment removes 22 percent of the covered
area each week.

\textbf{(a)} Find the weekly decay factor $b$, and check it by computing the
area still covered after one week.

\textbf{(b)} Write the model for the covered area, in square metres, after $w$
weeks.

\textbf{(c)} How much area is still covered after 5 weeks?
\par\vspace*{6.5cm}\answerline{m$^2$}}

\problem{\textbf{Drug half-life, two readings.} A patient is given 250 mg of
a drug. A blood test 2 hours later shows 160 mg still present. Assume the amount
decays by the same factor each hour.

\textbf{(a)} Write the equation the two readings give for the hourly factor $b$,
and solve it. (The two readings are 2 hours apart, so $b$ is applied twice.)

\textbf{(b)} Write the model for the amount in milligrams after $t$ hours.

\textbf{(c)} How much of the drug is left after 5 hours?
\par\vspace*{7cm}\answerline{mg}}

\problem[7cm]{\textbf{Is it exponential?} A crowdfunding campaign records its
running total, in dollars, at the end of each month.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
Month $t$ & 0 & 1 & 2 & 3 \\ \hline
Total (dollars) & 1500 & 1800 & 2160 & 2592 \\ \hline
\end{tabular}
\end{center}
\par\smallskip
\textbf{(a)} Subtract each total from the next one. Are the differences
constant? Now divide instead. What does that tell you about which kind of model
fits?

\textbf{(b)} Write the exponential model for the total in dollars after $t$
months.

\textbf{(c)} If the pattern holds, what is the total at the end of month 6, to
the nearest cent?}

\problem{\textbf{Challenge --- neither reading is the start.} A mould
colony is photographed on two days. On day 2 it covers 12 square centimetres,
and on day 5 it covers 96 square centimetres. The area grows by the same factor
each day.

\textbf{(a)} How many days separate the two photographs? Write the equation
those readings give for the daily factor $b$, and solve it.

\textbf{(b)} The model is $A = a\cdot b^{\,t}$ with $t$ in days. Day 0 was not
photographed --- work backwards from the day 2 reading to find $a$, and write
the full model.

\textbf{(c)} Predict the area on day 8, in square centimetres.
\par\vspace*{7cm}\answerline{cm$^2$}}
