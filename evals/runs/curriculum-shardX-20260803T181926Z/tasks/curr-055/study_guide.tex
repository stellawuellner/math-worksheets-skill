\sstitleblock{Rounding and Ordering on a Number Line}

\skillheading{1. Rounding to the nearest ten}

\begin{formulabox}
\textbf{Draw the picture, then decide.}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item Find the ten just below your number and the ten just above it. Those are
the two ends of your number line.
\item The middle mark is halfway between them --- it always ends in $5$.
\item Put your number on the line. \textbf{The end it is closer to is the
answer.}
\item Landed exactly on the middle? The rule is: go to the bigger end.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Round $37$ to the nearest ten.
\step{Rounding asks which benchmark is nearer, so first draw the two tens that
$37$ sits between.}
\step{The ends are $30$ and $40$, and the middle is $35$. Since $37$ is past
$35$, the right end is nearer.}
\begin{center}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, ->] (-0.6,0) -- (9.2,0);
  \foreach \x in {0,1,2,3,4,5,6,7,8} {\draw (\x,0.09) -- (\x,-0.09);}
  \draw[very thick] (0,0.26) -- (0,-0.26);
  \draw[very thick] (4,0.20) -- (4,-0.20);
  \draw[very thick] (8,0.26) -- (8,-0.26);
  \node[below=3pt, font=\small] at (0,-0.2) {$30$};
  \node[below=3pt, font=\small] at (4,-0.2) {$35$};
  \node[below=3pt, font=\small] at (8,-0.2) {$40$};
  \fill[red] (5.6,0) circle (2.4pt);
  \node[above=3pt, font=\small] at (5.6,0.1) {$37$};
\end{tikzpicture}
\end{center}

$\Rightarrow\quad \ans{40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Round $62$ to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{60}$}
\end{tryitbox}

\skillheading{2. Rounding to the nearest hundred}

\begin{formulabox}
\textbf{Same picture, bigger hops.} The ends are the hundred below and the
hundred above; the middle mark ends in $50$. Look only at where the number sits
between those two ends --- the ones digit never decides a hundreds rounding.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Round $349$ to the nearest hundred.
\step{Nearest hundred, so the benchmarks are hundreds: find the two the number
sits between.}
\step{The ends are $300$ and $400$ and the middle is $350$. $349$ is just short
of the middle, so the left end is nearer.}
\begin{center}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, ->] (-0.6,0) -- (9.2,0);
  \foreach \x in {0,1,2,3,4,5,6,7,8} {\draw (\x,0.09) -- (\x,-0.09);}
  \draw[very thick] (0,0.26) -- (0,-0.26);
  \draw[very thick] (4,0.20) -- (4,-0.20);
  \draw[very thick] (8,0.26) -- (8,-0.26);
  \node[below=3pt, font=\small] at (0,-0.2) {$300$};
  \node[below=3pt, font=\small] at (4,-0.2) {$350$};
  \node[below=3pt, font=\small] at (8,-0.2) {$400$};
  \fill[red] (3.92,0) circle (2.4pt);
  \node[above=3pt, font=\small] at (3.92,0.1) {$349$};
\end{tikzpicture}
\end{center}

$\Rightarrow\quad \ans{300}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Round $571$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{600}$}
\end{tryitbox}

\skillheading{3. Comparing two numbers}

\begin{formulabox}
\textbf{Further right means greater.} To find out which is further right without
drawing every tick: compare the \emph{hundreds} first, then the \emph{tens},
then the \emph{ones}. Stop at the first place where the digits differ --- that
place decides it, no matter what comes after.\\[3pt]
The small point of $<$ and $>$ always aims at the smaller number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $<$ or $>$ between $428$ and $482$.
\step{Comparing starts at the biggest place value, so check the hundreds
before anything else.}
\step{Both have $4$ hundreds, so move to the tens: $2$ tens against $8$ tens.
So $428$ sits to the left.}

$\Rightarrow\quad \ans{428 < 482}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $<$ or $>$ between $615$ and $609$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{615 > 609}$}
\end{tryitbox}

\skillheading{4. Putting numbers in order}

\begin{formulabox}
\textbf{Ordering is comparing, repeated.} Sort into groups by the hundreds
digit first, then order inside each group by the tens, then by the ones. Write
the finished list left to right, the way it would read on a number line.\\[3pt]
\textbf{Careful:} a longer-looking number is not automatically bigger, and the
last digit never decides anything on its own.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Order $204$, $240$ and $199$ from least to greatest.
\step{Ordering means placing them left to right on one number line, so compare
place by place, starting with the hundreds.}
\step{$199$ has $1$ hundred, so it is smallest. $204$ and $240$ both have $2$
hundreds, so compare tens: $0$ against $4$, putting $204$ first.}

$\Rightarrow\quad \ans{199,\ 204,\ 240}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Order $530$, $305$ and $503$ from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{305,\ 503,\ 530}$}
\end{tryitbox}
