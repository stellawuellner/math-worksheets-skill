\sstitleblock{Illegal Cancellation and Lost Solutions}

\skillheading{Skill 1 --- Factor, never cancel}

\begin{formulabox}
\textbf{Rule:} you may divide both sides by a quantity only if that quantity can never be zero. A trigonometric factor \emph{can} be zero, so dividing by it silently deletes solutions.\par
\textbf{Repair:} move everything to one side, factor, and set each factor to zero.\par
$AB = A \ \Longrightarrow\ A(B - 1) = 0$, \ never $B = 1$ alone.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\cos^2\theta = \sqrt{3}\cos\theta$ for $0^\circ \le \theta < 360^\circ$.
\step{The same cosine appears on both sides, which is the tempting moment to divide --- so collect and factor instead, keeping that cosine as a factor.}
\step{$\cos\theta\,(2\cos\theta - \sqrt{3}) = 0$: \ $\cos\theta = 0$ or $\cos\theta = \dfrac{\sqrt{3}}{2}$.}
$\Rightarrow\quad \theta = \ans{30}^\circ, \ans{90}^\circ, \ans{270}^\circ, \ans{330}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\sin^2\theta = \sqrt{2}\sin\theta$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{0}^\circ, \ans{45}^\circ, \ans{135}^\circ, \ans{180}^\circ$}
\end{tryitbox}

\skillheading{Skill 2 --- Keeping every solution}

\begin{formulabox}
\textbf{Three places solutions go missing:}\par
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item taking one square root: $u^2 = k$ means $u = \sqrt{k}$ \emph{or} $u = -\sqrt{k}$;
  \item a multiple angle: for $k\theta$, multiply both interval endpoints by $k$ before solving;
  \item declaring a factor impossible: $\cos\theta = 1$ and $\sin\theta = -1$ are attainable.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\cos 2\theta = \sqrt{3}$ for $0^\circ \le \theta < 360^\circ$.
\step{The angle inside is doubled, so the inside variable travels twice as far --- widen its interval before solving, or half the answers never appear.}
\step{$u = 2\theta$ on $[0^\circ, 720^\circ)$: $\cos u = \dfrac{\sqrt{3}}{2}$ at $u = 30^\circ, 330^\circ, 390^\circ, 690^\circ$; halve each.}
$\Rightarrow\quad \theta = \ans{15}^\circ, \ans{165}^\circ, \ans{195}^\circ, \ans{345}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\sin 3\theta = \sqrt{3}$ for $0^\circ \le \theta < 180^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{20}^\circ, \ans{40}^\circ, \ans{140}^\circ, \ans{160}^\circ$}
\end{tryitbox}

\skillheading{Skill 3 --- Testing for extraneous roots}

\begin{formulabox}
\textbf{Squaring is not reversible.} If $x = y$ then $x^2 = y^2$, but not the other way round: $-1$ and $1$ have the same square. So squaring both sides can \emph{invent} roots.\par
\textbf{Repair:} squaring is allowed, but every candidate must then be substituted into the \emph{original} equation and kept only if both sides agree.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\cos\theta + 1 = \sin\theta$ for $0^\circ \le \theta < 360^\circ$.
\step{Two different ratios and no factor in common, so squaring is the way in --- but it obliges me to test every candidate afterwards.}
\step{Squaring gives $2\cos\theta(\cos\theta + 1) = 0$, so the candidates are $90^\circ$, $270^\circ$ and $180^\circ$. Testing: $90^\circ$ gives $1 = 1$, $180^\circ$ gives $0 = 0$, but $270^\circ$ gives $1 = -1$.}
$\Rightarrow\quad \theta = \ans{90}^\circ, \ans{180}^\circ$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sin\theta + \cos\theta = 1$ for $0^\circ \le \theta < 360^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{0}^\circ, \ans{90}^\circ$}
\end{tryitbox}
