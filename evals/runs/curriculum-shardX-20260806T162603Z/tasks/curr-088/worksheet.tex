\wstitleblock{Mass and Liquid Volume: One-Step Problems}{Grade 3}{}

\begin{center}
\textit{Adding, subtracting, multiplying, and sharing masses and liquid volumes}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Grams (g) and kilograms (kg) measure \emph{mass}. Milliliters (mL) and
        liters (L) measure \emph{liquid volume}.
  \item Decide first: put together, take away, equal groups, or equal shares?
  \item Show your work, and write the unit with every answer.
\end{itemize}

% ---------------------------------------------------------------- Problem 1
\problem{A pear has a mass of 154 g. An apple has a mass of 138 g. What is the
total mass of the two fruits together?
\par\vspace*{3cm}\answerline{g}}

% ---------------------------------------------------------------- Problem 2
\problem{A bottle holds 500 mL of juice. Amir pours out 175 mL to fill a cup.
How many milliliters of juice are left in the bottle?
\par\vspace*{3cm}\answerline{mL}}

% ---------------------------------------------------------------- Problem 3
\problem{One brick has a mass of 2 kg. What is the mass of a stack of 9 bricks?
\par\vspace*{3cm}\answerline{kg}}

% ---------------------------------------------------------------- Problem 4
\problem{A cooler holds 24 L of lemonade. It is poured equally into 4 jugs.
How many liters of lemonade go into each jug?
\par\vspace*{3cm}\answerline{L}}

% ---------------------------------------------------------------- Problem 5
\problem{A bag of rice has a mass of 750 g. A bag of beans has a mass of 425 g.
How much more mass does the bag of rice have than the bag of beans?
\par\vspace*{3cm}\answerline{g}}

% ---------------------------------------------------------------- Problem 6
\problem{A fish tank holds 38 L of water. Keiko adds 17 L more. How many liters
of water are in the tank now?
\par\vspace*{3cm}\answerline{L}}

% ---------------------------------------------------------------- Problem 7
\problem{One spoonful of cough medicine is 5 mL. Dev takes 7 spoonfuls during
the whole week. How many milliliters of medicine is that in all?
\par\vspace*{3cm}\answerline{mL}}

% ---------------------------------------------------------------- Problem 8
\problem{A farmer divides 56 kg of potatoes equally into 8 sacks. How many
kilograms of potatoes go into each sack?
\par\vspace*{3cm}\answerline{kg}}
