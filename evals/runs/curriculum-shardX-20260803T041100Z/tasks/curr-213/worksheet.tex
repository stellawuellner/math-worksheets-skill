\wstitleblock{Linear Equations: One, None, or Infinitely Many Solutions}{Grade 8 / Pre-Algebra}{}

\begin{center}\small
\fbox{\begin{minipage}{0.93\linewidth}
\textbf{One solution, no solution, or infinitely many: deciding which, and what it means in context}
\begin{itemize}[itemsep=1pt, topsep=3pt]
  \item Simplify \emph{both} sides first (distribute, then combine like terms).
  \item \textbf{Different $x$-terms} $\Rightarrow$ the variable survives $\Rightarrow$ \textbf{one solution}.
  \item \textbf{Same $x$-term, different constants} $\Rightarrow$ a false statement such as $7 = 2$ $\Rightarrow$ \textbf{no solution}.
  \item \textbf{Same $x$-term and same constant} $\Rightarrow$ a true statement such as $6 = 6$ $\Rightarrow$ \textbf{infinitely many solutions}.
\end{itemize}
\end{minipage}}
\end{center}

\noindent\small\textbf{Directions:} Show the simplifying on both sides. Then write
your answer in one of these three forms: \textbf{one solution}, with the value of the
variable; \textbf{no solution}; or \textbf{infinitely many solutions}. Every quantity
below is stated with its units.\normalsize

\problem[6.5cm]{\textbf{Two saving plans.} Ana already has \textbf{5} dollars and saves
\textbf{3} dollars each week. Ben already has \textbf{9} dollars and saves \textbf{2}
dollars each week. After $w$ weeks their amounts are $3w + 5$ and $2w + 9$ dollars.
Solve $3w + 5 = 2w + 9$ and classify the equation.}

\problem[6cm]{\textbf{Two printers.} Each printer prints \textbf{4} pages per second.
Printer A had \textbf{7} pages already printed, printer B had \textbf{2}. After $t$
seconds the totals are $4t + 7$ and $4t + 2$ pages.

\begin{center}\small
\begin{tabular}{|l|c|c|c|}\hline
\textbf{Seconds $t$} & \textbf{1} & \textbf{2} & \textbf{3} \\\hline
Printer A: $4t + 7$ &  &  &  \\[0.35cm]\hline
Printer B: $4t + 2$ &  &  &  \\[0.35cm]\hline
\end{tabular}
\end{center}

Fill in the table, then solve $4t + 7 = 4t + 2$ and classify the equation.}

\problem[6cm]{\textbf{Perimeter two ways.} A rectangle is $x$ centimetres long and
\textbf{3} centimetres wide. Sam writes its perimeter as $2(x + 3)$; Dana writes it as
$2x + 6$. Setting the two expressions equal gives $2(x + 3) = 2x + 6$.
Expand the left side and classify the equation.}

\problem[6.5cm]{Solve if possible and classify: \quad $5(x - 2) = 3x + 4$}

\problem[6.5cm]{Solve if possible and classify: \quad $6x - 3(x + 2) = 3x + 1$}

\problem[6.5cm]{Solve if possible and classify: \quad $-2(4x - 6) + 5 = 17 - 8x$}

\problem[6.5cm]{\textbf{Two gyms.} Gym A charges a \textbf{25} dollar joining fee plus
\textbf{15} dollars each month. Gym B charges a \textbf{40} dollar joining fee plus
\textbf{10} dollars each month. After $m$ months the costs are $25 + 15m$ and
$40 + 10m$ dollars. Solve $25 + 15m = 40 + 10m$ and classify the equation.}

\problem{The equation $3(2x - 1) = ax + 7$ has \textbf{no solution}.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Expand the left side. \ansline
  \item What value must $a$ have? \ansline
  \item Explain why that value of $a$, together with the constant terms, forces the
        equation to have no solution. \ansline
\end{enumerate}}

\problem{Classify each equation below \emph{without} fully solving it where you can.
Write one solution (with its value), no solution, or infinitely many solutions.
\begin{enumerate}[label=(\alph*), itemsep=3.5cm]
  \item $8x - 3 = 8x + 1$ \ansline
  \item $4(x + 2) = 4x + 8$ \ansline
  \item $5x - 6 = 3x + 10$ \ansline
\end{enumerate}}

\problem{\textbf{Two ride prices.} Company P charges \textbf{3} dollars plus \textbf{2}
dollars per mile, so a ride of $n$ miles costs $2n + 3$ dollars. Company Q charges
\textbf{5} dollars plus \textbf{2} dollars per mile, so the same ride costs $2n + 5$
dollars.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Solve $2n + 3 = 2n + 5$ and classify the equation. \ansline
  \item Explain what your classification means for the two companies' prices: is there
        \emph{any} ride length for which they charge the same, and why? \ansline
\end{enumerate}}
