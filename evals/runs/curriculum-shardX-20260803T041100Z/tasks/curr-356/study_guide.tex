\sstitleblock{Adding, Subtracting, and Multiplying Polynomials}

\skillheading{Skill 1 --- Adding polynomials by like terms}

\begin{formulabox}\textbf{Rule:}\ Only \emph{like} terms combine --- same
variable, same exponent. Adding them adds the coefficients and leaves the
exponent alone, because $ax^n + bx^n = (a+b)x^n$. Write both polynomials in
descending order and stack the matching powers in columns.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $(2x^3 + 5x - 1) + (4x^3 - 3x + 6)$
\step{Both are already in descending order, so I can add column by column
instead of hunting for partners.}
\step{$x^3$: $2 + 4 = 6$. \quad $x$: $5 - 3 = 2$. \quad constant: $-1 + 6 = 5$.}
$\Rightarrow\quad \ans{6x^3+2x+5}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $(x^4 - 2x^2 + 3) + (3x^4 + 2x^2 - 8)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x^4-5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtracting: distribute the minus first}

\begin{formulabox}\textbf{Rule:}\ $A - B$ means $A + (-1)B$, so the minus sign
reaches \textbf{every} term of the second polynomial --- including the ones
already negative, which become positive. Rewrite the second polynomial with all
signs flipped, then add as in Skill 1.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $(5x^3 - 2x + 4) - (2x^3 + x - 9)$
\step{The parentheses carry a minus, so before combining anything I rewrite the
second polynomial with every sign flipped.}
\step{It becomes $-2x^3 - x + 9$; now add: $5 - 2 = 3$ on $x^3$, $-2 - 1 = -3$
on $x$, $4 + 9 = 13$ on the constant.}
$\Rightarrow\quad \ans{3x^3-3x+13}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $(4x^4 + x^2 - 6) - (x^4 - 3x^2 + 2)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^4+4x^2-8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Multiplying with an area box}

\begin{formulabox}\textbf{Rule:}\ Give the first factor's terms the rows and
the second factor's terms the columns. Each cell is one product (multiply
coefficients, add exponents); the answer is the sum of all the cells. An
$m$-term factor times an $n$-term factor always fills $m \times n$ cells --- if
you have fewer, you skipped one.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $(x + 4)(x^2 - 3x + 2)$
\step{Two terms times three terms means six products, so the box keeps me from
losing any of them.}
\step{Row $x$: $x^3,\ -3x^2,\ 2x$. \ Row $4$: $4x^2,\ -12x,\ 8$. \ Collect:
$-3x^2 + 4x^2 = x^2$ and $2x - 12x = -10x$.}
$\Rightarrow\quad \ans{x^3+x^2-10x+8}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $(x - 2)(x^2 + 5x - 1)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^3+3x^2-11x+2}$}
\end{tryitbox}

\skillheading{Skill 4 --- Checking a product two ways}

\begin{formulabox}\textbf{Rule:}\ Two independent checks:
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Substitute} a value into the factored form and the expanded
        form. Equal polynomials agree at every input, so any disagreement
        proves an error.
  \item \textbf{Factor back}. If the expansion factors to what you started
        with, the arithmetic held.
\end{itemize}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Does $(x+4)(x^2-3x+2) = x^3+x^2-10x+8$?
Test $x = 3$.
\step{One substitution cannot prove equality, but it can disprove it cheaply,
so I try a value that is easy to compute and not a root.}
\step{Factored: $(7)(9-9+2) = 14$. \ Expanded: $27 + 9 - 30 + 8$.}
$\Rightarrow\quad$ both give $\ans{14}$, so the expansion survives the test
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Factor $x^3 - x^2 - 6x$ back to a product
of three factors.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x(x-3)(x+2)}$}
\end{tryitbox}
