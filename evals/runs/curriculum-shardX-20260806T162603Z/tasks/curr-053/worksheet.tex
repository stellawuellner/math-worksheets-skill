\wstitleblock{Comparing Three-Digit Numbers}{Grade 2 Math}{}
\begin{center}
\textit{comparing with symbols, ordering, and real-world comparisons}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item Compare place by place: hundreds first, then tens, then ones.
  \item $>$ means \emph{greater than}, $<$ means \emph{less than}, $=$ means \emph{equal}.
\end{itemize}

\problem{Write $<$, $>$, or $=$.\par\vspace{6pt}
\hspace*{1.5cm} 372 \quad \ansblank \quad 327
\par\vspace*{2.6cm}}

\problem{Write $<$, $>$, or $=$.\par\vspace{6pt}
\hspace*{1.5cm} 458 \quad \ansblank \quad 485
\par\vspace*{2.6cm}}

\problem{Both sides are written in expanded form. Write $<$, $>$, or $=$.\par\vspace{6pt}
\hspace*{1.5cm} $300 + 40 + 2$ \quad \ansblank \quad $300 + 20 + 4$
\par\vspace*{2.6cm}}

\problem{Maya read \textbf{246} pages this year. Leo read \textbf{264} pages.\par\vspace{6pt}
Compare the page counts: \quad 246 \quad \ansblank \quad 264\par\vspace{6pt}
Who read more pages? \quad \ansblank
\par\vspace*{2.6cm}}

\problem{Write these numbers in order from \emph{least} to \emph{greatest}.\par\vspace{6pt}
\hspace*{1.5cm} 512 \qquad 251 \qquad 521\par\vspace{10pt}
\hspace*{1.5cm} \ansblank\ , \quad \ansblank\ , \quad \ansblank
\par\vspace*{2.6cm}}

\problem{Green Hill is \textbf{730} m tall. Stone Hill is \textbf{703} m tall.
Compare their heights in meters. Write $<$, $>$, or $=$.\par\vspace{6pt}
\hspace*{1.5cm} 730 m \quad \ansblank \quad 703 m
\par\vspace*{2.6cm}}

\problem[3cm]{Use the digits \textbf{2}, \textbf{5}, and \textbf{8} exactly once
each to make the \emph{greatest} possible three-digit number.}

\problem{Three classrooms collected cans for recycling. Room A collected
\textbf{318} cans, Room B collected \textbf{381} cans, and Room C collected
\textbf{313} cans. Write the three can counts in order from \emph{least} to
\emph{greatest}.\par\vspace{10pt}

\par\vspace*{2.6cm}}

\problem{Sam says \textbf{476} is greater than \textbf{512} because 7 tens beat

work space why Sam's tens-first thinking goes wrong.\par\vspace{6pt}
\hspace*{1.5cm} 476 \quad \ansblank \quad 512
