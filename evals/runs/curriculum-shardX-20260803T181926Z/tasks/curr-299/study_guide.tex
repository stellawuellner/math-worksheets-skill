\sstitleblock{Piecewise and Absolute-Value Models --- Boundaries and Endpoints}

\skillheading{Skill 1 --- Which rule owns the boundary?}

\begin{formulabox}
\textbf{The rule with the closed inequality owns the boundary.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $\le$ and $\ge$ \textbf{include} their endpoint; $<$ and $>$ \textbf{exclude} it.
  \item Test the input against the \emph{conditions} first, pick the one rule that
        is true, and only then substitute.
  \item Either rule may be the closed one --- top or bottom. There is no
        convention to memorise, so read the sign every time.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For
$P(x) = \begin{cases} 2x + 4, & x < 4 \\ x^2 - 10, & x \ge 4 \end{cases}$
find $P(4)$.

\step{The input is the boundary, so decide ownership before substituting: $4 < 4$
is false and $4 \ge 4$ is true, so the bottom rule owns it.}

\step{Substitute into the bottom rule: $4^2 - 10$.}

$P(4) = \ans{6}$. The top rule's value there would have been $12$, which is what
$P$ approaches from the left but never takes.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For
$Q(x) = \begin{cases} 6 - x, & x \le 1 \\ x + 6, & x > 1 \end{cases}$
find $Q(1)$.

\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Overlaps, gaps, and matching values}

\begin{formulabox}
\textbf{Every input must be owned by exactly one rule.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Both conditions closed at the boundary $\rightarrow$ \textbf{overlap}: two
        outputs for one input, so it is not a function. Open one side.
  \item Both strict $\rightarrow$ \textbf{gap}: the boundary has no output and is
        outside the domain. Close one side.
  \item \textbf{No jump} is a separate question: set the two formulas equal
        \emph{at} the boundary and solve for the unknown constant. Matching values
        does not change which rule owns the point.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $k$ so that
$R(x) = \begin{cases} 2x + k, & x < 5 \\ x^2 - 9, & x \ge 5 \end{cases}$
has no jump at its boundary.

\step{A jump is a mismatch of values at one input, so set the two rules equal at
the boundary and solve for the constant --- ownership is already settled by the
signs.}

\step{Set $2(5) + k = 5^2 - 9$ and solve for $k$.}

$\ans{k = 6}$, and the bottom rule still owns $x = 5$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $k$ so that
$S(x) = \begin{cases} 3x + k, & x < 2 \\ x^2 + 4, & x \ge 2 \end{cases}$
has no jump.

\rotatebox{180}{\footnotesize check: $\ans{k = 2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Endpoints of an absolute-value range}

\begin{formulabox}
\textbf{Solve the equation to find the endpoints; read the sign to include them.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $|A| = c$ splits into $A = c$ and $A = -c$. Those two solutions are the
        endpoints of the region $|A| \le c$.
  \item $|A| \le c$ keeps both endpoints (closed); $|A| < c$ drops both (open).
        The endpoints are the same numbers either way.
  \item In a tolerance context, the closed form accepts a part that hits the limit
        exactly --- reporting an open range rejects parts that meet the
        specification.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Give the endpoints of the region $|3x - 6| \le 9$ and say
whether they belong to it.

\step{The boundary of an absolute-value region is where the expression equals the
limit, so solve the equation $|3x - 6| = 9$ rather than guessing from a graph.}

\step{Split into $3x - 6 = 9$ and $3x - 6 = -9$, then solve each.}

$\ans{x = -1 \text{ or } x = 5}$. The sign is $\le$, so both endpoints belong and
the region is $-1 \le x \le 5$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Give the endpoints of the region $|2x + 1| \le 7$.

\rotatebox{180}{\footnotesize check: $\ans{x = -4 \text{ or } x = 3}$}
\end{tryitbox}
