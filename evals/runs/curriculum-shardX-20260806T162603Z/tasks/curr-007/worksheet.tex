\wstitleblock{Ten-Frames: See Numbers to Ten}{Kindergarten--Grade 1}{}
\begin{center}\textit{Five and more, almost ten, and building the frame}\end{center}

\noindent\textbf{Directions:} A ten-frame has two rows of five boxes.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item A full top row is \textbf{5} --- you never need to recount it.
  \item A full frame is \textbf{10}. Each empty box means one less than ten.
\end{itemize}
