\sstitleblock{Rewriting Expressions with Distributive Structure}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Distributing a factor}

\begin{formulabox}
\textbf{Rule:} $a(b + c) = ab + ac$. The outside factor meets \emph{every} term inside, signs
included. In a measurement problem this is ``$a$ copies of the whole thing'', so the unit of the
answer is the unit of the bracket.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A path is made of $6$ sections, each $(2x + 7)$ m long. Write the total
length in metres without brackets.
\step{Six copies of one length is a multiplication, and the length has two parts, so the 6 has to reach both of them.}
\step{$6\cdot 2x = 12x$ \ and \ $6\cdot 7 = 42$.}

$\Rightarrow\quad \ans{12x + 42}$ metres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Four sections, each $(3x - 5)$ m long: write the total length in metres
without brackets.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12x - 20}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Taking a common factor out}

\begin{formulabox}
\textbf{Rule:} $ab + ac = a(b + c)$ --- the same statement read right to left. Take the largest
number that divides every coefficient, and the lowest power of the variable that appears in
\emph{every} term. Check by re-distributing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ribbon of length $(10x + 35)$ cm is to be cut into equal pieces. Write
the length as a product.
\step{I want to see how many equal pieces the length splits into, so I look for the largest number that divides both coefficients.}
\step{$10$ and $35$ share $5$, and $10x \div 5 = 2x$, $35 \div 5 = 7$.}

$\Rightarrow\quad \ans{5(2x + 7)}$ --- five pieces of $(2x + 7)$ cm.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $8x^2 + 12x$ as a product of its greatest common monomial factor and
one remaining factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x(2x + 3)}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Choosing the form that answers the question}

\begin{formulabox}
\textbf{Rule:} the two forms are equal but they show different things.
\emph{Factored} form shows what the quantity is divisible by --- how many equal parts, what the
per-item rate is. \emph{Expanded} form shows the fixed part and the variable part as separate
terms. Rewrite into whichever one puts the asked-for quantity on the page.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A walkway $(5x + 20)$ m long is split into $5$ equal sections. How long is
one section, in metres?
\step{The question is about equal parts, so the factored form is the one that will show the answer without dividing term by term.}
\step{$5x + 20 = 5(x + 4)$, which reads as $5$ copies of $(x+4)$ metres.}
\step{One section is the second factor.}

$\Rightarrow\quad \ans{x + 4}$ metres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A walkway $(7x + 21)$ m long is split into $7$ equal sections. How long is
one section, in metres?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x + 3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a minus sign in front of a bracket is a factor of $-1$ and reaches every
term: $40x - 5(3x + 20) = 40x - 15x - 100$, not $40x - 15x + 100$.
\end{watchoutbox}
