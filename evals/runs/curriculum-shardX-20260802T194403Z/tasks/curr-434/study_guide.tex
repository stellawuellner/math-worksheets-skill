\sstitleblock{Quadrant and Parameter-Orientation Errors}

\noindent\small\textbf{The one habit that prevents all of these:} a calculator returns a
\emph{number}; you supply the \emph{position}. Before writing an angle, decide the quadrant from
the signs of $x$ and $y$ (or from the sign of $r$); before writing a parameter answer, decide how
far around the curve the parameter runs.\normalsize

\skillheading{1. Rectangular to polar: the quadrant decision}
\begin{formulabox}\textbf{Rule:}\quad $r = \sqrt{x^2 + y^2}$, \quad reference angle
$\alpha = \arctan\left|\dfrac{y}{x}\right|$, \quad then place $\alpha$:\\[3pt]
QI: $\theta = \alpha$ \quad QII: $\theta = 180^\circ - \alpha$ \quad
QIII: $\theta = 180^\circ + \alpha$ \quad QIV: $\theta = 360^\circ - \alpha$\\[3pt]
$\arctan$ only ever returns $(-90^\circ, 90^\circ)$ --- QI or QIV --- so it can never place a
QII or QIII point by itself.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Write $(-4, 3)$ in polar form with
$0^\circ \le \theta < 360^\circ$.
\step{$x < 0$, $y > 0$, so the point is in Quadrant II --- take the reference angle and reflect it, do not trust the raw arctan.}
\step{$\alpha = \arctan\left(\frac{3}{4}\right) = 36.87^\circ$, so $\theta = 180^\circ - 36.87^\circ$; also $r = \sqrt{16 + 9} = 5$.}
$\Rightarrow\ \ans{\theta = 143.13^\circ}$
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Write $(5, -12)$ in polar form with
$0^\circ \le \theta < 360^\circ$. (Which quadrant?)\\[2pt]
\rotatebox{180}{\footnotesize check: $r = 13$, $\ans{\theta = 292.62^\circ}$}\end{tryitbox}

\skillheading{2. Negative $r$ and equivalent names}
\begin{formulabox}\textbf{Rule:}\quad $(-r, \theta)$ is the same point as
$(r, \theta + 180^\circ)$ --- go out along $\theta$, then straight through the pole.\\[3pt]
Conversion formulas take the \emph{signed} $r$: $x = r\cos\theta$, $y = r\sin\theta$.
To rename a point with a positive $r$, add or subtract $180^\circ$ --- whichever keeps the angle
inside $[0^\circ, 360^\circ)$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Find the $x$-coordinate of the polar point
$(-3, 40^\circ)$.
\step{The sign of $r$ is part of the data, so substitute $r = -3$ rather than dropping the minus sign.}
\step{$x = r\cos\theta = -3\cos 40^\circ = -3(0.766\ldots)$ \ (equivalently, the point is $(3, 220^\circ)$).}
$\Rightarrow\ \ans{x = -2.30}$
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Find the $y$-coordinate of the polar point
$(-5, 110^\circ)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -4.70}$}\end{tryitbox}

\skillheading{3. Parametric orientation: which way, from where}
\begin{formulabox}\textbf{Rule:}\quad the same curve has many parametrizations; the parameter
carries the \emph{starting point} and the \emph{direction}, which the Cartesian equation does not.
\\[3pt]
To read orientation, tabulate $t = 0, \frac{\pi}{2}, \pi, \frac{3\pi}{2}$ and watch the points
move. Swapping $\sin$ and $\cos$ (or replacing $t$ by $-t$) keeps the curve and reverses the trip.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad For $x = 4\sin t$, $y = 4\cos t$, find $x$ at
$t = \frac{\pi}{6}$.
\step{Substitute into the rule for $x$ as written --- the $\sin$ belongs to $x$ here, which is exactly the swap that trips people up.}
\step{$x = 4\sin\frac{\pi}{6} = 4\left(\frac{1}{2}\right)$. \ (At $t = 0$ this curve is at $(0, 4)$ and moves right: clockwise.)}
$\Rightarrow\ \ans{x = 2}$
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad For $x = 4\sin t$, $y = 4\cos t$, find $x$ at
$t = \frac{\pi}{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3.46}$}\end{tryitbox}

\skillheading{4. Every parameter value, not just the first}
\begin{formulabox}\textbf{Rule:}\quad $\arcsin$, $\arccos$ and $\arctan$ each return ONE value.
On a full trip round a closed curve, a height or an $x$-value is usually reached \textbf{twice}.
\\[3pt]
$\sin t = k$: \ $t = \alpha$ and $180^\circ - \alpha$. \qquad
$\cos t = k$: \ $t = \alpha$ and $360^\circ - \alpha$. \\
Then keep only the values inside the stated parameter interval.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad On $0^\circ \le t < 360^\circ$, solve
$2\cos t - 1 = 0$.
\step{Cosine is positive in Quadrants I and IV, so expect two parameter values, not one.}
\step{$\cos t = \frac{1}{2}$ gives the reference angle $60^\circ$; the QIV partner is $360^\circ - 60^\circ$.}
\par\noindent$\Rightarrow\ \ans{t = 60^\circ,\ 300^\circ}$
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad On $0^\circ \le t < 360^\circ$, solve
$2\sin t + 1 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 210^\circ,\ 330^\circ}$}\end{tryitbox}
