\sstitleblock{Reading and Plotting Points on a Grid}

\skillheading{Skill 1 --- Naming a point you can see}

\begin{formulabox}
\textbf{Rule:} An ordered pair is $(x, y)$ --- \textbf{across first, then up}.\\[3pt]
$x$ counts squares to the right of the origin; $y$ counts squares above it. Swapping them names a different place, so ``across, then up'' is the whole of this skill.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table records where three points sit. Across: $A$ 3, $B$ 7, $C$ 5. Up: $A$ 6, $B$ 2, $C$ 9. Write the ordered pair for $B$.
\step{The pair must be written across-number first, so I read $B$ out of the across row before I touch the up row.}
\step{The across row gives $B$ as 7 and the up row gives $B$ as 2.}
$B = (\ans{7}, \ans{2})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From the same two rows, write the ordered pair for $C$.\\[2pt]
\rotatebox{180}{\footnotesize check: $C = (\ans{5}, \ans{9})$}
\end{tryitbox}

\skillheading{Skill 2 --- Moving from one point to another}

\begin{formulabox}
\textbf{Rule:} Moving \emph{right} changes only $x$; moving \emph{up} changes only $y$.\\[3pt]
new $x$ $=$ old $x$ $+$ steps right \qquad new $y$ $=$ old $y$ $+$ steps up\\[3pt]
Left and down subtract instead. Nothing you do sideways can change the up number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Point $R$ is at $(3, 4)$. Move 5 units right and 2 units up. Where do you land?
\step{Each direction touches its own coordinate, so I handle the sideways move and the upward move separately instead of adding everything together.}
\step{Across: $3 + 5 = 8$. Up: $4 + 2 = 6$.}
Landing point: $(\ans{8}, \ans{6})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Point $S$ is at $(6, 1)$. Move 2 units right and 7 units up. Where do you land?\\[2pt]
\rotatebox{180}{\footnotesize check: $(\ans{8}, \ans{8})$}
\end{tryitbox}

\skillheading{Skill 3 --- Plotting points and measuring between them}

\begin{formulabox}
\textbf{Rule:} To plot $(x, y)$, put a finger on the origin, count $x$ squares right, then $y$ squares up, and mark the corner where you stop.\\[3pt]
Two points with the same $y$ lie on a horizontal line, and the distance between them is the difference of their $x$-values. Two points with the same $x$ lie on a vertical line, and the difference of their $y$-values is the distance.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plot $D(2, 1)$ and $E(8, 1)$. How many units apart are they?
\step{Both points have the same up number, so they sit on one horizontal line and I can count squares across instead of measuring diagonally.}
\step{The across numbers are 2 and 8, so the gap is $8 - 2$.}
Distance: $\ans{6}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Plot $F(4, 2)$ and $G(4, 9)$. How many units apart are they?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$ units}
\end{tryitbox}

\skillheading{Skill 4 --- Turning a growing pattern into points}

\begin{formulabox}
\textbf{Rule:} A rule of the form ``start at $s$, change by $c$ each step'' makes ordered pairs (step number, value).\\[3pt]
value after $n$ steps $=$ $s + c \times n$ \quad (use $-$ when the quantity shrinks)\\[3pt]
The step number is the across coordinate and the value is the up coordinate. Plotted, a constant-change rule always makes a straight line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tree is 4 metres tall and grows 2 metres a year. Write the ordered pair for year 6.
\step{The year is the across number and the height is the up number, so I only have to compute the height and the pair is finished.}
\step{Six years of growth adds $2 \times 6 = 12$ to the starting 4, giving $4 + 12$.}
Year 6 pair: $(6, \ans{16})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tank holds 30 litres and loses 4 litres an hour. Write the ordered pair for hour 5.\\[2pt]
\rotatebox{180}{\footnotesize check: $(5, \ans{10})$}
\end{tryitbox}
