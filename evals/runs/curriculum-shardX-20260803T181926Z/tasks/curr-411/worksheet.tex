\wstitleblock{Solving Exponential Equations by Writing a Common Base}{Precalculus}{}

\begin{center}\small\textit{Equating exponents when the bases already match,
rewriting both sides over a common base, and solving exponential models the same
way}\end{center}

\noindent\textbf{The rule and the model.} For $b > 0$ and $b \neq 1$, the function
$b^u$ is one-to-one, so
\[ b^{\,u} = b^{\,v} \iff u = v. \]
Matching bases is therefore not a trick --- it is what licenses dropping the bases
and solving the ordinary equation left behind. Use this table of small powers to
spot a shared base:

\begin{center}\small
\begin{tabular}{|l|c|c|c|c|c|c|c|c|}\hline
\textbf{Power} & $b^{1}$ & $b^{2}$ & $b^{3}$ & $b^{4}$ & $b^{5}$ & $b^{6}$ & $b^{7}$ & $b^{8}$\\\hline
Base 2 & 2 & 4 & 8 & 16 & 32 & 64 & 128 & 256\\\hline
Base 3 & 3 & 9 & 27 & 81 & 243 & --- & --- & ---\\\hline
Base 5 & 5 & 25 & 125 & 625 & --- & --- & --- & ---\\\hline
\end{tabular}
\end{center}

\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item Read every number in the equation off the table. If they are all powers of
        one base, rewrite each side as that base raised to a power, using
        $(b^{m})^{n} = b^{mn}$ and $b^{m}b^{n} = b^{m+n}$.
  \item A reciprocal is a negative exponent: $\tfrac{1}{2} = 2^{-1}$ and
        $\tfrac{1}{125} = 5^{-3}$.
  \item Once both sides read $b^{\text{something}}$, set the two exponents equal
        and solve. Then substitute back to check.
  \item Show the rewriting line before the exponent equation --- that line is
        where the work is.
\end{itemize}
\vspace{0.1cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[5cm]{The bases already match. Solve for $x$.
\[ 2^{\,x+1} = 2^{\,3x-5} \]}

\problem[5cm]{The bases already match. Solve for $x$.
\[ 3^{\,2x} = 3^{\,x+4} \]}

\problem[5cm]{Write the right-hand side as a power of 3 first, then solve for $x$.
Leave the answer as a fraction.
\[ 9^{\,x} = 27 \]}

\problem[5cm]{Solve for $x$, then check your answer by substituting it back into
both exponents.
\[ 5^{\,3x-1} = 5^{\,x+9} \]}

\problem[5cm]{Both sides are powers of 2. Rewrite each side over base 2, then
solve for $x$.
\[ 4^{\,2x} = 8^{\,x+1} \]}

\problem[5cm]{Write both sides as powers of 2 --- remember that
$\tfrac{1}{2} = 2^{-1}$ --- then solve for $x$.
\[ \left(\tfrac{1}{2}\right)^{x} = 32 \]}

\problem[6.5cm]{\textbf{Doubling model.} A culture starts with 5 cells and doubles
every hour, so after $t$ hours the count is $N = 5 \cdot 2^{\,t}$.

\begin{center}\small
\begin{tabular}{|l|c|c|c|c|c|}\hline
\textbf{Hours }$t$ & 0 & 1 & 2 & 3 & 4\\\hline
\textbf{Cells }$N$ & 5 & 10 & 20 & 40 & 80\\\hline
\end{tabular}
\end{center}

\noindent Extend the pattern in your head, then solve $5 \cdot 2^{\,t} = 320$
algebraically: divide by 5 first so that the equation becomes a single power of 2
on each side. How many hours does it take?}

\problem[6.5cm]{\textbf{Check the claim.} A student says that $x = -2$ solves
\[ 25^{\,x-1} = 125^{\,x}. \]
Rewrite both sides over base 5 and solve the equation yourself. Then confirm the
claim by computing the difference $25^{\,x-1} - 125^{\,x}$ at $x = -2$, and state
what value that difference must have if the claim is true.}

\problem[7cm]{\textbf{Decay model.} A 240 mg sample loses half its mass every 5
years, so the mass remaining after $t$ years is $M = 240\left(\tfrac{1}{2}\right)^{t/5}$.
Solve
\[ 240\left(\tfrac{1}{2}\right)^{t/5} = 15 \]
for $t$ by writing both sides as powers of 2. Say how many half-lives that is, and
explain why the answer had to be a whole number of half-lives here.}

\problem[7cm]{\textbf{Challenge.} First write $9^{\,x}\cdot 27^{\,x+1}$ as a single
power of 3. Then use that form to solve
\[ 9^{\,x}\cdot 27^{\,x+1} = 81. \]
Give both the single-power form and the value of $x$.}
