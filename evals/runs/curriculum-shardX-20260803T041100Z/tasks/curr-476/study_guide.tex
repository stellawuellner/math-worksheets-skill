\sstitleblock{Using u-Substitution in Indefinite and Definite Integrals}

\begin{formulabox}
\textbf{The chain rule, read backwards.} If $u = g(x)$ then $du = g'(x)\,dx$, and
$\int f(g(x))\,g'(x)\,dx = \int f(u)\,du$.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \textbf{Choose $u$} to be the inner function --- what is under a root,
inside a bracket raised to a power, in an exponent, or in a denominator.
\item \textbf{Test it:} write $du$ and check that the rest of the integrand is a
\emph{constant} multiple of it. A leftover $x$ means the choice was wrong.
\item \textbf{Indefinite:} back-substitute and add $+\,C$.
\textbf{Definite:} convert both limits to $u$ and never return to $x$.
\end{itemize}
\end{formulabox}

\skillheading{1. When the derivative is already there}

\begin{formulabox}
The easiest case: the integrand is exactly $f(u)\,du$. Name $u$, write $du$, swap
everything, integrate with the power rule, back-substitute.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $\displaystyle\int 3x^2\left(x^3+2\right)^4 dx$.
\step{The bracket is the inner function, and its derivative $3x^2$ is already
outside, so no adjusting factor is needed.}
\step{With $u = x^3+2$ and $du = 3x^2\,dx$, the integral is
$\int u^4\,du = \frac{u^5}{5}$.}
\[ \ans{\tfrac{1}{5}\left(x^3+2\right)^5 + C} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $\displaystyle\int 2x\left(x^2-7\right)^5 dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{6}(x^2-7)^6 + C}$}
\end{tryitbox}

\skillheading{2. When a constant factor is missing}

\begin{formulabox}
If $du = k\,g'(x)\,dx$ but the integrand supplies only $g'(x)\,dx$, multiply the
integral by $\frac{1}{k}$. Constants move across the integral sign freely;
variables never do. If you have to move an $x$, change your choice of $u$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $\displaystyle\int x\sin\!\left(x^2\right)dx$.
\step{Take $u = x^2$ and compare $du = 2x\,dx$ with the $x\,dx$ actually present
--- the gap is a constant, so it is fixable.}
\step{$x\,dx = \frac{1}{2}du$, so the integral is
$\frac{1}{2}\int \sin u\,du = -\frac{1}{2}\cos u$.}
\[ \ans{-\tfrac{1}{2}\cos\!\left(x^2\right) + C} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $\displaystyle\int x^2 e^{x^3}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}e^{x^3} + C}$}
\end{tryitbox}

\skillheading{3. Definite integrals: change the limits}

\begin{formulabox}
Convert both bounds with the same rule $u = g(x)$, then evaluate entirely in $u$.
This is faster than back-substituting and it removes the classic error of
evaluating a $u$-antiderivative at the old $x$-limits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $\displaystyle\int_{0}^{1} 2x\left(x^2+1\right)^3 dx$.
\step{Convert the limits along with the integrand so the answer never has to
return to $x$.}
\step{$u = x^2+1$ gives $du = 2x\,dx$, with $u$ running from 1 to 2, so the
integral is $\int_{1}^{2} u^3\,du = \left[\frac{u^4}{4}\right]_{1}^{2}
= \frac{16-1}{4}$.}
\[ \ans{\tfrac{15}{4}} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\displaystyle\int_{0}^{1} 3x^2\left(x^3+1\right)^2
dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{7}{3}}$}
\end{tryitbox}

\skillheading{4. Improper integrals: substitute, then take a limit}

\begin{formulabox}
$\int_{a}^{\infty} f = \lim_{b\to\infty}\int_{a}^{b} f$. Do the substitution on
the proper integral over $[a,b]$, evaluate, and take the limit last. A finite
limit means the integral converges to it; an infinite limit means it diverges.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $\displaystyle\int_{0}^{b}\frac{2x}{\left(x^2+1\right)^2}dx$
equals $1 - \dfrac{1}{b^2+1}$. What happens as $b\to\infty$?
\step{Take the limit of the evaluated antiderivative, since the substitution
work is already finished.}
\step{As $b$ grows, $\frac{1}{b^2+1}\to 0$, so only the constant term survives
and the integral converges.}
\[ \ans{1} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\displaystyle\int_{0}^{\infty}
\frac{4x}{\left(x^2+1\right)^2}dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}
