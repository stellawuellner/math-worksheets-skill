\wstitleblock{Approximating Irrational Roots on a Number Line}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Bracketing a root between whole numbers, squeezing it to a decimal, and using the
estimate in measurement}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item None of these roots is exact, so every answer is an \emph{estimate} --- the
        symbol $\approx$ belongs in your work.
  \item \textbf{Bracket first:} name the perfect square just below the radicand and the
        one just above it. Their roots are the whole numbers your value sits between.
  \item \textbf{Then squeeze:} square candidate decimals until two of them trap the
        radicand, and round as the problem asks.
  \item Measurement answers must carry their unit.
\end{itemize}
