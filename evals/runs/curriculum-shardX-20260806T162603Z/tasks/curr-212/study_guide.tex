\sstitleblock{Equations with Brackets and Fractions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Distributing over a bracket}

\begin{formulabox}
\textbf{Rule:} \quad $a(b + c) = ab + ac$ \quad --- the outside factor multiplies
\emph{every} term inside, not just the first.
\par\smallskip
A minus sign outside counts as part of the factor, so it changes the sign of every term
inside. After distributing, the equation is an ordinary two-step equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4(x + 3) = 32$.
\step{A bracket blocks any collecting, so the first job is to distribute the outside factor across both terms inside.}
\step{That gives a two-step equation: subtract the constant from both sides, then divide both sides by the coefficient.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $6(x - 2) = 30$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Clearing a denominator}

\begin{formulabox}
\textbf{Rule:} multiply \emph{both whole sides} by the least common denominator. Every
term is multiplied --- including terms that carry no fraction at all.
\par\smallskip
That single move turns a fraction equation into an ordinary one, which is why it is worth
doing before anything else.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{5} + 2 = 6$.
\step{A denominator is in the way, so multiply both sides by it and the fraction disappears in one step.}
\step{Multiplying every term by five clears the fraction and leaves a two-step equation; subtract the constant, then divide.}

$\Rightarrow\quad \ans{x = 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{3x}{4} = 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 12}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the two classic slips are multiplying only the term that had the
fraction, and distributing to only the first term inside a bracket. In both cases something
on the page was left untouched --- check that every term took the move.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- When a bracket sits over a denominator}

\begin{formulabox}
\textbf{Method:} a fraction bar groups everything above it, so treat the numerator as a
bracket. Multiply both sides by the least common denominator first; the numerators are
then multiplied as whole brackets, and only afterwards are they distributed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x - 1}{3} = \dfrac{x + 5}{6}$.
\step{Both sides are single fractions with grouped numerators, so clear both denominators at once with their least common multiple.}
\step{Multiplying through leaves a bracket on the left to distribute; then collect the variable terms on one side and finish.}

$\Rightarrow\quad \ans{x = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x + 2}{2} = \dfrac{x + 8}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}
