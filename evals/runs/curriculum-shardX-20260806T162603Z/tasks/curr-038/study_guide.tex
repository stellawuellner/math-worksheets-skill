\sstitleblock{Finding a Missing Number}

% =================== SKILL 1 ===================
\skillheading{See the parts and the whole}

\begin{formulabox}
\textbf{Rule:} In $\square + \triangle = $ whole, the two parts go together
to make the whole. If one part is hiding, the whole and the other part can
find it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The whole is 11 and one part is 4. Find the other part.
\step{Put what we know in a part--whole box: whole 11 on top, part 4 below --- the empty spot is what we want.}
\step{4 and how many more make 11? \ 4 and 7 make 11.}

$\Rightarrow\quad 4 + \ans{7} = 11$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The whole is 12 and one part is 4. Find the other part.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Count on to the total}

\begin{formulabox}
\textbf{Rule:} Start at the part you know. Count up to the whole. The number
of jumps is the missing number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $9 + \square = 13$
\step{We know where to start and where to land --- count on from 9 up to 13 and keep track of the jumps.}
\step{9 $\to$ 10, 11, 12, 13. That is 4 jumps.}

$\Rightarrow\quad 9 + \ans{4} = 13$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $8 + \square = 13$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Use subtraction to find and check}

\begin{formulabox}
\textbf{Rule:} Addition and subtraction are in the same fact family. The
missing part is whole $-$ known part, and adding back checks your answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\square + 5 = 12$
\step{The missing part comes first, so counting on is hard --- subtract the known part from the whole instead.}
\step{$12 - 5 = 7$. \ Check: $7 + 5 = 12$. It works!}

$\Rightarrow\quad \ans{7} + 5 = 12$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\square + 6 = 13$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not add the two numbers you can see! In
$\square + 6 = 13$, the answer is NOT $19$. The whole is already 13 --- the
missing number is a part, so it must be smaller than the whole.
\end{watchoutbox}
