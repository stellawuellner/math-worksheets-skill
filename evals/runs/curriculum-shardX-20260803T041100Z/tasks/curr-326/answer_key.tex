\problem{\textbf{Distance from $A(1,2)$ to $B(4,6)$.}

\textbf{Solution:} The horizontal leg is $4 - 1 = 3$ and the vertical leg is
$6 - 2 = 4$, so the segment is the hypotenuse of a $3$--$4$ right triangle.
\begin{align*}
  AB &= \sqrt{(4-1)^2 + (6-2)^2} \\
     &= \sqrt{3^2 + 4^2} = \sqrt{9 + 16} = \sqrt{25}
\end{align*}
\[ \ans{AB = 5} \]}

\problem{\textbf{Midpoint of $A(2,-3)$ and $B(8,7)$.}

\textbf{Solution:} Average each coordinate separately --- the $x$'s with the $x$'s and
the $y$'s with the $y$'s.
\[ M = \left( \frac{2+8}{2},\ \frac{-3+7}{2} \right) = \left( \frac{10}{2},\ \frac{4}{2} \right) \]
\[ \ans{M = (5,\ 2)} \]
Sanity check: $5$ is halfway between $2$ and $8$, and $2$ is halfway between $-3$ and $7$.}

\problem{\textbf{Distance from $A(-3,1)$ to $B(2,13)$.}

\textbf{Solution:} Subtracting a negative adds: the horizontal leg is
$2 - (-3) = 5$, and the vertical leg is $13 - 1 = 12$.
\begin{align*}
  AB &= \sqrt{(2-(-3))^2 + (13-1)^2} \\
     &= \sqrt{5^2 + 12^2} = \sqrt{25 + 144} = \sqrt{169}
\end{align*}
\[ \ans{AB = 13} \]
This is the $5$--$12$--$13$ triple.

\noindent\textit{Watch for:} an answer of $17$ --- that is $5 + 12$, adding the legs
instead of using the Pythagorean relationship between them.}

\problem{\textbf{Midpoint of $A(-5,4)$ and $B(3,-2)$.}

\textbf{Solution:}
\[ M = \left( \frac{-5+3}{2},\ \frac{4+(-2)}{2} \right) = \left( \frac{-2}{2},\ \frac{2}{2} \right) \]
\[ \ans{M = (-1,\ 1)} \]
A negative $x$ with a positive $y$ places the midpoint in Quadrant II.}

\problem{\textbf{Exact distance from $A(0,0)$ to $B(6,6)$.}

\textbf{Solution:} Both legs measure $6$, so this is an isosceles right triangle and
the exact answer is a radical, not a decimal.
\begin{align*}
  AB &= \sqrt{(6-0)^2 + (6-0)^2} = \sqrt{36 + 36} = \sqrt{72} \\
     &= \sqrt{36 \cdot 2} = \sqrt{36}\,\sqrt{2}
\end{align*}
\[ \ans{AB = 6\sqrt{2}} \]
(That is about $8.49$, but the problem asked for exact radical form.)}

\problem{\textbf{Find $B$ given $M(3,5)$ and $A(-1,2)$.}

\textbf{Solution:} The midpoint formula is two averaging equations. Write each one and
solve for the missing coordinate.
\begin{align*}
  \frac{-1 + x_B}{2} = 3 \;&\Longrightarrow\; -1 + x_B = 6 \;\Longrightarrow\; x_B = 7 \\
  \frac{2 + y_B}{2} = 5 \;&\Longrightarrow\; 2 + y_B = 10 \;\Longrightarrow\; y_B = 8
\end{align*}
\[ \ans{B = (7,\ 8)} \]
Check by recomputing the midpoint of $A(-1,2)$ and $B(7,8)$:
$\left(\dfrac{-1+7}{2},\ \dfrac{2+8}{2}\right)$, which gives the given midpoint
$\ans{M = (3,\ 5)}$ $\checkmark$}

\problem{\textbf{Median of triangle $PQR$: $P(1,1)$, $Q(7,1)$, $R(4,6)$.}

\textbf{(a) Midpoint of $\overline{PQ}$.} Both endpoints have $y = 1$, so the segment is
horizontal and only the $x$-coordinate needs averaging.
\[ M = \left( \frac{1+7}{2},\ \frac{1+1}{2} \right) = \left( \frac{8}{2},\ \frac{2}{2} \right) \]
\[ \ans{M = (4,\ 1)} \]

\textbf{(b) Length of the median $\overline{RM}$.} Now use the distance formula on
$R(4,6)$ and $M(4,1)$. Both have $x = 4$, so the horizontal leg is $0$ and the segment
is vertical.
\[ RM = \sqrt{(4-4)^2 + (1-6)^2} = \sqrt{0 + 25} = \sqrt{25} \]
\[ \ans{RM = 5} \]
The median from $R$ is also the triangle's altitude here, because $\overline{PQ}$ is
horizontal and $\overline{RM}$ is vertical.}

\problem{\textbf{Coordinate proof: diagonals of $ABCD$ with $A(0,0)$, $B(6,2)$, $C(8,7)$, $D(2,5)$.}

\textbf{Solution --- computation.} Diagonal $\overline{AC}$ joins $A(0,0)$ to $C(8,7)$:
\[ \left( \frac{0+8}{2},\ \frac{0+7}{2} \right) = \ans{(4,\ 3.5)} \]
Diagonal $\overline{BD}$ joins $B(6,2)$ to $D(2,5)$:
\[ \left( \frac{6+2}{2},\ \frac{2+5}{2} \right) = \ans{(4,\ 3.5)} \]

\textbf{Model proof (open response --- grade the reasoning).}
The two diagonals have the same midpoint, $(4, 3.5)$. A midpoint of a segment is by
definition the point that divides it into two congruent halves, so that shared point
cuts $\overline{AC}$ into two equal parts and cuts $\overline{BD}$ into two equal parts.
Because the point lies on both diagonals at once, it is exactly where they cross.
Therefore each diagonal passes through the other's midpoint --- that is, the diagonals
bisect each other.

\noindent\textit{Look for:} both midpoints computed and stated to be equal; the word
\emph{bisect} tied to the definition of midpoint; the observation that a common
midpoint is the intersection point. A student who computes only one midpoint has not
proved anything.}
