\sstitleblock{Teen Numbers: One Ten and Some Ones}

\begin{formulabox}
\textbf{Every teen number is one ten and some ones.}\\[3pt]
$11 = 10 + 1$ \quad $12 = 10 + 2$ \quad $13 = 10 + 3$ \quad $14 = 10 + 4$
\quad $15 = 10 + 5$\\[2pt]
$16 = 10 + 6$ \quad $17 = 10 + 7$ \quad $18 = 10 + 8$ \quad $19 = 10 + 9$
\end{formulabox}

\skillheading{1. Building a teen number}

\begin{formulabox}
Fill one ten-frame all the way up --- that is the ten. Put the leftovers in the
next frame --- those are the ones. Then say it: \emph{ten and some ones}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} One full frame and one frame with four counters. How many?
\step{The first frame is full, and a full frame always holds ten --- so count
only the second frame.}
\step{The second frame has four counters, so the number is ten and four.}
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \draw[thick] (0,0) rectangle (5,2);
  \foreach \i in {1,2,3,4} { \draw (\i,0) -- (\i,2); }
  \draw (0,1) -- (5,1);
  \foreach \i in {0,1,2,3,4} { \fill[blue!55!black] (\i+0.5,1.5) circle (0.3); }
  \foreach \i in {0,1,2,3,4} { \fill[blue!55!black] (\i+0.5,0.5) circle (0.3); }
  \draw[thick] (6,0) rectangle (11,2);
  \foreach \i in {7,8,9,10} { \draw (\i,0) -- (\i,2); }
  \draw (6,1) -- (11,1);
  \foreach \i in {6,7,8,9} { \fill[blue!55!black] (\i+0.5,1.5) circle (0.3); }
\end{tikzpicture}
\end{center}
So the number is $\ans{14}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} One full frame, and two counters in the next frame. How
many?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\skillheading{2. Breaking a teen number apart}

\begin{formulabox}
Going the other way: take the ten out first, and whatever is left is the ones.
The ones of a teen number are never more than nine, because ten more would fill
a second frame.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} 16 is one ten and how many ones?
\step{Take out the ten first --- one full frame --- and only then count what is
left.}
\step{Sixteen with ten taken away leaves the ones.}
So 16 is one ten and $\ans{6}$ ones.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} 13 is one ten and how many ones?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ ones}
\end{tryitbox}

\skillheading{3. Comparing two teen numbers}

\begin{formulabox}
Both teen numbers already have one ten each, so the tens tie. Cover them up and
compare the \emph{ones}. The bigger ones make the bigger number. Write $<$ for
\emph{is less than} and $>$ for \emph{is greater than} --- the open mouth
always faces the bigger number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Which is less, $10 + 3$ or $10 + 8$?
\step{Both have one ten, so the tens cannot decide it --- compare the ones
instead.}
\step{Three ones against eight ones: three is the smaller pile.}
So $\ans{13 < 18}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $<$ or $>$ between $10 + 6$ and $10 + 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16 > 12}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the ones of a teen number are the \emph{leftovers}, not
the whole number. 15 is one ten and 5 ones --- never one ten and 15 ones.
\end{watchoutbox}
