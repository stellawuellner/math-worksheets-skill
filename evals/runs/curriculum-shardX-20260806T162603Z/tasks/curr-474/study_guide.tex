% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Limits: Order and Conversion}

\begin{formulabox}
\textbf{Order.} $\displaystyle\int_b^a f = -\int_a^b f$. A definite integral
written with the larger number on the bottom is not a typo to be tidied away ---
it is the negative of the usual one.\par
\textbf{Conversion.} After a substitution, the limits belong to the new
variable. Convert them at the same moment you convert the integrand.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int_4^1 2x\,dx$.
\step{The lower limit is larger than the upper one, so the value will be negative; evaluate in the order written rather than swapping the numbers.}
\step{$\left[x^2\right]_4^{1} = 1 - 16$}

$\Rightarrow\quad \ans{-15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_3^0 2x\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-9}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Signed Integral or Geometric Area}

\begin{formulabox}
\textbf{Two different questions.} $\displaystyle\int_a^b f$ counts area below
the axis as negative. \emph{Area} counts it as positive, so split the interval
at every sign change and add the magnitudes.\par
The same distinction separates displacement (signed) from total distance
travelled (unsigned) in a motion problem.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the area between $y = x$ and the $x$-axis on
$[-1, 1]$.
\step{The question asks for area, and the graph crosses the axis inside the interval, so the signed integral would cancel to zero and must not be used.}
\step{Split at $0$: each triangle has area $\tfrac{1}{2}$, so the total is $\tfrac{1}{2} + \tfrac{1}{2}$.}

$\Rightarrow\quad \ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the area between $y = x$ and the $x$-axis on
$[-2, 2]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Substitution Bookkeeping}

\begin{formulabox}
\textbf{Three things change together.} The integrand, the differential, and the
limits. If $u = g(x)$ then $du = g'(x)\,dx$ --- solve that for the piece you
actually have, and carry any constant factor out front.\par
An answer that still contains the old variable, or that uses the old limits, is
half-converted.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int_0^1 x\left(x^2+1\right)^2dx$.
\step{The integrand contains a function and a multiple of its derivative, so substitution is the tool; the limits must move with it.}
\step{$u = x^2+1$, $du = 2x\,dx$, so $x\,dx = \tfrac{1}{2}du$ and $u$ runs from $1$ to $2$: $\tfrac{1}{2}\left[\tfrac{u^3}{3}\right]_1^2 = \tfrac{1}{2}\left(\tfrac{8}{3} - \tfrac{1}{3}\right)$.}

$\Rightarrow\quad \ans{\tfrac{7}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_0^1 x\left(x^2+1\right)^3dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{15}{8}}$}
\end{tryitbox}
