\sstitleblock{Solving AAS and ASA Triangles with the Law of Sines}

\skillheading{Skill 1 --- The third angle first}

\begin{formulabox}
\textbf{Angle sum:} $A + B + C = 180^\circ$. Two angles always give the third.\par
\textbf{Naming:} side $a$ lies opposite angle $A$, $b$ opposite $B$, $c$ opposite $C$. A side and the angle across from it are a \emph{pair}.\par
Two angles fix the shape; one side fixes the size.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A triangle has $A = 47^\circ$, $B = 68^\circ$ and $c = 12$. Find $C$.
\step{Two angles are known and the third is asked for, so this is the angle sum, not the law of sines --- the side plays no part.}
\step{$C = 180^\circ - 47^\circ - 68^\circ$}
$\Rightarrow\quad C = \ans{65}^\circ$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A triangle has $A = 39^\circ$, $C = 84^\circ$ and $b = 9$. Find $B$.\\[2pt]
\rotatebox{180}{\footnotesize check: $B = \ans{57}^\circ$}
\end{tryitbox}

\skillheading{Skill 2 --- AAS: the side already has a partner}

\begin{formulabox}
\textbf{Law of sines:} $\dfrac{a}{\sin A} = \dfrac{b}{\sin B} = \dfrac{c}{\sin C}$.\par
\textbf{AAS} means the known side is opposite one of the known angles, so it already has a partner and you can cross straight to the side you want:
\[ \text{wanted side} = \frac{\text{known side} \times \sin(\text{angle opposite the wanted side})}{\sin(\text{angle opposite the known side})} \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $A = 36^\circ$, $B = 71^\circ$, $a = 10$. Find $b$.
\step{The known side $a$ sits opposite the known angle $A$, and the wanted side $b$ sits opposite the known angle $B$ --- two complete pairs, so no third angle is needed.}
\step{$b = \dfrac{10\sin 71^\circ}{\sin 36^\circ} = \dfrac{9.4552}{0.5878}$}
$\Rightarrow\quad b \approx \ans{16.09}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B = 41^\circ$, $C = 63^\circ$, $b = 15$. Find $c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $c \approx \ans{20.37}$}
\end{tryitbox}

\skillheading{Skill 3 --- ASA: find the missing angle first}

\begin{formulabox}
\textbf{ASA} means the known side lies \emph{between} the two known angles, so it has no partner yet. Compute the third angle with the angle sum, and the side gains one.\par
\textbf{Never} pair a side with an angle beside it --- only with the angle across from it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $A = 48^\circ$, $B = 57^\circ$, $c = 20$. Find $a$.
\step{The known side is $c$, but $C$ is not given, so the side has no partner --- the angle sum has to come first.}
\step{$C = 180^\circ - 48^\circ - 57^\circ = 75^\circ$, then $a = \dfrac{20\sin 48^\circ}{\sin 75^\circ}$}
$\Rightarrow\quad a \approx \ans{15.39}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B = 34^\circ$, $C = 76^\circ$, $a = 18$. Find $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $b \approx \ans{10.71}$}
\end{tryitbox}

\skillheading{Skill 4 --- Surveying with a baseline}

\begin{formulabox}
A surveyor measures a \textbf{baseline} between two stations and the angle from each station to a distant point. That is ASA: the baseline sits between the two measured angles.\par
\textbf{Check the answer:} the longest side must face the largest angle. If your computed distance breaks that, a pair was crossed wrongly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Stations $A$ and $B$ are $c = 400$ m apart. The angle at $A$ is $29^\circ$ and at $B$ is $54^\circ$. Find the distance from $B$ to the point $C$.
\step{The distance from $B$ to $C$ is the side opposite $A$, and the baseline lies between the two measured angles --- ASA, so the third angle comes first.}
\step{$C = 180^\circ - 29^\circ - 54^\circ = 97^\circ$, then $a = \dfrac{400\sin 29^\circ}{\sin 97^\circ}$}
$\Rightarrow\quad a \approx \ans{195.38}$ metres.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Stations $A$ and $B$ are $c = 250$ m apart, with $A = 41^\circ$ and $B = 36^\circ$. Find the distance from $A$ to $C$.\\[2pt]
\rotatebox{180}{\footnotesize check: $b \approx \ans{150.81}$ metres}
\end{tryitbox}
