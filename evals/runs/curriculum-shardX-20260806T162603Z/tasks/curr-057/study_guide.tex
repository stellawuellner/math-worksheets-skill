\sstitleblock{Adding Three-Digit Numbers with Regrouping}

\skillheading{Skill 1 --- Adding with one regroup}

\begin{formulabox}
\textbf{Rule:} Add one column at a time, starting at the ones. A column that reaches ten or more is \emph{regrouped}: write the ones digit of that column's total and carry the ten into the next column left.\par
\textbf{Place values:} ones, tens, hundreds. Ten ones make one ten; ten tens make one hundred.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $265 + 118$.
\step{The ones digits are the only pair here that can reach ten, so I expect exactly one carry and I work right to left to catch it before it is needed.}
\step{Ones: five plus eight is thirteen, so write three and carry one ten. Tens: six plus one, plus the carried ten, is eight. Hundreds: two plus one is three.}

$\ans{383}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $372 + 219$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{591}$}
\end{tryitbox}

\skillheading{Skill 2 --- Adding with two regroups}

\begin{formulabox}
\textbf{Rule:} A carried ten joins the next column \emph{before} you decide whether that column regroups too.\par
So a second regroup can appear where the two original digits alone would not have reached ten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $378 + 246$.
\step{Both the ones and the tens look close to ten here, so I add the carry into a column first and only then ask whether that column regroups.}
\step{Ones: eight plus six is fourteen, write four and carry. Tens: seven plus four is eleven, and the carried ten makes twelve, so write two and carry again. Hundreds: three plus two, plus the carry, is six.}

$\ans{624}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $465 + 289$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{754}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimating a sum, then adding}

\begin{formulabox}
\textbf{Rule:} To estimate to the nearest hundred, round \emph{each} addend first, then add the rounded numbers.\par
Look at the tens digit: below five rounds down, five or more rounds up.\par
The estimate is a check, not the answer --- if the exact sum lands far from it, one of the two is wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $523 + 291$ to the nearest hundred, then add exactly.
\step{An estimate is only useful if it is quick, so I round both addends to hundreds before adding anything, and keep the exact addition as a separate second pass.}
\step{Rounding gives five hundred and three hundred, whose sum is the estimate. Adding exactly: ones three plus one is four; tens two plus nine is eleven, write one and carry; hundreds five plus two, plus the carry, is eight.}

estimate $\ans{800}$ \qquad exact $\ans{814}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $618 + 247$ to the nearest hundred, then add exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: estimate $\ans{800}$, exact $\ans{865}$}
\end{tryitbox}

\skillheading{Skill 4 --- Finding a lost carry}

\begin{formulabox}
\textbf{Rule:} The commonest addition error is writing a column's whole total, or only its ones digit, and never carrying the ten.\par
To find it, redo the addition column by column and compare each digit --- the first digit that differs is where the carry went missing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Someone added $329 + 145$ and wrote $464$. What went wrong?
\step{A wrong sum this close to the right one usually means a single lost carry, so instead of guessing I redo the columns and stop at the first digit that disagrees.}
\step{Ones: nine plus five is fourteen, so four is written and one ten is carried --- that carry is what is missing. Tens: two plus four, plus the carried ten, is seven, not the six that was written. Hundreds: three plus one is four.}

correct sum $\ans{474}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Someone added $257 + 186$ and wrote $333$. Find the correct sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{443}$}
\end{tryitbox}
