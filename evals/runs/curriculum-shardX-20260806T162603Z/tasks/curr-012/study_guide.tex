\sstitleblock{Comparing Numerals 1 to 10}

\skillheading{Skill 1 --- Comparing two numbers}

\begin{formulabox}
\textbf{Rule:} $<$ is \emph{is less than}, $>$ is \emph{is greater than}, $=$ is \emph{is equal to}.\par
The wide open end of the symbol always faces the \emph{bigger} number; the point faces the smaller one.\par
\textbf{Counting order:} the later a number is said when you count up, the bigger it is.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$, $>$, or $=$ between $4$ and $7$.
\step{I need to know which numeral stands for more, and counting order tells me that, so I count up and listen for which one I say later.}
\step{Counting up I say $4$ first and $7$ later, so $7$ is the bigger number. The wide end faces $7$, which is on the right.}

$\ans{4 < 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$ between $9$ and $6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9 > 6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The symbol does not always point the same way. Look at the numbers first, then turn the wide end toward the bigger one.
\end{watchoutbox}

\skillheading{Skill 2 --- One more and one less}

\begin{formulabox}
\textbf{Rule:} One more is the next number you say counting up; one less is the number you say just before.\par
A number is always \emph{greater than} the number one less than it, and \emph{less than} the number one more than it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad From $6$: what is one more, and what is one less? Then compare.
\step{One more and one less are neighbours in the counting order, so I count on once and count back once from the same number.}
\step{Counting on from $6$ gives $7$; counting back from $6$ gives $5$. The neighbour ahead is bigger and the neighbour behind is smaller.}

one more: $\ans{7}$ \qquad one less: $\ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From $3$: write one more, then one less.\\[2pt]
\rotatebox{180}{\footnotesize check: one more $\ans{4}$, one less $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Putting three numbers in order}

\begin{formulabox}
\textbf{Rule:} Least to greatest means smallest first. Say the counting numbers and write each of your three numerals down as you reach it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Put $8$, $3$, $5$ in order from least to greatest.
\step{Ordering three numbers is just comparing, done twice, so I walk through the counting order once and take each numeral as it comes up.}
\step{Counting up I reach $3$ first, then $5$, then $8$. That is the order from least to greatest.}

$\ans{3,\ 5,\ 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Put $6$, $10$, $2$ in order from least to greatest.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2,\ 6,\ 10}$}
\end{tryitbox}

\skillheading{Skill 4 --- Checking a comparison}

\begin{formulabox}
\textbf{Rule:} To check a comparison someone else wrote, cover the symbol, decide for yourself which number is bigger, then look: does the wide end face that number?
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $9 < 2$ right? If not, write the correct symbol.
\step{The claim is that $9$ is smaller, so I test it by counting instead of by looking at the symbol.}
\step{Counting up I say $2$ first and $9$ later, so $9$ is the bigger number. The wide end must face $9$, not $2$.}

$\ans{9 > 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $3 > 8$ right? If not, write the correct symbol.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3 < 8}$}
\end{tryitbox}
