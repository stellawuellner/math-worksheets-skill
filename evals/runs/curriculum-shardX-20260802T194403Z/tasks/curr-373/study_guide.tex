\sstitleblock{Solving Radical Equations and Checking Extraneous Roots}

\skillheading{Skill 1 --- Isolate, then square (isolate-and-square)}
\begin{formulabox}\textbf{Method:}\quad
(1) get the radical alone on one side; (2) square both sides;
(3) solve what is left; (4) substitute every candidate into the
\emph{original} equation.\par
Squaring an un-isolated radical keeps a radical
($(\sqrt{x} + 2)^2 = x + 4\sqrt{x} + 4$), which is why step 1 comes first.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Solve $\sqrt{x + 2} = 4$.
\step{The radical is already alone and the right side is positive, so squaring
is safe and undoes the square root.}
\step{$x + 2 = 16$, so $\ans{x = 14}$. Check: $\sqrt{14 + 2} = \sqrt{16} = 4$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $\sqrt{2x - 1} = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 13}$}\end{tryitbox}

\skillheading{Skill 2 --- Reject extraneous roots (check-extraneous)}
\begin{formulabox}\textbf{Why they appear:}\quad squaring is not reversible.
From $a = b$ you may conclude $a^2 = b^2$, but $a^2 = b^2$ only gives
$a = \pm b$, so the squared equation can gain roots.\par
\textbf{Fast filter:} $\sqrt{\;\;} \geq 0$ always. If a candidate makes the
non-radical side negative, it is extraneous --- no arithmetic needed.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Solve $\sqrt{x + 3} = x - 3$.
\step{The right side can be negative here, so squaring may invent a root and
every candidate must be tested in the original equation.}
\step{$x + 3 = x^2 - 6x + 9$ gives $x^2 - 7x + 6 = 0$, so $x = 1$ or $x = 6$.
At $x = 1$ the right side is $-2 < 0$, so it is extraneous; $x = 6$ checks:
$\ans{x = 6}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $\sqrt{x + 1} = x - 1$ and say
which candidate is extraneous.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}\end{tryitbox}

\skillheading{Skill 3 --- Radical and rational-exponent form}
\begin{formulabox}\textbf{Translation:}\quad
$x^{m/n} = \left(\sqrt[n]{x}\right)^m = \sqrt[n]{x^m}$ --- the denominator is
the root, the numerator is the power.\par
To solve $x^{m/n} = k$, raise both sides to the reciprocal power $n/m$.
Take the root \emph{first} when evaluating: the numbers stay small.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Evaluate $27^{2/3}$.
\step{The denominator three tells me to take a cube root, and the numerator
two tells me to square, so I do the root first.}
\step{$27^{2/3} = \left(\sqrt[3]{27}\right)^2 = 3^2 = \ans{9}$}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Solve $x^{3/2} = 27$ (raise both sides
to the power $2/3$).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 9}$}\end{tryitbox}

\skillheading{Skill 4 --- Applied radical models}
\begin{formulabox}\textbf{Method:}\quad substitute the given value, solve the
radical equation, then ask two questions: does the answer check in the model,
and does it make sense for the quantity (a length, a time, and a speed are
never negative)?\par
Keep units with the answer; the model's variables are quantities, not just
letters.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Free fall $t = \sqrt{h/16}$ ($t$ in
seconds, $h$ in feet). Find $h$ when $t = 2$.
\step{The unknown sits under the radical, so I substitute the given time and
solve the radical equation for the height.}
\step{$2 = \sqrt{h/16}$, square: $4 = h/16$, so $\ans{h = 64\ \text{ft}}$.
Check: $\sqrt{64/16} = \sqrt{4} = 2$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Skid model $s = \sqrt{24x}$ ($s$ in
miles per hour, $x$ in feet). Find $x$ when $s = 30$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 37.5\ \text{ft}}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} An answer is not finished at the
solving step. Check every candidate in the ORIGINAL equation, not in the
squared one --- the squared equation accepts extraneous roots by construction.
\end{watchoutbox}
