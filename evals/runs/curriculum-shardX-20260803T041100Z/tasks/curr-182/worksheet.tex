\wstitleblock{Solving and Graphing One-Step Inequalities}{Grade 6--7 \quad Expressions and Equations}{}

\begin{center}
\textit{Adding and subtracting, multiplying and dividing, and flipping the sign
when you multiply or divide by a negative}
\end{center}

\noindent\textbf{Directions.} Solve each inequality, then graph the solution on
the number line printed with the problem.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Undo one operation, applied to \emph{both} sides --- the same move you
would make for an equation.
\item Multiplying or dividing both sides by a \emph{negative} number reverses
the inequality sign. Adding or subtracting never does.
\item Graphing: open circle $\circ$ for $<$ or $>$, solid dot $\bullet$ for
$\le$ or $\ge$, then shade the ray toward every number that works.
\end{itemize}
