% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Proving a number is rational}

\begin{formulabox}
\textbf{Definition:} a number is \emph{rational} when it can be written as one integer
over another. \textbf{The proof is the fraction} --- find it and the argument is finished.
\par\smallskip
\textbf{Three families that are always rational:} integers (put them over one),
terminating decimals (put them over a power of ten), and repeating decimals (use the
subtract-the-copy trick).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Show that $0.24$ is rational.
\step{The decimal stops, so it is already a fraction in disguise --- read the last digit's place value and write it down.}
\step{The last digit sits in the hundredths place, so $0.24 = \dfrac{24}{100}$, and dividing top and bottom by four gives lowest terms.}

$\Rightarrow\quad 0.24 = \ans{\tfrac{6}{25}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Show that $0.45$ is rational.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{9}{20}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Deciding whether a root is rational}

\begin{formulabox}
\textbf{Rule:} the radical sign decides nothing. $\sqrt{N}$ is rational exactly when $N$
is a perfect square (or a fraction of two perfect squares); otherwise it is irrational.
\par\smallskip
$\pi$ is irrational, and so is every non-terminating, non-repeating decimal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $\sqrt{225}$ rational?
\step{Rational-or-not for a root is a perfect-square question, so I test whether the radicand is a square rather than reaching for a calculator.}
\step{Counting up the squares reaches the radicand exactly, so the root is a whole number and a whole number is itself over one.}

$\Rightarrow\quad \sqrt{225} = \ans{15}, \text{ rational}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $\sqrt{196}$ rational? If so, give its value.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a root sign does not make a number irrational, and a decimal that
runs on does not either. $\sqrt{225}$ is rational; $0.\overline{3}$ is rational. Only a
decimal that never ends \emph{and} never repeats is irrational.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Placing an irrational number on a number line}

\begin{formulabox}
\textbf{Method:} trap the irrational root between the two perfect squares on either side
of the radicand. Their roots are the consecutive whole numbers it lies between; then
squeeze with tenths and hundredths to place the point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Place $\sqrt{20}$ on a number line.
\step{No fraction exists for this root, so the useful answer is an interval and a rounded decimal --- trap it between neighbouring perfect squares.}
\step{The radicand sits between the squares of the fourth and fifth whole numbers, so the root is between them; squeezing with hundredths lands just under one half of the way up.}

$\Rightarrow\quad \sqrt{20} \approx \ans{4.47}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Place $\sqrt{30}$ to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.48}$}
\end{tryitbox}
