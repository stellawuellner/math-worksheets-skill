\begin{center}
\small Building Taylor polynomials, reusing known series, and bounding the remainder
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--3 are a warm-up, one on each idea. After that the three are mixed --- decide first whether you should differentiate, substitute into a known series, or bound an error.
  \item Building from scratch: $P_{n}(x) = \sum_{k=0}^{n}\dfrac{f^{(k)}(a)}{k!}(x-a)^{k}$. Reusing a known series is almost always faster when one exists.
  \item Two remainder bounds. The Lagrange form $|R_{n}| \le \dfrac{M|x-a|^{n+1}}{(n+1)!}$ always applies, with $M$ a bound on the next derivative over the whole interval. The alternating bound --- error at most the first omitted term --- applies only when the series alternates.
  \item Round decimals to five places unless told otherwise.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{Let $f(x) = e^{2x}$, with third-degree Maclaurin polynomial $P_{3}(x) = 1 + 2x + 2x^{2} + \frac{4}{3}x^{3}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $f'''(x)$, the derivative that produces the cubic coefficient.
  \item[(b)] Use $P_{3}$ to estimate $f(0.1)$.
\end{itemize}}

\problem[5.5cm]{Using the Maclaurin series for $\cos x$, the series for $x\cos x$ begins $x - \dfrac{x^{3}}{2} + \dfrac{x^{5}}{24}$. Use those three terms to estimate the value of $x\cos x$ at $x = 0.4$.}

\problem[6.5cm]{The third-degree Maclaurin polynomial for $e^{x}$ is used to estimate $e^{x}$ on the interval $0 \le x \le 1$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Compute the Lagrange error bound at $x = 1$, taking $M = e$.
  \item[(b)] Explain why $M = e$ is the right bound to use on this interval.
\end{itemize}}

\problem[6.5cm]{Let $f(x) = \ln x$, expanded about $a = 1$, so $P_{3}(x) = (x-1) - \dfrac{(x-1)^{2}}{2} + \dfrac{(x-1)^{3}}{3}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $f'''(x)$.
  \item[(b)] Use $P_{3}$ to estimate $\ln(1.2)$.
\end{itemize}}

\problem[6cm]{Substituting into the geometric series $\dfrac{1}{1-u} = 1 + u + u^{2} + \cdots$ with $u = -3x^{2}$ gives $\dfrac{1}{1+3x^{2}} = 1 - 3x^{2} + 9x^{4} - \cdots$. Use those three terms to estimate the value at $x = 0.2$.}

\problem[6.5cm]{The Maclaurin series for $\cos x$ is used through the $x^{4}$ term to estimate $\cos(0.5)$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Compute the error bound the alternating series remainder guarantees.
  \item[(b)] Explain what you had to check about the series before using that bound.
\end{itemize}}

\problem[6.5cm]{Let $f(x) = \sqrt{x}$, expanded about $a = 4$, so $P_{2}(x) = 2 + \dfrac{x-4}{4} - \dfrac{(x-4)^{2}}{64}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $f''(x)$, the derivative that produces the quadratic coefficient.
  \item[(b)] Use $P_{2}$ to estimate $\sqrt{4.4}$.
\end{itemize}}

\problem[6.5cm]{Substituting $x = -1$ into the Maclaurin series for $e^{x}$ gives $\displaystyle\sum_{n=0}^{\infty}\frac{(-1)^{n}}{n!}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Write the exact sum of that series, and a decimal to four places.
  \item[(b)] If the first four terms ($n = 0$ through $n = 3$) are used as an estimate, give the exact error bound the alternating series remainder guarantees.
\end{itemize}}

\problem[6.5cm]{The third-degree Maclaurin polynomial for $f(x) = \ln(1+x)$ is used to estimate $f(x)$ on $0 \le x \le 0.5$. Its fourth derivative is $f^{(4)}(x) = -\dfrac{6}{(1+x)^{4}}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Compute the Lagrange error bound at $x = 0.5$.
  \item[(b)] Explain how you chose $M$, and why the largest value of $|f^{(4)}|$ occurs where it does.
\end{itemize}}

\problem[7cm]{Let $f(x) = x e^{x}$, with third-degree Maclaurin polynomial $P_{3}(x) = x + x^{2} + \dfrac{x^{3}}{2}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $f'''(x)$, and check that it gives the cubic coefficient shown.
  \item[(b)] Use $P_{3}$ to estimate $f(0.3)$.
  \item[(c)] Which remainder bound --- the Lagrange form or the alternating-series form --- would you use to measure the error in part (b), and why is the other one unavailable here?
\end{itemize}}
