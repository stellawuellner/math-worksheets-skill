\small\textbf{Interior} angles ($3, 4, 5, 6$) lie between the parallel lines;
\textbf{exterior} angles ($1, 2, 7, 8$) lie outside them.\normalsize

\skillheading{Skill 1 --- Naming the pair before you compute}

\begin{formulabox}\textbf{Two questions name every pair:} inside or outside the
parallel lines, and same side of $t$ or opposite sides?
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Corresponding} --- same corner at the two intersections
        ($1$ and $5$). \emph{Congruent.}
  \item \textbf{Alternate interior} --- inside, opposite sides ($4$ and $6$).
        \emph{Congruent.}
  \item \textbf{Alternate exterior} --- outside, opposite sides ($2$ and $8$).
        \emph{Congruent.}
  \item \textbf{Same-side interior} --- inside, same side ($4$ and $5$).
        \emph{Supplementary.}
\end{itemize}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $m\angle 2 = 130^\circ$. Find
$m\angle 6$.
\step{Angles $2$ and $6$ sit in the same corner at their own intersections, so
they are corresponding --- and corresponding angles on parallel lines are
congruent, not supplementary.}
\step{$m\angle 6 = m\angle 2 = \ans{130^\circ}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $m\angle 3 = 47^\circ$. Find $m\angle 5$.
\ \rotatebox{180}{\footnotesize check: $\ans{47^\circ}$}\end{tryitbox}

\skillheading{Skill 2 --- Finding a missing measure}

\begin{formulabox}\textbf{Rule:}\ Congruent pairs (corresponding, alternate
interior, alternate exterior) copy the measure. Supplementary pairs (linear
pairs, same-side interior) subtract it from $180^\circ$. Chain the two: name a
pair, use it, then name the next.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $m\angle 1 = 118^\circ$. Find
$m\angle 4$.
\step{Angles $1$ and $4$ are adjacent at the same intersection --- a linear
pair --- so this is a subtraction, not a copy.}
\step{$m\angle 4 = 180 - 118 = \ans{62^\circ}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $m\angle 4 = 84^\circ$. Find $m\angle 5$
(same-side interior).
\ \rotatebox{180}{\footnotesize check: $\ans{96^\circ}$}\end{tryitbox}

\skillheading{Skill 3 --- Solving for $x$ from an angle pair}

\begin{formulabox}\textbf{Rule:}\ The pair name chooses the equation:
congruent pair $\Rightarrow$ set the two expressions \textbf{equal};
supplementary pair $\Rightarrow$ set their \textbf{sum} to $180$. Solve, then
substitute back to get the measure the question actually asked for.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Alternate interior angles measure
$(5x - 12)^\circ$ and $(3x + 20)^\circ$. Find $x$.
\step{Alternate interior angles are congruent, so the two expressions are equal
--- no $180$ appears anywhere in this setup.}
\step{$5x - 12 = 3x + 20 \Rightarrow 2x = 32 \Rightarrow \ans{x = 16}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Same-side interior angles measure
$(4x + 10)^\circ$ and $(6x)^\circ$. Find $x$.
\ \rotatebox{180}{\footnotesize check: $\ans{x = 17}$}\end{tryitbox}
