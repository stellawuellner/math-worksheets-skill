\wstitleblock{See It Fast! Quantities to Five}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:} Some dot pictures you can know \emph{just by
looking} --- no counting one by one!
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Look at the whole pattern first. Try to see the number all at once.
  \item If a pattern is tricky, look for small parts you already know.
\end{itemize}
