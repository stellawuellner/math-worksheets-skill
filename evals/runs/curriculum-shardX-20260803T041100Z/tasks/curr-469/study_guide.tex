\sstitleblock{Related Rates and Optimization: Diagnosing the Setup}

\skillheading{Skill 1 --- Differentiate the Relation}

\begin{formulabox}
\textbf{Order of operations for a setup:} relate $\rightarrow$ eliminate $\rightarrow$ differentiate $\rightarrow$ substitute.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Use the geometry to remove every variable but one \emph{before} differentiating.
  \item A quantity with a non-zero rate must stay a symbol; only true constants may be written as numbers.
  \item Similar triangles are the usual eliminator for cones: $r/h$ is fixed by the tank's shape.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A cylinder always has $h = 2r$. Write $V$ in terms of $r$ and differentiate.
\step{Two variables cannot both survive into one equation with a single given rate, so eliminate $h$ using the fixed shape ratio first.}
\step{$V = \pi r^2 h = \pi r^2(2r) = 2\pi r^3$, giving $\dfrac{dV}{dr} = \ans{6\pi r^2}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A cone always has $h = 4r$, so $V = \tfrac{4}{3}\pi r^3$. Find $dV/dr$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4\pi r^2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Set Up a Related-Rates Equation}

\begin{formulabox}
\textbf{Chain rule on every varying symbol:} $\dfrac{dV}{dt} = \dfrac{dV}{dr}\cdot\dfrac{dr}{dt}$.\par
Sphere: $V = \tfrac{4}{3}\pi r^3 \Rightarrow \dfrac{dV}{dt} = 4\pi r^2\dfrac{dr}{dt}$. \quad
Pythagoras: $x^2 + y^2 = c^2 \Rightarrow \dfrac{dy}{dt} = -\dfrac{x}{y}\dfrac{dx}{dt}$.\par
\textbf{Solve for the wanted rate symbolically, then substitute} --- that single habit prevents the inverted ratio.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A sphere's volume grows at $50$ per second. Find $dr/dt$ when $r = 4$.
\step{The given rate is volume and the wanted rate is radius, so differentiate the volume relation and solve for $dr/dt$ before touching a number.}
\step{$\dfrac{dr}{dt} = \dfrac{dV/dt}{4\pi r^2} = \dfrac{50}{4\pi(4)^2} = \ans{0.249}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A $13$-ft ladder's base slides out at $3$ ft/s. Find $dy/dt$ when the base is $5$ ft out.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1.25}$}
\end{tryitbox}

\skillheading{Skill 3 --- Set Up an Optimization Problem}

\begin{formulabox}
\textbf{Write the constraint from the picture, not from memory.}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Objective $=$ the quantity being maximised or minimised; constraint $=$ the resource that is fixed.
  \item A pen against a wall fences three sides: $x + 2y = P$, not $2x + 2y = P$.
  \item Use the constraint to make the objective a function of one variable, then set the derivative to zero.
  \item State the physical domain with the setup; it is what lets you reject a critical value later.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A closed rectangle has perimeter $40$. Maximise its area.
\step{Perimeter is the fixed resource and area is the objective, so use the constraint to write area with one variable.}
\step{$2x + 2y = 40 \Rightarrow y = 20 - x$, so $A = x(20-x)$ and $A' = 20 - 2x = 0$ gives $\ans{x = 10}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $60$ ft of fence, three sides against a wall ($x + 2y = 60$). Which $x$ maximises the area?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 30}$}
\end{tryitbox}
