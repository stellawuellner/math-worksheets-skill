\sstitleblock{Solving AAS and ASA Triangles with the Law of Sines}

\skillheading{Skill 1 --- Naming the Case and Finding the Third Angle}

\begin{formulabox}
\textbf{Angle sum:} $A + B + C = 180^\circ$. \quad
\textbf{Naming rule:} side $a$ is opposite angle $A$, and so on.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item \textbf{AAS}: the given side is opposite one of the given angles $\Rightarrow$ a complete side--angle pair already exists.
  \item \textbf{ASA}: the given side lies \emph{between} the given angles $\Rightarrow$ no pair yet, so find the third angle first.
  \item Either way, two angles force the third, so the triangle's shape is fixed and one side sets its size.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $A = 37^\circ$, $B = 58^\circ$, $a = 10$. Find $C$ and name the case.
\step{Two angles are known, so the third is forced by the angle sum --- no trigonometry is needed for this step.}
\step{$C = 180^\circ - 37^\circ - 58^\circ = \ans{85^\circ}$. Side $a$ is opposite the given $A$, so this is AAS.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $A = 64^\circ$, $C = 51^\circ$, $b = 10$. Find $B$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{B = 65^\circ}$}
\end{tryitbox}

\skillheading{Skill 2 --- AAS: Use the Complete Pair Straight Away}

\begin{formulabox}
\textbf{Law of sines:}\quad
$\dfrac{a}{\sin A} = \dfrac{b}{\sin B} = \dfrac{c}{\sin C}$\par
To find an unknown \textbf{side}, put the unknown on top:
\[ \text{unknown} = \frac{(\text{known side})\, \sin(\text{angle opposite the unknown})}
                        {\sin(\text{angle opposite the known side})} \]
Sanity check: the bigger angle must always face the longer side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $A = 41^\circ$, $B = 79^\circ$, $a = 14$. Find $b$.
\step{Side $a$ is opposite the given angle $A$, so the pair $a/\sin A$ is complete and the law of sines can be used at once.}
\step{$b = \dfrac{14 \sin 79^\circ}{\sin 41^\circ} = \dfrac{14(0.981627)}{0.656059} = 20.9475\ldots \Rightarrow \ans{b \approx 20.95}$, and $B > A$ so $b > a$ as expected.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $A = 36^\circ$, $B = 68^\circ$, $a = 20$. Find $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{b \approx 31.55}$}
\end{tryitbox}

\skillheading{Skill 3 --- ASA: Third Angle First, Then the Law of Sines}

\begin{formulabox}
\textbf{Why the extra step:} in ASA the known side sits \emph{between} the known angles,
so the angle opposite it is the one you do not have. Build the pair first:
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item Step 1: third angle from $180^\circ$ minus the two given angles.
  \item Step 2: the known side is now paired with that new angle --- apply the law of sines.
  \item Keep full calculator precision and round only the final side.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $A = 54^\circ$, $B = 39^\circ$, $c = 25$. Find $a$.
\step{Side $c$ lies between the two given angles, so no pair is complete --- the angle opposite $c$ must be found before the law of sines can be used.}
\step{$C = 180^\circ - 54^\circ - 39^\circ = 87^\circ$, then $a = \dfrac{25 \sin 54^\circ}{\sin 87^\circ} = 20.2532\ldots \Rightarrow \ans{a \approx 20.25}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $A = 47^\circ$, $B = 62^\circ$, $c = 30$. Find $a$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{a \approx 23.20}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Match each side with the angle \emph{opposite} it, never with the
nearest one --- an upside-down ratio is the single most common error here. And in ASA,
skipping the third angle leaves you with no complete pair at all.
\end{watchoutbox}
