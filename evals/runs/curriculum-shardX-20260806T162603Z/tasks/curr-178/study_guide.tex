\sstitleblock{Writing Equations from Word Situations}

% =================== SKILL 1 ===================
\skillheading{Situations that Add or Take Away}

\begin{formulabox}
\textbf{Rule:} give the unknown starting amount a letter, then write what
happened to it. ``More were added'' becomes $x + a$; ``some were used, lost or
drained'' becomes $x - a$. The number after the word \emph{was}, \emph{left} or
\emph{now} goes on the other side of the equals sign.\par
\textbf{Signal words:} more than, added, gained, fewer, left, remaining, fell by.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A jar held $j$ sweets. After 19 were eaten, 27 were
left. Find $j$.
\step{The unknown is the starting number and sweets were taken away, so the
equation subtracts from $j$ and sets the result equal to what is left:
$j - 19 = 27$.}
\step{Undo the subtraction by adding 19 to both sides: $j = 27 + 19$, so
$j = $ $\ans{46}$ sweets.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A shelf held $s$ books. After 23 more were added there
were 51. Write the equation and find $s$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Situations that Share or Scale}

\begin{formulabox}
\textbf{Rule:} ``shared equally between $n$'' and ``cut into $n$ equal pieces''
divide the unknown: $\frac{x}{n}$. ``$n$ times as many'' and ``$n$ in each of
several groups'' multiply it: $nx$. Undo a multiplication by dividing, and a
division by multiplying.\par
\textbf{Signal words:} each, per, shared equally, times as many, equal pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ribbon $r$ centimetres long is cut into 7 equal pieces
of 12 centimetres. Find $r$.
\step{Cutting the whole ribbon into 7 equal pieces divides its length, and one
piece is 12, so $\frac{r}{7} = 12$.}
\step{Undo the division by multiplying both sides by 7:
$r = 12 \times 7$, so $r = $ $\ans{84}$ centimetres.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Five identical crates weigh 65 kilograms altogether.
Write an equation for the mass $m$ of one crate and find it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Two-Step Situations: a Rate plus a Fixed Charge}

\begin{formulabox}
\textbf{Rule:} when something is charged \emph{once} and something else repeats,
the equation has two terms: (repeating rate) $\times$ (unknown count) $+$ (one
time amount) $=$ total, or $ax + b = c$. The one-time amount is never multiplied
by the count --- that is the mistake this shape is built to catch.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A gym charges 20 dollars to join plus 6 dollars a week.
A member has paid 62 dollars. How many weeks $w$?
\step{The 6 dollars repeats every week so it multiplies $w$; the 20 dollars is
paid once so it is added on: $6w + 20 = 62$.}
\step{Subtract 20 from both sides to get $6w = 42$, then divide by 6:
$w = $ $\ans{7}$ weeks.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A printer costs 9 dollars to set up plus 4 dollars for
each poster. The bill was 41 dollars. Write an equation for the number $n$ of
posters and find it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} check the equation against the story before you solve
it. Substituting your answer back in should make both sides equal --- if it does
not, the equation is wrong, not the arithmetic.
\end{watchoutbox}
