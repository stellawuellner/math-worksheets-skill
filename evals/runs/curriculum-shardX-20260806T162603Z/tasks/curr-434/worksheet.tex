\wstitleblock{Diagnosing Quadrant and Parameter-Orientation Errors}{Precalculus}{}

\begin{center}
\small Find the mistake, fix the value, and say what went wrong in each conversion below.
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Every problem below contains one student's work with exactly one thing wrong in it.
  \item Part (a) asks for the corrected value. Part (b) asks you to name the error in a sentence or two.
  \item Angles are in radians. Give exact values unless a decimal is asked for.
\end{itemize}

\vspace{0.15cm}

angle they enter $\arctan\!\left(\dfrac{3}{-3}\right)$ on a calculator and write
$\theta \approx -0.79$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Give the correct exact value of $\theta$ in $[0, 2\pi)$. \quad $\theta = $~\ansblank
  \item[(b)] Name the error in the student's method.
\end{itemize}
