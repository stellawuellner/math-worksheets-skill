\sstitleblock{Adding Two-Digit Numbers with Place-Value Strategies}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Break apart into tens and ones}

\begin{formulabox}
\textbf{Rule:} Split both numbers into tens and ones. Add the tens, add the
ones, then put the two answers together. If the ones make ten or more, that
extra ten joins the tens.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $38 + 47$.
\step{Break both numbers apart into tens and ones, and add each kind separately.}
\step{Tens: $30 + 40 = \ans{70}$. \quad Ones: $8 + 7 = \ans{15}$ --- one extra ten and five ones.}
$\Rightarrow\quad \ans{85}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $29 + 56$ by breaking both apart.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{85}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When the ones make more than nine, do not write them
side by side with the tens --- trade ten ones for one more ten first.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Jump along the number line}

\begin{formulabox}
\textbf{Rule:} Start at the first number. Jump the \emph{tens} of the second
number in one big hop, then jump its \emph{ones}. Where you land is the sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $53 + 26$ with number-line jumps.
\step{Start at 53 and jump the tens of 26 first: one hop of 20.}
\step{$53 + 20 = \ans{73}$, then hop the ones of 26.}
$\Rightarrow\quad 73 + 6 = \ans{79}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $64 + 28$ with two jumps: tens first, then ones.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{92}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Friendly numbers}

\begin{formulabox}
\textbf{Rule:} If a number is close to a ten, add the friendly ten instead,
then fix the small difference at the end: add a bit less --- take the extra
away; add a bit more --- put the missing bit back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Add $49 + 35$ with a friendly number.
\step{The number 49 is one below the friendly number 50, so add 50 instead and take one away at the end.}
\step{$50 + 35 = \ans{85}$, then take away the extra one.}
$\Rightarrow\quad \ans{84}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Add $38 + 45$ using a friendly number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{83}$}
\end{tryitbox}
