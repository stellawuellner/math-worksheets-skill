\sstitleblock{Comparing Unlike Fractions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Comparing to the benchmark one half}

\begin{formulabox}
\textbf{Benchmark test:} a fraction is more than $\tfrac{1}{2}$ when its
numerator is more than half its denominator, and less than $\tfrac{1}{2}$
when the numerator is less than half the denominator. If one fraction is
above the benchmark and the other is below, you are done --- no common
denominator needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $\tfrac{3}{7}$ and $\tfrac{1}{2}$.
\step{Check the numerator against half the denominator instead of rewriting anything.}
\step{Half of 7 is $3.5$, and $3 < 3.5$, so $\tfrac{3}{7}\;\ans{<}\;\tfrac{1}{2}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compare: $\tfrac{5}{9} \;\square\; \tfrac{1}{2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Comparing with a common denominator}

\begin{formulabox}
\textbf{Plan:} rewrite both fractions with the same denominator (a number
both denominators divide into), then compare the numerators --- same-size
pieces make the comparison fair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $\tfrac{3}{5}$ and $\tfrac{2}{3}$.
\step{Both 5 and 3 divide into 15, so rewrite both fractions in fifteenths.}
\step{$\tfrac{3}{5} = \tfrac{\ans{9}}{15}$ and $\tfrac{2}{3} = \tfrac{\ans{10}}{15}$, and 9 pieces is fewer than 10, so $\tfrac{3}{5}\;\ans{<}\;\tfrac{2}{3}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite in twelfths, then compare:
$\tfrac{3}{4} \;\square\; \tfrac{5}{6}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Making equivalent fractions}

\begin{formulabox}
\textbf{Rule:} multiplying the numerator and denominator by the same number
makes an equivalent fraction --- the same amount cut into smaller pieces.
This is the move that builds every common denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fill in: $\tfrac{3}{4} = \tfrac{\square}{8}$
\step{Ask what number times 4 gives 8, then multiply the top by the same number.}
\step{$4 \times 2 = 8$, so the numerator is $3 \times 2 = \ans{6}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Fill in: $\tfrac{4}{5} = \tfrac{\square}{15}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} you cannot compare unlike fractions by looking at
numerators alone or denominators alone. Always compare same-size pieces (a
common denominator) or use a benchmark.
\end{watchoutbox}
