\wstitleblock{Separable Differential Equations with Initial Conditions}{AP Calculus AB/BC \quad Differential Equations}{}

\begin{center}
\textit{Separating and integrating, applying the initial condition to pin down
the constant, and using the particular solution}
\end{center}

\noindent\textbf{Directions.} Give the particular solution --- an equation with
no arbitrary constant left in it --- unless a problem asks for something else.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Separate:} get every $y$ (with $dy$) on one side and every $x$
(with $dx$) on the other.
\item \textbf{Integrate both sides} and put a single $+C$ on one side only.
\item \textbf{Apply the initial condition immediately}, before solving for $y$
if that is easier --- it usually is.
\item Round decimal answers to two places and keep the units.
\end{itemize}

\problem[7cm]{Solve $\dfrac{dy}{dx} = 6x^2$ with $y(0) = 1$. Give the
particular solution, then use it to find $y(2)$.}

\problem[7cm]{Solve $\dfrac{dy}{dx} = 3x^2 - 4$ with $y(2) = 5$. Find the value
of $C$ explicitly before writing the particular solution.}

\problem[7cm]{Solve $\dfrac{dy}{dx} = xy$ with $y(0) = 2$. Separate first,
integrate both sides, then exponentiate to isolate $y$. Use your particular
solution to find $y(2)$, rounded to two decimal places.}

\problem[7cm]{Solve $\dfrac{dy}{dx} = \dfrac{2x+1}{y}$ with $y(0) = 3$. Find
$C$ from the initial condition, then give $y$ as an explicit function of $x$
and compute $y(2)$ to two decimal places.}

\problem[7cm]{Solve $\dfrac{dy}{dx} = y^2$ with $y(1) = 1$. Give the particular
solution and evaluate it at $x = 1.5$. For which values of $x$ is your solution
actually defined?}

\problem[6cm]{A $80$ milligram sample of a radioactive isotope decays at a rate
proportional to the amount present, so $\dfrac{dA}{dt} = -kA$, and its
half-life is $10$ days. Find $k$ from the half-life, then find how much of the
sample remains after $25$ days, in milligrams, to two decimal places.
\answerline{mg}}

\problem[7cm]{Solve $\dfrac{dy}{dx} = y\cos x$ with $y(0) = 4$. Give the
particular solution, then evaluate it at $x = 1$ radian, to two decimal
places.}

\problem[7cm]{Solve $\dfrac{dy}{dx} = \dfrac{3x^2}{2y}$ with $y(1) = 2$.
Separate, integrate, apply the condition, and give $y$ explicitly. Then find
$y(3)$ to two decimal places, taking the branch the initial condition selects.}

\problem[6cm]{A population grows so that $\dfrac{dP}{dt} = 0.08P$, with
$P(0) = 500$ and $t$ measured in years. Solve the equation, then find how long
it takes the population to reach $1500$, to two decimal places.
\answerline{years}}

\problem[8cm]{Consider the logistic equation
$\dfrac{dy}{dx} = y(3 - y)$.
