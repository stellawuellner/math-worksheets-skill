\wstitleblock{Comparing Groups with Two-Way Tables}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Reading cells and totals, turning counts into row fractions, and comparing two groups}
\end{center}

\vspace{0.05cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Each cell of a two-way table counts the people in \textit{both} categories at once. The last row and last column hold the totals.
  \item To compare two groups fairly, turn each count into a fraction or a percentage \textit{of its own row total}. Raw counts only compare fairly when the row totals are the same.
  \item Give fractions in simplest form. Where a question asks for a percent, write the number without the percent sign.
\end{itemize}
