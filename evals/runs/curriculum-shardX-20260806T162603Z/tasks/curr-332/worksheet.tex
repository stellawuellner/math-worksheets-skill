\wstitleblock{Polygon Interior and Exterior Angle Sums}{High School Geometry}{}

\begin{center}
\small\textit{Interior angle sums, missing interior angles, exterior angles, and finding the number of sides}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Write the equation you set up before you solve it.
  \item Every answer on this sheet is a whole number --- no rounding is needed.
  \item ``Regular'' means all sides and all angles are equal.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.6cm]{Find the sum of the measures of the interior angles of a hexagon (a polygon with $6$ sides).}

\problem[4.6cm]{Find the sum of the measures of the interior angles of a polygon with $11$ sides.}

\problem[4.6cm]{A regular polygon has $9$ sides. Find the measure of each exterior angle.}

\problem[4.6cm]{Three of the angles of a quadrilateral measure $112^\circ$, $87^\circ$ and $63^\circ$. Find the measure of the fourth angle.}

\problem[4.6cm]{Each exterior angle of a regular polygon measures $24^\circ$. How many sides does the polygon have?}

\problem[4.6cm]{Four of the angles of a pentagon measure $102^\circ$, $118^\circ$, $96^\circ$ and $131^\circ$. Find the measure of the fifth angle.}

\problem{Each interior angle of a regular polygon measures $156^\circ$.
\par\vspace{0.15cm}
(a) Find the measure of each exterior angle. \quad \ansblank
\par\vspace*{2.7cm}
(b) Find the number of sides. \quad \ansblank
\par\vspace*{2.7cm}}

\problem[4.6cm]{The interior angles of a polygon add up to $1800^\circ$. How many sides does the polygon have?}

\problem{Marcus is finding the sum of the interior angles of a pentagon. He writes
\[ \text{sum} = 5 \cdot 180^\circ = 900^\circ. \]
(a) Find the correct sum of the interior angles of a pentagon. \quad \ansblank
\par\vspace*{2.6cm}
(b) Explain what Marcus did wrong, and say what he should have used instead.
\par\vspace*{3.2cm}}

\problem{In a regular polygon each interior angle is $5$ times as large as each exterior angle.
\par\vspace{0.15cm}
(a) Find the measure of each exterior angle. \quad \ansblank
\par\vspace*{3.0cm}
(b) Find the number of sides of the polygon. \quad \ansblank
\par\vspace*{3.0cm}}
