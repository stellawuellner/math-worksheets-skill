\sstitleblock{Ten-Frames: How Many Do You See?}

\skillheading{1. Say how many the frame shows}

\begin{formulabox}
\textbf{The frame.} Two rows of five boxes. A \textbf{full row is 5}. A
\textbf{full frame is 10}.\\[3pt]
\textbf{The shortcut.} When the top row is full, say ``five'' and then count
on for the dots below --- six, seven, eight. You never have to start again at
one.\\[3pt]
\textbf{First look, then count:} is the top row really full?
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many dots? \quad \tenframe{5}{2}
\step{The top row is full, so it is worth five without counting it one dot at
a time.}
\step{Count on for the bottom row: ``six, seven''. That is $5 + 2$.}

The last number said is the answer: $\ans{7}$ dots.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many dots? \quad \tenframe{5}{3}\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{2. How many more make ten?}

\begin{formulabox}
\textbf{The idea.} The empty boxes show what is missing. Count the dots, then
count the empty boxes --- together they always make \textbf{10}.\\[3pt]
So ``how many more?'' can be answered two ways: count the empty boxes, or
think ``what goes with this number to make ten?''
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many more dots would fill this frame?
\quad \tenframe{5}{1}
\step{The question is about the empty boxes, so find how many dots are there
first: $5 + 1 = 6$.}
\step{Ten boxes in all, six of them filled, so the empty ones are what is
left --- and they can simply be counted: four.}

$\Rightarrow\quad \ans{4}$ more dots.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many more dots would fill this frame?
\quad \tenframe{3}{0}\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{3. Which frame shows more?}

\begin{formulabox}
\textbf{The rule.} Finish \emph{both} counts first, then compare the two
numbers.\\[3pt]
$>$ means \emph{is greater than} \qquad $<$ means \emph{is less than} \qquad
$=$ means \emph{is the same as}\\[3pt]
The open end of the sign always faces the bigger number. Two frames can look
different and still be equal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $>$ or $<$: \quad \tenframe{5}{1} \; \tenframe{5}{4}
\step{Nothing can be compared until each frame has one number, so count both
frames before choosing a sign.}
\step{Left frame: $5 + 1 = 6$. Right frame: $5 + 4 = 9$. Six is less than
nine.}

$\Rightarrow\quad \ans{<}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $>$, $<$ or $=$: \quad
\tenframe{4}{4} \; \tenframe{5}{2}\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the five-shortcut only works when the top row is
\emph{full}. If the top row has gaps, count every dot. And read the question
twice --- ``how many dots?'' and ``how many empty boxes?'' have different
answers for the same picture.
\end{watchoutbox}
