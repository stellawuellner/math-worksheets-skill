\wstitleblock{Composing Functions from Symbolic Rules}{Algebra 2}{}

\begin{center}
\textit{Composing at a number, composing into a new rule, comparing the two orders, and solving an equation built from a composition}
\end{center}

\noindent\textbf{The model: two machines in a row}

\noindent A composition sends the output of one machine straight into the next.
For $f(x) = 2x+5$ and $g(x) = x^2-3$, the chain
$x \rightarrow \boxed{\,g\,} \rightarrow \boxed{\,f\,} \rightarrow$ output is
what $(f \circ g)(x) = f(g(x))$ means. Follow one column at a time:

\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\textbf{input $x$} & \textbf{after $g$:\; $x^2-3$} & \textbf{after $f$:\; $2(\,\cdot\,)+5$} \\
\hline
$0$ & $-3$ & $-1$ \\
\hline
$1$ & $-2$ & $1$ \\
\hline
$3$ & $6$ & $17$ \\
\hline
\end{tabular}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=3pt]
  \item Read $(f \circ g)(x)$ as ``$g$ first, then $f$'': the function written closest to $x$ runs first.
  \item To compose \emph{symbolically}, substitute the whole rule for $g$ wherever $f$'s formula has an $x$, then simplify.
  \item Keep the substituted rule in brackets while you simplify --- that is where most sign and squaring errors come from.
\end{itemize}
