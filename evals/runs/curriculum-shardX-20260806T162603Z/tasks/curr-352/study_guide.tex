% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Adding and Subtracting}

\begin{formulabox}
\textbf{Rule:} treat $i$ like a variable and collect like parts.\\[3pt]
$(a + bi) + (c + di) = (a + c) + (b + d)i$ \qquad
$(a + bi) - (c + di) = (a - c) + (b - d)i$\\[3pt]
A subtraction sign applies to \textit{both} parts of the number behind it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(6 + 2i) - (9 - 5i)$
\step{Nothing here needs multiplying --- both numbers are already in $a + bi$ form, so the job is only to distribute the minus sign and collect like parts.}
\step{$6 + 2i - 9 + 5i$, so the real parts give $6 - 9$ and the imaginary parts give $2 + 5$.}

$\Rightarrow\quad \ans{-3 + 7i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(4 - 9i) + (-6 + 3i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-2 - 6i}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Multiplying Two Complex Numbers}

\begin{formulabox}
\textbf{Rule:} multiply out all four products, then replace $i^2$ with $-1$
and collect.\\[3pt]
$(a + bi)(c + di) = (ac - bd) + (ad + bc)i$\\[3pt]
The $-bd$ in the real part is exactly the $i^2 = -1$ step written once and for all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(3 + 5i)(2 - 4i)$
\step{Two binomials multiplied together --- use the same four products as ordinary algebra, then convert the $i^2$ term.}
\step{$6 - 12i + 10i - 20i^2$, and $-20i^2 = +20$, so the real parts give $6 + 20$ and the imaginary parts give $-12 + 10$.}

$\Rightarrow\quad \ans{26 - 2i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(4 + i)(3 - 2i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14 - 5i}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Conjugate Pairs and Real Products}

\begin{formulabox}
\textbf{Rule:} $a + bi$ and $a - bi$ are \textit{conjugates}. Their product is
always real:\\[3pt]
$(a + bi)(a - bi) = a^2 + b^2$\\[3pt]
The two middle terms are opposites, so they cancel; the $i^2$ term flips sign
and \textit{adds} to the square of the real part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(7 + 3i)(7 - 3i)$
\step{The factors differ only in the sign of the imaginary part, so this is a conjugate pair --- expect the $i$ terms to cancel and the answer to be real.}
\step{The middle terms are $+21i$ and $-21i$, which add to zero, and what is left is the sum of the squares of the parts.}

$\Rightarrow\quad \ans{58}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(4 + 5i)(4 - 5i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{41}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Squaring a Complex Number}

\begin{formulabox}
\textbf{Rule:} a square is still a product of two binomials.\\[3pt]
$(a + bi)^2 = a^2 - b^2 + 2abi$ \qquad $(a - bi)^2 = a^2 - b^2 - 2abi$
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $(5 + 2i)^2$
\step{Squaring a binomial is the same product twice, so the middle term must appear --- write it out rather than squaring each piece.}
\step{$25 + 10i + 10i + 4i^2$, and $4i^2 = -4$, so the real parts give $25 - 4$ and the imaginary parts give $10 + 10$.}

$\Rightarrow\quad \ans{21 + 20i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $(6 - i)^2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35 - 12i}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(a + bi)^2$ is \textit{not} $a^2 + b^2i^2$. Losing the
middle term $2abi$ is the most common error on this page. And $i^2 = -1$, never
$+1$ --- if your real part came out too small, check that sign first.
\end{watchoutbox}
