\wstitleblock{Teen Numbers: Tens and Ones Fix-It}{Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Write your answers on the blanks.
  \item Some problems show a mistake. Fix the mistake, then tell how you know.
\end{itemize}

\problem[3cm]{1 ten and 4 ones make the number \ansblank.}

\problem[3cm]{The number 17 has \ansblank\ ten and \ansblank\ ones.}

\problem[4cm]{Ben used 3 tens and 1 one to show the number 13.
That is a mistake! Fix it:\par
13 has \ansblank\ ten and \ansblank\ ones.\par
Tell how you know.}

\problem[3cm]{Write the number that has 9 ones and 1 ten. \ansblank}

\problem[3cm]{Ana wrote 91. Dan wrote 19. The digits look the same, but the
numbers are not! Write $>$, $<$, or $=$ in the blank:\par
\begin{center} 91 \ansblank\ 19 \end{center}}

\problem[4cm]{Mia has 1 ten and 6 ones. She writes the number 61.
Find her mistake and fix it:\par
The number is really \ansblank.\par
Tell how you know.}
