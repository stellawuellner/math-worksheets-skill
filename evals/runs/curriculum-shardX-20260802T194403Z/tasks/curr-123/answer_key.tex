\problem{\textbf{Ana $\frac{2}{5}$ of a bottle vs.\ Ben $\frac{3}{4}$ of the same
bottle.} Half of five fifths is $\frac{2.5}{5}$, so $\frac{2}{5}$ is
\emph{less} than $\frac{1}{2}$. Half of four fourths is $\frac{2}{4}$, and
$\frac{3}{4}$ is more than that, so $\frac{3}{4}$ is \emph{greater} than
$\frac{1}{2}$. One fraction sits below the benchmark and the other above it, so
no common denominator is needed. Ben has more water.
$\ans{\frac{2}{5} < \frac{3}{4}}$}

\problem{\textbf{$\frac{3}{4}$ of a tray written in twelfths.} The tray is cut
into $12$ equal pieces, so the new denominator is $12$. Since
$4 \times 3 = 12$, multiply the numerator by the same factor:
$\frac{3}{4} = \frac{3 \times 3}{4 \times 3}$. Dev took $9$ of the $12$ pieces.
$\ans{\frac{9}{12}}$
\par\smallskip
\emph{Why multiplying top and bottom is legal:} $\frac{3}{3} = 1$, and
multiplying by $1$ cuts each piece into smaller pieces without changing how much
cornbread is on the plate.}

\problem{\textbf{Sam $\frac{2}{3}$ vs.\ Kim $\frac{5}{8}$.} Both are just above
$\frac{1}{2}$, so the benchmark cannot separate them --- use the common
denominator $24$ ($3 \times 8 = 24$).
\begin{align*}
\frac{2}{3} &= \frac{2 \times 8}{3 \times 8} = \frac{16}{24}, &
\frac{5}{8} &= \frac{5 \times 3}{8 \times 3} = \frac{15}{24}
\end{align*}
With the same size pieces, $16$ pieces beat $15$ pieces. Sam walked farther.
$\ans{\frac{2}{3} > \frac{5}{8}}$}

\problem{\textbf{Morning crew $\frac{4}{9}$ vs.\ afternoon crew $\frac{5}{6}$.}
Half of nine ninths is $\frac{4.5}{9}$, so $\frac{4}{9}$ falls just short of
$\frac{1}{2}$. Half of six sixths is $\frac{3}{6}$, and $\frac{5}{6}$ is well
past it. Below the benchmark is always less than above it.
$\ans{\frac{4}{9} < \frac{5}{6}}$}

\problem{\textbf{$\frac{7}{10}$ of a cup written in fortieths.} The target
denominator is $40$ and $10 \times 4 = 40$, so multiply the numerator by $4$
as well: $\frac{7}{10} = \frac{7 \times 4}{10 \times 4}$.
$\ans{\frac{28}{40}}$}

\problem{\textbf{Mia $\frac{5}{6}$ vs.\ Jonah $\frac{7}{9}$.} Both are above
$\frac{1}{2}$, so build a common denominator. The least common multiple of $6$
and $9$ is $18$.
\begin{align*}
\frac{5}{6} &= \frac{5 \times 3}{6 \times 3} = \frac{15}{18}, &
\frac{7}{9} &= \frac{7 \times 2}{9 \times 2} = \frac{14}{18}
\end{align*}
$15$ eighteenths is more than $14$ eighteenths, so Mia has read more.
$\ans{\frac{5}{6} > \frac{7}{9}}$}

\problem{\textbf{Order $\frac{3}{5}$, $\frac{1}{2}$, $\frac{7}{10}$ from least
to greatest.} Every denominator divides $10$, so tenths is the common unit:
$\frac{3}{5} = \frac{6}{10}$, $\frac{1}{2} = \frac{5}{10}$, and $\frac{7}{10}$
already has it. Order the numerators $5 < 6 < 7$, then rewrite the fractions in
their original form. The Baker family planted least, the Chen family most.
$\ans{\frac{1}{2} < \frac{3}{5} < \frac{7}{10}}$}

\problem{\textbf{$\frac{7}{8}$ full vs.\ $\frac{9}{10}$ full.} Both are above
$\frac{1}{2}$, so measure the missing part instead: the first tank is
$\frac{1}{8}$ from full, the second is $\frac{1}{10}$ from full. With the same
numerator, the fraction with the \emph{larger} denominator is the smaller
amount, so $\frac{1}{10} < \frac{1}{8}$: the second tank is missing less.
Missing less means holding more.
$\ans{\frac{7}{8} < \frac{9}{10}}$
\par\smallskip
\emph{Check with a common denominator:} $\frac{7}{8} = \frac{35}{40}$ and
$\frac{9}{10} = \frac{36}{40}$, and $35 < 36$. Same conclusion.}

\problem{\textbf{Recipe A $\frac{11}{15}$ vs.\ Recipe B $\frac{3}{4}$.} Both sit
between $\frac{1}{2}$ and $1$, so the benchmark is useless here. The least
common multiple of $15$ and $4$ is $60$.
\begin{align*}
\frac{11}{15} &= \frac{11 \times 4}{15 \times 4} = \frac{44}{60}, &
\frac{3}{4} &= \frac{3 \times 15}{4 \times 15} = \frac{45}{60}
\end{align*}
The recipes differ by only $\frac{1}{60}$ of a cup, and Recipe B is the larger.
$\ans{\frac{11}{15} < \frac{3}{4}}$}

\problem{\textbf{Order $\frac{7}{12}$, $\frac{5}{8}$, $\frac{2}{3}$,
$\frac{3}{4}$ from greatest to least.} All four denominators divide $24$, so
$24$ is the common unit.
\begin{align*}
\frac{7}{12} = \frac{14}{24}, \qquad \frac{5}{8} = \frac{15}{24}, \qquad
\frac{2}{3} = \frac{16}{24}, \qquad \frac{3}{4} = \frac{18}{24}
\end{align*}
Ordering the numerators from greatest to least gives $18 > 16 > 15 > 14$, so
Nia walked the farthest and Rosa the least.
$\ans{\frac{3}{4} > \frac{2}{3} > \frac{5}{8} > \frac{7}{12}}$
\par\smallskip
\emph{Grading note:} accept any correct common denominator (a student who uses
$48$ or $72$ gets the same order). The order is what is being assessed.}
