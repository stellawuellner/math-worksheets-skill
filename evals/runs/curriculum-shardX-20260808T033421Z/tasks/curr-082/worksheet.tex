\wstitleblock{Fractions on the Number Line}{Grade 3 --- Fractions as Numbers}{}

\begin{center}
\small Reading points, placing fractions, and comparing them on a number line
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions:}
\begin{itemize}
  \item On every line below, the space from $0$ to $1$ is cut into equal parts. Count the parts to
        find the size of one jump, then count the jumps from $0$.
  \item Write fractions as $\frac{\text{jumps}}{\text{parts in one whole}}$.
  \item Show your counting on the line if it helps, and write your final answer on the answer line.
\end{itemize}

\vspace{0.15cm}

point show?
