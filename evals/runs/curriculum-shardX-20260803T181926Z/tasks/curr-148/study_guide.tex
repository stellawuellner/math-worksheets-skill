\sstitleblock{Building and Comparing Two Number Patterns}

\noindent\small A rule says where to start and what to add. Two rules make two
patterns, and the interesting question is how the \emph{matching} terms of the
two patterns are related.\normalsize

\skillheading{Building a pattern from its rule}
\begin{formulabox}
\textbf{Rule:} Write the starting number as term $1$. Add the step to get each
new term.\par
Going from term $1$ to term $n$ takes $n - 1$ steps, so
term $n$ = start $+$ step $\times (n-1)$. Count the steps, not the terms: five
terms means four additions.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Start at $1$ and add $4$ each time. What is term $5$?
\step{Term $5$ is four steps after term $1$, so add the step four times rather
than five --- the start is already term $1$.}
\step{$1 + 4 \times 4 = 1 + 16$. Written out: $1$, $5$, $9$, $13$, $17$.}

Term $5 = \ans{17}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Start at $2$ and add $5$ each time. What is term $6$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{Comparing corresponding terms}
\begin{formulabox}
\textbf{Rule:} Corresponding terms are the terms in the same place. Line the
two patterns up in a table, one under the other, then divide a term of the
second by the matching term of the first.\par
When both patterns start at $0$, the answer is the same for every pair, and it
equals the second step divided by the first. When the patterns start somewhere
else, doubling the step does \emph{not} double the terms unless the starting
number doubles too.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Pattern A: $0$, $3$, $6$, $9$. Pattern B: $0$, $9$,
$18$, $27$. How many times B's term is A's term?
\step{Both patterns start at $0$, so pick any matching pair and divide --- B on
top, because the question asks what A was multiplied by.}
\step{Fourth terms: $27 \div 9$. Checking another pair, $18 \div 6$ gives the
same.}

Each B term is $\ans{3}$ times its A term
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Pattern A: $0$, $5$, $10$, $15$. Pattern B: $0$, $20$,
$40$, $60$. How many times B's term is A's term?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Turning two patterns into ordered pairs}
\begin{formulabox}
\textbf{Rule:} Take corresponding terms and write them as (first pattern's
term, second pattern's term). Plot each pair by going across first, then up.\par
The points always fall in a straight line, and the relationship between the
matching terms tells you the second number from the first one.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Pattern A: $0$, $2$, $4$, $6$. Pattern B: $0$, $10$,
$20$, $30$. Pairs: $(0,0)$, $(2,10)$, $(4,20)$, $(6,30)$. What is the second
number of the pair whose first number is $6$?
\step{Each B term is $10 \div 2 = 5$ times its A term, so the second number of
any pair is $5$ times the first --- that is quicker than extending the table.}
\step{$5 \times 6 = 30$, which matches the fourth pair already listed.}

Second number $= \ans{30}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Pattern A: $0$, $3$, $6$. Pattern B: $0$, $12$, $24$.
What is the second number of the pair whose first number is $9$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Term $1$ is the starting number, before any adding ---
so term $6$ takes five additions, not six. And when you plot an ordered pair,
the first number is the across number every time.
\end{watchoutbox}
