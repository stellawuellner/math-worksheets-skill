\wstitleblock{Systems of Two Linear Equations: Intersections as Solutions}{Grade 8 / Pre-Algebra}{}

\begin{center}\small\textit{Testing a point in both equations, reading an intersection from a graph or table, and solving to find where two lines meet}\end{center}

\noindent\textbf{The big idea.} A point is on a line exactly when its $(x,y)$ makes that
line's equation true. So the point where two lines \textbf{cross} is the one $(x,y)$ that
makes \textbf{both} equations true at once --- that is what ``solution of the system'' means.

\begin{center}
\begin{tikzpicture}[scale=0.52]
  \draw[step=1, gray!30, very thin] (-0.4,-0.4) grid (6.4,6.4);
  \draw[->] (-0.4,0) -- (6.6,0) node[right] {$x$};
  \draw[->] (0,-0.4) -- (0,6.6) node[above] {$y$};
  \draw[very thick, blue] (-0.4,0.6) -- (5.4,6.4) node[above right] {$y=x+1$};
  \draw[very thick, red] (-0.4,5.4) -- (5.4,-0.4) node[below right] {$y=-x+5$};
  \fill (2,3) circle (4pt);
  \node[above right] at (2,3) {\textbf{$(2,3)$}};
\end{tikzpicture}
\end{center}

\noindent\small \textit{How every graph on this sheet is read: the marked crossing point is
the only pair of numbers that satisfies both equations. Substitute it into each equation to
prove it. No values from this picture belong to any problem below --- use the numbers each
problem gives you.}\normalsize

\noindent\rule{\linewidth}{0.4pt}

\problem[5cm]{Show that $(3, 2)$ is the solution of this system by substituting
$x = 3$ and $y = 2$ into \textbf{each} equation and checking that both are true.
\[ x + y = 5 \qquad\text{and}\qquad x - y = 1 \]
Then finish the sentence: the graphs of these two lines cross at the point \ldots}

\problem[5cm]{Ella claims the two lines
\[ y = x - 3 \qquad\text{and}\qquad y = 2x - 7 \]
cross at $(4, 1)$. Test the point in both equations. Is Ella right? Write the
intersection point as your answer.}

\problem[6cm]{The table gives points on two lines. Find the one row where the two
lines share the same $y$-value --- that row is where the graphs cross.
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $y = 2x - 1$ & $y = -x + 8$ \\ \hline
$0$ & $-1$ & $8$ \\ \hline
$1$ & $1$  & $7$ \\ \hline
$2$ & $3$  & $6$ \\ \hline
$3$ & $5$  & $5$ \\ \hline
$4$ & $7$  & $4$ \\ \hline
\end{tabular}
\end{center}
Write the intersection point $(x, y)$, then check it in both equations.}

\problem[6cm]{Two more lines are tabulated below.
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $y = x + 2$ & $y = 3x - 2$ \\ \hline
$0$ & $2$ & $-2$ \\ \hline
$1$ & $3$ & $1$  \\ \hline
$2$ & $4$ & $4$  \\ \hline
$3$ & $5$ & $7$  \\ \hline
$4$ & $6$ & $10$ \\ \hline
\end{tabular}
\end{center}
Where do these graphs intersect? Explain in one sentence how the table shows it.}

\problem[6cm]{Solve this system without a table: set the two expressions for $y$
equal to each other, find $x$, then find $y$.
\[ y = 2x + 1 \qquad\text{and}\qquad y = -x + 7 \]
Write your answer as an ordered pair and say what it means about the two graphs.}

\problem[6.5cm]{Maya says the graphs of
\[ x + 2y = 8 \qquad\text{and}\qquad 3x - y = 3 \]
cross at $(8, 0)$. Substitute her point into \textbf{both} equations to show that it
satisfies only one of them. Then find the point that satisfies both, and write it
as your answer.}

\problem[6.5cm]{\textbf{Break-even story.} A skate park charges two ways.
Plan A costs a membership fee of $30$ dollars plus $5$ dollars per visit, so its
cost in dollars after $x$ visits is $y = 5x + 30$. Plan B has no fee and costs
$10$ dollars per visit, so $y = 10x$.

\noindent Find the number of visits where the two plans cost the same, and that cost
in dollars.
Write the answer as the point $(x, y)$ where the two graphs intersect.}

\problem[6.5cm]{\textbf{No intersection.} Compare the two lines in this table.
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $y = 2x + 1$ & $y = 2x - 4$ \\ \hline
$0$ & $1$ & $-4$ \\ \hline
$1$ & $3$ & $-2$ \\ \hline
$2$ & $5$ & $0$  \\ \hline
$3$ & $7$ & $2$  \\ \hline
\end{tabular}
\end{center}
Explain, using the table and the slopes, why these graphs never cross, and say what
that tells you about the number of solutions of the system.
Write two or three complete sentences.\noansline}

\problem[7cm]{Neither equation is solved for $y$ here:
\[ 2x + 3y = 12 \qquad\text{and}\qquad x - y = 1 \]
Find the intersection point. (One good route: solve the second equation for $x$ or
$y$, substitute into the first, then back-substitute.) Check your point in both
original equations.}

\problem[7cm]{\textbf{Build your own.} Write a system of two \emph{different} linear
equations whose graphs intersect at $(4, -1)$.

\noindent Then explain how you know your point is the solution of your system, and
sketch or describe what the two graphs look like near that point.\noansline}
