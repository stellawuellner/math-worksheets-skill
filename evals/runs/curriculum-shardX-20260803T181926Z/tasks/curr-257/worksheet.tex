\wstitleblock{Multi-Step Linear Equations with Rational Coefficients}{Algebra 1 \quad Equations and Inequalities}{}

\begin{center}
\textit{Clearing fractions with the LCD, equations with the variable on both
sides, and inequalities with rational coefficients}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Show every step. A step is legal only if you do the same thing to
\emph{both} sides of the equation.
\item When fractions appear, multiplying both sides by the least common
denominator (LCD) clears them all at once --- multiply \emph{every} term,
including the terms with no fraction.
\item Give exact answers (a fraction, not a rounded decimal), and check your
solution in the original equation.
\item In an inequality, multiplying or dividing both sides by a
\textbf{negative} number reverses the inequality symbol.
\end{itemize}

\problem[5cm]{Solve for $x$. \qquad $\dfrac{x}{3} + 5 = 9$}

\problem[5cm]{Solve for $x$. \qquad $\dfrac{2}{5}x - 3 = 7$}

\problem[5.5cm]{Solve for $x$. Multiply both sides by the LCD first.
\qquad $\dfrac{x}{4} + \dfrac{x}{6} = 10$}

\problem[5.5cm]{Solve for $x$. \qquad $\dfrac{3}{4}(x - 8) = 6$}

\problem[6cm]{Solve for $x$. The variable appears on both sides.
\qquad $\dfrac{2}{3}x + 1 = \dfrac{x}{6} + 4$}

\problem[5.5cm]{Solve the inequality and write the solution in symbols.
\qquad $\dfrac{x}{2} - 3 > 4$}

\problem[6.5cm]{Solve for $x$. Watch the binomial numerators when you multiply
by the LCD.
\[ \frac{x + 2}{3} + \frac{x - 1}{2} = 6 \]}

\problem[6.5cm]{Solve for $x$.
\[ \frac{3}{4}x - \frac{2}{3} = \frac{x}{6} + \frac{5}{3} \]}

\problem[6cm]{Solve the inequality and write the solution in symbols. Decide
carefully whether the symbol turns around.
\qquad $-\dfrac{2}{3}x + 1 \ge 5$}

\problem[6cm]{Clear the fractions from the equation below by multiplying
\emph{both sides} by the LCD, then write the equivalent equation that has only
integer coefficients. (Do not solve it.)
\[ \frac{5}{6}x - \frac{1}{2} = \frac{1}{3}x + 2 \]}

\problem[7cm]{Solve for $x$. Distribute the outside factor before you collect
like terms.
\[ \frac{1}{2}\left(\frac{x}{3} - 4\right) + \frac{5}{6}x = 3 \]}

\problem[8cm]{Solve for $x$. Multiply by the LCD of all three fractions at
once, then finish.
\[ \frac{3(2x - 1)}{4} - \frac{2(x + 1)}{3} = \frac{5}{12}x + 4 \]}
