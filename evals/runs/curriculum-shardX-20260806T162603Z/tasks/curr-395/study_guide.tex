\sstitleblock{Binomial Coefficients and Expansion Structure}

\skillheading{Skill 1 --- Expanding a binomial power}

\begin{formulabox}
\textbf{The pattern:} $(A + B)^n$ has $n + 1$ terms. The powers of $A$ fall from $n$ to $0$
while the powers of $B$ rise from $0$ to $n$, and the coefficients are row $n$ of Pascal's
triangle ($1, 3, 3, 1$ for $n = 3$; $1, 4, 6, 4, 1$ for $n = 4$).\par
\textbf{Sign and coefficient travel with the letter.} If $B$ is negative the signs alternate;
if $A$ is $2x$, then $A^3$ is $8x^3$, not $2x^3$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $(x + 2)^3$.
\step{The exponent is $3$, so the coefficients are row $3$ of the triangle and the powers of $2$ climb as the powers of $x$ fall --- the whole expansion can be written straight down.}
\step{$1, 3, 3, 1$ against $2^0, 2^1, 2^2, 2^3$: \ $x^3 + 3x^2(2) + 3x(4) + 8$.}

$\Rightarrow\quad \ans{x^3 + 6x^2 + 12x + 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $(x - 1)^4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^4 - 4x^3 + 6x^2 - 4x + 1}$}
\end{tryitbox}

\skillheading{Skill 2 --- Picking out one named term}

\begin{formulabox}
\textbf{Do not expand.} The term at index $k$ of $(A + B)^n$ is $\binom{n}{k}A^{n-k}B^{k}$,
so a question about one power of $x$ fixes $k$ and the rest is arithmetic.\par
\textbf{Find $k$ from the power you want.} If $A = x$ and you want $x^{m}$, then
$n - k = m$, so $k = n - m$. Index $k$ starts at $0$, so the third term has $k = 2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the coefficient of $x^2$ in $(x + 3)^4$.
\step{Only one term carries $x^2$, so the index is forced: $n - k = 2$ with $n = 4$ gives $k = 2$, and nothing else needs computing.}
\step{$\binom{4}{2}x^2 3^2 = 6 \cdot 9 \cdot x^2$}

$\Rightarrow\quad$ coefficient $= \ans{54}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the coefficient of $x^2$ in $(x + 4)^4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{96}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reading a binomial coefficient}

\begin{formulabox}
\textbf{The formula:} $\binom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$, which counts the ways of
choosing $k$ of the $n$ factors to contribute their second term.\par
\textbf{Symmetry:} $\binom{n}{k} = \binom{n}{n-k}$, because choosing $k$ to take is the same
as choosing $n - k$ to leave. That is the mirror symmetry you can see in every row of
Pascal's triangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compute $\binom{7}{2}$.
\step{The formula counts choices, so only the top two factors of $7!$ survive against $2!$ --- there is no need to write out either factorial in full.}
\step{$\dfrac{7 \cdot 6}{2} = \dfrac{42}{2}$}

$\Rightarrow\quad \ans{21}$ \quad (and by symmetry $\binom{7}{5}$ is the same number)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compute $\binom{8}{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56}$}
\end{tryitbox}

\skillheading{Skill 4 --- Summing across an expansion}

\begin{formulabox}
\textbf{Substituting $x = 1$ collapses an expansion to its coefficient total}, because every
power of $1$ is $1$. For $(1 + x)^n$ that total is $2^n$ --- one row of Pascal's triangle
adds to $2^n$.\par
\textbf{Those totals themselves form a geometric sequence} with ratio $2$, so a sum of
several rows is an ordinary geometric series: $a_1(r^{m} - 1)/(r - 1)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\sum_{k=0}^{4}\binom{4}{k}$.
\step{The sum runs over a whole row of Pascal's triangle, so it is the coefficient total of $(1 + x)^4$ --- the substitution shortcut applies.}
\step{Row $4$ is $1, 4, 6, 4, 1$, and $1 + 4 + 6 + 4 + 1 = 2^4$.}

$\Rightarrow\quad \ans{16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\sum_{n=0}^{4} 3^n$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{121}$}
\end{tryitbox}
