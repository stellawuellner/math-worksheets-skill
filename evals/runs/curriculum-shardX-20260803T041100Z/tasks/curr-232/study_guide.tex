\sstitleblock{Evaluating Functions from Rules, Tables, and Graphs}

\skillheading{Skill 1 --- Evaluate a function from its rule}

\begin{formulabox}
\textbf{Rule:} \ $f(\text{input}) = \text{output}$. To find $f(a)$, replace every
$x$ in the rule with $a$ --- in parentheses --- then simplify.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $f(3)$ is \emph{not} $f$ times 3. The parentheses hold the input.
  \item Keep the input in brackets when it is negative: $(-4)^2 = 16$,
        but $-4^2 = -16$.
  \item Order of operations still applies after substituting.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 4x - 7$. \ Find $f(3)$.
\step{The rule is given, so evaluating means substituting the input for $x$ and
      simplifying --- no table or graph is needed.}
\step{$f(3) = 4(3) - 7 = 12 - 7$.}
$\Rightarrow\quad \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = x^2 + 2x$. \ Find $f(-4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 2 --- Read a value from a table or a graph}

\begin{formulabox}
\textbf{Rule:} \ A table row and a graph point say the same thing as a rule:
they pair one input with one output.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Table:} find the input in the input row, read the output directly below.
  \item \textbf{Graph:} find the input on the $x$-axis, go up (or down) to the curve,
        then across to the $y$-axis. That height is the output.
  \item If a question asks ``for which input'', the answer is an $x$-value, not a height.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $q(2)$ from this table.
\par
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
$x$ & 0 & 1 & 2 & 3 \\\hline
$q(x)$ & 2 & 5 & 8 & 11 \\\hline
\end{tabular}
\end{center}
\step{The table already lists the outputs, so reading replaces computing --- locate
      the input column instead of substituting.}
\step{The column headed 2 has 8 directly beneath it.}
$\Rightarrow\quad \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Read $p(4)$ from the graph of $p$.
\par
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=center, xmin=-3, xmax=5.5, ymin=-2, ymax=3,
  xtick={-2,0,2,4}, ytick={-1,0,1,2},
  tick label style={font=\tiny}, width=5.6cm, height=3.8cm, enlargelimits=false]
  \addplot[thick, blue] coordinates {(-2,-1) (0,0) (2,1) (4,2)}
    node[pos=1, above left, font=\tiny] {$p$};
\end{axis}
\end{tikzpicture}
\end{center}
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Compare two functions at the same input}

\begin{formulabox}
\textbf{Rule:} \ To compare $f$ and $g$, evaluate \textbf{both at the same input},
then compare the two outputs.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item On a graph: at that input, the higher curve has the greater output.
  \item Where the graphs cross, the outputs are equal --- that input solves
        $f(x) = g(x)$.
  \item Comparing at \emph{different} inputs answers a different question.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 4x - 7$ and $g(x) = 2x + 1$. Compare $f(3)$ and $g(3)$.
\step{Same input, two rules --- evaluate each separately, then compare the outputs.}
\step{$f(3) = 4(3) - 7 = 5$ and $g(3) = 2(3) + 1 = 7$; 5 is less than 7.}
$\Rightarrow\quad f(3) \ans{<} g(3)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = 5x - 3$ and $g(x) = x^2 + 2$. Compare $f(2)$ and $g(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}
