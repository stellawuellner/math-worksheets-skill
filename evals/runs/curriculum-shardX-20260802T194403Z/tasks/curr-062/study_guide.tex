\sstitleblock{Subtracting Across a Zero with Regrouping}

\skillheading{1. Subtracting across ONE zero}

\begin{formulabox}
\textbf{The rule:} you cannot borrow from a $0$ --- a zero has nothing to give.
Move \emph{left} to the first digit that is not $0$, take one from it, and hand
the group back one place at a time.\\[3pt]
$1$ hundred $= 10$ tens \qquad $1$ ten $= 10$ ones\\[3pt]
\textbf{Say it as trading:} ``The tens are empty, so the hundreds give the tens
ten. Now the tens can give the ones ten.''
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $403 - 167$
\step{The ones need more ($3 < 7$) and the tens digit is $0$, so the tens cannot
help --- this is a trade across a zero, starting from the hundreds.}
\step{$403 = 3$ hundreds, $10$ tens, $3$ ones $\;\to\;$ $3$ hundreds, $9$ tens,
$13$ ones.}
\step{$13 - 7 = 6$ ones; $9 - 6 = 3$ tens; $3 - 1 = 2$ hundreds
$\;\Rightarrow\;$ $\ans{236}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $502 - 245$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{257}$}
\end{tryitbox}

\skillheading{2. Subtracting across TWO zeros}

\begin{formulabox}
\textbf{The rule:} when the tens \emph{and} the ones are both $0$, the trade
travels two places. Rewrite the whole number before you subtract anything:\\[3pt]
$600 = 5$ hundreds $+\ 9$ tens $+\ 10$ ones\\[3pt]
The digits $5$, $9$, $10$ are the pattern for every ``hundreds minus something''
problem: one less hundred, nine tens, ten ones.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $600 - 274$
\step{Ones and tens are both $0$, so nothing can be borrowed until the hundreds
break --- rewrite $600$ first, then subtract once.}
\step{$600 = 5$ hundreds, $9$ tens, $10$ ones.}
\step{$10 - 4 = 6$; $9 - 7 = 2$; $5 - 2 = 3$ $\;\Rightarrow\;$ $\ans{326}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $900 - 358$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{542}$}
\end{tryitbox}

\skillheading{3. Estimate the difference first}

\begin{formulabox}
\textbf{The rule:} round \emph{each} number to the same place, then subtract the
rounded numbers. Rounding to the nearest hundred: look at the tens digit ---
$5$ or more rounds up, less than $5$ rounds down.\\[3pt]
An estimate is a \emph{safety net}: if your exact answer is far from the
estimate, you dropped a trade somewhere.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Estimate $497 - 212$ to the nearest hundred.
\step{This is an estimate, not the exact answer, so round both numbers before
subtracting instead of regrouping.}
\step{$497 \to 500$ (tens digit $9$); $212 \to 200$ (tens digit $1$).}
\step{$500 - 200 = \ans{300}$, so the exact answer should be near $300$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Estimate $803 - 396$ to the nearest hundred.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{400}$}
\end{tryitbox}

\skillheading{4. Check by adding}

\begin{formulabox}
\textbf{The rule:} addition undoes subtraction.\\[3pt]
If $\text{top} - \text{bottom} = \text{answer}$, then
$\text{answer} + \text{bottom} = \text{top}$.\\[3pt]
Add your answer to the number you subtracted. If you get the number you started
with, the trades were right.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $405 - 218 = 187$ correct?
\step{Checking a difference means running it backwards, so add the answer to the
number that was subtracted.}
\step{$187 + 218$: \ $7 + 8 = 15$ (write $5$, carry $1$); \ $8 + 1 + 1 = 10$
(write $0$, carry $1$); \ $1 + 2 + 1 = 4$.}
\step{The check gives $\ans{405}$, the number we started from, so $187$ is
correct.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Check $300 - 126 = 174$ by adding $174 + 126$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{300}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Never subtract the smaller digit from the larger one just
because it is easier. In $903 - 458$ the ones column is $3 - 8$, \emph{not}
$8 - 3$. Turning it around gives $555$ instead of the correct $445$.
\end{watchoutbox}
