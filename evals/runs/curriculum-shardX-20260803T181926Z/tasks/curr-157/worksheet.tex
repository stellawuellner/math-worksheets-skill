\wstitleblock{Unit Rates with Fractions}{Grades 6--7 \quad Ratios and Proportional Relationships}{}

\begin{center}
\textit{Finding a unit rate by dividing one fraction by another, comparing two
rates by their unit rates, and pulling a unit rate out of a word problem}
\end{center}

\noindent\textbf{How to work these problems.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A \textbf{unit rate} tells you how much there is for exactly \emph{one}
of something --- one hour, one pound, one batch.
\item Write the rate as a division: the amount divided by how many units it
took. Then divide by multiplying by the reciprocal.
\item Turn every mixed number into an improper fraction \emph{before} you
divide.
\item Give each answer as a fraction in lowest terms (or a whole number when
it works out to one), and say what it counts per one unit.
\end{itemize}

\vspace{0.15cm}

\problem[5cm]{Zara jogs $\frac{1}{2}$ of a mile in $\frac{1}{4}$ of an hour.
How many miles does she jog per hour?}

\problem[5cm]{A printer prints $\frac{3}{4}$ of a page in $\frac{1}{2}$ of a
minute. How many pages does it print per minute?}

\problem[5cm]{A tap fills $\frac{2}{3}$ of a litre in $\frac{1}{6}$ of a
minute. How many litres does it fill per minute?}

\problem[6cm]{A recipe uses $1\frac{1}{2}$ cups of flour to make $\frac{3}{4}$
of a batch. How many cups of flour does one whole batch need?}

\problem{Rowan walks $\frac{3}{5}$ of a kilometre in $\frac{1}{4}$ of an hour.
Sadie walks $\frac{2}{3}$ of a kilometre in $\frac{1}{3}$ of an hour. Find each
walker's speed in kilometres per hour, then write $>$ or $<$ in the last blank
so that ``Rowan's speed \underline{\hspace{0.9cm}} Sadie's speed'' is true.
\par\vspace*{8cm}
\noindent Rowan: \ansblank \qquad Sadie: \ansblank \qquad Rowan
\ansblank\ Sadie}

\problem[5cm]{A tortoise covers $\frac{5}{8}$ of a mile in $\frac{5}{6}$ of an
hour. How many miles does it cover per hour?}

\problem[6cm]{A hose fills $\frac{7}{8}$ of a barrel in $\frac{1}{2}$ of an
hour. How much of a barrel does it fill per hour?}

\problem[6cm]{At a market, $\frac{3}{4}$ of a pound of cheese costs $6$
dollars. What is the cost of one pound, in dollars?}

\problem[6cm]{Painting $\frac{2}{5}$ of a wall takes $\frac{3}{4}$ of an hour.
What fraction of the wall gets painted in one hour?}

\problem{Brand A sells $\frac{2}{3}$ of a pound of trail mix for $2$ dollars.
Brand B sells $\frac{3}{4}$ of a pound for $3$ dollars. Find each brand's price
per pound in dollars, then write $>$ or $<$ in the last blank so that ``Brand
A's price per pound \underline{\hspace{0.9cm}} Brand B's price per pound'' is
true.
\par\vspace*{8cm}
\noindent Brand A: \ansblank \qquad Brand B: \ansblank \qquad A \ansblank\ B}

\problem[7cm]{A cyclist rides $4\frac{1}{2}$ miles in $\frac{3}{4}$ of an
hour. How many miles does the cyclist ride per hour?}

\problem[8cm]{A bottling machine fills $2\frac{1}{4}$ litres in $\frac{1}{3}$
of a minute. Working at that same rate, how many litres does it fill in $4$
minutes?}
