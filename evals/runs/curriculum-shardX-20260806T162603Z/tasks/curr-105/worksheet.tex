\begin{center}\textit{Expanded form, comparing numbers, and rounding
decisions}\end{center}

\noindent\textbf{Directions:} After the warm-up, the three skills are mixed
together --- read each problem carefully and decide \emph{which} tool it needs
before you start. The last problem uses all three at once.

\problem[2.5cm]{Write in standard form:
$60{,}000 + 2{,}000 + 400 + 9 = $~\ansblank}

\problem[2.5cm]{Fill in the missing addend:
\par\vspace{2pt}\noindent
\hspace*{1cm}$8{,}315 = 8{,}000 + $~\ansblank~$+ \ 10 + 5$}

\problem[2.5cm]{Write $<$, $>$, or $=$:
\par\vspace{2pt}\noindent
\hspace*{1.5cm}$36{,}254$~~\ansblank~~$36{,}245$}

\problem[2.5cm]{Round $7{,}438$ to the nearest hundred.}

\problem[2.5cm]{Which number is $30{,}000 + 500 + 20 + 8$?}

\problem[3cm]{Round $52{,}499$ to the nearest thousand. (Think: is 499 enough
to round up?)}

\problem[3.5cm]{Order from least to greatest: $8{,}742$; \ $8{,}724$; \
$8{,}772$
\par\vspace{2pt}\noindent
least~\ansblank\quad middle~\ansblank\quad greatest~\ansblank}

\problem[3cm]{Round $63{,}970$ to the nearest hundred.}

\problem[3.5cm]{First write the total in standard form:
\par\vspace{2pt}\noindent
\hspace*{1cm}$40{,}000 + 3{,}000 + 70 = $~\ansblank
\par\vspace{2pt}\noindent
Then write $<$, $>$, or $=$: \quad that total~~\ansblank~~$43{,}900$}

\problem[4cm]{A wildlife count finds $45{,}067$ ducks. One report rounds the
count to the nearest thousand:~\ansblank. Another rounds it to the nearest
hundred:~\ansblank.
\par\vspace{2pt}\noindent
Which rounded number is closer to the actual count? Say which and show how far
away each one is.}

\problem[4cm]{Ben says $76{,}099 > 76{,}105$ ``because $76{,}099$ ends in two
nines.'' Write the correct symbol: \ $76{,}099$~~\ansblank~~$76{,}105$.
\par\vspace{2pt}\noindent
Then explain what Ben did wrong.}

\problem[5.5cm]{Three games had attendances of $41{,}807$, \ $41{,}490$, and
$42{,}088$.
\par\vspace{2pt}\noindent
(a) Order them from greatest to least:
~\ansblank\quad\ansblank\quad\ansblank
\par\vspace{2pt}\noindent
(b) Round the greatest attendance to the nearest hundred:~\ansblank
\par\vspace{2pt}\noindent
(c) Which two attendances round to the \emph{same} nearest-thousand value?
Explain how you decided.}
