\problem{Find the inverse rule for $f(x) = 5x - 8$.}

\textbf{Solution:} Write $x = f(y)$ and solve for $y$.
\begin{align*}
  x &= 5y - 8 \\
  x + 8 &= 5y \\
  y &= \frac{x + 8}{5}
\end{align*}
Every step is reversible, so the rule found is the inverse. \ans{f^{-1}(x) = \dfrac{x + 8}{5}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the inverse rule for $g(x) = \dfrac{x}{4} + 3$.}

\textbf{Solution:} Write $x = g(y)$ and undo the two operations in reverse order.
\begin{align*}
  x &= \frac{y}{4} + 3 \\
  x - 3 &= \frac{y}{4} \\
  y &= 4(x - 3) = 4x - 12
\end{align*}
Check: $g(4x - 12) = \frac{4x - 12}{4} + 3 = x - 3 + 3 = x$. \ans{g^{-1}(x) = 4x - 12}

\problem{Let $p(x) = 2x + 9$ and $q(x) = \dfrac{x - 9}{2}$. (a) Simplify $p(q(x))$. (b) Explain what the result says about $p$ and $q$.}

\textbf{Solution (a):} Substitute $q(x)$ into $p$ and simplify.
\begin{align*}
  p\bigl(q(x)\bigr) &= 2\left(\frac{x - 9}{2}\right) + 9 \\
                    &= (x - 9) + 9 = x
\end{align*}
The composition collapses to the identity. \ans{x}

\vspace{0.15cm}
\textbf{Solution (b) --- instructor-judged.} A correct answer says that $p$ and $q$ undo
each other: feeding a number through $q$ and then through $p$ returns the original number,
so $q$ is the inverse of $p$ (and $p$ the inverse of $q$). Look for the word \emph{inverse}
(or ``they undo each other'') together with a reference to the part (a) result being $x$.
Saying only ``they are related'' or ``they are opposites'' is not enough. A student who adds
that $q(p(x)) = x$ must also hold has given the complete two-sided argument --- worth extra
credit, not required.

\problem{Find the inverse rule for $r(x) = \dfrac{2x + 1}{x - 3}$.}

\textbf{Solution:} Write $x = r(y)$, clear the fraction, then collect every $y$ on one side.
\begin{align*}
  x &= \frac{2y + 1}{y - 3} \\
  x(y - 3) &= 2y + 1 \\
  xy - 3x &= 2y + 1 \\
  xy - 2y &= 3x + 1 \\
  y(x - 2) &= 3x + 1
\end{align*}
Dividing by $x - 2$ finishes it. \ans{r^{-1}(x) = \dfrac{3x + 1}{x - 2}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the inverse rule for $s(x) = \sqrt{x + 7} - 2$, inputs $x \ge -7$.}

\textbf{Solution:} Isolate the radical first, then square.
\begin{align*}
  x &= \sqrt{y + 7} - 2 \\
  x + 2 &= \sqrt{y + 7} \\
  (x + 2)^2 &= y + 7 \\
  y &= (x + 2)^2 - 7
\end{align*}
Squaring is safe because $\sqrt{\ }$ is never negative, so $x + 2 \ge 0$ on the range of $s$. \ans{s^{-1}(x) = (x + 2)^2 - 7}

\problem{For $m(x) = 4x + 3$, find $m^{-1}(19)$.}

\textbf{Solution:} Build the inverse rule, then evaluate it at $19$.
\begin{align*}
  x &= 4y + 3 \quad\Longrightarrow\quad y = \frac{x - 3}{4} \\
  m^{-1}(19) &= \frac{19 - 3}{4} = \frac{16}{4}
\end{align*}
Equivalently, ask what input makes $4x + 3 = 19$. \ans{m^{-1}(19) = 4}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the inverse rule for $t(x) = \dfrac{3 - 2x}{5}$.}

\textbf{Solution:} Multiply by $5$ before touching the numerator.
\begin{align*}
  x &= \frac{3 - 2y}{5} \\
  5x &= 3 - 2y \\
  2y &= 3 - 5x
\end{align*}
The sign of the $x$ term flips because $y$ was subtracted. \ans{t^{-1}(x) = \dfrac{3 - 5x}{2}}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the inverse rule for $u(x) = \dfrac{4}{x + 6} + 1$.}

\textbf{Solution:} Peel the outside operations off in reverse order, then reciprocate.
\begin{align*}
  x &= \frac{4}{y + 6} + 1 \\
  x - 1 &= \frac{4}{y + 6} \\
  y + 6 &= \frac{4}{x - 1} \\
  y &= \frac{4}{x - 1} - 6
\end{align*}
The input $x = 1$ is excluded, matching the horizontal asymptote $y = 1$ of $u$. \ans{u^{-1}(x) = \dfrac{4}{x - 1} - 6}

\problem{Let $w(x) = (x - 4)^3 + 2$. Test the claim $w^{-1}(x) = \sqrt[3]{x - 2} + 4$ by simplifying $w\bigl(w^{-1}(x)\bigr)$.}

\textbf{Solution:} Substitute the claimed rule into $w$ and simplify.
\begin{align*}
  w\bigl(w^{-1}(x)\bigr) &= \Bigl(\bigl(\sqrt[3]{x - 2} + 4\bigr) - 4\Bigr)^3 + 2 \\
                         &= \bigl(\sqrt[3]{x - 2}\bigr)^3 + 2 \\
                         &= (x - 2) + 2 = x
\end{align*}
The composition returns the input, so the classmate is right. \ans{x}

\problem{Let $n(x) = 7 - 3x$. (a) Find $n^{-1}(x)$. (b) Simplify $n\bigl(n^{-1}(x)\bigr)$.}

\textbf{Solution (a):}
\begin{align*}
  x &= 7 - 3y \\
  3y &= 7 - x \\
  y &= \frac{7 - x}{3}
\end{align*}
Watch the subtraction: $3y$ moves across, not $-3y$. \ans{n^{-1}(x) = \dfrac{7 - x}{3}}

\textbf{Solution (b):}
\begin{align*}
  n\bigl(n^{-1}(x)\bigr) &= 7 - 3\left(\frac{7 - x}{3}\right) \\
                         &= 7 - (7 - x) = x
\end{align*}
The identity confirms part (a). \ans{x}

\problem{Let $v(x) = x^2 - 6x + 5$ with $x \ge 3$. (a) Confirm $v^{-1}(x) = 3 + \sqrt{x + 4}$. (b) Explain why the restriction is needed.}

\textbf{Solution (a):} Completing the square first makes the check short: $v(x) = (x - 3)^2 - 4$.
\begin{align*}
  v\bigl(v^{-1}(x)\bigr) &= \Bigl(\bigl(3 + \sqrt{x + 4}\bigr) - 3\Bigr)^2 - 4 \\
                         &= \bigl(\sqrt{x + 4}\bigr)^2 - 4 \\
                         &= (x + 4) - 4 = x
\end{align*}
The claimed rule really does undo $v$. \ans{x}

\vspace{0.15cm}
\textbf{Solution (b) --- instructor-judged.} Look for the one-to-one idea. Without a
restriction $v$ is a parabola with vertex $(3, -4)$, so it takes every output twice --- one
input on each side of $x = 3$ (for example $v(1) = v(5) = 0$). A rule that sends $0$ back to
a single input cannot exist, so $v$ has no inverse on all of its natural domain. Restricting
to $x \ge 3$ keeps only the increasing half, which is one-to-one; that half is also why the
inverse takes the $+\sqrt{\ }$ branch and not $3 - \sqrt{x + 4}$. Full credit: the
one-to-one/fails-horizontal-line idea plus a reference to the vertex or to two inputs
sharing an output.

\problem{Let $f(x) = \dfrac{x - 3}{2x + 1}$. (a) Find $f^{-1}(x)$. (b) Name the excluded input and explain.}

\textbf{Solution (a):} Clear the fraction and collect the $y$ terms.
\begin{align*}
  x(2y + 1) &= y - 3 \\
  2xy + x &= y - 3 \\
  2xy - y &= -x - 3 \\
  y(2x - 1) &= -(x + 3) \\
  y &= \frac{x + 3}{1 - 2x}
\end{align*}
Multiplying numerator and denominator by $-1$ gives the tidier form above. \ans{f^{-1}(x) = \dfrac{x + 3}{1 - 2x}}

\vspace{0.15cm}
\textbf{Solution (b) --- instructor-judged.} The answer is $x = \tfrac{1}{2}$. Accept any
response that names that value and says the denominator $1 - 2x$ is zero there, so the
expression is undefined. The deeper reason, worth extra credit: $\tfrac{1}{2}$ is the
horizontal asymptote of $f$, so it is the one output $f$ never actually produces --- an
input the inverse can never be asked about. Do not accept ``$x = -\tfrac{1}{2}$'' (that is
the exclusion for $f$ itself, not for $f^{-1}$).
