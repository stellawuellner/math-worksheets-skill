\problem{\textbf{$x^{2} + y^{2} = 25$ --- and the explicit check.}

\textbf{Implicitly.} Differentiate both sides with respect to $x$. The term $y^{2}$
carries a factor $\dfrac{dy}{dx}$ by the chain rule, and the constant differentiates
to 0:
\[ 2x + 2y\,\frac{dy}{dx} = 0 \quad\Longrightarrow\quad
   2y\,\frac{dy}{dx} = -2x \quad\Longrightarrow\quad \frac{dy}{dx} = -\frac{x}{y}. \]

\textbf{The explicit check.} On the upper half, $y = \sqrt{25 - x^{2}}$, so
substituting gives $\dfrac{dy}{dx} = \dfrac{-x}{\sqrt{25 - x^{2}}}$. Differentiating
$\sqrt{25 - x^{2}}$ directly by the chain rule gives
$\tfrac{1}{2}(25 - x^{2})^{-1/2}(-2x)$, which is the same expression. The two
methods agree, which is the point of the exercise: implicit differentiation is not
a different derivative, only a different route to it.
\[ \boxed{\frac{dy}{dx} = -\frac{x}{y} = \frac{-x}{\sqrt{25 - x^{2}}}} \]}

\problem{\textbf{$xy = 12$ at $(3, 4)$.}

\textbf{On the curve?} $3 \cdot 4 = 12$, so $xy - 12 = 0$ at that point
$\checkmark$. Do this first --- a slope formula evaluated at a point that is not on
the curve is a number with no meaning.

\textbf{Differentiating.} The left side is a product, so
\[ y + x\,\frac{dy}{dx} = 0 \quad\Longrightarrow\quad \frac{dy}{dx} = -\frac{y}{x}. \]

\textbf{At the point.} $\dfrac{dy}{dx}\Big|_{(3,4)} = -\dfrac{4}{3}$. The hyperbola
falls steeply here, which matches a slope of magnitude greater than 1.
\[ \boxed{\frac{dy}{dx} = -\frac{y}{x};\quad
   \left.\frac{dy}{dx}\right|_{(3,4)} = -\frac{4}{3}} \]}

\problem{\textbf{$x^{2} + xy + y^{2} = 7$ at $(1, 2)$.}

\textbf{On the curve?} $1 + 2 + 4 = 7$ $\checkmark$.

\textbf{Differentiating.} The middle term needs the product rule:
\[ 2x + \left(y + x\,\frac{dy}{dx}\right) + 2y\,\frac{dy}{dx} = 0. \]
Collect the $\dfrac{dy}{dx}$ terms: $(x + 2y)\dfrac{dy}{dx} = -(2x + y)$, so
$\dfrac{dy}{dx} = -\dfrac{2x + y}{x + 2y}$.

\textbf{At the point.} $-\dfrac{2(1) + 2}{1 + 2(2)} = -\dfrac{4}{5}$.
\[ \boxed{\frac{dy}{dx} = -\frac{2x + y}{x + 2y};\quad
   \left.\frac{dy}{dx}\right|_{(1,2)} = -\frac{4}{5}} \]}

\problem{\textbf{Folium: $x^{3} + y^{3} = 9xy$ at $(2, 4)$.}

\textbf{On the curve?} $8 + 64 = 72$ and $9(2)(4) = 72$ $\checkmark$.

\textbf{Differentiating.} The right side is a product, so both sides need care:
\[ 3x^{2} + 3y^{2}\,\frac{dy}{dx} = 9y + 9x\,\frac{dy}{dx}. \]
Collect: $\left(3y^{2} - 9x\right)\dfrac{dy}{dx} = 9y - 3x^{2}$, so
\[ \frac{dy}{dx} = \frac{9y - 3x^{2}}{3y^{2} - 9x} = \frac{3y - x^{2}}{y^{2} - 3x}. \]

\textbf{At the point.} $\dfrac{3(4) - 4}{16 - 6} = \dfrac{8}{10} = \dfrac{4}{5}$.
\[ \boxed{\frac{dy}{dx} = \frac{3y - x^{2}}{y^{2} - 3x};\quad
   \left.\frac{dy}{dx}\right|_{(2,4)} = \frac{4}{5}} \]}

\problem{\textbf{$x^{2}y + y^{3} = 10$ at $(1, 2)$.}

\textbf{On the curve?} $1 \cdot 2 + 8 = 10$ $\checkmark$.

\textbf{Differentiating.} $x^{2}y$ is a product \emph{and} contains $y$, so it needs
the product rule with a chain-rule factor on the second piece; $y^{3}$ needs the
power and chain rules:
\[ \left(2xy + x^{2}\,\frac{dy}{dx}\right) + 3y^{2}\,\frac{dy}{dx} = 0. \]
Collect: $\left(x^{2} + 3y^{2}\right)\dfrac{dy}{dx} = -2xy$, so
$\dfrac{dy}{dx} = -\dfrac{2xy}{x^{2} + 3y^{2}}$.

\textbf{At the point.} $-\dfrac{2(1)(2)}{1 + 3(4)} = -\dfrac{4}{13}$.
\[ \boxed{\frac{dy}{dx} = -\frac{2xy}{x^{2} + 3y^{2}};\quad
   \left.\frac{dy}{dx}\right|_{(1,2)} = -\frac{4}{13}} \]}

\problem{\textbf{Ellipse $x^{2} + 4y^{2} = 25$ at $(3, 2)$ --- tangent line.}

\textbf{On the curve?} $9 + 4(4) = 25$ $\checkmark$.

\textbf{Differentiating.} $2x + 8y\dfrac{dy}{dx} = 0$, so
$\dfrac{dy}{dx} = -\dfrac{2x}{8y} = -\dfrac{x}{4y}$.

\textbf{At the point.} $-\dfrac{3}{4(2)} = -\dfrac{3}{8}$.

\textbf{Tangent line.} Point-slope form through $(3, 2)$ with that slope:
\[ \boxed{\left.\frac{dy}{dx}\right|_{(3,2)} = -\frac{3}{8};\quad
   y = -\frac{3}{8}(x - 3) + 2} \]

\noindent\textit{Accept} the equivalent slope-intercept form
$y = -\tfrac{3}{8}x + \tfrac{25}{8}$.}

\problem{\textbf{Horizontal tangents on $x^{2} + xy + y^{2} = 12$.}

\textbf{Differentiating.} As in problem 3,
$\dfrac{dy}{dx} = -\dfrac{2x + y}{x + 2y}$.

\textbf{Setting the numerator to zero.} A horizontal tangent needs
$\dfrac{dy}{dx} = 0$, so $2x + y = 0$, that is $y = -2x$ --- provided the
denominator $x + 2y$ is not also zero there.

\textbf{Back into the relation.} Substituting $y = -2x$ into
$x^{2} + xy + y^{2} = 12$ gives
\[ x^{2} - 2x^{2} + 4x^{2} = 3x^{2} = 12 \quad\Longrightarrow\quad x = \pm 2. \]
With $y = -2x$ this gives the points $(2, -4)$ and $(-2, 4)$.

\textbf{Confirming.} At $(2, -4)$: $4 - 8 + 16 = 12$ $\checkmark$, and the
denominator is $2 + 2(-4) = -6 \neq 0$, so the tangent really is horizontal there.
The point $(-2, 4)$ checks the same way by symmetry.
\[ \boxed{(2,\,-4)\ \text{and}\ (-2,\,4)} \]}

\problem{\textbf{Challenge --- second derivative of $x^{2} + y^{2} = 25$.}

\textbf{Differentiating $\dfrac{dy}{dx} = -\dfrac{x}{y}$ again.} By the quotient
rule, treating $y$ as a function of $x$:
\[ \frac{d^{2}y}{dx^{2}} = -\frac{y \cdot 1 - x\,\dfrac{dy}{dx}}{y^{2}}. \]
Substitute $\dfrac{dy}{dx} = -\dfrac{x}{y}$:
\[ \frac{d^{2}y}{dx^{2}} = -\frac{y + \dfrac{x^{2}}{y}}{y^{2}}
   = -\frac{y^{2} + x^{2}}{y^{3}}. \]

\textbf{Using the relation.} On this curve $x^{2} + y^{2}$ is not just any
expression --- it equals 25. That substitution is what collapses the answer:
$\dfrac{d^{2}y}{dx^{2}} = -\dfrac{25}{y^{3}}$.

\textbf{In terms of $x$, and the concavity.} On the upper semicircle
$y = \sqrt{25 - x^{2}}$, so the second derivative is
$-25\left(25 - x^{2}\right)^{-3/2}$. There $y > 0$, so the value is negative
everywhere on that arc: the upper semicircle is concave down, which is what the
picture shows.
\[ \boxed{\frac{d^{2}y}{dx^{2}} = -\frac{25}{y^{3}}
   = -\frac{25}{(25 - x^{2})^{3/2}}} \]

\noindent\textit{Grading:} full credit requires substituting $\dfrac{dy}{dx}$ back
in and using $x^{2} + y^{2} = 25$; a student who stops at
$-\dfrac{y^{2} + x^{2}}{y^{3}}$ has done the calculus but not the simplification.}
