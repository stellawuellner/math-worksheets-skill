\sstitleblock{Diagnosing Quadrant and Parameter-Orientation Errors}

\skillheading{Skill 1 --- The arctangent only knows two quadrants}

\begin{formulabox}
\textbf{Rule:} $\tan\theta = y/x$ has two solutions per turn, but the $\arctan$ key returns
only the one between $-\pi/2$ and $\pi/2$. So: if $x < 0$, add $\pi$ to the calculator
value. If $x > 0$ and $y < 0$, add $2\pi$ to land in $[0, 2\pi)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\theta$ for the point $(-1, \sqrt{3})$.
\step{The $x$ is negative, so the direction is a left-hand one and the raw arctangent will be wrong by half a turn --- compute it, then correct it.}
\step{$\arctan\!\left(\dfrac{\sqrt{3}}{-1}\right) = -\dfrac{\pi}{3}$, and adding $\pi$ gives $-\dfrac{\pi}{3} + \pi$.}

$\Rightarrow\quad \theta = \ans{\tfrac{2\pi}{3}}$, a second-quadrant direction, as the signs demand.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\theta$ in $[0, 2\pi)$ for the point $(-4, -4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\theta = \ans{\tfrac{5\pi}{4}}$}
\end{tryitbox}

\skillheading{Skill 2 --- A negative $r$ reverses the ray}

\begin{formulabox}
\textbf{Rule:} $x = r\cos\theta$ and $y = r\sin\theta$ work with the \emph{signed} $r$.
Keep the minus sign in the multiplication; do not convert with $|r|$ and patch the quadrant
afterwards. A negative $r$ puts the point half a turn from the direction $\theta$ names.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $x$ for the polar point $\left(-6, \dfrac{\pi}{3}\right)$.
\step{The first coordinate is negative, so the only safe move is to substitute it signed and let the arithmetic place the point.}
\step{$x = (-6)\cos\dfrac{\pi}{3} = (-6)\left(\dfrac{1}{2}\right)$.}

$\Rightarrow\quad x = \ans{-3}$, so the point sits to the left of the pole, not the right.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $x$ for the polar point $\left(-10, \dfrac{2\pi}{3}\right)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Orientation lives in the parameter}

\begin{formulabox}
\textbf{Rule:} the rectangular equation says which \emph{points} are on the curve; only the
parameter says in which \emph{order}. Test the orientation by evaluating two nearby values of
$t$ and watching which way the point moves. A minus sign on the sine reverses a circle from
counterclockwise to clockwise without changing the rectangular equation at all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $x = 3\cos t$, \ $y = -3\sin t$, find $y$ at $t = \dfrac{\pi}{2}$.
\step{The curve starts at $(3, 0)$; one sample a quarter turn later settles whether it went up or down.}
\step{$y = -3\sin\dfrac{\pi}{2} = -3(1)$.}

$\Rightarrow\quad y = \ans{-3}$: the point moved \emph{down} to $(0, -3)$, so the circle runs clockwise.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $x = 5\cos t$, \ $y = -5\sin t$, find $y$ at $t = \dfrac{\pi}{6}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{-\tfrac{5}{2}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Losing part of a curve to elimination}

\begin{formulabox}
\textbf{Rule:} after eliminating $t$, ask what values $x$ and $y$ could actually take.
A square root forces $x \ge 0$; a sine or cosine forces a bounded range; a restricted
$t$-interval clips the ends. Carry the restriction with the equation, or the drawn graph will
contain points the parametric curve never visits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $x = \sqrt{t}$, \ $y = t + 1$, \ $t \ge 0$, find $y$ when $x = 3$.
\step{Eliminating gives $y = x^2 + 1$, and because $x$ is a square root only $x \ge 0$ counts --- so evaluating at a positive $x$ is legitimate.}
\step{$y = 3^2 + 1$.}

$\Rightarrow\quad y = \ans{10}$; the curve is the right half of that parabola only.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $x = \sqrt{t}$, \ $y = t - 5$, find $y$ when $x = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{11}$}
\end{tryitbox}
