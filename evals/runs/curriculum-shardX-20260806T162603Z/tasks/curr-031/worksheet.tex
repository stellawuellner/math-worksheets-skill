\wstitleblock{Teen Numbers: One Ten and Some Ones}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A full ten-frame holds exactly one ten. The dots outside the frame are the ones.
  \item Every teen number is one ten and some ones: \quad \textbf{teen number} $=$ ten $+$ ones.
  \item Use the frames on this page to check your thinking. You may draw dots in the empty boxes.
\end{itemize}
