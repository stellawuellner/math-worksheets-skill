\sstitleblock{Transformations with Multiple Parameters}

\begin{formulabox}
\textbf{The parameter map:} \quad
$g(x) = a\, f\bigl(b(x - h)\bigr) + k$\par\vspace{2pt}
\begin{tabular}{@{}llll@{}}
$a$ & \textbf{outside} & vertical stretch by $|a|$; reflect over the $x$-axis if $a < 0$ & does what it says\\
$k$ & \textbf{outside} & shift up $k$ & does what it says\\
$b$ & \textbf{inside} & horizontal stretch by $1/b$ & does the opposite\\
$h$ & \textbf{inside} & shift right $h$ & does the opposite\\
\end{tabular}\par\vspace{2pt}
\textbf{Factor first.} $h$ can only be read once the inside is written as $b(x-h)$:
$f(2x - 6) = f\bigl(2(x-3)\bigr)$ shifts 3, not 6.
\end{formulabox}

\skillheading{Skill 1 --- The input chain (table-transform)}

\begin{formulabox}
\textbf{Order of operations on a single input:} \ $x \to (x - h) \to b(x-h) \to f(\ldots)
\to a \cdot f(\ldots) \to a \cdot f(\ldots) + k$.\par
Build a column for each arrow. The parent function runs in the \emph{middle} --- everything
to its left is an input move, everything to its right is an output move.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^2$, \ $g(x) = 3f(x + 1) - 2$. Find $g(1)$.
\step{Two parameters act, one inside and one outside, so I walk the input through the chain in order rather than guessing at the end.}
\step{Input move: $1 + 1 = 2$. \ Parent: $f(2) = 4$. \ Output moves: $3(4) = 12$, then $12 - 2 = 10$.}
$\Rightarrow\quad \ans{g(1) = 10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^2$, \ $h(x) = -f(x - 2) + 6$. Find $h(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h(5) = -3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Factor before you read (parameters-from-equation)}

\begin{formulabox}
\textbf{Rule:} factor $b$ out of the \emph{entire} inside before naming the shift ---
$bx + c = b\left(x + \frac{c}{b}\right)$, so the shift is $c/b$, not $c$.\par
For $y = A\sin\bigl(B(x - C)\bigr) + D$: amplitude $|A|$, period $2\pi/B$, phase shift $C$,
midline $y = D$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $y = 2\sin\left(3x + \frac{\pi}{2}\right)$ in the form $A\sin\bigl(B(x - C)\bigr)$.
\step{The shift is unreadable while $3$ still multiplies only part of the inside, so the first move is to factor $B = 3$ out of everything.}
\step{$3x + \frac{\pi}{2} = 3\left(x + \frac{\pi}{6}\right)$, since $3 \cdot \frac{\pi}{6} = \frac{\pi}{2}$. So $A = 2$, $B = 3$, $C = -\frac{\pi}{6}$: shifted $\frac{\pi}{6}$ \emph{left}, not $\frac{\pi}{2}$.}
$\Rightarrow\quad \ans{2\sin\left(3\left(x + \tfrac{\pi}{6}\right)\right)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $y = 5\cos\left(4x - \pi\right)$ in the form $A\cos\bigl(B(x - C)\bigr)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5\cos\left(4\left(x - \tfrac{\pi}{4}\right)\right)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Composition order (composition-order)}

\begin{formulabox}
\textbf{Rule:} in $(u \circ v)(x) = u(v(x))$ the \emph{inner} map runs first. Swapping the
order changes the function, because a stretch applied after a shift scales that shift, while
a stretch applied before it does not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $s(x) = x + 5$ (shift), $d(x) = 3x$ (stretch). Simplify $(d \circ s)(x)$.
\step{The inner map $s$ runs first, so the shift happens and the stretch then acts on the already-shifted value.}
\step{$d(s(x)) = d(x + 5) = 3(x + 5)$. Distributing shows the shift was scaled: the graph moves 15, not 5.}
$\Rightarrow\quad \ans{3x + 15}$ \quad (whereas $(s \circ d)(x) = 3x + 5$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $t(x) = x - 2$, \ $w(x) = 4x$. Simplify $(w \circ t)(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x - 8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Reading a shift straight off an unfactored inside is the most common
multi-parameter error --- always divide the constant by $b$ (or factor, which does it for
you). And when you undo parameters to solve an equation, peel them off from the
\emph{outside in}: $k$ first, then $a$, then the parent, then $b$ and $h$.
\end{watchoutbox}
