\sstitleblock{Finding a Percent of a Quantity}

% =================== SKILL 1 ===================
\skillheading{Reading an Amount off a Hundredths Grid}

\begin{formulabox}
\textbf{Rule:} ``Percent'' means \emph{per one hundred}. A hundredths grid cuts
the whole quantity into 100 equal parts, so one part is $1\%$ of it.
\ \textbf{one part} $=$ whole $\div\ 100$, and \textbf{$p$ percent of the whole}
$= p \times$ (one part) $= \dfrac{p}{100} \times$ whole.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A hundredths grid stands for a jar of 300 marbles, and
40 parts are shaded. How many marbles are shaded?
\step{The grid always has 100 parts, so first find what one part is worth ---
that turns a percent into a count.}
\step{One part is $300 \div 100 = $ $\ans{3}$ marbles, so 40 parts are
$40 \times 3 = $ $\ans{120}$ marbles.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A hundredths grid stands for 250 stickers and 60 parts
are shaded. How many stickers are shaded?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{150}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Building a Percent from Ten-Percent Strips}

\begin{formulabox}
\textbf{Rule:} Cut a tape of the whole amount into 10 equal strips. One strip is
$10\%$ (whole $\div\ 10$); half a strip is $5\%$; a tenth of a strip is $1\%$.
Any friendly percent is a sum of strips: $35\% = 10\% + 10\% + 10\% + 5\%$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $35\%$ of 80 dollars without multiplying decimals.
\step{Thirty-five percent is not a single strip, so build it: three whole
ten-percent strips plus one half strip.}
\step{One strip is $80 \div 10 = $ $\ans{8}$ dollars and half a strip is
$\ans{4}$ dollars, so $35\%$ is $8 + 8 + 8 + 4 = $ $\ans{28}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $45\%$ of 60 dollars with ten-percent
strips.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Multiplying by the Decimal Form}

\begin{formulabox}
\textbf{Rule:} Divide the percent by 100 to get its decimal form --- the decimal
point moves two places to the left --- then multiply.
\ \textbf{$p$ percent of $q$} $= (p \div 100) \times q$. This is the fastest
route when the percent is not friendly, such as $7.5\%$ or $23\%$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $24\%$ of 150 people.
\step{Twenty-four percent is not a whole number of ten-percent strips, so
convert to a decimal and multiply.}
\step{$24 \div 100 = $ $\ans{0.24}$, and $0.24 \times 150 = $ $\ans{36}$
people.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $6\%$ of 450 kilometres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $8\%$ is $0.08$, not $0.8$. Moving the decimal point
only one place turns eight percent into eighty percent, and the answer comes out
ten times too big.
\end{watchoutbox}
