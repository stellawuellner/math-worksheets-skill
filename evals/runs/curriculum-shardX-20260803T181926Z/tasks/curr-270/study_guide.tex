\sstitleblock{Function Notation, Domain, and Range}

\begin{formulabox}
\textbf{One function, four faces.} A rule, a table, a graph and a story can all
describe the same function. Whichever you are handed, the notation is the
same: $f(\text{input}) = \text{output}$.\\[3pt]
\textbf{Rule:} $f(x) = 4x - 7$ \quad $\bullet$ \quad \textbf{Table:} an input
row and an output row \quad $\bullet$ \quad \textbf{Graph:} the point $(a, b)$
means $f(a) = b$\\[3pt]
\textbf{Domain} $=$ the inputs you are allowed to use. \textbf{Range} $=$ the
outputs they produce.
\end{formulabox}

\skillheading{1. Evaluating a rule at an input}

\begin{formulabox}
$f(3)$ is an instruction, not a product: write the rule again with the input in
place of every $x$, then simplify. Put a negative input in parentheses first
--- $(-2)^2 = 4$, while $-2^2 = -4$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 5x + 1$. Find $f(-2)$.
\step{The input is given and the output is wanted, so this is a substitution
--- no equation to solve.}
\step{$f(-2) = 5(-2) + 1 = -10 + 1$}
$\Rightarrow\quad \ans{-9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 5x + 1$. Find $f(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$}
\end{tryitbox}

\skillheading{2. Finding the input for an output}

\begin{formulabox}
When the \emph{output} is the number you are given, set the rule equal to it
and solve for the input. Evaluating and solving are the same rule run in
opposite directions, so each one checks the other.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 5x + 1$. For what $x$ is $f(x) = 26$?
\step{The $26$ is an output, so it belongs on the right of an equation rather
than inside the rule: $5x + 1 = 26$.}
\step{Subtract $1$: \ $5x = 25$. \ Divide by $5$: \ $x = 5$.}
$\Rightarrow\quad \ans{x = 5}$ \quad (check: $5(5) + 1 = 26$ $\checkmark$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 5x + 1$. For what $x$ is $f(x) = -14$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -3}$}
\end{tryitbox}

\skillheading{3. Reading a table or a graph}

\begin{formulabox}
\textbf{Graph:} to find $f(a)$, go across to $a$ on the input axis, then
straight to the plotted point and read the output axis. To find the input for a
known output, go the other way round.\\[3pt]
\textbf{Table:} if the outputs rise by the same amount every time the input
rises by $1$, the function is linear. That constant rise is the multiplier, and
one row then pins down the constant term.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The table gives $q(x)$. Find a rule, then find $q(7)$.
\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|}\hline
$x$ & $1$ & $2$ & $3$ \\ \hline
$q(x)$ & $5$ & $8$ & $11$ \\ \hline
\end{tabular}
\end{center}
\step{The table stops at $x = 3$ but the question asks about $7$, so a rule is
needed --- reading the table cannot answer it.}
\step{The outputs climb by $3$ each step, so $q(x) = 3x + b$; using $x = 1$
gives $5 = 3 + b$, so $b = 2$ and $q(x) = 3x + 2$.}

$\Rightarrow\quad q(7) = 3(7) + 2 = \ans{23}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $r(x)$ is $9$, $13$, $17$ at $x = 1$, $2$, $3$. Find
$r(6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{29}$}
\end{tryitbox}

\skillheading{4. Domain and range in a story}

\begin{formulabox}
A bare rule usually accepts every number. A \emph{story} does not: time starts
at $0$, a tank stops when it is empty, tickets come in whole numbers, an order
has a limit. Find where the story ends by setting the rule equal to the value
that ends it, then solving.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tank holds $W(t) = 300 - 20t$ litres $t$ minutes
after the valve opens, and the rule stops describing it once the tank is empty.
Where does the domain end?
\step{The algebra would run forever, so the story sets the limit: empty means
an output of zero.}
\step{$300 - 20t = 0$, so $20t = 300$ and $t = 15$.}

$\Rightarrow\quad \ans{t = 15}$, so a sensible domain is $0 \le t \le 15$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $B(t) = 250 - 25t$ litres, same story. Where does the
domain end?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f(3)$ never means $f \times 3$. And two inputs may
share one output (the drone is at $40$ feet twice), but one input may never
have two outputs --- that is what stops a picture from being a function.
\end{watchoutbox}
