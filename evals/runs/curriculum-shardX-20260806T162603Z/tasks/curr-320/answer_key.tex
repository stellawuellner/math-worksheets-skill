\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 5$ cm, $BC = 7$ cm and
$DE = 15$ cm. Find $EF$, in centimetres.}

\noindent\textbf{Solution.} $\overline{AB}$ matches $\overline{DE}$ and
$\overline{BC}$ matches $\overline{EF}$, so
$\frac{5}{15} = \frac{7}{EF}$, giving $EF = 7 \cdot \frac{15}{5}$.
Therefore \ans{EF = 21 \text{ cm}}

\problem{$\triangle PQR \sim \triangle STU$ with $PQ = 12$ in, $PR = 21$ in and
$ST = 8$ in. Find $SU$, in inches.}

\noindent\textbf{Solution.} $\overline{PR}$ matches $\overline{SU}$, so
$\frac{12}{8} = \frac{21}{SU}$ and $SU = 21 \cdot \frac{8}{12}$.
Therefore \ans{SU = 14 \text{ in}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Two similar triangular road signs: the smaller has base $6$ cm, the
larger has base $15$ cm. Find the scale factor from smaller to larger.}

\noindent\textbf{Solution.} The scale factor is the target length over the
starting length:
\[ k = \frac{15}{6} . \]
So \ans{k = 2.5}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 8$ m, $DE = 12$ m, and
perimeter of $\triangle ABC$ equal to $30$ m. Find the perimeter of
$\triangle DEF$, in metres.}

\noindent\textbf{Solution.} From one pair of matching sides,
$k = \frac{12}{8} = 1.5$. Perimeter is a sum of lengths, so it scales by the
same $k$, not by $k^2$:
\[ P_{DEF} = 30 \cdot \frac{12}{8} . \]
Therefore \ans{P_{DEF} = 45 \text{ m}}

\problem{Two similar tiles have corresponding sides $6$ cm and $10$ cm; the
smaller has area $27$ cm$^2$. Find the area of the larger, in square
centimetres.}

\noindent\textbf{Solution.} Side ratio $k = \frac{10}{6}$, and area scales by
$k^2$:
\[ A = 27 \left( \frac{10}{6} \right)^2 = 27 \cdot \frac{100}{36} . \]
Therefore \ans{A = 75 \text{ cm}^2}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\triangle ABC \sim \triangle DEF$ with perimeters $24$ cm and $60$ cm.
If $AB = 7$ cm, find $DE$, in centimetres.}

\noindent\textbf{Solution.} Perimeters scale by the same factor as sides, so
$k = \frac{60}{24} = 2.5$ and
\[ DE = 7 \cdot \frac{60}{24} . \]
Therefore \ans{DE = 17.5 \text{ cm}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Two similar triangular plots have areas $48$ m$^2$ and $108$ m$^2$.
Find the ratio of a side of the larger to the matching side of the smaller.}

\noindent\textbf{Solution.} Areas scale by $k^2$, so
$k^2 = \frac{108}{48} = 2.25$ and
\[ k = \sqrt{\frac{108}{48}} . \]
Therefore \ans{k = 1.5}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Two similar sails have areas $50$ cm$^2$ and $128$ cm$^2$; the smaller
has perimeter $30$ cm. (a) Find the perimeter of the larger, in centimetres.
(b) Justify why the perimeter ratio is the square root of the area ratio.}

\noindent\textbf{Solution (a).} The area ratio is
$\frac{128}{50} = 2.56 = k^2$, so $k = \sqrt{2.56} = 1.6$ and
\[ P = 30 \sqrt{\frac{128}{50}} . \]
Therefore \ans{P = 48 \text{ cm}}

\noindent\textbf{Solution (b) --- instructor-judged.} A complete response says
areas of similar figures grow as the \emph{square} of the side ratio, so
recovering the side ratio --- which is what perimeters follow --- means undoing
that square with a square root. Using the area ratio directly would scale a
length by a factor that belongs to a region. Award full credit only when the
squaring relationship is named.

\problem{A triangular window is enlarged so that its area becomes $6.25$ times
the original. (a) Find the scale factor applied to each side. (b) The original
perimeter is $84$ cm; find the new perimeter, in centimetres.}

\noindent\textbf{Solution (a).} Area scales by $k^2$, so $k^2 = 6.25$ and
$k = \sqrt{6.25}$, giving \ans{k = 2.5}

\noindent\textbf{Solution (b).} Perimeter scales by $k$, not by $k^2$:
\[ P = 84 \sqrt{6.25} = 84 \cdot 2.5 . \]
Therefore \ans{P = 210 \text{ cm}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 14$ cm, $DE = 21$ cm, and
area of $\triangle ABC$ equal to $56$ cm$^2$. (a) Find the area of
$\triangle DEF$, in square centimetres. (b) Explain why multiplying every side
by the same factor multiplies the area by the square of that factor.}

\noindent\textbf{Solution (a).} $k = \frac{21}{14} = 1.5$, and area scales by
$k^2 = 2.25$:
\[ A = 56 \left( \frac{21}{14} \right)^2 . \]
Therefore \ans{A = 126 \text{ cm}^2}

\noindent\textbf{Solution (b) --- instructor-judged.} A complete response says a
scaled triangle is stretched in two independent directions at once: the base and
the height are each multiplied by the factor, and the area, being half their
product, picks the factor up twice --- once from the base and once from the
height. Award full credit only when the response ties the squaring to the two
dimensions rather than simply quoting the rule.
