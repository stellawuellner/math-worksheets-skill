\wstitleblock{Equivalent Expressions: Expanded \& Factored Forms}{Grade 6--7 Mathematics}{}
\begin{center}
\textit{expanding, factoring, and combining like terms to write equivalent expressions}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item After the warm-up the problems are mixed: read each one and decide whether
  to expand, factor, or combine before you start writing.
  \item Answers are expressions --- keep the variable, and write factored answers
  with the greatest common factor outside.
\end{itemize}

\problem[2.8cm]{Expand: \quad $3(x + 7)$}

\problem[2.8cm]{Expand: \quad $5(2x + 3)$}

\problem[2.8cm]{Factor out the greatest common factor: \quad $4x + 12$}

\problem[3.2cm]{Combine like terms: \quad $8x + 2 - 3x + 5$}

\problem[3.2cm]{Factor out the greatest common factor: \quad $14x + 21$}

\problem[3.2cm]{Expand: \quad $-4(x - 6)$}

\problem[3.2cm]{Combine like terms: \quad $2x + 9 + 4x - 9$}

\problem[3.4cm]{Factor out the greatest common factor: \quad $18x + 24$}

\problem{Consider the two expressions $2(3x + 1)$ and $6x + 2$.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Evaluate $2(3x + 1)$ when $x = 4$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(b)] Evaluate $6x + 2$ when $x = 4$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(c)] Say whether the two expressions are equivalent for \emph{every} value
  of $x$, and how you know.\par\vspace*{1.5cm}
\end{itemize}}

\problem{A rectangular banner has area $(12x + 30)$ square centimeters. Its height
is 6 cm. Write and simplify an expression, in centimeters, for the width of the
banner.\par\vspace*{3.8cm}\answerline{cm}}

\problem[4.2cm]{Simplify completely: \quad $4(2x + 3) - 5x$}

\problem{\textbf{Challenge.} Consider the expression $6(x + 2) + 3x + 6$.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Write it in simplest \emph{expanded} form.\par\vspace*{1.8cm}\ansline
  \item[(b)] Now \emph{factor} your answer from part (a).\par\vspace*{1.8cm}\ansline
\end{itemize}}
