\sstitleblock{Coordinate Plane, Line Plots, and Numerical Patterns}

\skillheading{Skill 1 --- Two patterns make ordered pairs}

\begin{formulabox}
\textbf{Rule:} run both patterns from the same starting point, then pair \emph{matching
terms} --- 1st with 1st, 2nd with 2nd. Write each pair as $(x, y)$ and plot it: across
first, then up.\par
\smallskip
When both patterns start at zero, the $y$-value is always the same multiple of the
$x$-value, and that multiplier is
\[ \text{multiplier} = \frac{\text{step size of the } y \text{ pattern}}
                            {\text{step size of the } x \text{ pattern}} \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Pattern M adds 3 each time, Pattern N adds 6 each time, both from
zero. Find the $N$-value paired with the $M$-value 8.
\step{Both patterns start together, so matching terms grow in a fixed ratio --- find the
multiplier from the two step sizes instead of writing out the lists.}
\step{Multiplier $= 6 \div 3 = 2$, so $N = 2 \times 8 = 16$, and the ordered pair is
$(8, 16)$.}
$\Rightarrow\quad N = \ans{16}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Pattern M adds 2 each time, Pattern N adds 10 each time, both from
zero. Find the $N$-value paired with the $M$-value 7.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

\skillheading{Skill 2 --- Summarising a line plot}

\begin{formulabox}
\textbf{Read the plot twice.} The number \emph{under} a stack is the value; the number of
$\times$ marks \emph{in} the stack is how many times that value happened.\par
\smallskip
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Total} $=$ add (value $\times$ stack height) for every stack.
  \item \textbf{Mean} $=$ total $\div$ number of marks (not number of stacks).
  \item \textbf{Median} $=$ middle value, counting \emph{marks} from the left.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line plot shows six values: 1, 2, 3, 3, 4, 5. Find the mean.
\step{A mean needs a total and a count, and the plot gives both --- add the values, then
divide by how many marks there are.}
\step{$1 + 2 + 3 + 3 + 4 + 5 = 18$, and there are six marks, so the mean is
$18 \div 6 = 3$.}
$\Rightarrow\quad \text{mean} = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line plot shows six values: 2, 2, 4, 4, 5, 7. Find the
mean.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Running a rule on data you read}

\begin{formulabox}
\textbf{Two-piece problems:} one piece is a number you must \emph{read or compute} from
the plot or table; the other is a \emph{rule} stated in words.\par
\smallskip
Write the read value down on its own line first, then apply the rule to it. ``Points for
every book'' multiplies; ``beyond'' or ``more than'' subtracts first; ``each value goes up
by'' shifts the mean by that same amount.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A class read 30 books in all. The club gives 4 points for every
book. How many points did the class earn?
\step{The plot supplies the book total and the rule supplies the points per book --- write
the read value first, then apply the rule to it.}
\step{Read value: 30 books. Rule: 4 points a book, so $4 \times 30 = 120$.}
$\Rightarrow\quad \ans{120}$ points
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A class read 25 books in all, and the club gives 3 points for every
book. How many points did the class earn?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} on a line plot the $\times$ marks count \emph{how many people},
never \emph{how much}. Counting marks answers ``how many students''; adding value times
stack height answers ``how many books''. And when pairing two patterns, always pair terms
in the \emph{same position} --- one step out of step is the most common wrong ordered
pair there is.
\end{watchoutbox}
