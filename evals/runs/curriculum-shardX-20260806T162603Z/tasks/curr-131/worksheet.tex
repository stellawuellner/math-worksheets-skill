\begin{center}
\textit{Unit fractions of a number, fractions of a set, and finding the whole}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item ``$\tfrac{1}{4}$ of 12'' means: split 12 into 4 equal groups and take
        1 group. Drawing equal groups is a great way to check your thinking.
  \item Show how you split the number into equal groups in your work.
\end{itemize}

equal groups to help you find: $\tfrac{1}{4}$ of 12

\problem[3.5cm]{Find: $\tfrac{1}{5}$ of 30}

\problem[4cm]{Use equal groups to find two thirds of 18.
\par\vspace{2pt}

\par\vspace{2pt}
(b) $\tfrac{2}{3}$ of 18 $=$ \ansblank}

\problem[4cm]{Kofi has a sheet of 20 stickers. He gives $\tfrac{3}{5}$ of the
stickers to his sister. How many stickers does he give away?}

\problem[4cm]{A book has 56 pages. Lena reads $\tfrac{1}{8}$ of the book
before dinner. How many pages does she read?}

\problem[4.5cm]{There are 24 students in a class, and $\tfrac{3}{8}$ of the
class walks to school. Picture the class split into 8 equal groups.
\par\vspace{2pt}
(a) How many students are in $\tfrac{1}{8}$ of the class? \ansblank
\par\vspace{2pt}
(b) How many students walk to school? \ansblank}

\problem[4.5cm]{One fourth of Maya's marbles is 6 marbles. How many marbles
does Maya have in all? \emph{Hint: draw 4 equal groups and put 6 in one of
them.}}

\problem[4.5cm]{A farm stand has 42 apples, and $\tfrac{5}{6}$ of them are
red. How many apples are red?}

\problem{A gift card holds 20 dollars. Nia spends $\tfrac{5}{8}$ of the money
on a book. How many dollars does she spend?
\par\vspace*{5cm}\answerline{dollars}}

\problem[5.5cm]{\textbf{Challenge.} In a small theater, $\tfrac{3}{4}$ of the
seats are filled, and 21 seats are filled.
\par\vspace{2pt}
(a) How many seats does the theater have? \ansblank
\par\vspace{2pt}
(b) Explain how a drawing of equal groups shows your answer is right.}
