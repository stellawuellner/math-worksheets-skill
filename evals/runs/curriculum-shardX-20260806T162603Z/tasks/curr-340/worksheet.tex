\noindent\rule{\linewidth}{0.4pt}

\problem[4.4cm]{An inscribed angle of a circle intercepts an arc measuring $86^\circ$. Find the measure of the inscribed angle.}

\problem[4.4cm]{Inscribed angle $\angle ABC$ measures $37^\circ$ and intercepts arc $AC$. Find the measure of arc $AC$.}

\problem{A sector of a circle of radius $9$ cm has a central angle of $40^\circ$. Find the length of the sector's arc, to the nearest hundredth.
\par\vspace*{4.0cm}\answerline{cm}}

\problem[4.4cm]{Chords $\overline{AB}$ and $\overline{CD}$ of a circle intersect inside the circle at $E$. Given $AE = 6$, $EB = 10$ and $CE = 4$, find $ED$.}

\problem{A circular fountain on a park map (axes in metres) has the equation $(x - 5)^2 + (y + 2)^2 = 49$. A straight path runs tangent to the fountain at the point of the fountain's edge that lies directly above the centre.
\par\vspace{0.15cm}
(a) Find the radius of the fountain.
\par\vspace*{2.2cm}\answerline{m}
\par\vspace{0.25cm}
(b) Find the $y$-coordinate of the point where the path touches the fountain. \quad \ansblank
\par\vspace*{2.4cm}}

\problem{Quadrilateral $ABCD$ is inscribed in a circle, with $m\angle A = (2x + 15)^\circ$ and $m\angle C = (3x + 40)^\circ$.
\par\vspace{0.15cm}
(a) Find $x$. \quad \ansblank
\par\vspace*{2.6cm}
(b) Find $m\angle A$. \quad \ansblank
\par\vspace*{2.6cm}}

\problem{A sector of a circle of radius $12$ cm has a central angle of $120^\circ$. Find the area of the sector, to the nearest hundredth.
\par\vspace*{4.0cm}\answerline{cm$^2$}}

\problem[5.0cm]{From an external point $P$, a tangent segment touches a circle at $T$ with $PT = 12$, and a secant from $P$ cuts the circle at $A$ then $B$, with the external part $PA = 8$. Find the whole secant length $PB$.}

\problem{In a circle of radius $10$ cm, minor arc $AB$ measures $72^\circ$.
\par\vspace{0.15cm}
(a) Find the length of minor arc $AB$, to the nearest hundredth.
\par\vspace*{2.4cm}\answerline{cm}
\par\vspace{0.25cm}
(b) Find the area of sector $AOB$, to the nearest hundredth.
\par\vspace*{2.4cm}\answerline{cm$^2$}}

\problem{A circle has equation $x^2 + y^2 - 10x + 4y - 71 = 0$, with $x$ and $y$ in units on a coordinate grid.
\par\vspace{0.15cm}
(a) Rewrite the equation in centre-radius form and give the radius. \quad \ansblank
\par\vspace*{2.8cm}
(b) A chord of this circle is $16$ units long. Find the distance from the centre to that chord. \quad \ansblank
\par\vspace*{2.6cm}
(c) Explain why this circle cannot contain a chord $24$ units long.
\par\vspace*{3.0cm}}
