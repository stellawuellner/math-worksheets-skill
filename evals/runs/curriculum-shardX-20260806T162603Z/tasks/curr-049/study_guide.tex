\sstitleblock{Halves and Fourths That Really Are Equal}

\skillheading{Skill 1 --- Fixing two parts that are called halves}

\begin{formulabox}
\textbf{Rule:} \emph{Halves} means \textbf{two parts of the same size}. Two pieces alone are not enough --- they must match.\par
To check a strip of equal squares: count the squares in each part. If the counts differ, the parts are not halves.\par
To fix it: share all the squares equally between the two parts, \quad each half $=$ whole $\div$ two.
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \draw[thick] (0,0) rectangle (8,1);
  \foreach \x in {1,...,7} { \draw[gray!55, thin] (\x,0) -- (\x,1); }
  \draw[very thick] (4,0) -- (4,1);
  \node[below] at (2,-0.05) {equal};
  \node[below] at (6,-0.05) {equal};
\end{tikzpicture}
\end{center}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A strip of $10$ squares is cut into a part of $7$ squares and a part of $3$. Are they halves? If not, fix them.
\step{Halves have to be the same size, so before anything else I compare the two counts instead of trusting the label.}
\step{Seven squares against three squares is not a match, so these are not halves. Sharing all ten squares equally between two parts gives each part $10 \div 2$ squares, so the cut belongs two squares further along.}

not halves: $\ans{7 > 3}$ \qquad each true half: $\ans{5}$ squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A strip of $6$ squares is cut into a part of $4$ and a part of $2$. Are they halves? How many squares should each half have?\\[2pt]
\rotatebox{180}{\footnotesize check: not halves, $\ans{4 > 2}$; each true half $\ans{3}$ squares}
\end{tryitbox}

\skillheading{Skill 2 --- Fixing four parts that are called fourths}

\begin{formulabox}
\textbf{Rule:} \emph{Fourths} means \textbf{four parts of the same size}. Four pieces alone are not enough --- all four must match.\par
Check the biggest part against the smallest: if those two differ, the parts are not fourths.\par
To fix it: \quad each fourth $=$ whole $\div$ four.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A strip of $12$ squares is cut into parts of $6$, $2$, $2$ and $2$ squares, and called fourths. Are they?
\step{All four parts have to be the same size, so the quickest test is to set the biggest part against the smallest and see whether even those two match.}
\step{Six squares against two squares is not a match, so these are not fourths. Sharing all twelve squares among four equal parts gives $12 \div 4$ squares each, so the big part must give squares back to the others.}

not fourths: $\ans{6 > 2}$ \qquad each true fourth: $\ans{3}$ squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A strip of $8$ squares is cut into parts of $5$, $1$, $1$ and $1$, and called fourths. Are they? How many squares should each fourth have?\\[2pt]
\rotatebox{180}{\footnotesize check: not fourths, $\ans{5 > 1}$; each true fourth $\ans{2}$ squares}
\end{tryitbox}

\skillheading{Skill 3 --- Halves are bigger than fourths}

\begin{formulabox}
\textbf{Rule:} The more equal parts a whole is shared into, the \emph{smaller} each part gets --- the whole never grows.\par
So a half of a strip is always bigger than a fourth of that same strip.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A strip of $8$ squares is shared into equal halves, and a strip just like it into equal fourths. How big is each part?
\step{Both strips hold the same amount, so the only thing that changes is how many equal parts it is shared into --- that tells me to share the same total two ways and then four ways.}
\step{Shared between two: $8 \div 2$ squares each. Shared among four: $8 \div 4$ squares each. The same strip, more parts, smaller parts.}

each half: $\ans{4}$ squares \qquad each fourth: $\ans{2}$ squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A strip of $16$ squares is shared into equal halves, and another into equal fourths. How big is each part?\\[2pt]
\rotatebox{180}{\footnotesize check: each half $\ans{8}$ squares, each fourth $\ans{4}$ squares}
\end{tryitbox}
