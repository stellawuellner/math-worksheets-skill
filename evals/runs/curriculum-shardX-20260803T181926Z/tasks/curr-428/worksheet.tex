\begin{center}\small\textit{Testing an SSA measurement for how many triangles it
allows, solving both triangles when there are two, and showing that the second
candidate fails when there is only one}\end{center}

\noindent\textbf{Why SSA is different.} Two sides and a non-included angle do
not always pin down a triangle. With angle $A$, its opposite side $a$, and the
adjacent side $b$ known, the side $a$ swings from the far vertex like a hinge and
may reach the base line twice, once, or not at all.
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item Compute the height $h = b\sin A$ first. Then:
        $a < h$ gives \textbf{no} triangle; $a = h$ gives \textbf{one} (right
        angle at $B$); $h < a < b$ gives \textbf{two}; $a \ge b$ gives
        \textbf{one}.
  \item The law of sines returns only the acute $B$. The supplement
        $180^\circ - B$ has the same sine, so \emph{always} test it: it is a
        second triangle exactly when $A + (180^\circ - B) < 180^\circ$.
  \item Every figure is drawn to scale from the given data, with the requested
        quantity marked $?$. A dashed side marks the second position of the
        swinging side. Round angles and lengths to two decimal places, and keep
        full precision until the last step.
\end{itemize}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.5cm]{\textbf{Case test.} Here $A = 30^\circ$, $b = 10$ and $a = 6$.

\textbf{(a)} Compute the height $h = b\sin A$. \quad \textbf{(b)} Compare $h$,
$a$ and $b$, say how many triangles this measurement allows, and find the acute
angle $B$.
\probfig{1}}

\problem[5cm]{\textbf{Two sightings.} A surveyor records $A = 40^\circ$,
$b = 12$ and $a = 9$.

\textbf{(a)} Compute $h$, state how many triangles are possible, and find the
acute $B$. \quad \textbf{(b)} A second sighting reports the same $A$ and $b$ but
$a = 7$. Compute the value the law of sines gives for $\sin B$, and explain what
that value tells you about this measurement.
\probfig{2}}

\problem[6cm]{\textbf{Both triangles.} Here $A = 35^\circ$, $b = 14$ and
$a = 9$.

\textbf{(a)} Find both possible values of $B$, and show that each leaves room
for a third angle. \quad \textbf{(b)} Find side $c$ for \emph{each} triangle.
Label your answers Triangle 1 (acute $B$) and Triangle 2 (obtuse $B$).
\probfig{3}}

\problem[5.5cm]{\textbf{Only one.} Here $A = 48^\circ$, $b = 7$ and $a = 10$.

\textbf{(a)} Find the acute $B$. \quad \textbf{(b)} Compute
$A + (180^\circ - B)$ and use it to show the obtuse candidate is impossible.
\quad \textbf{(c)} Find side $c$ for the one triangle that exists.
\probfig{4}}

\problem{\textbf{Search and rescue.} A rescue boat at $A$ picks up a signal
$b = 40$ km away at $C$ and a second station $B$ lies on a straight shoreline
from $A$. The bearing at $A$ gives the angle $A = 33^\circ$, and the distance
from $B$ to the signal is $a = 25$ km.

\textbf{(a)} Find both possible values of the angle at $B$. \quad
\textbf{(b)} Find the shoreline distance $c$ from $A$ to $B$ for each case, in
kilometres. Give the far position first.
\probfig{5}
\par\vspace*{6cm}\answerline{km}}

\problem[5cm]{\textbf{The borderline.} Here $A = 30^\circ$, $b = 12$ and
$a = 6$ --- the same $a$ and the same $A$ as the first problem, with a longer
$b$.

\textbf{(a)} Compute $h$ and compare it with $a$. \quad \textbf{(b)} Find $B$,
and explain why the supplement gives no second triangle in this case. \quad
\textbf{(c)} Find side $c$.
\probfig{6}}

\problem{\textbf{Surveying a site.} From station $A$ a surveyor measures
$A = 62^\circ$ and the distance $b = 30$ m to a corner $C$. The distance from
the second station $B$ to that corner is $a = 45$ m.

\textbf{(a)} Explain from the sizes of $a$ and $b$ alone why this measurement
cannot be ambiguous, then find the angle at $B$. \quad \textbf{(b)} Find the
distance $c$ between the two stations, in metres.
\probfig{7}
\par\vspace*{5.5cm}\answerline{m}}

\problem[6.5cm]{\textbf{How much slack does the ambiguity have?} Here
$A = 30^\circ$, $b = 18$ and $a = 10$.

\textbf{(a)} Compute $h$ and find both possible values of $B$. \quad
\textbf{(b)} Find side $c$ for each triangle. \quad \textbf{(c)} By how much
does $a$ exceed $h$? If the measurement of $a$ shrank by exactly that much, how
many triangles would remain, and what would be special about the survivor?
\probfig{8}}

\problem[6.5cm]{\textbf{Two possible positions.} A drone at $C$ is tracked from
control point $A$, where the angle is $A = 44^\circ$ and the distance to the
drone is $b = 26$. A ground marker at $B$ lies $a = 20$ from the drone.

\textbf{(a)} Find both values of the angle at $B$ and the matching values of
$c$. \quad \textbf{(b)} The drone is known to be \emph{beyond} the marker rather
than in front of it. Say which of the two triangles that rules out, and name one
extra observation that would settle the case on its own.
\probfig{9}}

\problem[6.5cm]{\textbf{Challenge --- the whole case rule.} A measurement fixes
$A = 50^\circ$ and $b = 16$, and only $a$ varies.

\textbf{(a)} With $a = 14$, find the acute $B$ and say how many triangles exist.
\quad \textbf{(b)} Compute $h = b\sin A$. \quad \textbf{(c)} State the complete
rule for this $A$ and $b$: for which values of $a$ are there no triangles,
exactly one, or two? Justify each boundary by what the swinging side is doing.
\probfig{10}}
