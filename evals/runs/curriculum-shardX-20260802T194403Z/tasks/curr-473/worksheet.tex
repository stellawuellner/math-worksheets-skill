\wstitleblock{Differentiating Accumulation Functions with the FTC}{AP Calculus AB/BC --- Integration}{}

\noindent\textbf{Directions:} An accumulation function is a function defined by an integral
with a variable limit, e.g. $\displaystyle g(x) = \int_a^x f(t)\,dt$. The Fundamental Theorem
says $g'(x) = f(x)$: the integrand \emph{is} the derivative. Show the step that names which
form of the theorem you are using, then compute. Give exact values unless a decimal is
asked for; state units where the problem gives them.\par\vspace{0.15cm}
\noindent{\small\textit{This workshop covers: Differentiating accumulation functions,
evaluating accumulated change, the chain-rule form of the FTC, and reading an accumulation
function's behaviour.}}

\problem{Let $\displaystyle g(x) = \int_1^{x} \left(t^3 + 2\right) dt$.\par\vspace{0.1cm}
Find $g'(3)$. (You should not need to evaluate the integral.)
\par\vspace*{4.5cm}\ansline}

\problem{Water flows into a tank at the rate $r(t) = 6t + 4$ litres per minute, where $t$ is
measured in minutes. The volume added during the first $x$ minutes is
$\displaystyle V(x) = \int_0^{x} \left(6t + 4\right) dt$.\par\vspace{0.1cm}
Find $V(5)$, the volume added during the first $5$ minutes.
\par\vspace*{4.5cm}\answerline{L}}

\problem{Let $\displaystyle g(x) = \int_1^{x^2} \sqrt{1 + t^3}\,dt$.\par\vspace{0.1cm}
Find $g'(2)$. Give the exact value.
\par\vspace*{5cm}\ansline}

\problem{A continuous, decreasing function $f$ has the values in the table, and
$\displaystyle g(x) = \int_0^{x} f(t)\,dt$ for $0 \le x \le 7$.
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|c|}\hline
$t$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$\\\hline
$f(t)$ & $4$ & $3$ & $2$ & $0$ & $-2$ & $-3$ & $-4$ & $-5$\\\hline
\end{tabular}
\end{center}
On which intervals is $g$ increasing, and on which is it decreasing? Where on $[0,7]$ does
$g$ attain its absolute maximum? Justify each claim with the Fundamental Theorem --- this
one is graded on the reasoning, not just the answer.
\par\vspace*{5.5cm}\noansline}

\problem{Let $\displaystyle h(x) = \int_{x}^{4} \ln\!\left(t^2 + 1\right) dt$. Note that the
variable is the \textbf{lower} limit.\par\vspace{0.1cm}
Find $h'(2)$. Give the exact value.
\par\vspace*{5cm}\ansline}

\problem{A pump adds water to a reservoir at the rate $r(t) = 12 - t$ litres per minute for
$0 \le t \le 10$, and $\displaystyle V(x) = \int_0^{x} \left(12 - t\right) dt$ is the volume
added in the first $x$ minutes.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.2cm]
  \item Find $V(10)$, in litres. \ansline
  \item Find $V'(10)$, and state what it tells you about the reservoir at $t = 10$ minutes,
        with its units. \ansline
\end{enumerate}\vspace*{0.2cm}}

\problem{Let $\displaystyle g(x) = \int_0^{x} \left(4t^3 - 6t\right) dt$.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.2cm]
  \item Evaluate the integral to write an explicit formula for $g(x)$ (a polynomial in $x$,
        no integral sign). \ansline
  \item Differentiate your formula from part (a) to find $g'(x)$, and say what the
        Fundamental Theorem predicted the answer would be. \ansline
\end{enumerate}\vspace*{0.2cm}}

\problem{\textbf{Synthesis.} Let $\displaystyle g(x) = \int_1^{x^2} \frac{3}{t}\,dt$ for
$x > 0$.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.2cm]
  \item Find $g'(x)$ and simplify it completely. \ansline
  \item Evaluate $g'(2)$. \ansline
\end{enumerate}\vspace*{0.2cm}}
