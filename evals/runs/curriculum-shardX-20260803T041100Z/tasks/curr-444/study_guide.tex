\sstitleblock{Sigma Notation: Index Shifts and Convergence}

\skillheading{Skill 1 --- Shifting the index of a finite sum}

\begin{formulabox}
\textbf{Rule:} a sum is its summand \emph{and} its limits. Substituting $m = n - c$ moves
\textbf{every} index value, so both limits move by the same $c$:
\[ \sum_{n=p}^{q} f(n-c) \;=\; \sum_{m=p-c}^{q-c} f(m) \]
\textbf{Term count (fencepost):} the sum $\sum_{n=p}^{q}$ has $q - p + 1$ terms. Counting
terms before and after a shift is the fastest way to catch a bad one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Rewrite $\sum_{n=2}^{5}(n-1)^2$ with the index starting at 1, then
evaluate.
\step{The summand is a function of $n-1$, so substituting $m = n-1$ simplifies it --- and
that same substitution must be applied to both limits.}
\step{$n = 2$ gives $m = 1$ and $n = 5$ gives $m = 4$, so the sum is
$\sum_{m=1}^{4} m^2 = 1 + 4 + 9 + 16$. Leaving the upper limit at 5 would add a fifth term
and give 55.}
$\Rightarrow\quad \ans{30}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Rewrite $\sum_{n=3}^{7}(n-2)^3$ with the index starting at 1 and
evaluate it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{225}$}
\end{tryitbox}

\skillheading{Skill 2 --- Check the ratio before the sum formula}

\begin{formulabox}
\textbf{Rule:} \quad $\displaystyle\sum_{n=p}^{\infty} a r^{\,n-p} = \frac{a}{1-r}$
\quad \textbf{only if} $|r| < 1$; otherwise the series \textbf{diverges} and has no sum.\par
\smallskip
Two things must be named before dividing: $r$ (the constant multiplier between
consecutive terms) and $a$ (the term produced by the \emph{stated} lower limit --- not
automatically the $n = 0$ term).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\sum_{n=0}^{\infty} 6\left(\frac{1}{3}\right)^{n}$, or
state that it diverges.
\step{Each term is the previous one times $\frac{1}{3}$, so this is geometric --- but the
formula is only licensed once the ratio passes the condition.}
\step{Here $r = \frac{1}{3}$ and $\left|\frac{1}{3}\right| < 1$, so it converges; the
lower limit gives $a = 6$, and $\frac{6}{1 - \frac{1}{3}} = \frac{6}{\frac{2}{3}}$.}
$\Rightarrow\quad \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\sum_{n=0}^{\infty} 4\left(\frac{3}{5}\right)^{n}$, or
state that it diverges.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reading the limit of a sequence}

\begin{formulabox}
\textbf{Rule:} for a ratio of polynomials in $n$, compare \emph{degrees}: divide numerator
and denominator by the highest power of $n$ present, then send every $\frac{1}{n^k}$ to 0.
Equal degrees leave the ratio of leading coefficients.\par
\smallskip
\textbf{Careful:} $a_n \to 0$ does \textbf{not} make $\sum a_n$ converge --- the harmonic
series is the standard counterexample. The $n$th-term test can prove divergence only.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\lim_{n\to\infty}\frac{3n+5}{2n-1}$.
\step{Numerator and denominator both grow without bound, so comparing degrees is the tool
--- divide through by the highest power, $n$.}
\step{The quotient becomes $\dfrac{3 + \frac{5}{n}}{2 - \frac{1}{n}}$, and both fractions
tend to 0, leaving the ratio of leading coefficients.}
$\Rightarrow\quad \ans{\dfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\lim_{n\to\infty}\frac{n^2+4}{5n^2-n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{5}}$}
\end{tryitbox}
