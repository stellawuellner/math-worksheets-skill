\sstitleblock{Off-by-One Errors in Sequence Indexing}

\skillheading{Skill 1 --- Jumps are one fewer than terms}

\begin{formulabox}
\textbf{The whole topic in one line:} between $n$ terms there are $n-1$ jumps. Fence posts and fence panels --- five posts, four panels.\par
\textbf{So} $a_n = a_1 + (n-1)d$, and between $a_j$ and $a_k$ there are $k-j$ jumps, giving $d = \dfrac{a_k - a_j}{k - j}$.\par
\textbf{Self-check:} put $n = 1$ into your rule. If it does not return $a_1$, the index is off by one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $a_1 = 4$ and $d = 6$. Find $a_9$.
\step{Term $9$ is not nine steps from the start --- count the jumps from term 1, which is one fewer.}
\step{Eight jumps: $4 + 8(6) = 4 + 48$.}

$a_9 = \ans{52}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a_1 = 10$ and $d = 5$. Find $a_{12}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $a_{12} = \ans{65}$}
\end{tryitbox}

\skillheading{Skill 2 --- Where an explicit rule starts}

\begin{formulabox}
\textbf{Rule:} a sequence rule always comes with the index it starts at, almost always $n \ge 1$. The first term is the value at that starting index --- not at $n = 0$.\par
\textbf{Two rules, both correct, different shapes:} $a_n = a_1 + (n-1)d$ is built from the first term; a tidied form like $a_n = 4n + 1$ hides the $-1$ inside the arithmetic. Test either at $n = 1$ before trusting it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $a_n = 5n - 2$ with $n \ge 1$, find the first and the sixth terms.
\step{The rule is stated from $n = 1$, so the first term is the value there --- substituting zero would name a term the sequence does not have.}
\step{$5(1) - 2 = 3$ and $5(6) - 2 = 30 - 2$.}

$a_1 = \ans{3}$ \quad and \quad $a_6 = \ans{28}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $a_n = 7n + 4$ with $n \ge 1$, find $a_1$ and $a_5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $a_1 = \ans{11}$, $a_5 = \ans{39}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a rule written $a_n = a_1 + nd$ gives the wrong first term, and a step found by dividing by the larger index rather than the index \emph{difference} is wrong for every term after it. Both are the same miscount.
\end{watchoutbox}

\skillheading{Skill 3 --- Counting terms in a block}

\begin{formulabox}
\textbf{Rule:} the block $a_j, a_{j+1}, \ldots, a_k$ holds $k - j + 1$ terms. Subtracting alone counts the gaps.\par
\textbf{Sum of the first $n$ terms:} $S_n = \dfrac{n}{2}(a_1 + a_n)$ --- and $n$ here is the number of terms, so an index range that starts anywhere but $1$ must be counted with the rule above first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many terms are in $a_5$ through $a_{18}$? And what is the sum of the first five terms of $a_n = 2 + 3(n-1)$?
\step{A count of items is one more than a count of gaps, and a sum formula needs the count, so settle the count first.}
\step{$18 - 5 + 1 = 13 + 1$.}
\step{First five terms: $2, 5, 8, 11, 14$, so $\frac{5}{2}(2 + 14)$.}

count $= \ans{14}$ \quad and \quad $S_5 = \ans{40}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many terms in $a_7$ through $a_{23}$? And find $S_6$ for $a_n = 1 + 5(n-1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: count $= \ans{17}$, $S_6 = \ans{81}$}
\end{tryitbox}
