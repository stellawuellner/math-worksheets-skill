\sstitleblock{Congruent or Similar? Evidence from Transformations}

\begin{formulabox}
\textbf{The two tools.} \ Distance
$d = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}$ \ and slope
$m = \frac{y_2-y_1}{x_2-x_1}$ (undefined if the run is $0$).\\[3pt]
\textbf{What each proves.} Equal corresponding \emph{lengths} $\Rightarrow$
congruent (some rigid motion works). Proportional corresponding lengths, all with
the same factor $k$ $\Rightarrow$ similar. \emph{Slopes} are what tell you
\textbf{which} motion: translation keeps every slope; a half turn keeps every
slope; a quarter turn gives the negative reciprocal; a reflection across
$y = x$ gives the reciprocal.
\end{formulabox}

\skillheading{1. Translation: slide, nothing else changes}

\begin{formulabox}
Rule: $(x, y) \rightarrow (x + a, y + b)$, the same $a$ and $b$ for every point.
Evidence to collect: corresponding lengths are equal \emph{and} corresponding
slopes are equal, so each side of the image is parallel to its partner. If the
lengths match but a slope has changed, the motion was not a translation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A(1,2)$ and $B(5,5)$ slide by $(x,y) \rightarrow (x+3, y-2)$,
giving $A'(4,0)$ and $B'(8,3)$. Find $A'B'$.
\step{The claim is that lengths are unchanged, so compute the image length and
compare it with $AB = 5$ --- that comparison is the evidence.}
\step{$A'B' = \sqrt{(8-4)^2 + (3-0)^2} = \sqrt{16 + 9}$.}
$A'B' = \ans{5}$, equal to $AB$, so the segments are congruent.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $C(0,1)$ and $D(6,9)$ slide by $(x,y) \rightarrow (x+3, y-3)$.
Find $C'D'$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{2. Reflection: mirror across a line}

\begin{formulabox}
Rules worth knowing: across the $x$-axis, $(x,y) \rightarrow (x,-y)$; across the
$y$-axis, $(x,y) \rightarrow (-x,y)$; across the line $y = x$,
$(x,y) \rightarrow (y,x)$. Evidence: lengths are preserved, but the figure is
turned over --- across $y = x$ every slope becomes its reciprocal, which is the
easiest fingerprint to check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $P(1,1)$ and $Q(7,9)$ are reflected across the $y$-axis to
$P'(-1,1)$ and $Q'(-7,9)$. Find $P'Q'$.
\step{To show a reflection is rigid I need a length on each side of the mirror,
so compute the image length and compare it with $PQ = 10$.}
\step{$P'Q' = \sqrt{(-7-(-1))^2 + (9-1)^2} = \sqrt{36 + 64}$.}
$P'Q' = \ans{10}$, so $\overline{PQ} \cong \overline{P'Q'}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $R(1,2)$ and $S(4,6)$ are reflected across the $x$-axis. Find
$R'S'$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{3. Rotation: turn about a centre}

\begin{formulabox}
About the origin: a quarter turn counterclockwise is
$(x,y) \rightarrow (-y,x)$; a half turn is $(x,y) \rightarrow (-x,-y)$; a
quarter turn clockwise is $(x,y) \rightarrow (y,-x)$. Evidence: lengths are
preserved. Slopes change for every rotation \emph{except} the half turn, where
rise and run both change sign and the quotient survives.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $E(1,2)$ and $F(4,6)$ are turned a quarter turn
counterclockwise about the origin to $E'(-2,1)$ and $F'(-6,4)$. Find $E'F'$.
\step{A rotation should leave the length alone, so the check is to recompute the
distance after the turn and compare it with $EF = 5$.}
\step{$E'F' = \sqrt{(-6-(-2))^2 + (4-1)^2} = \sqrt{16 + 9}$.}
$E'F' = \ans{5}$, unchanged, so the rotation is a rigid motion.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $G(0,0)$ and $H(6,8)$ are turned a quarter turn
counterclockwise about the origin. Find $G'H'$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{4. Dilation: the one that changes size}

\begin{formulabox}
About the origin: $(x,y) \rightarrow (kx, ky)$. Lengths are multiplied by $k$,
so a dilation is \emph{not} rigid and the figures are similar, not congruent.
Evidence: compute the ratio image length over pre-image length for
\textbf{several} pairs of corresponding sides --- similarity needs the same $k$
every time, and one matching ratio proves nothing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $X(0,0)$ and $Y(4,3)$ are dilated about the origin by
$k = 3$, giving $X'(0,0)$ and $Y'(12,9)$. Find $X'Y'$.
\step{Here the length is expected to change, so compute the image length and
divide by $XY = 5$ to recover the scale factor.}
\step{$X'Y' = \sqrt{12^2 + 9^2} = \sqrt{144 + 81}$.}
$X'Y' = \ans{15}$, and $15 \div 5 = 3 = k$ as predicted.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $X(0,0)$ and $Z(6,8)$ are dilated about the origin by
$k = 2$. Find $X'Z'$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}
