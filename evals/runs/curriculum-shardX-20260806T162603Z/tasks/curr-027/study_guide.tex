\sstitleblock{Subtracting on a Number Path}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Count Back on the Path}

\begin{formulabox}
\textbf{Rule:} Put your finger on the \textbf{start} number. Hop back one
box for each one you take away. The box you land on is the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 - 2$
\step{Taking away 2 means two hops back from the start number 8.}
\step{Hop: 7, then 6 --- two hops, landing on 6. $\checkmark$}

$\Rightarrow\quad 8 - 2 = \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $7 - 3$ \ (start at 7, hop back 3)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Count Up on the Path}

\begin{formulabox}
\textbf{Rule:} When the number taken away is big, start at the \textbf{small}
number and count the hops \emph{up} to the start number. The number of hops
is the answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $8 - 5$
\step{Hopping back 5 times is a lot, so we start at 5 and count hops up to 8 instead.}
\step{From 5, hop: 6, 7, 8 --- that is 3 hops. $\checkmark$}

$\Rightarrow\quad 8 - 5 = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $10 - 8$ \ (start at 8, hop up to 10)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Find the Missing Hop}

\begin{formulabox}
\textbf{Rule:} If you know where a hop trip \textbf{starts} and where it
\textbf{lands}, count the boxes between them --- that count is the hidden
number: \ start $-$ hops $=$ landing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $9 - \square = 5$
\step{We know the start 9 and the landing 5, so the missing number is the count of hops between them.}
\step{From 9 down to 5: \ 8, 7, 6, 5 --- that is 4 hops, and $9-4=5$. $\checkmark$}

$\Rightarrow\quad$ the missing hop number is $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $7 - \square = 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not count the box you start on! For $7-2$, your
finger starts on 7 and the hops land on 6, then 5 --- counting ``7'' as a
hop lands you one too high.
\end{watchoutbox}
