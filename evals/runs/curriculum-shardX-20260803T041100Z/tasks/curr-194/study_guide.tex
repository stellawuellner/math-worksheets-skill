\sstitleblock{Radius or Diameter? Length or Covering?}

\skillheading{Skill 1 --- Halve a diameter before you square it}

\begin{formulabox}
\textbf{Radius $r$} reaches from the centre to the edge. \textbf{Diameter $d$}
crosses the whole circle through the centre, so $d = 2r$.\par
\smallskip
$C = 2\pi r = \pi d$ \quad (either number works) \qquad
$A = \pi r^2$ \quad (\emph{radius only})\par
\smallskip
Classify the given number first: ``across'', ``through the centre'',
``wide'' $\rightarrow$ diameter. ``From the centre'' $\rightarrow$ radius.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A circle is $10$ cm across at its widest point. Find its
area.
\step{``Across at its widest point'' is a diameter, and $A = \pi r^2$ accepts a
radius only, so the halving has to happen before the squaring.}
\step{$r = 10 \div 2$, so $A = \pi (10 \div 2)^2 = \pi \times 25 = 78.5398\ldots$}
The area is $\ans{78.54\ \text{cm}^2}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A circle is $6$ cm across at its widest point. Find its
area, rounded to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28.27\ \text{cm}^2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Around the circle, or across its face?}

\begin{formulabox}
\textbf{Circumference} is the trip around the rim: a \emph{length}, measured in
cm or m. \textbf{Area} is the face the circle covers: a \emph{covering},
measured in cm$^2$ or m$^2$.\par
\smallskip
Deciding words: \emph{ribbon, fence, trim, rolls once around} $\rightarrow$
$C = 2\pi r$. \emph{Paint, paper, covers, surface} $\rightarrow$ $A = \pi r^2$.
The unit of your answer must match the word you picked.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A circular badge has radius $4$ cm. How much ribbon runs
around its edge?
\step{``Around its edge'' names a length, so this is circumference --- the
squared formula would answer a question nobody asked.}
\step{$C = 2\pi r = 2\pi(4) = 25.1327\ldots$}
The ribbon needed is $\ans{25.13\ \text{cm}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A circular badge has radius $8$ cm. How much paper does it
take to cover the badge's face?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{201.06\ \text{cm}^2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Same rectangle, two different questions}

\begin{formulabox}
For a rectangle with sides $a$ and $b$:\par
\smallskip
\textbf{Perimeter} $P = 2a + 2b$ --- add your way around the outside (a length).\par
\textbf{Area} $A = a \times b$ --- multiply to cover the inside (a covering).\par
\smallskip
A composite shape has no single formula: cut it into rectangles, find each area,
and add. Never add side lengths when the question asks what the shape covers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A deck is $7$ m long and $3$ m wide. How much railing runs
around it?
\step{Railing follows the outside edge, so the question wants perimeter --- the
adding formula, not the multiplying one.}
\step{$P = 2(7) + 2(3) = 14 + 6$}
The railing needed is $\ans{20\ \text{m}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The same deck is $7$ m long and $3$ m wide. How many square
metres of decking board cover it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21\ \text{m}^2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Forgetting to halve a diameter multiplies an area by
four, not by two, because the radius is squared. And an answer whose unit
disagrees with the formula you wrote is wrong before you check the arithmetic.
\end{watchoutbox}
