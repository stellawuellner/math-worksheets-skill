\problem{Find the distance between $A(2,3)$ and $B(10,9)$.}

\noindent\textbf{Solution.} The horizontal gap is $10 - 2 = 8$ and the vertical
gap is $9 - 3 = 6$; those are the legs of a right triangle whose hypotenuse is
the distance.
\[ d = \sqrt{8^2 + 6^2} = \sqrt{64 + 36} = \sqrt{100} \]
Therefore \ans{d = 10 \text{ units}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the midpoint of $C(-4,7)$ and $D(6,1)$.}

\noindent\textbf{Solution.} Average each coordinate separately:
\[ M = \left( \frac{-4+6}{2}, \frac{7+1}{2} \right)
     = \left( \frac{2}{2}, \frac{8}{2} \right) \]
Therefore \ans{M = (1,\, 4)}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Find the distance between $P(-5,2)$ and $Q(7,-3)$.}

\noindent\textbf{Solution.} Horizontal gap $7 - (-5) = 12$; vertical gap
$-3 - 2 = -5$. Squaring removes the sign.
\[ d = \sqrt{12^2 + (-5)^2} = \sqrt{144 + 25} = \sqrt{169} \]
Therefore \ans{d = 13 \text{ units}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$M(3,5)$ is the midpoint of $\overline{AB}$ and $A(-1,2)$. Find $B$.}

\noindent\textbf{Solution.} The midpoint sits half way, so travelling from $A$
to $M$ and then the same distance again lands on $B$. In coordinates that is
$B = 2M - A$, one coordinate at a time:
\[ x_B = 2(3) - (-1) = 6 + 1, \qquad y_B = 2(5) - 2 = 10 - 2 \]
so \ans{x_B = 7}

\noindent and \ans{y_B = 8}

\noindent\textit{Check:} the midpoint of $(-1,2)$ and $(7,8)$ is
$\left(\frac{-1+7}{2}, \frac{2+8}{2}\right) = (3,5)$, as given.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{A park entrance at $(2,1)$ and a library at $(14,10)$, one unit per
block. Find the straight-line distance in blocks.}

\noindent\textbf{Solution.} Horizontal gap $14 - 2 = 12$ blocks, vertical gap
$10 - 1 = 9$ blocks.
\[ d = \sqrt{12^2 + 9^2} = \sqrt{144 + 81} = \sqrt{225} \]
Therefore \ans{d = 15 \text{ blocks}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Endpoints $E(-6,3)$ and $F(2,-5)$. (a) Find the midpoint.
(b) Explain why averaging the coordinates gives a point on the segment.}

\noindent\textbf{Solution (a).}
\[ M = \left( \frac{-6+2}{2}, \frac{3+(-5)}{2} \right)
     = \left( \frac{-4}{2}, \frac{-2}{2} \right) \]
Therefore \ans{M = (-2,\, -1)}

\noindent\textbf{Solution (b) --- instructor-judged.} A complete response says
averaging the $x$-coordinates moves exactly half of the horizontal run from one
endpoint, averaging the $y$-coordinates moves exactly half of the vertical rise,
and because the segment is straight those two half-steps are taken together and
land on the segment itself --- half the run always pairs with half the rise on a
straight line. A response that says only "it is between the endpoints" has not
explained why the point lies on the segment; award partial credit.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Endpoints $R(-3,-1)$ and $S(9,4)$. (a) Find $RS$. (b) Find the midpoint
of $\overline{RS}$.}

\noindent\textbf{Solution (a).} Horizontal gap $9 - (-3) = 12$, vertical gap
$4 - (-1) = 5$.
\[ RS = \sqrt{12^2 + 5^2} = \sqrt{144 + 25} = \sqrt{169} \]
Therefore \ans{RS = 13 \text{ units}}

\noindent\textbf{Solution (b).}
\[ M = \left( \frac{-3+9}{2}, \frac{-1+4}{2} \right)
     = \left( \frac{6}{2}, \frac{3}{2} \right) \]
Therefore \ans{M = (3,\, 1.5)}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Quadrilateral $W(-4,1)$, $X(0,5)$, $Y(6,-1)$, $Z(2,-5)$. (a) Midpoint of
$\overline{WY}$. (b) Midpoint of $\overline{XZ}$. (c) What do the two answers
tell you about the quadrilateral?}

\noindent\textbf{Solution (a).}
$\left( \frac{-4+6}{2}, \frac{1+(-1)}{2} \right) = \left( \frac{2}{2}, \frac{0}{2} \right)$,
so \ans{(1,\, 0)}

\noindent\textbf{Solution (b).}
$\left( \frac{0+2}{2}, \frac{5+(-5)}{2} \right) = \left( \frac{2}{2}, \frac{0}{2} \right)$,
so \ans{(1,\, 0)}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response
observes that the two diagonals have the \emph{same} midpoint, so each diagonal
cuts the other exactly in half, and concludes that the quadrilateral is a
parallelogram (a quadrilateral whose diagonals bisect each other is a
parallelogram). Award full credit only when both the shared midpoint and the
conclusion drawn from it appear.
