\sstitleblock{Reading a Line Plot with Fractions}

\skillheading{Skill 1 --- Reading counts off the plot}

\begin{formulabox}
\textbf{Rule:} On a line plot, the \emph{tick} tells you the amount and the \emph{height of the stack} tells you how many.\\[3pt]
So ``how many'' questions count crosses, and ``how much'' questions use the fraction under them. Reading the wrong one is the mistake this plot invites.\\[3pt]
Number of measurements $=$ the sum of all the stack heights.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A plot has 3 crosses at $\frac{1}{4}$, 5 crosses at $\frac{1}{2}$ and 2 crosses at $\frac{3}{4}$. How many measurements were taken?
\step{The question asks how many, not how much, so I add stack heights and ignore the fractions underneath.}
\step{$3 + 5 + 2 = 10$ crosses, and each cross is one measurement.}
Measurements taken: $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has 2 crosses at $\frac{1}{8}$, 4 at $\frac{3}{8}$ and 3 at $\frac{5}{8}$. How many measurements were taken?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 2 --- Adding up one whole stack}

\begin{formulabox}
\textbf{Rule:} A stack of $n$ crosses at amount $a$ holds $n \times a$ altogether.\\[3pt]
Multiply the numerator by $n$ and leave the denominator alone: $n \times \frac{p}{q} = \frac{n \times p}{q}$. Then simplify, and write anything over 1 as a mixed number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Three containers each hold $\frac{1}{4}$ of a litre. How much is that altogether?
\step{Equal amounts repeated is multiplication, so I multiply the amount by the number of containers rather than adding three times.}
\step{$3 \times \frac{1}{4} = \frac{3}{4}$, which is already in lowest terms and less than 1.}
Total: $\ans{\tfrac{3}{4}}$ of a litre
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Five containers each hold $\frac{3}{8}$ of a litre. How much is that altogether?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{15}{8}}$ litres}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the middle of the data}

\begin{formulabox}
\textbf{Rule:} The \textbf{median} is the value in the middle place once the data is in order --- and a line plot is already in order, left to right.\\[3pt]
With an odd number of crosses, count in to the middle one. With an even number, the median sits halfway between the two middle crosses. The \textbf{range} is the rightmost amount minus the leftmost.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Seven measurements: $\frac{1}{8}, \frac{1}{4}, \frac{1}{4}, \frac{3}{8}, \frac{1}{2}, \frac{1}{2}, \frac{3}{4}$. Find the median.
\step{The list is already ordered, so the median is a counting job: find the middle place rather than doing any arithmetic.}
\step{Seven values means the middle place is the 4th, and counting in from the left the 4th value is $\frac{3}{8}$.}
Median: $\ans{\tfrac{3}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Seven measurements: $\frac{1}{8}, \frac{1}{8}, \frac{1}{4}, \frac{1}{2}, \frac{5}{8}, \frac{3}{4}, \frac{3}{4}$. Find the median.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Pouring together and sharing out}

\begin{formulabox}
\textbf{Rule:} Two steps, in this order. Pour together: total $= n \times a$. Share equally: each share $=$ total $\div m$.\\[3pt]
Dividing a fraction by a whole number means multiplying by its reciprocal: $\frac{p}{q} \div m = \frac{p}{q \times m}$. Doing the sharing before the pouring gives the wrong answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Six cups each hold $\frac{1}{2}$ litre. They are poured together, then shared equally between 4 cups. How much is in each?
\step{Sharing only works on one pooled amount, so I total the water first and divide afterwards.}
\step{Pour: $6 \times \frac{1}{2} = 3$ litres. Share: $3 \div 4 = \frac{3}{4}$.}
Each cup: $\ans{\tfrac{3}{4}}$ of a litre
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Four cups each hold $\frac{3}{4}$ litre. They are poured together, then shared equally between 6 cups. How much is in each?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2}}$ of a litre}
\end{tryitbox}
