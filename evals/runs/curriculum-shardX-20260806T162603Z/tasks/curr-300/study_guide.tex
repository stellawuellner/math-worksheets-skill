\sstitleblock{Choosing a Model from the Evidence}

\skillheading{Skill 1 --- Spotting a linear model}

\begin{formulabox}
\textbf{Evidence:} equal steps in the input give \emph{equal} steps in the output --- the \textbf{first differences} are constant.\par
\textbf{Rule:} $y = (\text{start value}) + (\text{constant step}) \times x$.\par
In words: a fixed amount to begin with, then the same amount added again and again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A club charges a joining fee and then a fixed amount per badge: totals are $15, 19, 23, 27$ dollars for $n = 0, 1, 2, 3$ badges. Find the total for $9$ badges.
\step{Want a rule; the first differences are $4, 4, 4$ --- constant --- so the family is linear and I only need the start value and the step.}
\step{Start value $15$ at $n = 0$, step $4$, so the rule is $15 + 4n$; at $n = 9$ this is $15 + 36$.}

$\Rightarrow\quad \ans{51}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Totals are $20, 26, 32, 38$ dollars for $n = 0, 1, 2, 3$. Find the total for $7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{62}$}
\end{tryitbox}

\skillheading{Skill 2 --- Spotting a quadratic model}

\begin{formulabox}
\textbf{Evidence:} the first differences change, but by the \emph{same} amount each time --- the \textbf{second differences} are constant. Falling-object heights and any area built from two lengths that both move behave this way.\par
\textbf{Rule:} $y = ax^2 + bx + c$; for a height in feet, $h(t) = -16t^2 + v_0 t + h_0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ball's height in feet is $0, 40, 48, 24$ at $t = 0, 1, 2, 3$ seconds, matching $h(t) = -16t^2 + 56t$. Find the height at $t = 2.5$ seconds.
\step{First differences $40, 8, -24$ are not constant but the second differences are both $-32$, so the family is quadratic and I may substitute into the quadratic rule.}
\step{$h(2.5) = -16(6.25) + 56(2.5) = -100 + 140$.}

$\Rightarrow\quad \ans{40}$ feet
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A second ball follows $h(t) = -16t^2 + 48t$. Find its height at $t = 2.5$ seconds.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

\skillheading{Skill 3 --- Spotting a piecewise model}

\begin{formulabox}
\textbf{Evidence:} the rule itself \emph{changes} at a break point --- a flat charge that becomes a per-unit charge, a free allowance that runs out, a rate that switches.\par
\textbf{Rule:} one expression before the break, another after it. Past a break at $x = k$: $(\text{amount used up to } k) + (\text{new rate}) \times (x - k)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A hall costs $\$5$ for the first $3$ hours and $\$2$ for every hour after that. Find the cost of $7$ hours.
\step{The rate changes at the third hour, so the family is piecewise; $7$ is past the break, so I use the second piece and count only the hours beyond $3$.}
\step{Hours past the break: $7 - 3 = 4$, so the cost is $5 + 2(4) = 5 + 8$.}

$\Rightarrow\quad \ans{13}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A locker costs $\$4$ for the first $2$ days and $\$3$ for every day after. Find the cost of $6$ days.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``the numbers keep going up'' does not make a table linear. Rising values only rule out a decreasing rule --- always test the \emph{differences} before naming the family.
\end{watchoutbox}
