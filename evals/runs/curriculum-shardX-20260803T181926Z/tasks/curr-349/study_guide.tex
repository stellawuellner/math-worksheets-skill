\sstitleblock{Getting the Vector, the Scale Factor and the Centre Right}

\skillheading{Skill 1 --- Finding a translation vector}

\begin{formulabox}
\textbf{Rule:} \quad
$\langle a,\ b \rangle = \langle x' - x,\ \ y' - y \rangle$ \quad --- image
\emph{minus} pre-image, always in that order.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Reversing the order gives the \emph{inverse} translation, the one that
        sends the image back to the original.
  \item Check by applying it: $(x + a,\ y + b)$ must land on the image.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A translation maps $A(-2,\,3)$ to $A'(6,\,-1)$. Find the
vector.
\step{The vector records the trip from pre-image to image, so each component is
the image coordinate minus the pre-image coordinate --- set that subtraction up
before evaluating anything.}
\step{Horizontal: $6 - (-2) = \ans{8}$. \ Vertical: $-1 - 3 = \ans{-4}$. The
vector is eight right and four down.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A translation maps $B(4,\,-5)$ to $B'(-1,\,2)$. Find the
vector.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$ and $\ans{7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Finding a scale factor}

\begin{formulabox}
\textbf{Rule:} \quad
$k = \dfrac{\text{distance from centre to image}}{\text{distance from centre to
pre-image}}$ --- a \emph{ratio}, never a difference.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $k > 1$ enlarges, \ $0 < k < 1$ shrinks, \ $k = 1$ changes nothing.
  \item Measure both distances from the \emph{same} centre, or the ratio means
        nothing.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A dilation centred at the origin maps $P(3,\,4)$ to
$P'(6,\,8)$. Find $k$.
\step{A scale factor compares distances from the centre, so measure both
distances first and divide --- the coordinates themselves are not the
comparison.}
\step{$OP = \sqrt{9+16} = \ans{5}$ \ and \ $OP' = \sqrt{36+64} = \ans{10}$.}
\step{$k = \dfrac{10}{5} = \ans{2}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A point lies $13$ units from the centre and its image lies
$39$ units from the same centre. Find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Subtracting the two distances, or two matching
coordinates, gives a number that changes if you rescale the whole picture. A
scale factor must not.
\end{watchoutbox}

\skillheading{Skill 3 --- Dilating about a given centre}

\begin{formulabox}
\textbf{Rule:} \quad $A' = C + k\,(A - C)$, one coordinate at a time:
$x' = u + k(x - u)$ and $y' = v + k(y - v)$, where $C = (u,\,v)$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Multiplying the coordinates by $k$ is the shortcut for $C$ at the
        \emph{origin} only.
  \item The centre is the one point that does not move --- so it lies on every
        line joining a point to its image.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Dilate $A(3,\,5)$ by $k=3$ about $C(1,\,2)$.
\step{The centre is not the origin, so the coordinates cannot simply be
multiplied --- step out from the centre, scale that step, and step back.}
\step{$x' = 1 + 3(3-1) = 1 + 6 = \ans{7}$ \ and \
$y' = 2 + 3(5-2) = 2 + 9 = \ans{11}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Dilate $B(1,\,6)$ by $k=2$ about $C(-2,\,4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$ and $\ans{8}$}
\end{tryitbox}
