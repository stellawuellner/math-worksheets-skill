\wstitleblock{Teen Numbers: One Ten and Some Ones}{Kindergarten / Grade 1 \quad Number and Operations in Base Ten}{}

\begin{center}
\textit{Building a teen number from one ten and some ones, breaking one apart,
and comparing two of them}
\end{center}

\noindent\textbf{How to use this sheet.}
\begin{itemize}[leftmargin=1.2cm, itemsep=3pt, topsep=3pt]
\item A \textbf{full ten-frame} always holds ten. Fill it first, then count what
is left over --- those are the \textbf{ones}.
\item A \textbf{ten-stick} is ten cubes joined together. One stick and some
loose cubes make a teen number.
\item Every teen number is $10 + \text{some ones}$.
\end{itemize}
