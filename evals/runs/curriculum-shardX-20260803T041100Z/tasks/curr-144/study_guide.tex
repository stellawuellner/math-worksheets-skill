\sstitleblock{Square or Cubic? Getting the Unit Right}

\skillheading{Skill 1 --- Read the unit off the formula}

\begin{formulabox}
\textbf{Rule: count the lengths you multiplied.}\\[3pt]
\begin{itemize}[label={$\bullet$}, leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item one length (perimeter, an edge) $\rightarrow$ plain units: cm, m, in
  \item \textbf{two} lengths (area, surface area) $\rightarrow$ \textbf{square} units
  \item \textbf{three} lengths (volume) $\rightarrow$ \textbf{cubic} units
\end{itemize}
\smallskip
Area of a rectangle $= \ell \times w$. \quad Volume of a box $= \ell \times w \times h$.
\quad Surface area of a cube $= 6 \times (\text{edge} \times \text{edge})$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A box is 5 centimeters long, 3 centimeters wide and 2 centimeters
tall. Find the volume, with its unit.
\step{Volume fills a space, so I pack unit cubes in --- that means multiplying all three
edges, and three lengths always give cubic units.}
\step{One layer on the floor holds $5 \times 3 = 15$ cubes, and there are 2 layers:
$15 \times 2 = 30$.}
$\Rightarrow\quad$ volume $= \ans{30}$ cubic centimeters
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A box is 6 centimeters long, 2 centimeters wide and 4 centimeters
tall. Find its volume in cubic centimeters.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{48}$}
\end{tryitbox}

\skillheading{Skill 2 --- Find and fix a squared-for-cubed mistake}

\begin{formulabox}
\textbf{How to catch it:} check the \emph{number} and the \emph{label} separately.\par
Count how many lengths were actually multiplied. If only two were multiplied but the
label says cubic, the number is an area wearing the wrong coat --- and the fix is to
multiply by the missing third length, not to change the label.\par
\textbf{Cube shortcut:} face area $=$ edge $\times$ edge; \ volume $=$ edge $\times$ edge
$\times$ edge.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A student found the volume of a cube with edges 5 centimeters long
and wrote 25 cubic centimeters. Find and fix the mistake.
\step{Only two edges were multiplied, so 25 measures one flat face --- an area number with
a cubic label, which is the classic squared-for-cubed slip.}
\step{The cube is 5 layers deep, so the missing third edge goes in:
$25 \times 5 = 125$.}
$\Rightarrow\quad$ volume $= \ans{125}$ cubic centimeters
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A cube has edges 3 centimeters long. Find its volume in cubic
centimeters.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{Skill 3 --- Convert square and cubic units}

\begin{formulabox}
\textbf{Rule: the length factor is used once per direction.}\par
If the lengths convert with a factor $f$, then square units convert with $f \times f$ and
cubic units convert with $f \times f \times f$. Never reuse the plain length factor on an
area or a volume.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many square centimeters cover one square meter? (One meter is
100 centimeters.)
\step{A square meter is a square with a meter on each side, and a square has two
directions --- so the factor 100 is used twice.}
\step{The square holds 100 rows of 100 small squares: $100 \times 100 = 10000$.}
$\Rightarrow\quad \ans{10000}$ square centimeters in one square meter
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many cubic inches fill one cubic foot? (One foot is 12
inches.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1728}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The unit is not decoration you add at the end --- it is a record
of how many lengths you multiplied. If your number came from two lengths, no label can
make it a volume; go back and find the third length.
\end{watchoutbox}
