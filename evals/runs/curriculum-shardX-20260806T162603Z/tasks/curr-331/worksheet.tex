\wstitleblock{Classifying Quadrilaterals from Their Properties}{High School Geometry}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The interior angles of any quadrilateral add to $360^\circ$.
  \item \textbf{Most specific name} means the name furthest down the family tree
        below that the stated properties actually force. If a property is not
        given, you may not assume it.
  \item Every classification answer must be \emph{justified}: name the property
        you used and say what it rules in or out.
  \item In this booklet a \emph{trapezoid} has exactly one pair of parallel
        sides.
\end{itemize}

\problem{A quadrilateral has exactly one pair of parallel sides, and its four
interior angles measure $(3x)^\circ$, $(3x)^\circ$, $(2x)^\circ$ and
$(2x)^\circ$.
\par\vspace{0.1cm}
(a) Find $x$. \ansline
\par\vspace*{2.6cm}
(b) Give the most specific name for this quadrilateral, and justify it from the
properties you were given.
\par\vspace*{2.8cm}}

\problem{The diagonals of quadrilateral $JKLM$ cross at a point that divides
each of them into two equal parts. On one diagonal those two parts measure
$(2x+1)$ cm and $13$ cm.
\par\vspace{0.1cm}
(a) Find $x$. \ansline
\par\vspace*{2.6cm}
(b) Give the most specific name that the stated property forces, and justify it.
\par\vspace*{2.8cm}}

\problem{A quadrilateral has four congruent sides. One side measures $(3x-4)$ cm
and another measures $20$ cm. Its two diagonals are also congruent to each
other.
\par\vspace{0.1cm}
(a) Find $x$. \ansline
\par\vspace*{2.6cm}
(b) Give the most specific name for this quadrilateral, and justify it from both
stated properties.
\par\vspace*{2.8cm}}

\problem{In quadrilateral $ABCD$ both pairs of opposite angles are congruent.
The four interior angles measure $(2x+10)^\circ$, $(3x-20)^\circ$,
$(2x+10)^\circ$ and $(3x-20)^\circ$.
\par\vspace{0.1cm}
(a) Find $x$. \ansline
\par\vspace*{2.4cm}
(b) Find the measure of the larger pair of angles, in degrees. \ansline
\par\vspace*{2.4cm}
(c) Classify $ABCD$ as specifically as the information allows, and state one
extra piece of information that would make it a rectangle.
\par\vspace*{2.6cm}}

\problem{Decide whether each statement is \emph{always}, \emph{sometimes} or
\emph{never} true, and explain your decision from the defining properties.
\par\vspace{0.1cm}
(a) A rhombus is a square.
\par\vspace*{2.4cm}
(b) A square is a rhombus.
\par\vspace*{2.4cm}
(c) A trapezoid is a parallelogram.
\par\vspace*{2.4cm}}

\problem{The diagonals of quadrilateral $EFGH$ are congruent to each other and
bisect each other, but they are \emph{not} perpendicular. One diagonal measures
$(4x+5)$ cm and the other measures $33$ cm.
\par\vspace{0.1cm}
(a) Give the most specific name for $EFGH$, and justify it.
\par\vspace*{2.6cm}
(b) Find $x$. \ansline
\par\vspace*{2.2cm}
(c) Explain why $EFGH$ cannot be a square.
\par\vspace*{2.4cm}}

\problem{In quadrilateral $ABCD$, $AB = AD = (3x-1)$ cm and $CB = CD = 14$ cm.
It is also known that $AB = 17$ cm.
\par\vspace{0.1cm}
(a) Find $x$. \ansline
\par\vspace*{2.6cm}
(b) Give the most specific name for $ABCD$, and explain why it is not a rhombus.
\par\vspace*{2.8cm}}

\problem{A student claims that every quadrilateral with two pairs of congruent
sides must be a parallelogram.
\par\vspace{0.1cm}
(a) Give a counterexample: name a quadrilateral that has two pairs of congruent
sides but is not a parallelogram, and describe how its congruent sides are
arranged.
\par\vspace*{2.6cm}
(b) In a different quadrilateral one pair of opposite sides measures $(5x-8)$ cm
and $27$ cm. Find the value of $x$ that makes those two sides congruent.
\ansline
\par\vspace*{2.2cm}
(c) The other pair of opposite sides measures $19$ cm and $(2y+3)$ cm. Find the
value of $y$ that makes those two sides congruent. \ansline
\par\vspace*{2.2cm}}
