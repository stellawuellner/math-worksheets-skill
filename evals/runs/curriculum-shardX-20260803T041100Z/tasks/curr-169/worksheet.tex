\wstitleblock{Subtraction and Negative Signs: Diagnosing the Error}{Grade 6--7 The Number System}{}

\noindent\textbf{Before you compute, rewrite every subtraction as an addition:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $p - q$ means $p + (-q)$ --- ``add the opposite'' of the second number.
  \item Subtracting a negative adds a positive: $p - (-q) = p + q$, so the result moves \emph{right} on the number line.
  \item Subtracting a positive adds a negative, so the result moves \emph{left}.
  \item The sign of the answer follows the number with the larger distance from zero, not the one written first.
\end{itemize}

\noindent For every find-and-fix item: name the step where the sign was lost,
then redo it correctly.

\vspace{0.15cm}

\par\vspace{2pt}
Subtracting $-3$ has the same effect as \underline{\hspace{4cm}} because
\underline{\hspace{5cm}}.}

\problem[4cm]{Order these three values from \textbf{least to greatest}. Write the
three results, not the expressions.
\par\vspace{2pt}
\[ -2 - 7 \qquad 2 - 7 \qquad -2 + 7 \]}

\par

\par

\problem[5cm]{\textbf{Challenge --- find and fix.} A student evaluates a
three-term expression left to right and gets $-12.75$:
\par\vspace{2pt}

\par
Identify the single sign the student mishandled, rewrite all of the subtractions
as additions, and compute the correct value.}
