\sstitleblock{Writing Growth and Decay Models From Data}

\begin{formulabox}
\textbf{Every model on this sheet has the same shape:} $y = a\cdot b^{\,t}$,
where $a$ is the amount at $t = 0$ and $b$ is what the amount is multiplied by
in one time period.
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item $b > 1$ grows, $0 < b < 1$ decays, and $b$ is never negative.
\item \textbf{Writing a model is finding two numbers:} $a$ from the data at
time zero, $b$ from a ratio or from a percent rate.
\item Say what $t$ counts and what $y$ measures, with its unit. A model without
units answers no question.
\item Exponential data has a constant \emph{ratio}; linear data has a constant
\emph{difference}. Test the ratio before you assume.
\end{itemize}
\end{formulabox}

\skillheading{1. Reading the factor out of a table}

\begin{formulabox}
Divide each output by the one before it --- later $\div$ earlier. If every
quotient is the same number, that number is $b$. The output at $t = 0$ is $a$.
If the table has no $t = 0$ row, see section 3.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $t$: 0, 1, 2 with values 24, 36, 54. Write the model.
\step{The values are equally spaced in time, so test the ratios: a constant
ratio is exactly what an exponential model has.}
\step{$36 \div 24$ gives the same quotient as $54 \div 36$, so the factor is
$b$, and $a = 24$ is the value at $t = 0$, giving $y = 24(1.5)^{t}$.}
\[ \ans{b = 1.5} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $t$: 0, 1, 2 with values 5, 20, 80. Find $b$ and write the
model.

\rotatebox{180}{\footnotesize check: $\ans{b = 4}$, so $y = 5(4)^{t}$}
\end{tryitbox}

\skillheading{2. Turning a percent rate into a factor}

\begin{formulabox}
Growth of $r$ per period: $b = 1 + r$. Decay of $r$ per period: $b = 1 - r$.
Write the percent as a decimal first: 20 percent decay gives
$b = 1 - 0.20 = 0.8$. The factor already contains the original amount, so never
add the percent on afterwards.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A bike costs 600 dollars and loses 20 percent of its value a
year. What is it worth after 3 years?
\step{A percent loss each year is a constant multiplier, so build the factor
first and let the exponent count the years.}
\step{Keeping 80 percent gives $b = 0.8$ and $V = 600(0.8)^{t}$; at $t = 3$ this
is $600(0.512)$.}
\[ \ans{307.20} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A town of 5000 people grows 6 percent a year. Write the model
and find the population after 2 years.

\rotatebox{180}{\footnotesize check: $P = 5000(1.06)^{t}$, giving $\ans{5618}$}
\end{tryitbox}

\skillheading{3. Two readings, no starting value}

\begin{formulabox}
If the readings are $n$ periods apart, the factor is applied $n$ times:
$(\text{later}) = (\text{earlier})\cdot b^{\,n}$. Solve for $b^{\,n}$, then take
the $n$th root. Keep the positive root only. To get $a$ afterwards, divide any
reading by $b$ raised to that reading's own time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An account holds 800 dollars at $t = 0$ and 1250 dollars at
$t = 2$ years. Find the annual factor.
\step{The two readings are 2 years apart, so $b$ has been applied twice --- that
makes the ratio a square, not the factor itself.}
\step{From $800\,b^{2} = 1250$ comes $b^{2} = 1.5625$, and the positive square
root of 1.5625 is the annual factor.}
\[ \ans{b = 1.25} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A car is worth 1000 dollars at $t = 0$ and 640 dollars at
$t = 2$ years. Find the annual factor.

\rotatebox{180}{\footnotesize check: $\ans{b = 0.8}$, so $V = 1000(0.8)^{t}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{Watch out.} Dividing the \emph{earlier} value by the later one inverts
the factor and turns decay into growth --- always later $\div$ earlier. And a
constant difference is not a constant ratio: adding 3 percent of the starting
amount every year is linear, and always predicts less than the exponential model
it was mistaken for.
\end{watchoutbox}
