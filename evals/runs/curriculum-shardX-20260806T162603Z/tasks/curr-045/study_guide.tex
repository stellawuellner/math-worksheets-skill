\sstitleblock{Lengths and Picture Graphs}

% =================== SKILL 1 ===================
\skillheading{Compare two lengths}

\begin{formulabox}
\textbf{Rule:} Measure both things with the SAME unit. More units means
longer, and subtracting the counts tells you by how much.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The rope is 10 cubes long and the stick is 6 cubes
long. How many cubes longer is the rope?
\step{Both lengths use the same unit, so the difference of the two counts is
the difference in length.}
\step{$10 - 6 = 4$.}

$\Rightarrow\quad$ the rope is $\ans{4}$ cubes longer
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A scarf is 11 clips long and a sock is 5 clips long.
How many clips longer is the scarf?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$ clips}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Read a picture graph}

\begin{formulabox}
\textbf{Rule:} First read the key: each picture stands for one thing. Then
count the pictures in the row you need.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Pets chart (each $\bullet$ = one pet): \quad
cats: $\bullet$ $\bullet$ $\bullet$ \quad dogs: $\bullet$ $\bullet$ $\bullet$
$\bullet$ $\bullet$. \ How many dogs?
\step{Find the dogs row, then count its pictures one by one.}
\step{The dogs row has five dots.}

$\Rightarrow\quad$ there are $\ans{5}$ dogs
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Using the same pets chart: how many cats?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{How many more? Compare graph rows}

\begin{formulabox}
\textbf{Rule:} ``How many more'' and ``how many fewer'' both mean find the
DIFFERENCE: subtract the smaller count from the bigger one. Do not add!
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In the pets chart, how many more dogs than cats?
\step{``How many more'' asks for the gap between the rows, so subtract the
two counts.}
\step{Dogs: 5. Cats: 3. \ $5 - 3 = 2$.}

$\Rightarrow\quad$ there are $\ans{2}$ more dogs
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A chart shows 6 birds and 2 fish. How many more birds
than fish?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``How many MORE'' does not mean add the rows
together. Line the rows up in your mind and look at the extra part --- that
extra part is the answer.
\end{watchoutbox}
