\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item An ordered pair is written $(x, y)$: count \emph{across} first, then \emph{up}. The order is the whole idea.
  \item Problems 1 to 4 all read from Park Map below. Later problems have their own blank grid to draw on.
\end{itemize}

\problem[2.5cm]{Use the Park Map. Write the ordered pair for landmark $L$.
\par\vspace{0.3cm}
$L = ($~\ansblank~$,$~\ansblank~$)$}

\problem[2.5cm]{Use the Park Map. Which labelled landmark is farthest to the right?}

\problem[2.5cm]{Use the Park Map. Start at landmark $L$, move 4 units to the right, then 5 units up. Write the ordered pair where you land.
\par\vspace{0.3cm}
$($~\ansblank~$,$~\ansblank~$)$}

\problem[2.5cm]{Use the Park Map. Landmarks $L$ and $M$ sit on the same horizontal line. How far apart are they?

\problem[3cm]{A candle is 20 centimetres tall and burns down 3 centimetres each hour. Each hour is recorded as an ordered pair (hour, height in centimetres). Write the ordered pair for hour 5. The hour has been filled in for you.

$($~5~$,$~\ansblank~$)$}

\problem[3cm]{The candle from problem 6 gives these points for hours 0 to 4: $(0, 20)$, $(1, 17)$, $(2, 14)$, $(3, 11)$, $(4, 8)$.
\begin{itemize}
  \item[(a)] Write the mean of the five $y$-coordinates.
\end{itemize}
\answerline{cm}
\begin{itemize}
  \item[(b)] Write the range of the five $y$-coordinates.
\end{itemize}
\answerline{cm}}

\problem[3.5cm]{Two rules build points. Rule 1: the $y$-coordinate is double the $x$-coordinate. Rule 2: the $y$-coordinate is 6 more than the $x$-coordinate.
\begin{itemize}
  \item[(a)] Rule 1 at $x = 4$: \quad $($~4~$,$~\ansblank~$)$
  \item[(b)] Rule 2 at $x = 4$: \quad $($~4~$,$~\ansblank~$)$
  \item[(c)] At which $x$ do the two rules give the same point? \ansline
  \item[(d)] Explain how you found that $x$.
\end{itemize}
