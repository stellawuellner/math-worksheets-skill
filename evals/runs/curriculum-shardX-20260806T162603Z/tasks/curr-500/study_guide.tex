% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Building the polynomial from derivatives}

\begin{formulabox}
\textbf{Rule:} $P_{n}(x) = \displaystyle\sum_{k=0}^{n}\frac{f^{(k)}(a)}{k!}(x-a)^{k}$. Each coefficient needs one derivative evaluated \emph{at the centre}, divided by that order's factorial. Work down the list of derivatives once and record the values at $a$; the powers of $(x-a)$ take care of themselves.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = e^{3x}$, find $f'''(x)$ and hence the cubic coefficient of the Maclaurin polynomial.
\step{The cubic coefficient is the third derivative at zero divided by three factorial, so the only real work is differentiating three times.}
\step{Each differentiation multiplies by $3$, so $f'''(x) = 27e^{3x}$, giving $f'''(0)/3! = 27/6$.}

$\Rightarrow\quad \ans{27e^{3x}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = e^{4x}$, find $f'''(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{64e^{4x}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Reusing a series you already know}

\begin{formulabox}
\textbf{Four to memorise:} $e^{x} = \sum \frac{x^{n}}{n!}$, \; $\sin x = x - \frac{x^{3}}{3!} + \frac{x^{5}}{5!} - \cdots$, \; $\cos x = 1 - \frac{x^{2}}{2!} + \frac{x^{4}}{4!} - \cdots$, \; $\frac{1}{1-x} = 1 + x + x^{2} + \cdots$. To get a new series, \emph{substitute} into one of these, or multiply it by a power of $x$ --- never differentiate from scratch when a substitution will do.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Estimate $\sin(0.3)$ using terms through $x^{5}$.
\step{The sine series is already known, so substitute the input rather than building anything --- three terms is plenty at this size.}
\step{$0.3 - \dfrac{0.3^{3}}{6} + \dfrac{0.3^{5}}{120} = 0.3 - 0.0045 + 0.00002$.}

$\Rightarrow\quad \ans{0.29552}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Estimate $\sin(0.2)$ using terms through $x^{5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.19867}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Bounding the error}

\begin{formulabox}
\textbf{Two bounds, and the condition that chooses between them.} Lagrange: $|R_{n}(x)| \le \dfrac{M\,|x-a|^{n+1}}{(n+1)!}$, where $M$ bounds $\left|f^{(n+1)}\right|$ across the whole interval --- always available. Alternating: the error is at most the first omitted term --- available only when the series alternates and its term sizes decrease to zero. Pick $M$ by asking where the next derivative is largest on the interval.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Bound the error when $P_{2}$ of $e^{x}$ estimates $e^{x}$ at $x = 1$.
\step{The series for the exponential does not alternate at a positive input, so the Lagrange form is the one available, and it needs a bound on the third derivative.}
\step{Every derivative is $e^{x}$, largest at $x = 1$, so $M = e$ and the bound is $\dfrac{e\cdot 1^{3}}{3!}$.}

$\Rightarrow\quad \ans{\frac{e}{6} \approx 0.45305}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Bound the error when $P_{4}$ of $e^{x}$ estimates $e^{x}$ at $x = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{e}{120} \approx 0.02265}$}
\end{tryitbox}
