\wstitleblock{Slope or $y$-Intercept? Find the Mix-Up}{Grade 8 / Pre-Algebra}{}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item In $y = mx + b$, the number \emph{multiplying} the variable is the
        slope: how much $y$ changes for each 1 unit of $x$. The number
        \emph{added on} is the $y$-intercept: the value of $y$ when $x$ is 0.
  \item A quick test for the intercept: substitute $x = 0$ and see what is
        left.
  \item When you name an error, say which number belongs in which place and
        why --- a corrected equation on its own is not an explanation.
\end{itemize}

\vspace{0.15cm}

\problem{A gym charges $C = 12x + 40$ dollars for $x$ months of membership.
\begin{itemize}
  \item[(a)] What is the slope --- the change in cost for each extra month?
        \par\vspace*{2.1cm}\ansline
  \item[(b)] What is the $y$-intercept --- the cost before any months are
        used?
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{A line passes through $(0, 4)$ and $(5, 19)$. Tomas wrote its
equation as $y = 4x + 3$.
\begin{itemize}
  \item[(a)] Find the slope of the line.
        \par\vspace*{1.8cm}\ansline
  \item[(b)] Using the correct equation, find $y$ when $x = 2$.
        \par\vspace*{1.8cm}\ansline
  \item[(c)] Name the error in the equation Tomas wrote.
        \par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem{Nia says that the graph of $y = 7x + 2$ crosses the $y$-axis at 7 and
rises 2 units for each 1 unit across.
\begin{itemize}
  \item[(a)] Find the value of $y$ where the graph crosses the $y$-axis.
        \par\vspace*{1.8cm}\ansline
  \item[(b)] Find how much $y$ rises for each 1 unit increase in $x$.
        \par\vspace*{1.8cm}\ansline
  \item[(c)] Name the error in the description Nia gave.
        \par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem{A phone plan costs \$25 when 0 gigabytes are used and \$45 when 5
gigabytes are used. Reza wrote the cost equation as $C = 25g + 4$ dollars for
$g$ gigabytes.
\begin{itemize}
  \item[(a)] Find the cost per gigabyte.
        \par\vspace*{1.8cm}\ansline
  \item[(b)] Using the correct equation, find the cost when 8 gigabytes are
        used.
        \par\vspace*{1.8cm}\ansline
  \item[(c)] Name the error in the equation Reza wrote.
        \par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem{A taxi's total charge is shown in the table. Kai says the equation is
$y = 3x + 4$ dollars for $x$ miles.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Miles} & 0 & 2 & 4 & 6 \\ \hline
\textbf{Total charge (\$)} & 3 & 11 & 19 & 27 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What is the charge at 0 miles?
        \par\vspace*{1.7cm}\ansline
  \item[(b)] Find the charge per mile.
        \par\vspace*{1.7cm}\ansline
  \item[(c)] Name the error in the equation Kai wrote.
        \par\vspace*{1.7cm}\ansline
\end{itemize}}

\problem{A candle is 20 cm tall and burns down 3 cm per hour, so its height
after $t$ hours is $h = 20 - 3t$ centimetres. Lena wrote $h = 3 - 20t$ instead.
\begin{itemize}
  \item[(a)] Using the correct equation, find the candle's height after 4
        hours.
        \par\vspace*{2.1cm}\ansline
  \item[(b)] Name the error in the equation Lena wrote.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{Two lines are given by $y = 5x + 2$ and $y = 2x + 5$.
\begin{itemize}
  \item[(a)] Compare their slopes using $>$ or $<$, first line then second.
        \par\vspace*{1.7cm}\ansline
  \item[(b)] Compare their $y$-intercepts using $>$ or $<$, first line then
        second.
        \par\vspace*{1.7cm}\ansline
  \item[(c)] Explain why the two equations describe different lines even
        though they are built from the same two numbers.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem{A savings account holds \$60 at week 0 and \$132 at week 8. Marcus
wrote the balance equation as $y = 60x + 9$ dollars after $x$ weeks.
\begin{itemize}
  \item[(a)] Find the amount saved per week.
        \par\vspace*{1.7cm}\ansline
  \item[(b)] Using the correct equation, find the balance at week 10.
        \par\vspace*{1.7cm}\ansline
  \item[(c)] Explain how you could tell that the balance Marcus would predict
        for week 10 is far too large, without doing his calculation.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}
