\sstitleblock{Finding Inverses with Domain Restrictions}

\skillheading{Skill 1 --- Inverting a measured model}

\begin{formulabox}
\textbf{Method.} Write the rule as output $=$ (expression in the input), then solve for the input. The result is the inverse rule.\par
\textbf{Units swap with the variables.} If the rule takes hours and returns dollars, the inverse takes dollars and returns hours --- so answer in the inverse's unit, not the original's.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A taxi charges $p(k) = 3k + 8$ dollars for $k$ kilometres. A fare was $41$ dollars. How far was the trip?
\step{The fare is known and the distance is wanted, which is backwards from the rule --- so invert it rather than guessing values.}
\step{$p = 3k + 8$ gives $k = \dfrac{p-8}{3}$, and $\dfrac{41-8}{3} = \dfrac{33}{3}$.}

$\Rightarrow\quad \ans{k = 11 \text{ km}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A printer charges $p(n) = 4n + 6$ dollars for $n$ posters. A bill was $46$ dollars. How many posters?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Restricting the domain, then inverting}

\begin{formulabox}
\textbf{Why restrict.} A rule with $x^{2}$ in it sends two inputs to the same output, so it has no inverse until one side is thrown away.\par
\textbf{Recipe.} Complete the square to find the vertex, restrict to one side of it, solve for $x$, and let the restriction choose the sign of the square root. The domain of $f^{-1}$ is the range of the restricted $f$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = (x-2)^{2} - 9$ on $x \geq 2$. Find $f^{-1}(16)$.
\step{The restriction starts at the vertex, so the inverse takes the plus square root --- that decision has to be made before any arithmetic.}
\step{$f^{-1}(x) = 2 + \sqrt{x+9}$, so $f^{-1}(16) = 2 + \sqrt{25} = 2 + 5$.}

$\Rightarrow\quad \ans{f^{-1}(16) = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = (x-5)^{2} - 1$ on $x \geq 5$. Find $f^{-1}(35)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f^{-1}(35) = 11}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the minus square root is not wrong in general --- it is the inverse of the \emph{other} branch. Which one you keep is decided by the restriction, so write the restriction down first.
\end{watchoutbox}

\skillheading{Skill 3 --- Checking that two rules undo each other}

\begin{formulabox}
\textbf{Test.} $g = f^{-1}$ exactly when $f\bigl(g(x)\bigr) = x$ and $g\bigl(f(x)\bigr) = x$ on the stated domains.\par
For a rational rule, substitute and then clear the compound fraction by multiplying the top and bottom by the inner denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 5x - 2$, $g(x) = \dfrac{x+2}{5}$. Show $(f \circ g)(x) = x$.
\step{Composing is the definition of ``undoes'', so substitute $g$ into $f$ rather than comparing graphs.}
\step{$5 \cdot \dfrac{x+2}{5} - 2 = (x + 2) - 2$.}

$\Rightarrow\quad \ans{(f \circ g)(x) = x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = \dfrac{7x+1}{x-2}$, $g(x) = \dfrac{2x+1}{x-7}$. Show $(f \circ g)(x) = x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(f \circ g)(x) = x}$}
\end{tryitbox}
