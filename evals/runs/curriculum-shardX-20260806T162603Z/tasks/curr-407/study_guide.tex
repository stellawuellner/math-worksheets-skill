\sstitleblock{Locating Zeros and Turning-Point Constraints}

\skillheading{Skill 1 --- Factoring completely to locate the zeros}

\begin{formulabox}
\textbf{Order of attack:} common factor first, then a difference of squares, then a quadratic pattern (including a quadratic in $x^{2}$), then grouping or a known zero.\par
Pulling out the common $x$ is what keeps the zero at $0$ visible --- dividing it away loses a real zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find all real zeros of $p(x) = x^{3} - 2x^{2} - 8x$.
\step{Every term carries an $x$, so factor that out first; the leftover quadratic is then an ordinary trinomial.}
\step{$x\left(x^{2} - 2x - 8\right) = x(x - 4)(x + 2)$.}

$\Rightarrow\quad \ans{x = 0}, \ \ans{x = 4}, \ \ans{x = -2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find all real zeros of $p(x) = x^{3} - x^{2} - 12x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 0}$, $\ans{x = 4}$, $\ans{x = -3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Multiplicity against degree}

\begin{formulabox}
\textbf{Rule.} A factor $(x - r)^{m}$ gives a zero at $r$ of multiplicity $m$, and the multiplicities add up to the degree.\par
Odd multiplicity means the curve crosses; even multiplicity means it touches and turns back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the zeros of $p(x) = (x - 2)^{2}(x + 5)$ and check the degree.
\step{The factored form already shows both zeros, so read the exponents rather than expanding.}
\step{Multiplicity $2$ at $x = 2$ (a touch) and multiplicity $1$ at $x = -5$ (a crossing); $2 + 1 = 3$, the degree.}

$\Rightarrow\quad \ans{x = 2}, \ \ans{x = -5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the zeros of $p(x) = (x + 3)^{2}(x - 1)$ and say which one is a touch.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -3}$ touch, $\ans{x = 1}$ crossing}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a repeated factor does not always mean a touch. $x^{3}$ is a repeated factor with multiplicity $3$, which is odd, so the curve crosses there.
\end{watchoutbox}

\skillheading{Skill 3 --- Bounding the turning points}

\begin{formulabox}
\textbf{Rule.} A polynomial of degree $n$ has at most $n$ real zeros and at most $n - 1$ turning points --- its derivative has degree $n - 1$, so it can vanish at most $n - 1$ times.\par
From below: each gap between two consecutive simple zeros forces at least one turning point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A polynomial has degree $7$. What is the greatest possible number of turning points?
\step{The bound comes from the degree alone, so no coefficients are needed --- subtract one.}
\step{$7 - 1 = 6$.}

$\Rightarrow\quad \ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A polynomial has degree $10$. Greatest possible number of turning points?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}
