\wstitleblock{Writing and Interpreting Circle Equations}{High-School Geometry --- Circles}{}

\noindent\small Reading the center and radius from an equation, writing an equation from a center and radius, completing the square, and applying circle equations

\medskip
\noindent\textbf{Reference for every problem on this sheet:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Standard form: $(x-h)^2 + (y-k)^2 = r^2$ --- center $(h, k)$, radius $r$. The number on the right is $r^2$, never $r$.
  \item A point lies on the circle exactly when its coordinates make the left side equal $r^2$.
  \item General form $x^2 + y^2 + Dx + Ey + F = 0$ becomes standard form by completing the square in $x$ and in $y$.
  \item Give exact values unless the problem asks for a decimal, and state the unit named in the problem.
\end{itemize}
\normalsize

\problem[1cm]{An air-traffic radar screen shows every plane within range of the tower.
The edge of the range is the circle $(x-4)^2 + (y+3)^2 = 49$, where $x$ and $y$ are in miles.
\begin{enumerate}[label=(\alph*), itemsep=2.6cm, leftmargin=1.5cm, topsep=4pt]
  \item Write the coordinates of the tower (the center of the circle). \ansline
  \item Find the radius of the radar range, in miles. \answerline{mi}
  \item A plane is at $(11, -3)$. Show by computing a distance that it is exactly on the edge of the range. \ansline
\end{enumerate}}

\problem[6cm]{A sprinkler stands at the point $(-2, 5)$ on a garden plan, with distances
in meters. It waters every point within $6$ meters of itself.
Write the equation of the circle that forms the boundary of the watered region,
in standard form.}

\problem[1cm]{On a map with distances in kilometres, a cell tower stands at $(3, -2)$.
Its signal reaches exactly as far as a town at $(8, -2)$, and no farther.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the radius of the coverage circle, in kilometres. \answerline{km}
  \item Write the equation of the coverage boundary in standard form. \ansline
\end{enumerate}}

\problem[6cm]{A stone dropped in a pond makes a ripple whose edge, at one instant, is
$x^2 + (y-6)^2 = 20$, with $x$ and $y$ in meters.
Write the center of the ripple, then find its radius in meters, rounded to two decimal places.
\answerline{m}}

\problem[7cm]{A circular running track is described by
\[ x^2 + y^2 - 6x + 8y - 11 = 0 \]
with $x$ and $y$ in meters.
Complete the square in $x$ and in $y$ to rewrite the equation in standard form,
then state the center and the radius of the track in meters.
\answerline{m}}

\problem[7cm]{A circular fence is modelled by $(x-2)^2 + (y-1)^2 = 25$, in meters,
and a straight path runs along the line $y = 4$.
Find both $x$-coordinates where the path meets the fence.
Show the equation you solved after substituting $y = 4$.}

\problem[7cm]{Rewrite
\[ x^2 + y^2 + 10x - 4y + 20 = 0 \]
in standard form by completing the square, then state the center and the radius in units.
\answerline{units}}

\problem[1cm]{A classmate looks at $x^2 + y^2 + 4x + 6y + 20 = 0$ and says
``the center is $(-2, -3)$ and the radius is $20$.''
\begin{enumerate}[label=(\alph*), itemsep=3.2cm, leftmargin=1.5cm, topsep=4pt]
  \item Complete the square in $x$ and in $y$ and find the value that ends up on the right-hand side. \ansline
  \item Explain in two or three complete sentences what that value tells you about the graph of the equation, and say what is wrong with your classmate's claim. \noansline
\end{enumerate}}
