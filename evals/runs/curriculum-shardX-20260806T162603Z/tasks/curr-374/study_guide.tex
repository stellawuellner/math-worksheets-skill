\sstitleblock{Radicals Do Not Split Across a Sum}

\skillheading{Skill 1 --- Test a suspect rule with numbers}

\begin{formulabox}
\textbf{Method:} one counterexample kills a proposed identity. Pick two easy
numbers, evaluate both sides, and compare. If they differ even once, the rule is
false --- no further argument is needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $\sqrt{a+b} = \sqrt{a}+\sqrt{b}$?
\step{A claimed identity has to hold for every choice, so the cheapest test is to
try one pair of perfect squares and compare the two sides.}
\step{Take $a = 9$ and $b = 16$: the left side is $\sqrt{25}$, and the right side
is $3+4$.}

left side $\ans{5}$, \quad right side $\ans{7}$ --- so the rule is false.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Test the same claim with $a = 36$ and $b = 64$.\\[2pt]
\rotatebox{180}{\footnotesize check: left side $\ans{10}$, right side $\ans{14}$}
\end{tryitbox}

\skillheading{Skill 2 --- Roots split over products, not sums}

\begin{formulabox}
\textbf{The rules that do hold} (for $a, b \ge 0$):
$\sqrt{ab} = \sqrt{a}\,\sqrt{b}$ \quad and \quad
$\sqrt{\dfrac{a}{b}} = \dfrac{\sqrt{a}}{\sqrt{b}}$.
There is no rule for $\sqrt{a+b}$. To simplify a radical, look for a
perfect-square \emph{factor}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\sqrt{48}$.
\step{Simplifying means pulling out a perfect square, and that is legal only
because $48$ is being written as a product, not as a sum.}
\step{$48 = 16 \cdot 3$, so $\sqrt{48} = \sqrt{16}\sqrt{3} = 4\sqrt{3}$.}

$\Rightarrow\quad \ans{4\sqrt{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\sqrt{98}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7\sqrt{2}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Spotting the split-over-a-sum error}

\begin{formulabox}
\textbf{Warning sign:} a step that turns $\sqrt{\text{something} + \text{something}}$
into two separate roots, or turns $\sqrt{A}+\sqrt{B}$ into one root. Both are the
same illegal move. A sum of two squares such as $x^2+16$ does not factor over the
real numbers, so its square root is already in simplest form.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A student writes $\sqrt{x^2+16} = x+4$. Test it at
$x = 6$.
\step{The claim is an identity, so substituting one convenient value and
comparing the two sides is enough to accept or reject it.}
\step{Left side: $\sqrt{36+16} = \sqrt{52} = 7.2111\ldots$; right side:
$6+4$.}

left side $\ans{7.21}$, \quad right side $\ans{10}$ --- the claim fails.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Test the claim $\sqrt{x^2+25} = x+5$ at $x = 12$.\\[2pt]
\rotatebox{180}{\footnotesize check: left side $\ans{13}$, right side $\ans{17}$}
\end{tryitbox}
