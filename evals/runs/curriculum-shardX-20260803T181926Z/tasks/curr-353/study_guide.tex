\sstitleblock{Dividing Complex Numbers with Conjugates}

\noindent\small A quotient is not finished until the denominator is real.
The conjugate of $a + bi$ is $a - bi$: same real part, opposite imaginary
part. Multiplying top and bottom by it never changes the value, only the form.
\normalsize

\skillheading{1. Why the conjugate clears a denominator}
\begin{formulabox}
\textbf{Rule:} $(a+bi)(a-bi) = a^2 - b^2i^2 = a^2 + b^2$, a real number.
The outer and inner products cancel, and the $-b^2i^2$ becomes $+b^2$.
So the conjugate is the one multiplier guaranteed to make a denominator real.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Multiply $6 + 5i$ by its conjugate.
\step{The conjugate flips the sign of the imaginary part only, so it is
$6 - 5i$; the point is to watch the $i$ terms disappear.}
\step{$(6+5i)(6-5i) = 36 - 30i + 30i - 25i^2 = 36 + 25$.}
\par
$\ans{61}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Multiply $3 + 7i$ by its conjugate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{58}$}
\end{tryitbox}

\skillheading{2. Dividing by a pure imaginary number}
\begin{formulabox}
\textbf{Rule:} the conjugate of $bi$ is $-bi$, and $(bi)(-bi) = b^2$.
Useful shortcut: $\dfrac{1}{i} = -i$, so dividing by $i$ is multiplying by
$-i$. Write the answer as $a + bi$, real part first.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Simplify $\dfrac{10 + 15i}{5i}$.
\step{The denominator has no real part, so its conjugate is just $-5i$ ---
one multiplication clears it.}
\step{$\dfrac{(10+15i)(-5i)}{(5i)(-5i)} = \dfrac{-50i + 75}{25}
      = \dfrac{75 - 50i}{25}$.}
\par
$\ans{3-2i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Simplify $\dfrac{8 - 2i}{2i}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1-4i}$}
\end{tryitbox}

\skillheading{3. Dividing by a full complex denominator}
\begin{formulabox}
\textbf{Rule:} multiply top and bottom by the conjugate of the denominator,
then divide \emph{both} parts of the new numerator by the real denominator:
$\dfrac{z}{a+bi} = \dfrac{z(a-bi)}{a^2+b^2}$. Dividing real by real and
imaginary by imaginary is not a legal move.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Simplify $\dfrac{11 + 3i}{3 + 2i}$.
\step{The denominator is a full binomial, so the conjugate $3 - 2i$ is what
turns the bottom into the real number $9 + 4$.}
\step{$\dfrac{(11+3i)(3-2i)}{13} = \dfrac{33 - 22i + 9i + 6}{13}
      = \dfrac{39 - 13i}{13}$.}
\par
$\ans{3-i}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Simplify $\dfrac{5 + 5i}{1 + 3i}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2-i}$}
\end{tryitbox}

\skillheading{4. Circuits: voltage, current and impedance}
\begin{formulabox}
\textbf{Model:} in an AC circuit $V = I \cdot Z$, where voltage $V$ is in
volts, current $I$ in amps and impedance $Z$ in ohms --- all three complex.
So $Z = V / I$ and $I = V / Z$: every missing quantity is one division by the
quantity you already know.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $V = 1 + 7i$ volts across an impedance of $1 + 2i$
ohms. Find the current.
\step{Current is wanted and $V$ and $Z$ are known, so rearrange $V = IZ$ to
$I = V/Z$ --- a division by the impedance.}
\step{$\dfrac{(1+7i)(1-2i)}{1 + 4} = \dfrac{1 - 2i + 7i + 14}{5}
      = \dfrac{15 + 5i}{5}$.}
\par
$\ans{3+i}$ amps
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $V = 4 + 7i$ volts, $Z = 2 + i$ ohms. Find the current in
amps.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3+2i}$}
\end{tryitbox}
