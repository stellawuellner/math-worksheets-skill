% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Find the Missing Part of a Bond}

\begin{formulabox}
\textbf{Rule:} In a number bond, the two \textbf{parts} always add up to the
\textbf{whole}: \ part $+$ part $=$ whole.\par
If one part is hiding, count on from the part you know up to the whole.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The whole is 8. One part is 6. Find the missing part.
\step{We know the whole and one part, so we count on from the part we know up to the whole.}
\step{Start at 6 and count on to 8: \ 7, 8 --- that is 2 counts, and $6 + 2 = 8$. $\checkmark$}

$\Rightarrow\quad$ the missing part is $\ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The whole is 9. One part is 6. What is the missing
part?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Finish a Bond's Fact Family}

\begin{formulabox}
\textbf{Rule:} One bond makes \textbf{four facts} with the same three
numbers:\par
part $+$ part $=$ whole \ (both orders), and whole $-$ part $=$ other part
\ (both parts).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Bond: whole 7, parts 2 and 5. Three facts are done:
$2+5=7$, \ $5+2=7$, \ $7-5=2$. Write the missing fact.
\step{Every fact uses the same three numbers, so we look for the subtraction that has not been used: taking the part 2 from the whole.}
\step{$7 - 2$ leaves the other part: \ $7 - 2 = 5$. $\checkmark$}

$\Rightarrow\quad$ the missing fact is $7 - 2 = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Bond: whole 8, parts 3 and 5. Write the answer to the
family fact \ $8 - 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Check a Bond and Fix It}

\begin{formulabox}
\textbf{Rule:} To check a bond, add the two parts. If they do not make the
whole, one number is wrong --- fix it so part $+$ part $=$ whole again.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Mia's bond says the whole is 9 with parts 4 and 6.
Fix the second part.
\step{We check a bond by adding its parts, and $4 + 6$ makes more than the whole 9, so the 6 must change.}
\step{Count on from 4 up to 9: \ 5, 6, 7, 8, 9 --- that is 5 counts, and $4 + 5 = 9$. $\checkmark$}

$\Rightarrow\quad$ the fixed part is $\ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A bond says the whole is 7 with parts 3 and 5. The 5
is wrong --- what should it be?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} In a fact family, do not write the same fact two
times! Swapping the parts makes a \emph{new} addition fact, but a copied
subtraction fact is still just one fact.
\end{watchoutbox}
