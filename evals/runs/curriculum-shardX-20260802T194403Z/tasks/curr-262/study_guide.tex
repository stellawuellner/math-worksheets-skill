\sstitleblock{Solving Linear Systems by Substitution}

\skillheading{1. One variable is already isolated}

\begin{formulabox}
\textbf{The method:} substitution replaces one variable by an \emph{equal
expression}, turning two equations in two unknowns into \textbf{one} equation in
one unknown.\\[3pt]
\textbf{(1)} Find the equation solved for a variable. \quad
\textbf{(2)} Put that expression --- in parentheses --- into the \emph{other}
equation.\\
\textbf{(3)} Solve for the one remaining variable. \quad
\textbf{(4)} Back-substitute to get the partner. \quad
\textbf{(5)} Check in the untouched equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $y = x + 2$ \; and \; $2x + y = 11$
\step{The first equation already says what $y$ equals, so substituting it into the second leaves an equation in $x$ alone --- no isolating work needed.}
\step{$2x + (x + 2) = 11 \Rightarrow 3x + 2 = 11 \Rightarrow x = 3$; then $y = 3 + 2 = 5$. Check: $2(3) + 5 = 11$ $\checkmark$}
Solution: $\ans{(3,\,5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $y = 2x - 1$ \; and \; $3x + y = 19$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(4,\,7)}$}
\end{tryitbox}

\skillheading{2. Isolate a variable first (isolate-then-substitute)}

\begin{formulabox}
\textbf{Rule:} when nothing is isolated, isolate the variable whose coefficient
is $1$ or $-1$ --- that choice keeps fractions out of the substitution.\\[3pt]
From $x + y = 9$: \; $y = 9 - x$. \quad Substituting a subtraction needs
parentheses: $2x - (9 - x) = 2x - 9 + x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $x + y = 9$ \; and \; $2x - y = 3$
\step{Neither equation is solved for a variable, and $y$ in the first has coefficient $1$, so isolating there is the fraction-free choice.}
\step{$y = 9 - x$, so $2x - (9 - x) = 3 \Rightarrow 3x - 9 = 3 \Rightarrow x = 4$; then $y = 9 - 4 = 5$. Check: $2(4) - 5 = 3$ $\checkmark$}
Solution: $\ans{(4,\,5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $x + 2y = 11$ \; and \; $3x - y = 5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\,4)}$}
\end{tryitbox}

\skillheading{3. When both variables disappear (special-case)}

\begin{formulabox}
\textbf{Read what is left behind:}\\[3pt]
$-6 = 1$ or $12 = 5$ (\textbf{false}) $\Rightarrow$ \textbf{no solution}; the
lines are parallel.\\
$8 = 8$ or $0 = 0$ (\textbf{true}) $\Rightarrow$ \textbf{infinitely many
solutions}; the two equations describe the same line.\\[3pt]
Losing the variables is information, not a mistake --- never write ``$x = 0$''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $2x + y = 6$ \; and \; $4x + 2y = 5$
\step{Nothing is isolated, so isolate $y = 6 - 2x$ and substitute --- then read whatever statement survives.}
\step{$4x + 2(6 - 2x) = 5 \Rightarrow 4x + 12 - 4x = 5 \Rightarrow 12 = 5$, which is false.}
No pair can work: $\ans{\text{no solution}}$ (parallel lines).
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $y = 3x - 4$ \; and \; $6x - 2y = 8$. Substitute, say what
the surviving statement means, then give one pair that works.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{infinitely many; e.g. } (2,\,2)}$}
\end{tryitbox}

\skillheading{4. Building the system from a story (word-model)}

\begin{formulabox}
\textbf{Two unknowns need two equations.} Name both unknowns with units, then
write one equation per \emph{kind} of total in the story:\\[3pt]
a \textbf{count} equation (how many things) and a \textbf{value} equation (money,
acid, length). Solve the count equation for a variable --- its coefficients are
$1$ --- and substitute into the value equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $12$ coins, dimes and quarters, worth $255$ cents. How many of each?
\step{Let $x$ be dimes and $y$ quarters: the count total gives one equation and the money total the other, so substitution starts from the count equation.}
\step{$x + y = 12 \Rightarrow x = 12 - y$; then $10(12 - y) + 25y = 255 \Rightarrow 120 + 15y = 255 \Rightarrow y = 9$, so $x = 3$.}
Dimes and quarters: $\ans{(3,\,9)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $15$ tickets, adult at $6$ dollars and student at $4$
dollars, worth $76$ dollars. Find $(x, y)$ with $x$ adult tickets.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(8,\,7)}$}
\end{tryitbox}
