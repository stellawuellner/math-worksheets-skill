\wstitleblock{Binomial Coefficients in Probability and Expansion}{Precalculus}{}

\begin{center}
\small Pascal's triangle, binomial expansions, single terms, and binomial probability
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $\displaystyle \binom{n}{k} = \frac{n!}{k!\,(n-k)!}$ counts the ways to choose $k$ objects from $n$.
  \item Give exact values for counts and expansions; round probabilities as each problem asks.
  \item The four subskills are mixed on purpose after problems 1--2 --- decide which tool each problem needs before you start computing.
\end{itemize}

\vspace{0.15cm}

\problem[4cm]{Evaluate $\dbinom{6}{2}$.}

\problem[4.5cm]{Expand $(x + 2)^3$ and write the result in descending powers of $x$.}

\problem[4.5cm]{A fair coin is flipped $5$ times, giving $32$ equally likely sequences.
Exactly $\dbinom{5}{2} = 10$ of them contain exactly two heads. Find the probability of
getting exactly two heads.}

\problem[4.5cm]{In the expansion of $(x + 3)^5$, find the coefficient of $x^3$.}

\problem[4.5cm]{Row $7$ of Pascal's triangle contains the entries $\dbinom{7}{2} = 21$ and
$\dbinom{7}{3} = 35$. Use the addition rule for Pascal's triangle to find $\dbinom{8}{3}$
without computing any factorials.}

\problem[5cm]{Expand $(2x - 1)^4$ and write the result in descending powers of $x$.}

\problem[5cm]{A basketball player makes each free throw with probability $0.4$, independently.
She takes $5$ free throws. Find the probability that she makes exactly $3$ of them. Round to
four decimal places.}

\problem[4.5cm]{In the expansion of $\left(x^2 + 3\right)^3$, find the coefficient of $x^4$.}

\problem[3.5cm]{The same player takes $5$ free throws, making each with probability $0.4$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the probability that she makes at least $4$ of the $5$. Round to three decimal places. \quad \ansblank
  \item[(b)] Explain why this answer needs two binomial terms added together rather than one.
\end{itemize}
\vspace*{3cm}}

\problem[3.5cm]{Consider $(1 + x)^4$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Expand it in ascending powers of $x$. \quad \ansblank
  \item[(b)] Explain what the coefficient of $x^k$ counts, in terms of choosing from the four factors.
\end{itemize}
\vspace*{2.8cm}}

\problem[3.5cm]{Consider row $6$ of Pascal's triangle: $1, 6, 15, 20, 15, 6, 1$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the sum of the entries in that row. \quad \ansblank
  \item[(b)] Explain why the sum of row $n$ is always $2^n$.
\end{itemize}
\vspace*{2.8cm}}

\problem[3.5cm]{\textbf{Synthesis.} Consider the expansion of $(2x + 3)^7$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the coefficient of $x^5$. \quad \ansblank
  \item[(b)] Explain why the power of $3$ in that term must be $2$.
\end{itemize}
