\skillheading{Skill 1 --- End behavior: degree and lead coefficient}

\begin{formulabox}\textbf{Rule:}\ For large $|x|$ the leading term dominates
every other term, so only two facts matter. \textbf{Degree parity} says whether
the ends agree (even: agree; odd: disagree). \textbf{Sign of the leading
coefficient} says which way the right end goes (positive: up; negative: down).
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Degree} & \textbf{Lead. coeff.} & \textbf{Left} & \textbf{Right} \\ \hline
even & positive & up & up \\ \hline
even & negative & down & down \\ \hline
odd & positive & down & up \\ \hline
odd & negative & up & down \\ \hline
\end{tabular}
\end{center}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad End behavior of
$f(x) = -4x^3 + x$ as $x \to +\infty$.
\step{Only the leading term matters at the ends: odd degree with a negative
coefficient, so the ends disagree and the right one falls.}
\step{$f(x) \to \ans{-\infty}$ as $x \to +\infty$ (and $+\infty$ on the left)}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad What does $f(x) = 5x^4 - 2x$ do as
$x \to -\infty$?
\ \rotatebox{180}{\footnotesize check: $\ans{+\infty}$}\end{tryitbox}

\skillheading{Skill 2 --- Zeros and what multiplicity does there}

\begin{formulabox}\textbf{Rule:}\ Factored form hands you the zeros: set each
factor to zero. The exponent on that factor is its \emph{multiplicity}.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Odd} multiplicity: the graph \textbf{crosses} the axis (a
        multiplicity of three crosses flattened out).
  \item \textbf{Even} multiplicity: the graph \textbf{touches and turns back},
        and the sign does not change.
\end{itemize}
Degree $=$ the sum of the multiplicities, so the zeros also cap how many turns
the curve can have.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Zeros of $f(x) = (x - 4)(x + 1)^2$.
\step{The function is already factored, so read each factor and its exponent
rather than expanding anything.}
\step{$x - 4 = 0$ gives $x = 4$ (multiplicity one, crosses);
$(x + 1)^2 = 0$ gives $x = -1$ (multiplicity two, touches):
$\ans{x = 4 \text{ (crosses)}, \ x = -1 \text{ (touches)}}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad List the zeros of $f(x) = x^2(x - 5)$ and
say what the graph does at each.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 0}$ touches, $\ans{x = 5}$ crosses}\end{tryitbox}

\skillheading{Skill 3 --- Signs, the $y$-intercept, and the sketch}

\begin{formulabox}\textbf{Rule:}\ Between consecutive zeros the sign of a
polynomial cannot change, so \emph{one} test value decides a whole interval.
Order of work for a sketch: end behavior $\to$ zeros with multiplicities $\to$
$f(0)$ $\to$ one test value per remaining interval $\to$ join the pieces with a
smooth curve.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Find the $y$-intercept of
$f(x) = (x + 1)(x - 2)^2$ and give the sign of $f$ just right of $x = 0$.
\step{The $y$-intercept is the test value at $x = 0$, which also fixes the sign
on the whole interval between the zeros $-1$ and $2$.}
\step{$f(0) = (1)(-2)^2 = \ans{4}$, positive --- so $f > 0$ across
$(-1, 2)$}
\end{examplebox}
