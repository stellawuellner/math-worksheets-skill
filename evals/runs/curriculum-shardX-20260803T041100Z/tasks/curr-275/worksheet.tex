\wstitleblock{Arithmetic Sequences as Linear Functions}{Algebra 1 \quad Sequences and Linear Modeling}{}

\begin{center}
\textit{Explicit formulas for discrete linear change, solving for which term a
value is, finite arithmetic sums, and the link between a sequence and a linear
function}
\end{center}

\noindent\textbf{The three tools on this sheet.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Explicit formula:} $a_n = a_1 + (n-1)d$ --- the $n$th term of a
sequence whose first term is $a_1$ and whose common difference is $d$.
\item \textbf{As a linear function:} the same rule rearranges to $f(n) = dn + c$,
a line with slope $d$ sampled only at whole numbers $n = 1, 2, 3, \ldots$
\item \textbf{Finite sum:} $S_n = \dfrac{n(a_1 + a_n)}{2}$ --- pair the first
term with the last, the second with the second-to-last, and so on.
\end{itemize}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Decide first whether the question asks for \emph{one term}, for
\emph{which term}, or for a \emph{total}. The three questions use different
tools and the problems do not stay in one mode.
\item Show the values of $a_1$, $d$ and $n$ you used before substituting.
\item Write your final answer on the answer line.
\end{itemize}

\problem[5cm]{An arithmetic sequence has first term $a_1 = 3$ and common
difference $d = 5$.

\smallskip
Use the explicit formula to find the 7th term. Show the substitution before you
simplify.}

\problem[5cm]{The sequence $4, 7, 10, \ldots$ can be written as the linear
function $f(n) = 3n + 1$, where $n$ is the term number.

\smallskip
Find $f(10)$, the 10th term.}

\problem[8cm]{Find the total:
\[
1 + 3 + 5 + \cdots + 19
\]
These are the first 10 odd numbers, an arithmetic sequence with first term 1.

\smallskip
\textbf{(a)} Pair the first term with the last, the second with the second-last,
and so on. What is each pair worth, and how many pairs are there?

\smallskip
\textbf{(b)} Use the pairing to find the sum.}

\problem[6cm]{The sequence $5, 9, 13, \ldots$ keeps increasing by the same
amount.

\smallskip
One of its terms is 45. \textbf{Which} term is it? Set up the explicit formula
as an equation and solve for $n$.}

\problem[5.5cm]{An arithmetic sequence starts at $a_1 = 20$ and decreases by 3
each time.

\smallskip
\textbf{(a)} What is the common difference $d$? Write it with its sign.

\smallskip
\textbf{(b)} Find the 12th term.}

\problem[8cm]{An auditorium has 12 rows. The front row has 18 seats, and each
row behind it has 2 more seats than the row in front.

\smallskip
\textbf{(a)} How many seats are in the last row?

\smallskip
\textbf{(b)} Find the total number of seats in the auditorium. Put the total on
the answer line.}

\problem[6cm]{The sequence $7, 11, 15, \ldots$ is arithmetic.

\smallskip
\textbf{(a)} Written as a linear function of the term number it is
$f(n) = 4n + 3$. Explain in one sentence where the 4 and the 3 come from.

\smallskip
\textbf{(b)} Find $f(25)$.}

\problem[6.5cm]{Priya has 150 dollars saved at the end of month 1, and she adds
35 dollars every month after that. Her balance is an arithmetic sequence.

\smallskip
In which month does her balance first reach 500 dollars? Write the explicit
formula as an equation, solve for $n$, and check that your $n$ is a whole
number.}

\problem[8cm]{Find the sum of the first 20 terms of the arithmetic sequence
$3, 7, 11, \ldots$

\smallskip
\textbf{(a)} Find the 20th term first.

\smallskip
\textbf{(b)} Then find the sum. Be careful: the sum is not 20 copies of the
last term.}

\problem[7cm]{In an arithmetic sequence, the 4th term is 14 and the 9th term is
34.

\smallskip
\textbf{(a)} How many common differences separate the 4th term from the 9th
term?

\smallskip
\textbf{(b)} Use that to find $d$. Put $d$ on the answer line.

\smallskip
\textbf{(c)} Find the first term $a_1$.}

\problem[8cm]{A theatre has 15 rows. Row 1 has 20 seats and each row behind has
3 more seats than the one in front.

\smallskip
\textbf{(a)} Write the explicit formula for the number of seats in row $k$.

\smallskip
\textbf{(b)} Find the total seating capacity of the theatre.}

\problem[9cm]{\textbf{Challenge.} A small factory produces 120 units in week 1,
and production increases by 15 units every week after that.

\smallskip
\textbf{(a)} How many units are produced in week 20? Put this value on the
answer line.

\smallskip
\textbf{(b)} How many units are produced in total over the first 20 weeks?

\smallskip
\textbf{(c)} A manager estimates the 20-week total by multiplying the week-20
output by 20. Explain in one sentence why that overestimates, and say what the
correct averaging idea is.}
