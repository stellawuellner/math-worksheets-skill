\sstitleblock{Graphing Parabolas from Vertex or Standard Form}

\skillheading{Skill 1 --- Graphing from vertex form}

\begin{formulabox}
\textbf{Form:} \quad $y = a(x-h)^2 + k$ \quad gives the vertex $(h, k)$ straight away.\par\smallskip
$a > 0$ opens upward (vertex is the lowest point); $a < 0$ opens downward (highest point). $|a| > 1$ makes the curve narrower. The $x$-intercepts come from setting $y = 0$ and taking square roots; the $y$-intercept comes from putting $x = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the $x$-intercepts of $y = (x-3)^2 - 4$.
\step{The squared bracket is already alone once the $4$ moves across, so square roots are the route --- no factoring or formula needed.}
\step{$(x-3)^2 = 4$, so $x - 3 = \pm 2$, giving $x = 3+2$ and $x = 3-2$.}
\par
$x = \ans{1}$ \ or \ $x = \ans{5}$, and the vertex $(3, -4)$ sits midway between them.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the $x$-intercepts of $y = (x+2)^2 - 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-5}$ or $x = \ans{1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the vertex of $y = (x+2)^2 - 9$ is at $x = -2$, not $x = 2$. The form is $(x-h)$, so a plus sign inside the bracket means the graph has moved \emph{left}.
\end{watchoutbox}

\skillheading{Skill 2 --- Graphing from standard form}

\begin{formulabox}
\textbf{Form:} \quad $y = ax^2 + bx + c$.\par\smallskip
\textbf{Axis of symmetry:} $x = -\dfrac{b}{2a}$ --- find this first, because the vertex sits on it. \textbf{Vertex:} substitute that $x$ back into the equation. \textbf{$y$-intercept:} $c$. \textbf{$x$-intercepts:} solve $y = 0$ by factoring.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the axis of symmetry and the vertex of $y = x^2 - 4x + 3$.
\step{The equation is in standard form, so the vertex is not visible --- compute the axis first, then substitute, because the vertex always lies on the axis.}
\step{$a = 1$, $b = -4$, so the axis is $x = -\dfrac{-4}{2} = 2$; then $y = 2^2 - 4(2) + 3 = -1$.}
\par
Axis $x = \ans{2}$, \ vertex $(2,\, \ans{-1})$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the axis of symmetry and the vertex of $y = x^2 + 6x + 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: axis $x = \ans{-3}$, vertex $(-3, \ans{-4})$}
\end{tryitbox}

\skillheading{Skill 3 --- Transformations of the parent parabola}

\begin{formulabox}
Every parabola is the parent graph $y = x^2$ moved about. In $y = a(x-h)^2 + k$: \ $h$ shifts it sideways (\emph{right} when the bracket says $x - h$), $k$ shifts it up or down, a negative $a$ flips it over the $x$-axis, and $|a|$ stretches it. Do the sideways move first, then the stretch or flip, then the up-or-down move.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The parent graph is shifted $2$ right, flipped over the $x$-axis, then raised $9$. Where does it cross the $x$-axis?
\step{Build the equation one move at a time, then set it to zero --- the crossings are the check that the transformations were assembled correctly.}
\step{The moves give $y = -(x-2)^2 + 9$; setting $y = 0$ gives $(x-2)^2 = 9$, so $x - 2 = \pm 3$.}
\par
$x = \ans{-1}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The parent graph is shifted $1$ left and $16$ down. Where does it cross the $x$-axis?\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-5}$ or $x = \ans{3}$}
\end{tryitbox}
