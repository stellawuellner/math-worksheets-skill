\wstitleblock{Exponential Form and Logarithmic Form}{Algebra 2}{}

\begin{center}
\textit{Rewriting an exponential statement as a logarithm, rewriting a logarithm as a power, evaluating a logarithm, and solving by switching form}
\end{center}

\noindent\textbf{The model: one fact, two ways of writing it}

\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\textbf{In words} & \textbf{Exponential form} & \textbf{Logarithmic form} \\
\hline
2 to the 3rd power is 8 & $2^3 = 8$ & $\log_2 8 = 3$ \\
\hline
10 to the 2nd power is 100 & $10^2 = 100$ & $\log_{10} 100 = 2$ \\
\hline
5 to the 0th power is 1 & $5^0 = 1$ & $\log_5 1 = 0$ \\
\hline
\textit{base} to the \textit{answer} is \textit{input} & $b^{\,y} = x$ & $\log_b x = y$ \\
\hline
\end{tabular}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=3pt]
  \item Read the bottom row of the table both ways: \textbf{a logarithm is an exponent}. $\log_b x$ answers the question ``what power of $b$ gives $x$?''
  \item The base stays the base when you switch forms. Only the exponent and the input change places.
  \item Show the switched form as part of your work, then write the number asked for on the blank.
\end{itemize}
