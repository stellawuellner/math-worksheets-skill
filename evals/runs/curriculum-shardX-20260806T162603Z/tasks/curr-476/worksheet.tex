\wstitleblock{Substitution: Reading an Integral Backwards Through the Chain Rule}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Problem 1 shows the \textbf{substitution table} filled in. Build the
        same three-row table for every problem after it --- the table is the
        bookkeeping, and most substitution errors are bookkeeping errors.
  \item For an indefinite integral, convert back to $x$ and add $+\,C$. For a
        definite integral, convert the \emph{limits} instead and never return
        to $x$.
  \item Give exact values (fractions, radicals, $e$) throughout.
\end{itemize}

\vspace{0.2cm}

\problem[6cm]{Evaluate $\displaystyle\int 2x\left(x^2 + 7\right)^5 dx$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{inner function} & $u = x^2 + 7$ \\ \hline
\textbf{differential} & $du = 2x\,dx$ \\ \hline
\textbf{integral in $u$} & $\displaystyle\int u^5\,du$ \\ \hline
\end{tabular}
\end{center}
\vspace{0.15cm}
Finish it, and write your answer in terms of $x$.}

\problem[6cm]{Evaluate $\displaystyle\int x^2 e^{x^3}\,dx$. Build your own
substitution table first.}

\problem{Consider $\displaystyle\int_0^1 2x\left(x^2 + 1\right)^3 dx$ with
$u = x^2 + 1$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the value of $u$ at the lower limit $x = 0$.
    \par\vspace*{1.8cm}\ansline
  \item[(b)] Find the value of $u$ at the upper limit $x = 1$.
    \par\vspace*{1.8cm}\ansline
  \item[(c)] Evaluate the integral.
    \par\vspace*{3.4cm}\ansline
\end{itemize}}

\problem[6.5cm]{Evaluate $\displaystyle\int \frac{x}{\sqrt{x^2 + 9}}\,dx$.}

\problem{Consider $\displaystyle\int_0^{\pi/4}\tan x\,\sec^2 x\,dx$ with
$u = \tan x$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the value of $u$ at the upper limit $x = \dfrac{\pi}{4}$.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Evaluate the integral.
    \par\vspace*{3.4cm}\ansline
\end{itemize}}

\problem{Consider $\displaystyle\int \frac{\left(\ln x\right)^3}{x}\,dx$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Evaluate it.
    \par\vspace*{3.2cm}\ansline
  \item[(b)] Explain why the $\dfrac{1}{x}$ in the integrand is exactly what
    makes this substitution work.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{Consider $\displaystyle\int_1^{e}\frac{\ln x}{x}\,dx$ with $u = \ln x$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the value of $u$ at the upper limit $x = e$.
    \par\vspace*{2.0cm}\ansline
  \item[(b)] Evaluate the integral.
    \par\vspace*{3.2cm}\ansline
\end{itemize}}

\problem{Consider $\displaystyle\int_0^{b} x e^{-x^2}dx$ with $u = x^2$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Evaluate the integral for $b = 2$. Give an exact value.
    \par\vspace*{3.0cm}\ansline
  \item[(b)] In general the integral equals
    $\dfrac{1}{2}\left(1 - e^{-b^2}\right)$. Find its limit as
    $b \rightarrow \infty$.
    \par\vspace*{2.4cm}\ansline
  \item[(c)] Explain what that limit says about the area under
    $y = xe^{-x^2}$ for $x \ge 0$.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}
