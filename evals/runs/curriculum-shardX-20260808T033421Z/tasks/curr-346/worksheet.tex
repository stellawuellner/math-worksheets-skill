\wstitleblock{Rigid Motions on the Coordinate Plane}{High School Geometry}{}

\begin{center}
\small\textit{Every slide, flip and turn written as a rule on coordinates --- and every one of them keeps distances the same}
\end{center}

\begin{center}
\footnotesize
\begin{tabular}{@{}ll@{\qquad}ll@{}}
\toprule
\textbf{Motion} & \textbf{Rule} & \textbf{Motion} & \textbf{Rule} \\
\midrule
Translation & $(x + a,\ y + b)$ & Reflect in $y = x$ & $(y,\ x)$ \\
Reflect in $x$-axis & $(x,\ -y)$ & Reflect in $y = -x$ & $(-y,\ -x)$ \\
Reflect in $y$-axis & $(-x,\ y)$ & Reflect in $x = c$ & $(2c - x,\ y)$ \\
Half turn, origin & $(-x,\ -y)$ & Quarter turn ccw, origin & $(-y,\ x)$ \\
\bottomrule
\end{tabular}
\par\vspace{0.12cm}
{\footnotesize Quarter turn counterclockwise about $(h, k)$: \ $(x, y) \to \bigl(h - (y - k),\ \ k + (x - h)\bigr)$}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Read the rule, substitute, simplify --- then check the answer against the picture in your head.
  \item Distance between points: $\sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$.
\end{itemize}
