\wstitleblock{Summarizing Distributions with Center, Spread and Shape}{Precalculus / Statistics}{}

\begin{center}
\small Center, spread, shape, and reading a display
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Both data sets below record daily reading time, in minutes, already sorted from least to greatest.
  \item Write each answer on its answer line, with the unit.
  \item Where a problem asks you to describe a distribution, name the center, the spread, and the shape.
\end{itemize}

\vspace{0.2cm}
\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{Class A} (10 students, minutes) & 20, \ 25, \ 30, \ 30, \ 35, \ 40, \ 40, \ 45, \ 50, \ 85 \\
\textbf{Class B} (7 students, minutes)  & 26, \ 30, \ 34, \ 38, \ 42, \ 46, \ 50 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}

\problem[0.4cm]{Find the mean daily reading time for Class A.
\par\vspace*{3cm}\answerline{minutes}}

\problem[0.4cm]{Find the median daily reading time for Class A.
\par\vspace*{3cm}\answerline{minutes}}

\problem[0.4cm]{Find the range of the Class A times.
\par\vspace*{3cm}\answerline{minutes}}

\problem[0.4cm]{A school library recorded the books borrowed each day one week:
\vspace{0.15cm}
\begin{center}
\begin{tabular}{lccccc}
\toprule
\textbf{Day} & Mon & Tue & Wed & Thu & Fri \\
\midrule
\textbf{Books borrowed} & 12 & 19 & 15 & 22 & 27 \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=6pt, topsep=4pt]
  \item[(a)] Find the total number of books borrowed that week. \quad \ansblank
  \item[(b)] Find the number borrowed on the busiest day. \quad \ansblank
\end{itemize}
\vspace*{2.6cm}}

\problem[0.4cm]{For Class A, find the quartiles and the interquartile range.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt, topsep=4pt]
  \item[(a)] The first quartile $Q_1$. \answerline{minutes}
  \item[(b)] The third quartile $Q_3$. \answerline{minutes}
  \item[(c)] The interquartile range $Q_3 - Q_1$. \answerline{minutes}
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{A value is called an outlier when it lies more than $1.5 \times \text{IQR}$
beyond a quartile.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt, topsep=4pt]
  \item[(a)] Compute the upper fence $Q_3 + 1.5 \times \text{IQR}$ for Class A. \answerline{minutes}
  \item[(b)] Name the Class A value that lies beyond that fence, and describe the shape of the Class A distribution.
\end{itemize}
\vspace*{3.2cm}}

\problem[0.4cm]{Now work with Class B.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt, topsep=4pt]
  \item[(a)] Find the mean. \answerline{minutes}
  \item[(b)] Find the median. \answerline{minutes}
  \item[(c)] Explain what the agreement of these two numbers says about the shape of Class B, and how that differs from Class A.
\end{itemize}
\vspace*{3.2cm}}

\problem[0.4cm]{Find the standard deviation of the Class B times, treating the seven values as
the whole population. Round to the nearest whole minute.
\par\vspace*{3.4cm}\answerline{minutes}}

\problem[4.5cm]{One of the ten Class A students is chosen at random. Find the probability that
the chosen student read more than $45$ minutes.}

\problem[0.4cm]{\textbf{Synthesis.} The $85$ in Class A turns out to be a recording error: the
student actually read $45$ minutes. The corrected list is
$20, 25, 30, 30, 35, 40, 40, 45, 45, 50$.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt, topsep=4pt]
  \item[(a)] Find the new mean. \answerline{minutes}
  \item[(b)] Find the new median. \answerline{minutes}
  \item[(c)] Explain why one of these two measures moved a great deal and the other did not move at all.
\end{itemize}
