\wstitleblock{Reading Graphs Carefully --- Find the Mistake, Then Fix It}{Grades 2--3}{}

\noindent\small Everyone in these problems has made the same kind of slip. Slow
down and check three things before you answer.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{On a bar graph, read the numbers up the side.} The steps are not
        always worth one. Find two numbers that are next to each other on the
        side to see what one step is worth.
  \item \textbf{On a line plot, count the marks.} The number written underneath
        tells you \emph{what} was counted. The marks tell you \emph{how many}.
  \item \textbf{Name a shape by counting, not by looking.} Count the straight
        sides. Count the corners. Then decide.
\end{itemize}
\normalsize

\problem[4cm]{\textbf{Graph A.} How many books did Ana read?}

\problem[4cm]{\textbf{Find and fix. Graph A.} Nia said, ``Ben read $3$ books,
because his bar is $3$ steps tall.'' Nia forgot to look at the numbers up the
side. What is one step really worth, and how many books did Ben read?}

\problem[4cm]{\textbf{Graph A.} How many \emph{more} books did Cruz read than
Dee?}

\problem[4cm]{\textbf{Find and fix. Graph A.} Sam said, ``the tallest bar goes
up $6$ steps, so the most books anyone read is $6$.'' Sam counted steps instead
of reading the side. What is the most books anyone read?}

\problem[4cm]{\textbf{Find and fix. Graph B.} Rae said, ``$5$ boxes had $5$
pencils in them, because the number under the marks is $5$.'' Rae mixed up the
number on the line with the number of marks. How many boxes really had $5$
pencils?}

\problem[4cm]{\textbf{Graph B.} Which number of pencils happened most often?
Careful: the answer is the number written \emph{under} the tallest stack, not
how many marks are in it.}
