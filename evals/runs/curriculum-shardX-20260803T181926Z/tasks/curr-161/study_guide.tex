\sstitleblock{Finding a Percent of a Quantity}

\skillheading{Skill 1 --- Benchmark percents on a bar model}

\begin{formulabox}
\textbf{Model:} the whole bar is $100\%$. Cut it into ten equal parts and each part
is $10\%$; cut it in half and each part is $50\%$.\par
$10\%$ of a number $=$ the number $\div\,10$. \quad
$50\%$ $=$ half. \quad $5\%$ $=$ half of $10\%$. \quad $1\%$ $=$ the number $\div\,100$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $10\%$ of $60$
\step{Ten equal parts of the bar make the whole, so one part is the whole divided by ten --- no multiplication needed.}
\step{$60 \div 10 = \ans{6}$, and ten parts of six do rebuild sixty, so the split is right.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $10\%$ of $90$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 2 --- Multiply by the decimal}

\begin{formulabox}
\textbf{Rule:} a percent of a quantity is (percent $\div\,100$) $\times$ quantity.
Writing $35\%$ as $0.35$ turns ``of'' into ``$\times$'', so
$35\%$ of $80$ is $0.35 \times 80$. This route works for \emph{any} percent,
benchmark or not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $35\%$ of $80$
\step{Thirty-five is not a benchmark I can halve or split off a bar, so I convert it to a decimal and multiply.}
\step{$35\% = 0.35$, and $0.35 \times 80 = \ans{28}$; a check against $10\%$ of $80 = 8$ says the answer should sit near three and a half eights.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $45\%$ of $60$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{Skill 3 --- Use the fraction shortcut}

\begin{formulabox}
\textbf{Rule:} some percents are simple fractions, and the fraction is faster by hand.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item $50\% = \frac{1}{2}$, \quad $25\% = \frac{1}{4}$, \quad $75\% = \frac{3}{4}$
  \item $20\% = \frac{1}{5}$, \quad $10\% = \frac{1}{10}$, \quad $12.5\% = \frac{1}{8}$
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $75\%$ of $48$
\step{Seventy-five percent is the friendly fraction three quarters, so dividing and then multiplying beats a decimal multiplication.}
\step{One quarter is $48 \div 4 = 12$, and three quarters is $3 \times 12 = \ans{36}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $20\%$ of $85$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17}$}
\end{tryitbox}

\skillheading{Skill 4 --- Percents in a story}

\begin{formulabox}
\textbf{How to set it up:} find the \textbf{whole} the percent is taken \emph{of},
then take the percent of that whole.\par
A tip or tax is \emph{added} to the whole; a discount is \emph{taken off} it. When
two percents follow each other, the second one acts on the new amount, not the
original.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad An $18\%$ tip on a $50$ dollar bill
\step{The tip is a percent of the bill, so the bill is the whole and I take eighteen percent of it.}
\step{$18\% = 0.18$, and $0.18 \times 50 = \ans{9}$ dollars of tip.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A $40$ dollar shirt is $15\%$ off. What is the sale price?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{34}$ dollars}
\end{tryitbox}
