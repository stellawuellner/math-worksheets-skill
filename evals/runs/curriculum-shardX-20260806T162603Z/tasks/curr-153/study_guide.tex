% =================== SKILL 1 ===================
\skillheading{Sharing a Total with a Tape Diagram}

\begin{formulabox}
\textbf{Rule:} Draw one row of boxes per person, matching the ratio. Then:
add up \textbf{all} the boxes, divide the total by that count to find the value
of \textbf{one box}, and multiply by each person's boxes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two friends share 30 dollars in the ratio $2{:}3$. How
much is the smaller share?
\step{The tape has $2 + 3 = 5$ equal boxes for 30 dollars, so one box is worth
$30 \div 5 = 6$ dollars.}
\step{The smaller share has 2 boxes: $2 \times 6 = 12$ dollars, so it is
$\ans{12}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two teams share 28 stickers in the ratio $3{:}4$. How
many stickers does the \textbf{larger} team get?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Differences with a Tape Diagram}

\begin{formulabox}
\textbf{Rule:} When a problem gives the \textbf{difference} between two
quantities, count the \textbf{extra} boxes in the longer tape. Divide the
difference by the extra boxes to find the value of one box.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two jars hold marbles in the ratio $5{:}3$, and the
first jar holds 10 more. How many marbles are in the first jar?
\step{The first tape has $5 - 3 = 2$ extra boxes, and those hold the 10 extra
marbles, so one box is $10 \div 2 = 5$ marbles.}
\step{The first jar has 5 boxes: $5 \times 5 = 25$, so it holds $\ans{25}$
marbles.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two shelves hold books in the ratio $7{:}4$, and the
first shelf holds 9 more. How many books are on the first shelf?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Divide the difference by the \textbf{extra} boxes only
— never by the total number of boxes, and never by the number of people.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Reading a Double Number Line}

\begin{formulabox}
\textbf{Rule:} On a double number line, marks that line up show equal ratios.
To reach a new mark, multiply or divide \textbf{both} rows by the same number
— a handy first move is dividing down to the mark for \textbf{one} unit.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A walker covers 8 km every 2 hours. How far in 5 hours?
\step{Divide both rows by 2 to get the one-hour mark: $8 \div 2 = 4$ km per
hour.}
\step{Multiply the one-hour mark by 5: $4 \times 5 = 20$, so she walks
$\ans{20}$ km.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A printer prints 12 pages every 3 minutes. How many pages
in 7 minutes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

% =================== KEY VOCABULARY ===================
\vspace{0.3cm}
\skillheading{Key Vocabulary}

\begin{multicols}{2}
\begin{description}[leftmargin=0.5cm, itemsep=2pt, font=\normalfont\bfseries]
  \item[Tape diagram:] rows of equal boxes showing a ratio
  \item[Double number line:] two aligned scales growing in the same ratio
  \item[Unit value:] what one box (or one unit mark) is worth
  \item[Difference:] the amount by which one quantity exceeds another
\end{description}
\end{multicols}
