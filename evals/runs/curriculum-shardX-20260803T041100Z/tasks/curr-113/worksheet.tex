\textbf{Deciding what the remainder means: drop it, round up, or report what is left over}
\begin{itemize}[itemsep=1pt, topsep=3pt]
  \item \textbf{Drop it} --- the question asks how many \emph{complete} groups you can make. Leftovers do not make a whole group, so the quotient alone is the answer.
  \item \textbf{Round up} --- the question asks how many groups you \emph{need} so that nothing is left behind. The leftovers still need one more group, so add 1 to the quotient.
  \item \textbf{Report the leftover} --- the question asks how many are left over, or how many are in the last partly filled group. The remainder itself is the answer.
\end{itemize}
\end{minipage}}
\end{center}

\noindent\small\textbf{Directions:} For every problem, write the division you used
($\text{total} \div \text{group size}$), write the quotient and the remainder,
then decide which of the three meanings above the question is asking for.
Answer with the number \emph{and} what it counts.\normalsize

\problem[6cm]{A farm stand has \textbf{53} apples. Each bag holds \textbf{6} apples.
The stand only sells \emph{full} bags. How many full bags can be made?}

\problem[6cm]{Devon has \textbf{47} stickers and shares them equally among
\textbf{5} friends, giving each friend as many whole stickers as possible.
How many stickers are left over?}

\problem[6cm]{A school is taking \textbf{128} students on a trip. Each van seats
\textbf{9} students. Every student must have a seat. How many vans are needed?}

\problem[6cm]{A ribbon is \textbf{275} centimetres long. It is cut into pieces
that are each \textbf{8} centimetres long. How many whole pieces can be cut?}

\problem[6cm]{A bakery has \textbf{350} eggs. Each carton holds \textbf{6} eggs.
How many cartons are needed so that every egg is in a carton?}

\problem[6.5cm]{Rosa is putting \textbf{623} trading cards into an album.
Each page holds \textbf{9} cards. She fills each page before starting the next one.
How many cards are on the last page she uses?}

\problem{A field trip takes \textbf{237} students. Each bus seats \textbf{40} students.
Buses are filled one at a time before the next one is used.
\begin{enumerate}[label=(\alph*), itemsep=5cm]
  \item \textbf{Estimate first.} Round the number of students to the nearest ten,
        then divide by the number of seats. About how many buses will be needed? \ansline
  \item How many students ride on the last bus? \ansline
\end{enumerate}}

\problem[6.5cm]{A warehouse packs \textbf{1248} pencils into boxes.
Each box holds \textbf{9} pencils. How many boxes can be completely filled?}

\problem{A library is putting \textbf{2455} books onto shelves.
Each shelf holds \textbf{8} books, and shelves are filled one at a time.
\begin{enumerate}[label=(\alph*), itemsep=5cm]
  \item How many shelves are needed so that every book is on a shelf? \ansline
  \item How many books are on the last shelf? \ansline
\end{enumerate}}

\problem{\textbf{Find and fix the mistake.} Maya has \textbf{314} marbles and boxes
that each hold \textbf{6} marbles. She divided correctly and wrote
$314 \div 6 = 52\ \text{R}\ 2$, then answered: ``\textbf{52} boxes are needed to hold
all the marbles.''
\begin{enumerate}[label=(\alph*), itemsep=4.5cm]
  \item Maya's division is right but her answer is wrong. How many boxes are
        really needed so that every marble is in a box? \ansline
  \item How many marbles are in the last box? \ansline
\end{enumerate}}
