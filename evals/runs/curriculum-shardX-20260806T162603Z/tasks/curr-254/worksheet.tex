\wstitleblock{Finding Sign and Like-Term Errors in Polynomial Work}{Algebra 1}{}

\begin{center}
\textit{Distributing a minus sign, combining only like terms, and checking a simplification with a test value}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Four of these problems show somebody else's work. For each one, find the mistake, check the
        two expressions against a test value, and write the corrected expression.
  \item A test value is a quick way to catch an error: if two expressions really are equivalent,
        they agree at \emph{every} number you substitute. One disagreement proves a mistake.
  \item Write every corrected expression in standard form.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.5cm]{Simplify: \quad $(4x + 7) - (x - 3)$}

\problem[4.5cm]{Simplify: \quad $5x^2 + 3x - 2x^2 + x$}

\problem[5cm]{Dev simplified $(5x^2 - 3x) - (2x^2 + 4x)$ and wrote $3x^2 + x$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Name the error in Dev's work and say which step it happened at. \ansline
  \item[(b)] Substitute $x = 2$ into the \emph{original} expression to find its true value. \ansline
  \item[(c)] Write the corrected simplified expression. \ansline
\end{itemize}}

\problem[4.5cm]{Simplify: \quad $6x^3 - 2x^2 + x^3 + 5x^2$}

\problem[5cm]{Mara simplified $3x^2 + 4x + 2x^2$ and wrote $9x^3$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Name the error in Mara's work. There are two things wrong with the single step she
             took --- describe both if you can. \ansline
  \item[(b)] Substitute $x = 2$ into the \emph{original} expression to find its true value. \ansline
  \item[(c)] Write the corrected simplified expression. \ansline
\end{itemize}}

\problem[5cm]{Theo expanded $(2x - 5)(x - 3)$ and wrote $2x^2 - 11x - 15$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Only one term of Theo's answer is wrong. Name the error that produced it. \ansline
  \item[(b)] Substitute $x = 2$ into the \emph{original} product to find its true value. \ansline
  \item[(c)] Write the corrected expansion. \ansline
\end{itemize}}

\problem[6cm]{Simplify, working left to right and taking care with each bracket:
\par\smallskip
\hspace*{0.8cm} $(3x^2 - 5x + 2) - (x^2 - 5x - 6) + (2x - 1)$}

\problem[6cm]{A student simplified $(x + 4)^2 - (x - 2)(x + 2)$ like this:
\par\smallskip
\hspace*{0.8cm} $(x + 4)^2 - (x - 2)(x + 2) \;=\; x^2 + 16 - x^2 - 4 \;=\; 12$
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] There are two separate errors in that line. Name both. \ansline
  \item[(b)] Substitute $x = 3$ into the \emph{original} expression to find its true value. \ansline
  \item[(c)] Write the corrected simplified expression. \ansline
\end{itemize}}
