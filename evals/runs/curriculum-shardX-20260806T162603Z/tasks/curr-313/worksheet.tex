\wstitleblock{Corresponding Parts of Congruent Triangles}{High School Geometry}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A congruence statement lists vertices in matching order. In
        $\triangle ABC \cong \triangle DEF$, vertex $A$ matches $D$, $B$ matches
        $E$, and $C$ matches $F$; side $\overline{AB}$ matches $\overline{DE}$.
  \item Corresponding parts of congruent triangles are congruent, so matching
        sides have equal lengths and matching angles have equal measures.
  \item Set the two matching measures equal, solve, then answer the question
        that was asked. Show your work in the space provided and include the
        unit whenever the problem asks for a measurement.
\end{itemize}

\problem[5cm]{A machine shop cuts two congruent steel gusset plates, so that
$\triangle ABC \cong \triangle DEF$. The shop drawing lists $AB = (2x+3)$ cm and
$DE = 15$ cm. Find $x$.}

\problem[5cm]{The two spars of a kite frame form congruent triangles with
$\triangle PQR \cong \triangle STU$. The pattern lists
$m\angle P = (5x-10)^\circ$ and $m\angle S = 65^\circ$. Find $x$.}

\problem{A footbridge is stiffened by two congruent gusset triangles with
$\triangle JKL \cong \triangle MNP$. The fabrication drawing gives
$JK = (3y+4)$ ft and $MN = (5y-12)$ ft.
\par\vspace{0.15cm}
(a) Find $y$. \ansline
\par\vspace*{2.5cm}
(b) Find the length of $\overline{MN}$, in feet. \answerline{ft}
\par\vspace*{2.5cm}}

\problem{A roof frame is built from two congruent trusses with
$\triangle ABC \cong \triangle XYZ$. The plan gives $m\angle A = 48^\circ$,
$m\angle B = (4x+6)^\circ$, and $m\angle Y = 74^\circ$.
\par\vspace{0.15cm}
(a) Find $x$. \ansline
\par\vspace*{2.5cm}
(b) Find $m\angle Z$, in degrees. \ansline
\par\vspace*{2.5cm}}

\problem{A sailmaker cuts two congruent triangular panels with
$\triangle RST \cong \triangle UVW$. The cutting list gives $RS = 12$ m,
$ST = (2n+5)$ m, $RT = 20$ m, and $VW = 17$ m.
\par\vspace{0.15cm}
(a) Find $n$. \ansline
\par\vspace*{2.5cm}
(b) Find the perimeter of $\triangle UVW$, in metres. \answerline{m}
\par\vspace*{2.5cm}}

\problem{A bicycle frame joint is reinforced with two congruent plates,
$\triangle GHJ \cong \triangle KLM$, with $m\angle G = (3x+12)^\circ$ and
$m\angle K = (5x-18)^\circ$.
\par\vspace{0.15cm}
(a) Find $x$. \ansline
\par\vspace*{2.4cm}
(b) Find $m\angle G$, in degrees. \ansline
\par\vspace*{2.4cm}
(c) Explain how you know that $m\angle J = m\angle M$ without measuring either
angle.
\par\vspace*{2.6cm}}

\problem{Two congruent triangular tent panels satisfy
$\triangle ABC \cong \triangle DEF$. The pattern lists $AB = (4x-7)$ cm,
$DE = (2x+9)$ cm, $BC = (3y+1)$ cm, and $EF = (y+15)$ cm.
\par\vspace{0.15cm}
(a) Find $x$. \ansline
\par\vspace*{2.4cm}
(b) Find $y$. \ansline
\par\vspace*{2.4cm}
(c) Find the length of $\overline{DE}$, in centimetres. \answerline{cm}
\par\vspace*{2.4cm}}

\problem{A crane rests on two congruent support triangles with
$\triangle PQR \cong \triangle XYZ$. The specification gives $PQ = (2w+5)$ m,
$XY = 19$ m, $m\angle Q = (7k-4)^\circ$, and $m\angle Y = 66^\circ$.
\par\vspace{0.15cm}
(a) Find $w$. \ansline
\par\vspace*{2.4cm}
(b) Find $k$. \ansline
\par\vspace*{2.4cm}
(c) The engineer claims that knowing only $PQ = XY$ and
$m\angle Q = m\angle Y$ would already guarantee the two triangles are
congruent. Explain why that is not enough, and name one additional measurement
that would settle it.
\par\vspace*{3.0cm}}
