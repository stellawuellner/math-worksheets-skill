\sstitleblock{Two-Digit Subtraction with Place-Value Models}

\skillheading{Skill 1 --- Subtract the tens, then the ones}

\begin{formulabox}
\textbf{Model:} a two-digit number is ten rods $+$ ones cubes. $58$ is $5$ rods and $8$ cubes.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Take away the \textbf{tens} first: cross out that many rods.
  \item Then take away the \textbf{ones}: cross out that many cubes.
  \item Read what is left: rods first, then cubes.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $58 - 24$
\step{The ones digit of the top number is bigger than the ones digit of the bottom number, so every cube I need is already there and no rod has to be broken.}
\step{Tens: $50 - 20 = 30$. Ones: $8 - 4 = 4$. Put them together: $30 + 4 = \ans{34}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $76 - 31$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$}
\end{tryitbox}

\skillheading{Skill 2 --- Break a ten when there are not enough ones}

\begin{formulabox}
\textbf{Rule:} if the ones you have are fewer than the ones you must take away,
\textbf{break one ten rod into ten ones cubes} first. The number does not change:
$63 = 60 + 3 = 50 + 13$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $63 - 27$
\step{I must take away $7$ ones but only $3$ are showing, so I break one ten rod into ten cubes before subtracting.}
\step{Now $63 = 50 + 13$. Tens: $50 - 20 = 30$. Ones: $13 - 7 = 6$. Together: $30 + 6 = \ans{36}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $45 - 18$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimate by rounding to the nearest ten}

\begin{formulabox}
\textbf{Rule:} round each number to the nearest ten, then subtract the rounded numbers.
Look at the ones digit: $0$--$4$ rounds down, $5$--$9$ rounds up.
An estimate tells you about how big the answer should be.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Estimate $72 - 39$
\step{The question asks about how big, not exactly how big, so I round both numbers to friendly tens and subtract those instead.}
\step{$72$ rounds to $70$ and $39$ rounds to $40$, so the estimate is $70 - 40 = \ans{30}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Estimate $54 - 27$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never turn a column around. In $63 - 29$ the ones column is
$3 - 9$, not $9 - 3$. If the top digit is too small, break a ten --- do not swap.
\end{watchoutbox}
