\sstitleblock{Interpreting Remainders in Real Contexts}

\noindent\small Every problem below starts from the same division,
$74 \div 8 = 9$ R $2$. The division never changes --- the \emph{question} decides
which part of it is the answer.\normalsize

\skillheading{Skill 1 --- Drop the remainder: complete groups only}
\begin{formulabox}\textbf{When:} the question asks how many \emph{full}, complete or
whole groups can be made.\\[3pt]
\textbf{Answer:} the quotient. The leftovers cannot form a whole group, so they are not counted.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad 74 cookies are packed 8 to a bag. How many \emph{full} bags?
\step{``Full bags'' means only complete groups count, so the remainder is dropped and the quotient is the answer.}
\step{$8 \times 9 = 72$, and $74 - 72 = 2$, so $74 \div 8 = 9$ R $2$. The 2 extra cookies do not fill a bag: $\ans{9}$ full bags.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad 59 pencils are packed 7 to a pack. How many full packs?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$ full packs}\end{tryitbox}

\skillheading{Skill 2 --- Round up: nothing may be left behind}
\begin{formulabox}\textbf{When:} the question asks how many groups are \emph{needed} so that
every item (or person) is included.\\[3pt]
\textbf{Answer:} the quotient $+\ 1$ whenever there is a remainder. The leftovers still need one more group.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad 74 cookies, 8 per box. How many boxes to hold \emph{all} of them?
\step{``All of them'' means the leftovers still need a container, so the quotient must be rounded up.}
\step{$74 \div 8 = 9$ R $2$. Nine boxes hold only 72 cookies, so the last 2 need a tenth box: $\ans{10}$ boxes.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad 59 pencils, 7 per pack. How many packs to hold all of them?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$ packs}\end{tryitbox}

\skillheading{Skill 3 --- The remainder is the answer}
\begin{formulabox}\textbf{When:} the question asks how many are \emph{left over}, or how many
are in the last partly filled group.\\[3pt]
\textbf{Answer:} the remainder itself --- found as
$\text{remainder} = \text{total} - (\text{group size} \times \text{quotient})$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad 74 cookies, 8 per bag. How many are on the last, partly filled bag?
\step{The question asks about the leftovers only, so the answer is the remainder, not the quotient.}
\step{$74 - (8 \times 9) = 74 - 72$, so the last bag holds $\ans{2}$ cookies.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad 59 pencils, 7 per pack. How many pencils are in the last, partly filled pack?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ pencils}\end{tryitbox}

\skillheading{Skill 4 --- Estimate the quotient first}
\begin{formulabox}\textbf{When:} before dividing, to know roughly how big the answer should be
(and to catch an answer that is ten times too big or too small).\\[3pt]
\textbf{How:} round each number to the nearest ten, then divide the rounded numbers.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad 238 fans travel in buses that seat 40. Estimate the number of buses.
\step{An estimate only needs friendly numbers, so round first and divide the rounded numbers.}
\step{238 rounds to 240, and 40 stays 40, so $240 \div 40 = \ans{6}$ buses --- the exact answer should be near 6.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad 179 chairs are set out in rows of 20. Estimate the number of rows.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$ rows}\end{tryitbox}
