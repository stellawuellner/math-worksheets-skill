\wstitleblock{Equivalent Expressions: Expanded and Factored Forms}{Grades 6--7 \quad Expressions and Equations}{}

\begin{center}
\textit{Distributing to expand, factoring out the greatest common factor,
combining like terms, and testing equivalence by substitution}
\end{center}

\noindent\textbf{One idea, read in two directions.} The distributive property
says $a(b+c) = ab + ac$. Read left to right it \textbf{expands}; read right to
left it \textbf{factors}. Two expressions are \textbf{equivalent} when they give
the same value for \emph{every} number you substitute --- so a single
substitution can prove two expressions are \emph{not} equivalent, but it never
proves that they are.

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Decide first which direction the problem asks for: expand, factor, or
collect like terms. The problems do not stay in one mode.
\item Factor \emph{completely} --- if the terms left inside the parentheses
still share a factor, you are not finished.
\item Write your final expression on the answer line.
\end{itemize}

\problem[5cm]{Expand: \quad $3(x+4)$

\smallskip
Write the multiplication you do for each term inside the parentheses before you
simplify.}

\problem[5cm]{Factor: \quad $5x + 15$

\smallskip
Name the greatest common factor of the two terms first, then write the
expression as that factor times a sum.}

\problem[5cm]{Expand: \quad $-2(x-7)$

\smallskip
The multiplier is negative and the expression inside is a difference. Show both
products with their signs.}

\problem[5.5cm]{Write an equivalent expression with as few terms as possible:
\[
4x + 3x + 6
\]
Say which terms are like terms and why the remaining term cannot join them.}

\problem[6cm]{Ana wrote \quad $3(x+4) = 3x + 4$.

\smallskip
\textbf{(a)} Substitute $x = 5$ into $3(x+4)$ and find its value. Put that value
on the answer line.

\smallskip
\textbf{(b)} Substitute $x = 5$ into Ana's expression $3x+4$. She gets 19.

\smallskip
\textbf{(c)} The two values disagree, so the expressions are not equivalent.
Describe Ana's mistake in one sentence.}

\problem[6cm]{Factor completely: \quad $12x + 18$

\smallskip
Someone factors this as $2(6x+9)$ and stops. Explain why that is not finished,
then give the fully factored form.}

\problem[6cm]{Expand: \quad $-4(2x - 3y + 1)$

\smallskip
There are three terms inside the parentheses, so there are three products. Keep
track of every sign.}

\problem[6cm]{Write an equivalent expression with as few terms as possible:
\[
2(x+3) + 4x
\]
You cannot combine anything until the parentheses are gone. Show the expanding
step and the collecting step separately.}

\problem[6cm]{Dev claims that $4(2x-5)$ and $8x-5$ are equivalent.

\smallskip
\textbf{(a)} Substitute $x = 4$ into $4(2x-5)$ and put that value on the answer
line.

\smallskip
\textbf{(b)} Dev's expression $8x-5$ gives 27 at $x = 4$.

\smallskip
\textbf{(c)} Are the two expressions equivalent? Say how the single substitution
settles it, and write the correct expansion of $4(2x-5)$.}

\problem[6.5cm]{Factor completely: \quad $8x - 20y$

\smallskip
Two different letters appear, so no variable is common to both terms. Find the
greatest common \emph{number} and write the factored form.}

\problem[7cm]{Expand and simplify:
\[
-3(4 - 2x) + 2(x + 6)
\]
Expand each pair of parentheses first, then collect like terms. Something
surprising happens to the constant terms --- say what and why.}

\problem[8cm]{\textbf{Challenge.} A club is printing $18x + 24$ stickers.

\smallskip
\textbf{(a)} Factor $18x + 24$ completely, so the club can see how many equal
bundles it can make. Put the factored expression on the answer line.

\smallskip
\textbf{(b)} Substitute $x = 2$ into your factored form and give its value.

\smallskip
\textbf{(c)} Substitute $x = 2$ into the original $18x + 24$. Does it agree with
part (b)? Explain in one sentence why agreeing at $x = 2$ is good evidence but
not a proof of equivalence.}
