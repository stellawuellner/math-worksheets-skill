\wstitleblock{Choosing Expanded or Factored Form}{Algebra 1}{}

\begin{center}
\textit{Expanded form shows the constant term, factored form shows the zeros, and the question decides which one you want}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item The two forms are equal for every value of the variable. They differ only in what they
        make easy to see: expanded form puts the constant term and the leading coefficient on the
        page, factored form puts the zeros and the divisors there.
  \item After the first four problems the two directions are mixed on purpose --- read the
        question before choosing a method.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4.5cm]{Write $(x - 4)(x + 9)$ in standard form, so that its constant term can be read off.}

\problem[4.5cm]{Write $x^2 + 10x + 21$ in factored form, as a product of two binomials.}

\problem[5cm]{A ball is thrown upward from ground level. Its height after $t$ seconds is
$h(t) = -5t(t - 4)$ metres. The ball is at ground level when its height is $0$.
Use the form the height is already written in to find both times, in seconds, at which that
happens.
\par\vspace*{0.1cm}
\answerline{s}}

\problem[4.5cm]{Write $3x(x - 5) + 2$ in standard form, so that its constant term can be read off.}

\problem[5cm]{Write $2x^2 - 18$ in fully factored form. Take out the common numerical factor first.}

\problem[4cm]{$g(x) = x^2 - 6x + 8$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Rewrite $g$ in the form that puts its zeros on the page. \ansline
  \item[(b)] Using that form, give the two values of $x$ for which $g(x) = 0$. \ansline
\end{itemize}}

\problem[5cm]{Write $(2x - 3)^2$ in standard form. Give all three terms.}

\problem[5cm]{Write $x^3 - 4x$ in fully factored form. It has three factors.}

\problem[4.5cm]{$P(x) = (x + 5)(x - 1)(x - 3)$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Find the constant term of $P$ without expanding the whole product. \ansline
  \item[(b)] Explain which of the two forms you used and why it made the work shorter. \ansline
\end{itemize}}

\problem[6cm]{Write $6x^2 + x - 12$ in factored form, as a product of two binomials. The leading
coefficient is not $1$, so look for two numbers with product $6\times(-12)$ and sum $1$.}

\problem[5cm]{Consider $(x + 3)(x - 3) + (x + 3)^2$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Expand and simplify it. \ansline
  \item[(b)] Now write your simplified answer in fully factored form. \ansline
\end{itemize}}

\problem[5cm]{A rectangular garden has area $A(x) = x^2 + 7x + 10$ square metres.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Write $A$ in factored form. \ansline
  \item[(b)] Find the area of the garden when $x = 3$, in square metres.
             \par\vspace*{0.1cm}\answerline{m$^2$}
  \item[(c)] The gardener needs the two side lengths, not the area. Explain which of your two
             forms gives them straight away, and why the other one does not. \ansline
\end{itemize}}
