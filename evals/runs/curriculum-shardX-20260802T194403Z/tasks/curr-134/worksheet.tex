\wstitleblock{Multiplying and Dividing Fractions: Find the Flip Mistake}{Grade 4--5}{}

\begin{center}\small\itshape
Cancelling correctly before multiplying, flipping only the divisor, and
finding and fixing a misused reciprocal
\end{center}

\noindent\textbf{Two rules this page is about.}
\begin{itemize}[label={}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{R1. Multiplying:} multiply straight across. You may cancel a
        \emph{top} number with a \emph{bottom} number first --- never a top
        with a top, and never a bottom with a bottom.
  \item \textbf{R2. Dividing:} keep the first fraction, change $\div$ to
        $\times$, and flip \emph{only the second} fraction (the divisor).
        Cancel \textbf{after} the flip, never before.
\end{itemize}

\smallskip
\noindent\small For every ``find the mistake'' problem, name the mistake in
words \emph{and} show correct work. Leave answers as fractions (improper
fractions are fine).\normalsize

\problem[5cm]{Multiply. Cancel a top number with a bottom number first, then
multiply straight across.
\[ \dfrac{2}{3} \times \dfrac{3}{8} \]}

\problem[5.5cm]{Divide. Write down which fraction you flipped before you
compute, and why that is the one that flips.
\[ \dfrac{3}{4} \div \dfrac{2}{5} \]}

\problem[7cm]{\textbf{Find and fix.} Kai divided like this:
\[ \dfrac{2}{3} \div \dfrac{4}{5} \;=\; \dfrac{3}{2} \times \dfrac{4}{5}
   \;=\; \dfrac{6}{5} \]
(a) Name Kai's mistake in words. \quad (b) Show the correct work and give the
correct answer.}

\problem[7cm]{\textbf{Find and fix.} Maya multiplied like this, saying she
``cancelled the $4$ and the $8$'':
\[ \dfrac{4}{5} \times \dfrac{8}{9} \;=\; \dfrac{1}{5} \times \dfrac{2}{9}
   \;=\; \dfrac{2}{45} \]
(a) Explain why that cancelling is not allowed. \quad (b) Give the correct
product.}

\problem[5.5cm]{Find the missing factor. To undo multiplying by $\dfrac{3}{4}$
you divide by $\dfrac{3}{4}$, which means multiplying by its reciprocal
$\dfrac{4}{3}$.
\[ \dfrac{3}{4} \times \;\text{\ansblank}\; = \; \dfrac{1}{2} \]}

\problem[7cm]{\textbf{Find and fix.} Jordan cancelled before flipping:
\[ \dfrac{6}{7} \div \dfrac{3}{8} \;=\; \dfrac{3}{7} \div \dfrac{3}{4}
   \;=\; \dfrac{3}{7} \times \dfrac{4}{3} \;=\; \dfrac{4}{7} \]
(a) Explain why no cancelling is allowed until the division has been rewritten
as multiplication. \quad (b) Give the correct answer.}

\problem[7cm]{\textbf{Who is right?} Ali and Sam each computed
$\dfrac{5}{6} \div \dfrac{5}{8}$.

\smallskip
Ali: $\;\dfrac{6}{5} \times \dfrac{5}{8} = \dfrac{3}{4}$ \qquad
Sam: $\;\dfrac{5}{6} \times \dfrac{8}{5} = \dfrac{4}{3}$

\smallskip
Which student took the reciprocal of the right fraction? Name the other
student's mistake, and write the correct answer.}

\problem[7cm]{\textbf{Find and fix (word problem).} One batch of granola uses
$\dfrac{3}{4}$ cup of oats. You have $\dfrac{9}{2}$ cups of oats. How many
whole batches can you make?

\smallskip
Nia wrote $\;\dfrac{2}{9} \times \dfrac{3}{4} = \dfrac{1}{6}\;$ batch.
Explain why her answer cannot be right (you have more oats than one batch
needs), then give the correct number of batches.
\answerline{batches}}
