\wstitleblock{Modeling Discrete Linear Change and Arithmetic Sums}{Algebra 1}{}

\begin{center}
\small Building arithmetic models, solving for the term number, and adding up a finite run of terms
\end{center}
\vspace{0.15cm}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Each situation changes by the same amount at every stage, so it is an arithmetic sequence. State what $n$ counts before you compute anything.
  \item Reaching term $n$ from term $1$ takes $n - 1$ steps: $a_n = a_1 + (n-1)d$.
  \item A total over the first $n$ stages is $S_n = \dfrac{n}{2}\,(a_1 + a_n)$ --- pair the smallest with the largest.
  \item Some problems ask for a value, some for the stage number, and some for a running total. Read which one before choosing a formula. Keep units with your answers.
\end{itemize}

\problem[0.2cm]{A single stacking chair stands $82$ cm tall. Each chair added to the stack raises it by $9$ cm. How tall is a stack of $6$ chairs?
\par\vspace*{3cm}\answerline{cm}

\problem[4.5cm]{A gym charges a \$30 joining fee plus \$22 for each month of membership. How much has a member paid in total after $8$ months, in dollars?}

\problem[0.2cm]{A runner trains $12$ minutes in week $1$ and adds $4$ minutes in each week after that. How many minutes does she run in total over the first $8$ weeks?
\par\vspace*{4cm}\answerline{min}
\par\vspace*{0.3cm}}

\problem[5cm]{A water tank holds $240$ litres and loses $15$ litres each day, so after $n$ days it holds $240 - 15n$ litres. On which day is the tank empty?}

\problem[5cm]{A theatre has $20$ seats in row $1$ and $3$ more seats in each row after that. How many seats are in row $15$?}

\problem[5.5cm]{The same theatre has $20$ seats in row $1$ and $3$ more in each row after that. How many seats are there altogether in rows $1$ through $15$?}

\problem[5.5cm]{A savings account holds \$45 after week $1$, and \$18 is added every week after that. In which week does the balance first reach \$261?}

\problem[5cm]{An emergency plumber charges \$95 for the first hour on site and \$60 for each additional hour. What does a $7$-hour call-out cost, in dollars?}

\problem[5.5cm]{A brick pile has $48$ bricks in the bottom row, and each row above holds $3$ fewer than the row below it. The pile is $12$ rows high. How many bricks are in the pile altogether?}

\problem[5.5cm]{A phone plan gives $500$ MB of data in month $1$ and $250$ MB more in each month after that. In which month does the allowance first reach $3000$ MB?}

\problem[0.2cm]{Two job offers both raise pay by a fixed amount each year. Offer $A$ pays \$2400 in year $1$ and \$1500 more in each year after that. Offer $B$ pays \$3600 in year $1$ and \$900 more in each year after that.

(b) What does offer $A$ pay in total over its first $6$ years, in dollars? \quad total $=$~\ansblank
\par\vspace*{3cm}}
