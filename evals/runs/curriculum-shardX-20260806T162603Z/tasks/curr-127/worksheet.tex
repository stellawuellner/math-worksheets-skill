\wstitleblock{Common Denominators Practice}{Grade 4--5}{}
\begin{center}
\textit{Rewriting pairs, then adding and subtracting unlike fractions}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Show your work in the space under each problem.
  \item Give every fraction answer in simplest form.
\end{itemize}

\problem[3cm]{\textbf{Worked pattern to copy:} to add halves and sixths, first
rewrite both with the same denominator. Fill in the missing numerators:
$\tfrac{1}{2} = \tfrac{\square}{6}$: \ansblank \qquad
$\tfrac{1}{3} = \tfrac{\square}{6}$: \ansblank}

\problem[3cm]{What is the smallest common denominator you can use for
$\tfrac{1}{4}$ and $\tfrac{5}{12}$?}

\problem[4cm]{Add: $\tfrac{1}{2} + \tfrac{1}{6}$}

\problem[3cm]{Rewrite both fractions with denominator 20. Fill in the missing
numerators: $\tfrac{3}{4} = \tfrac{\square}{20}$: \ansblank \qquad
$\tfrac{2}{5} = \tfrac{\square}{20}$: \ansblank}

\problem[4cm]{Subtract: $\tfrac{5}{6} - \tfrac{1}{3}$}

\problem[4cm]{Add: $\tfrac{2}{3} + \tfrac{1}{4}$}

\problem[3.5cm]{Three friends each pick a common denominator for
$\tfrac{5}{8}$ and $\tfrac{3}{10}$: Nia picks 18, Owen picks 40, and Tessa
picks 80. Two of those choices work. Write the \textbf{smallest} choice that
works as a common denominator.}

\problem[4cm]{Subtract: $\tfrac{7}{10} - \tfrac{1}{4}$}

\problem{Ana ties a ribbon $\tfrac{3}{8}$ m long to a ribbon $\tfrac{1}{6}$ m
long. How long are the two ribbons together, in meters?
\par\vspace*{4.5cm}\answerline{m}}

\problem{A bag holds $\tfrac{7}{8}$ kg of flour. Leo uses $\tfrac{2}{3}$ kg
for bread. How many kilograms of flour are left in the bag?
\par\vspace*{4.5cm}\answerline{kg}}

\problem[4.5cm]{Find the missing fraction:
$\tfrac{5}{12} + \square = \tfrac{3}{4}$}

\problem[5cm]{\textbf{Challenge.} Add all three:
$\tfrac{1}{2} + \tfrac{1}{3} + \tfrac{1}{12}$}
