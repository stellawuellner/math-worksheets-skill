\sstitleblock{Linear, Squared and Cubed Scale Factors}

\begin{formulabox}
\textbf{One factor, three jobs.} If every \emph{length} of a figure is multiplied by $k$, then
\[ \text{length} \times k, \qquad \text{area} \times k^2, \qquad \text{volume} \times k^3. \]
\textbf{Why:} an area is a product of two lengths, so it collects $k$ twice; a volume is a product of three lengths, so it collects $k$ three times. Count the dimensions and you know the power.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Scaling an area}

\begin{formulabox}
\textbf{Rule:} new area $=$ old area $\times\, k^2$. This holds for any shape and for surface area as well as flat area, because both are two-dimensional.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A shape of area $15$ cm$^2$ has every length multiplied by $4$. Find the new area.
\step{The quantity asked for is an area, which is two-dimensional, so the scale factor is applied twice --- square it rather than using it as given.}
\step{new area $= 15 \times 4^2 = 15 \times 16$}

$\Rightarrow\quad$ $\ans{240}$ cm$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A shape of area $7$ cm$^2$ has every length multiplied by $5$. Find the new area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{175}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Scaling a volume or a capacity}

\begin{formulabox}
\textbf{Rule:} new volume $=$ old volume $\times\, k^3$. Capacity (litres, millilitres) is a volume, so it follows the same rule --- and so does mass, when the material is unchanged.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A solid of volume $20$ cm$^3$ has every length multiplied by $3$. Find the new volume.
\step{A volume is three-dimensional, so each of its three length factors is stretched --- the multiplier is the cube of the scale factor, not the factor itself.}
\step{new volume $= 20 \times 3^3 = 20 \times 27$}

$\Rightarrow\quad$ $\ans{540}$ cm$^3$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A solid of volume $12$ cm$^3$ has every length multiplied by $2$. Find the new volume.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{96}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Working back to the linear factor}

\begin{formulabox}
\textbf{Rules, run backwards:} given an area ratio, $k = \sqrt{\text{area ratio}}$. Given a volume ratio, $k = \sqrt[3]{\text{volume ratio}}$.\par
The ratio you are handed is \emph{already} $k^2$ or $k^3$, so undo the power before using $k$ for anything else.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two similar shapes have areas $9$ and $144$. Find the linear scale factor.
\step{The given ratio is a ratio of areas, so it equals $k$ squared, not $k$ --- the move is a square root before anything else.}
\step{$k^2 = 144/9 = 16$, so $k = \sqrt{16}$}

$\Rightarrow\quad k = $ $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two similar solids have volumes $8$ and $216$. Find the linear scale factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}
