\wstitleblock{Multiplying \& Dividing Fractions --- Error Detective}{Grade 5}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item Show every step of your work in the space under each problem.
  \item Write each answer as a fraction in simplest form (a mixed number is fine).
  \item Two problems show another student's work with a mistake in it. Find the
        mistake, then fix it.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Multiply: \quad $\dfrac{2}{3} \times \dfrac{4}{5}$}

\problem[4.5cm]{Multiply, and write your answer in simplest form: \quad
$\dfrac{5}{6} \times \dfrac{3}{10}$}

\problem[5cm]{Divide: \quad $\dfrac{3}{4} \div \dfrac{2}{5}$}

\problem{Jordan divides $\dfrac{5}{8} \div \dfrac{1}{2}$ like this:
\[ \frac{5}{8} \div \frac{1}{2} \;\to\; \frac{8}{5} \times \frac{1}{2} = \frac{8}{10} \]
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] Name the error in Jordan's work.\par\vspace*{1.8cm}\ansline
  \item[(b)] Find the correct quotient.\par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{A recipe problem: Mr.\ Diaz has $6$ cups of flour. Each small loaf of
bread uses $\dfrac{3}{4}$ cup of flour. How many loaves can he make?
\par\vspace*{5cm}\ansline}

\problem{Ava multiplies $\dfrac{2}{9} \times \dfrac{3}{4}$ like this:
\[ \frac{2}{9} \times \frac{3}{4} \;\to\; \frac{2 \times 4}{9 \times 3} = \frac{8}{27} \]
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] Name the error in Ava's work.\par\vspace*{1.8cm}\ansline
  \item[(b)] Find the correct product in simplest form.\par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{A ribbon is $\dfrac{5}{6}$ meter long. It is cut into pieces that are
each $\dfrac{1}{12}$ meter long. How many pieces are there?
\par\vspace*{5cm}\ansline}

\problem{\textbf{Challenge.} Divide, and write your answer as a whole number:
\quad $2\tfrac{1}{2} \div \dfrac{5}{8}$
\par\vspace*{5.5cm}\ansline}
