\sstitleblock{Slope: Upside Down or Sign Flipped?}

\begin{formulabox}
\textbf{The rule, and the two ways it goes wrong:}
\[
  m = \frac{\text{change in } y}{\text{change in } x}
    = \frac{y_2 - y_1}{x_2 - x_1}
\]
\textbf{Upside down} means the $x$ change ended up on top --- the answer is
the reciprocal of the true slope. \textbf{Sign flipped} means the two
subtractions started at different points --- the digits are right and the sign
is wrong.
\end{formulabox}

\skillheading{Skill 1 --- Which difference goes on top}

\begin{formulabox}
\textbf{Rule:} the change in $y$ (rise) always goes on top; the change in $x$
(run) always goes underneath. Check with units: a slope of dollars per hour
must have dollars on top.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the slope through $(1, 2)$ and $(5, 14)$.
\step{The question asks how fast $y$ grows per unit of $x$, so the $y$ change
belongs on top --- write the fraction before dividing anything.}
\step{Change in $y$ is $14 - 2 = 12$; change in $x$ is $5 - 1 = 4$; so the
fraction is $12/4$.}

The slope is $12$ over $4$. \quad $\ans{m = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope through $(2, 5)$ and $(6, 7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = \tfrac{1}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} if your slope looks like a small fraction where you
expected a big number (or the other way round), suspect an upside-down ratio
before you suspect the arithmetic.
\end{watchoutbox}

\skillheading{Skill 2 --- Keeping the sign right}

\begin{formulabox}
\textbf{Rule:} pick a starting point and subtract \emph{that same point}
in both differences. Reversing only one of them flips the sign of the answer.
A line that falls from left to right must end up with a negative slope.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the slope through $(2, 9)$ and $(6, 1)$.
\step{The $y$ value drops while $x$ grows, so the answer must come out
negative --- that is the check to make at the end.}
\step{Starting from the first point both times: $y$ change $1 - 9 = -8$, $x$
change $6 - 2 = 4$, giving $-8/4$.}

Negative, as predicted. \quad $\ans{m = -2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope through $(0, 7)$ and $(4, 3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = -1}$}
\end{tryitbox}

\skillheading{Skill 3 --- Rate of change from a table}

\begin{formulabox}
\textbf{Rule:} pick any two rows, subtract the outputs for the top of the
fraction and the matching inputs for the bottom, keeping the same row first
in both. The rate is the same for any pair of rows on a linear table.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the rate of change for this table.
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 0 & 2 & 4 & 6 \\ \hline
$y$ & 5 & 11 & 17 & 23 \\ \hline
\end{tabular}
\end{center}
\step{The rate wanted is the output change per input change, so use the first
and last rows and put the $y$ change on top.}
\step{$y$ change $23 - 5 = 18$; $x$ change $6 - 0 = 6$; so the rate is
$18/6$.}

Six units of $x$ buy $18$ units of $y$. \quad $\ans{m = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table has $x = 1, 5$ with $y = 20, 8$. Find the rate
of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = -3}$}
\end{tryitbox}
