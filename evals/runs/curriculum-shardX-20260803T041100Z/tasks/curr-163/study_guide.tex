\sstitleblock{Percent Applications: Tax, Tip, Discount, and Markup}

\skillheading{1. Sales tax --- a percent added to the price}

\begin{formulabox}
\textbf{Name the base first:} tax is charged \emph{on the price}, so the price is
the base ($100\%$).\\[3pt]
$\text{total} = \text{price} \times \left(1 + \dfrac{\text{tax rate}}{100}\right)$
\qquad $\text{tax} = \text{price} \times \dfrac{\text{tax rate}}{100}$\\[3pt]
A $8\%$ tax has multiplier $1.08$: the $1$ keeps the price, the $0.08$ adds the
tax. Read the question --- ``how much tax'' wants the second formula, ``how much
in all'' wants the first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A lamp costs \$$52$ and the tax rate is $8\%$. Find the
total paid, in dollars.
\step{The question asks for the amount paid in all, and tax is added to the
price, so use the price as the base and the add-on multiplier.}
\step{Multiplier: $1 + \dfrac{8}{100} = 1.08$; \ $52 \cdot 1.08 = 56.16$.}
\step{Total paid: $\ans{56.16}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A \$$38$ shirt is taxed at $6\%$. Find the total paid, in
dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40.28}$}
\end{tryitbox}

\skillheading{2. Tip --- a percent added to the bill}

\begin{formulabox}
\textbf{The base is the bill before the tip.} A tip works exactly like tax:\\[3pt]
$\text{amount paid} = \text{bill} \times \left(1 + \dfrac{\text{tip percent}}{100}\right)$\\[3pt]
\textbf{Mental math:} $10\%$ of a bill is the bill with the decimal point moved
one place left; $5\%$ is half of that; $20\%$ is double it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A bill is \$$46$ and the tip is $20\%$. Find the amount
paid, in dollars.
\step{The tip is added to the bill, so the bill is the base and the amount paid
uses the add-on multiplier.}
\step{Multiplier: $1 + \dfrac{20}{100} = 1.20$; \ $46 \cdot 1.20 = 55.2$.}
\step{Amount paid: $\ans{55.20}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A \$$62$ bill with a $15\%$ tip. Find the amount paid, in
dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{71.30}$}
\end{tryitbox}

\skillheading{3. Discount --- a percent taken off}

\begin{formulabox}
\textbf{Subtract inside the percent, not at the end.} If $30\%$ comes off, the
customer still pays $100 - 30 = 70\%$:
\[ \text{sale price} = \text{original} \times \frac{100 - \text{percent off}}{100} \]
\textbf{Backwards:} when the sale price is given, the original price is the
unknown base --- put $x$ where ``original'' stands above and solve for $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A \$$84$ coat is $30\%$ off. Find the sale price, in
dollars.
\step{The percent comes off the original price, so the original is the base and
the customer pays the remaining percent.}
\step{Remaining: $100 - 30 = 70$, so multiply by $\dfrac{70}{100} = 0.7$;
\ $84 \cdot 0.7 = 58.8$.}
\step{Sale price: $\ans{58.80}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A \$$50$ game is $35\%$ off. Find the sale price, in
dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32.50}$}
\end{tryitbox}

\skillheading{4. Markup --- a percent added to a store's cost}

\begin{formulabox}
\textbf{The base is what the store paid,} not what the customer pays:\\[3pt]
$\text{selling price} = \text{cost} \times \left(1 + \dfrac{\text{markup percent}}{100}\right)$\\[3pt]
Markup and tax use the same multiplier shape. Only the base changes: markup sits
on the store's cost, tax sits on the shelf price.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A store pays \$$24$ for a kite and marks it up $40\%$.
Find the selling price, in dollars.
\step{Markup is added to what the store paid, so the cost is the base and the
multiplier adds the markup on.}
\step{Multiplier: $1 + \dfrac{40}{100} = 1.40$; \ $24 \cdot 1.40 = 33.6$.}
\step{Selling price: $\ans{33.60}$ dollars.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A store pays \$$65$ for a jacket and marks it up $25\%$.
Find the selling price, in dollars.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{81.25}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Two percents in one problem act on \emph{different}
bases. Take $25\%$ off and then add $6\%$ tax by multiplying twice
($\times 0.75$, then $\times 1.06$). Folding them into one $19\%$ change is the
most common wrong answer on this topic.
\end{watchoutbox}
