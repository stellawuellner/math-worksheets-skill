\ssheader{Is This Relation a Function?}

\begin{document}

\sstitleblock{One Input, One Output}

\begin{formulabox}
\textbf{Definition:} a relation is a \emph{function} when every input is
paired with exactly one output. The rule is about \emph{inputs}. Repeated
outputs are always fine; a repeated \emph{input} with two different outputs is
what breaks it.
\end{formulabox}

\skillheading{Skill 1 --- Deciding from a table}

\begin{formulabox}
\textbf{Rule:} scan the input row for a value that appears twice. If one does,
check whether its two outputs differ --- if they do, the relation is not a
function.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is the table with inputs $2, 5, 8$ and outputs
$7, 9, 13$ a function, and what is the output at input 5?
\step{The definition is about inputs, so read the input row first and only
then look up the value asked for.}
\step{The inputs 2, 5 and 8 are all different, so no input has a choice of
outputs; the column under 5 gives its output.}

A function, and the output at 5 is $\ans{9}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Inputs $1, 3, 6$ with outputs $4, 4, 9$. What is the
output at input 3, and is it a function?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$, and yes --- repeated outputs
are allowed}
\end{tryitbox}

\skillheading{Skill 2 --- Deciding from a list of pairs}

\begin{formulabox}
\textbf{Rule:} look at the first number of every pair. If the same first
number turns up twice with different second numbers, the relation is not a
function.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $(3, 2)$, $(7, 5)$, $(3, 9)$ a function? Compare
the two outputs of the repeated input.
\step{Only the first numbers matter for the test, so list them and look for a
repeat before doing anything else.}
\step{The input 3 appears twice, with outputs 2 and 9 in that order, and those
two numbers are different.}

Not a function, because $\ans{2 < 9}$ rather than equal.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $(8, 10)$, $(5, 6)$, $(8, 4)$, compare the two
outputs of the repeated input, in the order listed.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10 > 4}$, so not a function}
\end{tryitbox}

\skillheading{Skill 3 --- Deciding from a graph}

\begin{formulabox}
\textbf{Vertical-line test:} if any vertical line meets the graph more than
once, two points share an input and the relation is not a function. Points
side by side at the same height are fine.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is the plotted relation a function, and what is the
output at input 3?
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=6, ymin=0, ymax=6,
  xtick={0,1,2,3,4,5,6}, ytick={0,1,2,3,4,5,6},
  grid=both, grid style={line width=0.25pt, draw=gray!55},
  major grid style={line width=0.4pt, draw=gray!70},
  tick label style={font=\tiny}, xlabel={$x$}, ylabel={$y$},
  width=5.4cm, height=4.4cm, enlargelimits=false]
  \fill[blue] (axis cs:1,2) circle (1.8pt);
  \fill[blue] (axis cs:3,5) circle (1.8pt);
  \fill[blue] (axis cs:5,1) circle (1.8pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{Sweep an imaginary vertical line across the picture: the question is
whether it ever catches two dots at once.}
\step{The three dots sit at three different horizontal positions, so it never
does; the dot above 3 is at height 5.}

A function, and the output at 3 is $\ans{5}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A graph shows the points $(2, 3)$, $(4, 6)$, $(6, 2)$.
What is the output at input 4, and is it a function?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$, and yes}
\end{tryitbox}

\skillheading{Skill 4 --- Relations given by a rule}

\begin{formulabox}
\textbf{Rule:} a relation written as a rule is always a function. The
arithmetic gives one result for each input, so an input can never come back
with two different outputs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For the rule $y = 5x - 2$, find the output when the
input is 4.
\step{A rule needs no repeated-input search --- one substitution settles both
the value and the fact that it is a function.}
\step{$5(4) - 2 = 20 - 2$, and there is no other answer the arithmetic could
give.}

The output is $\ans{18}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For the rule $y = 6x + 1$, find the output when the
input is 5.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{31}$}
\end{tryitbox}
