\sstitleblock{Choosing Expanded or Factored Form}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Writing it in standard form}

\begin{formulabox}
\textbf{What expanded form shows:} the constant term (the value at $x = 0$), the leading
coefficient, and the degree --- all readable off the page.
\textbf{How:} every term of the first bracket meets every term of the second, then collect.
$(x + p)(x + q) = x^2 + (p+q)x + pq$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $(x + 6)(x - 2)$ in standard form.
\step{The question asks for the constant term, which the factored form never shows on its own, so I expand.}
\step{$x\cdot x = x^2$, \ $x\cdot(-2) = -2x$, \ $6\cdot x = 6x$, \ $6\cdot(-2) = -12$.}
\step{Collect the middle: $-2x + 6x = 4x$.}

$\Rightarrow\quad \ans{x^2 + 4x - 12}$, so the constant term is $-12$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $(x - 5)(x + 8)$ in standard form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 3x - 40}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Writing it in factored form}

\begin{formulabox}
\textbf{What factored form shows:} what the expression is divisible by, and --- once you set it
equal to zero --- where it is zero.
\textbf{How, for $x^2 + bx + c$:} find two numbers with product $c$ and sum $b$. Always take out a
common factor first, and watch for a difference of squares: $x^2 - k^2 = (x-k)(x+k)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $x^2 + 9x + 20$ in factored form.
\step{Nothing is common to both terms, and it is a trinomial with leading coefficient 1, so this is a product-and-sum search.}
\step{Product $20$, sum $9$: the pair is $4$ and $5$, since $4\cdot 5 = 20$ and $4 + 5 = 9$.}

$\Rightarrow\quad \ans{(x + 4)(x + 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $x^2 - x - 12$ in factored form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x - 4)(x + 3)}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Letting the question choose the form}

\begin{formulabox}
\textbf{Zero-product property:} if $AB = 0$ then $A = 0$ or $B = 0$. That is why zeros are a
factored-form question and constant terms are an expanded-form question.
\begin{itemize}[leftmargin=0.75cm, itemsep=1pt, topsep=2pt]
  \item Asked \emph{where it is zero}, or \emph{what it divides by}? Factor.
  \item Asked for the \emph{constant term}, the \emph{leading coefficient}, or a sum with another
        polynomial? Expand.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the zeros of $x^2 - 2x - 8$.
\step{Zeros are asked for, so I want the form where a product equals zero --- that means factoring first, not substituting values.}
\step{Product $-8$, sum $-2$: the pair is $-4$ and $2$, giving $(x - 4)(x + 2)$.}
\step{Set each factor to zero: $x - 4 = 0$ or $x + 2 = 0$.}

$\Rightarrow\quad \ans{(x - 4)(x + 2)}$, so $x = \ans{4}$ or $x = \ans{-2}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the zeros of $x^2 - 5x + 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{2}$ or $x = \ans{3}$}
\end{tryitbox}
