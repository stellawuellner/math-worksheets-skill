\sstitleblock{Interior and Exterior Angles of Polygons}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The interior angle sum of a polygon}

\begin{formulabox}
\textbf{Rule:} Diagonals drawn from one vertex cut an $n$-sided polygon into $n-2$ triangles, and each triangle holds $180^\circ$:
\ \ $S = (n-2)\cdot 180^\circ$.\par
Read it forwards to get the sum from the sides, backwards to get the sides from the sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the interior angle sum of an octagon.
\step{An octagon has $8$ sides and I want the total of its angles --- that is exactly what the interior-angle-sum rule gives, so no equation is needed.}
\step{$S = (8-2)\cdot 180^\circ = 6 \cdot 180^\circ$}

$\Rightarrow\quad S = $ $\ans{1080^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the interior angle sum of a decagon ($10$ sides).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1440^\circ}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Finding one missing interior angle}

\begin{formulabox}
\textbf{Rule:} The listed angles plus the unknown one must equal $(n-2)\cdot 180^\circ$. Write that as an equation and solve for the unknown --- do not average, and do not assume the polygon is regular.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Three angles of a quadrilateral are $95^\circ$, $140^\circ$ and $72^\circ$. Find the fourth.
\step{The four angles must total a fixed amount, and three of them are known --- so this is an addition equation with one unknown, not a formula to evaluate.}
\step{$(4-2)\cdot 180^\circ = 360^\circ$, so $95 + 140 + 72 + x = 360$, i.e. $307 + x = 360$}

$\Rightarrow\quad x = $ $\ans{53^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Three angles of a quadrilateral are $108^\circ$, $76^\circ$ and $121^\circ$. Find the fourth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{55^\circ}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Exterior angles and the full turn}

\begin{formulabox}
\textbf{Rules:} Walking once around any convex polygon turns you through one full circle, so the exterior angles always add to $360^\circ$ --- no matter how many sides. In a \emph{regular} polygon they are all equal, so each one is $360^\circ / n$. At any single vertex, interior $+$ exterior $= 180^\circ$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find each exterior angle of a regular polygon with $12$ sides.
\step{The polygon is regular, so all its exterior angles are equal --- share the one full turn out evenly instead of using the interior-angle rule.}
\step{each exterior angle $= 360^\circ / 12$}

$\Rightarrow\quad$ $\ans{30^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find each exterior angle of a regular polygon with $20$ sides.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18^\circ}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- From an angle back to the sides}

\begin{formulabox}
\textbf{Rule:} $n = 360^\circ \div (\text{one exterior angle})$. If you are handed an \emph{interior} angle instead, subtract it from $180^\circ$ first to get the exterior angle, then divide.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Each exterior angle of a regular polygon is $45^\circ$. How many sides?
\step{I know one exterior angle and want the count of equal pieces the full turn was cut into --- that is a division, and it uses the exterior angle rather than the interior one.}
\step{$n = 360^\circ / 45^\circ$}

$\Rightarrow\quad n = $ $\ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Each interior angle of a regular polygon is $150^\circ$. How many sides?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(n-2)\cdot 180^\circ$ is the sum of \emph{all} the interior angles. To get \emph{one} interior angle of a regular polygon you must still divide by $n$. And the exterior-angle total is $360^\circ$ for every polygon --- it never depends on the number of sides.
\end{watchoutbox}
