\sstitleblock{Equivalent Fractions and Comparing --- Read Both Numbers}

\skillheading{Skill 1 --- When the tops or the bottoms match}

\begin{formulabox}
\textbf{What each number does:} the \textbf{bottom} tells how many equal pieces
the whole was cut into, so a bigger bottom makes \emph{smaller} pieces. The
\textbf{top} tells how many pieces you have.\par
\smallskip
\textbf{Bottoms match:} the pieces are the same size, so the bigger top wins.\par
\textbf{Tops match:} you hold the same number of pieces, so the SMALLER bottom
wins, because its pieces are bigger.\par
\smallskip
\textbf{Neither matches:} no shortcut. Go to Skill 3.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Which is greater, $\dfrac{4}{9}$ or $\dfrac{4}{5}$?
\step{The tops match, so this is the same number of pieces in each fraction and
only the size of a piece can decide it.}
\step{A whole cut into $9$ gives smaller pieces than a whole cut into $5$, and
four small pieces are less than four big ones.}
$\ans{\dfrac{4}{9} < \dfrac{4}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write $<$, $>$ or $=$ between $\dfrac{5}{6}$ and
$\dfrac{5}{12}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{5}{6} > \dfrac{5}{12}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Build an equal fraction}

\begin{formulabox}
\textbf{Rule:} multiply the top AND the bottom by the same number, and the
fraction keeps its value. Divide them both by the same number and it still
does.\par
\smallskip
Changing only one of the two numbers does not make an equal fraction --- it
makes a different fraction altogether.\par
\smallskip
\textbf{How to find the missing top:} see what the bottom was multiplied by,
then do that to the top.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\dfrac{2}{3}$ written with $12$ on the bottom is
$\dfrac{\square}{12}$. Find the top.
\step{Start from the bottom number, because it is the one that changed and it
tells you what was done to the whole fraction.}
\step{$3$ was multiplied by $4$ to reach $12$, so the top is multiplied by $4$
as well: $2 \times 4$.}
The missing top is $\ans{8}$, so $\dfrac{2}{3} = \dfrac{8}{12}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\dfrac{3}{4} = \dfrac{\square}{20}$. Find the missing top
number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\skillheading{Skill 3 --- Rewrite both, then compare}

\begin{formulabox}
\textbf{When neither number matches:} choose a bottom number that both bottoms
divide into (multiplying the two bottoms together always works), rewrite each
fraction with it using Skill 2, then compare the tops.\par
\smallskip
Once the bottoms are equal, the pieces really are the same size --- and only
then does counting tops mean anything.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Which is greater, $\dfrac{2}{5}$ or $\dfrac{3}{8}$?
\step{Neither the tops nor the bottoms match, so nothing can be read off
directly --- rewrite both with one shared bottom number first.}
\step{Both $5$ and $8$ divide into $40$: $\dfrac{2}{5}=\dfrac{16}{40}$ and
$\dfrac{3}{8}=\dfrac{15}{40}$, and sixteen pieces beat fifteen equal ones.}
$\ans{\dfrac{2}{5} > \dfrac{3}{8}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write $<$, $>$ or $=$ between $\dfrac{5}{8}$ and
$\dfrac{2}{3}$. (Try $24$ on the bottom.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{5}{8} < \dfrac{2}{3}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``bigger bottom number, bigger fraction'' is backwards.
More pieces always means smaller pieces --- $\dfrac{1}{100}$ is tiny, not huge.
\end{watchoutbox}
