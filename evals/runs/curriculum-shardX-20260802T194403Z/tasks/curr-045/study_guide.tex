\sstitleblock{Measuring and Picture Graphs}

\skillheading{Skill 1 --- Reading a picture graph (read-picture-graph)}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
A picture graph answers a question with \emph{pictures you count}. Always read the
\textbf{key} first --- it tells you what one picture is worth. Then find the \textbf{one
row} the question names and touch each picture once as you count it.\\[2pt]
{\small Pet Show: \ Cats $\star\,\star\,\star\,\star$ \quad Dogs
$\star\,\star\,\star\,\star\,\star\,\star\,\star$ \quad Birds $\star\,\star$ \quad
(each $\star$ = 1 pet)}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many \emph{dogs} are in the Pet Show graph?
\step{The question names one row --- dogs --- so I count only that row, not the whole graph.}
\step{Touch and count the dog stars: 1, 2, 3, 4, 5, 6, 7. (Cats show 4 and birds show 2, but those rows were not asked about.)}
$\Rightarrow\quad$ number of dogs $= \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Marble jar: Red $\star\,\star\,\star\,\star\,\star$ \quad Blue
$\star\,\star\,\star$ \quad Green $\star\,\star\,\star\,\star\,\star\,\star$. \ How many
\emph{green} marbles?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Measuring with same-size units}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
Lay the units end to end with \textbf{no gaps and no overlaps}, starting at one edge of
the object. The number of units is the length.\par
Two lengths can be compared by their numbers \textbf{only when the same unit was used
both times}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ribbon is 8 cubes long. A pencil is 5 cubes long. How many cubes
longer is the ribbon?
\step{Both were measured with the same cube, so ``how many longer'' is just the gap between the two counts --- subtract.}
\step{Count up from 5 to 8: 6, 7, 8 --- three counts. So $8 - 5 = 3$.}
$\Rightarrow\quad$ the ribbon is $\ans{3}$ cubes longer
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A straw is 10 cubes long and a marker is 6 cubes long. How many
cubes longer is the straw?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing from data (compare-from-data)}

\begin{formulabox}
\textbf{Rule:}\\[3pt]
``How many more'', ``who has the fewest'', ``is that true'' --- all of these need
\textbf{both} amounts first. Get each number from the graph, \emph{then} compare.\par
$>$ and $<$ always open toward the bigger amount: $7 > 4$ and $4 < 7$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad In the Pet Show graph, how many \emph{more} dogs than cats?
\step{``How many more'' compares two rows, so I read both counts before doing anything else.}
\step{Dogs 7, cats 4. Match them up one to one; the leftover dogs are the extra: $7 - 4 = 3$.}
$\Rightarrow\quad$ there are $\ans{3}$ more dogs than cats
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Marble jar again (Red 5, Blue 3, Green 6). How many \emph{more}
green marbles than blue?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A bigger \emph{count} does not always mean a bigger \emph{unit}.
If the same straw is 9 paper clips but only 5 blocks, the block is the longer unit --- it
takes fewer of a long unit to cover the same length. And when a question asks ``how many
more'', do not stop at the bigger row: the answer is the difference, not the total.
\end{watchoutbox}
