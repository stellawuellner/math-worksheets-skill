\wstitleblock{Average Value, Accumulated Change, and Motion}{AP Calculus AB/BC \quad Applications of Integration}{}

\begin{center}
\textit{Average value, accumulated change from a rate, net displacement, and
total distance, mixed so the integral you set up depends on what is asked}
\end{center}

\noindent\textbf{Directions.} Every problem on this sheet is an integral, but
the integral is not the same one each time. Decide what is being asked
\emph{before} you set anything up, then show the setup, the antiderivative, and
the evaluation. Give exact values unless a problem says otherwise.

\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=3pt]
\item \textbf{Average value} of $f$ on $[a, b]$: \ $\dfrac{1}{b-a}\displaystyle\int_a^b f\,dt$ --- a
number in the units of $f$, not of the integral.
\item \textbf{Accumulated change} from a rate $r$: \ $\displaystyle\int_a^b r\,dt$; a starting
amount plus the accumulated change gives the amount at the end.
\item \textbf{Net displacement} of a particle: \ $\displaystyle\int_a^b v\,dt$ --- signed, so
motion backwards cancels motion forwards.
\item \textbf{Total distance} travelled: \ $\displaystyle\int_a^b |v|\,dt$ --- split the interval
at every time the velocity changes sign, then add the absolute values.
\end{itemize}

\vspace{0.1cm}
\noindent Round only where a problem asks you to; otherwise leave answers exact.

\problem[6cm]{Find the average value of $f(x) = x^2$ on the interval
$[0, 3]$.}

\problem{Water flows into a tank at a rate of $r(t) = 4t + 3$ litres per
minute, where $t$ is measured in minutes.

How much water enters the tank during the first $5$ minutes?
\par\vspace*{6cm}\answerline{L}}

\problem{A particle moves along a straight line with velocity
$v(t) = 3t^2 - 12$ metres per second, for $0 \le t \le 3$ seconds.

Find the net displacement of the particle over this interval.
\par\vspace*{6.5cm}\answerline{m}}

\problem{The particle of problem $3$ has the same velocity
$v(t) = 3t^2 - 12$ metres per second on $0 \le t \le 3$.

Find the \emph{total distance} the particle travels. Show where you split the
interval and why.
\par\vspace*{7.5cm}\answerline{m}}

\problem{On a spring day the temperature inside a greenhouse is modelled by
$T(t) = 18 + 6\sin\!\left(\dfrac{\pi t}{12}\right)$ degrees, where $t$ is the
number of hours after $6$ a.m.

Find the average temperature inside the greenhouse over the $12$ hours from
$t = 0$ to $t = 12$. Round to two decimal places.
\par\vspace*{7cm}\answerline{degrees}}

\problem{A generator's tank holds $14$ litres of fuel at $t = 0$. Fuel is
consumed at a rate of $c(t) = 0.4 + 0.1t$ litres per hour for
$0 \le t \le 4$ hours.

\textbf{(a)} How much fuel is consumed over the four hours?

\textbf{(b)} How much fuel remains in the tank at $t = 4$? Give this amount on
the answer line.
\par\vspace*{7.5cm}\answerline{L}}

\problem[8cm]{A particle moves along a line with velocity
$v(t) = t^2 - 4t + 3$ metres per second, for $0 \le t \le 3$ seconds.

\textbf{(a)} At what times in $[0, 3]$ is the particle at rest?

\textbf{(b)} Find the total distance the particle travels on $[0, 3]$.

\textbf{(c)} Find the particle's average speed on $[0, 3]$, in metres per
second, rounded to three decimal places. Write this value on the answer line.}

\problem{A cyclist starting from rest has velocity $v(t) = 3\sqrt{t}$ metres
per second for $0 \le t \le 4$ seconds.

Find the cyclist's average velocity over those $4$ seconds.
\par\vspace*{7cm}\answerline{m/s}}

\problem{A holding tank contains $20$ litres of water at $t = 0$. Water enters
the tank at a rate of $10 + 2t$ litres per minute and simultaneously drains out
at a rate of $3t$ litres per minute, for $0 \le t \le 6$ minutes.

\textbf{(a)} Write and evaluate one integral giving the change in the amount of
water in the tank over the six minutes.

\textbf{(b)} How much water is in the tank at $t = 6$? Write this amount on the
answer line.
\par\vspace*{8cm}\answerline{L}}

\problem[9cm]{\textbf{Synthesis challenge.} A particle moves along a line with
velocity $v(t) = 4 - t^2$ metres per second for $0 \le t \le 3$ seconds. At
$t = 0$ its position is $s(0) = 2$ metres.

\textbf{(a)} Find the particle's net displacement on $[0, 3]$.

\textbf{(b)} Find its position $s(3)$.

\textbf{(c)} Find the total distance it travels on $[0, 3]$, rounded to three
decimal places.

\textbf{(d)} Find its average velocity on $[0, 3]$.

\textbf{(e)} The particle's average \emph{speed} on $[0, 3]$ is not equal to
your answer to part (d). Explain in a short paragraph why not, and say which of
the two is larger for this particle.}
