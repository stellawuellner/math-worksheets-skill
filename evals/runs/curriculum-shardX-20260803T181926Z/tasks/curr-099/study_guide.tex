\sstitleblock{Reading a Graph Without Being Tricked by It}

\skillheading{Skill 1 --- Reading the numbers up the side}

\begin{formulabox}
\textbf{One step is not always one.} Before you read any bar, look at two
numbers next to each other up the side.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $0, 2, 4, 6 \ldots$ means one step is worth $2$.
  \item $0, 5, 10, 15 \ldots$ means one step is worth $5$.
  \item Then: \ \textbf{steps in the bar} $\times$ \textbf{what one step is
        worth} $=$ \textbf{the amount}.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The side numbers go $0, 2, 4, 6, 8$. A bar is $4$ steps
tall. How much does it show?
\step{The bar is being counted in steps, so first find what one step is worth by
looking at two side numbers next to each other.}
\step{From $2$ to $4$ is one step, so one step is worth $2$. Four steps is
$4 \times 2$, which is $\ans{8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The side numbers go $0, 5, 10, 15, 20$. A bar is $3$ steps
tall. How much does it show?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Counting the steps is not the answer. It is only half of
it. Always turn steps into the thing being counted.
\end{watchoutbox}

\skillheading{Skill 2 --- Counting the marks on a line plot}

\begin{formulabox}
\textbf{Two different numbers, two different jobs.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The number written \emph{under} the line says \textbf{what} was measured.
  \item The marks stacked \emph{above} it say \textbf{how many} times that
        happened. One mark $=$ one thing.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad On this line plot, one $\times$ is one bag. How many bags
held $7$ marbles?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}[scale=0.8]
  \draw[thick] (0.5,0) -- (3.9,0);
  \node at (1.1,0.32) {$\times$}; \node at (1.1,0.7) {$\times$};
  \node at (1.1,1.08) {$\times$};
  \node at (2.2,0.32) {$\times$}; \node at (2.2,0.7) {$\times$};
  \node at (2.2,1.08) {$\times$}; \node at (2.2,1.46) {$\times$};
  \node at (2.2,1.84) {$\times$};
  \node at (3.3,0.32) {$\times$}; \node at (3.3,0.7) {$\times$};
  \draw (1.1,0.09) -- (1.1,-0.09); \node[below] at (1.1,-0.09) {$6$};
  \draw (2.2,0.09) -- (2.2,-0.09); \node[below] at (2.2,-0.09) {$7$};
  \draw (3.3,0.09) -- (3.3,-0.09); \node[below] at (3.3,-0.09) {$8$};
\end{tikzpicture}
\end{center}
\step{The question asks how many \emph{bags}, so the answer comes from counting
marks --- not from the number printed under the line.}
\step{Find the $7$ on the line and count the marks stacked above it: there are
$\ans{5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad On another line plot, $4$ marks sit above the $2$, $1$ mark
sits above the $3$, and $6$ marks sit above the $4$. How many things were
counted at $4$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Naming a shape by counting}

\begin{formulabox}
\textbf{Count, then name.} Count the straight sides, count the corners, and
check whether the sides are all the same length.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $3$ sides: triangle \ $\cdot$ \ $4$ sides: quadrilateral \ $\cdot$ \
        $5$ sides: pentagon \ $\cdot$ \ $6$ sides: hexagon.
  \item A \textbf{square} is a quadrilateral with four square corners
        \emph{and} four sides of the same length. Four sides alone is not
        enough.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A shape has four square corners and sides of $5$, $3$,
$5$, $3$. Is it a square?
\step{Four sides makes it a quadrilateral, so the question left to settle is
whether all four sides are the same length --- compare the long side with the
short one.}
\step{$\ans{5 > 3}$, so the sides are not all equal. It is a rectangle, not a
square.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A shape has four square corners and sides of $4$, $4$,
$4$, $4$. Is it a square? Compare a long side with a short side.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4 = 4}$, so yes --- it is a square}
\end{tryitbox}
