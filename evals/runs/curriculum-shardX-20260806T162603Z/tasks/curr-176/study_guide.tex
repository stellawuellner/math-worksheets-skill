\sstitleblock{One-Step Equations with Rational Coefficients}

\skillheading{Skill 1 --- Undo adding or subtracting}

\begin{formulabox}
\textbf{Rule:} an equation is a balance --- do the same thing to \emph{both} sides.
To undo $x + a = b$, subtract $a$ from both sides; to undo $x - a = b$, add $a$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x + 6 = 13$
\step{The $x$ has 6 added to it, so undo the addition by subtracting 6 from both sides of the balance.}
\step{$x = 13 - 6 = 7$. Check: $7 + 6 = 13$. \checkmark}

$\Rightarrow\quad \ans{x = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x - 5 = 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 14}$}
\end{tryitbox}

\skillheading{Skill 2 --- Divide both sides}

\begin{formulabox}
\textbf{Rule:} to undo $ax = b$, divide both sides by $a$:
\ $x = b/a$. A fraction answer in simplest form is a perfectly good answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4x = 30$
\step{The $x$ is multiplied by 4, so undo it by dividing both sides by 4, then simplify the fraction.}
\step{$x = \tfrac{30}{4} = \tfrac{15}{2}$ after dividing top and bottom by 2.}

$\Rightarrow\quad \ans{x = \tfrac{15}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $8x = 20$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{5}{2}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Multiply by the reciprocal}

\begin{formulabox}
\textbf{Rule:} to undo a \emph{fraction} coefficient $\tfrac{a}{b}x = c$ in one
step, multiply both sides by the reciprocal $\tfrac{b}{a}$, because
$\tfrac{b}{a} \cdot \tfrac{a}{b} = 1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\tfrac{3}{5}x = 12$
\step{The coefficient is the fraction three-fifths, so multiply both sides by its reciprocal five-thirds to leave $x$ alone.}
\step{$x = 12 \cdot \tfrac{5}{3} = \tfrac{60}{3} = 20$.}

$\Rightarrow\quad \ans{x = 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\tfrac{2}{7}x = 6$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 21}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} dividing by a fraction and multiplying by its reciprocal
are the same move --- never divide by only the numerator.
\end{watchoutbox}

\skillheading{Skill 4 --- Undo a division}

\begin{formulabox}
\textbf{Rule:} to undo $\dfrac{x}{a} = b$, multiply both sides by $a$: \ $x = ab$.
Picture a tape cut into $a$ equal parts, each worth $b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{6} = 7$
\step{The $x$ is divided by 6, so undo the division by multiplying both sides by 6.}
\step{$x = 7 \cdot 6 = 42$. Check: $42 \div 6 = 7$. \checkmark}

$\Rightarrow\quad \ans{x = 42}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{4} = 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 36}$}
\end{tryitbox}
