\problem{Multiply $4 \times \tfrac{2}{3}$; write the answer as a mixed
number.}

\textbf{Solution:} Multiply the whole number by the numerator:
$4 \times \tfrac{2}{3} = \tfrac{4 \times 2}{3} = \tfrac{8}{3}$. Since
$8 \div 3 = 2$ remainder $2$, that is
$\ans{2\tfrac{2}{3}}$

\problem{Multiply: $\tfrac{1}{2} \times \tfrac{1}{3}$}

\textbf{Solution:} Numerators together, denominators together:
$\tfrac{1 \times 1}{2 \times 3} = \ans{\tfrac{1}{6}}$

\problem{Multiply and simplify: $\tfrac{2}{3} \times \tfrac{3}{4}$}

\textbf{Solution:} $\tfrac{2 \times 3}{3 \times 4} = \tfrac{6}{12}$, and
$\tfrac{6}{12}$ simplifies (divide top and bottom by 6) to
$\ans{\tfrac{1}{2}}$

\problem{Multiply: $\tfrac{2}{5} \times 15$}

\textbf{Solution:} $\tfrac{2}{5} \times 15 = \tfrac{2 \times 15}{5} =
\tfrac{30}{5} = 6$. Answer: \ans{6}

\problem{Multiply: $1\tfrac{1}{2} \times 4$}

\textbf{Solution:} Rewrite the mixed number first:
$1\tfrac{1}{2} = \tfrac{3}{2}$. Then
$\tfrac{3}{2} \times 4 = \tfrac{12}{2} = 6$. Answer: \ans{6}

\problem{Multiply and simplify: $\tfrac{3}{8} \times \tfrac{4}{9}$}

\textbf{Solution:} $\tfrac{3 \times 4}{8 \times 9} = \tfrac{12}{72}$, and
dividing top and bottom by 12 gives
$\ans{\tfrac{1}{6}}$

\problem{Multiply: $2\tfrac{1}{3} \times 1\tfrac{1}{2}$}

\textbf{Solution:} Rewrite both:
$2\tfrac{1}{3} = \tfrac{7}{3}$ and $1\tfrac{1}{2} = \tfrac{3}{2}$. Then
$\tfrac{7}{3} \times \tfrac{3}{2} = \tfrac{21}{6} = \tfrac{7}{2}$, which is
$\ans{3\tfrac{1}{2}}$

\problem{One batch uses $\tfrac{3}{4}$ cup of milk. How much milk do 5
batches use?}

\textbf{Solution:} $5 \times \tfrac{3}{4} = \tfrac{5 \times 3}{4} =
\tfrac{15}{4} = 3\tfrac{3}{4}$. The 5 batches use
$\ans{3\tfrac{3}{4}\ \text{cups}}$

\problem{Find the missing factor:
$\square \times \tfrac{3}{4} = \tfrac{1}{2}$}

\textbf{Solution:} The missing factor is
$\tfrac{1}{2} \div \tfrac{3}{4} = \tfrac{1}{2} \times \tfrac{4}{3} =
\tfrac{4}{6}$, which simplifies to $\ans{\tfrac{2}{3}}$
\emph{Check:} $\tfrac{2}{3} \times \tfrac{3}{4} = \tfrac{6}{12} =
\tfrac{1}{2}$. \checkmark

\problem{A shelf is 3 boards, each $2\tfrac{1}{4}$ ft long. How long is the
shelf?}

\textbf{Solution:} $2\tfrac{1}{4} = \tfrac{9}{4}$, so
$3 \times \tfrac{9}{4} = \tfrac{27}{4} = 6\tfrac{3}{4}$. The shelf is
$\ans{6\tfrac{3}{4}\ \text{ft}}$

\problem{Multiply: $1\tfrac{1}{2} \times 2\tfrac{1}{2}$}

\textbf{Solution:} $1\tfrac{1}{2} = \tfrac{3}{2}$ and
$2\tfrac{1}{2} = \tfrac{5}{2}$, so the product is
$\tfrac{3}{2} \times \tfrac{5}{2} = \tfrac{15}{4}$. Since
$15 \div 4 = 3$ remainder 3, that is $3\tfrac{3}{4}$, or as a decimal
$\ans{3.75}$

\problem{A garden is $3\tfrac{1}{2}$ m long and $2\tfrac{1}{4}$ m wide. What
is its area?}

\textbf{Solution:} Area $=$ length $\times$ width:
$3\tfrac{1}{2} = \tfrac{7}{2}$ and $2\tfrac{1}{4} = \tfrac{9}{4}$, so the
area is $\tfrac{7}{2} \times \tfrac{9}{4} = \tfrac{63}{8} = 7\tfrac{7}{8}$.
Area: $\ans{7\tfrac{7}{8}\ \text{m}^2}$
