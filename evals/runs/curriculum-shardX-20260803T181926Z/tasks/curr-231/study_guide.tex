\sstitleblock{Deciding Whether a Relation Is a Function}

\begin{formulabox}
\textbf{Definition.} A relation is a \textbf{function} when every input has
exactly one output.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item \textbf{The count test, in every representation:} count the pairs
(arrows, rows, points), then count the \emph{different} inputs. Equal counts
$\Rightarrow$ function. Fewer inputs than pairs $\Rightarrow$ some input was
used twice $\Rightarrow$ not a function.
\item \textbf{Outputs may repeat. Inputs may not.} Two inputs sharing one
output is fine; one input with two outputs is fatal.
\item To \emph{disprove} it, name the offending input and its two outputs. That
one pair of values is the whole proof.
\end{itemize}
\end{formulabox}

\skillheading{1. From a mapping diagram}

\begin{formulabox}
Count the arrows, then count the dots on the left that have an arrow leaving
them. A left-hand dot with two arrows is the only thing that can go wrong ---
two arrows arriving at the same right-hand dot is harmless.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Is this mapping diagram a function?
\begin{center}
\begin{tikzpicture}[scale=0.62]
  \draw[thick, rounded corners=10pt] (-0.8,-1.6) rectangle (0.8,1.6);
  \draw[thick, rounded corners=10pt] (3.2,-1.6) rectangle (4.8,1.6);
  \node at (0,1) {$1$}; \node at (0,0) {$2$}; \node at (0,-1) {$3$};
  \node at (4,0.6) {$8$}; \node at (4,-0.6) {$5$};
  \draw[->, thick] (0.35,1) -- (3.6,0.6);
  \draw[->, thick] (0.35,0) -- (3.6,-0.6);
  \draw[->, thick] (0.35,-1) -- (3.6,-0.6);
\end{tikzpicture}
\end{center}
\step{Do not look at the right-hand side first --- only a repeated \emph{input}
can break the rule, so count arrows against left-hand dots.}
\step{Arrows: three. Left-hand dots with an arrow leaving: three.}
So $\ans{3 = 3}$: no input is used twice, and it is a function. The two arrows
landing on the same output do not matter.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A diagram has arrows $1 \rightarrow 4$, $6 \rightarrow 2$,
$6 \rightarrow 9$, $8 \rightarrow 4$. Compare the arrow count with the count of
different inputs.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3 < 4}$ --- not a function, since 6
has two outputs}
\end{tryitbox}

\skillheading{2. From a set of pairs or a table}

\begin{formulabox}
In $\{(x,y),\dots\}$ the \emph{first} coordinate is the input; in a table it is
the top row (or the left column). List the inputs, cross out repeats, and
compare what is left with the number of pairs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Is $\{(2,3),\;(4,5),\;(6,7),\;(2,9),\;(8,1)\}$ a function?
\step{The pairs are given, so the count test needs no drawing --- just list the
first coordinates and look for a repeat.}
\step{First coordinates: $2, 4, 6, 2, 8$. Crossing out the repeated $2$ leaves
four different inputs among five pairs.}
So $\ans{4 < 5}$: the input $2$ has outputs $3$ and $9$, so this is not a
function.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A table lists hours $1, 2, 3$ with costs $5, 5, 12$. Compare
the number of rows with the number of different hours.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3 = 3}$ --- a function; the repeated
cost is an output, so it is allowed}
\end{tryitbox}

\skillheading{3. From a graph: the vertical line test}

\begin{formulabox}
Slide a vertical line across the graph. Each position of the line is one input,
and the number of crossings is the number of outputs it has. \textbf{More than
one crossing anywhere $\Rightarrow$ not a function.} One failing line is enough
to decide it, so name the line and the two points it hits.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} The curve below is $x = y^2$. Is it a function?
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=center, xmin=-1, xmax=6, ymin=-3, ymax=3,
  xtick={4}, ytick={-2,2}, tick label style={font=\tiny},
  xlabel={$x$}, ylabel={$y$}, width=5.6cm, height=3.6cm,
  enlargelimits=false, samples=80]
  \addplot[very thick, blue!70!black, domain=-2.3:2.3] ({x^2},{x});
  \draw[dashed, red] (axis cs:4,-3) -- (axis cs:4,3);
  \fill[red] (axis cs:4,2) circle (1.6pt);
  \fill[red] (axis cs:4,-2) circle (1.6pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{One failing vertical line settles it, so look for an input the curve
reaches twice rather than checking the whole picture.}
\step{The dashed line meets the curve at two marked points, so that one input
has two outputs, against the one a function allows.}
So $\ans{1 < 2}$: the vertical line test fails and this is not a function.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} For the parabola $y = x^2$, how many times does a vertical
line meet the curve, against the one output a function allows?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1 = 1}$ --- exactly one crossing, so
it is a function}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} being a function depends on which way the arrows point.
"Student $\rightarrow$ locker" can be a function while "locker
$\rightarrow$ student" is not, from the very same table. Always check which
column you are calling the input.
\end{watchoutbox}
