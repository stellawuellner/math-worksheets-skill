\wstitleblock{Right-Triangle Trigonometry --- Wrong Ratio, Wrong Inverse, Wrong Mode}{High School Geometry}{}

\noindent\small Three errors account for most lost marks in this unit: choosing
the ratio that does not contain the two sides you have, putting the sides into
an inverse function upside down, and leaving the calculator in radian mode.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Before computing, name the two sides involved (opposite / adjacent /
        hypotenuse) and write the ratio you chose.
  \item Round side lengths and angle measures to two decimal places.
  \item Sanity-check every answer: a leg is always shorter than the hypotenuse,
        and an acute angle is between $0^\circ$ and $90^\circ$.
\end{itemize}
\normalsize

\refrt

\problem[4.5cm]{The hypotenuse is $12$ and $\angle A = 35^\circ$. Find $x$, the
side opposite $\angle A$. Write the ratio you chose before you compute.\par
\probfig{1}}

\problem[4.5cm]{\textbf{Find and fix.} Dana needs $x$, the side opposite
$\angle A$, from the adjacent leg $14$ and $\angle A = 52^\circ$. She writes the
correct setup $x = 14 \tan 52^\circ$, presses enter, and her calculator returns
$-84.75$.\par
Her setup is right, so the fault is in the calculator. Say what it is, then give
the correct value of $x$.\par
\probfig{2}}

\problem[4.5cm]{\textbf{Find and fix.} Eli must find $\angle A$ from the two
legs, $7$ opposite $\angle A$ and $24$ adjacent to it. He writes
$\angle A = \tan^{-1}\!\left(\frac{24}{7}\right) = 73.74^\circ$.\par
His arithmetic is fine; the ratio inside the inverse is not. Name the error,
then give $\angle A$ correctly.\par
\probfig{3}}

\problem[4.5cm]{The hypotenuse is $20$ and $\angle A = 41^\circ$. Find $b$, the
leg \emph{adjacent} to $\angle A$. A classmate reaches for sine out of habit;
decide whether that is the ratio this side needs, then compute.\par
\probfig{4}}

\problem{\textbf{Find and fix.} Ravi stands $40$ feet from the base of a
flagpole on level ground. The angle of elevation from his eye level to the top
of the pole is $38^\circ$. He sets up the height as $40 \tan 38^\circ$ and his
screen shows $12.41$.\par
That height is less than half the distance he is standing away, even though the
angle is close to $45^\circ$ --- so the number cannot be right. Find what his
calculator is doing and give the correct height of the pole above his eye level,
in feet.\par
\probfig{5}
\par\vspace*{4.5cm}\answerline{ft}}

\problem[5cm]{This is a $30$-$60$-$90$ triangle: the hypotenuse is $10$ and
$\angle A = 30^\circ$.\par
\textbf{(a)} Find $a$, the side opposite the $30^\circ$ angle.\par
\textbf{(b)} Sasha answers $8.66$ for part (a). Which side of this triangle is
$8.66$, and which ratio did Sasha use to get it? Answer in one sentence beside
your work.\par
\probfig{6}}
