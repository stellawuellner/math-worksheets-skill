\wstitleblock{Adding, Subtracting, and Multiplying Complex Numbers}{Algebra 2}{}

\noindent\small
Adding and subtracting complex numbers, multiplying them, simplifying powers
of the imaginary unit, and building or checking complex solutions. Write every answer in
standard form $a + bi$ with $a$ and $b$ real.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $i = \sqrt{-1}$, so $i^2 = -1$. That single substitution is what turns
        a product back into $a + bi$.
  \item Add and subtract by combining real parts with real parts and imaginary
        parts with imaginary parts --- exactly like combining like terms.
  \item Multiply by distributing (FOIL), then replace every $i^2$ with $-1$
        and re-collect.
  \item Powers of $i$ cycle with period $4$: $i, -1, -i, 1$. Divide the
        exponent by $4$ and keep the remainder.
\end{itemize}
\normalsize

\vspace{0.15cm}

\problem[4cm]{Simplify: \quad $(3 + 5i) + (4 - 2i)$}

\problem[4cm]{Simplify: \quad $(8 - i) - (3 + 6i)$}

\problem[4.5cm]{Multiply and write in standard form: \quad $3i(4 - 7i)$}

\problem[4.5cm]{Multiply and write in standard form: \quad $(2 + 3i)(5 - i)$}

\problem[4.5cm]{Simplify $i^{18}$. State the remainder you used when you
divided the exponent by $4$.}

\problem[5cm]{Simplify: \quad $(-4 + 7i) + (9 - 7i) - (2 - 3i)$
\par\smallskip
Say in one sentence what happened to the imaginary parts of the first two
numbers.}

\problem[5cm]{Multiply $(5 + 2i)(5 - 2i)$ and write the result in standard
form. Then say in one sentence why the answer has no imaginary part.}

\problem[5.5cm]{Expand $(4 - 3i)^2$ and write the result in standard form.
Remember that squaring a binomial means multiplying it by itself.}

\problem[5.5cm]{Simplify: \quad $i^{43} + i^{20}$}

\problem[6cm]{Solve $x^2 - 6x + 13 = 0$ over the complex numbers. Give both
solutions in standard form.}

\problem[6cm]{Multiply the three factors and write the result in standard
form: \quad $(2 + i)(3 - 2i)(1 + i)$
\par\smallskip
Multiply the first two factors, simplify to standard form, and only then
multiply by the third.}

\problem[6.5cm]{\textbf{Synthesis challenge.} Multiply out
\[ \bigl(x - (5 + i)\bigr)\bigl(x - (5 - i)\bigr) \]
and write the product as a quadratic in standard form. Then state the two
zeros of that quadratic, and say in one sentence why the product came out with
real coefficients.}
