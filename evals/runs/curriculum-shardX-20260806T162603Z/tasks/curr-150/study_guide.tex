\sstitleblock{Putting Patterns, Data and Points Together}

\skillheading{Skill 1 --- Jumping to any term of a pattern}

\begin{formulabox}
\textbf{Rule:} The first term is the start, so term $n$ is $n - 1$ steps out.\\[3pt]
term $n$ $=$ start $+$ step size $\times (n - 1)$\\[3pt]
Two patterns meet when their terms are equal; the one that starts behind closes the gap by the difference of the step sizes each step.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at 5 and adds 7 each time. Write the 9th term.
\step{Listing nine terms invites a miscount, so I count steps instead: the 9th term is eight steps past the start.}
\step{$5 + 7 \times 8 = 5 + 56$.}
9th term: $\ans{61}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at 2 and adds 9 each time. Write the 7th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56}$}
\end{tryitbox}

\skillheading{Skill 2 --- Counting from a line plot}

\begin{formulabox}
\textbf{Rule:} On a line plot the tick is the \emph{value} and the stack height is \emph{how many}.\\[3pt]
``How many students'' counts crosses. ``How many books'' uses the numbers under them. Comparing two stacks means subtracting their heights, not their ticks.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A plot has 3 crosses at 1, 5 crosses at 2, 2 crosses at 3 and 4 crosses at 4. How many entries sit at value 2?
\step{The question asks how many, not how much, so I read the height of the stack above the tick and ignore the tick itself.}
\step{The stack above the tick labelled 2 is 5 crosses tall.}
Entries at value 2: $\ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same plot, how many more entries sit at value 2 than at value 3?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Turning a pattern into ordered pairs}

\begin{formulabox}
\textbf{Rule:} Pair the term number with the term value: (term number, value).\\[3pt]
Across is the term number, up is the value. Because the step never changes, the marks always land on a straight line --- so a bent line is a signal that a term was generated wrongly, not a new shape.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pattern starts at 0 and adds 3. Write the ordered pair for term 7.
\step{The term number is already the across coordinate, so the only work is generating the value that goes up.}
\step{Term 7 is six steps out: $0 + 3 \times 6 = 18$.}
Ordered pair: $(7, \ans{18})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A pattern starts at 1 and adds 5. Write the ordered pair for term 6.\\[2pt]
\rotatebox{180}{\footnotesize check: $(6, \ans{26})$}
\end{tryitbox}

\skillheading{Skill 4 --- Totalling a whole plot}

\begin{formulabox}
\textbf{Rule:} Multiply each value by its stack height, then add the products.\\[3pt]
total $=$ (height $\times$ value) $+$ (height $\times$ value) $+ \ldots$\\[3pt]
Once the total is known, mean $=$ total $\div$ number of entries --- and running that backwards (mean $\times$ number of entries) is a genuine check on the total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A plot has 3 crosses at 1, 2 crosses at 4 and 5 crosses at 2. What is the total?
\step{Adding the ticks alone would ignore how many entries sit on each, so I weight every value by its stack height first.}
\step{$3 \times 1 + 2 \times 4 + 5 \times 2 = 3 + 8 + 10$.}
Total: $\ans{21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has 2 crosses at 3, 4 crosses at 5 and 1 cross at 6. What is the total?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32}$}
\end{tryitbox}
