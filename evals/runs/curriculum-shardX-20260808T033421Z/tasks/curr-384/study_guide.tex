\sstitleblock{Composition Order and Inverse Domains}

\skillheading{Skill 1 --- Order changes a composition}

\begin{formulabox}
\textbf{Read it inside out:} $(f \circ g)(x) = f\bigl(g(x)\bigr)$ --- the rule written
\emph{next to} the input acts first.\par
\textbf{Not commutative:} $f \circ g$ and $g \circ f$ are different rules. If a quoted answer
looks wrong, compute both compositions; the wrong one is usually the other order.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x + 4$, $g(x) = 3x$. Find $f\bigl(g(5)\bigr)$ and $g\bigl(f(5)\bigr)$.
\step{Each answer needs the inner rule evaluated first, so read each expression inside out and do that rule before the outer one.}
\step{$g(5) = 15$, then $f(15) = 15 + 4 = 19$. As one rule, $f(g(x)) = 3x + 4$.}
\step{$f(5) = 9$, then $g(9) = 27$. As one rule, $g(f(x)) = 3(x + 4)$.}

$\Rightarrow\quad f\bigl(g(5)\bigr) = \ans{19}, \qquad g\bigl(f(5)\bigr) = \ans{27}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $p(x) = x - 2$, $q(x) = x^2$. Find $p\bigl(q(3)\bigr)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(g \circ f)(x)$ with $f(x) = x - 4$ and $g(x) = x^2$ is $(x-4)^2$, \emph{not} $x^2 - 4$. Subtract first, square second.
\end{watchoutbox}

\skillheading{Skill 2 --- Testing a claimed inverse}

\begin{formulabox}
\textbf{The two-sided test:} a rule is $f^{-1}$ only when $f^{-1}\bigl(f(x)\bigr) = x$
\emph{and} $f\bigl(f^{-1}(x)\bigr) = x$.\par
\textbf{Fast refutation:} pick one convenient input, run it through $f$, then through the
claimed rule. If you do not get your input back, the claim is dead --- one counterexample is
enough.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test the claim that $\dfrac{x + 3}{5}$ inverts $f(x) = 5x - 3$.
\step{A claimed inverse is worth testing with a single number before any algebra, so run one input out through $f$ and back through the claim.}
\step{$f(4) = 5(4) - 3 = 17$, then $\dfrac{17 + 3}{5} = \dfrac{20}{5} = 4$ --- the input came back.}
\step{In general $5\left(\dfrac{x + 3}{5}\right) - 3 = (x + 3) - 3 = x$, so the other round trip works too.}

$\Rightarrow\quad$ round trip returns $\ans{4}$, and the claim holds.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = 2x + 7$ and the claim $g^{-1}(x) = \dfrac{x - 7}{2}$, run the input $6$ out and back.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- The domain an inverse inherits}

\begin{formulabox}
\textbf{Inputs of $f^{-1}$ are outputs of $f$.} Solving for $y$ gives the formula; the
\emph{range} of the original gives the restriction the formula must carry.\par
\textbf{Square roots:} $\sqrt{\ \ }$ is never negative, so a rule built from
$\sqrt{x - h} + k$ has outputs $x \ge k$ --- and its inverse only accepts inputs $x \ge k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $f^{-1}$ for $f(x) = \sqrt{x - 1} + 2$, and state its domain.
\step{The formula comes from solving $x = f(y)$ for $y$, and the restriction comes separately from asking what outputs $f$ can actually produce.}
\step{$x - 2 = \sqrt{y - 1}\;\Rightarrow\;(x - 2)^2 = y - 1$}
\step{Outputs of $f$: the root is at least $0$, so $f(x) \ge 2$. The inverse therefore accepts only $x \ge 2$.}

$\Rightarrow\quad \ans{f^{-1}(x) = (x - 2)^2 + 1}$ \ for $x \ge 2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $g^{-1}$ for $g(x) = \sqrt{x + 4} - 1$ (and say what its inputs must satisfy).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g^{-1}(x) = (x + 1)^2 - 4}$, inputs $x \ge -1$}
\end{tryitbox}
