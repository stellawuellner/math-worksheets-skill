\sstitleblock{Modeling with Systems and Matrices}

\skillheading{Skill 1 --- Systems in two unknowns}

\begin{formulabox}
\textbf{Choose the method from the shape of the equations.} If one equation is already
solved for a variable (or costs one step to solve), substitute. If both are in the
$ax + by = c$ form with awkward coefficients, eliminate by scaling one equation until a
column matches.\par
\textbf{Always define the variables in words first} --- ``$x$ = price of one adult ticket, in
dollars''. A system you cannot label is a system you cannot check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2x + y = 13$ and $x - y = 2$.
\step{The $y$ terms already have opposite signs, so adding the two equations removes $y$ in one move --- elimination is cheaper here than substitution.}
\step{Adding: $3x = 15$, so $x = 5$. Then $5 - y = 2$.}

$\Rightarrow\quad \ans{x = 5}, \quad \ans{y = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $3x + 2y = 16$ and $x + y = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$, $\ans{y = 2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Systems in three unknowns}

\begin{formulabox}
\textbf{Reduce three to two.} Pair the equations up and eliminate the \emph{same} variable
from each pair. That leaves a two-unknown system you already know how to finish; back
substitute for the third.\par
\textbf{Look for a free gift first:} an equation like $y = x + 3$, or two equations that
differ in only one term, does the elimination for you.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x + y + z = 9$, $2x + y + z = 11$, $x + 2y + z = 13$.
\step{The three equations differ from the first in a single term each, so subtracting the first from the others isolates one unknown at a time.}
\step{Second minus first: $x = 2$. Third minus first: $y = 4$.}
\step{Back into the first: $2 + 4 + z = 9$.}

$\Rightarrow\quad \ans{x = 2}, \quad \ans{y = 4}, \quad \ans{z = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x + y + z = 12$, $x + 2y + z = 15$, $2x + y + z = 16$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$, $\ans{y = 3}$, $\ans{z = 5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Writing a system as a matrix equation}

\begin{formulabox}
\textbf{The translation is mechanical:} coefficients fill the matrix, one row per condition
and one column per unknown; the unknowns stack into a column; the right-hand sides stack
into another. That is $A\mathbf{x} = \mathbf{b}$.\par
\textbf{What it buys you} is a picture of the model: each row is one real-world condition,
each column is one real-world quantity. Solve it with the same elimination as before.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $4x + 3y = 18$ and $2x + 5y = 16$ as a matrix equation, and solve.
\step{Each condition becomes one row, so the unknowns line up in columns and the constants stand alone on the right.}
\step{$\begin{bmatrix} 4 & 3 \\ 2 & 5 \end{bmatrix}\begin{bmatrix} x \\ y \end{bmatrix} = \begin{bmatrix} 18 \\ 16 \end{bmatrix}$}
\step{Doubling the second row gives $4x + 10y = 32$; subtracting the first row leaves $7y = 14$.}

$\Rightarrow\quad \ans{x = 3}, \quad \ans{y = 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $5x + 2y = 24$ and $3x + 4y = 20$ as a matrix equation, and solve.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$, $\ans{y = 2}$}
\end{tryitbox}

\skillheading{Skill 4 --- Where a line meets a curve}

\begin{formulabox}
\textbf{Substitute the linear equation into the nonlinear one.} One quadratic in one unknown
comes out, so expect \emph{two} answers --- and each $x$ carries its own $y$.\par
\textbf{Finish honestly:} pair the coordinates up, then ask whether the situation allows
both. A negative length is rejected by the context; a second valid design is not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Where does $y = 2x + 4$ meet $y = x^2 + 1$?
\step{Both equations give $y$, so setting the right-hand sides equal turns the pair into one quadratic --- the standard move for a line against a curve.}
\step{$x^2 + 1 = 2x + 4 \;\Rightarrow\; x^2 - 2x - 3 = 0 \;\Rightarrow\; (x - 3)(x + 1) = 0$}
\step{At $x = 3$: $y = 2(3) + 4 = 10$. At $x = -1$: $y = 2(-1) + 4 = 2$.}

$\Rightarrow\quad \ans{x = 3,\ y = 10} \quad \text{and} \quad \ans{x = -1,\ y = 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Where does $y = 3x$ meet $y = x^2 - 4$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4,\ y = 12}$ and $\ans{x = -1,\ y = -3}$}
\end{tryitbox}
