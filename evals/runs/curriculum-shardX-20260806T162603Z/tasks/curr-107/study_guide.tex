\sstitleblock{Multiplying by a One-Digit Number}

\skillheading{Skill 1 --- The Standard Algorithm}

\begin{formulabox}
\textbf{Work right to left, one column at a time.} Multiply the one-digit number by the ones, then
the tens, then the hundreds, then the thousands.\par
When a column's product is $10$ or more, write the ones digit in that column and \textbf{carry} the
rest above the next column, then add it in after you multiply there.\par
A $0$ digit still gets its turn: $0$ times anything is $0$, but a carry can land in that column.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $47 \times 6$
\step{Both digits of $47$ have to be multiplied, so I go column by column from the right and keep track of anything that spills over ten.}
\step{Ones: $6 \times 7 = 42$, write $2$ and carry $4$ tens. Tens: $6 \times 4 = 24$ tens, plus the $4$ carried tens is $28$ tens, which is $280$.}

$47 \times 6 = \ans{282}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $2{,}043 \times 8$ \ (watch the zero in the hundreds column)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16344}$}
\end{tryitbox}

\skillheading{Skill 2 --- Partial Products}

\begin{formulabox}
\textbf{Split the big number by place value, multiply each part, then add.}\par
$1{,}325 = 1000 + 300 + 20 + 5$, so $1{,}325 \times 4$ is
$4000 + 1200 + 80 + 20$.\par
Nothing is carried --- every part keeps its own size --- which makes this the method to fall back on
when a carry goes wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $132 \times 5$ using partial products
\step{I want a method with no carrying to keep track of, so I break the number into its place values and multiply each piece on its own.}
\step{$132 = 100 + 30 + 2$, so the parts are $5 \times 100 = 500$, $5 \times 30 = 150$ and $5 \times 2 = 10$; adding gives $500 + 150 + 10$.}

$132 \times 5 = \ans{660}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $2{,}140 \times 3$ using partial products\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6420}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimate Before You Multiply}

\begin{formulabox}
\textbf{Round the big number, keep the one-digit number exact, multiply.}\par
To round: look at the digit \emph{one place to the right} of the place you are rounding to. If it is
$5$ or more, round up; otherwise round down.\par
The estimate tells you roughly how big the answer should be, so a product that is wildly bigger or
smaller is a signal to check the work.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $687$ to the nearest hundred, ready to estimate $687 \times 4$.
\step{I only round the large number, because rounding the one-digit factor would throw the estimate away entirely.}
\step{Rounding to the hundred means looking at the tens digit, $8$; that is $5$ or more, so $687$ rounds up to the next hundred.}

$687$ to the nearest hundred is $\ans{700}$, so the estimate is $4 \times 700 = 2800$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $3{,}140$ to the nearest thousand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3000}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never drop a $0$ digit out of a number to make it shorter. Turning
$5{,}704$ into $574$ removes an entire place value and shrinks the answer by thousands. The zero
holds its column open --- often for a carry.
\end{watchoutbox}
