\sstitleblock{Modeling Loci and Intersections with Conic Sections}

\noindent\small Two moves cover this whole unit. A \emph{locus} is a sentence
about distance: write the distances, set them equal (or in the stated ratio),
square, simplify. An \emph{intersection} is a system: substitute, solve one
variable, substitute back for the other.
\normalsize

\skillheading{1. Circles from a distance condition}
\begin{formulabox}
\textbf{Center-radius form:} $(x-h)^2 + (y-k)^2 = r^2$ has center $(h, k)$ and
radius $r$. The signs in the brackets are the \emph{opposite} of the center
coordinates: a center of $(3, -4)$ gives $(x-3)^2 + (y+4)^2$.\\[3pt]
\textbf{From general form,} complete the square in $x$ and in $y$: halve each
linear coefficient, square it, and add it to \emph{both} sides. The radius is
the square root of the new right-hand side, never of the original constant.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Write $x^2 + y^2 + 10x - 4y + 13 = 0$ in center-radius form.
\step{The equation has no visible center, so completing the square in each
variable is the tool that exposes one.}
\step{Move $13$ across and add $(10/2)^2 = 25$ and $(-4/2)^2 = 4$ to both
sides: $(x+5)^2 + (y-2)^2 = -13 + 25 + 4 = 16$, so $r = \sqrt{16}$.}
\par
$\ans{\text{center } (-5,\ 2), \quad r = 4}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Write $x^2 + y^2 - 2x + 6y + 6 = 0$ in center-radius
form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{center } (1,\ -3), \; r = 2}$}
\end{tryitbox}

\skillheading{2. Parabolas from a focus and a directrix}
\begin{formulabox}
\textbf{Definition:} a parabola is every point whose distance to the
\emph{focus} equals its distance to the \emph{directrix}.\\[3pt]
With vertex at the origin, focus $(0, p)$ and directrix $y = -p$, that
condition simplifies to $x^2 = 4py$, so $y = x^2/(4p)$.\\[3pt]
\textbf{(!)} $p$ is the focal \emph{distance}; the number in the equation is
$4p$. A focus at $(0,3)$ gives $x^2 = 12y$, not $x^2 = 3y$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} A parabola has focus $(0, 4)$ and directrix $y = -4$. Find
$y$ when $x = 8$.
\step{The focus is $4$ above the vertex and the directrix $4$ below it, so
$p = 4$ and the standard form $x^2 = 4py$ applies directly.}
\step{$x^2 = 16y$, so $y = 8^2/16 = 64/16$.}
\par
$\ans{4}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} A parabola has focus $(0, 2)$ and directrix $y = -2$. Find
$y$ when $x = 8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{3. Where a line meets a conic}
\begin{formulabox}
\textbf{Substitute, do not guess.} A line solved for $y$ drops straight into
the conic and leaves one quadratic in $x$.\\[3pt]
\textbf{The root count is geometry:} two real roots means the line cuts the
curve twice (a secant); one repeated root means it touches once (a tangent);
no real roots means it misses. Check the discriminant if the question asks
about tangency.\\[3pt]
\textbf{Finish the job:} put each root back into the \emph{line} --- the
easier equation --- and report ordered pairs.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Where does $y = x - 1$ meet $x^2 + y^2 = 13$?
\step{The line is already solved for $y$, so substituting it into the circle
turns a two-variable system into one quadratic.}
\step{$x^2 + (x-1)^2 = 13$ gives $2x^2 - 2x - 12 = 0$, so $x^2 - x - 6 = 0$ and
$(x-3)(x+2) = 0$; then $y = x - 1$ gives $2$ and $-3$.}
\par
$\ans{(3,\ 2) \text{ and } (-2,\ -3)}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Where does $y = x + 2$ meet $x^2 + y^2 = 10$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(1,\ 3) \text{ and } (-3,\ -1)}$}
\end{tryitbox}

\skillheading{4. Where two conics meet}
\begin{formulabox}
\textbf{Eliminate the squared variable, not the linear one.} If one curve
gives $x^2$ in terms of $y$ (a parabola does), substitute that whole
expression into the other equation --- you get a quadratic in $y$ with no
square roots anywhere.\\[3pt]
For $xy = c$, substitute $y = c/x$ and clear the denominator; the result is a
quadratic in $x^2$.\\[3pt]
\textbf{(!) Test every value you find.} A $y$ that forces $x^2 < 0$ gives no
real point, and $x^2 = 0$ gives only one. Two conics can meet in $0$, $1$, $2$,
$3$ or $4$ points.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Where does $x^2 + y^2 = 25$ meet $y = x^2 - 13$?
\step{The parabola gives $x^2 = y + 13$, so substituting for $x^2$ removes
$x$ completely and leaves a quadratic in $y$.}
\step{$(y+13) + y^2 = 25$ gives $y^2 + y - 12 = 0$, so $y = 3$ or $y = -4$;
then $x^2 = 16$ and $x^2 = 9$ give $x = \pm 4$ and $x = \pm 3$.}
\par
$\ans{(4,\ 3),\ (-4,\ 3),\ (3,\ -4) \text{ and } (-3,\ -4)}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Where does $x^2 + y^2 = 20$ meet $xy = 8$?\\[2pt]
\rotatebox{180}{\footnotesize check:
$\ans{(2,\ 4),\ (4,\ 2),\ (-2,\ -4) \text{ and } (-4,\ -2)}$}
\end{tryitbox}
