\sstitleblock{Counting Steps, the Line Behind a Sequence, and Counting Terms}

\skillheading{Skill 1 --- Count steps, not terms}

\begin{formulabox}
\textbf{Explicit rule:} $a_n = a_1 + (n-1)d$, where $d$ is the common difference.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The first term costs no steps, so reaching term $n$ takes $n-1$ of them.
  \item \textbf{The $n=1$ test:} substitute $n = 1$. A correct rule returns
        $a_1$ exactly; a rule using $nd$ returns the second term instead.
  \item Going backwards works too: to find \emph{which} term equals a value $V$,
        solve $a_1 + (n-1)d = V$, then substitute your $n$ back in.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad An arithmetic sequence has $a_1 = 4$ and $d = 9$. Find the
$11$th term.
\step{The explicit rule jumps straight to any term, and its whole risk is the
step count, so write $(n-1)$ before substituting a single number.}
\step{$a_{11} = 4 + (11-1)(9) = 4 + 90$}

The $11$th term is $\ans{94}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad An arithmetic sequence has $a_1 = 7$ and $d = 5$. Find the
$13$th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{67}$}
\end{tryitbox}

\skillheading{Skill 2 --- The line behind the sequence}

\begin{formulabox}
\textbf{Linear form:} $a_n = dn + b$ with $b = a_1 - d$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The slope is the common difference $d$: one step along, one $d$ up.
  \item The constant $b$ is the value at $n = 0$ --- one step \emph{before} the
        sequence starts. It is the $y$-intercept of the line, never $a_1$.
  \item Same $n=1$ test: $d(1) + (a_1 - d) = a_1$ $\checkmark$. If your rule
        returns $a_2$ at $n=1$, you used $a_1$ as the constant.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The sequence $5$, $12$, $19$, \ldots\ has $a_1 = 5$ and
$d = 7$. Write it in linear form and find the $9$th term.
\step{The slope is the common difference, and the constant is one step below the
first term, so build $b = a_1 - d$ before touching the index.}
\step{$b = 5 - 7 = -2$, so $a_n = 7n - 2$ and $a_9 = 7(9) - 2 = 63 - 2$}

The $9$th term is $\ans{61}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The sequence $8$, $13$, $18$, \ldots\ has $a_1 = 8$ and
$d = 5$. Write it in linear form and find the $12$th term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{63}$}
\end{tryitbox}

\skillheading{Skill 3 --- How many terms are in the sum?}

\begin{formulabox}
\textbf{Term count:} from term $p$ through term $q$ inclusive there are
$q - p + 1$ terms. \qquad
\textbf{Sum:} $S = \dfrac{(\text{number of terms})}{2}\,(\text{first} + \text{last})$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Subtracting index numbers counts \emph{gaps}; adding $1$ turns gaps back
        into terms.
  \item The last term of the first $n$ terms is $a_1 + (n-1)d$ --- using
        $a_1 + nd$ sums one term too many.
  \item Doubtful? Test your counting rule on a three-term stretch you can count
        on your fingers.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the sum of the first $10$ terms of $2$, $7$, $12$,
\ldots
\step{The sum formula needs the true last term, so find $a_{10}$ with the
$(n-1)$ rule first rather than reaching for $a_1 + nd$.}
\step{$a_{10} = 2 + 9(5) = 47$, so $S_{10} = \dfrac{10}{2}(2 + 47) = 5 \times 49$}

The sum is $\ans{245}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the sum of the first $8$ terms of $3$, $10$, $17$,
\ldots\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{220}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An off-by-one index almost always produces a clean, whole
number, so nothing looks wrong. Run the $n=1$ test on every rule you write, and
substitute your answer back into the sequence before you trust it.
\end{watchoutbox}
