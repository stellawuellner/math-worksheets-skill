\sstitleblock{Radical Equations, Domains, and Decimals}

\skillheading{Skill 1 --- The domain of a radical}

\begin{formulabox}
\textbf{Rule:} $\sqrt{A}$ is defined exactly where $A \ge 0$. Set the radicand
$\ge 0$ and solve. If the radical sits in a \emph{denominator}, use $> 0$
instead: the radicand may not be zero there.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Give the domain of $f(x) = \sqrt{3x-12}$.
\step{The only restriction a square root carries is that its radicand cannot be
negative, so the domain question becomes an inequality.}
\step{$3x - 12 \ge 0$ gives $3x \ge 12$, and dividing by the positive $3$ keeps
the direction.}

$\Rightarrow\quad \ans{x \ge 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Give the domain of $g(x) = \sqrt{10-5x}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x \le 2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Isolate, then undo the radical}

\begin{formulabox}
\textbf{Method:} strip everything off the radical first, then square both sides.
\[ \sqrt{A} = B \ \Longrightarrow \ A = B^2 \]
Squaring before isolating leaves a cross term and makes the equation harder.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\sqrt{x-1} + 4 = 9$.
\step{Squaring is only worth doing once the root stands alone, so undo the
$+4$ first.}
\step{$\sqrt{x-1} = 5$, so $x - 1 = 25$. Check: $\sqrt{25}+4 = 9$.}

$\Rightarrow\quad \ans{x = 26}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sqrt{x+3} - 2 = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 22}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reject the extraneous candidate}

\begin{formulabox}
\textbf{Why it happens:} squaring loses the sign, so the squared equation can
have roots the original never had. Being inside the domain is \emph{not} enough
--- every candidate must be substituted into the original equation. A candidate
that makes the isolated radical equal a negative quantity always fails.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\sqrt{x+12} = x$.
\step{The right side is a variable and could be negative, which is exactly when
squaring manufactures a false root, so both candidates will need testing.}
\step{$x + 12 = x^2$ gives $x^2 - x - 12 = 0$, so $(x-4)(x+3) = 0$. Testing:
$\sqrt{16} = 4$ works; $\sqrt{9} = 3$ is not $-3$.}

$\Rightarrow\quad \ans{x = 4}$ \ (the candidate $-3$ is extraneous)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sqrt{x+20} = x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$ (the candidate $-4$ is extraneous)}
\end{tryitbox}

\skillheading{Skill 4 --- Reporting a rounded decimal}

\begin{formulabox}
\textbf{Method:} carry full precision through the working and round only at the
very end, to the number of places asked for. Rounding an intermediate value and
then squaring it magnifies the error.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\sqrt{x-3} = 2.5$, giving $x$ to two decimal
places.
\step{The radical is already alone, so squaring releases $x$ in one step, and the
decimal only has to survive that one operation.}
\step{$x - 3 = 2.5^2 = 6.25$, so $x = 9.25$.}

$\Rightarrow\quad \ans{9.25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sqrt{x+1} = 3.8$, giving $x$ to two decimal
places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13.44}$}
\end{tryitbox}
