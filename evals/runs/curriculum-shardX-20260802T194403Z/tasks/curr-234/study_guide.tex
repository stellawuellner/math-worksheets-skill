\sstitleblock{Diagnosing Swapped Input--Output and Notation Errors}

\skillheading{Skill 1 --- What $f(a)$ actually names}
\begin{formulabox}\textbf{Read the notation out loud:}\\[3pt]
$f(4) = 14$ says ``the function $f$ turns the \textbf{input} $4$ into the
\textbf{output} $14$.'' The number \emph{inside} the parentheses is always the
input; the number \emph{after the equals sign} is always the output.\\[3pt]
$f(\ )$ is the \textbf{name of a rule}, not a multiplication. So
$f(2+3)$ means ``apply the rule to $5$'' --- it is not $f \cdot (2+3)$, and it
is not $f(2) + f(3)$.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $f(x) = 3x + 2$. \ Find $f(4)$.
\step{The $4$ sits inside the parentheses, so it is the input --- this is an
evaluate question: substitute and compute.}
\step{Replace every $x$ in the rule with $4$: \ $f(4) = 3(4) + 2 = 12 + 2$, so
$f(4) = \ans{14}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $g(x) = 5x - 1$. \ Find $g(4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{19}$}\end{tryitbox}

\skillheading{Skill 2 --- Evaluate, or solve?}
\begin{formulabox}\textbf{The one-question test:} \emph{where is the number
written?}\\[3pt]
\begin{tabular}{@{}l@{\quad}l@{\quad}l@{}}
\textbf{Given} & \textbf{Question type} & \textbf{What you do}\\
inside the parentheses, $f(3)$ & evaluate & substitute, then compute\\
across the equals sign, $f(x) = 3$ & solve & write an equation, solve for $x$\\
\end{tabular}\\[3pt]
The two questions run through the same rule in \emph{opposite directions}, so
they almost never have the same answer. Always check a solved input by
evaluating it forwards.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $f(x) = 3x + 2$. \ Find $x$ when $f(x) = 17$.
\step{The $17$ is across the equals sign from $f(x)$, so it is an output --- that
makes this a solve question, not a substitution.}
\step{Write the equation and undo the rule: \ $3x + 2 = 17$, so $3x = 15$ and
$x = \ans{5}$. Check forwards: $3(5) + 2 = 17$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $g(x) = 5x - 1$. \ Find $x$ when $g(x) = 34$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}\end{tryitbox}
\begin{watchoutbox}\textbf{(!) Watch out:} substituting the output is the most
common slip on this page. If you are given $f(x) = 17$ and you compute $f(17)$,
you have answered a question nobody asked.
\end{watchoutbox}

\skillheading{Skill 3 --- Reading a function table both ways}
\begin{formulabox}\textbf{Rule:} a table has an \textbf{input row} (the letter
in the parentheses) and an \textbf{output row} (the $f(\ )$ row). To find
$h(3)$, find $3$ in the \emph{input} row and read \emph{down}. To answer ``which
input gives $3$?'', find $3$ in the \emph{output} row and read \emph{up}. Same
table, opposite directions --- so always ask which row you started in.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Find $h(3)$ from this table.
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
$t$ & 1 & 2 & 3 & 4 \\\hline
$h(t)$ & 6 & 9 & 1 & 3 \\\hline
\end{tabular}
\end{center}
\step{$h(3)$ gives an input, so start in the $t$ row --- reading the $h(t)$ row
first is the error this table is built to catch.}
\step{Find $3$ in the $t$ row (fourth column of the header) and read down:
$h(3) = \ans{1}$. Reading the other row instead would have found $3$ under
$t = 4$, an answer to the \emph{reverse} question.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Find $k(2)$.
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|}\hline
$t$ & 1 & 2 & 3 & 4 \\\hline
$k(t)$ & 9 & 5 & 8 & 2 \\\hline
\end{tabular}
\end{center}
\rotatebox{180}{\footnotesize check: $\ans{5}$}\end{tryitbox}
