\sstitleblock{Measuring with Units: Gaps and Overlaps}

% =================== SKILL 1 ===================
\skillheading{Fix a count with gaps or overlaps}

\begin{formulabox}
\textbf{Rule:} Units must sit end to end.\par
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Gaps skip some length $\to$ the count is too SMALL $\to$ add the
    hidden units back.
  \item Overlaps count some length twice $\to$ the count is too BIG $\to$
    take the extra units away.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Sam overlapped blocks, counting 2 too many. His count
was 9 blocks. What is the real length?
\step{Overlaps make a count too big --- so the fix is to take the extra
blocks away.}
\step{$9 - 2 = 7$.}

$\Rightarrow\quad$ the real length is $\ans{7}$ blocks
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Pia left gaps that hid 2 clips. She counted 8 clips.
What is the real length?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$ clips}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A wrong measuring count is not fixed by measuring
louder --- look at HOW the units were laid down. Ask: did any unit overlap?
Was any length left bare?
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Compare two measurements}

\begin{formulabox}
\textbf{Rule:} Measure both things with the SAME unit. Then compare the
counts: more units means longer, and subtracting the counts tells you by how
much.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The pencil is 9 cubes long and the pen is 6 cubes
long. How many cubes longer is the pencil?
\step{Both counts use the same unit, so the difference of the counts is the
difference in length.}
\step{$9 - 6 = 3$.}

$\Rightarrow\quad$ the pencil is $\ans{3}$ cubes longer
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One string is 12 clips long, another is 7 clips long.
How many clips longer is the first?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$ clips}
\end{tryitbox}
