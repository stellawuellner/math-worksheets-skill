\wstitleblock{Triangle Congruence: SSS, SAS, ASA, AAS and HL}{High-School Geometry}{}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Matching marks mean congruent parts: one tick $=$ one tick, two ticks $=$ two
        ticks, one arc $=$ one arc, and a small square means a right angle.
  \item A criterion needs \textbf{three} pairs, and \emph{where} they sit matters.
        In SAS the angle is \textbf{between} the two sides; in ASA the side is
        \textbf{between} the two angles.
  \item Two arrangements are \emph{not} criteria: SSA (two sides and a
        non-included angle) and AAA. HL is the one legal SSA-shaped rule, and only
        for right triangles.
  \item When you name a criterion, name the three pairs it uses. ``Looks congruent''
        is not evidence.
\end{itemize}
\normalsize

\problem[5cm]{\textbf{Why SSA is not a criterion.} Both triangles below share the
angle at $A$, the side $\overline{AR}$, and a congruent swinging side (single tick)
--- yet they are not congruent.
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \coordinate (A) at (0,0);
  \coordinate (R) at (3.18,3.1);
  \coordinate (P) at (1.35,0);
  \coordinate (Q) at (5.0,0);
  \draw[thick] (A) -- (Q);
  \draw[thick] (A) -- (R);
  \draw[thick] (R) -- node[sloped]{$|$} (P);
  \draw[thick, dashed] (R) -- node[sloped]{$|$} (Q);
  \node[below left] at (A) {$A$};
  \node[above] at (R) {$R$};
  \node[below] at (P) {$P$};
  \node[below] at (Q) {$Q$};
  \pic[draw, angle radius=8mm] {angle = Q--A--R};
\end{tikzpicture}

congruence criterion, and say what extra condition turns it back into a valid rule.
