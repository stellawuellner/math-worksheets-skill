\wstitleblock{Writing Inequalities from Verbal Constraints}{Grade 6--7 \quad Expressions and Equations}{}

\begin{center}
\textit{Turning a comparison phrase into a symbol, building a one-variable
inequality from a story, and solving it and reading the answer back into the
situation}
\end{center}

\noindent\textbf{The four phrases that decide the symbol.}
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{@{}llp{6.4cm}@{}}
\toprule
\textbf{Phrase} & \textbf{Symbol} & \textbf{Is the boundary allowed?} \\
\midrule
more than, over & $>$  & no --- open circle on the number line \\
less than, under & $<$ & no --- open circle \\
at least, no less than, minimum & $\ge$ & yes --- closed circle \\
at most, no more than, maximum & $\le$ & yes --- closed circle \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
\item Say in words what the variable counts or measures before you write
anything --- that sentence is part of the answer.
\item Write the inequality first, then solve it. Graph the solution on the
number line under the problem, using an open or closed circle to match your
symbol.
\item If a step multiplies or divides both sides by a negative number, the
inequality symbol flips.
\end{itemize}

\problem[4cm]{A ride at the fair requires riders to be \emph{at least} $48$
inches tall. Let $h$ be a rider's height in inches. Write an inequality for the
heights that may ride, and graph it.
\begin{center}
\begin{tikzpicture}[x=0.72cm]
  \draw[thick,<->] (-0.8,0) -- (8.8,0);
  \foreach \i in {0,...,8} \draw (\i,0.12) -- (\i,-0.12);
  \foreach \i/\lab in {0/44, 2/46, 4/48, 6/50, 8/52}
    \node[below=3pt, font=\small] at (\i,0) {$\lab$};
\end{tikzpicture}
\end{center}}
