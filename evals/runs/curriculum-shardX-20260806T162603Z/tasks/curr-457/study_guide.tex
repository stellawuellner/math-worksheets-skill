\sstitleblock{Differentiation Rules Reference}

\begin{formulabox}
\textbf{Choose the rule before you differentiate.} A sum of powers: power rule
term by term. Two non-constant factors multiplied: product rule. One expression
divided by another: quotient rule. A constant over a power (like $5/x^3$) is
\emph{not} a quotient-rule problem --- rewrite it as $5x^{-3}$.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Power Rule and Rewriting First}

\begin{formulabox}
\textbf{Rule:} for every real $n$,
\[
  \frac{d}{dx}\,x^n = nx^{n-1},
\]
and constants differentiate to $0$. Rewrite before you start: $c/x^n$ is
$cx^{-n}$, and $\sqrt{x}$ is $x^{1/2}$. The exponent goes down by one even when
it is negative or fractional --- $x^{-3}$ becomes $-3x^{-4}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = 3x^4 - 7x + 2$
\step{Every term is already a power of $x$, so the power rule applies term by term --- no product or quotient rule is involved.}
\step{$3(4)x^3 = 12x^3$, \ $-7x^1$ gives $-7$, \ and the constant $2$ gives $0$.}

$\Rightarrow\quad y' = $ $\ans{12x^3 - 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = 5x^3 + 2/x$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15x^2 - \tfrac{2}{x^2}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The Product Rule}

\begin{formulabox}
\textbf{Rule:}
\[
  (uv)' = u'v + uv'
\]
Derivative of the first times the second, plus the first times the derivative of
the second. It is a \emph{sum}, not a product: $(uv)' \neq u'v'$. Label $u$ and
$v$ in the margin before differentiating, then expand and collect.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = (x+3)\left(x^2-1\right)$
\step{Two non-constant factors are multiplied, so this is the product rule --- take $u = x+3$ and $v = x^2-1$.}
\step{$u'v + uv' = (1)\left(x^2-1\right) + (x+3)(2x) = x^2 - 1 + 2x^2 + 6x = 3x^2 + 6x - 1$.}

$\Rightarrow\quad y' = $ $\ans{3x^2 + 6x - 1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = (2x-1)\left(x^2+4\right)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6x^2 - 2x + 8}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The Quotient Rule}

\begin{formulabox}
\textbf{Rule:}
\[
  \left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}
\]
The order in the numerator matters --- swapping the two terms flips every sign
in the answer. Leave the denominator as $v^2$ and simplify only the numerator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = (x+1)/(x-2)$
\step{One expression is divided by another and neither is constant, so use the quotient rule with $u = x+1$ and $v = x-2$.}
\step{$\frac{(1)(x-2) - (x+1)(1)}{(x-2)^2} = \frac{x-2-x-1}{(x-2)^2} = \frac{-3}{(x-2)^2}$.}

$\Rightarrow\quad y' = $ $\ans{\tfrac{-3}{(x-2)^2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = 3x/(x+4)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{12}{(x+4)^2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the two commonest slips are $u'v'$ for a product and
$u'/v'$ for a quotient. Both are wrong, and both give an answer that looks
tidier than the right one --- tidiness is not evidence.
\end{watchoutbox}
