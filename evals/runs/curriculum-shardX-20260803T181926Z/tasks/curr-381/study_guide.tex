\sstitleblock{Composing Functions from Symbolic Rules}

\skillheading{1. Run a composition at a number, inside rule first}

\begin{formulabox}
\textbf{Notation.} $(f \circ g)(x)$ means $f\big(g(x)\big)$. The rule written
next to the input --- here $g$ --- is the \emph{inner} rule, and it always runs
first.\\[3pt]
\textbf{At a number:} compute $g(\text{number})$, then feed that single output
into $f$. Two machines in a row, never both at once.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x - 6$ and $g(x) = 5x$. Find $(f \circ g)(4)$.
\step{The circle notation puts $g$ next to the input, so $g$ runs first and its
output becomes the input of $f$.}
\step{$g(4) = 5 \cdot 4 = 20$, and then $f(20) = 20 - 6$.}

$(f \circ g)(4) = \ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 2x$ and $g(x) = x + 3$. Find $(f \circ g)(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

\skillheading{2. Build the composed rule from two symbolic rules}

\begin{formulabox}
\textbf{The substitution rule.} Wherever the outer rule shows its input
variable, write the \emph{whole} inner rule in parentheses --- in every slot,
not just the first one. Only then expand and combine like terms.\\[3pt]
$f(x) = x^2 + 2x$ with input $\square$ becomes $f(\square) = \square^2 + 2\square$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^2 + 2x$, $g(x) = x - 1$. Write $(f \circ g)(x)$.
\step{$g$ is inner, so every input slot of $f$ takes $(x-1)$ --- and $f$ has two
slots, one squared and one linear.}
\step{$(x-1)^2 + 2(x-1) = x^2 - 2x + 1 + 2x - 2$, and the $-2x$ and $+2x$
cancel.}

$(f \circ g)(x) = \ans{x^2 - 1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^2 - 4$, $g(x) = x + 2$. Write $(f \circ g)(x)$
in simplest form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 4x}$}
\end{tryitbox}

\skillheading{3. Order matters: $f(g(x))$ is not $g(f(x))$}

\begin{formulabox}
Swapping which rule is inner usually changes the function, so
$f \circ g$ and $g \circ f$ must be built separately.\\[3pt]
\textbf{Quick test:} pick one input and run it both ways. If the two outputs
differ, the two compositions are different rules --- one counterexample
settles it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 3x - 2$, $g(x) = x^2 + 1$. Build both orders.
\step{Each order names a different inner rule, so do two separate
substitutions rather than reusing the first answer.}
\step{$f(g(x)) = 3(x^2+1) - 2 = 3x^2 + 3 - 2$, while
$g(f(x)) = (3x-2)^2 + 1 = 9x^2 - 12x + 4 + 1$.}

$(f \circ g)(x) = \ans{3x^2 + 1}$ \quad but \quad $(g \circ f)(x) = \ans{9x^2 - 12x + 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x + 5$, $g(x) = x^2$. Write $(g \circ f)(x)$
expanded.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + 10x + 25}$}
\end{tryitbox}

\skillheading{4. Inverses: composition hands the input straight back}

\begin{formulabox}
Two rules are inverses exactly when composing them --- \emph{both ways} ---
returns the input untouched: $f\big(g(x)\big) = x$ and $g\big(f(x)\big) = x$.
One direction alone is not the definition, so always check the pair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Are $f(x) = 2x - 6$ and $g(x) = \frac{x+6}{2}$ inverses?
\step{The definition is a composition statement, so substitute rather than
compare the two formulas by eye.}
\step{$f(g(x)) = 2\!\left(\frac{x+6}{2}\right) - 6 = (x + 6) - 6$, because the
factor $2$ cancels the division.}

$f\big(g(x)\big) = \ans{x}$, and $g\big(f(x)\big)$ works out the same way
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 5x + 1$, $g(x) = \frac{x-1}{5}$. Simplify
$g\big(f(x)\big)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} composition is not multiplication.
$(f \circ g)(x)$ is $f\big(g(x)\big)$, while $f(x) \cdot g(x)$ multiplies the
two outputs --- different operations, different answers. And when you
substitute a whole rule, keep it in parentheses: forgetting them turns
$(x-3)^2$ into $x - 3^2$.
\end{watchoutbox}
