\wstitleblock{Using Zero and Negative Exponents}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{The zero exponent, rewriting a negative exponent as a fraction, and the laws when
the result is zero or negative}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Any non-zero base to the power zero is $1$. Dividing a power by itself proves it.
  \item A negative exponent means \emph{reciprocal}, never a negative answer:
        $x^{-n} = \dfrac{1}{x^n}$ for $x \neq 0$.
  \item Final answers use positive exponents only, and numerical answers are exact
        fractions --- not decimals.
\end{itemize}
