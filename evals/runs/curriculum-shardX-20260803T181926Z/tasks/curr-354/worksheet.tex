\wstitleblock{Sign and Conjugate Errors with Complex Numbers}{Algebra 2}{}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Write every answer in the form $a + bi$ with $a$ and $b$ real, unless the problem asks for something else.
  \item Remember two facts all the way through: $i^2 = -1$, and the conjugate of $c + di$ is $c - di$ --- only the sign in front of the $i$ changes.
  \item On a ``find the mistake'' problem, name the wrong move first, then redo the work correctly. Both parts are graded.
\end{itemize}
\normalsize

\problem[4cm]{Subtract and write the result in the form $a + bi$:
\begin{center}\large $(3 - 8i) - (5 - 2i)$\end{center}}

\problem[4.5cm]{Multiply and write the result in the form $a + bi$:
\begin{center}\large $(4 + 3i)(2 - 5i)$\end{center}}

\problem[4.5cm]{Multiply the conjugate pair and show why the result has no $i$ in it:
\begin{center}\large $(5 + 2i)(5 - 2i)$\end{center}}

\problem[6cm]{\textbf{Find the mistake.} Nadia wants a product that is a real number.
She writes
\begin{center}\large $(7 - 3i)(7 - 3i) = 58$\end{center}
and says, ``multiplying by the conjugate clears the $i$.''
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain what went wrong: which factor does the conjugate rule actually call for?
  \item[(b)] Compute $(7 - 3i)(7 - 3i)$ correctly.
\end{itemize}}

\problem[5.5cm]{Divide. Multiply by the conjugate of the denominator and write the result in the form $a + bi$:
\begin{center}\large $\dfrac{8 + i}{2 - 3i}$\end{center}}

\problem[6.5cm]{\textbf{Find the mistake.} Marcus simplifies $\dfrac{20}{3 + i}$.
He multiplies the numerator and the denominator by $3 - i$, and then writes the new
denominator as $8$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain why $8$ is wrong, and give the correct denominator.
  \item[(b)] Finish the division correctly and write the result in the form $a + bi$.
\end{itemize}}

\problem[6cm]{Solve. Give both solutions in the form $a + bi$, and state how the two answers are related:
\begin{center}\large $x^2 - 6x + 13 = 0$\end{center}}

\problem[6cm]{Find all four zeros of $P(x) = x^4 - 16$, real and non-real.
Check that the non-real zeros you write down form a conjugate pair.}
