\wstitleblock{Arithmetic Sequences as Linear Functions}{Algebra 1}{}

\begin{center}
Writing recursive rules from listed sequences and from real contexts, running a
recursive rule forward, and converting a recursive rule to an explicit linear
function
\end{center}

\noindent\small\textbf{Directions.} A recursive formula for an arithmetic
sequence always has \textbf{two} lines --- the starting value and the step:
\[ a_1 = \text{first term}, \qquad a_n = a_{n-1} + d \quad (n \ge 2). \]
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt, topsep=2pt]
  \item Write \emph{both} lines for every problem, then answer the question
        asked. A rule with no starting value names infinitely many sequences.
  \item Keep the units of the situation in your answer where the problem
        states them.
  \item Getting from $a_1$ to $a_n$ takes $n-1$ steps, not $n$.
\end{itemize}
\normalsize

\problem[8cm]{\textbf{From a list.} A tiling pattern uses this many tiles in
each stage:
\[ 5, \quad 12, \quad 19, \quad 26, \quad \ldots \]
Write the recursive formula for $a_n$, then find $a_6$, the number of tiles in
stage 6.}

\problem[8cm]{\textbf{From a list, decreasing.} A drone descends in equal steps,
and its altitude in meters at each reading is:
\[ 48, \quad 41, \quad 34, \quad 27, \quad \ldots \]
Write the recursive formula for $a_n$, then find $a_9$. Say what a negative
value of $a_9$ means for the drone.}

\problem{\textbf{Stacked chairs.} One chair on the floor stands 82
centimeters tall. Each extra chair stacked on top adds 9 centimeters to the
height of the stack. Let $a_n$ be the height in centimeters of a stack of $n$
chairs.\\[3pt]
Write the recursive formula for $a_n$, stating $a_1$ and the step, then find the
height of a stack of 12 chairs.\par\vspace*{8cm}
\answerline{cm}}

\problem[8cm]{\textbf{Run the rule forward.} A sequence is defined by
\[ a_1 = 7, \qquad a_n = a_{n-1} + 6 \quad (n \ge 2). \]
Find $a_{15}$ without listing every term, and explain in one sentence how many
times the step of 6 is applied.}

\problem[8.5cm]{\textbf{Run the rule backward.} A sequence is defined by
\[ a_1 = 100, \qquad a_n = a_{n-1} - 12 \quad (n \ge 2). \]
Which term of this sequence equals 16? Set up an equation in $n$ and solve it;
do not answer by listing terms.}

\problem{\textbf{Draining tank.} A tank holds 240 liters of water. At the
end of every hour a pump removes 15 liters. Let $a_n$ be the number of liters
left at the end of hour $n$.\\[3pt]
Write the recursive formula for $a_n$, being careful about what $a_1$ is, then
find how many liters are left at the end of hour 8.\par\vspace*{9cm}
\answerline{L}}

\problem[9cm]{\textbf{Gym membership.} A gym charges a one-time joining fee of
60 dollars, then 35 dollars at the end of each month. Let $a_n$ be the total
amount paid, in dollars, after $n$ months of membership.\\[3pt]
Write the recursive formula for $a_n$, convert it to an explicit rule of the
form $a_n = mn + b$, and use the explicit rule to find the total paid after 24
months.}

\problem[8.5cm]{\textbf{Recursive to explicit.} A sequence is defined by
\[ a_1 = -5, \qquad a_n = a_{n-1} + 4 \quad (n \ge 2). \]
Write the explicit rule as a linear function of $n$, name the slope and the
value of the function at $n = 0$, and find $a_{20}$.}

\problem[9cm]{\textbf{Missing first term.} A table of an arithmetic sequence
records only two of its terms.
\begin{center}\large
\begin{tabular}{|l|c|c|}\hline
$n$ & 3 & 7 \\[0.5cm]\hline
$a_n$ & 23 & 47 \\[0.5cm]\hline
\end{tabular}
\end{center}
Find the common difference, then work back to $a_1$ and write the full recursive
formula. Give the common difference and $a_1$ as your two answers.}

\problem[9.5cm]{\textbf{Synthesis: theater seating.} Row 1 of a theater has 18
seats, and each row behind it has 3 more seats than the row in front of it.\\[3pt]
Write the recursive formula for $a_n$, the number of seats in row $n$. Then find
the \textbf{total} number of seats in the first 15 rows, and explain why adding
the first and last row and multiplying by half the number of rows gives the same
total.}
