\sstitleblock{Building Equivalent Ratio Tables}

\noindent\small A ratio table is one ratio written over and over at different
sizes. Every column is the first column times some number $k$ --- the same $k$
for both rows. Find $k$ first, and every question on the worksheet becomes one
multiplication.
\normalsize

\skillheading{1. Filling in a missing table entry}
\begin{formulabox}
\textbf{Rule:} if $a : b$ is the ratio, then every column is $ak : bk$.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Use the row where \emph{both} entries are known to get
        $k = \dfrac{\text{new}}{\text{old}}$.
  \item Multiply the partner entry by the same $k$.
  \item Never add: $a + n : b + n$ is a \emph{different} ratio.
\end{itemize}
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A table built from $3 : 4$ has a column whose first
entry is $12$. Find the second entry.
\step{The first row has both a known old and a known new value, so it is the
row that reveals the scale factor.}
\step{$k = 12 \div 3 = 4$, so the second entry is $4 \times 4$.}
\par
$\ans{16}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A table built from $5 : 9$ has a column starting with
$35$. Find its second entry.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{63}$}
\end{tryitbox}

\skillheading{2. Finding the scale factor between two columns}
\begin{formulabox}
\textbf{Rule:} $k = \dfrac{\text{new entry}}{\text{old entry}}$, taken from
the row where both numbers are given.\\[3pt]
The factor does not have to be a whole number --- $2.5$, $\tfrac{3}{2}$ and
$0.4$ are all legal scale factors, and $k < 1$ simply scales the table down.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A ratio table's first row goes from $6$ to $15$. What
is the scale factor?
\step{Scale factor is a multiplier, not a difference, so divide the new entry
by the old one instead of subtracting.}
\step{$k = 15 \div 6 = \dfrac{5}{2}$, and $6 \times 2.5 = 15$ checks out.}
\par
$\ans{2.5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A row of a ratio table goes from $8$ to $28$. Find the
scale factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3.5}$}
\end{tryitbox}

\skillheading{3. Adding the unit-rate column}
\begin{formulabox}
\textbf{Rule:} the unit-rate column is the one whose \emph{second} row entry
is $1$. Divide both entries of a known column by that column's second entry:
\[ \frac{a}{b} : 1 \]
The top entry of that column is the rate --- ``how many per one''.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A runner covers $12$ laps in $8$ minutes. Add the
per-minute column.
\step{A unit rate is the entry paired with $1$, so divide the whole column by
the time entry.}
\step{$12 \div 8 = \dfrac{3}{2}$ laps for $8 \div 8 = 1$ minute.}
\par
$\ans{1.5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A printer prints $27$ pages in $6$ minutes. Find the
pages-per-minute unit rate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4.5}$}
\end{tryitbox}

\skillheading{4. Deciding whether two tables are equivalent}
\begin{formulabox}
\textbf{Rule:} write each ratio as a fraction and reduce, or compare unit
rates. Two tables are equivalent exactly when the reduced fractions match.\\[3pt]
Cross-products say the same thing: $\dfrac{a}{b} = \dfrac{c}{d}$ precisely
when $ad = bc$. Use $<$, $>$, or $=$ to report the result.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Is the table $6 : 9$ equivalent to the table
$10 : 15$?
\step{Equivalent means the same ratio at a different size, so reduce both to
lowest terms and compare.}
\step{$6 : 9$ reduces to $2 : 3$, and $10 : 15$ also reduces to $2 : 3$.}
\par
$\ans{6/9 = 10/15}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Compare the tables $2 : 3$ and $5 : 7$ with $<$, $>$,
or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2/3 < 5/7}$}
\end{tryitbox}
