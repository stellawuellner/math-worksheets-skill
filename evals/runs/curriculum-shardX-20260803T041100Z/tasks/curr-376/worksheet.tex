\wstitleblock{Converting Between Exponential and Logarithmic Forms}{Algebra 2 \quad Exponential and Logarithmic Functions}{}

\begin{center}
\textit{Rewriting an exponential statement as a logarithm, rewriting a logarithm
as an exponential statement, evaluating logarithms by converting, changing base,
and solving equations by switching forms}
\end{center}

\noindent\textbf{The one idea behind every problem.} A logarithm is an exponent.
The two forms below say exactly the same thing about the same three numbers ---
only the question changes. Exponential form asks ``what is the result?'';
logarithmic form asks ``what is the exponent?''

\begin{center}
\fbox{\parbox{0.92\linewidth}{\centering
$b^{\,y} = x$ \qquad$\Longleftrightarrow$\qquad $\log_b x = y$
\\[4pt]
\textbf{base} $b$ stays the base in both forms
\\[2pt]
\textbf{exponent} $y$ is what the logarithm equals
\\[2pt]
\textbf{result} $x$ is what the logarithm is taken of}}

\smallskip
\small\textit{Reference card. No values are filled in --- use the numbers given
in each problem.}\normalsize
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Before converting, label which given number is the base, which is the
exponent, and which is the result.
\item Give exact answers unless the problem asks for a decimal; then round to
three decimal places.
\item Write your final answer on the answer line.
\end{itemize}

\problem[5.5cm]{The statement $5^3 = 125$ is in exponential form.

\smallskip
\textbf{(a)} Name the base, the exponent, and the result.

\smallskip
\textbf{(b)} Rewrite the statement in logarithmic form.

\smallskip
\textbf{(c)} What number does $\log_5 125$ equal?}

\problem[5.5cm]{The statement $\log_2 32 = 5$ is in logarithmic form.

\smallskip
\textbf{(a)} Which number is the base, and which one is the exponent?

\smallskip
\textbf{(b)} Rewrite the statement in exponential form.

\smallskip
\textbf{(c)} Evaluate the power you wrote in part (b).}

\problem[6cm]{Evaluate $\log_3 81$ without a calculator.

\smallskip
\textbf{(a)} Fill in the table of powers.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
power & $3^1$ & $3^2$ & $3^3$ & $3^4$ \\
\hline
value & \strut\qquad & \qquad & \qquad & \qquad \\
\hline
\end{tabular}
\end{center}
\par
\textbf{(b)} $\log_3 81$ asks which of those exponents produces 81. Write the
exponential statement, then the value of the logarithm.}

\problem[6cm]{A logarithm can be negative, because an exponent can be negative.

\smallskip
Consider the exponential statement $10^{-2} = 0.01$.

\smallskip
\textbf{(a)} Rewrite it in logarithmic form.

\smallskip
\textbf{(b)} What is $\log_{10} 0.01$?

\smallskip
\textbf{(c)} In one sentence, say why the logarithm of a number between 0 and 1
comes out negative when the base is bigger than 1.}

\problem[6.5cm]{Solve $2^x = 40$.

\smallskip
\textbf{(a)} The unknown is the exponent, so the logarithmic form isolates it.
Rewrite the equation in logarithmic form.

\smallskip
\textbf{(b)} Between which two consecutive whole numbers must $x$ lie? Say how
you know from powers of 2.

\smallskip
\textbf{(c)} Give $x$ to three decimal places.}

\problem[6.5cm]{Evaluate $\log_8 2$ exactly.

\smallskip
\textbf{(a)} Write the conversion: $\log_8 2 = y$ becomes what exponential
statement?

\smallskip
\textbf{(b)} 2 is smaller than the base 8, so the exponent is a fraction between
0 and 1. Find it exactly, as a fraction in lowest terms.}

\problem[6.5cm]{Solve $\log_4 x = \dfrac{3}{2}$.

\smallskip
\textbf{(a)} Here the unknown sits inside the logarithm, so exponential form is
the one that isolates it. Rewrite the equation.

\smallskip
\textbf{(b)} Evaluate the power. A fractional exponent is a root and a power:
the denominator is the root.

\smallskip
\textbf{(c)} Check your answer by substituting it back into the original
logarithmic statement.}

\problem[7cm]{Most calculators only have $\ln$ and $\log_{10}$, so a logarithm
in another base has to be rewritten.

\smallskip
\textbf{(a)} Starting from $\log_3 x = y$, convert to exponential form, take the
natural logarithm of both sides, and solve for $y$. Write $\log_3 x$ as a
quotient of natural logarithms.

\smallskip
\textbf{(b)} Use your formula to approximate $\log_3 20$ to three decimal
places.

\smallskip
\textbf{(c)} Write both your part (a) expression and your part (b) decimal on
the answer line.}

\problem[7cm]{Solve $\log_5(2x - 3) = 2$.

\smallskip
\textbf{(a)} Convert to exponential form. The entire expression $2x - 3$ is what
the logarithm was taken of, so it becomes the result.

\smallskip
\textbf{(b)} Solve the equation that remains.

\smallskip
\textbf{(c)} Verify that your value makes $2x - 3$ positive, and say in one
sentence why that check is required.}

\problem[7.5cm]{\textbf{Challenge.} An investment of 1200 dollars grows by 4.5
percent each year, so its value after $t$ years is $1200 \cdot 1.045^{\,t}$.

\smallskip
\textbf{(a)} Write the equation that says the investment has doubled.

\smallskip
\textbf{(b)} Divide both sides by the starting amount, then convert the result
to logarithmic form to isolate $t$.

\smallskip
\textbf{(c)} Find the doubling time to three decimal places.}
