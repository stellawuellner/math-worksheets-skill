\sstitleblock{Building a Coherent Proof from the Givens}

\skillheading{1. A reason for every line}

\begin{formulabox}
\textbf{A justified solution is a proof.} Each line must be the line above it
plus \emph{one} named move:
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item Addition / Subtraction / Multiplication / Division Property of Equality
--- the same operation on both sides.
\item Distributive Property --- rewrites \emph{one} side; it is not a property
of equality.
\item Substitution Property --- replaces a quantity by something already shown
equal to it.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $4x - 9 = 15$, naming each reason.
\step{Every line needs one named move, so undo the operations one at a time
rather than in a single jump.}
\step{$4x - 9 = 15$ (Given); $4x = 24$ (Addition Property of Equality);
$x = 6$ (Division Property of Equality).}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $3x + 8 = 29$ and name the reason on each line.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$ --- Subtraction, then Division}
\end{tryitbox}

\skillheading{2. From a segment picture to an equation}

\begin{formulabox}
\textbf{Segment Addition Postulate:} if $B$ is between $A$ and $C$, then
$AB + BC = AC$. Betweenness gives a \textbf{sum}.\\[3pt]
\textbf{Definition of midpoint:} if $M$ is the midpoint of $\overline{AB}$,
then $AM = MB$, and each equals $\frac{1}{2}AB$. A midpoint gives an
\textbf{equality between the parts} --- and nothing else does. If no midpoint
or congruence is stated, setting two parts equal does not follow.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $B$ is between $A$ and $C$, $AB = x + 4$, $BC = 2x$,
$AC = 19$. Find $x$.
\step{The only stated relationship is betweenness, so the postulate that
applies is Segment Addition --- add the parts, do not equate them.}
\step{$(x + 4) + 2x = 19$, so $3x + 4 = 19$ and $3x = 15$.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $B$ is between $A$ and $C$, $AB = 3x - 2$, $BC = x + 6$,
$AC = 28$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{3. From an angle relationship to an equation}

\begin{formulabox}
\textbf{Each relationship supplies a different number.}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item Angle bisector: the two halves are \emph{equal} to each other.
\item Complementary: the two measures add to $90$.
\item Linear pair / supplementary: they add to $180$.
\item Vertical angles: equal to each other.
\item Same-side interior angles, parallel lines: add to $180$.
\end{itemize}
Name the relationship first --- it is the reason, and it decides the equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Two complementary angles measure $(2x)^\circ$ and
$(3x + 15)^\circ$. Find $x$.
\step{Complementary is the stated relationship, and complementary angles add
to a right angle --- so the equation is a sum, not an equality.}
\step{$2x + (3x + 15) = 90$, so $5x + 15 = 90$ and $5x = 75$.}

$\Rightarrow\quad \ans{x = 15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Two supplementary angles measure $(4x + 10)^\circ$ and
$(2x + 20)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 25}$}
\end{tryitbox}

\skillheading{4. Writing the two-column proof}

\begin{formulabox}
\textbf{Shape of the argument:} start with the givens, end with the statement
you were asked to prove, and let every line in between be licensed by the
lines above it. Work \emph{backwards} first --- ask what would immediately
give the conclusion, then what would give that.\\[3pt]
\textbf{Reasons come from four places only:} Given, a definition, a postulate
or theorem, or a property of equality. ``It looks that way'' is not one of
them.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Given $M$ is the midpoint of $\overline{AB}$, prove
$AM \cong \overline{MB}$.
\step{The conclusion is about the two halves, and the only given is the
midpoint, so the proof is the definition of midpoint plus the link between
equal measures and congruence.}
\step{Each line below carries exactly one reason, ending on the statement that
was to be proved.}

\ssproof{
1.\ $M$ is the midpoint of $\overline{AB}$ & 1.\ Given \\ \hline
2.\ $AM = MB$ & 2.\ Definition of midpoint \\ \hline
3.\ $\overline{AM} \cong \overline{MB}$ & 3.\ Definition of congruent segments
\\ \hline}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $\angle A$ and $\angle B$ are vertical angles. Prove
$\angle A \cong \angle B$ in three lines --- supply both columns.\\[2pt]
\rotatebox{180}{\footnotesize check: Given; Vertical Angles Theorem; Definition of congruent angles}
\end{tryitbox}
