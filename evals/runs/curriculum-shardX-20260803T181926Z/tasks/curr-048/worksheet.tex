\wstitleblock{Where Is the Shape? Position Words on a Shelf}{Kindergarten--Grade 1}{}

\begin{center}
Above and below, left and right, putting shapes in order by position, and
telling what is between
\end{center}

\noindent\textbf{Directions --- read these with your grown-up or teacher:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item Every question is about the shelf picture below. Point to the shape
        first, then answer.
  \item The shelves are numbered. Shelf 1 is the bottom shelf. A shape on a
        \emph{bigger} shelf number is \emph{higher up}.
  \item The spots are numbered from the left. Spot 1 is farthest left. A shape
        in a \emph{bigger} spot number is farther \emph{right}.
\end{itemize}

\problem[5cm]{The circle is on shelf 3. The square is on shelf 2. Write the two
shelf numbers in that order and put $<$ or $>$ between them:
\ 3 \ \underline{\hspace{1.1cm}} \ 2.
\par\smallskip
Now circle the right word. \ The circle is \ \textbf{above} \ / \
\textbf{below} \ the square.}

\problem[5cm]{The rectangle is in spot 1. The diamond is in spot 4. Write the
two spot numbers in that order and put $<$ or $>$ between them:
\ 1 \ \underline{\hspace{1.1cm}} \ 4.
\par\smallskip
Now circle the right word. \ The rectangle is to the \ \textbf{left} \ / \
\textbf{right} \ of the diamond.}

\problem[5cm]{The oval is on shelf 1. The hexagon is on shelf 2. Write the two
shelf numbers in that order and put $<$ or $>$ between them:
\ 1 \ \underline{\hspace{1.1cm}} \ 2.
\par\smallskip
Now circle the right word. \ The oval is \ \textbf{above} \ / \ \textbf{below}
\ the hexagon.}

\problem[6cm]{Three shapes and their shelf numbers: the hexagon is on shelf 2,
the triangle is on shelf 3, and the rectangle is on shelf 1.
\par\smallskip
Write the three shelf numbers in order from the \emph{bottom} shelf to the
\emph{top} shelf:
\ \underline{\hspace{1.1cm}} \, \underline{\hspace{1.1cm}} \,
\underline{\hspace{1.1cm}}
\par\smallskip
Then write the name of the shape that is lowest and the name of the shape that
is highest.}

\problem[5cm]{The hexagon is in spot 4. The square is in spot 2. Write the two
spot numbers in that order and put $<$ or $>$ between them:
\ 4 \ \underline{\hspace{1.1cm}} \ 2.
\par\smallskip
Now circle the right word. \ The hexagon is to the \ \textbf{left} \ / \
\textbf{right} \ of the square.}

\problem[6cm]{Go back to the shelf picture at the top of this page. On shelf 2,
draw a small triangle in the empty space \emph{between} the square and the
hexagon.
\par\smallskip
Now finish this sentence with a position word that tells where your triangle is:
\par\medskip
The triangle I drew is \underline{\hspace{4.5cm}} the square.
\par\medskip
Then tell your grown-up one more true thing about where your triangle is,
using the word \emph{between}.
\noansline}

\problem[6.5cm]{Four shapes and their spot numbers: the square is in spot 2, the
diamond is in spot 4, the circle is in spot 1, and the triangle is in spot 3.
\par\smallskip
Write the four spot numbers in order from the \emph{right} side of the shelf to
the \emph{left} side:
\ \underline{\hspace{1.1cm}} \, \underline{\hspace{1.1cm}} \,
\underline{\hspace{1.1cm}} \, \underline{\hspace{1.1cm}}
\par\smallskip
Then write the name of the shape that is farthest right.}

\problem[7cm]{The triangle is on shelf 3, in spot 3. The oval is on shelf 1, in
spot 2. This shape needs \emph{two} position words.
\par\smallskip
\textbf{(a)} Compare the shelf numbers: \ 3 \ \underline{\hspace{1.1cm}} \ 1.
\ The triangle is \ \textbf{above} \ / \ \textbf{below} \ the oval.
\par\smallskip
\textbf{(b)} Compare the spot numbers: \ 3 \ \underline{\hspace{1.1cm}} \ 2.
\ The triangle is to the \ \textbf{left} \ / \ \textbf{right} \ of the oval.}
