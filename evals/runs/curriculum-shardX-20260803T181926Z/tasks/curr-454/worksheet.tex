\wstitleblock{Diagnosing Direct-Substitution and One-Sided-Sign Errors}{AP Calculus AB / BC --- Limits and Continuity}{}

\noindent\textbf{What direct substitution can and cannot settle:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{A real number} --- the function is continuous there and the substitution \emph{is} the limit.
  \item \textbf{The form $0/0$} --- indeterminate, not undefined. It reports that the substitution failed, nothing more; factor, rationalize, or combine fractions and try again.
  \item \textbf{Nonzero over zero} --- the limit is infinite, and \emph{which} infinity depends on the sign the denominator carries as $x$ approaches from that side. Test the side, never assume it.
\end{itemize}

\noindent On every find-and-fix item, quote the line of the student's reasoning
that fails, then redo the problem from that line on. Write $\infty$ or
$-\infty$ where the limit is infinite, and write ``does not exist'' where the
two sides disagree --- those are different statements.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3cm]{Evaluate $\displaystyle\lim_{x \to 3}\, (x^2 - 4x + 7)$, and state
in one line why direct substitution is legitimate here.}

\problem[5cm]{Evaluate $\displaystyle\lim_{x \to 3}\, \frac{x^2 - 2x - 3}{x - 3}$.
Write down what direct substitution produces before you do anything else, then
show the algebra that repairs it.}

\problem[5.5cm]{Evaluate $\displaystyle\lim_{x \to 0}\, \frac{\sqrt{x + 4} - 2}{x}$.
Substitution gives an indeterminate form; multiply by the conjugate to remove
it. Give the exact value, not a decimal approximation.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student writes
\[ \lim_{x \to 2}\, \frac{1}{x - 2} = \infty \]
arguing that the denominator shrinks to zero, so the fraction grows without
bound. Evaluate the two one-sided limits separately, then state what the
two-sided limit is and why.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student writes
\[ \lim_{x \to 0}\, \frac{|x|}{x} = 1 \]
saying that $|x|$ and $x$ ``cancel''. Evaluate the limit from the left and from
the right, and state what the two-sided limit is.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student claims that
$\displaystyle\lim_{x \to 4^{+}} \frac{x - 7}{x - 4} = \infty$, because
``approaching from the right always gives positive infinity''. Determine both
one-sided limits at $x = 4$ and explain what the student's rule left out.}

\problem[6cm]{\textbf{Find and fix the mistake.} Let
\[ f(x) = \begin{cases} x^2 + k, & x < 3 \\ 5x - 1, & x \ge 3 \end{cases} \]
A student is asked which value of $k$ makes $f$ continuous at $x = 3$ and
answers $k = 14$, ``because $5(3) - 1 = 14$''. Write the continuity condition as
an equation, solve it for $k$, and give the common value of the two one-sided
limits.}

\problem[5cm]{\textbf{Explain.} Write a short paragraph for a classmate who
believes that substitution either gives the answer or shows the limit does not
exist. Say what the form $0/0$ actually reports and what you do next; say what
nonzero-over-zero reports; and explain why two infinite one-sided limits with
opposite signs mean the two-sided limit does not exist. \noansline}
