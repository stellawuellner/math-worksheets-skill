\problem{\textbf{Taxi fare table.}\\
\textbf{(a) Step 1.} The question asks for a value the table already holds, so
this is a read, not a computation: find the column headed $3$ miles.\\
\textbf{(a) Step 2.} That column shows a fare of $\ans{19}$ dollars.\\
\textbf{(b) Step 1.} Rate of change needs two rows of the table. Take the first
and last columns, $(1, 9)$ and $(4, 24)$, so the arithmetic stays clean.\\
\textbf{(b) Step 2.} $\dfrac{24 - 9}{4 - 1} = \dfrac{15}{3} = \ans{5}$ dollars
per mile.\\
\textit{Check:} every step across the table adds $5$, and $9 - 5 = 4$ is the
fixed flag-fall, so the rule is $\text{fare} = 5d + 4$.}

\problem{\textbf{Slope of Graph A.}\\
\textbf{Step 1.} Slope needs two points that sit exactly on grid intersections,
so read the $y$-intercept and one clean lattice point rather than estimating.\\
\textbf{Step 2.} The line passes through $(0, 3)$ and $(4, 11)$.\\
\textbf{Step 3.} $\dfrac{11 - 3}{4 - 0} = \dfrac{8}{4} = \ans{2}$}

\problem{\textbf{Gym membership.}\\
\textbf{Step 1.} The joining fee is paid once, so it is the initial value; the
monthly charge repeats, so it is the rate. That fixes the shape
$C = (\text{rate}) \cdot m + (\text{initial value})$.\\
\textbf{Step 2.} $C = 15m + 20$.\\
\textbf{Step 3.} Substitute $m = 8$: \; $15(8) + 20 = 120 + 20 = \ans{140}$
dollars.}

\problem{\textbf{Slope of Graph B.}\\
\textbf{Step 1.} The line falls from left to right, so expect a negative rate
of change --- that sign check catches the most common slip before it happens.\\
\textbf{Step 2.} The line passes through $(0, 12)$ and $(5, 2)$.\\
\textbf{Step 3.} $\dfrac{2 - 12}{5 - 0} = \dfrac{-10}{5} = \ans{-2}$}

\problem{\textbf{Graph C: solve $4x = 28$.}\\
\textbf{Step 1.} Here $y$ is known and $x$ is not, so the equation is used
\emph{backwards} --- solve rather than substitute.\\
\textbf{Step 2.} $4x = 28$, so $x = \dfrac{28}{4}$.\\
\textbf{Step 3.} $x = \ans{7}$}

\problem{\textbf{Draining tank.}\\
\textbf{(a) Step 1.} The minute asked for appears in the table, so read the
column headed $2$ minutes.\\
\textbf{(a) Step 2.} The tank holds $\ans{36}$ litres.\\
\textbf{(b) Step 1.} Water is leaving, so the rate of change must be negative;
use the first and last columns, $(0, 60)$ and $(3, 24)$.\\
\textbf{(b) Step 2.} $\dfrac{24 - 60}{3 - 0} = \dfrac{-36}{3} = \ans{-12}$
litres per minute.}

\problem{\textbf{Matching $y = 12 - 2x$ to a graph.}\\
\textbf{Step 1.} An equation matches a graph when a point computed from the
equation lies on that graph, so generate a point instead of eyeballing the
picture.\\
\textbf{Step 2.} Substitute $x = 5$: \; $12 - 2(5) = 12 - 10 = \ans{2}$.\\
\textbf{Step 3.} The equation therefore passes through $(5, 2)$. Graph B is the
only reference graph containing that point --- Graph A is rising and is at
$13$ when $x$ is $5$, and Graph C passes through the origin. So the equation
matches \textbf{Graph B}. Its $y$-intercept, $12$, and its falling direction
agree as well.}

\problem{\textbf{Filling tank.}\\
\textbf{Step 1.} Two (time, volume) pairs are given in words rather than in a
table, but they are still two points of the same linear relationship:
$(3, 26)$ and $(8, 51)$.\\
\textbf{Step 2.} $\dfrac{51 - 26}{8 - 3} = \dfrac{25}{5} = \ans{5}$ litres per
minute.\\
\textit{Check:} the tank rises, so a positive rate is expected.}

\problem{\textbf{Seedling height.}\\
\textbf{Step 1.} The present height is what you have before any weeks pass, so
it is the initial value; the weekly growth repeats, so it is the rate.\\
\textbf{Step 2.} $h = 3w + 12$.\\
\textbf{Step 3.} Substitute $w = 9$: \; $3(9) + 12 = 27 + 12 = 39$, so the
height is $\ans{39 \text{ cm}}$.}

\problem{\textbf{Table of a linear relationship.}\\
\textbf{(a) Step 1.} The value is in the table, so read the column headed
$x = 4$ rather than building an equation.\\
\textbf{(a) Step 2.} $y = \ans{19}$.\\
\textbf{(b) Step 1.} Rate of change compares a change in $y$ with the matching
change in $x$; the columns step by $2$, not by $1$, so dividing is essential
here.\\
\textbf{(b) Step 2.} Using the outer columns $(0, 7)$ and $(6, 25)$:
$\dfrac{25 - 7}{6 - 0} = \dfrac{18}{6} = \ans{3}$.}

\problem{\textbf{Comparing Graph A and Graph C.}\\
\textbf{Step 1.} Both grids are identical, so the steeper line is the one with
the larger rate of change; compute the unknown one and compare.\\
\textbf{Step 2.} Graph C runs through $(0, 0)$ and $(3, 12)$, so its slope is
$\dfrac{12 - 0}{3 - 0} = \ans{4}$.\\
\textbf{Step 3.} Graph A's slope is $2$, so \textbf{Graph C climbs faster}.
Without computing: on the same grid, Graph C rises four squares for every one
square across while Graph A rises two, and Graph C reaches the top of the grid
in half the horizontal distance.}

\problem{\textbf{Challenge: phone plan.}\\
\textbf{(a) Step 1.} The two bills are two points of one linear relationship,
$(2, 31)$ and $(5, 46)$, so the per-gigabyte charge is their rate of change.\\
\textbf{(a) Step 2.} $\dfrac{46 - 31}{5 - 2} = \dfrac{15}{3} = \ans{5}$ dollars
per gigabyte.\\
\textbf{(b) Step 1.} The fixed fee is the value at zero gigabytes. Work back
from $(2, 31)$: two gigabytes cost $2 \times 5 = 10$, so the fee is
$31 - 10 = 21$. The equation is $B = 5g + 21$.\\
\textbf{(b) Step 2.} A bill of $61$ dollars means $5g + 21 = 61$, so
$5g = 40$.\\
\textbf{(b) Step 3.} $g = \ans{8}$ gigabytes.\\
\textit{Check:} $5(8) + 21 = 61$.}
