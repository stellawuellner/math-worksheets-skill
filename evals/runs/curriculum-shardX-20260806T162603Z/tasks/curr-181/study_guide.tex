\sstitleblock{Writing Inequalities from Verbal Constraints}

% =================== SKILL 1 ===================
\skillheading{At Most and At Least: the Boundary Counts}

\begin{formulabox}
\textbf{Rule:} ``at most'', ``no more than'', ``maximum'', ``up to'' all
\emph{include} the limit, so they become $\le$. ``At least'', ``no fewer than'',
``minimum'' include it too, and become $\ge$.\par
\textbf{Test-value model:} try the limit itself. If that value is allowed, the
sign has the bar under it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A van may carry at most 900 kilograms, and each crate
weighs 60 kilograms. Write and solve an inequality for the number $c$ of crates.
\step{``At most'' allows exactly 900 kilograms, so the sign is $\le$ and the
total mass of the crates is $60c$: \ $60c \le 900$.}
\step{Divide both sides by 60, which is positive so the direction does not
change: $c \le 15$, that is $\ans{c \leq 15}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A test is passed with at least 45 points, and each
correct question is worth 5 points. Write and solve an inequality for the number
$q$ of correct questions.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{q \geq 9}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{More Than and Fewer Than: the Boundary is Out}

\begin{formulabox}
\textbf{Rule:} ``more than'', ``over'', ``above'' become $>$; ``fewer than'',
``less than'', ``under'', ``below'' become $<$. The limit itself is \emph{not}
allowed, so the solution stops just short of it.\par
\textbf{Test-value model:} try the limit. If it is not allowed, the sign has no
bar.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fewer than 30 cakes are packed into boxes of 5. Write
and solve an inequality for the number $x$ of boxes.
\step{$x$ boxes of 5 cakes is $5x$ cakes, and ``fewer than'' excludes 30 itself,
so the sign is strict: \ $5x < 30$.}
\step{Divide both sides by 5: $x < 6$, that is $\ans{x < 6}$ --- six boxes would
be exactly 30 cakes, which is one too many.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A hall has more than 48 seats, arranged in 8 equal rows.
Write and solve an inequality for the number $r$ of seats in one row.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{r > 6}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Limits with a Starting Amount}

\begin{formulabox}
\textbf{Rule:} when one amount is charged \emph{once} and another repeats, the
inequality has two terms: (rate) $\times$ (unknown) $+$ (one-time amount),
compared with the limit. The one-time amount is added, never multiplied by the
count. Undo the addition first, then the multiplication.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A phone plan costs 15 dollars a month plus 2 dollars for
each gigabyte, and the budget is at most 27 dollars. Write and solve an
inequality for the number $g$ of gigabytes.
\step{The 2 dollars repeats for each gigabyte and the 15 dollars is paid once,
and ``at most'' includes the limit: \ $2g + 15 \le 27$.}
\step{Subtract 15 from both sides to get $2g \le 12$, then divide by 2:
$g \le 6$, that is $\ans{g \leq 6}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A club charges 25 dollars to join plus 4 dollars a month,
and a member has at most 65 dollars. Write and solve an inequality for the
number $m$ of months.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m \leq 10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an inequality has a whole \emph{set} of solutions, and
the situation may allow only some of them. ``$n \le 15$ people'' really means
the whole numbers up to 15 --- the algebra cannot know that people come in whole
numbers.
\end{watchoutbox}
