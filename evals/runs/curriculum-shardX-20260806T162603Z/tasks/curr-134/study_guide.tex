% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Multiply Straight Across}

\begin{formulabox}
\textbf{Rule:} \quad
$\dfrac{a}{b} \times \dfrac{c}{d} = \dfrac{a \times c}{b \times d}$
\quad --- top times top, bottom times bottom.\par
You may cancel a \emph{top} number with a \emph{bottom} number before
multiplying. Never cancel top with top.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $\dfrac{3}{5} \times \dfrac{5}{9}$.
\step{This is multiplication, so no flipping --- multiply straight across,
cancelling a top with a bottom first when you can.}
\step{Cancel the $5$s, and $3$ with $9$: \quad
$\dfrac{3}{5} \times \dfrac{5}{9} = \dfrac{3 \times 5}{5 \times 9}
= \dfrac{15}{45}$}

$\Rightarrow\quad \dfrac{3}{5} \times \dfrac{5}{9} = \ans{\tfrac{1}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $\dfrac{2}{7} \times \dfrac{7}{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{4}}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Divide: Flip the Second Fraction}

\begin{formulabox}
\textbf{Rule (keep--change--flip):} \quad
$\dfrac{a}{b} \div \dfrac{c}{d} = \dfrac{a}{b} \times \dfrac{d}{c}$\par
\emph{Keep} the first fraction, \emph{change} $\div$ to $\times$, \emph{flip}
only the \emph{divisor} (the second fraction).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $\dfrac{4}{5} \div \dfrac{2}{3}$.
\step{Division by a fraction means multiply by its reciprocal --- keep the
first fraction exactly as it is.}
\step{Flip only the second: \quad
$\dfrac{4}{5} \times \dfrac{3}{2} = \dfrac{12}{10}$, then simplify.}

$\Rightarrow\quad \dfrac{4}{5} \div \dfrac{2}{3} = \ans{\tfrac{6}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $\dfrac{3}{8} \div \dfrac{3}{4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{2}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Spot the Mistake, Then Fix It}

\begin{formulabox}
\textbf{Error checklist} --- when work looks wrong, ask:
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item Was the \emph{second} fraction (the divisor) the one that got flipped?
  \item In a multiply, did the work go \emph{straight across} (not diagonally)?
  \item Does the answer make sense? Dividing by a fraction smaller than 1
        should make the answer \emph{bigger}.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Leo says $\dfrac{1}{2} \div \dfrac{1}{6} =
\dfrac{1}{12}$. Find the mistake and the correct answer.
\step{Check the answer against sense first: dividing by $\tfrac{1}{6}$ asks how
many sixths fit in one half, so the answer must be bigger than 1 --- Leo
multiplied instead of flipping the divisor.}
\step{Fix it: \quad $\dfrac{1}{2} \times \dfrac{6}{1} = \dfrac{6}{2}$}

$\Rightarrow\quad \dfrac{1}{2} \div \dfrac{1}{6} = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Maya says $\dfrac{3}{4} \div \dfrac{1}{8} =
\dfrac{3}{32}$. Find her mistake, then compute the correct quotient.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Flip and multiply'' does \emph{not} mean flip
whichever fraction you like --- only the fraction \emph{after} the $\div$ sign
gets flipped, and only in a \emph{division} problem.
\end{watchoutbox}
