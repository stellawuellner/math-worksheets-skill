\sstitleblock{Graphing Basic Polar Curves from Equations}

\skillheading{Skill 1 --- Plotting one point of a polar curve}

\begin{formulabox}
\textbf{Rule:} a polar equation is a recipe. Feed it an angle $\theta$, it returns a
directed distance $r$. Plot the point by facing along the ray $\theta$ and walking
$r$ units --- forwards if $r > 0$, backwards through the pole if $r < 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $r$ on $r = 2 + 3\sin\theta$ at $\theta = \dfrac{\pi}{6}$.
\step{One angle in, one distance out --- substitute and evaluate, no graphing needed yet.}
\step{$\sin\dfrac{\pi}{6} = \dfrac{1}{2}$, \ so $r = 2 + 3\left(\dfrac{1}{2}\right)$.}

$\Rightarrow\quad r = \ans{\tfrac{7}{2}}$, so the point sits $3.5$ units out along the $\pi/6$ ray.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $r$ on $r = 5 + \cos\theta$ at $\theta = \dfrac{\pi}{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{\tfrac{11}{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Circles and lines in polar form}

\begin{formulabox}
\textbf{The three shapes to recognise on sight:}
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=1pt]
  \item $r = k$ --- circle of radius $k$ centred at the pole.
  \item $\theta = k$ --- straight line through the pole at angle $k$.
  \item $r = a\cos\theta$ --- circle of diameter $a$ resting on the polar axis;
        $r = a\sin\theta$ --- the same circle rotated onto the vertical axis.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Locate a point of the circle $r = 4\sin\theta$ at $\theta = \dfrac{\pi}{6}$.
\step{The shape is already known to be a circle, so one substituted point is enough to pin its position.}
\step{$\sin\dfrac{\pi}{6} = \dfrac{1}{2}$, \ so $r = 4\left(\dfrac{1}{2}\right)$.}

$\Rightarrow\quad r = \ans{2}$ at $\theta = \pi/6$; the circle's far point is at $\theta = \pi/2$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $r$ on the circle $r = 8\cos\theta$ at $\theta = \dfrac{\pi}{3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Limacons and cardioids}

\begin{formulabox}
\textbf{Form:} $r = a \pm b\cos\theta$ or $r = a \pm b\sin\theta$. Compare $a$ with $b$:
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=1pt]
  \item $a = b$ --- cardioid (heart), reaches the pole exactly once.
  \item $a > b$ --- dimpled or convex limacon, never reaches the pole.
  \item $a < b$ --- limacon with an inner loop, traced where $r < 0$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the largest $r$ on $r = 3 + 2\cos\theta$.
\step{$r$ is largest where the cosine is largest, and $\cos\theta$ peaks at $1$ when $\theta = 0$.}
\step{$r = 3 + 2(1)$ at $\theta = 0$.}

$\Rightarrow\quad r_{\max} = \ans{5}$, so the curve stretches farthest along the polar axis.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $r$ on $r = 4 - 3\sin\theta$ at $\theta = \dfrac{\pi}{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{1}$}
\end{tryitbox}

\skillheading{Skill 4 --- Rose curves}

\begin{formulabox}
\textbf{Form:} $r = a\cos(n\theta)$ or $r = a\sin(n\theta)$. Each petal has length $|a|$;
the petal count is $n$ when $n$ is odd and $2n$ when $n$ is even. The curve returns to the
pole wherever the sine or cosine is zero, and those angles separate the petals.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the tip length of a petal of $r = 4\sin(2\theta)$.
\step{A petal tip is where $|r|$ is largest, so look for the angle that makes the sine equal $1$.}
\step{$2\theta = \dfrac{\pi}{2}$ gives $\theta = \dfrac{\pi}{4}$, and $r = 4\sin\dfrac{\pi}{2}$.}

$\Rightarrow\quad r = \ans{4}$, and since $n = 2$ is even this rose has four petals of that length.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $r$ on $r = 6\cos(3\theta)$ at $\theta = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a negative $r$ is not an error. Plot it backwards through the pole ---
that is exactly how an inner loop and half the petals of an even rose get drawn.
\end{watchoutbox}
