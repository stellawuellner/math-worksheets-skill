\sstitleblock{Unit Rates from Fractions and Mixed Numbers}

\skillheading{Skill 1 --- A rate when both quantities are fractions}

\begin{formulabox}
\textbf{Rule:} A unit rate is one quantity per \emph{one} of the other, so it is always a division.
\[ \text{rate} = \frac{\text{amount}}{\text{time}} = \frac{a}{b} \div \frac{c}{d} = \frac{a}{b} \times \frac{d}{c} \]
Dividing by a fraction smaller than 1 makes the answer \emph{larger} --- that is the built-in check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tap delivers $\frac{2}{3}$ of a litre in $\frac{1}{6}$ of a minute. Litres per minute?
\step{Per minute means litres divided by minutes, so I divide the amount by the time and turn the division into multiplication by the reciprocal.}
\step{$\frac{2}{3} \div \frac{1}{6} = \frac{2}{3} \times \frac{6}{1} = \frac{12}{3}$.}
Rate: $\ans{4}$ litres per minute
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tap delivers $\frac{3}{5}$ of a litre in $\frac{1}{4}$ of a minute. Litres per minute?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{12}{5}}$ litres per minute}
\end{tryitbox}

\skillheading{Skill 2 --- Rates that start with a mixed number}

\begin{formulabox}
\textbf{Rule:} Convert every mixed number to an improper fraction \emph{before} dividing.
\[ w\frac{p}{q} = \frac{w \times q + p}{q} \]
Dividing a mixed number part by part gives a wrong answer, because the whole and the fraction do not divide separately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A conveyor moves $1\frac{3}{4}$ metres in $\frac{1}{2}$ of a minute. Metres per minute?
\step{The mixed number cannot be divided as it stands, so I rewrite it as an improper fraction first and only then divide.}
\step{$1\frac{3}{4} = \frac{7}{4}$, and $\frac{7}{4} \div \frac{1}{2} = \frac{7}{4} \times \frac{2}{1} = \frac{14}{4}$.}
Rate: $\ans{\tfrac{7}{2}}$ metres per minute
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A conveyor moves $4\frac{1}{2}$ metres in $\frac{3}{4}$ of a minute. Metres per minute?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$ metres per minute}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two rates fairly}

\begin{formulabox}
\textbf{Rule:} Two deals can only be compared once both are expressed \emph{per one} of the same thing.\\[3pt]
A bigger total, a smaller price, or a shorter time proves nothing on its own. Work out both unit rates, put them over a common denominator, and then compare the numerators.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which is the greater rate, $\frac{9}{4}$ per hour or $\frac{7}{3}$ per hour?
\step{Both are already unit rates, so the only remaining job is to compare two fractions --- I rewrite them over a common denominator.}
\step{Over 12: $\frac{9}{4} = \frac{27}{12}$ and $\frac{7}{3} = \frac{28}{12}$, so 27 twelfths against 28 twelfths.}
$\dfrac{9}{4} \ans{<} \dfrac{7}{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which is the greater rate, $\frac{8}{3}$ per hour or $\frac{5}{2}$ per hour?\\[2pt]
\rotatebox{180}{\footnotesize check: $\dfrac{8}{3} \ans{>} \dfrac{5}{2}$}
\end{tryitbox}

\skillheading{Skill 4 --- Turning the rate the other way up}

\begin{formulabox}
\textbf{Rule:} ``How long for one whole thing'' is the \emph{reciprocal} question, so divide the time by the fraction completed.
\[ \text{time for one whole} = \frac{\text{time taken}}{\text{fraction completed}} \]
Check the direction: finishing more than was done must take longer than the time already spent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A machine completes $\frac{3}{4}$ of a job in $\frac{1}{2}$ of an hour. How long for the whole job?
\step{The question wants hours per job rather than jobs per hour, so I divide the time by the fraction of the job that is done.}
\step{$\frac{1}{2} \div \frac{3}{4} = \frac{1}{2} \times \frac{4}{3} = \frac{4}{6}$, and $\frac{2}{3} > \frac{1}{2}$ as it must be.}
Whole job: $\ans{\tfrac{2}{3}}$ of an hour
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A machine completes $\frac{2}{5}$ of a job in $\frac{1}{3}$ of an hour. How long for the whole job?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{6}}$ of an hour}
\end{tryitbox}
