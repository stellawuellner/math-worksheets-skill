\sstitleblock{Multistep Spatial and Coordinate Problems}

\skillheading{Skill 1 --- A missing side of a right triangle}

\begin{formulabox}
\textbf{The theorem:} \quad $a^2 + b^2 = c^2$, with $c$ the hypotenuse (the longest side, opposite
the right angle).\\[3pt]
Missing the hypotenuse: add, $c = \sqrt{a^2 + b^2}$ --- e.g. legs $3$ and $4$ give $\sqrt{25} = 5$.\\[2pt]
Missing a leg: subtract, $a = \sqrt{c^2 - b^2}$. Which one you need is decided by whether the
longest side is already known.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Hypotenuse $26$, one leg $10$. Find the other leg.
\step{The longest side is given, so the missing side is a leg --- subtract the squares.}
\step{$\sqrt{26^2 - 10^2} = \sqrt{676 - 100} = \sqrt{576}$}
$\Rightarrow\quad \ans{24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Hypotenuse $37$, one leg $12$. Find the other leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{35}$}
\end{tryitbox}

\skillheading{Skill 2 --- Distance between two points on a grid}

\begin{formulabox}
\textbf{Rule:} the horizontal gap and the vertical gap are the two legs, and the segment joining
the points is the hypotenuse. So a distance is always found by \textit{adding}:
\[ d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2} \]
Squaring removes any minus sign, so it does not matter which point you subtract first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the distance from $(2, 1)$ to $(10, 7)$.
\step{Two points and a length wanted, so build the right triangle from the two gaps and add.}
\step{Gaps: $10 - 2 = 8$ and $7 - 1 = 6$, so $d = \sqrt{64 + 36} = \sqrt{100}$.}
$\Rightarrow\quad d = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the distance from $(1, -2)$ to $(13, 3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 3 --- The diagonal of a box, in two stages}

\begin{formulabox}
\textbf{Stage 1:} the diagonal across the base, $d = \sqrt{\ell^2 + w^2}$.\\[2pt]
\textbf{Stage 2:} treat that answer as a leg and the height as the other leg,
$D = \sqrt{d^2 + h^2}$.\\[3pt]
Doing both stages at once gives $D = \sqrt{\ell^2 + w^2 + h^2}$ --- the same thing, with the
first square root never written down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A box has a base $6$ by $8$ and a height of $24$. Find both diagonals.
\step{A box needs two stages: flat across the base first, then upwards using that as a leg.}
\step{$d = \sqrt{6^2 + 8^2} = \sqrt{100}$, then $D = \sqrt{10^2 + 24^2} = \sqrt{676}$.}
$\Rightarrow\quad d = \ans{10} \quad \text{and} \quad D = \ans{26}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A box has a base $4$ by $3$ and a height of $12$. Find the diagonal of the
box.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 4 --- Multistep journeys and outlines}

\begin{formulabox}
\textbf{Method:} find the missing side first, then do the ordinary arithmetic the question really
asks for --- a total, a difference, a perimeter.\\[3pt]
Going along two sides of a rectangle costs $\ell + w$; cutting the diagonal costs
$\sqrt{\ell^2 + w^2}$, which is always less. The saving is the difference of the two.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A field is $9$ m by $12$ m. How much shorter is the diagonal than walking
two sides?
\step{The question is a difference, so find the diagonal first and then subtract it from the two-side route.}
\step{Diagonal $= \sqrt{9^2 + 12^2} = \sqrt{225} = 15$; two sides cost $9 + 12 = 21$; the saving is $21 - 15$.}
$\Rightarrow\quad \text{diagonal} = \ans{15} \text{ m},\ \text{saving} = \ans{6} \text{ m}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A field is $5$ m by $12$ m. How many metres does the diagonal save?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an answer that comes out \textit{longer} than the hypotenuse you started
with means the squares were added when they should have been subtracted.
\end{watchoutbox}
