\begin{center}\small
\fbox{\begin{minipage}{0.93\linewidth}
\textbf{Fitting a model to two data points, interpreting its parameters, solving for time, and rewriting the base}\\[3pt]
Every model on this sheet has the form $y = a \cdot b^{\,t}$, where $a$ is the value at
$t = 0$ and $b$ is the factor the quantity is multiplied by each time step.
A growth rate $r$ per step means $b = 1 + r$; a decay rate means $b = 1 - r$.
To fit from two points, divide the two equations so $a$ cancels; to solve for $t$, take
logarithms of both sides; to change the base, use $b^{\,t} = e^{(\ln b)t}$.
\end{minipage}}
\end{center}

\noindent\small\textbf{Directions:} Units and time origins are stated in each problem.
Give exact values where they are exact, and round decimals to the precision the problem
asks for. Show the equation you solved, not just the answer.\normalsize

\problem[5.5cm]{A streaming service models its subscribers by
$S(t) = 320 \cdot (1.25)^{\,t}$ thousand subscribers, where $t$ is the number of years
since launch. State the number of subscribers at launch, and state the growth rate per
year as a decimal.}

\problem[6.5cm]{A bacteria culture is modelled by $N(t) = a \cdot b^{\,t}$ cells, with $t$
in hours. The culture holds $240$ cells at $t = 0$ hours and $810$ cells at $t = 3$ hours.
Find the hourly growth factor $b$. (Divide the second equation by the first so that $a$
cancels.)}

\problem[6.5cm]{The same culture is modelled by $N(t) = 240 \cdot (1.5)^{\,t}$ cells.
At what time $t$, in hours, does the culture reach $1215$ cells?}

\problem[6.5cm]{Rewrite the culture model $N(t) = 240 \cdot (1.5)^{\,t}$ in the continuous
form $N(t) = 240 \cdot e^{k t}$. Give the exact value of $k$ as a logarithm and then as a
decimal rounded to three places.}

\problem{A dose of medicine in a patient's bloodstream is modelled by
$A(t) = 80 \cdot (0.85)^{\,t}$ milligrams, with $t$ in hours.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item What decimal fraction of the medicine is lost each hour? \ansline
  \item How much medicine remains after $4$ hours, in milligrams, to two decimal places? \ansline
\end{enumerate}}

\problem{Algae coverage on a pond is modelled by $C(t) = a \cdot b^{\,t}$ square metres,
with $t$ in weeks. Two measurements were recorded:

\begin{center}
\begin{tabular}{|c|c|}\hline
\textbf{Week $t$} & \textbf{Coverage $C(t)$} \\\hline
$1$ & $36$ \\\hline
$4$ & $121.5$ \\\hline
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Find the weekly growth factor $b$. \ansline
  \item Find $a$, the coverage the model gives at $t = 0$. \ansline
\end{enumerate}}

\problem[6.5cm]{Use the streaming model $S(t) = 320 \cdot (1.25)^{\,t}$ again.
How long does the subscriber count take to double? Give the number of years to two
decimal places. (Notice that the answer does not depend on $320$.)}

\problem{The streaming company wants to report growth per \emph{month} instead of per year.
Because $12$ months make a year, the monthly factor $b$ satisfies $b^{12} = 1.25$.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Find the monthly growth factor to four decimal places. \ansline
  \item State the monthly growth rate as a decimal to four places. \ansline
\end{enumerate}}

\problem{\textbf{Find and fix the mistake.} A town's population is modelled by
$P(t) = 500 \cdot (1.08)^{\,t}$ people, with $t$ in years. A student reads the model and
says: ``the town gains $8$ people every year''.
\begin{enumerate}[label=(\alph*), itemsep=4cm]
  \item Compute the increase in population from $t = 0$ to $t = 1$. \ansline
  \item Compute the increase from $t = 5$ to $t = 6$, to two decimal places.
        Your two answers show why the student's reading is wrong. \ansline
\end{enumerate}}

\problem{A chemical sample decays exponentially: it measures $12$ milligrams at
$t = 0$ hours and $3$ milligrams at $t = 6$ hours.
\begin{enumerate}[label=(\alph*), itemsep=4.5cm]
  \item Fit the model $M(t) = a \cdot b^{\,t}$: find the hourly decay factor $b$ to four
        decimal places. \ansline
  \item Using that model, at what time does the sample fall to $1$ milligram?
        Give the answer to two decimal places. \answerline{hours}
\end{enumerate}}
