\wstitleblock{The Zero That Holds a Place}{Grade 2 Math}{}

\noindent\textbf{Directions:}
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item A zero inside a number holds a place open: 507 has 5 hundreds,
        \emph{zero} tens, and 7 ones.
  \item Some problems show work with a mistake in it. Find the mistake,
        fix it, and be ready to say what went wrong.
\end{itemize}
