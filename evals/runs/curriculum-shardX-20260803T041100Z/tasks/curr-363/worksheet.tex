\begin{center}\small\textit{Factoring to find zeros, reading multiplicity as
crossing or touching, describing end behavior, and building a polynomial from
its zeros}\end{center}

\noindent\textbf{What multiplicity looks like on a graph.} A factor's exponent
tells you the shape of the curve where it meets the $x$-axis: an odd exponent
passes through, an even exponent bounces off, and a higher odd exponent
flattens as it passes.

\begin{center}
\begin{tikzpicture}[scale=0.62]
  \begin{scope}
    \draw[gray, ->] (-1.7,0) -- (1.7,0);
    \draw[thick] plot[domain=-1.4:1.4, samples=40] (\x, {1.1*\x});
    \node[below, font=\footnotesize] at (0,-2.0) {odd exponent: crosses};

    \draw[thick] plot[domain=-1.35:1.35, samples=60] (\x, {0.6*\x*\x*\x});

\noindent\textbf{Directions.}
\begin{itemize}[itemsep=2pt]
  \item Factor completely before reading anything off a polynomial --- the
        factored form is what shows the zeros and their multiplicities.
  \item When a problem asks for behavior, name the zero, its multiplicity, and
        whether the curve crosses or touches.
  \item End behavior is set by the degree (odd or even) and the sign of the
        leading coefficient.
\end{itemize}

\problem[6cm]{Factor $f(x) = x^3 - 4x$ completely, then list every real zero of
$f$. Write the factored form on the answer line.}

\problem[6cm]{A polynomial is given in factored form:
\[ g(x) = (x+1)(x-3)^2 \]
List each real zero with its multiplicity, and state whether the graph of $g$
crosses or touches the $x$-axis at each one. Write the zeros on the answer line.}

\problem[6cm]{Factor $h(x) = x^3 - 3x^2 - 10x$ completely, then list every real
zero of $h$. Write the factored form on the answer line.}

\problem[6cm]{Describe the end behavior of $f(x) = -2x^3 + 5x - 1$ by completing
both statements, and say in one sentence which two features of the polynomial
decided them.
\[ \text{as } x \to \infty,\ f(x) \to \underline{\hspace{1.8cm}}
   \qquad\qquad
   \text{as } x \to -\infty,\ f(x) \to \underline{\hspace{1.8cm}} \]}

\problem[6cm]{A polynomial is given in factored form:
\[ p(x) = x^2(x-4)^3 \]
State the degree of $p$, list each real zero with its multiplicity, and say
whether the graph crosses, touches, or flattens as it passes at each zero.
Write the zeros on the answer line.}

\problem[6cm]{An engineer needs a curve that meets the $x$-axis exactly at
$x = -2$ and at $x = 1$, passing straight through at $x = -2$ but bouncing off
at $x = 1$. Write the polynomial of least degree with leading coefficient $1$
that does this, in expanded form.}

\problem[7cm]{Solve $x^4 - 5x^2 + 4 = 0$ by factoring. (Hint: the left side is a
quadratic in $x^2$.) List all four real zeros on the answer line.}

\problem[6cm]{Describe the end behavior of $g(x) = 3x^4 - x^3$ by completing
both statements, and say in one sentence why both ends behave the same way here
but not in problem 4.
\[ \text{as } x \to \infty,\ g(x) \to \underline{\hspace{1.8cm}}
   \qquad\qquad
   \text{as } x \to -\infty,\ g(x) \to \underline{\hspace{1.8cm}} \]}

\problem[7cm]{A designer sketches a profile curve that crosses the $x$-axis at
$x = -3$ and just touches it at $x = 2$, with no other $x$-intercepts. Write the
polynomial of least degree with leading coefficient $1$ that matches the sketch,
in expanded form.}

\problem[11cm]{Sketch the graph of $y = x(x+2)^2(x-3)$ on the grid below. Mark
every $x$-intercept, label each one as crossing or touching, and make both ends
of your sketch match the end behavior. Each square is one unit wide; choose your
own vertical scale and label it. Write the zeros on the answer line.
