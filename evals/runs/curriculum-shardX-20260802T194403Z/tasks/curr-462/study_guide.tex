\sstitleblock{Derivatives of Inverse and Inverse-Trigonometric Functions}

\noindent\small Every rule below comes from one idea: if $y$ is the inverse of something you can
already differentiate, implicit differentiation turns the known derivative upside down. That is why
each arc-formula is a reciprocal, and why $(f^{-1})'$ is a reciprocal too.\normalsize

\skillheading{1. Arc-functions with the chain rule}

\begin{formulabox}
\textbf{With inner function $u = u(x)$:}\\[4pt]
$\dfrac{d}{dx}\arcsin u = \dfrac{u'}{\sqrt{1 - u^{2}}}$ \qquad
$\dfrac{d}{dx}\arccos u = \dfrac{-u'}{\sqrt{1 - u^{2}}}$ \qquad
$\dfrac{d}{dx}\arctan u = \dfrac{u'}{1 + u^{2}}$\\[4pt]
The $\arccos$ formula is the $\arcsin$ formula with a minus sign; the $\arctan$ denominator has
\emph{no} radical. Substitute $u$ into the denominator \emph{before} multiplying by $u'$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = \arcsin(4x)$
\step{This is an arc-function of an inner function, so use the $\arcsin$ formula with $u = 4x$ and
carry the chain-rule factor $u'$.}
\step{$u' = 4$ and $u^{2} = 16x^{2}$, so $y' = \dfrac{4}{\sqrt{1 - 16x^{2}}}$.}
$\Rightarrow\quad y' = \ans{\dfrac{4}{\sqrt{1 - 16x^{2}}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = \arctan(5x)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{5}{\,1 + 25x^{2}}}$}
\end{tryitbox}

\skillheading{2. The inverse-function derivative}

\begin{formulabox}
\textbf{Rule:} if $f$ is one-to-one and $f(b) = a$, then
$\left(f^{-1}\right)'(a) = \dfrac{1}{f'(b)}$.\\[3pt]
\textbf{Order of work:} (1) find the input $b$ that $f$ sends to $a$ --- usually by inspection or a
quick solve, (2) differentiate $f$, (3) evaluate $f'$ \textbf{at $b$}, (4) take the reciprocal.
Evaluating $f'$ at $a$ instead of $b$ is the classic lost mark.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{3} + x + 2$. Find $\left(f^{-1}\right)'(4)$.
\step{The question asks for the inverse's slope at an output value, so find the matching input
first and then reciprocate $f'$ there.}
\step{$f(1) = 4$, so $b = 1$; $f'(x) = 3x^{2} + 1$, so $f'(1) = 4$.}
$\Rightarrow\quad \left(f^{-1}\right)'(4) = \ans{\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{3} + 2x$, and $f(1) = 3$. Find $\left(f^{-1}\right)'(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{5}}$}
\end{tryitbox}

\skillheading{3. Products with an arc-function}

\begin{formulabox}
\textbf{Product rule:} $(fg)' = f'g + fg'$, with the arc-derivative used for whichever factor is
the arc-function.\\[3pt]
Differentiate the arc-factor \emph{in place} --- do not try to simplify $\arctan x$ away first. The
answer usually keeps an arc-function term \emph{and} a rational term; that is normal, not
unfinished.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Differentiate $y = x^{2}\arctan(x)$
\step{Two functions are multiplied and neither is a constant, so this is the product rule with the
$\arctan$ derivative supplying $g'$.}
\step{$f = x^{2}$, $f' = 2x$; $g = \arctan x$, $g' = \dfrac{1}{1 + x^{2}}$.}
$\Rightarrow\quad y' = \ans{2x\arctan(x) + \dfrac{x^{2}}{1 + x^{2}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Differentiate $y = x\arcsin(x)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\arcsin(x) + \dfrac{x}{\sqrt{1 - x^{2}}}}$}
\end{tryitbox}

\skillheading{4. Second derivatives of arc-functions}

\begin{formulabox}
\textbf{Method:} write the first derivative as a \textbf{power}, then use the chain rule again:\\[3pt]
$\dfrac{1}{1 + x^{2}} = (1 + x^{2})^{-1}$ \qquad
$\dfrac{1}{\sqrt{1 - x^{2}}} = (1 - x^{2})^{-1/2}$\\[3pt]
Negative and fractional exponents make the second differentiation ordinary power-and-chain work ---
no quotient rule needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $y''$ for $y = \arcsin(x)$
\step{The first derivative is a radical in the denominator, so rewrite it as a power before
differentiating a second time.}
\step{$y' = (1 - x^{2})^{-1/2}$, so
$y'' = -\tfrac{1}{2}(1 - x^{2})^{-3/2}(-2x) = x(1 - x^{2})^{-3/2}$.}
$\Rightarrow\quad y'' = \ans{\dfrac{x}{(1 - x^{2})\sqrt{1 - x^{2}}}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $y''$ for $y = \arccos(x)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{-x}{(1 - x^{2})\sqrt{1 - x^{2}}}}$}
\end{tryitbox}
