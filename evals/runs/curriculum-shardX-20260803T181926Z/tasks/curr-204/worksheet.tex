\wstitleblock{Diagnosing Invalid Simplification of Sums of Radicals}{Grade 8 / Pre-Algebra --- Irrational numbers and roots}{}

\noindent\textbf{The one rule that does not exist:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $\sqrt{a} + \sqrt{b}$ is \emph{not} $\sqrt{a + b}$. Roots distribute over multiplication, never over addition.
  \item Two radical terms combine only when their radicands are identical \emph{after} each one is simplified: $\sqrt{8} + \sqrt{18}$ looks unlike, but both hide $\sqrt{2}$.
  \item When they do combine, you add the coefficients and the radicand is untouched: $2\sqrt{2} + 3\sqrt{2} = 5\sqrt{2}$, not $5\sqrt{4}$.
\end{itemize}

\noindent Give exact answers in simplest radical form. Where a decimal is
asked for, round to two decimal places --- a decimal is how you \emph{test} a
claim, not how you state the answer.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3.5cm]{Write $\sqrt{72}$ in simplest radical form.}

\problem[4cm]{Simplify $5\sqrt{3} + 2\sqrt{3} - \sqrt{3}$.}

\problem[5cm]{Simplify $\sqrt{8} + \sqrt{18}$ in simplest radical form. Then
give its decimal value to two places, and say in one line why the answer is not
$\sqrt{26}$.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student writes
\[ \sqrt{9} + \sqrt{16} = \sqrt{25} = 5 \]
Compute the left side correctly, state which of the two values is greater, and
name the rule the student invented.}

\problem[5cm]{\textbf{Find and fix the mistake.} A student simplifies
$4\sqrt{5} + 3\sqrt{2}$ to $7\sqrt{7}$. Explain what was added that should not
have been, decide whether $4\sqrt{5} + 3\sqrt{2}$ can be simplified at all, and
give its decimal value to two places.}

\problem[5.5cm]{Simplify $\sqrt{12} + \sqrt{27} - \sqrt{3}$ in simplest radical
form, and give its decimal value to two places. Show the simplified form of each
radical before you combine anything.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} A student claims
$\sqrt{2} + \sqrt{8} = \sqrt{10}$. Give the decimal value of $\sqrt{2} +
\sqrt{8}$ to two places, write the correct comparison symbol between
$\sqrt{2} + \sqrt{8}$ and $\sqrt{10}$, and write $\sqrt{2} + \sqrt{8}$ in
simplest radical form.}

\problem[5cm]{\textbf{Explain.} Write a short paragraph for a classmate who
believes that square roots may be added the way fractions with the same
denominator are. Say why $\sqrt{a} + \sqrt{b}$ is not $\sqrt{a + b}$, state
exactly when two radical terms may be combined, and give one numerical
counterexample with the arithmetic shown. \noansline}
