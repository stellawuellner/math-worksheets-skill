\sstitleblock{Transformations, Compositions and Inverses}

\skillheading{Skill 1 --- Transforming a graph}

\begin{formulabox}
\textbf{Master form:} $g(x) = a\,f(x - h) + k$. Inside changes ($h$) move the graph horizontally and act \emph{opposite} to their sign; outside changes ($a$, $k$) act on the output as written, with the stretch applied before the shift.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2}$ and $g(x) = -2f(x - 1) + 3$. Find $g(4)$.
\step{Feed the shifted input to $f$ first, then apply the outside factor before adding --- the $3$ must not be multiplied.}
\step{$g(4) = -2f(3) + 3 = -2(9) + 3$.}

$\Rightarrow\quad \ans{g(4) = -15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2}$ and $g(x) = 3f(x + 2) - 1$. Find $g(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g(1) = 26}$}
\end{tryitbox}

\skillheading{Skill 2 --- Composing two functions}

\begin{formulabox}
\textbf{Definition.} $(f \circ g)(x) = f\bigl(g(x)\bigr)$: replace every $x$ in the rule for $f$ by the whole expression $g(x)$.\par
If a composition simplifies to $x$, the two functions are inverses --- that is the test, not a coincidence.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2} + 2$, $g(x) = 3x - 1$. Write $(f \circ g)(x)$.
\step{The inner rule goes in wherever $x$ appears in $f$, so square the whole binomial rather than its parts.}
\step{$(3x-1)^{2} + 2 = 9x^{2} - 6x + 1 + 2$.}

$\Rightarrow\quad \ans{(f \circ g)(x) = 9x^{2} - 6x + 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2} - 3$, $g(x) = 2x + 1$. Write $(f \circ g)(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(f \circ g)(x) = 4x^{2} + 4x - 2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{x^{2}}$ is $|x|$, not $x$. A square root undoes a square only on non-negative inputs, which is why so many of these problems come with a restriction attached.
\end{watchoutbox}

\skillheading{Skill 3 --- Finding and using an inverse}

\begin{formulabox}
\textbf{Method.} Write $y = f(x)$, solve for $x$, and rename. \par
\textbf{Reversal rule.} $(g \circ f)^{-1} = f^{-1} \circ g^{-1}$: the last operation applied is the first one undone. A restricted domain decides the sign of any square root the inverse needs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 4x + 9$. Find $f^{-1}(29)$.
\step{The rule multiplies then adds, so the inverse subtracts then divides --- reverse order, opposite operations.}
\step{$f^{-1}(x) = \dfrac{x - 9}{4}$, so $f^{-1}(29) = \dfrac{20}{4}$.}

$\Rightarrow\quad \ans{f^{-1}(29) = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = \dfrac{x + 6}{3}$. Find $f^{-1}(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f^{-1}(5) = 9}$}
\end{tryitbox}
