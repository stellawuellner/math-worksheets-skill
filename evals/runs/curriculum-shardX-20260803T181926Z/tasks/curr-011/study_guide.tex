\sstitleblock{Comparing Groups of Objects: More, Fewer, and the Same}

\skillheading{Skill 1 --- Which Group Has More, Fewer, or the Same?}

\begin{formulabox}
\textbf{The model:} put the two groups in two rows, one object under one object.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item The row that \textbf{runs out first} has \textbf{fewer}.
  \item The row with \textbf{leftovers} has \textbf{more}.
  \item If \textbf{nothing} is left over in either row, the groups are \textbf{the same}.
\end{itemize}
\textbf{The notation:} the symbol opens its mouth toward the bigger number.
$8 > 5$ says ``8 is more than 5''; \ $5 < 8$ says ``5 is fewer than 8''; \
$4 = 4$ says ``4 is the same as 4''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad One row has 8 buttons. The other row has 5 beads.
\step{The question is which group is bigger, so match the rows one to one instead of guessing by how long the rows look.}
\step{Five buttons find a bead. Three buttons are left over, so the buttons are more, and the comparison is written $\ans{8 > 5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A shelf holds 4 books. A box holds 7 blocks.
Write $>$, $<$, or $=$ between 4 and 7.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4 < 7}$}
\end{tryitbox}

\skillheading{Skill 2 --- How Many More? How Many Fewer?}

\begin{formulabox}
\textbf{Rule:} ``how many more'' and ``how many fewer'' ask for the \emph{leftover}
part --- the objects with no partner --- not for the whole row.\\[3pt]
\textbf{The notation:} bigger number $-$ smaller number $=$ how many more.
The same subtraction answers both questions, because ``Ana has 3 more'' and
``Ben has 3 fewer'' describe the same picture.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad There are 8 cars and 3 drivers. How many more cars than drivers?
\step{The question asks for the extra cars, so I match a driver to a car and look only at the cars left without one.}
\step{Three cars get a driver. Counting the cars still empty gives 1, 2, 3, 4, 5, and the sentence $8 - 3 = \ans{5}$ records it.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A tray holds 10 cookies and 6 napkins.
How many more cookies than napkins?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Making Two Groups the Same}

\begin{formulabox}
\textbf{Rule:} to make the smaller group match the bigger one, add exactly as many
objects as there are leftovers in the bigger row.\\[3pt]
\textbf{The notation:} smaller $+\ \underline{\hspace{0.7cm}}\ =$ bigger.
Count \textbf{up} from the smaller number to the bigger one; every count you say
is one object you still have to add. Then check by adding.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Rosa has 3 stickers. Theo has 7. How many more does Rosa need?
\step{Rosa's row is shorter, so the missing objects are the ones under Theo's leftovers --- that makes this an add-to-match question, not a take-away one.}
\step{Count up from 3 to 7: ``4, 5, 6, 7'' is 4 counts. Check: $3 + 4 = 7$ $\checkmark$, so Rosa needs $\ans{4}$ more.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Mia has 2 marbles. Kai has 8. How many more does Mia need?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a longer-\emph{looking} row is not always the bigger group ---
objects spread far apart still have to be counted or matched. And ``how many more''
never means add the two numbers together: it asks only for the leftovers.
\end{watchoutbox}
