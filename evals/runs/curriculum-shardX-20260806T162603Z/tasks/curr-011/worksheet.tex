\wstitleblock{More, Fewer, or the Same?}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Count each group and write its number.
  \item \emph{More} means a bigger count. \emph{Fewer} means a smaller count.
        \emph{The same} means the counts match.
\end{itemize}

\problem[2.8cm]{Some cats want dinner! Count the cats and the bowls. Write the
numbers.\par\vspace{0.2cm}
\begin{center}
Cats: {\LARGE $\bullet\quad\bullet\quad\bullet$}\qquad\qquad
Bowls: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet\quad\bullet$}
\end{center}
\vspace{0.2cm}\par Cats: \ansblank\qquad Bowls: \ansblank\qquad
Are there more cats or more bowls? \ansblank}

\problem[2.8cm]{Count each row. Write the numbers.\par\vspace{0.2cm}
\begin{center}

Row B: {\LARGE $\bullet\quad\bullet\quad\bullet\quad\bullet$}
\end{center}
\vspace{0.2cm}\par Row A: \ansblank\qquad Row B: \ansblank\qquad
Which row has fewer? Write A or B: \ansblank}

\problem[2.8cm]{Count the stars and the moons. Write the
numbers.\par\vspace{0.2cm}
\begin{center}
