\sstitleblock{Finding Parameters That Make Piecewise Functions Continuous}

\begin{formulabox}
\textbf{The continuity test at $x = p$.} $f$ is continuous at $p$ when
(i) $f(p)$ is defined, (ii) $\lim_{x \to p} f(x)$ exists, and (iii) the two are
equal. On a piecewise function only the \emph{junctions} can fail --- each
piece is already continuous on its own interval.\\[3pt]
\textbf{Turn it into an equation:} left-hand limit $=$ right-hand limit
$= f(p)$. One junction gives one equation, so one junction determines one
unknown constant and two junctions determine two.
\end{formulabox}

\skillheading{1. Solve the junction equation for a constant}

\begin{formulabox}
Substitute the junction value $p$ into the formula on each side --- for
polynomial and rational pieces the one-sided limit is just that substitution.
Set the two results equal and solve. With two unknowns, start at the junction
that involves only one of them.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x) = 3x + a$ for $x < 4$ and $f(x) = 2x + 7$ for
$x \geq 4$. Find $a$.
\step{Both pieces are polynomials, so the only risk is the junction at
$x = 4$ --- set the one-sided limits equal there.}
\step{Left: $3(4) + a = 12 + a$. Right: $2(4) + 7 = 15$, which is also $f(4)$.
So $12 + a = 15$.}

$\Rightarrow\quad \ans{a = 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x) = x^2 + a$ for $x < 3$ and $f(x) = 4x - 1$ for
$x \geq 3$. Find $a$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{a = 2}$}
\end{tryitbox}

\skillheading{2. Fill a removable hole with the limit value}

\begin{formulabox}
When a piece is defined for $x \neq p$ and a constant $k$ is assigned at $p$,
condition (iii) reads $k = \lim_{x \to p} f(x)$ --- so the whole problem is a
limit problem.\\[3pt]
Substitution giving $\frac{0}{0}$ means the form is indeterminate, not that the
limit is $0$: factor and cancel, multiply by the conjugate for radicals, or use
a known limit such as $\lim_{u \to 0} \frac{e^u - 1}{u} = 1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x) = \dfrac{x^2 - 25}{x - 5}$ for $x \neq 5$, and
$f(5) = k$. Find $k$.
\step{Continuity at a hole forces $k$ to be the limit, and substitution gives
$\frac{0}{0}$ --- so factor and cancel first.}
\step{$\dfrac{(x - 5)(x + 5)}{x - 5} = x + 5$ for $x \neq 5$, and substituting
$x = 5$ into $x + 5$ gives $10$.}

$\Rightarrow\quad \ans{k = 10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x) = \dfrac{x^2 - 16}{x - 4}$ for $x \neq 4$, and
$f(4) = k$. Find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k = 8}$}
\end{tryitbox}

\skillheading{3. Show that no constant can work}

\begin{formulabox}
A constant supplies $f(p)$ only. If the \emph{limit} fails to exist, no choice
of $k$ can repair the function.\\[3pt]
Compute both one-sided limits. Two different finite values means a jump
discontinuity; an infinite one-sided limit means an infinite discontinuity.
Both are non-removable, and the honest answer is ``no such $k$ exists''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(x) = x + 1$ for $x < 0$ and $f(x) = x^2 - 3$ for $x > 0$,
with $f(0) = k$. Can any $k$ make $f$ continuous at $x = 0$?
\step{The value $k$ only settles condition (i), so test condition (ii) first:
does the limit exist?}
\step{From the left, $x + 1 \to 1$. From the right, $x^2 - 3 \to -3$.}

The one-sided limits are $\ans{1}$ and $\ans{-3}$; they differ, so the limit
does not exist and no $k$ works --- a jump discontinuity.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $f(x) = 2x - 5$ for $x < 1$ and $f(x) = x + 1$ for $x > 1$.
Find both one-sided limits at $x = 1$ and decide whether some $k$ works.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$ and $\ans{2}$ --- they differ, so no $k$ works}
\end{tryitbox}
