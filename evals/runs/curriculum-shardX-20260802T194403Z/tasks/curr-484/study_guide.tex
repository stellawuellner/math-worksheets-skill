\sstitleblock{Separation and Initial-Condition Errors}

\skillheading{Skill 1 --- Separate the variables before integrating}

\begin{formulabox}
\textbf{Rule:} an integral cannot be evaluated while the unknown function is
still on the wrong side.
\[ \frac{dy}{dx} = g(x)h(y) \ \Longrightarrow \
   \int \frac{dy}{h(y)} = \int g(x)\,dx \]
\textbf{Test before integrating:} does the right side contain $y$? If it does,
move it across first. Treating $y$ as a constant inside
$\int \ldots dx$ is the single most common wrong first line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $\dfrac{dy}{dx} = 4xy$ with $y(0) = 2$ and find
$y(1)$.
\step{The right side contains $y$, so nothing may be integrated until the
variables are separated.}
\step{$\dfrac{dy}{y} = 4x\,dx$ gives $\ln|y| = 2x^{2} + C$, so
$y = Ae^{2x^{2}}$; $y(0) = A = 2$, and $y(1) = 2e^{2} \approx \ans{14.78}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $\dfrac{dy}{dx} = 6xy$ with $y(0) = 1$ and find
$y(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20.09}$}
\end{tryitbox}

\skillheading{Skill 2 --- Where the constant of integration goes}

\begin{formulabox}
\textbf{Rule:} exponentiating a \emph{sum} produces a \emph{product}.
\[ \ln|y| = kx + C \ \Longrightarrow \ y = e^{C}e^{kx} = Ae^{kx} \]
So the additive $+C$ becomes the multiplying factor $A$. Writing
$y = e^{kx} + C$ is the classic error, and it is caught in one line by
differentiating: $y' = ke^{kx}$ while $ky = ke^{kx} + kC$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $\dfrac{dy}{dx} = 3y$ with $y(0) = 4$ and find
$y(1)$.
\step{The equation is already separable in $y$ alone, so the whole question is
where the constant lands after exponentiating.}
\step{$\ln|y| = 3x + C$ gives $y = Ae^{3x}$; $y(0) = A = 4$, so
$y = 4e^{3x}$ and $y(1) = 4e^{3} \approx \ans{80.34}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $\dfrac{dy}{dx} = 2y$ with $y(0) = 5$ and find
$y(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36.95}$}
\end{tryitbox}

\skillheading{Skill 3 --- When to apply the initial condition}

\begin{formulabox}
\textbf{Rule:} integrate first, solve for the constant second, simplify third.
An initial condition can only be used once a constant exists to determine.\par
\smallskip
\textbf{And solve for $y$ carefully:} from $y^{2} = x^{2} + K$ the root applies
to the \emph{whole} side, $y = \pm\sqrt{x^{2}+K}$, with the sign chosen by the
initial condition. $\sqrt{u^{2}+v^{2}} \neq u + v$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $\dfrac{dy}{dx} = \dfrac{x}{y}$ with $y(0) = 3$ and
find $y(4)$.
\step{Separate and integrate before touching the initial condition; the
condition then fixes the constant inside the radical, not outside it.}
\step{$y\,dy = x\,dx$ gives $y^{2} = x^{2} + K$; $y(0) = 3$ gives $K = 9$, so
$y = \sqrt{x^{2}+9}$ and $y(4) = \sqrt{16+9} = \ans{5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $\dfrac{dy}{dx} = \dfrac{x}{y}$ with $y(0) = 5$ and
find $y(12)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} every particular solution can be checked in one line ---
differentiate it and compare with the original right-hand side. A solution that
fails that test is wrong no matter how tidy the algebra looked.
\end{watchoutbox}
