\sstitleblock{Arithmetic Sequences as Linear Functions}

\begin{formulabox}
\textbf{Three questions, three tools.} For an arithmetic sequence with first
term $a_1$ and common difference $d$:
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \emph{What is term $n$?} \quad $a_n = a_1 + (n-1)d$
\item \emph{Which term is this value?} \quad solve $a_1 + (n-1)d = $ value for $n$
\item \emph{What is the total of $n$ terms?} \quad
$S_n = \dfrac{n(a_1 + a_n)}{2}$
\end{itemize}
Read the question first: a term, a term \emph{number}, and a total are three
different answers.
\end{formulabox}

\skillheading{1. The explicit formula for a term}

\begin{formulabox}
$a_n = a_1 + (n-1)d$. The $n-1$ is the count of \textbf{steps}, not terms:
reaching term 10 from term 1 takes nine steps. A decreasing sequence simply has
a negative $d$ --- the formula never changes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the 10th term of the sequence with $a_1 = 6$ and
$d = 4$.
\step{One specific term is wanted, so use the explicit formula and count steps
rather than listing all ten terms.}
\step{Nine steps separate term 1 from term 10, so
$a_{10} = 6 + (10-1)(4) = 6 + 36$.}
\[ a_{10} = \ans{42} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the 8th term of the sequence with $a_1 = 11$ and
$d = 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{46}$}
\end{tryitbox}

\skillheading{2. Solving for which term a value is}

\begin{formulabox}
When the value is known and the position is not, set the explicit formula equal
to the value and solve for $n$.\\[2pt]
$n$ must be a \textbf{positive whole number}. A fractional $n$ means the value is
not a term of the sequence at all --- that is a real answer, not a mistake.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Which term of $4, 10, 16, \ldots$ equals 58?
\step{The value is known but its position is not, so this is an equation to
solve rather than a formula to evaluate.}
\step{$4 + (n-1)(6) = 58$ becomes $6n - 2 = 58$, so $6n = 60$.}
\[ n = \ans{10} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Which term of $9, 16, 23, \ldots$ equals 65?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{3. Summing a finite arithmetic sequence}

\begin{formulabox}
$S_n = \dfrac{n(a_1 + a_n)}{2}$ --- the number of terms times the
\textbf{average} of the first and last.\\[2pt]
The sum formula needs the \emph{last} term, so compute $a_n$ first whenever the
problem gives only $a_1$, $d$ and $n$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the sum of the first 10 terms of $5, 8, 11, \ldots$
\step{A total is wanted, so the sum formula applies --- but it needs the last
term, which is not given, so find that first.}
\step{$a_{10} = 5 + 9(3) = 32$, and then
$S_{10} = \dfrac{10(5+32)}{2} = 5 \times 37$.}
\[ S_{10} = \ans{185} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find the sum of the first 12 terms of $2, 6, 10, \ldots$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{288}$}
\end{tryitbox}

\skillheading{4. A sequence as a linear function}

\begin{formulabox}
Expanding $a_1 + (n-1)d$ gives $f(n) = dn + (a_1 - d)$: a line with
\textbf{slope $d$}, sampled only at $n = 1, 2, 3, \ldots$\\[2pt]
So the common difference is the slope, and the constant is the ``term zero''
value one step \emph{before} the first term --- not $a_1$ itself. That is the
whole difference between discrete and continuous linear change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} The sequence $7, 12, 17, \ldots$ is $f(n) = 5n + 2$. Find
$f(12)$.
\step{The sequence is already in function form, so the term number is just an
input --- no stepping needed.}
\step{Substituting the term number gives $f(12) = 5(12) + 2 = 60 + 2$.}
\[ f(12) = \ans{62} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The sequence $5, 11, 17, \ldots$ is $f(n) = 6n - 1$. Find
$f(9)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{53}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} use $n-1$, not $n$ --- one extra common difference is the
most common wrong answer on this topic. Keep the sign of $d$ from the story. And
never multiply the number of terms by the \emph{last} term to get a sum: multiply
by the average of the first and last.
\end{watchoutbox}
