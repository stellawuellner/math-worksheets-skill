\sstitleblock{Integration by Parts: Choosing the Pieces}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- One Pass: Choosing u and dv}

\begin{formulabox}
\textbf{Formula.} $\displaystyle\int u\,dv = uv - \int v\,du$.\par
\textbf{Choosing.} Take as $u$ the factor that gets \emph{simpler} when
differentiated (a power of $x$, a logarithm, an inverse trig function), and as
$dv$ the factor you can integrate without making things worse (an exponential,
a sine or cosine, or just $dx$).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int x e^{3x}dx$.
\step{Differentiating $x$ removes it while the exponential integrates back to itself, so $x$ is the factor to call $u$.}
\step{$u = x$, $v = \tfrac{e^{3x}}{3}$, so the integral is $\tfrac{xe^{3x}}{3} - \tfrac{1}{3}\int e^{3x}dx$.}

$\Rightarrow\quad \ans{\tfrac{xe^{3x}}{3} - \tfrac{e^{3x}}{9} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int x e^{4x}dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{xe^{4x}}{4} - \tfrac{e^{4x}}{16} + C}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Logarithms and Inverse Trig}

\begin{formulabox}
\textbf{Why they go on top.} $\ln x$ and $\arctan x$ have no elementary
antiderivative you already know, but their \emph{derivatives} are simple
algebraic fractions. So they are always $u$, never $dv$.\par
When a logarithm is the only factor, take $dv = dx$ and $v = x$: parts still
applies with nothing else in sight.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int x^2\ln x\,dx$.
\step{The logarithm cannot be integrated directly but differentiates to $1/x$, so it must be $u$ and the power must be $dv$.}
\step{$v = \tfrac{x^3}{3}$, so the integral is $\tfrac{x^3\ln x}{3} - \tfrac{1}{3}\int x^2 dx$.}

$\Rightarrow\quad \ans{\tfrac{x^3\ln x}{3} - \tfrac{x^3}{9} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int x^3\ln x\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{x^4\ln x}{4} - \tfrac{x^4}{16} + C}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Repeating Until It Closes}

\begin{formulabox}
\textbf{Two ways a second pass ends it.} With $x^n$ times an exponential or a
trig function, each pass lowers the power by one until the polynomial is gone.
With $e^x\sin x$ the integral \emph{returns} after two passes --- call it $I$,
collect, and solve for $I$ algebraically.\par
Keep the same kind of factor as $dv$ on both passes, or the second pass undoes
the first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int x^2\sin x\,dx$.
\step{One pass drops the power from two to one but leaves the same shape, so plan on two passes with $dv$ trigonometric both times.}
\step{First pass gives $-x^2\cos x + 2\int x\cos x\,dx$, and the second gives $2\left(x\sin x + \cos x\right)$.}

$\Rightarrow\quad \ans{-x^2\cos x + 2x\sin x + 2\cos x + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int x^2\cos x\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2\sin x + 2x\cos x - 2\sin x + C}$}
\end{tryitbox}
