\wstitleblock{Random Samples and Simple Probability Models}{Grade 7}{}

\begin{center}
\textit{Probability as favourable outcomes over equally likely ones, scaling a
sample up to a population, and judging whether a sample can be trusted}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item When every outcome is equally likely,
        $P(\text{event}) = \dfrac{\text{number of favourable outcomes}}
        {\text{total number of outcomes}}$. Give probabilities as fractions in
        lowest terms.
  \item To scale a sample up to a population, multiply the population size by
        the fraction the sample gives you.
  \item A sample is \textbf{random} when every member of the population has the
        same chance of being chosen. An estimate from a sample that is not
        random tells you about the group you asked, not about the population.
  \item Every quantity below is stated with its units. Answer in the same units.
\end{itemize}

\problem[5cm]{A bag holds 12 counters, and 3 of them are red. One counter is
drawn at random. What is the probability that it is red?}

\problem[5cm]{A spinner is divided into 8 equal sections and 3 of them are

shaded section?}
