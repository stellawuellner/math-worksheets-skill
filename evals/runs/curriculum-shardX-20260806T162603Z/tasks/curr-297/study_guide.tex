\sstitleblock{Transformations of the Absolute-Value Graph}

\skillheading{Skill 1 --- Moving the V left, right, up and down}

\begin{formulabox}
\textbf{Parent graph:} \quad $y = |x|$ is a V with its corner at the origin and arms of slope $1$ and $-1$.\par\smallskip
\textbf{Form:} \quad $y = |x-h| + k$ puts the corner at $(h, k)$. The corner is wherever the \emph{inside of the bars is zero}, which is why $|x-3|$ sits three units to the right and $|x+3|$ three units to the left.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the $x$-intercepts of $y = |x-2| - 3$.
\step{An intercept means $y = 0$, so isolate the bars first --- and expect two answers, because two numbers lie the same distance from zero.}
\step{$|x-2| = 3$ splits into $x - 2 = 3$ and $x - 2 = -3$, giving $x = 5$ and $x = -1$.}
\par
$x = \ans{-1}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the $x$-intercepts of $y = |x+3| - 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-5}$ or $x = \ans{-1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the corner of $y = |x+3| - 2$ is at $x = -3$, not $x = 3$. Ask ``what makes the inside zero?'' instead of reading the sign, and the direction is never wrong.
\end{watchoutbox}

\skillheading{Skill 2 --- Flipping and steepening the arms}

\begin{formulabox}
\textbf{Form:} \quad $y = a|x-h| + k$. The number $a$ acts on the arms only, never on the corner's position.\par\smallskip
$a < 0$ turns the V upside down. $|a| > 1$ makes the arms steeper (narrower V); $|a| < 1$ makes them shallower (wider V). Substituting: bars first, then multiply by $a$, then add $k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $y = -2|x| + 6$, find the value at $x = 1$.
\step{This is a substitution, so the only decision is order --- the bars close before the $-2$ is applied, so take the distance first.}
\step{$|1| = 1$; then $-2 \cdot 1 = -2$; then $-2 + 6$.}
\par
$y = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $y = 3|x-1| - 5$, find the value at $x = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{1}$}
\end{tryitbox}

\skillheading{Skill 3 --- Drawing the graph from the equation}

\begin{formulabox}
\textbf{Three steps, always in this order:} plot the corner $(h,k)$; find the $x$-intercepts by solving $y = 0$; join each intercept to the corner with a \emph{straight} arm and continue it.\par\smallskip
The two intercepts are always the same distance from the corner --- if yours are not, a sign is wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Where does $y = -|x-1| + 4$ cross the horizontal axis?
\step{The corner is at $(1,4)$ and the minus flips the V downward, so the graph must come back down to the axis on both sides --- set $y = 0$ to find where.}
\step{$|x-1| = 4$ gives $x - 1 = \pm 4$, so $x = 5$ and $x = -3$, each four units from the corner.}
\par
$x = \ans{-3}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Where does $y = |x+1| - 3$ cross the horizontal axis?\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-4}$ or $x = \ans{2}$}
\end{tryitbox}
