\wstitleblock{Polynomial Zeros and Turning-Point Constraints}{Precalculus \quad Polynomial and Rational Functions}{}

\begin{center}
\textit{Factoring to locate zeros, reading multiplicity from a factored form,
bounding turning points by degree, and end behaviour of rational functions}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Give exact zeros, and list every distinct real zero.
\item A zero of \textbf{even} multiplicity touches the $x$-axis and turns
back; a zero of \textbf{odd} multiplicity crosses it.
\item A polynomial of degree $n$ has at most $n$ real zeros and at most
$n - 1$ turning points. Both bounds are ceilings, not counts.
\item For end behaviour of a rational function, compare the degrees of the
numerator and denominator.
\end{itemize}

\problem[5cm]{Find all real zeros. \qquad $f(x) = x^{2} - 7x + 10$}

\problem[6cm]{Let $g(x) = x^{3} - 4x$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] Factor $g(x)$ completely.
\item[(b)] State all real zeros of $g$.
\end{itemize}}

\problem[6cm]{Let $p(x) = (x - 3)^{2}(x + 1)$. List each distinct real zero
with its multiplicity, and say whether the graph crosses or touches the
$x$-axis there.}

\problem[6cm]{A polynomial function $h$ has degree $5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] What is the greatest number of turning points the graph of $h$ can
have?
\item[(b)] What is the greatest number of distinct real zeros $h$ can have?
\end{itemize}}

\problem[6cm]{Find the end behaviour of the rational function below by
evaluating the limit, and state the equation of the horizontal asymptote.
\[ \lim_{x \to \infty} \frac{3x^{2} - 5}{x^{2} + 4} \]}

\problem[7cm]{Find all real zeros. One zero is a small integer --- test the
factors of the constant term first, then divide.
\[ f(x) = x^{3} - 2x^{2} - 5x + 6 \]}

\problem[7cm]{Find all distinct real zeros of $q(x) = x^{4} - 6x^{3} + 9x^{2}$
and give the multiplicity of each. How many times does the graph of $q$ cross
the $x$-axis?}

\problem[6cm]{The graph of a polynomial function $f$ has exactly $4$ turning
points. What is the smallest possible degree of $f$? Explain in one sentence
how the turning-point bound forces your answer.}

\problem[7cm]{Let $r(x) = x^{4} - 5x^{2} + 4$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] Factor $r(x)$ completely. (Treat it as a quadratic in $x^{2}$.)
\item[(b)] State all real zeros, and say how many turning points the graph of
$r$ must therefore have.
\end{itemize}}

\problem[3cm]{Let $s(x) = (x + 2)(x - 1)^{2}(x - 3)$, a quartic with positive
leading coefficient.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item[(a)] List each distinct real zero with its multiplicity.
\item[(b)] Sketch a graph of $s$ on the grid: mark the zeros, show a crossing
or a touch at each one, and make the end behaviour correct. Exact
$y$-values are not needed.
\end{itemize}

\problem[7cm]{Evaluate the limit and state the horizontal asymptote of the
graph of $y = \dfrac{4x^{2} + 3}{2x^{2} - x}$.
\[ \lim_{x \to \infty} \frac{4x^{2} + 3}{2x^{2} - x} \]}

\problem[8cm]{A student claims to have found a polynomial of degree $5$ whose
graph has $5$ distinct real $x$-intercepts and only $2$ turning points.
Explain why no such polynomial exists. Your explanation should say what has to
happen to the graph between two consecutive $x$-intercepts. \noansline}
