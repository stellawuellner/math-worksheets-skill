\wstitleblock{Algebraic Limits: Factoring and Rationalizing}{AP Calculus AB/BC}{}

\noindent\small Each limit below is indeterminate of the form $\tfrac{0}{0}$ on direct
substitution. Rewrite first, then substitute. Show the algebra that removes the
common factor.\normalsize

\problem[4cm]{Evaluate $\displaystyle\lim_{x\to 3}\frac{x^2-9}{x-3}$.

\smallskip\noindent\small The table below shows the function approaching $x=3$. Use it to
predict the limit, then confirm it algebraically.\normalsize
\begin{center}\small
\begin{tabular}{|c|c|c|c|c|c|}\hline
$x$ & 2.9 & 2.99 & 3 & 3.01 & 3.1 \\\hline
$f(x)$ & 5.9 & 5.99 & undefined & 6.01 & 6.1 \\\hline
\end{tabular}
\end{center}}

\problem[4cm]{Evaluate $\displaystyle\lim_{x\to 2}\frac{x^2-5x+6}{x-2}$.}

\problem[4cm]{Evaluate $\displaystyle\lim_{x\to 2}\frac{x^3-8}{x-2}$.
\smallskip\noindent\small (Hint: $a^3-b^3=(a-b)(a^2+ab+b^2)$.)\normalsize}

\problem[4.5cm]{Evaluate $\displaystyle\lim_{x\to 0}\frac{\sqrt{x+4}-2}{x}$ by
multiplying by the conjugate.}

\problem[4.5cm]{Evaluate $\displaystyle\lim_{x\to 9}\frac{\sqrt{x}-3}{x-9}$.}

\problem[4.5cm]{Evaluate $\displaystyle\lim_{x\to 1}\frac{2-\sqrt{x+3}}{x-1}$.
Watch the sign: the radical is being subtracted \emph{from} a constant.}

\problem[4cm]{Let
\[
f(x)=\begin{cases}\dfrac{x^2-5x-14}{x-7}, & x\neq 7\\[8pt] c, & x=7\end{cases}
\]
Find the value of $c$ that makes $f$ continuous at $x=7$.}

\problem[5cm]{In problem 1 you cancelled the factor $(x-3)$ from
$\dfrac{x^2-9}{x-3}$ to get $x+3$. Explain, in a sentence or two, why that
cancellation is legitimate when computing the limit, even though
$\dfrac{x^2-9}{x-3}$ and $x+3$ are \emph{not} the same function at $x=3$.
\noansline}
