% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Derivatives of a parametrised curve}

\begin{formulabox}
\textbf{Slope:} $\dfrac{dy}{dx} = \dfrac{dy/dt}{dx/dt}$ --- never $dy/dt$ on its own. \textbf{Concavity:} differentiate that slope with respect to $t$ and divide by $dx/dt$ \emph{again}:
\[ \frac{d^2y}{dx^2} = \frac{\frac{d}{dt}\!\left(\frac{dy}{dx}\right)}{dx/dt}. \]
The second division is the step almost everyone drops.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $x = t^3$, $y = t^2 - t$. Find $\dfrac{dy}{dx}$ at $t = 2$.
\step{The curve is given through a parameter, so the slope is a ratio of the two $t$-derivatives rather than either one alone.}
\step{$\dfrac{dy}{dt} = 2t - 1$ and $\dfrac{dx}{dt} = 3t^2$, so at $t = 2$ the ratio is $\dfrac{3}{12}$.}

$\Rightarrow\quad \ans{0.25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $x = t^3$, $y = t^2 + t$. Find $\dfrac{dy}{dx}$ at $t = 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.417}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Polar bounds that trace the region once}

\begin{formulabox}
\textbf{Area:} $A = \dfrac{1}{2}\displaystyle\int_{\alpha}^{\beta} r^2\,d\theta$, where $[\alpha, \beta]$ traces the boundary \emph{exactly once}. Find the bounds from the equation, by solving $r = 0$ and taking two consecutive solutions. Circles $r = a\sin\theta$ and $r = a\cos\theta$ close after an interval of length $\pi$, not $2\pi$; a petal of $r = a\cos(n\theta)$ runs from $-\frac{\pi}{2n}$ to $\frac{\pi}{2n}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the area enclosed by $r = 6\cos\theta$.
\step{This is a circle, so the first job is the interval that draws it once: solving $6\cos\theta = 0$ gives the endpoints.}
\step{The zeros are $\theta = \pm\frac{\pi}{2}$, so $A = \frac{1}{2}\int_{-\pi/2}^{\pi/2} 36\cos^2\theta\,d\theta = 18\cdot\frac{\pi}{2}$.}

$\Rightarrow\quad \ans{A = 9\pi \approx 28.274}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the area enclosed by $r = 4\cos\theta$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{A = 4\pi \approx 12.566}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Speed is a length, not a sum}

\begin{formulabox}
\textbf{Speed} at time $t$ is $\sqrt{\big(x'(t)\big)^2 + \big(y'(t)\big)^2}$ --- the length of the velocity vector, computed from the \emph{derivatives}, never from the position. \textbf{Distance travelled} is $\int_a^b \text{speed}\,dt$; a square root cannot be split across the sum inside it. To decide speeding up or slowing down, take the sign of $v \cdot a$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A particle has $x'(t) = 3$ and $y'(t) = 4t$. Find its speed at $t = 2$.
\step{Speed asks for the length of the velocity vector, so evaluate both derivatives at the instant and then take the magnitude.}
\step{At $t = 2$ the velocity is $\langle 3, 8\rangle$, so the speed is $\sqrt{9 + 64} = \sqrt{73}$.}

$\Rightarrow\quad \ans{8.544}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A particle has $x'(t) = 5$ and $y'(t) = 3t$. Find its speed at $t = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}
