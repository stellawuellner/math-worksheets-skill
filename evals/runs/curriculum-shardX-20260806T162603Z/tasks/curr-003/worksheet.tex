\wstitleblock{Numerals, Number Words, and Quantities}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:} Numbers can be shown three ways --- as
\textbf{dots} you can count, as a \textbf{numeral} like 4, and as a
\textbf{number word} like \emph{four}.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Count every dot picture one dot at a time.
  \item Write numerals carefully so a friend could read them.
\end{itemize}
