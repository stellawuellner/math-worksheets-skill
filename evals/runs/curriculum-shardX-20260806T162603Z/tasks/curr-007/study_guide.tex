\sstitleblock{Ten-Frames: Seeing Numbers to Ten}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Five and Some More}

\begin{formulabox}
\textbf{Rule:} A \textbf{full top row is 5}. When the top row is full, do not
recount it --- just count on through the bottom row: ``six, seven, \ldots''
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ten-frame has a full top row and 2 dots in the
bottom row. How many in all?
\step{The full top row is five without counting, so start at five and count on the bottom dots.}
\step{``Six, seven'' --- $5 + 2 = \ans{7}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A ten-frame has a full top row and 3 dots in the bottom
row. How many in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Almost Ten}

\begin{formulabox}
\textbf{Rule:} A \textbf{full frame is 10}. When a frame is nearly full, count
the \emph{empty boxes} instead --- each empty box means one less than ten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ten-frame has just 1 empty box. How many dots?
\step{Counting nine dots is slow --- count the one empty box and take it away from ten.}
\step{$10 - 1 = \ans{9}$ --- nine at a glance.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A ten-frame has 3 empty boxes. How many dots?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Empty boxes count \emph{down} from ten, dots in the
bottom row count \emph{up} from five. Pick the one with less to count!
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Building a Number on the Frame}

\begin{formulabox}
\textbf{Rule:} To show a number, \textbf{fill the top row first} (that is 5),
then put the rest in the bottom row. Numbers bigger than five always look like
``a full row and some more.''
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Show 7 on a ten-frame. How many dots go in each row?
\step{Fill the top row with five first, then figure out how many are still needed to reach seven.}
\step{Count on from five: ``six, seven'' --- $7 - 5 = \ans{2}$ dots go in the bottom row.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Show 9 on a ten-frame. After filling the top row, how
many dots go in the bottom row?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}
