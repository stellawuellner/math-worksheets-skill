\wstitleblock{Zeros That Hold a Place}{Grades 2--3}{}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Every number has a \textbf{hundreds} place, a \textbf{tens} place, and a \textbf{ones} place.
  \item A \textbf{0} means there is nothing in that place --- but the 0 still holds the place open, so the other digits stay where they belong.
  \item Use the charts and the space to show your thinking. Then write your answer on the line.
\end{itemize}
\normalsize

\problem[4cm]{Write the number this chart shows.\par
\begin{center}\large
\begin{tabular}{|c|c|c|}\hline
\textbf{hundreds} & \textbf{tens} & \textbf{ones} \\\hline
\rule{0pt}{0.9cm}4 & 0 & 7 \\\hline
\end{tabular}
\end{center}
Then finish the expanded form: \quad $\underline{\hspace{1.6cm}} + \underline{\hspace{1.2cm}} + \underline{\hspace{1.2cm}}$}

\problem[4cm]{This number is written in expanded form. Write it the usual way.
\begin{center}\large $600 + 0 + 5 = \underline{\hspace{2cm}}$\end{center}}

\problem[4cm]{Write $<$ or $>$ in the box to compare these two numbers.
\begin{center}\large $406$ \ \fbox{\rule{0pt}{0.7cm}\makebox[1.1cm]{}} \ $460$\end{center}
The digits are the same. Explain on the answer line which place makes the difference.}

\problem[5cm]{\textbf{Find the mistake.} Jamal writes the expanded form of $308$ like this:
\begin{center}\large $308 = 300 + 80$\end{center}
His pieces add up to $380$, not $308$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Tell what Jamal did wrong with the $0$ in the tens place.
  \item[(b)] Write the expanded form of $308$ correctly, and write the number it makes.
\end{itemize}}

\problem[5cm]{\textbf{Find the mistake.} Ruby writes the expanded form of $720$ like this:
\begin{center}\large $720 = 700 + 2$\end{center}
Her pieces add up to $702$, not $720$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Tell what Ruby did wrong with the $2$.
  \item[(b)] Write the expanded form of $720$ correctly, and write the number it makes.
\end{itemize}}

\problem[5cm]{Nina has base-ten blocks: $3$ hundreds, $0$ tens, and $9$ ones.
Her teacher gives her $4$ more tens.
What number do Nina's blocks show now?}

\problem[4.5cm]{Put these three numbers in order from \textbf{least} to \textbf{greatest}.
\begin{center}\large $380 \qquad 830 \qquad 308$\end{center}
Write them on the lines: \quad $\underline{\hspace{1.8cm}} \ , \ \underline{\hspace{1.8cm}} \ , \ \underline{\hspace{1.8cm}}$}

\problem[5.5cm]{\textbf{Challenge.} One number has $7$ hundreds, $0$ tens, and $0$ ones.
A second number has $0$ hundreds, $7$ tens, and $0$ ones.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Write both numbers.
  \item[(b)] Both numbers use the digit $7$. Write $<$ or $>$ between them, and tell how you know.
\end{itemize}}
