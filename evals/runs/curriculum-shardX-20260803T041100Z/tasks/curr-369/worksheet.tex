\wstitleblock{Rational Functions: Canceled-Factor Holes vs Vertical Asymptotes}{Algebra 2 --- Rational Functions}{}

\noindent\textbf{The diagnosis, in the order you should run it:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Factor the numerator and the denominator completely. Do not cancel yet.
  \item Every zero of the \emph{original} denominator is a break in the domain --- the function is undefined there either way.
  \item A denominator factor that cancels \textbf{completely} leaves a \emph{hole}: the height is the reduced form evaluated at that $x$.
  \item A denominator factor that \textbf{survives} the reduction gives a \emph{vertical asymptote}: the values grow without bound there.
  \item A repeated factor cancels only as many times as it appears on top. $(x-2)^2$ over one $(x-2)$ still leaves one in the denominator.
\end{itemize}

\noindent For every find-and-fix item, name the factor that was mishandled
before you write the corrected answer.

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[4cm]{Factor both parts of $f(x) = \dfrac{x^2 - 9}{x^2 + x - 12}$
completely. Do not reduce yet --- write the two factored polynomials, and circle
the factor that appears in both.}

\problem[4.5cm]{Reduce $f(x) = \dfrac{x^2 - 9}{x^2 + x - 12}$ to lowest terms.
\par\vspace{2pt}
State the restriction on the domain that the reduced form no longer shows.}

\problem[5cm]{The factor $x - 3$ cancels in $f(x) = \dfrac{x^2 - 9}{x^2 + x - 12}$,
so the graph has a hole there.
\par\vspace{2pt}
Find the \emph{height} of that hole, i.e. the value the outputs approach as $x$
approaches $3$. Give an exact fraction.}

\problem[4cm]{Still with $f(x) = \dfrac{x^2 - 9}{x^2 + x - 12}$: one denominator
factor survives the reduction. Solve for the $x$-value of the vertical asymptote.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} Nina factors
$g(x) = \dfrac{x^2 - 25}{x^2 - 3x - 10}$ and writes: ``The denominator is zero at
$x = 5$ and $x = -2$, so there are vertical asymptotes at both.''
\par\vspace{2pt}
Name the factor Nina mishandled. Then give the one real vertical asymptote and
the exact coordinates of the hole.}

\problem[5.5cm]{\textbf{Find and fix the mistake.} Ray looks at
$h(x) = \dfrac{2x^2 - 8}{x^2 - 4x + 4}$, sees a factor of $x - 2$ on the top and
on the bottom, and concludes there is a hole at $x = 2$.
\par\vspace{2pt}
Factor the denominator completely, reduce $h$, and state what is really at
$x = 2$.}

\problem[5.5cm]{\textbf{Both at once.} For $k(x) = \dfrac{x^2 - x - 6}{x^2 - 4}$,
find the exact coordinates of the hole and the equation of the vertical
asymptote. Say which factor produced each.}

\problem[5cm]{\textbf{Explain.} A classmate says, ``If the denominator is zero
there, the graph shoots off to infinity --- that is what a zero denominator
means.'' Write a short paragraph explaining why that is only half true. Use
one-sided behaviour or a table of nearby outputs to contrast the two cases, and
say explicitly whether the function has a value at a hole. \noansline}
