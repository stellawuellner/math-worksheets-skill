\problem{\textbf{Translation: $A(1,1)$, $B(5,1)$ and $A'(4,3)$, $B'(8,3)$.}

\textbf{Step 1.} $AB = \sqrt{(5-1)^2 + (1-1)^2} = \sqrt{16 + 0} = 4$ and
$A'B' = \sqrt{(8-4)^2 + (3-3)^2} = \sqrt{16 + 0} = 4$.

\textbf{Step 2.} The lengths are equal, which is what a translation must do: it
slides every point by the same amount, so no distance can change.

\textbf{Step 3.} Equal corresponding lengths are evidence of congruence, not
proof on their own --- the full argument checks all three sides.
\ans{AB = A'B' = 4\ \text{units}}}

\problem{\textbf{Reflection across the $x$-axis: $D(2,3)$, $E(6,6)$.}

\textbf{Step 1.} $DE = \sqrt{(6-2)^2 + (6-3)^2} = \sqrt{16 + 9} = 5$.

\textbf{Step 2.} A reflection across the $x$-axis negates each $y$-coordinate,
giving $D'(2,-3)$ and $E'(6,-6)$, so
$D'E' = \sqrt{(6-2)^2 + (-6-(-3))^2} = \sqrt{16 + 9} = 5$.

\textbf{Step 3.} Equal lengths again: reflection is a rigid motion, so
$\overline{DE} \cong \overline{D'E'}$.
\ans{DE = D'E' = 5\ \text{units}}}

\problem{\textbf{Dilation: $F(0,0)$, $G(3,4)$ to $F'(0,0)$, $G'(6,8)$.}

\textbf{Step 1.} $FG = \sqrt{3^2 + 4^2} = \sqrt{9 + 16} = 5$.
\ans{FG = 5\ \text{units}}

\textbf{Step 2.} $F'G' = \sqrt{6^2 + 8^2} = \sqrt{36 + 64} = 10$.

\textbf{Step 3.} The segments are \emph{not} congruent --- the image is twice as
long --- so a dilation is not a rigid motion. The scale factor is
$k = \frac{10}{5} = 2$, and the two segments are similar.
\ans{F'G' = 10\ \text{units}}}

\problem{\textbf{Rotation about the origin: $H(1,2)$, $J(4,6)$.}

\textbf{Step 1.} $HJ = \sqrt{(4-1)^2 + (6-2)^2} = \sqrt{9 + 16} = 5$.

\textbf{Step 2.} $H'J' = \sqrt{(-6-(-2))^2 + (4-1)^2} = \sqrt{16 + 9} = 5$.

\textbf{Step 3.} Yes --- the rotation preserves length. (The mapping here is
$(x,y) \rightarrow (-y, x)$, a quarter turn counterclockwise.)
\ans{HJ = H'J' = 5\ \text{units}}}

\problem{\textbf{Translation and slope: $K(-2,1)$, $L(2,9)$.}

\textbf{Step 1.} Length alone cannot tell a translation from a rotation, so use
direction: slope of $KL = \frac{9-1}{2-(-2)} = \frac{8}{4} = 2$.

\textbf{Step 2.} Slope of $K'L' = \frac{7-(-1)}{5-1} = \frac{8}{4} = 2$.

\textbf{Step 3.} Equal slopes mean the segments are parallel. A translation
never turns a figure, so parallel corresponding sides are exactly what it
predicts --- and a rotation of anything other than a half turn would have
changed the slope. \ans{m = 2 \quad \text{for both}}}

\problem{\textbf{Dilation of triangle $MNP$, scale factor found from the sides.}

\textbf{Step 1.} $MN = \sqrt{4^2 + 0^2} = 4$ and $M'N' = \sqrt{6^2 + 0^2} = 6$.

\textbf{Step 2.} $NP = \sqrt{(0-4)^2 + (3-0)^2} = \sqrt{16 + 9} = 5$ and
$N'P' = \sqrt{(0-6)^2 + (4.5-0)^2} = \sqrt{36 + 20.25} = 7.5$.

\textbf{Step 3.} The ratios are $\frac{6}{4} = 1.5$ and $\frac{7.5}{5} = 1.5$.
Because \emph{both} corresponding ratios give the same number, the triangles are
similar with scale factor $1.5$ --- one equal ratio would not have been enough.
\ans{4,\ 6,\ 5,\ 7.5\ \text{units}}}

\problem{\textbf{Full SSS evidence for $QRS$ and $Q'R'S'$.}

\textbf{Step 1.} $QR = \sqrt{(4-1)^2 + 0^2} = 3$; \quad
$Q'R' = \sqrt{0^2 + (4-1)^2} = 3$.

\textbf{Step 2.} $RS = \sqrt{(1-4)^2 + (5-1)^2} = \sqrt{9+16} = 5$; \quad
$R'S' = \sqrt{(5-1)^2 + (1-4)^2} = \sqrt{16+9} = 5$.

\textbf{Step 3.} $QS = \sqrt{0^2 + (5-1)^2} = 4$; \quad
$Q'S' = \sqrt{(5-1)^2 + 0^2} = 4$. All three corresponding pairs are equal, so
the triangles are congruent by SSS. (The motion is a reflection across the line
$y = x$, which swaps each point's coordinates.)
\ans{3,\ 5,\ 4\ \text{units for each triangle}}}

\problem{\textbf{Half turn: $T(2,1)$, $U(6,4)$.}

\textbf{Step 1 (part a).} $TU = \sqrt{(6-2)^2 + (4-1)^2} = \sqrt{16+9} = 5$ and
$T'U' = \sqrt{(-6-(-2))^2 + (-4-(-1))^2} = \sqrt{16+9} = 5$.
\ans{TU = T'U' = 5}

\textbf{Step 2 (part b).} Slope of $TU = \frac{4-1}{6-2} = \frac{3}{4}$; slope
of $T'U' = \frac{-4-(-1)}{-6-(-2)} = \frac{-3}{-4} = \frac{3}{4}$.
\ans{\tfrac{3}{4} \quad \text{for both}}

\textbf{Step 3 (part c).} A half turn sends $(x,y)$ to $(-x,-y)$, so both the
rise and the run change sign and the quotient is unchanged. A half turn is the
only rotation that leaves every segment parallel to itself; a quarter turn, for
instance, would have given the negative reciprocal $-\frac{4}{3}$.}

\problem{\textbf{Error analysis: Ana's congruence claim.}

\textbf{Step 1 (part a).} $AB = \sqrt{(4-0)^2 + 0^2} = 4$ and
$A'B' = \sqrt{(8-0)^2 + 0^2} = 8$.

\textbf{Step 2 (part b).} Congruent figures have \emph{equal} corresponding
sides, and $8 \neq 4$, so the triangles cannot be congruent. Being right
triangles that share a vertex says nothing about size --- shape agreement is not
size agreement.

\textbf{Step 3 (part c).} Every corresponding side doubles ($4 \rightarrow 8$,
and the vertical legs $3 \rightarrow 6$), so the triangles are \emph{similar},
produced by a dilation centred at the origin with scale factor $2$.

\textit{Manual review (parts b and c):} accept any sentence that contrasts equal
lengths with proportional lengths, and any answer naming similarity via a
dilation of factor $2$ about the origin. \ans{AB = 4,\ A'B' = 8}}

\problem{\textbf{Identify the rigid motion: $VWX \rightarrow V'W'X'$.}

\textbf{Step 1 (part a).} $VW = \sqrt{(4-1)^2 + 0^2} = 3$ and
$V'W' = \sqrt{(-4-(-1))^2 + 0^2} = 3$. Equal, so the motion is rigid.
\ans{VW = V'W' = 3}

\textbf{Step 2 (part b).} $W(4,1)$ and $X(4,3)$ share an $x$-coordinate, so the
run is zero and the slope is undefined; the same is true of $W'(-4,1)$ and
$X'(-4,3)$. \ans{\text{undefined for both}}

\textbf{Step 3 (part c).} Every $x$-coordinate has changed sign while every
$y$-coordinate is unchanged: $(x,y) \rightarrow (-x, y)$. That is a reflection
across the $y$-axis. The equal lengths show the motion is rigid; the unchanged
vertical direction and the mirrored $x$-coordinates identify which rigid motion
it is.

\textit{Manual review (part c):} accept ``reflection across the $y$-axis'' with
any justification that cites the equal side length plus the sign change in the
$x$-coordinates only.}

\problem{\textbf{Similar or not: $ABC$ and $DEF$.}

\textbf{Step 1.} $AB = \sqrt{(5-1)^2 + 0^2} = 4$ and
$DE = \sqrt{(10-2)^2 + 0^2} = 8$, so that ratio is $2$.

\textbf{Step 2.} $AC = \sqrt{0^2 + (4-1)^2} = 3$ and
$DF = \sqrt{0^2 + (9-3)^2} = 6$, again a ratio of $2$.

\textbf{Step 3.} $BC = \sqrt{(1-5)^2 + (4-1)^2} = \sqrt{16+9} = 5$ and
$EF = \sqrt{(2-10)^2 + (9-3)^2} = \sqrt{64+36} = 10$, a ratio of $2$ once more.
All three ratios agree, so the triangles are similar with scale factor $2$ --- a
dilation of factor $2$ followed by a translation carries $ABC$ onto $DEF$.
\ans{4,\ 8,\ 3,\ 6,\ 5,\ 10\ \text{units}}}

\problem{\textbf{Synthesis: $JKL \rightarrow J'K'L'$.}

\textbf{Step 1 (part a).} $JK = \sqrt{(-1-(-4))^2 + 0^2} = 3$ and
$J'K' = \sqrt{0^2 + (-1-(-4))^2} = 3$. Equal lengths, so the motion is rigid.
\ans{JK = J'K' = 3}

\textbf{Step 2 (part b).} Slope of $JL = \frac{6-2}{-1-(-4)} = \frac{4}{3}$.
\ans{\tfrac{4}{3}}

Slope of $J'L' = \frac{-1-(-4)}{6-2} = \frac{3}{4}$. The two slopes are
reciprocals of each other. \ans{\tfrac{3}{4}}

\textbf{Step 3 (part c).} Look at the coordinates: $(-4,2) \rightarrow (2,-4)$,
$(-1,2) \rightarrow (2,-1)$, $(-1,6) \rightarrow (6,-1)$. Every point has had its
coordinates swapped, $(x,y) \rightarrow (y,x)$, which is a reflection across the
line $y = x$. The reciprocal slopes are exactly the fingerprint of that mirror:
swapping coordinates swaps rise and run.

\textbf{Step 4 (part d).} Equal side lengths only establish that \emph{some}
rigid motion works --- a translation, a rotation and a reflection all preserve
length, so the lengths cannot tell them apart. Direction is what distinguishes
them: a translation keeps every slope, a half turn keeps every slope, a quarter
turn takes each slope to its negative reciprocal, and a reflection across
$y = x$ takes each slope to its reciprocal.

\textit{Manual review (parts c and d):} accept ``reflection across the line
$y = x$'' justified by the coordinate swap or by the reciprocal slopes, and any
part (d) answer making the point that several different rigid motions all
preserve length, so length alone cannot single one out.}
