% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Four-Step Cycle of Powers of $i$}

\begin{formulabox}
\textbf{The cycle:} \quad
\begin{tabular}{c@{\;=\;}c@{\qquad}c@{\;=\;}c@{\qquad}c@{\;=\;}c@{\qquad}c@{\;=\;}c}
$i^{1}$ & $i$ & $i^{2}$ & $-1$ & $i^{3}$ & $-i$ & $i^{4}$ & $1$
\end{tabular}\\[4pt]
\textbf{Rule:} divide the exponent by $4$ and keep the remainder. Remainder
$0 \to 1$, \ $1 \to i$, \ $2 \to -1$, \ $3 \to -i$. A coefficient in front of
the power just rides along.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $2i^{19} + i^{24}$
\step{Both terms are powers of $i$ with big exponents, so reduce each exponent modulo $4$ before doing any arithmetic.}
\step{For the first term the remainder is $3$, so that power is $-i$ and the term becomes $-2i$. For the second the remainder is $0$, so that power is $1$.}

$\Rightarrow\quad \ans{1 - 2i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $3i^{22} - i^{13}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3 - i}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Points and Distances on the Plane}

\begin{formulabox}
\textbf{The picture:} $a + bi$ is the point $(a, b)$ --- $a$ along the
horizontal \textit{real} axis, $b$ up the vertical \textit{imaginary} axis.\\[3pt]
\textbf{Distance from the origin:} the two parts are the legs of a right
triangle, so the distance is $\sqrt{a^{2} + b^{2}}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Locate $w = -3 - 4i$ and find its distance from the origin.
\step{Negative real part means left of the origin and negative imaginary part means below it, so $w$ sits in the lower-left quarter --- then the two parts are the legs of a right triangle.}
\step{The legs are $3$ and $4$, so the distance is the hypotenuse $\sqrt{9 + 16}$.}
\begin{center}
\begin{tikzpicture}[scale=0.32]
  \draw[gray!55, thin] (-6,-6) grid (6,6);
  \draw[->] (-6.5,0) -- (6.5,0) node[right, font=\tiny] {re};
  \draw[->] (0,-6.5) -- (0,6.5) node[above, font=\tiny] {im};
  \draw[thick] (0,0) -- (-3,-4);
  \fill (-3,-4) circle (10pt);
  \node[below left, font=\tiny] at (-3,-4) {$w$};
\end{tikzpicture}
\end{center}

$\Rightarrow\quad \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How far is $6 + 8i$ from the origin?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiplying by $i$ Is a Quarter Turn}

\begin{formulabox}
\textbf{Rule:} $i(a + bi) = -b + ai$. On the plane this is a quarter turn
($90^\circ$) counterclockwise about the origin, and the distance from the origin
does not change.\\[3pt]
Two turns ($i^{2}$) is a half turn, which negates both parts; four turns
($i^{4}$) is all the way round, back to the start.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compute $i(4 - 3i)$
\step{Multiplying by $i$ should turn the point a quarter turn, so expect the parts to swap and one sign to flip --- distribute and check that against the answer.}
\step{$4i - 3i^{2}$, and $-3i^{2} = +3$, so the real part is $3$ and the imaginary part is $4$: the point $(4, -3)$ has turned to $(3, 4)$.}

$\Rightarrow\quad \ans{3 + 4i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compute $i(-2 + 5i)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5 - 2i}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the remainder $2$ gives $-1$, not $+1$ --- the most
common slip on this page is losing that sign in a term like $3i^{14}$. And a
remainder of $0$ means the power is $1$, not $0$.
\end{watchoutbox}
