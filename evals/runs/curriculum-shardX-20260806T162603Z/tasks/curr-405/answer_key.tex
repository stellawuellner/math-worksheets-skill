\problem{$f(x) = x^{2}$, $g(x) = f(x-2) + 3$. Find $g(5)$.}

\textbf{Solution:} The $-2$ is inside $f$, so it changes the input: $g(5) = f(3) + 3 = 9 + 3$. \ans{g(5) = 12}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = 2x+1$, $g(x) = x^{2}$. Find $(f \circ g)(3)$.}

\textbf{Solution:} Inner first: $g(3) = 9$, then $f(9) = 2(9) + 1$. \ans{(f \circ g)(3) = 19}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = 5x - 3$. Find $f^{-1}(12)$.}

\textbf{Solution:} From $y = 5x - 3$, $x = \dfrac{y+3}{5}$, so $f^{-1}(x) = \dfrac{x+3}{5}$ and $f^{-1}(12) = \dfrac{15}{5}$. \ans{f^{-1}(12) = 3}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \sqrt{x}$, $g(x) = -2f(x+1)$. Find $g(8)$.}

\textbf{Solution:} $g(8) = -2\sqrt{8 + 1} = -2\sqrt{9} = -2(3)$. \ans{g(8) = -6}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = x^{2} - 1$, $g(x) = x+4$. Write $(f \circ g)(x)$.}

\textbf{Solution:} Substitute $x + 4$ into $f$:
\begin{align*}
  (x + 4)^{2} - 1 &= x^{2} + 8x + 16 - 1
\end{align*}
\ans{(f \circ g)(x) = x^{2} + 8x + 15}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \dfrac{x-7}{2}$. Write $f^{-1}$ and find $f^{-1}(4)$.}

\textbf{Solution:} Undo in reverse: multiply by $2$, then add $7$, so $f^{-1}(x) = 2x + 7$. Then $f^{-1}(4) = 8 + 7$. \ans{f^{-1}(4) = 15}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \sqrt{x-1}$, $g(x) = x^{2}+1$. Simplify $(f \circ g)(x)$ and solve $(f \circ g)(x) = 3$.}

\textbf{Solution:} $f\bigl(g(x)\bigr) = \sqrt{\left(x^{2} + 1\right) - 1} = \sqrt{x^{2}} = |x|$. The composition is the absolute-value function, not $x$ --- the square root undoes the square only for non-negative inputs. Then $|x| = 3$ gives two solutions.
\[ \boxed{x = -3 \quad \text{or} \quad x = 3} \]

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = x^{2} - 17$, $g(x) = 3f(x-2) + 5$. \textbf{(a)} $g(6)$. \textbf{(b)} Describe the transformations.}

\textbf{Solution (a):} $g(6) = 3f(4) + 5 = 3\left(16 - 17\right) + 5 = 3(-1) + 5$. \ans{g(6) = 2}

\textbf{Solution (b):} \emph{Instructor-judged.} A complete answer gives a horizontal shift $2$ units \textbf{right}, a vertical stretch by $3$, and a vertical shift $5$ units \textbf{up}, with the stretch applied before the shift. Naming all three but shifting up before stretching is wrong: the $5$ would be tripled. The horizontal shift may come at any point, but ``left'' is not acceptable.

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = (x-3)^{2} + 1$ on $x \geq 3$. Find $f^{-1}(17)$.}

\textbf{Solution:} Solve $y = (x-3)^{2} + 1$: $(x-3)^{2} = y - 1$, so $x = 3 \pm \sqrt{y-1}$. The restriction $x \geq 3$ selects the plus sign, giving $f^{-1}(x) = 3 + \sqrt{x-1}$. Then $f^{-1}(17) = 3 + \sqrt{16} = 3 + 4$. \ans{f^{-1}(17) = 7}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = \dfrac{1}{x+2}$, $g(x) = \dfrac{1}{x} - 2$. \textbf{(a)} Simplify $(f \circ g)(x)$. \textbf{(b)} What it means.}

\textbf{Solution (a):} The $-2$ inside $g$ and the $+2$ inside $f$ cancel:
\[ f\bigl(g(x)\bigr) = \frac{1}{\left(\tfrac{1}{x} - 2\right) + 2} = \frac{1}{\tfrac{1}{x}} \]
and the reciprocal of $\tfrac{1}{x}$ is $x$. \ans{(f \circ g)(x) = x}

\textbf{Solution (b):} \emph{Instructor-judged.} Returning $x$ means $f$ undoes $g$, so the two are inverses of each other. A strong answer adds one of: the other order should also be checked, or the identity holds only where both rules are defined, so $x = 0$ and $x = -2$ are excluded. Simply restating that the expression simplifies is not an interpretation.

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = 2x+1$, $h(x) = f^{-1}(x-4)$. Find $h(11)$.}

\textbf{Solution:} $f^{-1}(x) = \dfrac{x-1}{2}$. The $x - 4$ is the input to $f^{-1}$, so $h(11) = f^{-1}(7) = \dfrac{7-1}{2}$. \ans{h(11) = 3}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$f(x) = 3x-5$, $g(x) = x^{2}$ on $x \geq 0$. \textbf{(a)} $(g \circ f)^{-1}(9)$. \textbf{(b)} Why the order reverses.}

\textbf{Solution (a):} Undo the outer function first. From $g$: the input to $g$ was $3x - 5$, and $g$ is inverted on $x \geq 0$ by the positive square root, so $3x - 5 = \sqrt{9} = 3$. Then undo $f$: $3x = 8$.
\[ \boxed{(g \circ f)^{-1}(9) = \tfrac{8}{3}} \]

\textbf{Solution (b):} \emph{Instructor-judged.} Composition applies $f$ first and $g$ second, so undoing it must remove $g$ first and $f$ second --- which is $f^{-1}$ applied \emph{after} $g^{-1}$, i.e. $f^{-1} \circ g^{-1}$. A concrete illustration with these rules (take the square root, then add $5$ and divide by $3$) is acceptable as long as the general reversal point is made. Asserting the identity with no reasoning earns no credit.
