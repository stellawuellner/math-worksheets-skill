% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Division That Comes Out Even}

\begin{formulabox}
\textbf{The four-step loop:} divide the leading terms, multiply the divisor by
that result, subtract the whole line, bring down the next term. Repeat until
what is left has lower degree than the divisor.\\[3pt]
\textbf{Check:} quotient $\times$ divisor should rebuild the dividend.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $\dfrac{x^{2} + 9x + 20}{x + 4}$
\step{Both leading terms are $x$-terms, so the first piece of the quotient is whatever multiplies $x$ into $x^{2}$ --- start the loop there.}
\step{$x(x + 4) = x^{2} + 4x$; subtracting leaves $5x + 20$. Then $5(x + 4) = 5x + 20$, and the subtraction leaves nothing.}

$\Rightarrow\quad \ans{x + 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $\dfrac{x^{2} - x - 12}{x - 4}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x + 3}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Quotient Plus a Remainder}

\begin{formulabox}
\textbf{Stop rule:} stop when the leftover has lower degree than the divisor.
That leftover is the remainder.\\[3pt]
\textbf{Report it as} quotient $+$ remainder over divisor. A negative remainder
is reported with a minus sign in front of the fraction, not hidden.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $\dfrac{2x^{2} + 3x - 1}{x + 1}$
\step{The divisor has degree $1$, so the loop runs until the leftover is a constant --- that constant is the remainder, not an error.}
\step{$2x(x + 1) = 2x^{2} + 2x$, leaving $x - 1$. Then $1(x + 1) = x + 1$, leaving $-2$, which has degree $0$: stop.}

$\Rightarrow\quad \ans{2x + 1 - \dfrac{2}{x + 1}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $\dfrac{3x^{2} + 2x + 7}{x + 2}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x - 4 + \dfrac{15}{x + 2}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Zero Placeholders for Missing Powers}

\begin{formulabox}
\textbf{Rule:} before dividing, rewrite the dividend with every descending power
present, using a $0$ coefficient where a power is missing --- for example
$x^{3} - 8$ becomes $x^{3} + 0x^{2} + 0x - 8$.\\[3pt]
Without the placeholders the columns slide and terms get subtracted from the
wrong partners.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $\dfrac{x^{3} + 27}{x + 3}$
\step{Two powers are missing from the dividend, so write them in with zero coefficients before starting the loop --- that is the whole difficulty here.}
\step{Dividing $x^{3} + 0x^{2} + 0x + 27$ gives $x^{2}$, then $-3x$, then $9$, and the final subtraction leaves nothing.}

$\Rightarrow\quad \ans{x^2 - 3x + 9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $\dfrac{x^{3} - 1}{x - 1}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2 + x + 1}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- The Remainder Theorem Shortcut}

\begin{formulabox}
\textbf{Remainder theorem:} dividing a polynomial $P$ by $x - a$ leaves the
remainder $P(a)$. So the remainder can be found by one substitution, with no
division at all.\\[3pt]
\textbf{Factor theorem:} $P(a) = 0$ exactly when $x - a$ is a factor. For a
divisor $x + 3$, the value to substitute is $a = -3$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the remainder when $x^{3} - 4x^{2} + x + 6$ is
divided by $x - 1$.
\step{Only the remainder is wanted, not the quotient, so the theorem replaces the whole division with one substitution at $a = 1$.}
\step{$1 - 4 + 1 + 6$, which is the same number the last subtraction of a full long division would leave.}

$\Rightarrow\quad \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the remainder when $2x^{3} + x - 5$ is divided by
$x + 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-23}$}
\end{tryitbox}
