\sstitleblock{Quotients and Remainders with Models}

\skillheading{Skill 1 --- Reading a Quotient off a Model}

\begin{formulabox}
\textbf{Every division fits one sentence:}\par
total $=$ (number of groups) $\times$ (size of a group) $+$ remainder.\par
On a model, the \textbf{quotient} is what you counted --- full groups, or how many are in each group ---
and the \textbf{remainder} is what would not fit into any group.\par
The remainder is always \emph{smaller} than the group size. If it is not, one more group can still be
made.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $17$ counters circled in groups of $4$. How many full groups?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}
  \foreach \i in {1,...,17}{\fill[blue!65] (0.38*\i,0) circle (2.4pt);}
\end{tikzpicture}
\end{center}
\step{The model asks how many complete groups fit, so I count groups of the given size and stop when the counters run out rather than dividing straight away.}
\step{Circling fours uses $4, 8, 12, 16$ counters --- four full groups --- and one counter is left, which is less than $4$, so no fifth group is possible.}

full groups $= \ans{4}$, remainder $1$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $29$ counters in groups of $7$. How many are left over?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

\skillheading{Skill 2 --- Deciding What the Remainder Means}

\begin{formulabox}
\textbf{The division is the same; the question decides what to do with the remainder.}\par
\textbf{Drop it} --- ``how many full boxes,'' ``how many complete rows.''\par
\textbf{Add one more group} --- ``how many vans are needed,'' ``how many boxes to hold them all.''\par
\textbf{The remainder IS the answer} --- ``how many are left over.''\par
Read the question again before you write the number down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $26$ people, tables seat $4$. How many tables are needed?
\step{Dividing gives a quotient and a leftover, so before writing anything I ask what the leftover people do --- and they still need somewhere to sit.}
\step{$26 \div 4$ is $6$ with $2$ left over. Six tables seat $24$, so the last two people need a seventh table.}

tables needed $= \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same numbers, different question: $26$ people at tables of $4$. How many tables
are \emph{completely full}?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimating Before You Divide}

\begin{formulabox}
\textbf{Round the total to a number that divides easily, keep the divisor exact, then divide.}\par
The estimate says roughly how big the quotient should be, so a slip in the long division shows up at
once.\par
Rounding \emph{up} makes the estimate a little too big; rounding \emph{down} makes it a little too
small. Knowing which way you went tells you which side the exact answer should land on.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $418$ to the nearest hundred, ready to estimate $418 \div 4$.
\step{I round only the total, because changing the divisor would change the size of the answer completely.}
\step{Rounding to the hundred means looking at the tens digit, $1$; it is below $5$, so $418$ rounds down.}

$418$ to the nearest hundred is $\ans{400}$, so the estimate is $400 \div 4 = 100$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $2{,}740$ to the nearest hundred, ready to estimate a division.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2700}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a remainder is never written as part of the quotient. $23 \div 5$ is
``$4$ remainder $3$,'' not $43$ and not $7$. And a remainder equal to or larger than the divisor
always means a group was missed.
\end{watchoutbox}
