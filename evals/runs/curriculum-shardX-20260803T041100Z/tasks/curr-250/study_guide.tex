\sstitleblock{Scatter Plots and Bivariate Association}

\skillheading{Skill 1 --- Reading a scatter plot and its trend}

\begin{formulabox}
\textbf{Read the plot in this order:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Direction} --- dots rising left to right is a positive association;
        falling is negative; no slant is no association.
  \item \textbf{Strength} --- tightly packed around a line is strong; widely scattered is
        weak.
  \item \textbf{Outliers} --- one dot far off the pattern of the rest. Note it before you
        summarise anything, because it moves means and it weakens the trend.
  \item \textbf{Range} --- the smallest and largest $x$-values in the data. Write them
        down: every prediction gets judged against them.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A plot has the points $(1, 12)$, $(2, 15)$, $(3, 26)$, $(4, 21)$,
$(5, 24)$. Which $x$-value carries the highest $y$-value, and does that point fit the
pattern?
\step{Highest $y$ means scan up the vertical axis first and only then read down to the
horizontal axis --- do not assume the largest $x$ carries it.}
\step{The top point is $26$, sitting above $x = 3$, while its neighbours at $x = 2$ and
$x = 4$ are only $15$ and $21$. The rest of the plot climbs gently, so this dot sits well
above the pattern: it is an outlier.}
$\Rightarrow\quad x = \ans{3}$, and it does \emph{not} fit the trend
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has the points $(2, 40)$, $(4, 44)$, $(6, 68)$, $(8, 55)$,
$(10, 60)$. Which $x$-value carries the highest $y$-value?\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{6}$}
\end{tryitbox}

\skillheading{Skill 2 --- Predicting from a trend line}

\begin{formulabox}
\textbf{Rule:} a trend line $y = mx + b$ turns any input into a prediction. Substitute the
$x$-value and evaluate --- the slope $m$ \emph{multiplies}, it never adds.\par
\smallskip
\textbf{Slope in words:} $m$ is the predicted change in $y$ for each one-unit increase in
$x$. \ \textbf{Intercept in words:} $b$ is the prediction at $x = 0$, which is often
outside the data and therefore often meaningless.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A trend line is $y = 1.5x + 20$. Predict $y$ when $x = 12$.
\step{A prediction from a line is just the line evaluated at that input --- substitute and
compute, keeping the slope as a multiplier.}
\step{$y = 1.5(12) + 20 = 38$, and in words: each extra unit of $x$ is worth about
$1.5$ more $y$.}
$\Rightarrow\quad y = \ans{38}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A trend line is $y = 2.5x + 10$. Predict $y$ when $x = 14$.\\[2pt]
\rotatebox{180}{\footnotesize check: $y = \ans{45}$}
\end{tryitbox}

\skillheading{Skill 3 --- Qualifying a prediction}

\begin{formulabox}
\textbf{Every prediction needs a sentence about how far to trust it.} Check all four:
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Inside the range?} Predicting inside the observed $x$-values is
        \emph{interpolation} (supported). Outside it is \emph{extrapolation} (assumed).
  \item \textbf{Possible?} An answer past a real limit --- a score above 100, a negative
        length --- shows the line has been pushed past where it can hold.
  \item \textbf{Outlier-driven?} Re-summarise without the odd point and see how far the
        answer moves.
  \item \textbf{Cause or company?} An association is not evidence that one variable
        \emph{makes} the other change.
\end{itemize}
\textbf{Residual} $=$ actual $-$ predicted. Positive means the line under-predicted that
point; negative means it over-predicted.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The line is $y = 1.5x + 20$. A real data point is $(20, 47)$. Find
the residual and say what it tells you.
\step{Residual is actual minus predicted, in that order, because the sign is the whole
message --- so compute the prediction and subtract it from the observed value.}
\step{$47 - (1.5(20) + 20) = -3$, and the negative sign says the observed value sits below
the line.}
$\Rightarrow\quad \ans{-3}$, so the line \emph{over}-predicted this point by three
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The line is $y = 2x + 12$. A real data point is $(25, 66)$. Find
the residual.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the arithmetic of a prediction is never the hard part --- a line
returns a number for any input, including inputs no data covers and outputs the situation
cannot produce. Before writing a prediction down, say out loud which observed range it
sits in. And never upgrade ``these two move together'' into ``this one causes that one''.
\end{watchoutbox}
