\sstitleblock{Converting Between Exponential and Logarithmic Forms}

\begin{formulabox}
\textbf{A logarithm is an exponent.} The two forms hold the same three numbers:
\[ b^{\,y} = x \qquad\Longleftrightarrow\qquad \log_b x = y \]
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item $b$ is the \textbf{base} --- the subscript in log form, the bottom number
in exponential form. It never changes role.
\item $y$ is the \textbf{exponent} --- and it is what the logarithm equals.
\item $x$ is the \textbf{result} --- and it is what the logarithm is taken of.
Only positive $x$ is allowed.
\end{itemize}
Choose the form that puts your unknown alone: unknown exponent $\to$ log form;
unknown inside the log $\to$ exponential form.
\end{formulabox}

\skillheading{1. Exponential to logarithmic}

\begin{formulabox}
Label base, exponent, result first. Then write $\log_{\text{base}}(\text{result})
= \text{exponent}$. The base drops to the subscript; nothing else moves.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Rewrite $4^3 = 64$ in logarithmic form and give the value.
\step{Name the roles before rearranging: base 4, exponent 3, result 64.}
\step{The base becomes the subscript and the exponent becomes the value, so
$\log_4 64 = 3$.}
\[ \log_4 64 = \ans{3} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Rewrite $7^2 = 49$ in logarithmic form and give the value of
$\log_7 49$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{2. Logarithmic to exponential}

\begin{formulabox}
$\log_b x = y$ rebuilds as $b^{\,y} = x$: the subscript is the base, the value of
the logarithm is the exponent. Use this direction whenever the unknown sits
inside the logarithm.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $\log_6 x = 3$.
\step{The unknown is what the logarithm is taken of, so exponential form is the
one that isolates it.}
\step{$\log_6 x = 3$ becomes $x = 6^3$, and $6^3 = 216$.}
\[ x = \ans{216} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $\log_9 x = \dfrac{1}{2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\skillheading{3. Evaluating a logarithm by converting}

\begin{formulabox}
To evaluate $\log_b x$, set it equal to $y$, convert to $b^{\,y} = x$, and write
both sides as powers of the same number. A result below 1 forces a negative
exponent; a result between 1 and the base forces a fraction between 0 and 1.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $\log_2 \dfrac{1}{16}$.
\step{Convert to a power question, then write both sides on the same base so the
exponents can be compared directly.}
\step{$2^{\,y} = \frac{1}{16} = 2^{-4}$, so $y = -4$.}
\[ \log_2 \tfrac{1}{16} = \ans{-4} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $\log_{25} 5$ exactly.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1/2}$}
\end{tryitbox}

\skillheading{4. Solving when the unknown is the exponent}

\begin{formulabox}
$b^{\,x} = k$ converts to $x = \log_b k$. For a decimal, change base:
$\log_b k = \dfrac{\ln k}{\ln b}$ --- a quotient of two logarithms, never the
logarithm of a quotient. Bracket the answer with nearby whole powers first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $3^{\,x} = 50$ to three decimals.
\step{The unknown is the exponent, so convert to logarithmic form rather than
dividing.}
\step{$x = \log_3 50 = \dfrac{\ln 50}{\ln 3}$, and since $3^3 = 27$ and
$3^4 = 81$, the answer must sit between 3 and 4.}
\[ x = \ans{3.561} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $5^{\,x} = 90$ to three decimals.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.796}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the subscript is always the base --- $\log_2 32 = 5$
rebuilds as $2^5$, never $5^2$. And $\dfrac{\ln 20}{\ln 3}$ is not
$\ln\!\left(\frac{20}{3}\right)$. Finally, $\log_b x$ exists only for positive
$x$, so check the inside of every logarithm after solving.
\end{watchoutbox}
