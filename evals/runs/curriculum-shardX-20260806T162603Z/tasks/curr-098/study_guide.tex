\sstitleblock{Sorting Quadrilaterals by Their Attributes}

\skillheading{Skill 1 --- Checking One Shape's Attributes}

\begin{formulabox}
\textbf{A quadrilateral} is any closed shape with four straight sides. Three attributes tell them
apart:\par
\textbf{square corners} --- corners that the corner of a card fits exactly;\par
\textbf{parallel sides} --- two sides that stay the same distance apart all the way along;\par
\textbf{equal sides} --- sides of the same length. Check each attribute separately.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many square corners does shape P have?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}[thick]
  \draw (0,0) rectangle (1.4,0.85);
  \node[below, font=\small] at (0.7,-0.1) {P};
  \begin{scope}[xshift=2.6cm]
    \draw (0,0.45) -- (0.8,0.85) -- (1.6,0.45) -- (0.8,0.05) -- cycle;
    \node[below, font=\small] at (0.8,-0.1) {Q};
  \end{scope}
  \begin{scope}[xshift=5.2cm]
    \draw (0,0) -- (1.4,0) -- (1.4,0.85) -- (0.5,0.85) -- cycle;
    \node[below, font=\small] at (0.8,-0.1) {R};
  \end{scope}
\end{tikzpicture}
\end{center}
\step{Corners are what this question is about, so I test each corner on its own and ignore the side lengths completely.}
\step{Holding a card corner against each of P's four corners, all four fit exactly.}

square corners on P $= \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Shape R above has two corners that fit a card corner and two that do not.
How many square corners does R have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Sorting a Set into Attribute Groups}

\begin{formulabox}
\textbf{One shape, many groups.} A shape joins \emph{every} group whose attribute it has, so the
groups overlap --- a square has four square corners \emph{and} four equal sides \emph{and} two pairs
of parallel sides.\par
To sort: pick one attribute, walk through every shape, and count only the shapes that pass.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A set of shapes is sorted. How many shapes have four square corners?
\step{The group is defined by one attribute, so I go shape by shape and ask only that one question, instead of trying to name each shape.}
\step{Testing the corners of every shape in the set, three of them pass; the rest have corners that do not fit a card corner.}

shapes with four square corners $= \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad In that same sorted set, two of the shapes have four sides of the same length.
How many shapes are in the four-equal-sides group?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Using Attribute Groups in a Real Count}

\begin{formulabox}
\textbf{Two steps, in this order:} (1) decide which kinds of shape belong to the group the question
names; (2) add only those counts.\par
Adding every number in the table answers ``how many altogether,'' which is a different question ---
read what the group is before you add.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A shop sold $6$ square tiles, $8$ rectangle tiles and $3$ trapezoid tiles.
How many quadrilateral tiles is that?
\step{Every one of the three kinds has four straight sides, so all three belong to the group named, and no tile is left out.}
\step{Add all three counts: $6 + 8 + 3$.}

quadrilateral tiles $= \ans{17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad From the same sales ($6$ square, $8$ rectangle, $3$ trapezoid), how many tiles
had four square corners?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} equal sides and square corners are different attributes. A rhombus has four
equal sides but no square corners; a rectangle has four square corners but not four equal sides. Only
a square has both, which is why a square lands in so many groups at once.
\end{watchoutbox}
