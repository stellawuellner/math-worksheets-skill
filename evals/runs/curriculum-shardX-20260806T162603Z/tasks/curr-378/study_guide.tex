\sstitleblock{Solving Exponential and Logarithmic Equations}

\skillheading{Skill 1 --- Match the bases}

\begin{formulabox}
\textbf{Rule:} if $b^{\,M} = b^{\,N}$ with $b>0$, $b \ne 1$, then $M = N$.
Rewrite both sides as powers of one base, then set the exponents equal. This is
the exact method --- use it whenever the numbers are powers of a common base.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2^{\,3x} = 32$.
\step{The right side is a power of the same base as the left, so no logarithm is
needed --- matching the bases gives an exact answer.}
\step{$32 = 2^5$, so $2^{3x} = 2^5$ and therefore $3x = 5$.}

$\Rightarrow\quad \ans{x = \tfrac{5}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $4^{\,x-1} = 64$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Take a logarithm of both sides}

\begin{formulabox}
\textbf{Method:} when the two sides are not powers of a common base, apply a
logarithm to both sides and use the power rule to bring the unknown down:
\[ b^{\,x} = c \ \Longrightarrow \ x\ln b = \ln c \ \Longrightarrow \
   x = \frac{\ln c}{\ln b} . \]
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $5^{\,x} = 40$, to two decimal places.
\step{$40$ is not a power of $5$, so matching bases is impossible and a logarithm
is the tool that frees an exponent.}
\step{$x\ln 5 = \ln 40$, so $x = \dfrac{\ln 40}{\ln 5} = \dfrac{3.68887\ldots}{1.60943\ldots}
= 2.29202\ldots$}

$\Rightarrow\quad \ans{2.29}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $3^{\,x} = 50$, to two decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3.56}$}
\end{tryitbox}

\skillheading{Skill 3 --- Logarithmic equations}

\begin{formulabox}
\textbf{Method:} condense to a single logarithm if there is more than one, then
rewrite in exponential form and solve. \textbf{Always finish with a domain
check}: every logarithm's input must be positive at your answer, or the candidate
is discarded.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\log_3(x-4) = 3$.
\step{The unknown is locked inside the logarithm, so rewriting the statement as a
power is what releases it.}
\step{$3^3 = x - 4$, so $27 = x - 4$. The input at $x = 31$ is $27 > 0$, so the
answer survives the domain check.}

$\Rightarrow\quad \ans{x = 31}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\log_2(x+5) = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 11}$}
\end{tryitbox}

\skillheading{Skill 4 --- Growth and decay models}

\begin{formulabox}
\textbf{Shape:} $A = A_0 \cdot r^{\,t}$, where $A_0$ is the starting amount and
$r$ is the factor per period ($r>1$ grows, $0<r<1$ decays). To find an
\emph{amount}, substitute $t$. To find a \emph{time}, isolate the power and take
logarithms.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A town of $8000$ people grows by $3\%$ a year, so
$P = 8000(1.03)^{\,t}$. Find the population after $12$ years.
\step{The unknown here is the amount, not the time, so the model only has to be
evaluated --- no logarithm is involved.}
\step{$1.03^{12} = 1.425760\ldots$, and $8000 \times 1.425760\ldots
= 11406.087\ldots$}

$\Rightarrow\quad \ans{11406.09}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An account of $5000$ dollars grows by $4\%$ a year:
$A = 5000(1.04)^{\,t}$. Find its value after $10$ years, to the nearest
cent.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7401.22}$}
\end{tryitbox}
