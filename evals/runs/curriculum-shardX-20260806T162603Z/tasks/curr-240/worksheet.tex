\wstitleblock{Transformation Evidence for Congruence and Similarity}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Congruence from lengths, similarity from ratios, slope evidence, and naming the transformation}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Distance between two points: $\sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$. \quad Slope: $\dfrac{y_2 - y_1}{x_2 - x_1}$.
  \item \textbf{Congruent} figures have equal matching lengths. \textbf{Similar} figures have matching lengths in one constant ratio.
  \item The problems switch between the four kinds of evidence on purpose --- decide which one the question needs before you calculate. All lengths are in grid units.
\end{itemize}
