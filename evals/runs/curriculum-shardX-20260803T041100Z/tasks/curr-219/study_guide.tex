\sstitleblock{Slope and Rate of Change --- Inverted and Sign-Reversed Errors}

\skillheading{Skill 1 --- Slope from two points, right side up}

\begin{formulabox}
\textbf{Formula:}\\[4pt]
\[ m = \frac{\text{rise}}{\text{run}} = \frac{y_2 - y_1}{x_2 - x_1} \]
\textbf{Two rules that prevent both classic errors:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Rise on top.} If your numerator is a difference of $x$'s, the
        fraction is \emph{inverted} --- your answer is the reciprocal of the slope.
  \item \textbf{Same order top and bottom.} Pick which point is $1$ and which is
        $2$, then use that order in both differences. Swapping one of them
        \emph{reverses the sign}.
  \item \textbf{Predict the sign first:} if $y$ goes down as $x$ goes up, the slope
        is negative. A sign you predicted rejects a reversed answer instantly.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the slope through $(-1, 6)$ and $(3, -2)$.
\step{$y$ drops from $6$ to $-2$ while $x$ rises, so the answer must be negative
--- fix that prediction before computing.}
\step{Point $1$ is $(-1, 6)$, point $2$ is $(3, -2)$: \
$m = \dfrac{-2 - 6}{3 - (-1)} = \dfrac{-8}{4}$.}
The slope is $\ans{m = -2}$, negative as predicted.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the slope through $(2, -3)$ and $(6, 9)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = 3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Rate of change from a table}

\begin{formulabox}
\textbf{Rule:} a table's rate of change is a slope --- the change in the measured
amount over the change in time:
\[ \text{rate} = \frac{\text{later amount} - \text{earlier amount}}
                     {\text{later time} - \text{earlier time}} \]
\textbf{Later minus earlier, in both.} Written that way the sign takes care of
itself: a falling quantity gives a negative numerator and therefore a negative
rate. Never subtract "big minus small" to keep the numbers friendly --- that
throws away the sign that carries the meaning.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A tank holds $45$ litres at hour $0$ and $17$ litres at hour
$7$. Find the rate of change.
\step{The amount is falling, so the rate must come out negative --- keep both
subtractions later-minus-earlier so the sign appears on its own.}
\step{$\dfrac{17 - 45}{7 - 0} = \dfrac{-28}{7}$.}
The rate is $\ans{-4}$ litres per hour.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A tank holds $39$ litres at hour $1$ and $9$ litres at hour
$6$. Find the rate of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-6}$ litres per hour}
\end{tryitbox}

\skillheading{Skill 3 --- Rate of change in a story}

\begin{formulabox}
\textbf{Rule:} in a story, the quantity being measured goes on top and the time
goes underneath --- ask ``how much per one unit of time?'' and the units name the
fraction for you (cm per minute $=$ cm $\div$ minutes).\par
\smallskip
\textbf{Two checks that catch an inverted answer:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Read the units of your fraction back. Minutes per centimetre is not what
        the question asked.
  \item Multiply your rate by the elapsed time. It must reproduce the total change.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A pool's depth is $12$ cm at minute $2$ and $3$ cm at
minute $8$. Find the rate of change of the depth.
\step{The question asks for cm per minute, so depth goes on top and time
underneath --- that decision alone prevents the inverted answer.}
\step{$\dfrac{3 - 12}{8 - 2} = \dfrac{-9}{6}$, and $6 \times 1.5 = 9$ cm lost in
those $6$ minutes $\checkmark$}
The rate is $\ans{-1.5}$ cm per minute.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A mug's temperature is $15$ degrees at minute $1$ and $3$
degrees at minute $5$. Find the rate of change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$ degrees per minute}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An inverted slope and a sign-reversed slope both keep the
\emph{digits} you computed, so re-checking the arithmetic never finds them. Check
the \emph{set-up} instead: what is on top, and did both differences run the same
way?
\end{watchoutbox}
