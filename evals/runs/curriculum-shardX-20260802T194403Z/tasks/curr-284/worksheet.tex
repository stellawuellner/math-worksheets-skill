\wstitleblock{Sign Errors in Parabola Transformations}{Algebra 1 \quad Quadratic Functions}{}

\begin{center}\small\textit{Reading vertex form, writing transformed equations, and finding sign errors}\end{center}

\noindent\small\textbf{Directions:} Every problem on this sheet turns on a \emph{sign}.
Work each one carefully and, where a student's work is shown, say exactly which step went
wrong before you write the correct answer. Showing the substitution you used is part of the
answer.\normalsize

\vspace{0.2cm}
\noindent\fbox{\begin{minipage}{0.965\linewidth}
\small\textbf{The form everything here is written in}\par\vspace{2pt}
\[ y = a(x - h)^2 + k \qquad \text{vertex } (h,\,k),\quad \text{axis of symmetry } x = h \]
The subtraction in $(x - h)$ is part of the \emph{form}, so the vertex $x$ is the value
that makes the parentheses zero --- not the number you see printed. Decide $h$ by asking
``what makes $x - h$ equal $0$?'' A negative $a$ reflects the parabola over the $x$-axis;
it does \emph{not} change which way the graph shifts.
\end{minipage}}

\problem[5cm]{Let $f(x) = (x - 3)^2 + 5$.\\[2pt]
First decide where the vertex is: is it at $x = 3$ or at $x = -3$? Then evaluate $f$ at the
vertex to find the \textbf{minimum value} of $f$. Show the substitution you used.}

\problem[5.5cm]{The graph of $y = x^2$ is translated \textbf{6 units left} and
\textbf{2 units down}.\\[2pt]
Write the equation of the new parabola in vertex form, then evaluate \emph{your} equation
at $x = -6$. What value do you get?}

\problem[5.5cm]{Let $p(x) = (x - 2)^2 - 9$.\\[2pt]
Find both \textbf{$x$-intercepts} of $p$. (Check your signs against the vertex: the two
intercepts must sit the same distance on either side of the axis of symmetry.)}

\problem[6cm]{\textbf{Find and fix.} Jamal is asked for the maximum value of
$y = -(x - 4)^2 + 3$. He writes:
\quoted{``The vertex is at $x = -4$, so the maximum is $-(-4 - 4)^2 + 3 = -61$.''}
Name the sign error in his work, then give the \textbf{correct maximum value}.}

\problem[6cm]{The graph of $y = x^2$ is \textbf{reflected over the $x$-axis},
\textbf{stretched vertically by a factor of 2}, then shifted \textbf{1 unit left} and
\textbf{8 units up}.\\[2pt]
Write the equation of the new parabola, then evaluate \emph{your} equation at $x = 1$.}

\problem[6cm]{\textbf{Find and fix.} Mia writes:
\quoted{``$g(x) = (x + 5)^2 - 4$ is $y = x^2$ shifted 5 units right and 4 units down, so its lowest point is $g(5) = 96$.''}
Mia made one sign error that then produced a wrong minimum. Say which shift she described
incorrectly, then give the \textbf{actual minimum value} of $g$.}

\problem[6.5cm]{The graph of $y = x^2$ is reflected over the $x$-axis, shifted
\textbf{3 units right}, and shifted \textbf{16 units up}.\\[2pt]
Write the equation, then find both \textbf{$x$-intercepts}. Explain in one line why the
reflection does \emph{not} move the axis of symmetry.}

\problem[8cm]{\textbf{Error analysis --- study card.} Ravi made this card for the test.
Exactly one of the three rules is correct.\par\vspace{3pt}
\noindent\cardbox{
\textbf{R1.} In $y = a(x - h)^2 + k$, the graph shifts $h$ units in the direction of the
sign you see: $(x + 4)$ means right 4.\par
\textbf{R2.} $k$ shifts the graph up when $k > 0$ and down when $k < 0$.\par
\textbf{R3.} When $a < 0$, the parabola is reflected, so all its shifts reverse direction
too.
}\par\vspace{3pt}
For \textbf{each} rule: say whether it is right or wrong, explain \emph{why} in terms of
what makes $(x - h)$ equal zero or what $a$ does to the outputs, and rewrite every wrong
rule so it is correct. \noansline}
