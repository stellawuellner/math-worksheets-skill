\wstitleblock{Integer Exponent Laws: Product and Quotient}{Grade 8 --- Expressions and Equations}{}

\noindent\textbf{Product of powers, quotient of powers, zero and negative exponents}

\vspace{0.15cm}
\noindent\textbf{The model.} A power is a count of equal factors. Expanding the
factors is what makes every law on this page obvious, so expand first and state
the law second.

\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\textbf{Power} & \textbf{Expanded form} & \textbf{Factors} \\ \hline
$2^3$ & $2 \cdot 2 \cdot 2$ & three \\ \hline
$2^4$ & $2 \cdot 2 \cdot 2 \cdot 2$ & four \\ \hline
$2^3 \cdot 2^4$ & $(2 \cdot 2 \cdot 2)(2 \cdot 2 \cdot 2 \cdot 2)$ & three $+$ four \\ \hline
\end{tabular}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item Show the expansion or the exponent arithmetic --- not just the answer.
  \item Leave every answer as a single power of the given base unless the
        problem asks for a number.
  \item Bases never change. Only the exponents move.
\end{itemize}

\problem[4cm]{Expand both powers, then rewrite the product as one power of $3$.
\[ 3^2 \cdot 3^3 = (3 \cdot 3)(3 \cdot 3 \cdot 3) = \underline{\hspace{2cm}} \]}

\problem[4cm]{Use the product law. Write the answer as a single power of $x$.
\[ x^4 \cdot x^6 \]}

\problem[4.5cm]{Write the quotient as a single power of $5$. Expanding the top
and the bottom and cancelling pairs is a good way to see why the law works.
\[ \frac{5^7}{5^4} \]}

\problem[4cm]{Write as a single power of $y$.
\[ \frac{y^9}{y^3} \]}

\problem[4.5cm]{Simplify $a^3 \cdot a^2$ first, then substitute $a = 2$ and give
the numerical value.}

\problem[4.5cm]{Simplify completely. The coefficients divide; the powers of $x$
follow the quotient law.
\[ \frac{12x^5}{4x^2} \]}

\problem[5cm]{Two ways of reading the same expression:
\[ \frac{6^4}{6^4} \]
(a) Cancel the identical factors on top and bottom. What number is left?
(b) Apply the quotient law without cancelling: subtract the exponents and write
the power that results. Your two answers must agree --- write the value that
$6^0$ must therefore equal.}

\problem[5cm]{Apply the quotient law to
\[ \frac{4^2}{4^5} \]
even though the top exponent is smaller. Write the result as a power of $4$ with
a negative exponent, then write it again as a fraction with no negative
exponent.}

\problem[5cm]{Decide which expression is larger, or whether they are equal, and
write $<$, $>$ or $=$ in the blank. Simplify each side to a single power of $3$
before you compare --- do not multiply the numbers out.
\[ 3^4 \cdot 3^2 \underline{\hspace{1.5cm}} \frac{3^7}{3^1} \]}

\problem[5.5cm]{Jae simplified $2^5 \cdot 2^3$ by multiplying the exponents,
writing $2^{5 \cdot 3}$, and answered $32768$. Explain in one sentence which law
Jae should have used, then give the correct value of $2^5 \cdot 2^3$.}
