\sstitleblock{Completing Short Algebraic and Segment Proofs}

\noindent\small A proof is a table of \textbf{statements} and
\textbf{reasons}. The statements are the algebra you would have written anyway;
the reasons say what gives you the right to write them. Line 1 is always the
given, and the last line is always what you were asked to prove.
\normalsize

\skillheading{1. Justifying the steps of an equation}
\begin{formulabox}
\textbf{Properties of equality} --- one per move you make:
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=2pt]
  \item add the same amount to both sides: \emph{Addition Property}
  \item take the same amount off both sides: \emph{Subtraction Property}
  \item divide both sides by the same nonzero number: \emph{Division Property}
  \item clear parentheses: \emph{Distributive Property}
  \item swap equal things for each other: \emph{Substitution}
\end{itemize}
Both sides, every time: a move made on one side only is not a property.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Given $4x - 9 = 23$, prove $x = 8$.
\step{Two moves separate $x$ from the given equation, so the proof is two
lines of algebra and the two properties that permit them.}
\step{$4x = 32$ by the Addition Property (add $9$ to both sides); then divide
both sides by $4$ by the Division Property.}
\par
$\ans{x = 8}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Given $3(x + 2) = 21$, prove the value of $x$. Name a
reason for every line.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

\skillheading{2. Segment proofs with the addition postulate}
\begin{formulabox}
\textbf{Segment Addition Postulate:} if $B$ is between $A$ and $C$, then
$AB + BC = AC$. It is the only statement that lets you add segment lengths, so
name it before the algebra starts. For four collinear points it applies twice.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $B$ is between $A$ and $C$, $AB = x + 4$,
$BC = 2x - 1$, $AC = 21$ cm. Find $x$.
\step{Betweenness is given, so the Segment Addition Postulate supplies the
equation before any solving happens.}
\step{$(x+4) + (2x-1) = 21$ gives $3x + 3 = 21$, then $3x = 18$ by the
Subtraction Property and divide both sides by $3$.}
\par
$\ans{x = 6}$ \quad (check: $10 + 11 = 21$ cm)
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $Q$ is between $P$ and $R$, $PQ = 5x$, $QR = 3x + 4$,
$PR = 52$ cm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{3. Midpoints and congruent segments}
\begin{formulabox}
\textbf{Definition of midpoint:} $M$ is the midpoint of $\overline{AB}$ means
$\overline{AM} \cong \overline{MB}$.\\[3pt]
\textbf{Definition of congruent segments:} $\overline{AM} \cong \overline{MB}$
means $AM = MB$.\\[3pt]
Two separate lines in a proof: congruence is about segments, equality is about
their lengths, and only the second one can be solved.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $M$ is the midpoint of $\overline{AB}$ with
$AM = 6x + 1$ and $MB = 4x + 11$ cm. Find $x$.
\step{Midpoint gives congruence and congruence gives equal lengths, so the
equation to solve is $AM = MB$ --- not $AM + MB$.}
\step{$6x + 1 = 4x + 11$ gives $2x + 1 = 11$ by the Subtraction Property, then
$2x = 10$.}
\par
$\ans{x = 5}$ \quad (check: both halves are $31$ cm)
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $N$ is the midpoint of $\overline{XY}$ with
$XN = 7x - 2$ and $NY = 3x + 18$ cm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Combined like terms'' is not a reason when the terms
sit on opposite sides of the equals sign --- that move is the Subtraction
Property, and it has to happen on both sides.
\end{watchoutbox}
