\wstitleblock{Expressions Workshop: Distribute, Combine, Factor}{Grade 6--7 Expressions and Equations}{}

\noindent\textbf{Focus:} Distributing over a sum, combining like terms,
distributing then combining, and factoring out a common factor.

\vspace{4pt}
\noindent Every problem comes with a representation --- an area model, a tape
diagram, or a sorting table --- that stands for the expression printed with it.
Fill the representation in first, then write the simplified expression. Two
expressions are \emph{equivalent} when they give the same value for every
number you could substitute, so a simplified expression must keep every term
you started with.

\vspace{0.2cm}

\problem[3.6cm]{A rectangle is $4$ metres tall. Its base is cut into a piece
$x$ metres long and a piece $7$ metres long, so its total area is $4(x + 7)$
square metres. Write each piece's area in the model, then write the total area
as an expanded expression (no parentheses).
\begin{center}
\begin{tabular}{|c|c|c|}\hline
 & $x$ & $7$ \\\hline
$4$ & \hspace{3cm} & \hspace{3cm} \\[0.9cm]\hline
\end{tabular}
\end{center}}

\problem[3.6cm]{A crew paints $5$ identical rooms. Each room takes $(3x - 2)$
minutes, where $x$ is the number of minutes per wall. The tape diagram shows
the whole job. Write the total time as an expanded expression in minutes.
\begin{center}
\begin{tabular}{|c|c|c|c|c|}\hline
$3x-2$ & $3x-2$ & $3x-2$ & $3x-2$ & $3x-2$ \\[0.5cm]\hline
\end{tabular}
\end{center}}

\problem[3.6cm]{Two boxes of tiles are combined: $7x + 3$ tiles in the first and
$2x + 8$ tiles in the second, where $x$ is the number of tiles in one full row.
Sort the terms into the table, then write the simplified expression for
$7x + 3 + 2x + 8$.
\begin{center}
\begin{tabular}{|p{5.2cm}|p{5.2cm}|}\hline
\textbf{terms with $x$} & \textbf{number terms} \\[0.9cm]\hline
\end{tabular}
\end{center}}

\problem[3.6cm]{Tia has $8$ packs of markers with $m$ markers in each pack, plus
$5$ loose markers. She gives away $3$ whole packs and then receives $2$ more
loose markers. That is $8m + 5 - 3m + 2$ markers. Sort the terms and simplify.
\begin{center}
\begin{tabular}{|p{5.2cm}|p{5.2cm}|}\hline
\textbf{terms with $m$} & \textbf{number terms} \\[0.9cm]\hline
\end{tabular}
\end{center}}

\problem[4cm]{A walking path has $3$ curved sections, each $(x + 4)$ metres long,
followed by a straight run of $2x$ metres. The total length in metres is
$3(x + 4) + 2x$. Fill in the model, then write the simplified total length.
\begin{center}
\begin{tabular}{|c|c|c|c|}\hline
$x+4$ & $x+4$ & $x+4$ & $2x$ \\[0.8cm]\hline
\end{tabular}
\end{center}}

\problem[4cm]{Now run an area model \emph{backwards}. A rectangle is split into
two pieces of area $6x$ square units and $15$ square units, so its total area is
$6x + 15$ square units. Both pieces share the same height. Find the greatest
common factor of $6x$ and $15$, write it as the height, and write the total area
as a product.
\begin{center}
\begin{tabular}{|c|c|c|}\hline
 & \hspace{2.6cm} & \hspace{2.6cm} \\[0.2cm]\hline
\hspace{1cm} & $6x$ & $15$ \\[0.8cm]\hline
\end{tabular}
\end{center}}

\problem[4.2cm]{A club orders $4$ boxes holding $(2x + 3)$ badges each, then
returns $3$ boxes holding $(x - 2)$ badges each. The badges kept are
$4(2x + 3) - 3(x - 2)$. Distribute both products in the table --- watch the
subtraction --- then combine like terms.
\begin{center}
\begin{tabular}{|l|c|c|}\hline
add $4$ groups of $(2x+3)$ & \hspace{3cm} & \hspace{3cm} \\[0.7cm]\hline
subtract $3$ groups of $(x-2)$ & & \\[0.7cm]\hline
\end{tabular}
\end{center}}

\problem[4cm]{A banner starts as a rectangle of area $12n$ square inches. A strip
of area $18$ square inches is cut off, leaving $12n - 18$ square inches. The
rectangle and the strip have the same height. Use the backwards area model to
write $12n - 18$ as a product with the greatest common factor outside.
\begin{center}
\begin{tabular}{|c|c|c|}\hline
 & \hspace{2.6cm} & \hspace{2.6cm} \\[0.2cm]\hline
\hspace{1cm} & $12n$ & $-18$ \\[0.8cm]\hline
\end{tabular}
\end{center}}

\problem[4.4cm]{A tutoring centre charges $5$ dollars per session for each of
$(x + 3)$ students and adds a materials fee of $4x$ dollars, so the cost in
dollars is $5(x + 3) + 4x$.
\begin{center}
\begin{tabular}{|l|c|}\hline
\textbf{Given} & \textbf{Cost in dollars} \\\hline
$5$ dollars per session, $(x+3)$ students & $5(x+3)$ \\[0.4cm]\hline
materials fee & $4x$ \\[0.4cm]\hline
\end{tabular}
\end{center}
(a) Simplify the total cost expression. \ansline \\[0.5cm]
(b) There are $7$ students in the group, so $x = 7$. Find the total cost in
dollars. \ansline}

\problem[4.2cm]{\textbf{Challenge --- find and fix the error.} Jamal simplified

\begin{center}
\begin{tabular}{|l|}\hline
Jamal's work: \quad $3(2x - 5) = 6x - 5$ \\[0.7cm]\hline
\end{tabular}
\end{center}
Say which number he failed to multiply and why the distributive property
