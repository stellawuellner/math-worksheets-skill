\sstitleblock{Separable Differential Equations with Initial Conditions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Separating and integrating}

\begin{formulabox}
\textbf{When is an equation separable?} When the right-hand side factors as (something in $x$) times (something in $y$):
\[ \frac{dy}{dx} = f(x)\,g(y) \quad\Longrightarrow\quad \frac{dy}{g(y)} = f(x)\,dx. \]
Integrate each side in its own variable. \textbf{One} constant of integration is enough --- write $+\,C$ on the $x$ side. The two integrals you will meet most often are $\displaystyle\int \frac{dy}{y} = \ln|y|$ and $\displaystyle\int y\,dy = \tfrac{1}{2}y^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the general solution of $\dfrac{dy}{dx} = 6x^2 y$.
\step{The right side is already a product of $6x^2$ and $y$, so the equation is separable: divide by $y$ and multiply by $dx$ to put each variable with its own differential.}
\step{$\displaystyle\int \frac{dy}{y} = \int 6x^2\,dx$, and the two antiderivatives are $\ln|y|$ and $2x^3$.}

$\Rightarrow\quad \ans{\ln|y| = 2x^3 + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the general solution of $\dfrac{dy}{dx} = 8x^3 y$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\ln|y| = 2x^4 + C}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Finding C from the start value}

\begin{formulabox}
\textbf{Order of operations that saves algebra:} integrate first, \emph{then} substitute the initial values, \emph{then} solve for the constant, \emph{then} rearrange into $y = f(x)$. Exponentiating $\ln|y| = F(x) + C$ gives $y = Ae^{F(x)}$ with $A = e^{C}$, and for a positive initial value $A$ is simply $y$ at the starting point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{dy}{dx} = 4xy$ with $y(0) = 2$, then find $y(1)$.
\step{An initial value is given, so integrate the separated equation and use the starting point to pin the constant before evaluating anything.}
\step{$\displaystyle\int \frac{dy}{y} = \int 4x\,dx$ gives $\ln|y| = 2x^2 + C$, so $y = Ae^{2x^2}$; the condition $y(0) = 2$ forces $A = 2$, hence $y = 2e^{2x^2}$.}

$\Rightarrow\quad y(1) = 2e^{2}$, so $\ans{y(1) \approx 14.778}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{dy}{dx} = 6xy$ with $y(0) = 3$, then find $y(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y(1) \approx 60.257}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Using the particular solution}

\begin{formulabox}
\textbf{Two questions, one solution.} Once $y = f(x)$ is known: to find a \emph{value}, substitute the given input; to find \emph{when} a value occurs, set $f(x)$ equal to it and solve. For $y = Ae^{kx}$ the second question always ends in a logarithm: $Ae^{kx} = B$ gives $x = \frac{1}{k}\ln\!\left(\frac{B}{A}\right)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\dfrac{dy}{dt} = 0.05\,y$ with $y(0) = 40$. When does $y$ reach $100$?
\step{The question asks for a time rather than a value, so solve the particular solution for $t$ instead of substituting into it.}
\step{Separating gives $y = 40e^{0.05t}$; setting $40e^{0.05t} = 100$ gives $e^{0.05t} = 2.5$, so $t = 20\ln 2.5$.}

$\Rightarrow\quad \ans{t \approx 18.326}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\dfrac{dy}{dt} = 0.04\,y$ with $y(0) = 50$. When does $y$ reach $150$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t \approx 27.465}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A constant added \emph{inside} an exponential is not the same as one added outside. From $\ln|y| = F(x) + C$ you get $y = e^{F(x)}e^{C} = Ae^{F(x)}$ --- the constant multiplies, it does not add. Also: substitute the initial values \emph{before} you rearrange, and never write a separate $C$ on each side.
\end{watchoutbox}
