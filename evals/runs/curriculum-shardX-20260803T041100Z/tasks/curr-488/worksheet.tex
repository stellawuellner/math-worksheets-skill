\wstitleblock{Volumes by Known Cross Sections and Shells}{AP Calculus AB/BC \quad Applications of Integration}{}

\begin{center}
\textit{Volumes by integrating a known cross-sectional area, volumes by
cylindrical shells, and choosing between the two}
\end{center}

\noindent\textbf{The two set-ups.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Known cross sections:} $V = \displaystyle\int_a^b A(x)\,dx$, where
$A(x)$ is the area of the slice at $x$. Square: $A = s^2$. Semicircle on a
diameter $d$: $A = \tfrac{\pi}{8}d^2$. Equilateral triangle of side $s$:
$A = \tfrac{\sqrt{3}}{4}s^2$.
\item \textbf{Cylindrical shells} (revolving about the $y$-axis):
$V = \displaystyle\int_a^b 2\pi \, (\text{radius})(\text{height})\,dx$, with
radius $x$ and height the vertical extent of the region at $x$.
\item Give exact answers unless a decimal is asked for, and state the units of
the volume.
\end{itemize}

\vspace{0.1cm}
\noindent\textbf{The shell picture used throughout.} A vertical strip of the
region, at distance $x$ from the axis, sweeps out a thin cylinder when the
region turns. No values are shown here --- use the functions given in each
problem.

\problem{The base of a solid is the region bounded by $y = x$, the
$x$-axis, and the line $x = 4$. Every cross section perpendicular to the
$x$-axis is a square whose side lies in the base.

Set up and evaluate an integral for the volume of the solid.
\par\vspace*{4.5cm}\answerline{cubic units}}

\problem{The base of a solid is the region bounded by $y = \sqrt{x}$,
the $x$-axis, and the line $x = 9$. Every cross section perpendicular to the
$x$-axis is a semicircle whose \emph{diameter} lies in the base.

Find the exact volume of the solid.
\par\vspace*{4.8cm}\answerline{cubic units}}

\problem{The region under $y = x^2$ from $x = 0$ to $x = 3$, above the
$x$-axis, is revolved about the $y$-axis.

Use the method of cylindrical shells to find the exact volume of the solid of
revolution.
\par\vspace*{4.8cm}\answerline{cubic units}}

\problem{A monument sits on a base shaped like the region bounded by
$y = \sqrt{16 - x^2}$ and the $x$-axis. Every cross section perpendicular to
the $x$-axis is a square whose side lies in the base.

Find the exact volume of the monument.
\par\vspace*{4.8cm}\answerline{cubic units}}

\problem{The region under $y = \sqrt{x}$ from $x = 0$ to $x = 4$, above
the $x$-axis, is revolved about the $y$-axis.

Use cylindrical shells to find the exact volume. (Solving for $x$ in terms of
$y$ is exactly the work shells let you skip.)
\par\vspace*{4.8cm}\answerline{cubic units}}

\problem{The base of a solid is the region bounded by $y = 4 - x^2$ and
the $x$-axis. Every cross section perpendicular to the $x$-axis is an
equilateral triangle with one side in the base.

Find the exact volume, then give it to two decimal places.
\par\vspace*{5cm}\answerline{cubic units}}

\problem{The region bounded by $y = 2x$ and $y = x^2$ is revolved about
the $y$-axis.

First find the $x$-values where the two curves meet, then use cylindrical
shells to find the exact volume. Be careful about which curve gives the height
of a shell.
\par\vspace*{6cm}\answerline{cubic units}}

\problem{The region bounded by $y = x^2$, the $y$-axis, and the line
$y = 4$ is revolved about the $y$-axis.

Explain in one or two sentences why cylindrical shells in $x$ is a reasonable
choice for this region, then set up the shell integral and evaluate it exactly.
\par\vspace*{6cm}\answerline{cubic units}}
