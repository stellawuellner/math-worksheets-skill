\sstitleblock{Converting Between Standard Form and Slope-Intercept Form}

\skillheading{1. Standard $\rightarrow$ slope-intercept (standard-to-slope-intercept)}

\begin{formulabox}
\textbf{The two forms:}\quad standard $Ax + By = C$ \qquad slope-intercept $y = mx + b$\\[3pt]
\textbf{Method:} solve for $y$. Move the $x$ term to the right, then divide \emph{every}
term by $B$:
\[ Ax + By = C \;\Longrightarrow\; By = -Ax + C \;\Longrightarrow\; y = -\frac{A}{B}x + \frac{C}{B} \]
So $m = -A/B$ and $b = C/B$ --- but derive it each time; the reason to divide every term is
that dividing only one changes the line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $2x + y = 8$ in slope-intercept form.
\step{The form $y = mx + b$ needs $y$ alone on the left, so undo what is attached to $y$:
here only the $2x$ has to move.}
\step{Subtract $2x$ from both sides: $y = 8 - 2x$, which reorders to
$\ans{y = -2x + 8}$. Slope $-2$, $y$-intercept $8$: from $(0, 8)$ the line drops to $(2, 4)$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $5x + y = 9$ in slope-intercept form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -5x + 9}$}
\end{tryitbox}

\skillheading{2. Slope-intercept $\rightarrow$ standard (slope-intercept-to-standard)}

\begin{formulabox}
\textbf{Standard form wants:} $x$ and $y$ on the left, a number on the right, \textbf{integer}
coefficients, and $A > 0$.\\[3pt]
\textbf{Method:} (1) clear fractions by multiplying every term by the denominator;
(2) move the $x$ term to the left; (3) if $A$ came out negative, multiply the whole equation
by $-1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $y = 4x - 7$ in standard form.
\step{No fractions to clear, so the only work is collecting $x$ and $y$ on the left and
keeping $A$ positive.}
\step{Subtract $4x$: $-4x + y = -7$; multiply by $-1$ so $A > 0$: $\ans{4x - y = 7}$.
Check at $x = 2$: $y = 1$ and $4(2) - 1 = 7$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $y = 6x - 2$ in standard form with $A > 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6x - y = 2}$}
\end{tryitbox}

\skillheading{3. Building the equation from a table or two points (slope-from-representation)}

\begin{formulabox}
\textbf{Slope from two points:} \quad $m = \dfrac{y_2 - y_1}{x_2 - x_1}$ (change in $y$ per
$1$ unit of $x$).\\[3pt]
\textbf{Intercept:} $b$ is the $y$-value when $x = 0$. If the table has no $x = 0$ row,
substitute one known point into $y = mx + b$ and solve for $b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tank drains steadily: $(x, y) = (0, 50)$ and $(4, 38)$ litres.
Write $y = mx + b$.
\step{Two points on a straight line give the slope directly, and one of them already has
$x = 0$, so it hands over the intercept as well.}
\step{$m = \dfrac{38 - 50}{4 - 0} = \dfrac{-12}{4} = -3$ litres per minute, and $b = 50$,
so $\ans{y = -3x + 50}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A candle burns steadily: $(0, 24)$ and $(3, 12)$ centimetres.
Write $y = mx + b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -4x + 24}$}
\end{tryitbox}

\skillheading{4. Converting inside a context (context-conversion)}

\begin{formulabox}
A context usually arrives in \textbf{standard form}, because it counts two things that add to
a total: $(\text{cost}_1)x + (\text{cost}_2)y = \text{total}$.\\[3pt]
Converting to $y = mx + b$ answers two questions at once: $b$ is how much $y$ you need when
$x = 0$, and $m$ is the \textbf{trade-off rate} --- how much $y$ falls for each extra unit of
$x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Tickets cost \$6 (child) and \$3 (adult), and the total is \$60:
$6x + 3y = 60$. Convert and read the intercept.
\step{The question asks how many adult tickets are needed, so make $y$ the subject --- that
is exactly what slope-intercept form gives.}
\step{$3y = 60 - 6x$, divide every term by $3$: $y = -2x + 20$, so with no child tickets
sold, $\ans{20}$ adult tickets are needed, and each child ticket sold replaces $2$ adult
tickets.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $5x + 2y = 50$. Convert to slope-intercept form and state the
$y$-intercept.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -\tfrac{5}{2}x + 25}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Divide \emph{every} term by $B$, not just the $x$ term ---
$4y = 12 - 3x$ becomes $y = 3 - \tfrac{3}{4}x$, never $y = 12 - \tfrac{3}{4}x$. And a
negative $B$ flips both signs: $-6x + 2y = 10$ gives $y = 3x + 5$, not $y = -3x + 5$.
\end{watchoutbox}
