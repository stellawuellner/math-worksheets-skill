\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Divide, then \emph{stop and read the question again}. The same division can have three different answers.
  \item Show the division you used. Write the quotient and the remainder before you answer.
  \item Answer with the number the question actually asks for, and name what it counts.
\end{itemize}

\vspace{0.25cm}

\problem[5cm]{A camp counselor has 63 water bottles to load into coolers. Each cooler holds exactly 8 bottles. How many coolers can she fill completely?}

\problem[5cm]{A baker makes 95 dinner rolls and puts 6 rolls in each bag. She fills as many bags as she can. How many rolls are left over?}

\problem[5cm]{A total of 138 students are going to the science museum. Each bus seats 40 students. How many buses must the school order so that every student has a seat?}

\problem[6.5cm]{A school office has 476 pencils. Each classroom is given a set of 20 pencils.
\begin{itemize}
  \item[(a)] Estimate the number of classrooms by first rounding 476 to the nearest ten. \ansline
  \item[(b)] Now find exactly how many classrooms can be given a full set of 20 pencils. \ansline
\end{itemize}}

\problem[5cm]{A librarian is shelving 1{,}000 books. Each shelf holds 24 books. How many shelves will be completely filled?}

\problem[7cm]{A field trip takes 154 riders. Each van carries 12 riders.
\begin{itemize}
  \item[(a)] How many vans are needed so that everyone gets a ride? \ansline
  \item[(b)] Explain why the trip leader cannot order the number of vans that the plain quotient gives.
\end{itemize}

\problem[6.5cm]{A roll of ribbon is 500 centimetres long. Each bookmark uses a piece 35 centimetres long.
\begin{itemize}
  \item[(a)] How many whole bookmark pieces can be cut? \ansline
  \item[(b)] How much ribbon is left over?
\end{itemize}
\answerline{cm}}

\problem[6.5cm]{A food drive collected 296 cans. The volunteers pack 28 cans into each carton.
\begin{itemize}
  \item[(a)] Estimate the number of cartons by rounding both numbers to the nearest ten. \ansline
  \item[(b)] After every full carton is packed, how many cans are left over? \ansline
\end{itemize}}

\problem[7cm]{A school orders 8 boxes of markers with 144 markers in each box. The markers are then shared equally among 25 classrooms.
\begin{itemize}
  \item[(a)] How many markers does each classroom receive? \ansline
  \item[(b)] How many markers are left over? \ansline
\end{itemize}}

\problem[7.5cm]{A charity has 2{,}000 meal kits to ship. One pallet holds 144 kits.
\begin{itemize}
  \item[(a)] How many pallets will be completely full? \ansline
  \item[(b)] How many pallets must the charity order so that every kit ships? \ansline
  \item[(c)] Explain why the answers to (a) and (b) are different.
\end{itemize}
