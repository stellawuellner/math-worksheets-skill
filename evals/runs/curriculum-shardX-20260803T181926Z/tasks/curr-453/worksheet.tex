\wstitleblock{Finding Parameters That Make Piecewise Functions Continuous}{AP Calculus AB and BC \quad Limits and Continuity}{}

\begin{center}
\textit{Solving for the constant that makes a piecewise function continuous,
filling a removable hole with the limit value, and showing when no value can
work}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item $f$ is continuous at $x = p$ exactly when all three of these hold:
$f(p)$ exists, $\lim_{x \to p} f(x)$ exists, and the two are equal.
\item At a junction between two pieces, ``the limit exists'' means the
left-hand and right-hand limits agree. Set them equal and solve for the
unknown constant.
\item Show the one-sided limits explicitly before you solve. An answer with no
limit statement earns no credit on the AP exam.
\item Give exact values (fractions, not decimals).
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Let
\[
f(x) = \begin{cases} 3x + a, & x < 2 \\[2pt] x^2 + 1, & x \geq 2 \end{cases}
\]
Find the value of $a$ that makes $f$ continuous at $x = 2$.}

\problem[5cm]{Let
\[
f(x) = \begin{cases} \dfrac{x^2 - 9}{x - 3}, & x \neq 3 \\[6pt] k, & x = 3 \end{cases}
\]
Find the value of $k$ that makes $f$ continuous at $x = 3$.}

\problem[5cm]{An oven's temperature, in degrees Celsius, $t$ minutes after it is
switched on is modelled by
\[
T(t) = \begin{cases} 4t + c, & 0 \leq t < 5 \\[2pt] 32 - 2t, & t \geq 5 \end{cases}
\]
Find the value of $c$, in degrees Celsius, that makes $T$ continuous at
$t = 5$.}

\problem[6cm]{Let
\[
f(x) = \begin{cases} x^2, & x \leq 1 \\[2pt] ax + b, & 1 < x < 4 \\[2pt] 10, & x \geq 4 \end{cases}
\]
Find the values of $a$ and $b$ that make $f$ continuous on all of
$(-\infty, \infty)$.}

\problem[6cm]{Let
\[
f(x) = \begin{cases} \dfrac{\sqrt{x + 7} - 3}{x - 2}, & x \neq 2 \\[6pt] k, & x = 2 \end{cases}
\]
Find the exact value of $k$ that makes $f$ continuous at $x = 2$. (Rationalize
the numerator first.)}

\problem[5cm]{Let
\[
f(x) = \begin{cases} \dfrac{e^{4x} - 1}{2x}, & x \neq 0 \\[6pt] k, & x = 0 \end{cases}
\]
Find the value of $k$ that makes $f$ continuous at $x = 0$.}

\problem[7cm]{Let
\[
f(x) = \begin{cases} 2x + a, & x < 3 \\[2pt] x^2 + b, & 3 \leq x < 5 \\[2pt] 4x + 1, & x \geq 5 \end{cases}
\]
Find the values of $a$ and $b$ that make $f$ continuous everywhere. Start with
the junction that involves only one unknown.}

\problem[6cm]{A student claims that setting $k = 5$ makes
\[
f(x) = \begin{cases} \dfrac{1}{x - 3}, & x \neq 3 \\[6pt] k, & x = 3 \end{cases}
\]
continuous at $x = 3$. Using one-sided limits, explain why \emph{no} value of
$k$ makes $f$ continuous there, and name the type of discontinuity at
$x = 3$.}
