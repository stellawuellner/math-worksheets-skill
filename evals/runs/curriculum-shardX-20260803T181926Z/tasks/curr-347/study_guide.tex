\sstitleblock{Compositions of Rigid Motions and Symmetry}

\skillheading{Skill 1 --- Doing a composition one step at a time}

\begin{formulabox}
\textbf{Coordinate rules.} Reflection in the $x$-axis: $(x, y) \to (x, -y)$. \quad Reflection
in the $y$-axis: $(x, y) \to (-x, y)$. \quad Reflection in $y = x$: $(x, y) \to (y, x)$. \quad
Translation by $\langle h, k \rangle$: $(x, y) \to (x + h,\ y + k)$. \quad Rotation about the
origin: $90^\circ$ counterclockwise $(x,y) \to (-y, x)$; $180^\circ$ $(x,y) \to (-x, -y)$;
$90^\circ$ clockwise $(x,y) \to (y, -x)$.\par
\smallskip
\textbf{Order matters.} Apply the first transformation to the original, then apply the second
to \emph{that} image. Swapping the order usually changes the answer, so write $A'$ down before
you go looking for $A''$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect $B(5, 2)$ in the $y$-axis, then translate by
$\langle 1, -3 \rangle$.
\step{Two named steps in a stated order, so use the coordinate rule for each in turn rather
than trying to see the finished position on the grid.}
\step{Reflection in the $y$-axis: $B' = (-5, 2)$. Then add the vector:
$x = -5 + 1$ and $y = 2 + (-3)$.}

$\Rightarrow\quad B'' = (\ans{-4},\ \ans{-1})$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Reflect $D(-2, 6)$ in the $x$-axis, then translate by
$\langle 4, 1 \rangle$.\\[2pt]
\rotatebox{180}{\footnotesize check: $D'' = (\ans{2},\ \ans{-5})$}
\end{tryitbox}

\skillheading{Skill 2 --- Naming the single transformation}

\begin{formulabox}
Two rigid motions done one after the other always add up to a single rigid motion, and it has a
one-step name.\par
\smallskip
\textbf{Two reflections in PARALLEL mirrors} $=$ a translation, perpendicular to the mirrors,
of \emph{twice} the distance between them, in the direction from the first mirror to the
second.\par
\smallskip
\textbf{Two reflections in INTERSECTING mirrors} $=$ a rotation about the point where they
cross, through \emph{twice} the angle between them, turning from the first mirror towards the
second. Perpendicular mirrors therefore give a half turn.\par
\smallskip
Naming one is not finished until you have said the kind and, for a translation, how far and
which way, or, for a rotation, the angle, the direction and the centre.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Reflect in $y = 1$, then in $y = 5$. Name the single transformation.
\step{The two mirrors are parallel, so the answer is a translation --- and the fastest way to
pin it down is to track one convenient point.}
\step{Take the origin: reflecting in $y = 1$ gives $(0, 2)$, and reflecting that in $y = 5$
gives $(0, 8)$. The mirrors are $4$ apart, and the point moved twice that.}

$\Rightarrow\quad$ a translation of $\ans{8}$ units straight up, $\langle 0, 8 \rangle$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Reflect in $x = 2$, then in $x = 5$. Name the single
transformation.\\[2pt]
\rotatebox{180}{\footnotesize check: a translation of $\ans{6}$ units right}
\end{tryitbox}

\skillheading{Skill 3 --- Line and rotational symmetry}

\begin{formulabox}
A figure has a \textbf{symmetry} when a rigid motion maps it exactly onto itself.\par
\smallskip
\textbf{Line symmetry:} a fold line with the figure matching itself on both sides.
\textbf{Rotational symmetry:} a turn about the centre that lands the figure back on itself. Its
\emph{order} is how many such turns there are in a full revolution, and the smallest angle is
$360^\circ \div \text{order}$.\par
\smallskip
A regular $n$-gon has exactly $n$ lines of symmetry and rotational symmetry of order $n$. A
parallelogram that is not a rectangle or rhombus has \emph{no} line of symmetry but still has
order $2$: the half turn about the common midpoint of its diagonals.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What is the smallest angle of rotational symmetry of a regular
pentagon?
\step{Its five vertices are equally spaced around the centre, so the smallest turn that works
is the one carrying each vertex to its neighbour --- one fifth of a full turn.}
\step{$360 \div 5$, and the pentagon also has $5$ lines of symmetry, one through each vertex
and the midpoint of the opposite side.}

$\Rightarrow\quad \ans{72}$ degrees, with rotational symmetry of order $5$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What is the smallest angle of rotational symmetry of a regular
octagon?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$ degrees, order $8$}
\end{tryitbox}
