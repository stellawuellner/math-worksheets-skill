\sstitleblock{Writing Recursive Rules for Arithmetic Sequences}

\skillheading{Skill 1 --- Finding the step and the starting value}

\begin{formulabox}
\textbf{The recursive rule has two parts:} $a_1 = \text{(starting value)}$ and $a_n = a_{n-1} + d$ for $n \ge 2$. Neither part alone describes a sequence.\par
\textbf{Step from any two terms:} $d = \dfrac{a_k - a_j}{k - j}$ --- divide by the number of \emph{jumps}, which is one less than the number of terms involved.\par
\textbf{Walk back to term 1:} $a_1 = a_j - (j-1)d$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An arithmetic sequence has $a_3 = 14$ and $a_7 = 30$. Write its recursive rule.
\step{Two known terms fix the step, and the step walks the sequence back to term 1 --- that is exactly the two pieces the rule needs.}
\step{Terms $3$ and $7$ are four jumps apart, so the step is $\frac{30 - 14}{4}$.}
\step{Term $1$ is two jumps below term $3$: $14 - 2(4)$.}

$d = \ans{4}$, \quad $a_1 = \ans{6}$, \quad so $a_n = a_{n-1} + 4$ for $n \ge 2$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a_2 = 9$ and $a_6 = 29$. Write the recursive rule.\\[2pt]
\rotatebox{180}{\footnotesize check: $d = \ans{5}$, $a_1 = \ans{4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Recursive rules from a situation}

\begin{formulabox}
\textbf{Model:} a constant increase per stage is an arithmetic sequence. Decide what the index $n$ counts (rows, weeks, rungs) and what $a_n$ measures, with units.\par
\textbf{Then it is Skill 1 again:} the two reported stages are two known terms, so the same two formulas give $d$ and $a_1$. Check the rule against a reported stage before using it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A hall gains the same number of chairs each row. Row $3$ holds $22$ chairs and row $7$ holds $34$. Write the recursive rule.
\step{Equal chairs added per row makes the row counts an arithmetic sequence, so treat the two rows as two known terms.}
\step{Rows $3$ and $7$ are four rows apart: $\frac{34 - 22}{4}$ chairs per row.}
\step{Row $1$ is two rows below row $3$: $22 - 2(3)$.}

$d = \ans{3}$ chairs per row, \quad $a_1 = \ans{16}$ chairs, \quad $a_n = a_{n-1} + 3$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Row $2$ holds $15$ chairs and row $5$ holds $27$. Find the step and the first row.\\[2pt]
\rotatebox{180}{\footnotesize check: $d = \ans{4}$, $a_1 = \ans{11}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} count \emph{jumps}, not terms. From term $3$ to term $7$ is four jumps, and reaching term $n$ from term $1$ takes $n-1$ jumps --- the off-by-one here is the most common lost mark on this topic.
\end{watchoutbox}

\skillheading{Skill 3 --- Extending and adding up}

\begin{formulabox}
\textbf{Far-off terms:} applying the step $n-1$ times gives $a_n = a_1 + (n-1)d$, which is the same rule written so you can jump straight to a term instead of listing every one.\par
\textbf{Sum of the first $n$ terms:} $S_n = \dfrac{n}{2}\,(a_1 + a_n)$ --- pair the smallest term with the largest.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $a_1 = 7$ and $a_n = a_{n-1} + 6$. Find $a_5$ and the sum of the first $8$ terms.
\step{Listing eight terms works but invites slips, so use the two shortcut forms the recursive rule generates.}
\step{Term $5$ is four jumps past the start: $7 + 4(6)$.}
\step{Term $8$ is $7 + 7(6) = 49$, so the sum pairs up as $\frac{8}{2}(7 + 49)$.}

$a_5 = \ans{31}$ \quad and \quad $S_8 = \ans{224}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a_1 = 2$ and $a_n = a_{n-1} + 9$. Find $a_5$ and the sum of the first $6$ terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $a_5 = \ans{38}$, $S_6 = \ans{147}$}
\end{tryitbox}
