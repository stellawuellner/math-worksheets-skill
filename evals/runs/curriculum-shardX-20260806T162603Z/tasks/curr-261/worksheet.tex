\wstitleblock{Interpreting the Solution of a System on a Graph}{Algebra 1}{}

\begin{center}
\textit{Reading a solution off a graph, telling one solution from none or infinitely many, and graphing to solve}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item A point is a solution of a system only if it lies on \emph{both} lines at once.
  \item Where a grid is already drawn on, read the point where the lines cross and write it as an
        ordered pair $(x, y)$. Where the grid is empty, graph both lines first.
  \item Two lines can cross once, never (parallel), or everywhere (the same line). Say which.
\end{itemize}
