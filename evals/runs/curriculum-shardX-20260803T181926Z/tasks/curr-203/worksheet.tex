\wstitleblock{Approximating Irrational Roots on a Number Line}{Grade 8 / Pre-Algebra \quad The Number System}{}

\begin{center}
\textit{Trapping a square root between two whole numbers, estimating it to the
nearest tenth, plotting it on a number line, and ordering roots with rational
numbers}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A square root that is not a perfect square is \emph{irrational}: its
decimal never ends and never repeats, so every answer here is an
\textbf{approximation}, written with $\approx$.
\item Trap the root first: find the perfect square just below the radicand and
the perfect square just above it. The root lies between their roots.
\item To get the tenth, test tenths by squaring: if $6.8^2$ is too small and
$6.9^2$ is too big, the root is between them --- take whichever tenth is
closer.
\item Mark each answer on the number line printed with the problem, and write
the numerical answer on the answer blank.
\item No calculators.
\end{itemize}
