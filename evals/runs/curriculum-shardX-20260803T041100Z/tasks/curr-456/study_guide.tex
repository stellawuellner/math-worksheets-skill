\sstitleblock{The Derivative as a Limit and an Instantaneous Rate}

\skillheading{Skill 1 --- The limit definition at a point}

\begin{formulabox}\textbf{Definition:}
\[
f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}
\]
Simplify the quotient \textbf{first}. Substituting $h = 0$ before simplifying
gives $0/0$, which is not a value; cancelling the $h$ is legal because the
limit only uses $h$ near $0$, never $h = 0$.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $f(x) = x^2$; find $f'(2)$.
\step{The quotient will be $0/0$ at $h = 0$, so the job is to cancel the $h$
algebraically before the limit is taken.}
\step{$\dfrac{(2+h)^2 - 4}{h} = \dfrac{4h + h^2}{h} = 4 + h$, and $h \to 0$.}
$\Rightarrow\quad f'(2) = \ans{4}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $f(x) = x^2$; find $f'(5)$ from the
definition.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 2 --- From average rate to instantaneous}

\begin{formulabox}\textbf{Rule:}\ Over a window the quotient is an
\emph{average} rate; the limit as the window shrinks is the
\emph{instantaneous} rate. With only a table of data there is no limit to take,
so estimate with the window \textbf{centred} on the point --- it straddles the
instant instead of leaning to one side.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Distance $8$ m at $t = 1$ s and
$20$ m at $t = 3$ s. Estimate the velocity at $t = 2$.
\step{No formula means no limit is available, so I use the window centred on
$t = 2$ and read its average rate as the estimate.}
\step{$\dfrac{20 - 8}{3 - 1} = \dfrac{12}{2}$.}
$\Rightarrow\quad$ about $\ans{6}$ metres per second
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Distance $22$ m at $t = 2$ s and $50$ m
at $t = 6$ s. Estimate the velocity at $t = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$ metres per second}
\end{tryitbox}

\skillheading{Skill 3 --- Reading a derivative as a rate}

\begin{formulabox}\textbf{Rule:}\ $f'(t)$ is not a number attached to the
function --- it is the rate at which the output changes per unit of input, at
one instant. Its \textbf{units are output units per input unit}, inherited
straight from the difference quotient. The sign says direction: positive means
increasing, negative means decreasing.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $V(t) = t^3 - 3t^2$ gallons after $t$
minutes. Find $V'(t)$ and say what it measures.
\step{The question asks how fast the volume is changing, which is the
derivative, so I differentiate rather than evaluate.}
\step{Term by term: $3t^2$ from $t^3$, and $-6t$ from $-3t^2$.}
$\Rightarrow\quad V'(t) = \ans{3t^2-6t}$ gallons per minute
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $s(t) = 2t^3 - 5t$ metres after $t$
seconds. Find the velocity function $s'(t)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6t^2-5}$ metres per second}
\end{tryitbox}

\skillheading{Skill 4 --- Spotting a limit that is a derivative}

\begin{formulabox}\textbf{Rule:}\ If a limit has the shape
$\lim_{h \to 0} \frac{(\text{something involving } a+h) - (\text{a constant})}{h}$,
read $a$ off the $a+h$ and read $f$ off the operation applied to it; the
constant should equal $f(a)$. Then answer with $f'(a)$ instead of grinding the
algebra.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad
$\displaystyle \lim_{h \to 0} \frac{(4+h)^3 - 64}{h}$
\step{The quotient matches the definition's shape, so instead of expanding the
cube I identify which function and which point it belongs to.}
\step{$a = 4$ and $f(x) = x^3$ (check: $f(4) = 64$), so the limit is $f'(4)$,
and $f'(x) = 3x^2$.}
$\Rightarrow\quad \ans{48}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad
$\displaystyle \lim_{h \to 0} \frac{(2+h)^4 - 16}{h}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{32}$}
\end{tryitbox}
