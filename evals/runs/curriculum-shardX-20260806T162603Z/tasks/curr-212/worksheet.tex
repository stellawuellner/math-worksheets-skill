\wstitleblock{Equations with Brackets and Fractions}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Distributing over a bracket, clearing a denominator, and equations that need both}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Bracket first:} the factor outside multiplies \emph{every} term inside ---
        all of them, not just the first.
  \item \textbf{Fractions next:} multiply \emph{both whole sides} by the least common
        denominator. Every term gets multiplied, including the ones with no fraction.
  \item Then finish as usual: collect the variable terms on one side, undo the constant,
        undo the coefficient.
  \item Substitute your answer back into the original equation to check it.
\end{itemize}
