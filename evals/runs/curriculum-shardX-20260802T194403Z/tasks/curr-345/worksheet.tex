\wstitleblock{Composite Solids, Surface Area, and Density Design}{High School Geometry --- Modeling with Measurement}{}

\noindent\textbf{Directions:} Decide first what each solid is \emph{made of} --- which basic
solids are joined, and which faces or circles disappear where they join. Then choose volume,
surface area, or density. Use $\pi$ from your calculator and round each answer to the
precision asked for. Every figure carries only the values that problem gives you.\par
\vspace{0.15cm}
\noindent{\small\textit{This review interleaves: Composite volume, composite surface area,
density and mass, and reverse design problems.}}\par\vspace{0.1cm}
\noindent{\small\textbf{Reference:} $V_{\text{cyl}} = \pi r^2 h$, \quad
$V_{\text{cone}} = \tfrac{1}{3}\pi r^2 h$, \quad $V_{\text{sphere}} = \tfrac{4}{3}\pi r^3$,
\quad $V_{\text{prism}} = Bh$, \quad
lateral cylinder $= 2\pi r h$, \quad lateral cone $= \pi r \ell$, \quad
sphere surface $= 4\pi r^2$, \quad density $=$ mass $\div$ volume.}

\problem{\textbf{Synthesis challenge.} A concrete bollard is a cylinder topped by a
hemisphere of the same radius. It must contain exactly $0.30$ cubic metres of concrete, and
concrete has density $2400$ kilograms per cubic metre.
\begin{center}
\begin{tikzpicture}[scale=0.38]
  \draw[thick] (-1.2,2.6) arc (180:0:1.2 and 1.2);
  \draw[dashed] (0,2.6) ellipse (1.2 and 0.35);
  \draw[thick] (-1.2,2.6) -- (-1.2,0);
  \draw[thick] (1.2,2.6) -- (1.2,0);
  \draw[thick] (-1.2,0) arc (180:360:1.2 and 0.35);
  \draw[dashed] (1.2,0) arc (0:180:1.2 and 0.35);
  \draw (0,0) -- (1.2,0);
  \node[below=2pt] at (0.6,0) {$r = 0.25$ m};
  \node[right] at (1.25,1.3) {$h = ?$};
\end{tikzpicture}
\end{center}
\begin{enumerate}[label=(\alph*), itemsep=2.7cm, leftmargin=1.2cm]
  \item Option A uses radius $0.25$ m. Find the cylinder height $h$, in metres, that makes
        the total volume exactly $0.30$ cubic metres. Round to the nearest hundredth. \ansline
  \item Find the mass of the finished bollard, in kilograms. \ansline
  \item Option B uses radius $0.30$ m instead, and is also built to hold exactly $0.30$
        cubic metres. Explain what that forces to be true about the two bollards' masses,
        and give one practical reason a designer might still prefer Option B. \ansline
\end{enumerate}\vspace*{0.2cm}}
