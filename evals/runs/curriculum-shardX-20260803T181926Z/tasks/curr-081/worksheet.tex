\wstitleblock{Unit Fractions in Area Models}{Grade 2--3 \quad Fractions as Numbers}{}

\begin{center}
\textit{Naming the unit fraction for one shaded part, checking that the parts
really are equal, and comparing unit fractions drawn in same size shapes}
\end{center}

\noindent\textbf{How to name one shaded part.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item First check that all the parts are the \emph{same size}. If they are not,
the piece has no fraction name yet.
\item Count how many equal parts the whole shape is cut into. That number goes
on the \emph{bottom}.
\item One shaded part means the number on \emph{top} is $1$. A fraction with
$1$ on top is called a \textbf{unit fraction}.
\end{itemize}

  \draw[thick] (4,0) -- (4,1.5);
