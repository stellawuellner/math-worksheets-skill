\ssheader{Exponential vs. Linear Change}
\begin{document}
\sstitleblock{Distinguishing Exponential from Linear Change}

\begin{formulabox}
\textbf{One question decides everything: add or multiply?}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item \textbf{Linear} --- same amount \emph{added} each step. Constant
difference. $f(n) = a + d\,n$.
\item \textbf{Exponential} --- same factor \emph{multiplied} each step. Constant
ratio. $f(n) = a\,r^{\,n}$.
\item Neither row constant? Then it is neither model. ``Not linear'' does not
mean exponential.
\end{itemize}
Language clues: ``per year,'' ``each month it adds'' $\to$ linear. ``doubles,''
``percent of its current value,'' ``times as many'' $\to$ exponential.
\end{formulabox}

\skillheading{1. Decide from a table: differences or ratios}

\begin{formulabox}
Write \emph{both} rows under the table. Subtract consecutive values for the
difference row; divide consecutive values for the ratio row. Whichever row is
constant names the model, and that constant is $d$ or $r$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $f(n)$ is 3, 7, 11, 15 at $n = 0, 1, 2, 3$. Find $f(6)$.
\step{Test both rows before naming a model, since increasing values alone prove
nothing.}
\step{Differences $7-3$, $11-7$, $15-11$ are all 4 --- constant --- so $f$ is
linear with $a = 3$, $d = 4$, and $f(6) = 3 + 4 \cdot 6$.}
\[ f(6) = \ans{27} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $g(n)$ is 2, 6, 18, 54 at $n = 0, 1, 2, 3$. Decide the model,
then find $g(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{486}$}
\end{tryitbox}

\skillheading{2. Write the rule from a description}

\begin{formulabox}
Find the starting value $a$ at step 0. Then read the verb: an amount added is
$d$, a factor multiplied is $r$. ``Doubles'' is $r = 2$; ``grows 20 percent'' is
$r = 1.2$; ``loses 20 percent'' is $r = 0.8$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A count starts at 5 and doubles each step. Find the value at
step 4.
\step{``Doubles'' is a repeated multiplication, so the model is exponential with
$r = 2$ rather than a constant amount added.}
\step{The rule is $f(n) = 5 \cdot 2^{\,n}$, so $f(4) = 5 \cdot 16$.}
\[ f(4) = \ans{80} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A count starts at 7 and triples each step. Find the value at
step 3.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{189}$}
\end{tryitbox}

\skillheading{3. Compare a linear model with an exponential one}

\begin{formulabox}
Early steps are misleading: the linear model is usually ahead at first. Compare
by evaluating both rules at the same input and subtracting. An exponential model
with $r > 1$ always overtakes any linear model eventually.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $L(n) = 20 + 10n$ and $E(n) = 20 \cdot 1.5^{\,n}$. How far
ahead is $E$ at $n = 6$?
\step{Evaluate both rules at the same input and subtract, instead of comparing
their step sizes.}
\step{$E(6) = 20 \cdot 1.5^6 = 227.8125$ and $L(6) = 20 + 60 = 80$, so the gap
is $227.8125 - 80$.}
\[ E(6) - L(6) = \ans{147.81} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} With the same two rules, how far ahead is $E$ at $n = 4$?
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{41.25}$}
\end{tryitbox}

\skillheading{4. Classify a real situation, then compute}

\begin{formulabox}
A percent \emph{of the current amount} is exponential; a fixed quantity per
period is linear. Losing 20 percent a year multiplies by 0.8 each year; losing
20 litres an hour subtracts 20 each hour. Classify before you compute --- the
arithmetic is easy, the choice is where marks are lost.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A 600-dollar phone loses 20 percent of its value each year.
What is it worth after 3 years?
\step{The loss is a percent of the current value, so each year multiplies by
$0.8$ --- exponential decay, not a fixed subtraction.}
\step{$V(n) = 600 \cdot 0.8^{\,n}$, so $V(3) = 600 \cdot 0.512$.}
\[ V(3) = \ans{307.2} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A 600-litre tank drains 20 litres every hour. How much is left
after 3 hours?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{540}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $a \cdot r \cdot n$ is not $a \cdot r^{\,n}$ ---
multiplying by the number of steps is the linear move. And a percent taken from
the \emph{original} amount every period is linear; a percent of the
\emph{current} amount is exponential.
\end{watchoutbox}
