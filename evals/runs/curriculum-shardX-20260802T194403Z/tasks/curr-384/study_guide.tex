\sstitleblock{Composition Order and Inverse Domains}

\skillheading{1. Which function goes first? (composition-order)}

\begin{formulabox}
\textbf{Definition:} $(f \circ g)(x) = f\bigl(g(x)\bigr)$ --- the \textbf{inner}
function $g$ (the one written on the right, closest to $x$) acts
\textbf{first}.\\[3pt]
Read it inside-out, exactly like nested parentheses. Composition is \textbf{not}
commutative: $(f \circ g)(x) \neq (g \circ f)(x)$ for most $f$ and $g$, so one
numerical counterexample is enough to prove they differ.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = x + 3$, $g(x) = 2x$. Find $(f \circ g)(5)$.
\step{The inner function is $g$, so $5$ goes into $g$ first and only that output reaches $f$.}
\step{$g(5) = 2 \cdot 5 = 10$, then $f(10) = 10 + 3 = 13$.}
$(f \circ g)(5) = \ans{13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Same $f(x) = x + 3$ and $g(x) = 2x$: find
$(g \circ f)(5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Substituting the input into the \emph{outer} function
first is the single most common composition error. In $(f \circ g)(4)$ the
number $4$ never touches $f$ --- only $g(4)$ does.
\end{watchoutbox}

\skillheading{2. Proving a proposed inverse really is one}

\begin{formulabox}
\textbf{To build $f^{-1}$:} write $y = f(x)$, swap $x$ and $y$, solve for
$y$.\\[3pt]
\textbf{To verify it, compose BOTH ways:}
$f\bigl(f^{-1}(x)\bigr) = x$ \; \textbf{and} \; $f^{-1}\bigl(f(x)\bigr) = x$.\\
One direction alone can hide a domain restriction, so a single check is never a
proof.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 4x - 6$ with proposed inverse $f^{-1}(x) = \dfrac{x + 6}{4}$.
\step{A proposed inverse earns the name only by composing back to the identity, so substitute the whole expression into $f$.}
\step{$f\bigl(f^{-1}(x)\bigr) = 4 \cdot \dfrac{x + 6}{4} - 6 = (x + 6) - 6$, and the $4$s cancel.}
$f\bigl(f^{-1}(x)\bigr) = \ans{x}$ for every real $x$, so the proposal survives this direction.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = \dfrac{x - 1}{2}$ with proposed inverse
$f^{-1}(x) = 2x + 1$. Simplify $f\bigl(f^{-1}(x)\bigr)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x}$}
\end{tryitbox}

\skillheading{3. The domain is part of the inverse (inverse-domain)}

\begin{formulabox}
\textbf{Rule:} only a \textbf{one-to-one} function has an inverse. $x^2$ is not
one-to-one ($3$ and $-3$ share the output $9$), so an inverse exists only after
the domain is restricted --- and the restriction decides which branch of the
square root you keep.\\[3pt]
\textbf{Swap rule:} domain of $f^{-1}$ $=$ range of $f$, and range of $f^{-1}$
$=$ domain of $f$. Any output of $f^{-1}$ that falls outside the domain of $f$
proves the rule is wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $h(x) = (x - 2)^2$ restricted to $x \ge 2$. Find $h^{-1}(25)$.
\step{Outputs of $h^{-1}$ must land in $h$'s domain $x \ge 2$, so the plus branch of the square root is the only branch that can be right.}
\step{$y = (x-2)^2 \Rightarrow \sqrt{y} = x - 2$ (positive since $x \ge 2$) $\Rightarrow h^{-1}(x) = 2 + \sqrt{x}$, so $h^{-1}(25) = 2 + 5$.}
$h^{-1}(25) = \ans{7}$, and $7 \ge 2$ sits inside the domain $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $k(x) = (x + 1)^2$ restricted to $x \ge -1$. Find
$k^{-1}(36)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}
