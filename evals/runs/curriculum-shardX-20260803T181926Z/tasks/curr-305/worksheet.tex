\wstitleblock{Building a Coherent Proof from the Givens}{High-School Geometry \quad Definitions, Postulates, and Proof Logic}{}

\begin{center}
\textit{Justifying each step of an algebraic solution, building an equation
from a segment argument, building one from an angle argument, and writing a
complete two-column proof}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Every line you write needs a reason. ``Given'', a definition, a
postulate, or a property of equality --- one of those, named.
\item Start from what is \emph{stated}. A diagram can suggest a relationship,
but only the givens and what follows from them may be used as reasons.
\item When a problem asks for a value, show the equation you built \emph{and}
say which definition or postulate licensed it.
\item Angle measures are in degrees.
\end{itemize}

\vspace{0.2cm}

\problem[7cm]{Solve $3x + 7 = 22$, writing one equation per line. Beside each
line, name the property of equality that justifies it.}

\problem[4.5cm]{Point $B$ lies between $A$ and $C$ on $\overline{AC}$, with
$AB = 2x + 3$, $BC = 3x - 1$, and $AC = 27$.

\begin{center}
\begin{tikzpicture}
  \draw[thick] (0,0) -- (7,0);
  \fill (0,0) circle (2pt);
  \fill (2.6,0) circle (2pt);
  \fill (7,0) circle (2pt);
  \node[below=3pt] at (0,0) {$A$};
  \node[below=3pt] at (2.6,0) {$B$};
  \node[below=3pt] at (7,0) {$C$};
\end{tikzpicture}
\end{center}

Name the postulate that lets you combine the three givens into one equation.
Then find $x$, and use it to find $AB$.}

\problem[4.5cm]{$\overrightarrow{BD}$ bisects $\angle ABC$. The two halves
measure $m\angle ABD = (2x + 5)^\circ$ and $m\angle DBC = (4x - 19)^\circ$.

\begin{center}
\begin{tikzpicture}
  \coordinate (B) at (0,0);
  \draw[thick] (B) -- (4,1.7);
  \draw[thick] (B) -- (4,-1.7);
  \draw[thick, dashed] (B) -- (4.2,0);
  \node[right] at (4,1.7) {$A$};
  \node[right] at (4.2,0) {$D$};
  \node[right] at (4,-1.7) {$C$};
  \node[left] at (B) {$B$};
\end{tikzpicture}
\end{center}

State the definition that makes the two expressions equal, find $x$, and then
find $m\angle ABC$.}

\problem[8cm]{Solve $5(x - 2) = 3x + 8$. Write one equation per line, and name
the property that justifies each line. One of your reasons is \emph{not} a
property of equality --- say which line it is on and what it is called
instead.}

\problem[4.5cm]{$\angle AEC$ and $\angle CEB$ form a linear pair, with
$m\angle AEC = (3x + 10)^\circ$ and $m\angle CEB = (x + 30)^\circ$.

\begin{center}
\begin{tikzpicture}
  \draw[thick] (-3.4,0) -- (3.4,0);
  \draw[thick] (0,0) -- (1.6,2.4);
  \fill (0,0) circle (2pt);
  \node[left] at (-3.4,0) {$A$};
  \node[right] at (3.4,0) {$B$};
  \node[above right] at (1.6,2.4) {$C$};
  \node[below=3pt] at (0,0) {$E$};
\end{tikzpicture}
\end{center}

Name the postulate that tells you what the two measures add to, then find $x$
and $m\angle AEC$.}

\problem[8cm]{Point $P$ lies between $Q$ and $R$, with $QP = 4x - 5$,
$PR = 2x + 7$, and $QR = 50$.

Dana started her solution by writing $4x - 5 = 2x + 7$.

\smallskip
(a) Explain, in one or two sentences, why that equation does not follow from
the givens.

\smallskip
(b) Write the equation that \emph{does} follow, name the postulate behind it,
and solve for $x$.}

\problem[8cm]{Lines $m$ and $n$ are parallel and are cut by transversal $t$.
The two marked angles, $\angle p$ and $\angle q$, are same-side interior
angles, with $m\angle p = (5x - 12)^\circ$ and $m\angle q = (3x + 16)^\circ$.

\begin{center}
\begin{tikzpicture}
  \draw[thick, <->] (-3.4,1.2) -- (3.4,1.2) node[right] {$m$};
  \draw[thick, <->] (-3.4,-1.2) -- (3.4,-1.2) node[right] {$n$};
  \draw[thick, <->] (-2.6,-2.4) -- (2.6,2.4) node[above right] {$t$};
  \node[font=\small] at (0.35,1.75) {$p$};
  \node[font=\small] at (-0.35,-1.75) {$q$};
\end{tikzpicture}
\end{center}

Name the theorem that relates $\angle p$ and $\angle q$, then find $x$,
$m\angle p$, and $m\angle q$.}

\problem[2.6cm]{\textbf{Given:} $\overline{AB} \cong \overline{DE}$,
$\overline{BC} \cong \overline{EF}$, and $\angle B \cong \angle E$. \quad
\textbf{Prove:} $\overline{AC} \cong \overline{DF}$.

\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[thick] (0,0) -- (3.2,0) -- (1.1,2.1) -- cycle;
  \node[below left] at (0,0) {$A$};
  \node[below right] at (3.2,0) {$C$};
  \node[above] at (1.1,2.1) {$B$};
  \draw[thick] (5,0) -- (8.2,0) -- (6.1,2.1) -- cycle;
  \node[below left] at (5,0) {$D$};
  \node[below right] at (8.2,0) {$F$};
  \node[above] at (6.1,2.1) {$E$};
\end{tikzpicture}
\end{center}

Write a two-column proof. Your last reason should explain how corresponding
parts follow once the triangles are congruent.

\problem[9cm]{\textbf{Synthesis.} $\overrightarrow{BD}$ is drawn inside
$\angle ABC$ so that $\angle ABD \cong \angle DBC$. It is also given that
$m\angle ABC = (7x - 4)^\circ$ and $m\angle ABD = (2x + 10)^\circ$.

\smallskip
(a) Write a short argument, in complete sentences, that starts from the given
congruence and ends with the equation $7x - 4 = 2(2x + 10)$. Name every
definition or postulate you use, in the order you use it.

\smallskip
(b) Solve your equation for $x$, then find $m\angle ABC$.}
