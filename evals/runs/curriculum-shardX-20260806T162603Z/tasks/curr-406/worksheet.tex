\wstitleblock{Using End Behavior and Multiplicity to Sketch Polynomials}{Precalculus}{}

\begin{center}
\small\textit{Finding zeros and their multiplicity, reading end behavior from the leading term, and building a sketch from both}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A zero with an \emph{odd} multiplicity is a crossing; a zero with an \emph{even} multiplicity is a turning point on the axis.
  \item End behavior is decided by two things only: the degree (odd or even) and the sign of the leading coefficient.
  \item On the sketching problems, plot the zeros and the $y$-intercept first, then join them so both ends match the end behavior. A smooth freehand curve is fine --- exact heights between zeros are not being graded.
\end{itemize}
\vspace{0.15cm}

\problem[5cm]{Factor $p(x) = x^{3} - 4x$ completely.}

\problem[5cm]{For $p(x) = -2x^{3} + 5x - 1$, state $\displaystyle\lim_{x \to \infty} p(x)$.}

\problem[0pt]{Let $p(x) = (x - 1)^{2}(x + 3)$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] List all real zeros of $p$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Say which zero the graph crosses and which it only touches, and give the reason from the exponents.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Let $p(x) = 3x^{4} - x^{2} + 2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] State $\displaystyle\lim_{x \to -\infty} p(x)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] State $\displaystyle\lim_{x \to \infty} p(x)$.
  \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[0pt]{Let $p(x) = (x + 2)(x - 1)^{2}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find the $y$-intercept $p(0)$.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Sketch $y = p(x)$ on the grid. Mark the zeros and the $y$-intercept. \noansline
\end{itemize}
