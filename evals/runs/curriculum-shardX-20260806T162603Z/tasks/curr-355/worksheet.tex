\wstitleblock{Complex Solution Sets}{Algebra 2}{}

\begin{center}
{\small\textit{Solving equations and reporting complete complex solution sets}}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item Report \emph{every} solution, real and non-real. A degree-$n$ equation
    has $n$ solutions counted with multiplicity, so count yours before you stop.
  \item Write non-real solutions in standard form $a + bi$. A pair such as
    $2 + 3i$ and $2 - 3i$ may be written $2 \pm 3i$.
  \item The methods are mixed on purpose: decide for each problem whether to
    isolate a square, use the quadratic formula, or factor first.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Solve over the complex numbers: \quad $x^{2} + 49 = 0$}

\problem[4.5cm]{Solve over the complex numbers: \quad $3x^{2} + 75 = 0$}

\problem[5.5cm]{Solve over the complex numbers: \quad $x^{2} - 4x + 13 = 0$}

\problem[6cm]{Solve over the complex numbers, and list all four solutions:
\quad $x^{4} - 81 = 0$}

\problem{A student solved $x^{2} + 16 = 0$ and reported the solution set
$\{\,4i\,\}$.
\par\vspace{0.15cm}
(a) Give the complete solution set.
\par\vspace*{3cm}\ansline
\par\vspace{0.15cm}
(b) Explain what is missing and why the equation must have that many solutions.
\par\vspace*{2.6cm}\ansline}

\problem[6cm]{Solve over the complex numbers: \quad $2x^{2} + 4x + 10 = 0$}

\problem[5.5cm]{Solve over the complex numbers: \quad $(x - 5)^{2} + 9 = 0$}

\problem[6cm]{Solve over the complex numbers, and list all four solutions:
\quad $x^{4} + 3x^{2} - 4 = 0$}

\problem{Consider the equation $x^{2} - 2x + 5 = 0$.
\par\vspace{0.15cm}
(a) Substitute $x = 1 + 2i$ into the left side and simplify. What value do you get?
\par\vspace*{3.4cm}\ansline
\par\vspace{0.15cm}
(b) Using part (a), write the complete solution set of the equation.
\par\vspace*{2.4cm}\ansline}

\problem[6.5cm]{Find all zeros of \quad $f(x) = x^{3} - 2x^{2} + 9x - 18$.
\ Group the terms in pairs to factor first.}

\problem[6cm]{Solve over the complex numbers: \quad $x^{2} + 6x + 34 = 0$}

\problem{A quadratic with \emph{real} coefficients and leading coefficient $1$
has $4 - i$ as one of its solutions.
\par\vspace{0.15cm}
(a) Write the quadratic in standard form.
\par\vspace*{4cm}\ansline
\par\vspace{0.15cm}
(b) Justify why $4 + i$ must be the other solution.
\par\vspace*{2.6cm}\ansline}
