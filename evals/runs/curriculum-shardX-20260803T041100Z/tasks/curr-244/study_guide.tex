\sstitleblock{Pythagorean Theorem: Which Side Goes Where?}

\skillheading{Skill 1 --- Find the hypotenuse (both legs known)}

\begin{formulabox}
\textbf{Rule:} \quad $a^2 + b^2 = c^2$, and $c$ is \textbf{always} the hypotenuse: the
side opposite the right angle, and the longest side of the triangle.\par
\smallskip
\textbf{Name the hypotenuse before you substitute.} Both givens are legs $\Rightarrow$
\emph{add} the squares and take the square root. The answer must come out
\emph{longer} than either given side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A right triangle has legs of 5 and 12. Find the hypotenuse.
\step{Both given sides touch the right angle, so both are legs and the unknown is $c$ ---
that is the adding direction of the theorem.}
\step{$c = \sqrt{5^2 + 12^2} = 13$, and 13 is longer than both givens, as a hypotenuse
must be.}
$\Rightarrow\quad c = \ans{13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A right triangle has legs of 8 and 15. Find the hypotenuse.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{17}$}
\end{tryitbox}

\skillheading{Skill 2 --- Find a missing leg (hypotenuse known)}

\begin{formulabox}
\textbf{Rule:} \quad when the hypotenuse $c$ is given, the missing leg comes from
\emph{subtracting}: \ $b^2 = c^2 - a^2$.\par
\smallskip
\textbf{Check:} a leg must be \emph{shorter} than the hypotenuse. If your ``leg'' comes
out longer, you added when you should have subtracted --- the hypotenuse landed in a leg
slot.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A right triangle has hypotenuse 25 and one leg 7. Find the other
leg.
\step{The longest side is named as the hypotenuse, so it stands alone on the $c$ side and
the unknown is a leg --- subtract.}
\step{$b = \sqrt{25^2 - 7^2} = 24$. Adding instead would give about 25.96, longer than the
hypotenuse and therefore impossible.}
$\Rightarrow\quad b = \ans{24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A right triangle has hypotenuse 10 and one leg 6. Find the other
leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Distance between two points}

\begin{formulabox}
\textbf{Rule:} the horizontal run and the vertical rise are the two \textbf{legs}; the
segment joining the points is \textbf{always the hypotenuse}, because the run and the
rise meet at a right angle.\par
\smallskip
So a distance question is always the adding direction --- there is no version of it that
subtracts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the distance between $A(-6, 0)$ and $B(0, 8)$.
\step{Sliding across and then up builds a right triangle whose legs are the run and the
rise, so $\overline{AB}$ is its hypotenuse and the squares get added.}
\step{run $= 6$ and rise $= 8$, so $AB = \sqrt{6^2 + 8^2} = 10$.}
$\Rightarrow\quad AB = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the distance between $P(-3, 1)$ and $Q(2, 13)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``Longest number given'' does not mean ``hypotenuse''. Only
position decides: the hypotenuse is the side \emph{across from the right angle}. If a
problem calls two sides legs, the hypotenuse is the thing you are solving for --- add. If
it names the hypotenuse, subtract. Then check the size of your answer before you write it
down.
\end{watchoutbox}
