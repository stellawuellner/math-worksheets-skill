\skillheading{Skill 1 --- Finding the partner in a factor pair}

\begin{formulabox}
\textbf{Rule:} Factors come in pairs. If one factor is known, its partner is the number divided by it.\\[3pt]
number $\div$ known factor $=$ partner factor\\[3pt]
The two sides of a rectangle made from unit squares are always a factor pair of its area.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Fill in the missing factor: $8 \times \underline{\hspace{0.9cm}} = 104$
\step{One factor and the product are known, so the missing factor is the partner --- divide the product by the factor I have.}
\step{$104 \div 8 = 13$, and checking, $8 \times 13 = 104$.}
Missing factor: $\ans{13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Fill in the missing factor: $6 \times \underline{\hspace{0.9cm}} = 84$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\skillheading{Skill 2 --- Digit-sum tests for 3 and for 9}

\begin{formulabox}
\textbf{Rules:} Add the digits.\\[3pt]
divisible by 3 $\Leftrightarrow$ the digit sum is a multiple of 3 \qquad divisible by 9 $\Leftrightarrow$ the digit sum is a multiple of 9\\[3pt]
divisible by 2 $\Leftrightarrow$ the last digit is even \qquad divisible by 6 $\Leftrightarrow$ it passes the test for 2 \emph{and} the test for 3
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which digit goes in the box so that $3\,\square\,5$ is divisible by 9?
\step{Divisibility by 9 depends only on the digit sum, so I add the digits I know and see how far the sum is from a multiple of 9.}
\step{$3 + 5 = 8$, and the next multiple of 9 is 9 itself, so the missing digit is $9 - 8$.}
Missing digit: $\ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which digit goes in the box so that $7\,\square\,2$ is divisible by 9?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\skillheading{Skill 3 --- When two repeating events meet again}

\begin{formulabox}
\textbf{Rule:} Two events that repeat every $m$ and every $n$ steps meet again at the \textbf{least common multiple}.\\[3pt]
List multiples of each and take the first number on both lists. Or use the shared factor: $m \times n \div (\text{factor they share})$.\\[3pt]
Multiplying $m \times n$ always gives \emph{a} meeting time, but usually not the \emph{first} one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Buses leave every 4 minutes and every 6 minutes, starting together. When do they next leave together?
\step{I want the first time that is a multiple of both gaps, so I list multiples of each and look for the first number they share.}
\step{Multiples of 4: 4, 8, 12. Multiples of 6: 6, 12. They share 12, which is also $4 \times 6 \div 2$ because 2 is the factor they share.}
Next together: $\ans{12}$ minutes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two trams leave every 9 minutes and every 15 minutes, starting together. When do they next leave together?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$ minutes}
\end{tryitbox}

\skillheading{Skill 4 --- Reading a number from its prime factors}

\begin{formulabox}
\textbf{Rule:} Split a number until only primes are left. That list is the number's building plan.\\[3pt]
Multiply the whole list back and you get the number; any group you can pull out of the list is a factor of it.\\[3pt]
So dividing the number by the small primes already found leaves the remaining prime factor behind.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write 84 as a product of primes and give its largest prime factor.
\step{To find prime factors I keep splitting off the smallest prime that divides, and the leftover at the end is the largest one.}
\step{$84 = 2 \times 42 = 2 \times 2 \times 21 = 2 \times 2 \times 3 \times 7$, so dividing out $2 \times 2 \times 3$ leaves $84 \div 12$.}
Largest prime factor: $\ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write 165 as a product of primes and give its largest prime factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}
