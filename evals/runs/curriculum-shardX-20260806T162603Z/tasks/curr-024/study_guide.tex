\sstitleblock{Counting On Without Slipping}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Count On the Right Way}

\begin{formulabox}
\textbf{Rule:} Say the start number --- but do not count it! Then hop once
for each one you are adding: \ for $5+2$, say ``5\,'', hop ``6, 7\,''.
The number of hops matches the number being added.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $7 + 2$
\step{Counting on means the hops come after the start number, so we say 7 first without counting it.}
\step{Hop two times: 8, 9 --- two hops for the 2. $\checkmark$}

$\Rightarrow\quad 7 + 2 = \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $6 + 2$ \ (say 6, then hop two times)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Spot the One-Short Mistake}

\begin{formulabox}
\textbf{Rule:} If the start number was counted as a hop, the answer lands
\textbf{one short}. Clue: the written count begins with the start number
itself.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Jo counted $5+3$ as ``5, 6, 7\,'' and wrote 7. What
is the correct sum?
\step{Jo's count starts with 5 itself, the one-short clue, so one hop is missing and we recount.}
\step{Say 5, then hop 6, 7, 8 --- three hops for the 3. $\checkmark$}

$\Rightarrow\quad$ the correct sum is $5 + 3 = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Li counted $4+3$ as ``4, 5, 6\,'' and wrote 6. What
is the correct sum?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Fix It and Check}

\begin{formulabox}
\textbf{Rule:} To check a sum, count on again from the \textbf{bigger}
number, hopping on your fingers. If you land somewhere new, fix the sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A friend says $2+7=8$. Check it and fix it.
\step{We recheck by counting on from the bigger number 7, using fingers to keep the hops honest.}
\step{Say 7, hop 8, 9 --- two hops for the 2, landing on 9, not 8. $\checkmark$}

$\Rightarrow\quad$ the fixed sum is $2 + 7 = \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A friend says $3+5=7$. Count on from 5 and fix the
sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Fingers help! Put up one finger per hop. If you are
adding 3 and you used only 2 fingers, your count started too early.
\end{watchoutbox}
