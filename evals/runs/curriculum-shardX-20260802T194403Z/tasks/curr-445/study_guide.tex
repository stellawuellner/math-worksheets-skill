\sstitleblock{Three Views of a Limit}

\begin{formulabox}
$\displaystyle\lim_{x\to a}f(x)=L$ means: as $x$ gets close to $a$ from both
sides --- \emph{without ever equalling $a$} --- the outputs get close to $L$. The
value $f(a)$ is irrelevant. It exists exactly when
$\lim_{x\to a^-}f(x)=\lim_{x\to a^+}f(x)=L$. Tables give evidence, graphs give a
picture, algebra gives proof.
\end{formulabox}

\skillheading{1. From a table of values}

\begin{formulabox}
Sample inputs closing in on $a$ from \emph{both} sides. If both columns settle
on the same number, that is what the limit should be --- but a table sees only
finitely many inputs, so it can never prove one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A table for $f(x)=\dfrac{x^2-25}{x-5}$ gives $f(4.99)=9.99$
and $f(5.01)=10.01$. Find $\displaystyle\lim_{x\to5}f(x)$.
\step{Both one-sided columns close on the same value, so read the trend instead
of evaluating at the missing point.}
\step{The entries straddle $10$ and tighten toward it; cancelling
$\dfrac{(x-5)(x+5)}{x-5}=x+5$ confirms it.}
$\Rightarrow\quad \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A table for $f(x)=\dfrac{x^2-49}{x-7}$ gives $f(6.99)=13.99$
and $f(7.01)=14.01$. Find $\displaystyle\lim_{x\to7}f(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

\skillheading{2. From algebra}

\begin{formulabox}
Substitute first: a number \emph{is} the limit. $\tfrac{0}{0}$ means a hidden
common factor --- rewrite, then substitute. Factor and cancel
($a^2-b^2=(a-b)(a+b)$, $a^3-b^3=(a-b)(a^2+ab+b^2)$); use the conjugate when a
root causes it; divide by the highest power when $x\to\infty$. Cancelling
changes the function at one point, never the limit.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $\displaystyle\lim_{x\to3}\frac{x^2-x-6}{x-3}$
\step{Substitution gives $\tfrac{0}{0}$, a signal to rewrite rather than an
answer, so factor and hunt for the common factor.}
\step{$\dfrac{(x-3)(x+2)}{x-3}=x+2$ for $x\neq3$, and at $x=3$ that is $3+2$.}
$\Rightarrow\quad \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $\displaystyle\lim_{x\to3}\frac{x^3-27}{x-3}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{27}$}
\end{tryitbox}

\skillheading{3. From a graph}

\begin{formulabox}
Trace the curve toward $x=a$ once from each side and read the \emph{height} you
head for, not the dot. An open circle means the point is missing (the limit can
still exist); a jump between the sides means the two-sided limit does not exist.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} From the graph of $q$, find
$\displaystyle\lim_{x\to2^-}q(x)$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[width=6.6cm, height=3.5cm, axis lines=middle,
  xmin=-0.4, xmax=4.4, ymin=-0.4, ymax=3.9,
  xtick={1,2,3,4}, ytick={1,2,3},
  tick label style={font=\scriptsize}, yticklabel style={fill=white, inner sep=1pt},
  samples=2, enlargelimits=false]
  \addplot[thick, blue, domain=-0.4:2] {x-1};
  \addplot[thick, blue, domain=2:4.4] {5-x};
  % tikz paths, not an only-marks plot --- a marks plot layered on a function
  % plot makes the engine typeset a stray 0.1pt in nullfont (fatal "Missing
  % character" faults). See references/latex-templates.md, "Completed graph".
  \draw[blue, thick, fill=white] (axis cs:2,1) circle (2.2pt);
  \fill[blue] (axis cs:2,3) circle (2.2pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{The minus sign asks only about inputs left of $2$, so follow the left
branch and ignore the dot on the right.}
\step{That branch is the line $y=x-1$, climbing to the open circle at height
$1$.}
$\Rightarrow\quad \ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} From the same graph, find
$\displaystyle\lim_{x\to2^+}q(x)$. Do the two sides agree?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ --- no, so the two-sided limit does not exist}
\end{tryitbox}

\skillheading{4. A series sum is a limit}

\begin{formulabox}
An infinite sum is \emph{defined} as the limit of its partial sums:
$\sum_{n=0}^{\infty}a_n=\lim_{k\to\infty}S_k$ with $S_k=a_0+\cdots+a_k$.
Geometric ($a_n=ar^n$) has $S_k=a\frac{1-r^{k+1}}{1-r}$, so for $|r|<1$ the
$r^{k+1}$ term dies and $\sum_{n=0}^{\infty}ar^n=\frac{a}{1-r}$; for $|r|\ge1$
the limit fails and the series diverges.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate
$\displaystyle\sum_{n=0}^{\infty}5\left(\tfrac{1}{3}\right)^{n}$.
\step{Each term is the previous one times a fixed ratio, so it is geometric and
the partial-sum limit has a closed form.}
\step{$a=5$, $r=\tfrac{1}{3}$, $|r|<1$, so
$S=\dfrac{5}{1-\frac{1}{3}}=\dfrac{5}{\ \frac{2}{3}\ }$.}
$\Rightarrow\quad \ans{\tfrac{15}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate
$\displaystyle\sum_{n=0}^{\infty}4\left(\tfrac{1}{5}\right)^{n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}
