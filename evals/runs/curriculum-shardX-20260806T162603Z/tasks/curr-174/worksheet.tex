\wstitleblock{Expressions: Find the Distribution \& Like-Term Errors}{Grade 6--7 Mathematics}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Distribute to \emph{every} term, and combine only \emph{like} terms.
  \item Several problems show another student's work with a mistake in it ---
  your job is to find the mistake and fix it.
  \item When you test two expressions with a number, remember: equivalent
  expressions agree for \emph{every} value of the variable.
\end{itemize}

\problem[2.6cm]{Combine like terms: \quad $5x + 3x$}

\problem[2.6cm]{Expand: \quad $6(x + 4)$}

\problem[3.2cm]{Combine like terms: \quad $4x + 7 + 3x - 2$}

\problem{Consider the two expressions $3(x + 2)$ and $3x + 2$.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Evaluate $3(x + 2)$ when $x = 5$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(b)] Evaluate $3x + 2$ when $x = 5$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(c)] Are the two expressions equivalent? Say how you know.\par\vspace*{1.4cm}
\end{itemize}}

\problem{Deshawn evaluates $4(x + 3)$ when $x = 2$. He writes:
\[ 4(x + 3) = 4 \cdot 2 + 3 = 11. \]
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] What error did Deshawn make? Answer in one sentence.\par\vspace*{1.5cm}
  \item[(b)] Find the correct value of $4(x + 3)$ when $x = 2$.\par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem{Ava simplifies $6x - 4x$ and writes:
\[ 6x - 4x = 2 \qquad \text{``because the $x$'s cancel.''} \]
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] What error did Ava make? Answer in one sentence.\par\vspace*{1.5cm}
  \item[(b)] Find the value of $6x - 4x$ when $x = 7$.\par\vspace*{1.8cm}\ansline
\end{itemize}}

\problem[3.4cm]{Expand: \quad $-3(2x - 5)$}

\problem{\textbf{Challenge.} Jaylen claims that $2(3x + 4) = 6x + 4$.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Evaluate $2(3x + 4)$ when $x = 5$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(b)] Evaluate $6x + 4$ when $x = 5$: \quad \ansblank\par\vspace*{1.3cm}
  \item[(c)] Explain what Jaylen forgot when he distributed, and write the
  correct expansion.\par\vspace*{1.6cm}
\end{itemize}}
