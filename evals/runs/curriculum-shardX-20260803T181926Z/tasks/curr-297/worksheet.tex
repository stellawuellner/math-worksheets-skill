\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Every function on this sheet is built from the parent function $f(x) = |x|$ by shifts, reflections, and stretches. Vertex form is $g(x) = a\,|x - h| + k$.
  \item Show the work that gets you to each answer, then write the final answer on the answer line.
  \item On sketching problems, plot the vertex first, then one point on each arm of the V.
\end{itemize}

\vspace{0.2cm}

\problem[4cm]{Describe in words how the graph of $g(x) = |x + 3| - 5$ is obtained from the graph of $f(x) = |x|$, then give the vertex of $g$.}

\problem[4.5cm]{For $h(x) = -2\,|x - 1| + 6$: give the vertex, say whether the V opens up or down, and say whether the graph is narrower or wider than $y = |x|$.}

\problem[6cm]{A drone flies straight along a marked trail. Its distance from the trailhead, in kilometres, after $t$ minutes is $d(t) = |2t - 6| + 1$.
\par\vspace{0.15cm}
(a) Rewrite $d(t)$ in the vertex form $a\,|t - h| + k$ and give the vertex.
\par\vspace{0.1cm}
(b) Explain what the vertex means for the drone.
\par\vspace{0.1cm}
(c) Find every time $t$ at which the drone is $5$ kilometres from the trailhead.}
