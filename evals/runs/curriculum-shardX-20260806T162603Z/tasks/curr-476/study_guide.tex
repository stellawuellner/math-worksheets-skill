\sstitleblock{Substitution: the Chain Rule, Read Backwards}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Spotting the Inner Function}

\begin{formulabox}
\textbf{Look for a pair.} An inner function and a constant multiple of its
derivative, both in the same integrand. The inner function becomes $u$.\par
Build the three-row table every time: $u = \ldots$, then
$du = \ldots\,dx$, then the integral rewritten with no $x$ left in it. If an $x$
survives, $u$ was the wrong choice.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int 3x^2\left(x^3+2\right)^4dx$.
\step{The bracket is an inner function and the factor in front is exactly its derivative, which is the signal to substitute rather than expand.}
\step{$u = x^3+2$, $du = 3x^2\,dx$, so the integral is $\int u^4\,du = \tfrac{u^5}{5} + C$.}

$\Rightarrow\quad \ans{\tfrac{(x^3+2)^5}{5} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int 4x^3\left(x^4+1\right)^2dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{(x^4+1)^3}{3} + C}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Indefinite: Convert Back to x}

\begin{formulabox}
\textbf{Missing constant?} If $du$ is a multiple of what you have, solve for it:
$du = 2x\,dx$ means $x\,dx = \tfrac{1}{2}du$, and that $\tfrac{1}{2}$ rides
outside the integral.\par
An indefinite answer must come back to $x$ and carry $+\,C$. Differentiate it
mentally as a check --- the chain rule should hand back the integrand.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int x\sqrt{x^2+4}\;dx$.
\step{The radicand is the inner function and $x$ is half its derivative, so substitute and carry the missing one half outside.}
\step{$u = x^2+4$, $x\,dx = \tfrac{1}{2}du$, so $\tfrac{1}{2}\int u^{1/2}du = \tfrac{1}{3}u^{3/2} + C$.}

$\Rightarrow\quad \ans{\tfrac{(x^2+4)^{3/2}}{3} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int x\sqrt{x^2+1}\;dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{(x^2+1)^{3/2}}{3} + C}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Definite: Convert the Limits}

\begin{formulabox}
\textbf{Rule.} In a definite integral, move the limits into the new variable at
the same moment the integrand moves:
\[ \int_a^b f(g(x))g'(x)\,dx = \int_{g(a)}^{g(b)} f(u)\,du . \]
Then evaluate in $u$ and stop --- there is nothing to convert back, and no
$+\,C$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int_0^1 3x^2\left(x^3+1\right)^2dx$.
\step{This is a definite integral, so converting the limits is faster and safer than converting the antiderivative back to $x$.}
\step{$u = x^3+1$ runs from $1$ to $2$, so the value is $\left[\tfrac{u^3}{3}\right]_1^2 = \tfrac{8-1}{3}$.}

$\Rightarrow\quad \ans{\tfrac{7}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_0^2 2x\left(x^2+1\right)^2dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{124}{3}}$}
\end{tryitbox}
