\sstitleblock{Intervals of Convergence, Endpoint by Endpoint}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The radius from the coefficients}

\begin{formulabox}
\textbf{Rule:} for $\sum c_{n}(x-a)^{n}$, the radius is $R = \lim_{n\to\infty}\left|\dfrac{c_{n}}{c_{n+1}}\right|$. Convergence is certain on $(a-R,\ a+R)$ and impossible outside it. Two special cases: $R = \infty$ means every real $x$ works, and $R = 0$ means only $x = a$ does. Neither has endpoints to test.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the radius of convergence of $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n\,4^{n}}$.
\step{The coefficients are $c_{n} = 1/(n4^{n})$, so the radius comes straight from the limit of their ratio --- no need to carry $x$ through the ratio test.}
\step{$\left|\dfrac{c_{n}}{c_{n+1}}\right| = \dfrac{(n+1)4^{n+1}}{n\,4^{n}} = \dfrac{4(n+1)}{n}$, whose limit is the constant part.}

$\Rightarrow\quad \ans{R = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the radius of convergence of $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n\,7^{n}}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{R = 7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Testing an endpoint}

\begin{formulabox}
\textbf{Substitute the endpoint back into the original series} and test the ordinary series you get. The ratio test is useless here --- it returns $1$ at every endpoint by construction. The three you will meet again and again: $\sum \frac1n$ diverges, $\sum \frac{(-1)^n}{n}$ converges, and $\sum \frac{1}{n^p}$ converges exactly when $p>1$. The two endpoints of one series can easily disagree.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n}$ (radius $1$, centre $0$), test the endpoint $x = -1$.
\step{Substituting the endpoint turns the power series into an ordinary series, and the alternating signs point at the alternating series test.}
\step{At $x = -1$ the series is $\sum \frac{(-1)^{n}}{n}$: the sizes $\frac1n$ decrease to zero, so it converges, and its value is known.}

$\Rightarrow\quad \ans{-\ln 2 \approx -0.693}$, so this endpoint is included
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\displaystyle\sum_{n=1}^{\infty}\frac{x^{n}}{n^{2}}$, test the endpoint $x = 1$ and give the sum.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{\pi^{2}}{6}}$, so this endpoint is included}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- When the variable carries a coefficient}

\begin{formulabox}
\textbf{Factor the constant out of the shifted power first.} A series in $(kx - b)^{n}$ is a series in $(x - a)^{n}$ with $a = b/k$ and coefficients multiplied by $k^{n}$ --- so its radius is $k$ times smaller than a careless reading suggests. Then finish as usual: endpoints at $a \pm R$, each tested on its own, square bracket for a convergent endpoint and round for a divergent one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the radius of convergence of $\displaystyle\sum_{n=1}^{\infty}\frac{(2x-8)^{n}}{n}$.
\step{Rewrite $(2x-8)^{n}$ as $2^{n}(x-4)^{n}$ so the series is visibly centred at $4$, with coefficients $c_{n} = 2^{n}/n$.}
\step{$\left|\dfrac{c_{n}}{c_{n+1}}\right| = \dfrac{2^{n}}{n}\cdot\dfrac{n+1}{2^{n+1}} = \dfrac{n+1}{2n}$, whose limit is the reciprocal of the factored constant.}

$\Rightarrow\quad \ans{R = \frac{1}{2}}$, so the interval is centred at $4$ with half-width one half
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the radius of convergence of $\displaystyle\sum_{n=1}^{\infty}\frac{(4x-12)^{n}}{n}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{R = \frac{1}{4}}$}
\end{tryitbox}
