\sstitleblock{Finding and Verifying Inverse Functions}

\skillheading{Skill 1 --- Inverting a linear rule}

\begin{formulabox}
\textbf{The method (works for every rule on this sheet):} start from $x = f(y)$, then solve for $y$. Whatever you get is $f^{-1}(x)$.\par
\textbf{Why it works:} $f$ sends inputs to outputs; the inverse asks the reverse question, so the old output $x$ becomes the new input and the old input $y$ becomes the unknown.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the inverse rule for $f(x) = 4x + 9$.
\step{The rule multiplies by $4$ and then adds $9$, so the inverse must subtract $9$ and then divide by $4$ --- write $x = f(y)$ and solve for $y$ to get that automatically.}
\step{$x = 4y + 9 \;\Rightarrow\; x - 9 = 4y \;\Rightarrow\; y = \dfrac{x - 9}{4}$}

$\Rightarrow\quad \ans{f^{-1}(x) = \tfrac{x - 9}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the inverse rule for $h(x) = 6x - 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h^{-1}(x) = \tfrac{x + 5}{6}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Inverting a rational rule}

\begin{formulabox}
\textbf{Extra step:} when $y$ appears in a denominator, clear the fraction first, then gather \emph{every} $y$ term on one side and factor $y$ out.\par
\textbf{Pattern:} $x(\text{denominator in } y) = \text{numerator in } y \;\Rightarrow\; y \cdot (\ldots) = (\ldots) \;\Rightarrow\; y = \dfrac{\ldots}{\ldots}$
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the inverse rule for $g(x) = \dfrac{3x}{x + 2}$.
\step{The unknown sits in both the numerator and the denominator, so no amount of peeling operations will work --- clear the fraction and factor $y$ out instead.}
\step{$x(y + 2) = 3y \;\Rightarrow\; xy + 2x = 3y \;\Rightarrow\; xy - 3y = -2x \;\Rightarrow\; y(x - 3) = -2x$}

$\Rightarrow\quad \ans{g^{-1}(x) = \tfrac{2x}{3 - x}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the inverse rule for $k(x) = \dfrac{x - 1}{x + 5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k^{-1}(x) = \tfrac{5x + 1}{1 - x}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\dfrac{3x}{x+2}$ does \emph{not} invert to $\dfrac{x+2}{3x}$. The reciprocal of a rule is not its inverse.
\end{watchoutbox}

\skillheading{Skill 3 --- Inverting roots and powers}

\begin{formulabox}
\textbf{Undo a square root by squaring; undo a square by taking a root.} For $f(x) = \sqrt{x - h} + k$ the inverse is $(x - k)^2 + h$; for $f(x) = (x - h)^3 + k$ it is $\sqrt[3]{x - k} + h$.\par
\textbf{One-to-one first:} an even power needs a restricted domain (such as $x \ge h$) before it has an inverse; an odd power never does.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the inverse rule for $p(x) = \sqrt{x - 5} + 2$.
\step{The last operation applied to the input is ``add $2$'', so the first thing the inverse must undo is that $2$ --- isolate the radical, then square.}
\step{$x = \sqrt{y - 5} + 2 \;\Rightarrow\; x - 2 = \sqrt{y - 5} \;\Rightarrow\; (x - 2)^2 = y - 5$}

$\Rightarrow\quad \ans{p^{-1}(x) = (x - 2)^2 + 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the inverse rule for $q(x) = \sqrt{x + 3} - 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{q^{-1}(x) = (x + 1)^2 - 3}$}
\end{tryitbox}

\skillheading{Skill 4 --- Checking and using an inverse}

\begin{formulabox}
\textbf{The check:} a rule $f^{-1}$ is the inverse of $f$ exactly when $f\bigl(f^{-1}(x)\bigr) = x$ and $f^{-1}\bigl(f(x)\bigr) = x$. Simplify the composition; if it collapses to $x$, the rule is right.\par
\textbf{Shortcut for a value:} $f^{-1}(c)$ is the input that $f$ sends to $c$, so you can also just solve $f(x) = c$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $m(x) = 3x - 12$, check that $m^{-1}(x) = \dfrac{x + 12}{3}$ and then find $m^{-1}(9)$.
\step{Two separate jobs: composition settles whether the rule is right, and substitution answers the numerical question --- do the composition first so the value can be trusted.}
\step{$m\bigl(m^{-1}(x)\bigr) = 3\left(\dfrac{x + 12}{3}\right) - 12 = (x + 12) - 12 = x$ $\checkmark$}
\step{$m^{-1}(9) = \dfrac{9 + 12}{3} = \dfrac{21}{3}$}

$\Rightarrow\quad \ans{m^{-1}(9) = 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $n(x) = 5x + 4$, find $n^{-1}(24)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n^{-1}(24) = 4}$}
\end{tryitbox}
