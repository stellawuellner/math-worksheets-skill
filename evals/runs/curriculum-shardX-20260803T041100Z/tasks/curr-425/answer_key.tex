\problem{\textbf{Simplify $\dfrac{1 - \cos^2 x}{\sin x}$.}\\
\textbf{Step 1.} The numerator is one of the two rearrangements of the
Pythagorean identity, so replace it rather than expanding anything.\\
\textbf{Step 2.} $1 - \cos^2 x = \sin^2 x$, so the expression is
$\dfrac{\sin^2 x}{\sin x}$.\\
\textbf{Step 3.} Cancel one factor of $\sin x$: $\ans{\sin x}$
(valid where $\sin x \neq 0$).}

\problem{\textbf{Simplify $\sec x - \sec x \sin^2 x$.}\\
\textbf{Step 1.} Both terms share $\sec x$; factoring first exposes a
Pythagorean form instead of leaving two unrelated terms.\\
\textbf{Step 2.} $\sec x(1 - \sin^2 x) = \sec x \cos^2 x$.\\
\textbf{Step 3.} $\sec x = \dfrac{1}{\cos x}$, so
$\dfrac{\cos^2 x}{\cos x} = \ans{\cos x}$.}

\problem{\textbf{Rewrite $\cos 2x$ in terms of $\sin x$ alone.}\\
\textbf{Step 1.} There are three forms of the cosine double angle. Choosing is
the whole point: pick the one whose surviving function is the one you were
asked for.\\
\textbf{Step 2.} From $\cos 2x = \cos^2 x - \sin^2 x$, substitute
$\cos^2 x = 1 - \sin^2 x$: \; $(1 - \sin^2 x) - \sin^2 x$.\\
\textbf{Step 3.} $\cos 2x = \ans{1 - 2\sin^2 x}$}

\problem{\textbf{Solve $2\sin x - 1 = 0$ on $[0, 2\pi)$.}\\
\textbf{Step 1.} One trigonometric function, first power --- isolate it and
read the unit circle; no identity is needed here.\\
\textbf{Step 2.} $\sin x = \dfrac{1}{2}$.\\
\textbf{Step 3.} Sine is positive in quadrants I and II. The reference angle is
$\dfrac{\pi}{6}$, so the quadrant II partner is
$\pi - \dfrac{\pi}{6} = \dfrac{5\pi}{6}$.\\
\textbf{Step 4.} $\ans{x = \dfrac{\pi}{6}, \; \dfrac{5\pi}{6}}$}

\problem{\textbf{$\sin\theta = \tfrac{3}{5}$ in quadrant I; find $\sin 2\theta$.}\\
\textbf{Step 1.} A double angle with both $\sin\theta$ and $\cos\theta$ known
calls for $\sin 2\theta = 2\sin\theta\cos\theta$ --- no quadrant work is left,
since quadrant I makes both values positive.\\
\textbf{Step 2.} $2 \left(\dfrac{3}{5}\right)\left(\dfrac{4}{5}\right)
= \dfrac{24}{25}$.\\
\textbf{Step 3.} $\sin 2\theta = \ans{\,\dfrac{24}{25}}$\\
\textit{If they got $1.2$: doubled the sine itself. $\sin 2\theta$ is never
$2\sin\theta$ --- that value even exceeds the range of sine.}}

\problem{\textbf{Simplify $\tan x \cos x + \sin x$.}\\
\textbf{Step 1.} A lone tangent among sines and cosines is the signal to
rewrite it as a quotient; nothing else in the expression needs touching.\\
\textbf{Step 2.} $\tan x \cos x = \dfrac{\sin x}{\cos x}\cos x = \sin x$.\\
\textbf{Step 3.} The expression is $\sin x + \sin x = \ans{2\sin x}$.}

\problem{\textbf{Solve $2\cos^2 x - \cos x - 1 = 0$ on $[0, 2\pi)$.}\\
\textbf{Step 1.} One function appears, but squared --- so this is a quadratic
in $\cos x$. Factor rather than isolating.\\
\textbf{Step 2.} $(2\cos x + 1)(\cos x - 1) = 0$, giving
$\cos x = -\dfrac{1}{2}$ or $\cos x = 1$.\\
\textbf{Step 3.} $\cos x = -\dfrac{1}{2}$ has reference angle
$\dfrac{\pi}{3}$ and is negative in quadrants II and III, giving
$\dfrac{2\pi}{3}$ and $\dfrac{4\pi}{3}$. \; $\cos x = 1$ gives $x = 0$.\\
\textbf{Step 4.} $\ans{x = 0, \; \dfrac{2\pi}{3}, \; \dfrac{4\pi}{3}}$}

\problem{\textbf{Simplify $\dfrac{\sin 2x}{1 + \cos 2x}$.}\\
\textbf{Step 1.} Both double angles must go, and the denominator decides which
cosine form to use: $\cos 2x = 2\cos^2 x - 1$ is the one that makes
$1 + \cos 2x$ collapse.\\
\textbf{Step 2.} Numerator: $\sin 2x = 2\sin x \cos x$. Denominator:
$1 + (2\cos^2 x - 1) = 2\cos^2 x$.\\
\textbf{Step 3.} $\dfrac{2\sin x\cos x}{2\cos^2 x}
= \dfrac{\sin x}{\cos x} = \ans{\tan x}$}

\problem{\textbf{Combine $\dfrac{1}{1 - \sin x} + \dfrac{1}{1 + \sin x}$.}\\
\textbf{Step 1.} The two denominators are conjugates, so their product is a
difference of squares --- which is exactly the shape the Pythagorean identity
simplifies.\\
\textbf{Step 2.} Common denominator $(1 - \sin x)(1 + \sin x) =
1 - \sin^2 x = \cos^2 x$; the numerators add to
$(1 + \sin x) + (1 - \sin x) = 2$.\\
\textbf{Step 3.} $\dfrac{2}{\cos^2 x} = \ans{2\sec^2 x}$}

\problem{\textbf{Solve $\sin 2x = \sin x$ on $[0, 2\pi)$.}\\
\textbf{Step 1.} The two sides carry different angles, so expand the double
angle first. Then move everything to one side --- dividing by $\sin x$ would
destroy the solutions where $\sin x = 0$.\\
\textbf{Step 2.} $2\sin x\cos x - \sin x = 0$, so
$\sin x(2\cos x - 1) = 0$.\\
\textbf{Step 3.} $\sin x = 0$ gives $x = 0$ and $x = \pi$. \;
$\cos x = \dfrac{1}{2}$ gives $x = \dfrac{\pi}{3}$ and
$x = \dfrac{5\pi}{3}$.\\
\textbf{Step 4.} $\ans{x = 0, \; \dfrac{\pi}{3}, \; \pi, \; \dfrac{5\pi}{3}}$}

\problem{\textbf{$\cos\theta = -\tfrac{5}{13}$ in quadrant II; find
$\cos 2\theta$.}\\
\textbf{Step 1.} Only the cosine is known, so choose the double-angle form
built from cosine alone: $\cos 2\theta = 2\cos^2\theta - 1$. The quadrant does
not matter here, because the cosine is squared.\\
\textbf{Step 2.} $2\left(-\dfrac{5}{13}\right)^{2} - 1
= 2 \cdot \dfrac{25}{169} - 1 = \dfrac{50}{169} - \dfrac{169}{169}$.\\
\textbf{Step 3.} $\cos 2\theta = \ans{\,\dfrac{-119}{169}}$\\
\textit{If they got $-0.77$: doubled the cosine instead of squaring it first.}}

\problem{\textbf{Challenge: solve $2\cos^2 x + \sin x = 1$ on $[0, 2\pi)$.}\\
\textbf{Step 1.} Two different functions appear and one is squared, so the
first move is an identity that leaves a single function --- replace
$\cos^2 x$ with $1 - \sin^2 x$ so everything is in sine.\\
\textbf{Step 2.} $2(1 - \sin^2 x) + \sin x = 1$, so
$2 - 2\sin^2 x + \sin x - 1 = 0$, i.e.
$2\sin^2 x - \sin x - 1 = 0$.\\
\textbf{Step 3.} Factor: $(2\sin x + 1)(\sin x - 1) = 0$, so
$\sin x = -\dfrac{1}{2}$ or $\sin x = 1$.\\
\textbf{Step 4.} $\sin x = 1$ gives $x = \dfrac{\pi}{2}$. \;
$\sin x = -\dfrac{1}{2}$ is negative in quadrants III and IV, with reference
angle $\dfrac{\pi}{6}$, giving $\dfrac{7\pi}{6}$ and $\dfrac{11\pi}{6}$.\\
\textbf{Step 5.} $\ans{x = \dfrac{\pi}{2}, \; \dfrac{7\pi}{6}, \;
\dfrac{11\pi}{6}}$}
