\wstitleblock{Multiplying and Dividing Decimals by Place-Value Reasoning}{Grade 4--5 Number and Operations in Base Ten}{}

\noindent\small Place-value shifts, decimal products, decimal quotients, and estimating or comparing without computing

\medskip
\noindent\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Say every number in place-value words first: $0.4$ is \emph{4 tenths}, $0.25$ is \emph{25 hundredths}.
  \item Multiply or divide the digits as if they were whole numbers, then decide where the decimal point belongs by thinking about the size of the answer.
  \item Show the place-value thinking, not just the algorithm. Label every answer with its unit.
\end{itemize}
\normalsize

\problem[5cm]{One paper clip has a mass of $1.24$ grams.
A box holds $100$ identical paper clips.
Find the total mass of the paper clips in the box.
Use the place-value chart to show how each digit moves.
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}\hline
\textbf{hundreds} & \textbf{tens} & \textbf{ones} & \textbf{tenths} & \textbf{hundredths} & \textbf{thousandths}\\\hline
\rule{0pt}{0.9cm} & & 1 & 2 & 4 & \\\hline
\rule{0pt}{0.9cm} & & & & & \\\hline
\end{tabular}
\end{center}
\answerline{g}}

\problem[5cm]{A ribbon is $4.5$ metres long. It is cut into $10$ equal pieces.
How long is each piece, in metres?
Explain which way the digits shift and why the answer must be smaller than $4.5$.
\answerline{m}}

\problem[5.5cm]{A soup recipe uses $0.4$ litres of stock for one batch.
The kitchen makes $6$ batches.
How many litres of stock are needed in all?
Write the product first in tenths (for example, ``4 tenths, six times'') and then as a decimal.
\answerline{L}}

\problem[5cm]{A rectangular garden bed measures $6.8$ metres by $4.2$ metres.
\textbf{Estimate} its area by rounding each measurement to the nearest whole metre,
then multiplying. (Do not compute the exact area.)
State the two rounded measurements you used.
\answerline{m$^2$}}

\problem[6cm]{A rope $4.8$ metres long is cut into pieces that are each $0.6$ metres long.
How many pieces are there?
Show the place-value reasoning: rewrite both lengths in tenths, then divide.}

\problem[6cm]{Without computing either value exactly, decide which is greater:
\begin{center}\large
$0.6 \times 12$ \qquad or \qquad $12 \div 0.6$
\end{center}
Write $<$ or $>$ between them, and explain your reasoning in one sentence using the
words \emph{smaller than one}.}

\problem[5.5cm]{Nine identical books have a total mass of $7.2$ kilograms.
Each book has the same mass. Find the mass of one book, in kilograms.
Show the division in tenths.
\answerline{kg}}

\problem[6cm]{\textbf{Find the mistake.} A strip of fabric measures $1.4$ metres by
$0.35$ metres. Sam multiplies and writes the area as $4.9$ square metres.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Use estimation to explain, in one sentence, why $4.9$ cannot be correct.
  \item Give the correct area in square metres.
\end{itemize}
\answerline{m$^2$}}
