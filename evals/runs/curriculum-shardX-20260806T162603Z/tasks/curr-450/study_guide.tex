% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Choosing a Summary: Mean or Median}

\begin{formulabox}
\textbf{Two models for ``typical''.} The mean adds every value and divides, so
one extreme value moves it a long way. The median is the middle value in order,
so an extreme value moves it hardly at all. Report the mean with the standard
deviation; report the median with the interquartile range. When the two
summaries disagree by a lot, an outlier is usually the reason --- say so rather
than picking silently.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Five days of rainfall: $3, 5, 6, 8, 28$ mm. Compute both
summaries and compare.
\step{The list has one value far above the rest, so the mean and the median will not agree --- compute both before choosing.}
\step{Mean $= \frac{3+5+6+8+28}{5} = \frac{50}{5} = 10$; median is the third value in order, $6$. The $28$ mm day is what pulls the mean up.}

$\Rightarrow\quad$ mean $\ans{10}$, \ median $\ans{6}$ --- report the median for a typical day
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Five days: $1, 3, 4, 6, 26$ mm. Compute the mean and the
median.\\[2pt]
\rotatebox{180}{\footnotesize check: mean $\ans{8}$, median $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Comparing Fits With Residuals}

\begin{formulabox}
\textbf{Rule:} for each data point, residual $= y - \hat{y}$, the observed value
minus the model's prediction. A negative residual means the model predicted too
high. Square every residual and add: that single number scores the model, and
between two models the \emph{smaller} score is the closer fit. Squaring is what
stops over- and under-predictions cancelling out.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Model $\hat{y} = 2 + 2x$ against the points $(1,5)$,
$(2,7)$, $(3,7)$.
\step{Scoring a fit means comparing each observation with what the model predicts there, so compute all three predictions first.}
\step{Predictions $4, 6, 8$ give residuals $5-4 = 1$, \ $7-6 = 1$, \ $7-8 = -1$; squaring and adding gives $1 + 1 + 1 = 3$.}

$\Rightarrow\quad$ first residual $\ans{1}$, \ fit score $\ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Model $\hat{y} = 1 + 3x$ against $(1,6)$, $(2,8)$,
$(3,9)$. Find the residual at $x = 1$ and the fit score.\\[2pt]
\rotatebox{180}{\footnotesize check: residual $\ans{2}$, fit score $\ans{6}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Saying Only What the Data Supports}

\begin{formulabox}
\textbf{Three limits to state on every conclusion.} \textbf{Who} --- a
proportion describes the group that was actually surveyed, and reaches further
only if that group was randomly sampled from the wider one. \textbf{Why not} ---
observational data show that two things go together, never that one causes the
other; name a confounding variable that could explain both. \textbf{How far} ---
a fitted model is trustworthy inside the range of the data; predicting outside it
is extrapolation, and two models that agree inside can disagree badly outside.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Of $60$ surveyed members of a school's chess club,
$24$ walk to school. What proportion is that, and who does it describe?
\step{The count is a proportion of the group actually asked, so compute it and then name that group before saying anything wider.}
\step{$\frac{24}{60} = \frac{2}{5}$. It describes the chess club members surveyed --- club members are not a random sample of the school, so it says nothing about the school as a whole.}

$\Rightarrow\quad$ $\ans{\tfrac{2}{5}}$ of the surveyed club members
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $90$ students leaving the library, $27$ took the bus
home. What proportion is that?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{10}}$ of the students surveyed}
\end{tryitbox}
