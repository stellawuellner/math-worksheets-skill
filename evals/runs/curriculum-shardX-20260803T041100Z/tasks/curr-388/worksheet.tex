\textit{Adding, subtracting and scaling matrices, multiplying matrices
including a real inventory model, and using determinants and matrix equations}
\end{center}

\noindent\textbf{The rules you need.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{Add / subtract} only matrices of the same size, entry by
corresponding entry.
\item \textbf{Scalar multiple:} multiply \emph{every} entry by the scalar.
\item \textbf{Product:} the entry in row $i$, column $j$ of $AB$ is row $i$ of
$A$ paired term by term with column $j$ of $B$, then summed. $AB$ exists only
when the number of columns of $A$ equals the number of rows of $B$.
\item \textbf{Determinant} of a $2 \times 2$ matrix: the product of the main
diagonal minus the product of the other diagonal.
\item \textbf{Equal matrices} have equal corresponding entries --- that is what
turns a matrix equation into ordinary equations.
\end{itemize}

\problem[3.6cm]{Let
$A = \begin{bmatrix} 3 & -2 \\ 5 & 4 \end{bmatrix}$ \quad and \quad
$B = \begin{bmatrix} -1 & 6 \\ 2 & -7 \end{bmatrix}$.

Find $A + B$.}

\problem[3.6cm]{Let $A = \begin{bmatrix} 4 & -3 \\ 2 & 6 \end{bmatrix}$.

Find $3A$.}

\problem[4.6cm]{Let
$A = \begin{bmatrix} 2 & 1 \\ 3 & -1 \end{bmatrix}$ \quad and \quad
$B = \begin{bmatrix} 4 & 3 \\ -2 & 5 \end{bmatrix}$.

Find the product $AB$. Show the row-by-column pairing for at least one entry.}

\problem[3.6cm]{Let
$A = \begin{bmatrix} 5 & -2 & 7 \\ 1 & 4 & -3 \end{bmatrix}$ \quad and \quad
$B = \begin{bmatrix} 2 & 3 & -1 \\ 6 & -4 & 5 \end{bmatrix}$.

Find $A - B$.}

\problem[4.2cm]{Let
$A = \begin{bmatrix} 1 & -2 \\ 3 & 5 \end{bmatrix}$ \quad and \quad
$B = \begin{bmatrix} 2 & 4 \\ -1 & 2 \end{bmatrix}$.

Find $2A + 3B$.}

\problem[4.6cm]{A chain has two stores. The stock matrix $S$ lists how many
units of three products each store holds, and the price column $P$ lists the
price of each product in dollars.
\[
S = \begin{bmatrix} 12 & 8 & 5 \\ 9 & 14 & 6 \end{bmatrix}
\qquad
P = \begin{bmatrix} 40 \\ 25 \\ 60 \end{bmatrix}
\]
Row $1$ of $S$ is the north store and row $2$ is the south store. Find the
product $SP$. Each entry of the answer is one store's total stock value in
dollars.}

\problem[3.4cm]{Find the determinant of
$M = \begin{bmatrix} 7 & 3 \\ 2 & 5 \end{bmatrix}$.}

\problem[5cm]{Let
$A = \begin{bmatrix} 1 & 2 & -1 \\ 3 & 2 & 4 \end{bmatrix}$ \quad and \quad
$B = \begin{bmatrix} 2 & 1 \\ -1 & 3 \\ 4 & 2 \end{bmatrix}$.

Check that the product $AB$ is defined, state its size, then compute it.}

\problem[4.2cm]{Find the values of $x$ and $y$ that make these two matrices
equal.
\[
\begin{bmatrix} x + 2 & 9 \\ 2 & y - 1 \end{bmatrix}
=
\begin{bmatrix} 7 & 9 \\ 2 & 9 \end{bmatrix}
\]}

\problem[5cm]{A ticket seller sold adult tickets and child tickets. The
situation gives the system
\[
2x + 3y = 12, \qquad x - y = 1 .
\]
Write this system as a single matrix equation of the form $AX = B$, where $A$
is the coefficient matrix, $X$ is the column of variables and $B$ is the column
of constants. Then find $x$ and $y$.}
