\sstitleblock{Solving SAS and SSS Triangles with the Law of Cosines}

\skillheading{Skill 1 --- SAS: the third side}

\begin{formulabox}
$c^2 = a^2 + b^2 - 2ab\cos C$ \quad --- and the same shape for $a$ or $b$. The side on the left is the one \emph{opposite} the known angle.\par
\textbf{Use it when} you know two sides and the angle between them. If $C = 90^\circ$ the last term is zero and this is the Pythagorean theorem.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $a = 7$, $b = 9$, included angle $C = 52^\circ$. Find $c$.
\step{Two sides with the angle between them, and no side has a known opposite angle --- that rules out the law of sines and points straight at the law of cosines.}
\step{$c^2 = 49 + 81 - 2(7)(9)\cos 52^\circ = 130 - 77.5733 = 52.4267$}
$\Rightarrow\quad c = \sqrt{52.4267} \approx \ans{7.24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a = 6$, $b = 11$, included angle $C = 64^\circ$. Find $c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $c \approx \ans{9.96}$}
\end{tryitbox}

\skillheading{Skill 2 --- SSS: an angle from three sides}

\begin{formulabox}
Rearranged: \ $\cos C = \dfrac{a^2 + b^2 - c^2}{2ab}$. \ The side \emph{opposite} the wanted angle is the one that gets subtracted.\par
\textbf{Read the sign first:} a negative cosine means the angle is obtuse, a positive one means acute. The largest angle always faces the longest side.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $a = 6$, $b = 9$, $c = 11$. Find $C$.
\step{All three sides and no angles, so the law of sines has nothing to start from; rearranging the law of cosines for the cosine is the only route.}
\step{$\cos C = \dfrac{36 + 81 - 121}{2(6)(9)} = \dfrac{-4}{108} = -0.0370$, which is negative, so expect an obtuse angle.}
$\Rightarrow\quad C = \cos^{-1}(-0.0370) \approx \ans{92.12}^\circ$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a = 7$, $b = 8$, $c = 12$. Find $C$.\\[2pt]
\rotatebox{180}{\footnotesize check: $C \approx \ans{106.07}^\circ$}
\end{tryitbox}

\skillheading{Skill 3 --- SAS, then a remaining angle}

\begin{formulabox}
\textbf{Two steps.} The law of cosines gives the third side; now that side and its opposite angle form a pair, so the law of sines finishes the triangle.\par
\textbf{Safety rule:} after finding the third side, aim the law of sines at the angle opposite the \emph{shorter} of the remaining sides. That angle must be acute, so the inverse sine cannot mislead you.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $a = 5$, $b = 8$, included angle $C = 66^\circ$. Find $A$.
\step{No side has a known opposite angle, so nothing can start with the law of sines --- find the third side first, then a pair exists.}
\step{$c^2 = 25 + 64 - 80\cos 66^\circ = 56.4600$, so $c = 7.5141$; then $\sin A = \dfrac{5\sin 66^\circ}{7.5141} = 0.6079$.}
$\Rightarrow\quad A \approx \ans{37.44}^\circ$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $a = 4$, $b = 9$, included angle $C = 58^\circ$. Find $A$.\\[2pt]
\rotatebox{180}{\footnotesize check: $A \approx \ans{26.24}^\circ$}
\end{tryitbox}

\skillheading{Skill 4 --- Distances and turns}

\begin{formulabox}
Two routes leaving one point, with the angle between them, is SAS: the two route lengths are the sides and the angle at the shared point is included between them. The direct distance between the far ends is the third side.\par
\textbf{Sanity check:} an obtuse included angle makes the third side longer than either route; an acute one makes it shorter than their sum.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two tracks leave a gate, one $a = 250$ m long and one $b = 180$ m long, meeting at $C = 41^\circ$. How far apart are the far ends?
\step{The angle sits at the shared gate, between the two known lengths, so the direct distance is the side opposite it --- SAS.}
\step{$c^2 = 62500 + 32400 - 2(250)(180)\cos 41^\circ = 26976.1$}
$\Rightarrow\quad c \approx \ans{164.24}$ metres, shorter than either track, as an acute angle requires.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A triangular course has legs $a = 90$ m, $b = 140$ m and $c = 200$ m. Find the turn angle $C$ opposite the longest leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $C \approx \ans{119.22}^\circ$}
\end{tryitbox}
