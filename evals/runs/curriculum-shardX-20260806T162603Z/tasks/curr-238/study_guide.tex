\sstitleblock{Dilations and Similar Figures}

\skillheading{Skill 1 --- Dilating a point about the origin}

\begin{formulabox}
\textbf{Rule:} a dilation centred at the origin with scale factor $k$ sends
$(x, y) \mapsto (kx, ky)$.\\[3pt]
Multiply \textit{both} coordinates by the same $k$. Signs come along unchanged, and $k > 1$
moves the point away from the origin while $0 < k < 1$ moves it closer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Dilate $(4, -3)$ about the origin with $k = 2$.
\step{The centre is the origin, so no subtraction is needed --- multiply each coordinate by $k$.}
\step{$x' = 2 \cdot 4 = 8$ \quad and \quad $y' = 2 \cdot (-3) = -6$}
$\Rightarrow\quad \ans{(8, -6)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Dilate $(-2, 5)$ about the origin with $k = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(-6, 15)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Finding the scale factor from two lengths}

\begin{formulabox}
\textbf{Rule:} \quad $k = \dfrac{\text{image length}}{\text{original length}}$\\[3pt]
Always divide a length by the length it corresponds to. The same $k$ must work for every pair of
matching sides --- if two pairs give different numbers, the figures are not similar.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A side of $5$ becomes a side of $20$. Find $k$.
\step{We know a matching pair of lengths, so divide image by original.}
\step{$k = \dfrac{20}{5} = 4$, and $4 > 1$ tells us the image is an enlargement.}
$\Rightarrow\quad \ans{k = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A side of $12$ becomes a side of $9$. Find $k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{k = \tfrac{3}{4}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Why a dilation keeps sides parallel}

\begin{formulabox}
\textbf{Rule:} \quad slope $= \dfrac{y_2 - y_1}{x_2 - x_1}$\\[3pt]
Dilating multiplies every rise and every run by the same $k$, so the fraction is unchanged. Equal
slopes on two different segments mean the segments are parallel --- that is the evidence that a
dilation gives a similar figure with matching angles.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $(2, 1)$ to $(4, 4)$, dilated by $k = 2$ to $(4, 2)$ and $(8, 8)$.
Compare the slopes.
\step{To test for parallel sides, compute the slope of each segment and compare them.}
\step{Original: $\dfrac{4 - 1}{4 - 2} = \dfrac{3}{2}$. \quad Image: $\dfrac{8 - 2}{8 - 4} = \dfrac{6}{4}$, which is the same fraction.}
$\Rightarrow\quad \text{both slopes} = \ans{\tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $(1, 3)$ to $(5, 6)$, dilated by $k = 4$ to $(4, 12)$ and $(20, 24)$.
Find both slopes.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{4}}$}
\end{tryitbox}

\skillheading{Skill 4 --- Finding a missing length in similar figures}

\begin{formulabox}
\textbf{Rule:} in similar figures, matching sides are in the same ratio, so
\[ \frac{\text{missing image side}}{\text{its original}} = \frac{\text{known image side}}{\text{its original}} \]
Find $k$ from the pair you know, then multiply the other original length by that same $k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rectangle $5$ cm by $8$ cm is enlarged so the $5$ cm side becomes
$15$ cm. Find the other side.
\step{One matching pair is known, so find $k$ from it and apply it to the side we want.}
\step{$k = \dfrac{15}{5} = 3$, so the other side is $8 \cdot 3$ cm.}
$\Rightarrow\quad \ans{24} \text{ cm}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A rectangle $6$ cm by $10$ cm is enlarged so the $6$ cm side becomes
$9$ cm. Find the other side, in centimetres.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} one scale factor must fit every direction. If the width scales by $2$ and
the height by $3$, the picture has been stretched, not dilated --- and stretched figures are not
similar.
\end{watchoutbox}
