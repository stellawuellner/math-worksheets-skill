\sstitleblock{Diagnosing Transformation-Order and Inverse-Domain Errors}

\skillheading{Skill 1 --- Transformations Inside the Function}

\begin{formulabox}
\textbf{Factor the inside before you read anything off it:} $f(bx - c) = f\!\left(b\left(x - \tfrac{c}{b}\right)\right)$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Inside (input) operations act in the \emph{reverse} of the written order and in the \emph{opposite} direction: stretch or compress first, then shift.
  \item Outside (output) operations act in the order written and in the direction written: $f(x) + 2$ really is up $2$.
  \item Horizontal factor $b$ means a compression by $1/b$; the shift is $c/b$, never $c$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Describe the graph of $y = f(3x - 12)$.
\step{A shift cannot be read from an unfactored inside, so pull the coefficient of $x$ out first.}
\step{$\ans{3(x - 4)}$ is the inside, so the graph compresses horizontally by $\tfrac{1}{3}$ and then shifts right $4$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Factor the inside of $y = f(2x + 10)$ and read off the shift.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2(x + 5)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Composing in the Right Order}

\begin{formulabox}
\textbf{In $f \circ g$ the function on the right runs first:} $(f \circ g)(x) = f(g(x))$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Compose symbolically before substituting a number --- the order is visible in the formula and invisible in the arithmetic.
  \item $f \circ g$ and $g \circ f$ are different functions except by accident.
  \item An input must be in the domain of the inner function \emph{and} its output in the domain of the outer one.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = 3x - 1$ and $g(x) = x^2$. Find $(f \circ g)(2)$.
\step{The inner function $g$ takes the input, so build $f(g(x))$ symbolically before any number goes in.}
\step{$f(g(x)) = 3x^2 - 1$, so at $x = 2$ the value is $\ans{11}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Same $f$ and $g$: find $(g \circ f)(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25}$}
\end{tryitbox}

\skillheading{Skill 3 --- Inverses and Their Domains}

\begin{formulabox}
\textbf{Swap and solve gives the formula; the graph gives the domain.} The domain of $f^{-1}$ is the range of $f$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Write $y = f(x)$, swap $x$ and $y$, solve for $y$, then state the domain separately.
  \item Squaring both sides is reversible only on the side where the radical was non-negative --- that is what selects one branch of a $\pm$.
  \item A function that is not one-to-one needs a restriction first; without one it has no inverse function.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = \sqrt{x - 1} + 2$. Find $f^{-1}$ in completed-square form.
\step{Swap the variables and undo the operations from the outside in, keeping track of where the square root was non-negative.}
\step{Solving gives $x^2 - 4x + 5$, which is $\ans{(x - 2)^2 + 1}$, valid on $x \ge 2$ because $f$ never outputs less than $2$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = \sqrt{x + 3} - 1$. Find $f^{-1}$ in completed-square form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x + 1)^2 - 3}$}
\end{tryitbox}
