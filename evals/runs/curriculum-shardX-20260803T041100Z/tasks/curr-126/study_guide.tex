\sstitleblock{Adding and Subtracting Fractions with Like Denominators}

\skillheading{Skill 1 --- Adding Fractions with Like Denominators}

\begin{formulabox}
\textbf{Rule:}\quad $\dfrac{a}{n} + \dfrac{b}{n} = \dfrac{a+b}{n}$\\[3pt]
\textbf{Why:} the denominator $n$ names the \emph{size} of one piece. Adding does not
re-cut the bar, so the pieces keep their size and only the count changes.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=2pt]
  \item Add the numerators (top numbers).
  \item Copy the denominator (bottom number) unchanged.
  \item If the numerator ends up larger than the denominator, the answer is more than one whole.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\dfrac{3}{8} + \dfrac{2}{8}$
\step{Both fractions count eighths, so the pieces are already the same size --- just count pieces.}
\step{$3$ eighths and $2$ more eighths make $3 + 2 = 5$ eighths, and the bar is still cut in 8:
$\dfrac{3}{8} + \dfrac{2}{8} = \ans{\dfrac{5}{8}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\dfrac{4}{9} + \dfrac{3}{9}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{7}{9}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Subtracting, then Simplifying}

\begin{formulabox}
\textbf{Rule:}\quad $\dfrac{a}{n} - \dfrac{b}{n} = \dfrac{a-b}{n}$\\[3pt]
\textbf{Simplify:} divide the numerator and the denominator by the same number until
the only number that divides both is 1. Simplifying renames the amount; it never changes it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\dfrac{7}{10} - \dfrac{3}{10}$
\step{Take away means cross out pieces of the same size, so subtract the counts and keep the tenths.}
\step{$7 - 3 = 4$ tenths, and $4$ and $10$ both divide by $2$:
$\dfrac{7}{10} - \dfrac{3}{10} = \dfrac{4}{10} = \ans{\dfrac{2}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\dfrac{5}{8} - \dfrac{1}{8}$, in simplest form\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{2}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding a Missing Numerator}

\begin{formulabox}
\textbf{Shape of the question:}\quad $\dfrac{a}{n} + \underline{\hspace{0.9cm}} = \dfrac{c}{n}$\\[3pt]
Since every piece is an $n$th, ask ``how many more $n$ths?'' and \textbf{count up}
from $a$ to $c$. The missing count is $c - a$, written over the same denominator.
Always check by adding your answer back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $\dfrac{1}{7} + \underline{\hspace{0.9cm}} = \dfrac{6}{7}$
\step{Both known fractions are sevenths, so the missing piece must be sevenths too --- only its numerator is unknown.}
\step{Count up from $1$ seventh to $6$ sevenths: $6 - 1 = 5$ more sevenths.
Check: $\dfrac{1}{7} + \dfrac{5}{7} = \dfrac{6}{7}$, so the blank is $\ans{\dfrac{5}{7}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $\dfrac{2}{11} + \underline{\hspace{0.9cm}} = \dfrac{9}{11}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{7}{11}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Never add or subtract the denominators.
$\dfrac{3}{7} + \dfrac{2}{7}$ is $\dfrac{5}{7}$, not $\dfrac{5}{14}$ --- adding cannot
make the total \emph{smaller}, and nothing re-cut the bar into fourteenths.
\end{watchoutbox}
