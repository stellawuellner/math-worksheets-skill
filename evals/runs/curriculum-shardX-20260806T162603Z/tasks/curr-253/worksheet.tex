\wstitleblock{Rewriting Expressions with Distributive Structure}{Algebra 1}{}

\begin{center}
\textit{Distributing a factor, taking a common factor out, and choosing the form that answers the question}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Every quantity below is stated with its unit. Give your answer in the unit the problem names.
  \item ``Without brackets'' means expanded form; ``as a product'' means factored form.
  \item Show the distributing or factoring step, not just the final expression.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[5cm]{A rectangular dog run measures $(2x + 9)$ m along its long side and $4$ m along its
short side. Write its perimeter, in metres, as an expression with no brackets.
\par\vspace*{0.1cm}
\answerline{m}}

\problem[5cm]{A school shop sells one notebook for $(x + 3)$ dollars. A teacher buys $15$ identical
notebooks. Write the total cost, in dollars, as an expression with no brackets.}

\problem[5cm]{A hallway floor has area $(6x + 30)$ square metres and a constant width of $6$ m.
Its area is width $\times$ length. Find the length of the hallway, in metres.
\par\vspace*{0.1cm}
\answerline{m}}

\problem[4cm]{A club orders class sets of books. Plan A charges $8(x + 5)$ dollars and Plan B
charges $8x + 40$ dollars, where $x$ is the number of books a class needs.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Rewrite Plan A without brackets to confirm the two plans always charge the same. \ansline
  \item[(b)] One printed form shows the price per book and the $5$ spare books; the other shows a
             flat $40$-dollar charge as its own term. Say which is which and explain what each
             form makes visible. \ansline
\end{itemize}}

\problem[5.5cm]{A patio is a rectangle measuring $(x + 6)$ m by $(x + 2)$ m. Write its area, in
square metres, in standard form.
\par\vspace*{0.1cm}
\answerline{m$^2$}}

\problem[5cm]{A crew has painted $(9x^2 + 15x)$ square metres of wall. Write that area as a product
of the greatest common monomial factor and one remaining factor.}

\problem[5.5cm]{A stall sells $x$ bunches of flowers. It takes in $40x$ dollars and its costs are
$5(3x + 20)$ dollars. Profit is income minus costs. Write the profit, in dollars, as an expression
with no brackets.}

\problem[4cm]{A field is $(3x + 12)$ m long. It is divided along its length into $3$ strips of
equal length.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Find the length of one strip, in metres.
             \par\vspace*{0.1cm}\answerline{m}
  \item[(b)] Explain how writing the field length in the factored form $3(x + 4)$ makes that
             division something you can read off rather than calculate. \ansline
\end{itemize}}

\problem[6cm]{A border is built from three straight runs of edging: $3$ runs of $(2x + 5)$ m,
$4$ runs of $(x - 2)$ m, and then $2$ runs of $(x + 1)$ m are removed again. Write the total length
of edging left, in metres, as an expression with no brackets.
\par\vspace*{0.1cm}
\answerline{m}}

\problem[5.5cm]{A rectangular metal sheet measures $(3x + 6)$ cm by $(x + 4)$ cm.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Write the area of the sheet in standard form, in square centimetres.
             \par\vspace*{0.1cm}\answerline{cm$^2$}
  \item[(b)] Write the same area in fully factored form. \ansline
  \item[(c)] The workshop needs to know that the area is always a whole number of square
             centimetres divisible by $3$ whenever $x$ is a whole number. Say which of your two
             forms shows that immediately, and explain why. \ansline
\end{itemize}}
