\sstitleblock{Placing Fractions on a Number Line}

\begin{formulabox}
\textbf{The two counts.} On a number line, a fraction is a \emph{distance from
$0$}, not a picture of a pizza.\\[3pt]
\textbf{Bottom number (denominator):} how many equal parts one whole was cut
into. Count the \emph{parts}, never the tick marks.\\[2pt]
\textbf{Top number (numerator):} how many of those parts you travel, starting
at $0$.\\[3pt]
So $\frac{3}{4}$ means: cut the whole into $4$ equal parts, then go $3$ of them.
\end{formulabox}

\skillheading{1. Name the point that is already marked}

\begin{formulabox}
Read the labels at the ends first --- they tell you where one whole finishes.
Then count the equal parts between $0$ and $1$ to get the bottom number, and
count parts from $0$ to the dot to get the top number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Name point $M$.
\begin{center}
\begin{tikzpicture}[x=1.3cm,y=1cm]
  \draw[thick] (0,0) -- (6,0);
  \foreach \i in {0,1,2,3,4,5,6} \draw[thick] (\i,0.14) -- (\i,-0.14);
  \node[below=3pt] at (0,0) {$0$};
  \node[below=3pt] at (6,0) {$1$};
  \fill (5,0) circle (2.2pt);
  \node[above=4pt] at (5,0) {$M$};
\end{tikzpicture}
\end{center}
\step{The dot is already placed, so the job is to read it --- count parts for
the bottom number, then count travel for the top number.}
\step{Equal parts from $0$ to $1$: six, so the parts are sixths. Parts
travelled from $0$ to $M$: five.}
Five parts of a whole cut into sixths: $M = \ans{\tfrac{5}{6}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A line from $0$ to $1$ is cut into $3$ equal parts. A dot sits
on the first tick after $0$. What fraction names the dot?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$}
\end{tryitbox}

\skillheading{2. Place a fraction you are given}

\begin{formulabox}
Work from the bottom number outward: it tells you how many equal parts the
whole needs. Then count that many parts from $0$ --- the top number of them ---
and put the dot there. Two fractions that land on the same point name the same
number, so $\frac{2}{4}$ and $\frac{1}{2}$ are the same place.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Mark $\frac{6}{8}$, then name the point with a simpler
fraction.
\begin{center}
\begin{tikzpicture}[x=1.0cm,y=1cm]
  \draw[thick] (0,0) -- (8,0);
  \foreach \i in {0,1,2,3,4,5,6,7,8} \draw[thick] (\i,0.14) -- (\i,-0.14);
  \node[below=3pt] at (0,0) {$0$};
  \node[below=3pt] at (8,0) {$1$};
  \fill (6,0) circle (2.2pt);
\end{tikzpicture}
\end{center}
\step{The bottom number $8$ says the whole is cut into eighths, so count six
eighths from $0$ and mark the sixth tick.}
\step{Now group the eighths in twos: two eighths make one fourth, so the eight
parts become four fourths and the dot lands on the third of them.}
The same point, named in bigger parts: $\ans{\tfrac{3}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} On a line cut into $6$ equal parts, mark $\frac{2}{6}$. What
simpler fraction names that same point?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$}
\end{tryitbox}

\skillheading{3. Compare two fractions on the line}

\begin{formulabox}
Whichever fraction lands \emph{farther from $0$} is the greater one --- that is
all comparing means on a number line.\\[2pt]
$>$ means \emph{is greater than} \qquad $<$ means \emph{is less than}\\[2pt]
When the top numbers match, the fraction with the \emph{smaller} bottom number
wins, because fewer cuts make longer parts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Which is greater, $\frac{3}{4}$ or $\frac{3}{6}$?
\step{Both travel three parts, so the winner is decided by which parts are
longer --- compare the sizes of the cuts, not the numbers.}
\step{Cutting a whole into $4$ makes longer parts than cutting it into $6$, so
three of them reach farther. On a line, $\frac{3}{4}$ sits past the middle
while $\frac{3}{6}$ sits exactly on it.}
$\frac{3}{4}$ is farther from $0$, so the sign is $\ans{>}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Put $\frac{1}{2}$ and $\frac{5}{8}$ on one eighths line
(remember $\frac{1}{2}$ is four eighths). Which sign makes
$\frac{1}{2}$ \underline{\hspace{0.9cm}} $\frac{5}{8}$ true?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} count the \emph{spaces} between ticks, not the ticks
themselves --- a line cut into fourths has five tick marks. And check the label
at the right-hand end before you start: if it says $2$, the parts between $0$
and $1$ are still the unit, so a dot past the middle can be more than one
whole.
\end{watchoutbox}
