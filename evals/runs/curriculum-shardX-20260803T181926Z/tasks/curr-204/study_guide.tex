\sstitleblock{Diagnosing Invalid Simplification of Sums of Radicals}

\skillheading{Skill 1 --- Simplifying One Radical}

\begin{formulabox}
\textbf{Roots split over products, never over sums:} $\sqrt{ab} = \sqrt{a}\,\sqrt{b}$, and there is no rule for $\sqrt{a+b}$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Look for the largest perfect-square factor: $4, 9, 16, 25, 36, 49, 64, 81, 100$.
  \item Take its root out front and leave the rest under the sign.
  \item A radicand with no perfect-square factor is already simplest.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write $\sqrt{45}$ in simplest radical form.
\step{Hunt for the largest perfect square that divides the radicand, because only a product may be split.}
\step{$45 = 9 \cdot 5$, so $\sqrt{45} = \sqrt{9}\,\sqrt{5} = \ans{3\sqrt{5}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write $\sqrt{98}$ in simplest radical form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7\sqrt{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Combining Like Radicals}

\begin{formulabox}
\textbf{Radicals combine like terms:} $2\sqrt{2} + 3\sqrt{2} = 5\sqrt{2}$, exactly as $2x + 3x = 5x$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Simplify every term \emph{first} --- unlike radicands often match once simplified.
  \item Add only the coefficients. The radicand is never added and never changes.
  \item If the radicands still differ after simplifying, the expression is finished as it stands.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Simplify $\sqrt{20} + \sqrt{45}$.
\step{The radicands look unlike, so simplify each term before deciding whether anything combines.}
\step{$\sqrt{20} = 2\sqrt{5}$ and $\sqrt{45} = 3\sqrt{5}$, so the coefficients add to give $\ans{5\sqrt{5}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Simplify $\sqrt{75} - \sqrt{12}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3\sqrt{3}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Testing a Claim with Decimals}

\begin{formulabox}
\textbf{When a classmate's rule looks plausible, price it out.} Put a decimal on both sides; if the two numbers differ, the rule is dead.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Perfect squares make the sharpest counterexamples --- no rounding is involved at all.
  \item A decimal test disproves a rule but never proves one; state the answer in exact radical form.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Is $\sqrt{4} + \sqrt{9}$ equal to $\sqrt{13}$?
\step{Both radicands are perfect squares, so each root can be taken exactly and the claim needs no rounding to settle.}
\step{The left side is $2 + 3$, while $\sqrt{13}$ is about $3.61$, so $\ans{5 > \sqrt{13}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Compare $\sqrt{1} + \sqrt{9}$ with $\sqrt{10}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4 > \sqrt{10}}$}
\end{tryitbox}
