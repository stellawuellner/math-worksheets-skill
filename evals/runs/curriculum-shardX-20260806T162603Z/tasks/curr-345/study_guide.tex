% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Volume of a composite solid}

\begin{formulabox}
\textbf{Method:} cut the solid into standard pieces, find each volume, then \emph{add} pieces that are stuck on and \emph{subtract} pieces that are drilled or scooped out.\par
$V_{\text{cyl}} = \pi r^2 h$ \quad $V_{\text{cone}} = \frac{1}{3}\pi r^2 h$ \quad $V_{\text{sphere}} = \frac{4}{3}\pi r^3$ \quad (hemisphere: half of that)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A cone of radius $2$ and height $6$ sits on a cylinder of radius $2$ and height $5$. Find the volume.
\step{The solid is two standard shapes joined, and neither is cut away, so the volume is a sum --- find each piece and add.}
\step{$V = \pi(2)^2(5) + \frac{1}{3}\pi(2)^2(6) = 20\pi + 8\pi = 28\pi$}

$\Rightarrow\quad V \approx$ $\ans{87.96}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A cone of radius $3$ and height $7$ sits on a cylinder of radius $3$ and height $4$. Find the volume, to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{179.07}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Surface area of a composite solid}

\begin{formulabox}
\textbf{Method:} count only the faces that are actually \emph{on the outside}. A face where two pieces join is buried and is never included, and an open top means no lid.\par
cylinder wall $= 2\pi r h$ \quad circle $= \pi r^2$ \quad cone curved surface $= \pi r \ell$ (slant height $\ell$, not the height)
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An open-topped cylinder has radius $3$ and height $10$. Find its outside surface area.
\step{Only the wall and the base are real surfaces here --- the top is open, so the usual second circle does not belong.}
\step{$S = 2\pi(3)(10) + \pi(3)^2 = 60\pi + 9\pi = 69\pi$}

$\Rightarrow\quad S \approx$ $\ans{216.77}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An open-topped cylinder has radius $4$ and height $5$. Find its outside surface area, to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{175.93}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Density, mass and volume}

\begin{formulabox}
\textbf{One relationship, three forms:} density $= \dfrac{\text{mass}}{\text{volume}}$, so mass $=$ volume $\times$ density and volume $=$ mass $\div$ density.\par
Check the units before you compute: g and cm$^3$ go together, kg and m$^3$ go together. A mass in kilograms against a density in g/cm$^3$ needs converting first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A casting of volume $350$ cm$^3$ has density $2.7$ g/cm$^3$. Find its mass.
\step{Volume and density are known and mass is wanted, so the relationship is used in its multiplying form --- density counts grams per cubic centimetre, so multiply.}
\step{mass $= 350 \times 2.7$}

$\Rightarrow\quad$ $\ans{945}$ g
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An object has mass $3200$ g and density $1.6$ g/cm$^3$. Find its volume.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2000}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Designing to a constraint}

\begin{formulabox}
\textbf{Method:} write the formula for the quantity that is fixed, put the required value on one side, and solve for the dimension you are free to choose.\par
For a cylinder of fixed capacity $V$ and fixed radius $r$: \ $h = \dfrac{V}{\pi r^2}$. Changing $r$ therefore changes $h$ too --- the two are never independent once the capacity is fixed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A cylindrical tank of radius $2$ must hold $40$ cubic units. Find the height.
\step{The capacity is the fixed requirement and the radius is already chosen, so the height is the only unknown --- rearrange the volume formula for it.}
\step{$h = \dfrac{40}{\pi(2)^2} = \dfrac{40}{4\pi}$}

$\Rightarrow\quad h \approx$ $\ans{3.18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A cylindrical tank of radius $2.5$ must hold $50$ cubic units. Find the height, to the nearest hundredth.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.55}$}
\end{tryitbox}
