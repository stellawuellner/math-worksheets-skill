\sstitleblock{Number Bonds to 10 --- Fixing an Incomplete or Duplicated Family}

\skillheading{Skill 1 --- Finding the missing part of a bond}

\begin{formulabox}
\textbf{The model:}\\[4pt]
A number bond holds the \textbf{whole} on top and its two \textbf{parts} below.
\begin{center}
\begin{tikzpicture}
  \node[draw, thick, circle, minimum size=0.85cm] (w) at (0,1.25) {whole};
  \node[draw, thick, circle, minimum size=0.85cm] (p) at (-1.5,0) {part};
  \node[draw, thick, circle, minimum size=0.85cm] (q) at (1.5,0) {part};
  \draw[thick] (w) -- (p);
  \draw[thick] (w) -- (q);
\end{tikzpicture}
\end{center}
\textbf{part $+$ part $=$ whole} \qquad \textbf{whole $-$ part $=$ the other part}\par
\smallskip
A part is always \emph{smaller} than the whole. To find a missing part, take the
part you know \textbf{away} from the whole --- never add.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad The whole is $7$ and one part is $2$. Find the other part.
\step{A part is missing and the whole is known, so this is a take-away: whole minus
the known part.}
\step{Take $2$ away from $7$: two steps back from $7$ lands on $5$. Check by adding
the parts back: $2 + 5 = 7$ $\checkmark$}
The missing part is $\ans{5}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad The whole is $8$ and one part is $6$. Find the other part.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Spotting a fact written twice}

\begin{formulabox}
\textbf{How to check a family for a repeat:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Count the \textbf{adding} facts. There must be exactly \textbf{two}.
  \item Count the \textbf{take-away} facts. There must be exactly \textbf{two}.
  \item Each take-away fact starts with the whole and removes a \textbf{different}
        part. If both remove the \emph{same} part, one is a repeat.
\end{itemize}
Cross out the repeat, then write the fact that takes away the \emph{other} part.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A family for parts $3$ and $7$: \ $3+7=10$, \ $7+3=10$, \
$10-7=3$, \ $10-7=3$.
\step{Both take-away facts remove the same part $7$, so the second one is a repeat
and the fact that removes the part $3$ is missing.}
\step{Remove the other part instead: $10 - 3$ leaves the part $7$, and $3 + 7 = 10$
checks it $\checkmark$}
The missing fact is $10 - 3 = \ans{7}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A family for parts $2$ and $6$: \ $2+6=8$, \ $6+2=8$, \
$8-6=2$, \ $6+2=8$. Write the missing take-away fact $8 - 2 = ?$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Writing the whole family of four facts}

\begin{formulabox}
\textbf{Rule --- every bond makes four facts:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item part $+$ other part $=$ whole
  \item other part $+$ part $=$ whole
  \item whole $-$ part $=$ other part
  \item whole $-$ other part $=$ part
\end{itemize}
Two adding facts, two take-away facts. If you have written fewer than four, the
family is \textbf{incomplete}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write the whole family for the parts $4$ and $5$.
\step{The two parts are known and the whole is not, so add the parts first --- the
whole is what the family is built on.}
\step{$4 + 5 = 9$, so the family is $4+5=9$, \ $5+4=9$, \ $9-4=5$, \ $9-5=4$.}
The whole is $\ans{9}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the whole for the parts $2$ and $5$, then write all four
facts.\\[2pt]
\rotatebox{180}{\footnotesize check: whole $= \ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Turning an adding fact around ($5+4$ instead of $4+5$) makes
a \emph{new} fact, but turning a take-away fact around does not --- $10-3$ and
$10-7$ are different facts, while writing $10-7$ twice is the same fact twice.
\end{watchoutbox}
