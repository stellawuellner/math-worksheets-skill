  \item Most problems below show a student's work. The arithmetic is usually

  \item When a part asks you to name or describe an error, write a sentence.

  \item Keep exact values (fractions, $\pi$) unless a decimal is requested.
\end{itemize}

\problem[5.5cm]{Air is pumped into a spherical balloon so that its radius grows
at the constant rate $\dfrac{dr}{dt} = 3$ cm/s. Using $V = \frac{4}{3}\pi r^3$,
find $\dfrac{dV}{dt}$ at the instant when $r = 5$ cm. Round to two decimal
places (cubic centimetres per second).}

\problem{A cube's edge length $s$ grows at $\dfrac{ds}{dt} = 0.5$ cm/s. Asked
for $\dfrac{dV}{dt}$ when $s = 4$ cm, a student writes:
\[ V = s^3 \qquad\Longrightarrow\qquad \frac{dV}{dt} = 3s^2 = 3(4)^2 = 48. \]
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error in the student's differentiation.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Give the corrected value of $\dfrac{dV}{dt}$.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{A rectangular pen is fenced against a straight barn wall, so only
three sides need fence and $60$ m of fence is available. A student writes:
\emph{``Let $x$ be each side perpendicular to the barn, so $2x + y = 60$ and
$A = xy = x(60 - 2x)$. Setting $A = 0$ gives $x = 0$ or $x = 30$, so the answer
is $x = 30$ m.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Explain what went wrong in the student's last step.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Find the corrected value of $x$ that maximizes the area.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{A $13$ ft ladder leans against a vertical wall. Its foot slides away
from the wall at $\dfrac{dx}{dt} = 2$ ft/s. Asked for $\dfrac{dy}{dt}$ when the
foot is $x = 5$ ft from the wall (so $y = 12$ ft), a student writes:
\emph{``From $x^2 + y^2 = 169$ I get $2x\frac{dx}{dt} + 2y\frac{dy}{dt} = 0$,
so $\frac{dy}{dt} = \frac{5}{6}$ ft/s.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Describe the error in the student's final line.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Give the corrected value of $\dfrac{dy}{dt}$, in feet per second.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{An open-top box is folded from a $20$ cm by $20$ cm sheet by cutting a
square of side $x$ from each corner. A student writes $V = x(20 - x)^2$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Explain which factor of the student's volume formula is wrong and why.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] The corrected volume satisfies $V'(x) = (20 - 2x)(20 - 6x)$.
    Find both critical values.
    \par\vspace*{2.6cm}\ansline
  \item[(c)] Say which critical value maximizes the volume, and why the other is rejected.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{A student minimizes $f(x) = x + \dfrac{9}{x}$ on the domain $x > 0$.
They compute $f'(x) = 1 - \dfrac{9}{x^2}$, solve $f'(x) = 0$, and report
\emph{``the minimum is at $x = -3$.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Name the error in the student's conclusion.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] List both solutions of $f'(x) = 0$.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Give the minimum value of $f$ on $x > 0$.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A street lamp sits atop a $15$ ft pole. A person $6$ ft tall walks away
from the base at $4$ ft/s. Let $x$ be the person's distance from the pole and
$s$ the length of their shadow. A student writes
$\dfrac{6}{15} = \dfrac{s}{x}$ and reports that the shadow lengthens at
$1.6$ ft/s.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Describe the error in the student's proportion.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Give the corrected value of $\dfrac{ds}{dt}$, in feet per second.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{A closed cylindrical can must hold $1000$ cm$^3$. A student minimizes
its surface area: \emph{``Constraint $\pi r^2 h = 1000$; objective
$S = 2\pi r^2 + \pi r h$. Then $h = \dfrac{1000}{\pi r^2}$ and
$S(r) = 2\pi r^2 + \dfrac{1000}{r}$, so $r \approx 4.30$ cm.''}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Identify the error in the student's objective function.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Write the corrected $S(r)$ as a function of $r$ alone.
    \par\vspace*{2.4cm}\ansline
  \item[(c)] Find the radius that minimizes the corrected $S$, to two decimals.
    \par\vspace*{2.4cm}\ansline
\end{itemize}}
