\sstitleblock{Convergence Tests and the Endpoints of an Interval}

\skillheading{Skill 1 --- The nth-term test goes one way}

\begin{formulabox}
\textbf{If $\lim a_n \neq 0$, then $\sum a_n$ diverges. That is the whole test.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item A limit of $0$ licenses \textbf{nothing}. It is necessary for
        convergence, never sufficient --- $\sum 1/n$ is the standing
        counterexample.
  \item So compute the limit first: a non-zero answer ends the problem, and a
        zero answer means the real work has not started.
  \item For a ratio of polynomials, divide through by the highest power of $n$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Apply the $n$th-term test to the series with
$a_n = \dfrac{5n - 2}{3n + 4}$.

\step{The test only needs the limit of the terms, so compute that before
reaching for any comparison --- if it is non-zero the question is already
answered.}

\step{Divide numerator and denominator by $n$: $\dfrac{5 - 2/n}{3 + 4/n}$, and
let $n \to \infty$.}

The limit is $\ans{\tfrac{5}{3}}$, which is not $0$, so the series diverges.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\lim_{n \to \infty}
\dfrac{2n + 9}{6n - 1}$, then say what the $n$th-term test concludes.

\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{3}}$, so it diverges}
\end{tryitbox}

\skillheading{Skill 2 --- When the ratio test says nothing}

\begin{formulabox}
\textbf{The ratio test concludes only when its limit misses $1$.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $L < 1$ converges, $L > 1$ diverges, $L = 1$ is \textbf{inconclusive} ---
        both $\sum 1/n$ and $\sum 1/n^{2}$ give $L = 1$ and behave differently.
  \item At $L = 1$, switch tools: $p$-series, integral test, telescoping, or a
        direct comparison.
  \item Geometric with $|r| < 1$: the sum is $\dfrac{a}{1 - r}$, where $a$ is the
        \emph{first term the sum actually starts with}. Dividing by $r$ instead
        of by $1 - r$ is the classic slip.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the exact sum of
$\displaystyle\sum_{n=0}^{\infty} 5\left(\dfrac{1}{4}\right)^{n}$.

\step{The terms have a constant ratio, so this is geometric and the closed-form
sum applies --- no test is needed beyond checking that the ratio is under $1$ in
size.}

\step{The index starts at $0$, so the first term is $5$ and the ratio is
$\dfrac{1}{4}$: the sum is $\dfrac{5}{1 - 1/4}$.}

The sum is $\ans{\tfrac{20}{3}}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the exact sum of
$\displaystyle\sum_{n=0}^{\infty} 3\left(\dfrac{2}{5}\right)^{n}$.

\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Endpoints of the interval}

\begin{formulabox}
\textbf{The ratio test gives the radius. The endpoints are yours to test.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Endpoints are $c - r$ and $c + r$: the radius is a \emph{distance} from
        the centre, never a coordinate.
  \item At either endpoint the ratio-test limit is exactly $1$, so that test is
        guaranteed to be silent there. Substitute and run a different test.
  \item The two ends are independent --- one may be included and the other
        excluded, so an interval like $[c - r,\ c + r)$ is entirely normal.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A power series is centred at $c = -4$ with radius
$r = 2$. Give the two endpoints of its interval of convergence.

\step{The radius measures distance from the centre, so step one radius to each
side of the centre rather than reading the radius off as a position.}

\step{Compute $c - r = -4 - 2$ and $c + r = -4 + 2$.}

The endpoints are $\ans{-6}$ and $\ans{-2}$, and each must now be substituted and
tested on its own.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A power series is centred at $c = 7$ with radius $r = 5$.
Give the two endpoints.

\rotatebox{180}{\footnotesize check: $\ans{2}$ and $\ans{12}$}
\end{tryitbox}
