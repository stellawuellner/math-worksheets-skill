\sstitleblock{Graphing Basic Polar Curves from Equations}

\skillheading{Skill 1 --- Evaluate a polar equation at an angle}

\begin{formulabox}
\textbf{Idea:} \ A polar equation $r = f(\theta)$ is a rule: feed it an angle, get a
signed distance from the pole.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The point is $(r, \theta)$: go out $r$ along the ray at angle $\theta$.
  \item \textbf{Negative $r$ is legal.} Plot $|r|$ in the direction \emph{opposite}
        $\theta$; that is exactly how inner loops are drawn.
  \item Exact values only: $\cos 60^\circ = \tfrac{1}{2}$,
        $\cos 120^\circ = -\tfrac{1}{2}$, $\cos 180^\circ = -1$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $r = 3\cos\theta$, find $r$ at $\theta = 60^\circ$.
\step{A single angle is given and a distance is wanted, so this is plain
      substitution --- no equation to solve.}
\step{$r = 3\cos 60^\circ = 3 \cdot \tfrac{1}{2}$.}
$\Rightarrow\quad \ans{\,\dfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $r = 4\sin\theta$, find $r$ at $\theta = 30^\circ$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Pole crossings and maximum radius}

\begin{formulabox}
\textbf{Two questions that build the picture:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Zeros.} Solve $f(\theta) = 0$ on $0^\circ \le \theta < 360^\circ$.
        Each solution is an angle at which the curve passes through the pole ---
        the boundary between petals, a cardioid's cusp, or the ends of an inner loop.
  \item \textbf{Maximum $r$.} Solve $f(\theta) = r_{\max}$. Those angles point at the
        petal tips, and for a circle or limacon they give the direction the curve is
        offset from the pole.
  \item With $\sin(k\theta)$ or $\cos(k\theta)$, solve for $k\theta$ first, list every
        value up to $k \cdot 360^\circ$, and only then divide by $k$ --- dividing too
        early loses solutions.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find every $\theta$ in $[0^\circ, 360^\circ)$ with
$3\sin(2\theta) = 0$.
\step{Asking where the rose returns to the pole means solving $r = 0$; the doubled
      angle means the window for $2\theta$ is twice as wide.}
\step{$\sin(2\theta) = 0$ gives $2\theta = 0^\circ, 180^\circ, 360^\circ, 540^\circ$;
      halving each gives four angles.}
$\Rightarrow\quad \ans{\theta = 0^\circ,\ 90^\circ,\ 180^\circ,\ 270^\circ}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $r = 5\cos\theta$, solve $5\cos\theta = 5$ on
$[0^\circ, 360^\circ)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = 0^\circ}$}
\end{tryitbox}

\skillheading{Skill 3 --- Identify the family, then sketch it}

\begin{formulabox}
\textbf{Recognise the form before plotting:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item $r = a\cos\theta$ or $a\sin\theta$: a \textbf{circle} through the pole,
        offset along the $0^\circ$ or $90^\circ$ ray.
  \item $r = a\sin(k\theta)$ or $a\cos(k\theta)$: a \textbf{rose} --- $k$ petals when
        $k$ is odd, $2k$ petals when $k$ is even, each of length $|a|$.
  \item $r = a \pm b\cos\theta$: a \textbf{limacon}. Equal $a$ and $b$ gives a
        cardioid (cusp); $a/b < 1$ gives an inner loop; $a/b > 1$ gives no loop.
  \item Then sketch from four values: $\theta = 0^\circ, 90^\circ, 180^\circ,
        270^\circ$, plus the zeros and the maximum.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Sketch $r = 2 + 2\cos\theta$. Start with its greatest radius.
\step{Constant and coefficient are equal, so this is a cardioid; the sketch is
      anchored by the largest radius, which occurs where $\cos\theta$ is largest.}
\step{$\cos\theta$ peaks at $\theta = 0^\circ$, giving $r = 2 + 2(1)$; the curve then
      shrinks to 0 at $180^\circ$, the cusp.}
$\Rightarrow\quad \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $r = 3 + 3\sin\theta$, what is the greatest value of $r$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}
