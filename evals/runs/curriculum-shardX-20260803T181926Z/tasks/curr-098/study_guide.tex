\sstitleblock{Sorting Four-Sided Shapes by What They Share}

\skillheading{Sorting by square corners}
\begin{formulabox}
\textbf{Rule:} A \textbf{square corner} matches the corner of a sheet of paper.
Slide the paper corner in: if it fits with no gap and no overlap, that corner
is square. A four-sided shape with four square corners is a
\textbf{rectangle}. Long sides never spoil a square corner.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad How many of these cards have four square corners?\quad
\begin{tikzpicture}[scale=0.3, baseline=(current bounding box.south), every node/.style={font=\scriptsize}]
  \draw[thick] (0,0) -- (1.7,0) -- (1.7,1.7) -- (0,1.7) -- cycle;
  \node[below] at (0.85,-0.2) {A};
  \begin{scope}[shift={(3.0,0)}]
    \draw[thick] (0,0) -- (2.4,0) -- (2.4,1.2) -- (0,1.2) -- cycle;
    \node[below] at (1.2,-0.2) {B};
  \end{scope}
  \begin{scope}[shift={(6.4,0)}]
    \draw[thick] (0,0) -- (1.7,0) -- (2.3,1.6) -- (0.6,1.6) -- cycle;
    \node[below] at (1.15,-0.2) {C};
  \end{scope}
\end{tikzpicture}
\step{The question is about corners only, so test corners and ignore how long
the sides are.}
\step{A fits four times; B fits four times; C leans, so two corners are too
narrow and two too wide.}

A and B fit, which is $\ans{2}$
\end{examplebox}

\skillheading{Sorting by side lengths}
\begin{formulabox}
\textbf{Rule:} Compare the four sides. All four the same length means the shape
joins the ``four equal sides'' group: the \textbf{rhombus} and the
\textbf{square}. A leaning shape can still have four equal sides.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad How many of these cards have four sides the same
length?\quad
\begin{tikzpicture}[scale=0.3, baseline=(current bounding box.south), every node/.style={font=\scriptsize}]
  \draw[thick] (0,0) -- (1.7,0) -- (2.3,1.6) -- (0.6,1.6) -- cycle;
  \node[below] at (1.15,-0.2) {A};
  \begin{scope}[shift={(3.6,0)}]
    \draw[thick] (0,0) -- (2.4,0) -- (2.4,1.2) -- (0,1.2) -- cycle;
    \node[below] at (1.2,-0.2) {B};
  \end{scope}
  \begin{scope}[shift={(7.0,0)}]
    \draw[thick] (0,0) -- (1.7,0) -- (1.7,1.7) -- (0,1.7) -- cycle;
    \node[below] at (0.85,-0.2) {C};
  \end{scope}
\end{tikzpicture}
\step{The group is named by side length, so compare sides and leave the corners
out of it.}
\step{A leans but its four sides match; B has a long pair and a short pair;
C is a square.}

A and C fit, which is $\ans{2}$
\end{examplebox}

\skillheading{Sorting by parallel sides}
\begin{formulabox}
\textbf{Rule:} Two sides are \textbf{parallel} when they run the same way and
would never meet, like train rails. Check both pairs of opposite sides:
exactly one pair is a \textbf{trapezoid}, two pairs is a
\textbf{parallelogram} (rectangles, rhombuses and squares are all in it).
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad How many of these have two pairs of parallel
sides?\quad
\begin{tikzpicture}[scale=0.3, baseline=(current bounding box.south), every node/.style={font=\scriptsize}]
  \draw[thick] (0,0) -- (2.2,0) -- (2.8,1.4) -- (0.6,1.4) -- cycle;
  \node[below] at (1.4,-0.2) {A};
  \begin{scope}[shift={(4.2,0)}]
    \draw[thick] (0,0) -- (2.6,0) -- (2.0,1.5) -- (0.6,1.5) -- cycle;
    \node[below] at (1.3,-0.2) {B};
  \end{scope}
  \begin{scope}[shift={(7.6,0)}]
    \draw[thick] (0,0) -- (1.7,0) -- (1.7,1.7) -- (0,1.7) -- cycle;
    \node[below] at (0.85,-0.2) {C};
  \end{scope}
\end{tikzpicture}
\step{Two pairs are asked for, so check both pairs on every card --- one pair
is not enough to stop.}
\step{A: top and bottom run the same way, and so do the slanted sides. B: only
the top and bottom do. C: both pairs do.}

A and C fit, which is $\ans{2}$
\end{examplebox}

\skillheading{Reading a shape-sort graph}
\begin{formulabox}
\textbf{Rule:} Each bar or tally row is one group, and its number tells how
many shapes went in it. ``How many in all'' means add every group; ``how many
more'' means subtract the smaller number from the larger one.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A class sorted its cards: squares $7$, trapezoids $4$.
How many more squares than trapezoids?
\step{``How many more'' compares two groups, so this is a subtraction.}
\step{Larger group $7$, smaller group $4$, so subtract: $7 - 4$.}

There are $\ans{3}$ more squares
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Same sort --- squares $7$, trapezoids $4$. How many cards
in all?\quad
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}
