\problem[5cm]{Write $18x^3 + 30x^2$ as a product of its greatest common monomial factor and one
remaining polynomial factor.}

\problem[4cm]{Kian writes the polynomial as $3x(2x-5)$. Nina writes it as $6x^2 - 15x$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Multiply out Kian's form to confirm the two expressions are equivalent. \ansline
  \item[(b)] One form shows the factors of the polynomial; the other shows its degree and its
             coefficients. Say which is which, and explain what each form makes easy to see. \ansline
\end{itemize}}

\problem[6.5cm]{Multiply $(2x+3)(x-5)$ and write the result in standard form.
State it so that the coefficient of each power of $x$ is visible.}

\problem[5cm]{$Q(x) = (x+4)(x-1)(x+2)$. The constant term of a polynomial is its value at $x = 0$.
Use that fact to find the constant term of $Q$ without multiplying the whole product out.}

\par\medskip

\par\medskip}

\problem[5cm]{Factor $x^2 - 3x - 28$ into two binomial factors.}

\problem[4.5cm]{Consider $(x+3)^2 - (x-3)^2$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Expand both squares and simplify the difference. \ansline
  \item[(b)] Both starting pieces have degree $2$, yet the answer does not. Explain what happened
             to the squared terms. \ansline
\end{itemize}}

\problem[5cm]{A rectangle has area $6x^2 + 19x + 10$ and one side of length $(2x+5)$.
\par\smallskip

  \item[(a)] Find the length of the other side. \ansline
  \item[(b)] The area has degree $2$ and each side has degree $1$. Explain why the degrees of the
             two sides must add up to the degree of the area. \ansline
\end{itemize}}
