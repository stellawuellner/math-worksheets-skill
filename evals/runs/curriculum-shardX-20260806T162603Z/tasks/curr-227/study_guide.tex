\sstitleblock{Systems by Graphing}

\skillheading{Skill 1 --- Reading a crossing point off a graph}

\begin{formulabox}
\textbf{Rule:} the crossing point lies on both lines, so its coordinates
satisfy both equations. Read $x$ across, then $y$ up, and always substitute to
check.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Read the solution of $y = x + 3$ and $y = 7 - x$ from
the graph.
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=7, ymin=0, ymax=8,
  xtick={0,1,2,3,4,5,6,7}, ytick={0,2,4,6,8},
  grid=both, grid style={line width=0.25pt, draw=gray!55},
  major grid style={line width=0.4pt, draw=gray!70},
  tick label style={font=\tiny}, xlabel={$x$}, ylabel={$y$},
  width=5.6cm, height=4.2cm, enlargelimits=false]
  \addplot[thick, blue, domain=0:5, samples=2] {x + 3};
  \addplot[thick, red, domain=0:7, samples=2] {7 - x};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The lines cross at a grid corner, so read the coordinates there rather
than solving --- then confirm them in both equations.}
\step{Two across and five up: $5 = 2 + 3$ and $5 = 7 - 2$, both true.}

On both lines at once. \quad $\ans{(2,\, 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = x + 1$ and $y = 5 - x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,\, 3)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Graphing two lines from $y = mx + b$}

\begin{formulabox}
\textbf{Rule:} plot the $y$-intercept, then step by the slope (right by the
denominator, up by the numerator). Do this for both lines on one grid and read
where they meet.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $y = 2x + 1$ and $y = x + 4$ by graphing.
\step{Both equations are already solved for $y$, so each one hands over a
starting dot and a step --- no rearranging is needed first.}
\step{Start at 1 stepping up 2, and at 4 stepping up 1. The dots coincide at
three across, seven up; $2x + 1 = x + 4$ confirms $x = 3$.}

Checked in both equations. \quad $\ans{(3,\, 7)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = 2x$ and $y = x + 3$ by graphing.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\, 6)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Graphing a line given as $Ax + By = C$}

\begin{formulabox}
\textbf{Rule:} do not fight the form. Set $x = 0$ to get one point, set
$y = 0$ to get another, draw the line through them --- or solve for $y$ first
if the numbers are friendly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x + y = 6$ and $y = 2x$ by graphing.
\step{Only the first equation is in standard form, so get its two intercepts
instead of rearranging it.}
\step{Intercepts $(0, 6)$ and $(6, 0)$; the other line runs through the origin
rising 2 each step. They meet where $x + 2x = 6$, so $x = 2$.}

Two across, four up. \quad $\ans{(2,\, 4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x + y = 10$ and $y = 4x$ by graphing.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,\, 8)}$}
\end{tryitbox}

\skillheading{Skill 4 --- When the lines never meet}

\begin{formulabox}
\textbf{Rule:} equal slopes with different intercepts give parallel lines and
\emph{no solution}. Identical equations give one line drawn twice and
\emph{infinitely many} solutions. Say which, rather than guessing a point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $y = 2x + 1$ and $y = 2x + 5$ by graphing.
\step{Compare the slopes before drawing anything: both are 2, so the lines can
only be parallel or identical.}
\step{The intercepts differ, 1 against 5, so the lines stay 4 units apart and
never cross.}

Nothing lies on both lines. \quad $\ans{\text{no solution}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = 4x + 2$ and $y = 4x + 7$ by graphing.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{no solution}}$}
\end{tryitbox}
