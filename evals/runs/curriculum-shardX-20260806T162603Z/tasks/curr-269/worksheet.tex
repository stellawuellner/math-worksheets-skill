\wstitleblock{Function Notation Is Not Multiplication}{Algebra 1}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Every problem below shows work that treats $f(x)$ as ``$f$ times $x$''. Your job is to repair the computation \emph{and} to say what the notation really asks for.
  \item $f(3)$ means: put $3$ in for the input everywhere it appears in the rule, then simplify. The letter naming the function is never a factor.
  \item Write the corrected value on the blank, and the diagnosis on the ruled space.
\end{itemize}

\vspace{0.2cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[0.2cm]{Let $f(x) = 4x - 7$. A student wrote
\[ f(3) = f \cdot 3 = 3f. \]
(a) Evaluate $f(3)$ correctly. \quad $f(3) =$~\ansblank
\par\vspace*{2.2cm}
(b) In one sentence, say what the notation $f(3)$ actually instructs you to do.
\par\vspace*{2cm}}

\problem[0.2cm]{Let $g(x) = x^2 - 2x$. A student wrote
\[ g(5) = g \cdot 5^2 - 2 \cdot g \cdot 5 = 25g - 10g = 15g. \]
(a) Evaluate $g(5)$ correctly. \quad $g(5) =$~\ansblank
\par\vspace*{2.4cm}
(b) Name the error in the student's first step.
\par\vspace*{2cm}}

\problem[0.2cm]{Let $f(x) = x^2 + 1$. A student claims that $f(2+3) = f(2) + f(3)$, ``because the parentheses just multiply''.
\par\vspace*{0.1cm}
(a) Compute both sides. \quad $f(5) =$~\ansblank \quad and \quad $f(2) + f(3) =$~\ansblank
\par\vspace*{2.6cm}
(b) Explain why the claim fails for this function.
\par\vspace*{2.2cm}}

\problem[0.2cm]{Let $f(x) = 6x - 4$. Asked to solve $f(x) = 20$, a student wrote
\[ f \cdot x = 20, \qquad\text{so}\qquad x = \frac{20}{6} = \frac{10}{3}. \]
(a) Solve $f(x) = 20$ correctly. \quad $x =$~\ansblank
\par\vspace*{2.6cm}
(b) Name the error in the student's first line.
\par\vspace*{2cm}}

\problem[0.2cm]{A print shop charges a one-time setup fee plus a fixed price for each shirt. The cost of an order of $n$ shirts, in dollars, is $C(n) = 12n + 25$. A student found the cost of an order of $4$ shirts by writing $C \cdot 4 = 4C$.
\par\vspace*{0.1cm}
(a) Find $C(4)$, the cost of that order in dollars. \quad $C(4) =$~\ansblank
\par\vspace*{2.4cm}
(b) Explain what the $25$ represents, and why $C(4)$ is not ``$4$ times $C$''.
\par\vspace*{2.4cm}}

\problem[0.2cm]{A tank holds $45$ litres and drains at $5$ litres per minute, so the amount left after $t$ minutes is $A(t) = 45 - 5t$. Asked when the tank is empty, a student wrote ``$A$ times $t$ is $0$, so $t = 0$''.
\par\vspace*{0.1cm}
(a) Solve $A(t) = 0$ correctly, in minutes.
\par\vspace*{2.6cm}\answerline{min}
\par\vspace*{0.25cm}
(b) Name the error in the student's reasoning.
\par\vspace*{2.2cm}}

\problem[0.2cm]{Let $f(x) = 4x - 12$. The two questions below look almost alike and are not the same question.
\par\vspace*{0.1cm}
(a) Find $f(0)$. \quad $f(0) =$~\ansblank
\par\vspace*{2.2cm}
(b) Solve $f(x) = 0$. \quad $x =$~\ansblank
\par\vspace*{2.4cm}
(c) In one sentence, explain the difference between what (a) asks for and what (b) asks for.
\par\vspace*{2.2cm}}

\problem[0.2cm]{Let $f(x) = x^2 + 2x$. A student argues: ``$f(x)$ means $f$ times $x$, so doubling the input doubles the output --- that is, $f(6) = 2f(3)$.''
\par\vspace*{0.1cm}
(a) Compute both values. \quad $f(3) =$~\ansblank \quad and \quad $f(6) =$~\ansblank
\par\vspace*{2.6cm}
(b) Show that $f(6) \neq 2f(3)$, and explain what $f(6)$ really instructs you to do.
\par\vspace*{2.6cm}}
