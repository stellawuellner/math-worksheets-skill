% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Is Direct Substitution Even Legal?}

\begin{formulabox}
\textbf{The test:} substitute the target into the \emph{denominator} first.
If it is not zero, the function is continuous there and the limit is the value
--- substitute and stop. If it is zero and the numerator is not, you have a
vertical asymptote. If both are zero you have $\frac{0}{0}$, which is an
\textbf{indeterminate form}: it is not an answer, it is a message saying
``rewrite me''. Factor and cancel, or multiply by the conjugate.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to 3} \frac{x^2-9}{x-3}$
\step{Substituting $3$ gives $\frac{0}{0}$, so substitution has failed --- that is a signal to rewrite, never a value.}
\step{$\dfrac{x^2-9}{x-3} = \dfrac{(x-3)(x+3)}{x-3} = x+3$ for $x \neq 3$, and $3+3 = 6$.}

$\Rightarrow\quad$ $\ans{6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to 7} \frac{x^2-49}{x-7}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{14}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- The Sign of the Denominator Decides}

\begin{formulabox}
\textbf{The test:} at a vertical asymptote the size is always infinite, so the
only question is the \emph{direction}. Read two signs, not one: the sign the
numerator is heading to, and the sign of the denominator \emph{on the side you
are approaching from}. Same signs give $+\infty$; opposite signs give $-\infty$.
An even power in the denominator is positive on both sides.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\displaystyle F(x) = \frac{5}{x-1}$ find the limit
from each side of $1$.
\step{The numerator is a fixed positive $5$, so both answers are decided by the sign of $x-1$ on each side.}
\step{Left of $1$: $x-1$ is a small negative, so $5$ over it is large negative. Right of $1$: $x-1$ is a small positive, so it is large positive.}

$\Rightarrow\quad$ left: $\ans{-\infty}$ \quad right: $\ans{+\infty}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $G(x) = \dfrac{3}{x+4}$ find the limit from each side
of $-4$.\\[2pt]
\rotatebox{180}{\footnotesize check: left $\ans{-\infty}$, \ right $\ans{+\infty}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Diagnosing Someone Else's Work}

\begin{formulabox}
\textbf{How to write a find-and-fix answer:} a complete answer has two parts,
and both are graded. \textbf{Fix:} redo the limit correctly and state the value.
\textbf{Diagnose:} name the specific move that was illegal --- ``treated
$\frac{0}{0}$ as a value'', ``cancelled across a $+$ instead of a factor'',
``never checked the sign of the denominator''. ``It is wrong'' is not a
diagnosis.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A student says $\displaystyle\lim_{x \to 3^+}
\frac{x-8}{x-3} = +\infty$ ``because it blows up''. Fix it.
\step{The size is right but the direction was never argued, so check both signs on the right of $3$.}
\step{Numerator near $3-8 = -5$, negative; denominator $x-3$ a small positive. Negative over positive is large negative, so the direction is downward.}

$\Rightarrow\quad$ $\ans{-\infty}$ --- the diagnosis is that only the denominator was looked at.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A student says $\displaystyle\lim_{x \to 1^-}
\frac{x-9}{x-1} = -\infty$ ``because the top is negative''. Fix it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{+\infty}$}
\end{tryitbox}
