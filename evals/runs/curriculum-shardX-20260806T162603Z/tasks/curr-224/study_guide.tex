\sstitleblock{Which Number Is the Slope?}

\begin{formulabox}
\textbf{In $y = mx + b$, position decides the role:}\par
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item $m$ \emph{multiplies} the variable. It is the slope: the change in $y$
        for each 1 unit of $x$. On a graph it is the steepness.
  \item $b$ is \emph{added on}. It is the $y$-intercept: the value of $y$ when
        $x$ is 0. On a graph it is where the line crosses the vertical axis.
  \item Test for $b$: substitute $x = 0$ and see what survives.
\end{itemize}
\end{formulabox}

\skillheading{Skill 1 --- Reading both numbers out of an equation}

\begin{formulabox}
\textbf{Rule:} the coefficient of the variable is the slope; the lone number
is the $y$-intercept. Swapping them changes the line completely, so check by
substituting $x = 0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the slope and $y$-intercept of $y = 6x + 11$.
\step{Two different questions are being asked, so answer them with two
different moves: one about a change, one about a single value.}
\step{Change: going from $x$ to $x + 1$ adds 6 to $y$. Value at zero:
$6(0) + 11 = 11$.}

Slope $\ans{m = 6}$, \quad $y$-intercept $\ans{b = 11}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the slope and $y$-intercept of $y = 4x + 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = 4}$ and $\ans{b = 9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $y = 5x + 2$ and $y = 2x + 5$ use the same two
numbers and are different lines. Never read a pair of numbers without asking
which one is multiplying.
\end{watchoutbox}

\skillheading{Skill 2 --- Building the equation from two points}

\begin{formulabox}
\textbf{Rule:} the slope is the change in $y$ over the change in $x$. When one
of the points sits on the $y$-axis, its $y$-value is $b$ --- it goes on its
own, not in front of $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line passes through $(0, 7)$ and $(4, 23)$. Find its
slope.
\step{One point already has $x = 0$, so its height is the intercept; the slope
still has to come from the change between the two points.}
\step{$\dfrac{23 - 7}{4 - 0} = \dfrac{16}{4}$, so the equation is
$y = 4x + 7$.}

The 7 stays alone; the 4 multiplies. \quad $\ans{y = 4x + 7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line passes through $(0, 5)$ and $(3, 20)$. Write its
equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 5x + 5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Building the equation from a table}

\begin{formulabox}
\textbf{Rule:} the row where the input is 0 gives $b$; the step from one row
to the next, divided by the step in the input, gives $m$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $m$ and $b$ for this table.
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 \\ \hline
$y$ & 8 & 13 & 18 & 23 \\ \hline
\end{tabular}
\end{center}
\step{The table already contains the input value 0, so read $b$ straight off
and get $m$ from the change across the whole table.}
\step{At $x = 0$ the output is 8. From the first column to the last,
$\dfrac{23 - 8}{3 - 0} = \dfrac{15}{3}$.}

$\ans{b = 8}$ \quad and \quad $\ans{m = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $x = 0, 2, 4$ with $y = 12, 20, 28$. Find
$b$ and $m$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{b = 12}$ and $\ans{m = 4}$}
\end{tryitbox}
