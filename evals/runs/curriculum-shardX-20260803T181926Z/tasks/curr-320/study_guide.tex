\sstitleblock{Similar Triangles: Sides, Perimeter, and Area}

\begin{formulabox}
\textbf{One factor runs the whole picture.} If two triangles are similar with
scale factor $k$ (new $\div$ old), then every side $\times k$, the perimeter
$\times k$, and the area $\times k^2$.\par
\textbf{Going backwards:} from an area ratio, $k = \sqrt{\text{area ratio}}$.
\end{formulabox}

\skillheading{1. Scale factor and a missing side}

\begin{formulabox}
Write the similarity statement first --- $\triangle ABC \sim \triangle DEF$
pairs $A$ with $D$, $B$ with $E$, $C$ with $F$, so $AB$ pairs with $DE$.\par
Then set two matching ratios equal and cross-multiply:
$\frac{AB}{DE} = \frac{BC}{EF}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\triangle ABC \sim \triangle DEF$ with $AB = 4$,
$BC = 6$, $DE = 10$. Find $EF$.
\step{Two corresponding pairs are known ($AB$ with $DE$) and one is half
missing ($BC$ with $EF$), so a single proportion settles it.}
\step{$\dfrac{4}{10} = \dfrac{6}{EF}$, so $4 \cdot EF = 60$.}

$EF = \ans{15}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\triangle PQR \sim \triangle STU$ with $PQ = 9$,
$QR = 12$, $ST = 6$. Find $TU$.\\[2pt]
\rotatebox{180}{\footnotesize check: $TU = \ans{8}$}
\end{tryitbox}

\skillheading{2. Perimeter scales by $k$ --- not by $k^2$}

\begin{formulabox}
A perimeter is a sum of sides, so multiplying every side by $k$ multiplies the
total by $k$ as well.\par
This runs both ways: if you are given the two perimeters, their ratio IS $k$,
even when no single side is known.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Corresponding sides are $5$ cm and $8$ cm; the smaller
triangle's perimeter is $20$ cm. Find the larger perimeter.
\step{Perimeter is built from sides only, so the side ratio is the factor ---
no squaring belongs here.}
\step{$k = \dfrac{8}{5} = 1.6$, so the larger perimeter is
$20 \times 1.6$.}

The perimeter is $\ans{32}$ cm.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Corresponding sides $6$ cm and $9$ cm; the smaller
perimeter is $22$ cm. Find the larger perimeter.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{33}$ cm}
\end{tryitbox}

\skillheading{3. Area scales by $k$ squared}

\begin{formulabox}
Area covers two directions at once, so both of them stretch: the area ratio is
$k^2$. With $k = 3$, a triangle's sides triple but its area becomes $9$ times
as large.\par
Given the areas instead, undo the square: $k = \sqrt{\text{area ratio}}$, and
only then scale a length or a perimeter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Corresponding sides are $4$ cm and $10$ cm; the smaller
triangle's area is $12$ cm$^2$. Find the larger area.
\step{The question asks for an area, and areas scale by the square of the side
ratio, so square $k$ before multiplying.}
\step{$k = \dfrac{10}{4} = 2.5$ and $k^2 = 6.25$, so the area is
$12 \times 6.25$.}

The area is $\ans{75}$ cm$^2$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Corresponding sides $3$ cm and $6$ cm; the smaller area
is $18$ cm$^2$. Find the larger area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{72}$ cm$^2$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} do not scale an area by $k$, and do not scale a
perimeter by $k^2$. When a problem hands you \emph{areas} and asks for a
\emph{length}, take the square root first --- that is the step most often
skipped.
\end{watchoutbox}
