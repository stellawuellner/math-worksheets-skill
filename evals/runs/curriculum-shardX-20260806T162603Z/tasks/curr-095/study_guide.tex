% =================== SKILL 1: rectangle-area ===================
\skillheading{Rectangle Area}

\begin{formulabox}
\textbf{Rule:} Area of a rectangle $=$ length $\times$ width, in
\emph{square} units (cm$^2$, m$^2$, square units).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle is 6 cm long and 5 cm wide. What is its area?
\step{Area is the space covered inside, so multiply the length by the width.}
\step{$6 \times 5 = 30$.}

Area: $\ans{30\ \text{cm}^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle is 7 cm long and 3 cm wide. What is its
area?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21\ \text{cm}^2}$}
\end{tryitbox}

% =================== SKILL 2: rectangle-perimeter ===================
\skillheading{Rectangle Perimeter}

\begin{formulabox}
\textbf{Rule:} Perimeter $=$ add \emph{all four} sides:
$P = 2 \times (\text{length} + \text{width})$, in plain length units.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rectangle is 8 cm long and 2 cm wide. What is its
perimeter?
\step{Perimeter is the distance around the edge, so add two lengths and two
widths.}
\step{$2 \times (8 + 2) = 2 \times 10 = 20$.}

Perimeter: $\ans{20\ \text{cm}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rectangle is 6 m long and 5 m wide. What is its
perimeter?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{22\ \text{m}}$}
\end{tryitbox}

% =================== SKILL 3: composite-area ===================
\skillheading{Composite Area: Split, Then Add}

\begin{formulabox}
\textbf{Rule:} Split an L-shape (or any rectilinear shape) into rectangles
with a straight cut. Find each rectangle's area, then \textbf{add} the pieces.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An L-shape splits into a bottom rectangle 6 units by
2 units and a top rectangle 3 units by 2 units. What is its area?
\step{The shape is not one rectangle, so cut it into two rectangles and add
their areas.}
\step{Bottom: $6 \times 2 = 12$. Top: $3 \times 2 = 6$. Add: $12 + 6 = 18$.}

Area: $\ans{18\ \text{square units}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} An L-shape splits into a rectangle 5 units by 2 units and a
rectangle 2 units by 3 units. What is its area?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16\ \text{square units}}$}
\end{tryitbox}

% =================== SKILL 4: composite-perimeter ===================
\skillheading{Composite Perimeter: Walk Every Side}

\begin{formulabox}
\textbf{Rule:} Walk all the way around and \textbf{add every side}. If a side
label is missing, find it first: opposite directions must match (the small
tops must add up to the whole bottom, the small rights to the whole left).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} An L-shape has sides 6, 2, 3, 2, 3, and 4 cm. What is its
perimeter?
\step{Perimeter needs every side of the walk around the shape, so add all six
labels --- none may be skipped.}
\step{$6 + 2 + 3 + 2 + 3 + 4 = 20$.}

Perimeter: $\ans{20\ \text{cm}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} An L-shape has sides 8, 3, 4, 2, 4, and 5 cm. What is its
perimeter?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{26\ \text{cm}}$}
\end{tryitbox}
