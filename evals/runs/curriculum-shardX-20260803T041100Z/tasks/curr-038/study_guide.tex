\skillheading{Skill 1 --- Count on with a ten-frame}
\begin{formulabox}\textbf{Model:}\\[4pt]
part $+$ \fbox{?} $=$ whole. The counters you can see are the \textbf{known part};
the total you want is the \textbf{whole}.\par
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Start at the known part. Do \emph{not} say it as your first count.
  \item Count on one at a time until you say the whole. The number of counts is the missing addend.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $7 + \underline{\hspace{1cm}} = 12$
\step{The known part is 7 and the whole is 12, so count on from 7 up to 12 and keep track of how many counts.}
\step{8, 9, 10, 11, 12 --- five counts, so the missing addend is $\ans{5}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $8 + \underline{\hspace{1cm}} = 15$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}\end{tryitbox}

\skillheading{Skill 2 --- Subtract to find the part}
\begin{formulabox}\textbf{Rule:}\\[4pt]
whole $-$ known part $=$ missing part.\par
Every missing-addend equation can be rewritten as a subtraction, and the subtraction
is usually faster when the missing part is large.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\underline{\hspace{1cm}} + 9 = 17$
\step{The whole is 17 and one part is 9, so take the known part away from the whole instead of counting one by one.}
\step{$17 - 9 = 8$, and the check $9 + 8 = 17$ works, so the missing part is $\ans{8}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\underline{\hspace{1cm}} + 4 = 13$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}\end{tryitbox}

\skillheading{Skill 3 --- Missing addends in a story}
\begin{formulabox}\textbf{How to set it up:}\\[4pt]
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Find the \textbf{whole}: the number the story wants \emph{in all}.
  \item Find the \textbf{known part}: what the person already has.
  \item Write part $+$ \fbox{?} $=$ whole, then subtract. Answer with the unit from the story.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad Ben has 6 stickers. He wants 14 stickers in all. How many more?
\step{``In all'' names the whole (14 stickers) and ``has'' names the known part (6 stickers), so this is $6 + ? = 14$.}
\step{$14 - 6 = 8$, and $6 + 8 = 14$ checks, so Ben needs $\ans{8}$ more stickers.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Rosa has 9 shells and wants 16 shells in all. How many more shells?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} The two numbers you can see are \emph{not}
both parts. The bigger one is the whole, so do not add them. In $5 + ? = 13$ the
answer is $13 - 5 = 8$, never $5 + 13$.
\end{watchoutbox}
