\sstitleblock{Comparing Distributions and Compound Events}

\noindent\small Two questions decide every problem on the worksheet. For data:
am I describing \emph{where} the distribution sits (center) or \emph{how spread
out} it is? For chance: is this one event, or two stages happening together?
\normalsize

\skillheading{1. Comparing the centers of two distributions}
\begin{formulabox}
\textbf{Mean} $=$ (sum of the values) $\div$ (how many values). \quad
\textbf{Median} $=$ the middle value of the \emph{ordered} list.\\[3pt]
With an \textbf{even} count there are two middle values: the median is halfway
between them, so it need not be a value anyone actually got.\\[3pt]
\textbf{(!)} A few unusually high or low values drag the \emph{mean} toward
them but barely move the \emph{median}. When the two disagree, that gap is
itself information about the distribution.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Find the median of $4,\ 7,\ 9,\ 10,\ 12,\ 15$.
\step{The list is already ordered and there are 6 values --- an even count, so
the median sits between the two middle ones rather than on a value.}
\step{The 3rd and 4th values are $9$ and $10$, and halfway between them is
$(9+10) \div 2$.}
\par
$\ans{9.5}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Find the median of $5,\ 8,\ 9,\ 13,\ 14,\ 19$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

\skillheading{2. Comparing the spreads of two distributions}
\begin{formulabox}
\textbf{Range} $=$ greatest value $-$ least value. One pair of extremes decides
it, so a single unusual value can double it.\\[3pt]
\textbf{Interquartile range} $=$ $Q_3 - Q_1$, the width of the middle half.
Split the ordered list at the median; $Q_1$ is the median of the lower half and
$Q_3$ is the median of the upper half. The extremes are deliberately left
out.\\[3pt]
\textbf{Same center, different spread} is the usual way two data sets differ.
The smaller IQR belongs to the more consistent set.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} Find the IQR of $3,\ 5,\ 8,\ 10,\ 12,\ 14,\ 15,\ 19$.
\step{IQR is about the middle half, so I split the ordered list in two and take
the median of each half rather than using the extremes.}
\step{Lower half $3, 5, 8, 10$ gives $Q_1 = (5+8) \div 2 = 6.5$; upper half
$12, 14, 15, 19$ gives $Q_3 = (14+15) \div 2 = 14.5$; and $14.5 - 6.5$ is}
\par
$\ans{8}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} Find the IQR of $2,\ 4,\ 6,\ 8,\ 12,\ 14,\ 16,\ 18$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{3. Probability of a single event}
\begin{formulabox}
$P(\text{event}) = \dfrac{\text{number of favourable outcomes}}{\text{number of
equally likely outcomes}}$, always written in lowest terms.\\[3pt]
Build the denominator before anything else: it counts \emph{outcomes}, not
categories. Four clubs with 40 students in them give a denominator of 40, not
of 4.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} A jar holds 10 counters and 4 of them are yellow. One counter
is drawn at random. Find $P(\text{yellow})$.
\step{One draw, so this is a simple probability: count the favourable outcomes
over all the equally likely outcomes.}
\step{$4$ yellow out of $10$ counters gives $\dfrac{4}{10}$, which reduces by a
factor of $2$.}
\par
$\ans{\dfrac{2}{5}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} A jar holds 8 counters and 3 of them are green. Find
$P(\text{green})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{3}{8}}$}
\end{tryitbox}

\skillheading{4. Probability of a compound event}
\begin{formulabox}
Two stages happening together: find each stage's probability and
\textbf{multiply}. Adding gives the probability of \emph{one or the other},
which is a different question.\\[3pt]
\textbf{Independent} (spin then roll, or draw \emph{with} replacement): the
second probability is unchanged. \textbf{Dependent} (draw \emph{without}
replacement): recount the second stage from what is left --- both the top and
the bottom of the fraction shrink.\\[3pt]
\textbf{``At least one''} is fastest through its opposite:
$P(\text{at least one}) = 1 - P(\text{none})$.
\end{formulabox}
\begin{examplebox}
\textbf{Example:} A fair coin is flipped and a number cube is rolled. Find
$P(\text{heads and a } 6)$.
\step{Two stages happen together and the coin cannot change the cube, so the
stages are independent and their probabilities multiply.}
\step{$P(\text{heads}) = \dfrac{1}{2}$ and $P(6) = \dfrac{1}{6}$, so the
product is $\dfrac{1}{2} \times \dfrac{1}{6}$.}
\par
$\ans{\dfrac{1}{12}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:} A spinner has 4 equal sections and 1 is blue. It is spun once
and a number cube is rolled. Find $P(\text{blue and an even number})$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{1}{8}}$}
\end{tryitbox}
