% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Repeated Differentiation}

\begin{formulabox}
\textbf{Notation.} $f'$, $f''$, $f'''$, then $f^{(4)}$, $f^{(5)}$, \ldots\par
Each one is the derivative of the one before it: $f'' = (f')'$ and
$f''' = (f'')'$. Differentiate once, simplify, differentiate the result.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $g''(x)$ for $g(x) = x^5 - 4x^3 + 2x$.
\step{The order asked for is two, so this is the power rule applied twice --- no new tool is needed.}
\step{$g'(x) = 5x^4 - 12x^2 + 2$, and differentiating that gives $20x^3 - 24x$.}

$\Rightarrow\quad \ans{g''(x) = 20x^3 - 24x}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $h''(x)$ for $h(x) = x^4 - 6x^2 + 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h''(x) = 12x^2 - 12}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Acceleration from a Position Model}

\begin{formulabox}
\textbf{Rule.} If $s(t)$ is position, then $v(t) = s'(t)$ is velocity and
$a(t) = s''(t)$ is acceleration.\par
Units divide once per derivative: metres $\rightarrow$ metres per second
$\rightarrow$ metres per second squared.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $s(t) = t^3 - 6t^2 + 5t$ metres. Find $a(t)$ and $a(4)$.
\step{Acceleration is the second derivative of position, so differentiate twice and only then substitute the time.}
\step{$s'(t) = 3t^2 - 12t + 5$, so $a(t) = s''(t) = 6t - 12$; at $t = 4$, $a(4) = 24 - 12$.}

$\Rightarrow\quad \ans{a(t) = 6t - 12}$ \quad and \quad $\ans{a(4) = 12}$ metres per second squared.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $s(t) = 2t^3 - 9t^2 + 4t$ metres. Find $a(t)$ and $a(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{a(t) = 12t - 18}$, $\ans{a(3) = 18}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Second Derivative of an Implicit Curve}

\begin{formulabox}
\textbf{Recipe.} Differentiate the equation term by term, remembering
$\dfrac{d}{dx}\big(y^n\big) = ny^{n-1}\dfrac{dy}{dx}$. Solve for
$\dfrac{dy}{dx}$, differentiate again, then replace any surviving
$\dfrac{dy}{dx}$ by the expression you already found.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\dfrac{d^2y}{dx^2}$ for the curve $xy = 6$.
\step{The curve solves cleanly for $y$, so the fastest honest route is to write $y$ explicitly and differentiate twice; implicit differentiation gives the same answer.}
\step{$y = 6x^{-1}$, so $y' = -6x^{-2}$ and $y'' = 12x^{-3}$.}

$\Rightarrow\quad \ans{y'' = \tfrac{12}{x^3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\dfrac{d^2y}{dx^2}$ for the curve $xy = 10$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y'' = \tfrac{20}{x^3}}$}
\end{tryitbox}
