\sstitleblock{Complex Numbers: Complete Solution Sets}

\skillheading{1. Multiplying complex numbers}

\begin{formulabox}
\textbf{The one rule:} $i^2 = -1$, so $\sqrt{-k} = i\sqrt{k}$ for $k > 0$.\\[3pt]
\textbf{Multiply} like binomials, then replace $i^2$: \
$(a + bi)(c + di) = (ac - bd) + (ad + bc)i$.\\[3pt]
\textbf{Powers of $i$ cycle with period 4:} $i^1 = i$, $i^2 = -1$,
$i^3 = -i$, $i^4 = 1$. Divide the exponent by $4$ and keep the remainder.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Multiply $(3 + 2i)(1 - 4i)$.
\step{Two binomials, so distribute all four products --- the only special step
is turning $i^2$ back into a real number.}
\step{$3 \cdot 1 = 3$, \ $3 \cdot (-4i) = -12i$, \ $2i \cdot 1 = 2i$, \
$2i \cdot (-4i) = -8i^2 = +8$. Real: $3 + 8 = 11$; imaginary: $-12i + 2i = -10i$.}

$\Rightarrow\quad (3 + 2i)(1 - 4i) = \ans{11 - 10i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Multiply $(5 + i)(2 - 3i)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13 - 13i}$}
\end{tryitbox}

\skillheading{2. Quadratics with complex solutions}

\begin{formulabox}
\textbf{Quadratic formula:} $x = \dfrac{-b \pm \sqrt{b^2 - 4ac}}{2a}$.\\[3pt]
\textbf{Read the discriminant} $\Delta = b^2 - 4ac$ first: $\Delta > 0$ two
real solutions; $\Delta = 0$ one repeated real solution; $\Delta < 0$ a
\emph{conjugate pair} $p + qi$ and $p - qi$. There is never ``no solution''
once complex numbers are allowed.\\[3pt]
\textbf{Shortcut:} if the variable sits inside a single square, take square
roots of both sides instead --- and keep both signs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $x^2 + 2x + 10 = 0$.
\step{Nothing multiplies to $10$ and adds to $2$, so the quadratic formula is
the tool; check the discriminant to see what kind of answer to expect.}
\step{$\Delta = 4 - 40 = -36$, so $\sqrt{\Delta} = 6i$ and
$x = \dfrac{-2 \pm 6i}{2} = -1 \pm 3i$.}

$\Rightarrow\quad \ans{x = -1 + 3i}$ \ or \ $\ans{x = -1 - 3i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $x^2 - 6x + 34 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3 + 5i \ \text{or} \ x = 3 - 5i}$}
\end{tryitbox}

\skillheading{3. Testing a proposed solution}

\begin{formulabox}
\textbf{Substitute and simplify to one number.} A value is a solution exactly
when the left side collapses to $0$.\\[3pt]
\textbf{Square carefully:} $(p + qi)^2 = p^2 + 2pqi + q^2i^2 = (p^2 - q^2) +
2pq\,i$ --- the $q^2$ term changes sign, and that sign change is the whole
check.\\[3pt]
\textbf{Free information:} with real coefficients, if $p + qi$ is a solution
then $p - qi$ is one too. Never report just one of a pair.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Is $x = 2 - i$ a solution of $x^2 - 4x + 5 = 0$?
\step{``Is it a solution?'' is settled by substitution, not by re-solving ---
replace $x$ and simplify.}
\step{$(2 - i)^2 = 4 - 4i + i^2 = 3 - 4i$; \ $-4(2 - i) = -8 + 4i$; \ adding
$5$ gives $(3 - 8 + 5) + (-4 + 4)i$.}

$\Rightarrow\quad$ the substitution gives $\ans{0}$, so yes, it is a solution.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Substitute $x = -1 + 2i$ into $x^2 + 2x + 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0}$, so it is a solution}
\end{tryitbox}

\skillheading{4. Reporting the complete solution set}

\begin{formulabox}
\textbf{Count first:} a degree-$n$ polynomial equation has exactly $n$
solutions, counted with multiplicity. If your list is short, you stopped
early.\\[3pt]
\textbf{Route:} factor as far as you can over the reals; every leftover
$x^2 + k$ contributes the pair $x = \pm i\sqrt{k}$.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item Factor out the common $x$ before dividing by it, or the solution
$x = 0$ disappears.
\item Only even powers? Factor as a quadratic in $x^2$ first.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find every zero of $x^4 - 1$.
\step{Degree $4$, so four zeros are owed; it is a difference of squares, so
factor rather than expand.}
\step{$x^4 - 1 = (x^2 - 1)(x^2 + 1)$. The first gives $x = 1$ and $x = -1$;
the second gives $x^2 = -1$, so $x = \pm i$.}

$\Rightarrow\quad \ans{x = 1,\ -1,\ i,\ -i}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find every zero of $x^4 + 5x^2 + 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = i,\ -i,\ 2i,\ -2i}$}
\end{tryitbox}
