\sstitleblock{Derivative Applications and Modeling}

\begin{formulabox}
\textbf{Which kind of problem is this?} Read the question before touching the
algebra --- all three types differentiate, and they differ in what you
differentiate \emph{with respect to}.\\[3pt]
$\bullet$ \textbf{Related rates:} two quantities change in time. Differentiate
the relation with respect to $t$, keeping every changing letter.\\[2pt]
$\bullet$ \textbf{Optimization:} one quantity is largest or smallest. Reduce to
one variable, set the derivative to zero, then check the domain.\\[2pt]
$\bullet$ \textbf{Curve analysis:} the shape of a graph. $f'$ gives
increasing/decreasing and critical points; $f''$ gives concavity and inflection
points.
\end{formulabox}

\skillheading{1. Curve analysis with $f'$ and $f''$}

\begin{formulabox}
Critical numbers: where $f'(x) = 0$ or $f'$ is undefined. $f' > 0$ means $f$
increases; $f' < 0$ means it decreases.\\[3pt]
$f'' > 0$ is concave up, $f'' < 0$ concave down, and an inflection point needs
$f''$ to \emph{change sign} --- $f''(x) = 0$ by itself is not enough
($g(x) = x^4$ at the origin is the standard counterexample).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the critical numbers of $f(x) = 2x^3 - 9x^2 + 12x$.
\step{The question is about the shape of the graph, so the tool is $f'$, and
critical numbers are exactly its zeros for a polynomial.}
\step{$f'(x) = 6x^2 - 18x + 12 = 6(x - 1)(x - 2)$, which is zero when a factor
is zero.}

$\Rightarrow\quad \ans{x = 1 \text{ and } x = 2}$ \quad ($f'$ is $+$, then $-$,
then $+$: a maximum at $1$, a minimum at $2$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the critical numbers of $g(x) = x^3 - 12x + 5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -2 \text{ and } x = 2}$}
\end{tryitbox}

\skillheading{2. Related rates}

\begin{formulabox}
\textbf{Order matters.} Write the geometric relation, use any fixed ratio to
remove extra variables, differentiate with respect to $t$, and only \emph{then}
substitute the instant's numbers. Substituting first freezes a variable that is
still moving.\\[3pt]
Circle: $A = \pi r^2 \Rightarrow \frac{dA}{dt} = 2\pi r\,\frac{dr}{dt}$. \quad
Sphere: $V = \frac{4}{3}\pi r^3 \Rightarrow
\frac{dV}{dt} = 4\pi r^2\,\frac{dr}{dt}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A circular ripple's radius grows at $3$ cm/s. How fast
is its area growing when $r = 4$ cm?
\step{Two quantities change with time and the link between them is the area
formula, so differentiate that link in $t$ before any number goes in.}
\step{$\frac{dA}{dt} = 2\pi r\,\frac{dr}{dt} = 2\pi(4)(3)$.}

$\Rightarrow\quad \ans{24\pi}$ square centimetres per second
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A square's side grows at $5$ in/s. How fast is its area
growing when the side is $7$ in? (Use $\frac{dA}{dt} = 2s\,\frac{ds}{dt}$.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{70}$}
\end{tryitbox}

\skillheading{3. Optimization with a domain check}

\begin{formulabox}
Name the quantity to be optimized, write it in \emph{one} variable using the
constraint, differentiate, and solve. Then finish the job: list the critical
numbers \emph{inside} the physical domain plus the endpoints, and compare --- or
give a sign chart for the derivative. A root outside the domain is discarded
with a reason, not ignored.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $40$ m of fence, three sides of a rectangle against a
wall. Which $x$ (the two equal sides) maximizes the area $A(x) = x(40 - 2x)$?
\step{The area is already a function of the single variable $x$ on
$0 < x < 20$, so the maximum can only be at a zero of $A'$ or at an endpoint.}
\step{$A(x) = 40x - 2x^2$, so $A'(x) = 40 - 4x = 0$.}

$\Rightarrow\quad \ans{x = 10}$ \quad ($A'' = -4 < 0$, so this critical point
is the absolute maximum, and both endpoints give area $0$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same fence problem with $24$ m of fence:
$A(x) = x(24 - 2x)$. Which $x$ maximizes the area?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the two most expensive habits on this topic are
substituting the instant's numbers into a related-rates relation \emph{before}
differentiating, and stopping an optimization at ``$f'(x) = 0$ here'' without
saying why that point is the absolute extreme on the domain. Both lose points
even when the number is right.
\end{watchoutbox}
