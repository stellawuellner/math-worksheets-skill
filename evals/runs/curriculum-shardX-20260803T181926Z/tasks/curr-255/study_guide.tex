\sstitleblock{Choosing the Form That Shows the Structure}

\begin{formulabox}
\textbf{One expression, two forms.} $(x-4)(x-5)$ and $x^2-9x+20$ are the same
expression written two ways. Neither is ``simpler'' --- they answer different
questions.\\[3pt]
\textbf{Expanded} (a sum of terms) shows: the degree, the leading coefficient,
and the constant term (the value at $x=0$).\\[3pt]
\textbf{Factored} (a product) shows: the zeros, and any factor shared by every
term.\\[3pt]
So read the \emph{question} first, then choose the form that puts the answer in
plain sight.
\end{formulabox}

\skillheading{1. Expand a product to reveal its terms}

\begin{formulabox}
Every term of the first factor multiplies every term of the second, then like
terms are collected. Useful patterns:
$(A+B)^2=A^2+2AB+B^2$ and $(A-B)(A+B)=A^2-B^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Write $(x+4)(x+1)$ in expanded form.
\step{The question asks for the terms, so the product has to be distributed.}
\step{$x\cdot x=x^2$; the cross terms are $x\cdot 1+4\cdot x=5x$; the last
product is $4\cdot 1=4$.}

Collecting gives $\ans{x^2+5x+4}$, so the constant term is $4$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Write $(x+6)(x+2)$ in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^2+8x+12}$}
\end{tryitbox}

\skillheading{2. Factor a sum to reveal its structure}

\begin{formulabox}
Pull out the greatest common factor first, then look for a pattern.
For $x^2+bx+c$, find two numbers with product $c$ and sum $b$.
No middle term and two perfect squares means $A^2-B^2=(A-B)(A+B)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Factor $x^2-5x+6$ completely.
\step{The zeros are wanted, so the expression must be turned into a product.}
\step{Two numbers with product $6$ and sum $-5$ are $-2$ and $-3$, and there is
no common factor to remove first.}

So $x^2-5x+6=\ans{(x-2)(x-3)}$, which shows zeros at $2$ and $3$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Factor $x^2-25$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-5)(x+5)}$}
\end{tryitbox}

\skillheading{3. Choose the form that answers the question}

\begin{formulabox}
Asked for a zero, or ``when is this $0$?'' --- use the \textbf{factored} form
and set one factor to zero. Asked for a value, a constant term or a
cancellation --- use the \textbf{expanded} form. A hard product of numbers is
often an $A^2-B^2$ in disguise.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $(n-2)(n+2)$ when $n=50$.
\step{Multiplying $48$ by $52$ by hand is slow, and the product matches
$A^2-B^2$, so expanding is the cheaper route here.}
\step{$(n-2)(n+2)=n^2-4$, and at $n=50$ that is $2500-4$.}

The value is $\ans{2496}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $(n-5)(n+5)$ when $n=40$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1575}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(2x-3)^2$ is not $4x^2+9$ --- squaring a binomial
always leaves a middle term. And ``factor completely'' means the greatest
common factor comes out \emph{before} you hunt for a pattern: $2x^3-18x$ is
$2x(x-3)(x+3)$, not $(2x^2-18)(x)$ left half-done.
\end{watchoutbox}
