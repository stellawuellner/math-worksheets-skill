\problem{Area of the triangle $A(0,0)$, $B(9,0)$, $C(9,12)$.}

\noindent\textbf{Solution.} The legs $\overline{AB}$ and $\overline{BC}$ are
horizontal and vertical, of lengths $9$ and $12$, so the triangle is right-angled
at $B$:
\[ \text{area} = \tfrac{1}{2}(9)(12) = \tfrac{1}{2}(108) . \]
Therefore \ans{54 \text{ square units}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Triangle $A(-3,2)$, $B(5,8)$, $C(5,2)$. (a) $AB$ (b) $BC$ (c) $CA$
(d) perimeter.}

\noindent\textbf{Solution (a).} $AB = \sqrt{(5-(-3))^2 + (8-2)^2}
= \sqrt{64+36} = \sqrt{100}$, so \ans{AB = 10 \text{ units}}

\noindent\textbf{Solution (b).} $\overline{BC}$ is vertical, so its length is the
difference of the $y$-values: $8 - 2$, giving \ans{BC = 6 \text{ units}}

\noindent\textbf{Solution (c).} $\overline{CA}$ is horizontal:
$5 - (-3)$, giving \ans{CA = 8 \text{ units}}

\noindent\textbf{Solution (d).} $10 + 6 + 8$, so
\ans{\text{perimeter} = 24 \text{ units}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Area of the quadrilateral $(1,1)$, $(7,1)$, $(9,5)$, $(3,5)$.}

\noindent\textbf{Solution.} Shoelace, going round in the given order:
\[ (1\cdot 1 - 7\cdot 1) + (7\cdot 5 - 9\cdot 1) + (9\cdot 5 - 3\cdot 5)
   + (3\cdot 1 - 1\cdot 5) = -6 + 26 + 30 - 2 = 48 , \]
and half of $48$ is the area. Therefore \ans{24 \text{ square units}}

\noindent\textit{Check:} the figure is a parallelogram with base $6$ and height
$4$.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Area of the park $(0,0)$, $(10,0)$, $(12,6)$, $(2,8)$, one metre per
unit.}

\noindent\textbf{Solution.} Shoelace:
\[ (0\cdot 0 - 10\cdot 0) + (10\cdot 6 - 12\cdot 0) + (12\cdot 8 - 2\cdot 6)
   + (2\cdot 0 - 0\cdot 8) = 0 + 60 + 84 + 0 = 144 . \]
Half of $144$ gives \ans{72 \text{ m}^2}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{L-shaped patio $(0,0)$, $(8,0)$, $(8,4)$, $(4,4)$, $(4,7)$, $(0,7)$,
one metre per unit. (a) Area. (b) Check by splitting into two rectangles.}

\noindent\textbf{Solution (a).} Shoelace round the six vertices:
\[ 0 + 32 + 16 + 12 + 28 + 0 = 88 , \]
and half of $88$ gives \ans{44 \text{ m}^2}

\noindent\textbf{Solution (b) --- instructor-judged.} A complete response splits
the patio into two rectangles --- for example the $8 \times 4$ block along the
bottom and the $4 \times 3$ block above its left half --- computes $32$ and $12$,
and adds them to $44$, matching the shoelace result. Any valid decomposition with
both partial areas and their sum earns full credit; naming the pieces without
computing them does not.

\problem{Quadrilateral $W(-2,1)$, $X(3,1)$, $Y(7,4)$, $Z(2,4)$. (a) $WX$
(b) $XY$ (c) perimeter (d) area.}

\noindent\textbf{Solution (a).} $\overline{WX}$ is horizontal:
$3 - (-2)$, so \ans{WX = 5 \text{ units}}

\noindent\textbf{Solution (b).} $XY = \sqrt{(7-3)^2 + (4-1)^2} = \sqrt{16+9}
= \sqrt{25}$, so \ans{XY = 5 \text{ units}}

\noindent\textbf{Solution (c).} All four sides measure $5$, so the perimeter is
$4 \times 5$, giving \ans{\text{perimeter} = 20 \text{ units}}

\noindent\textbf{Solution (d).} Shoelace:
\[ (-2 - 3) + (12 - 7) + (28 - 8) + (2 + 8) = -5 + 5 + 20 + 10 = 30 , \]
so the area is half of $30$: \ans{15 \text{ square units}}

\noindent\textit{Note:} the figure is a rhombus, not a square --- equal sides do
not make the area $5 \times 5$.

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Two triangular plots: $(0,0)$, $(10,0)$, $(4,6)$ and $(0,0)$, $(8,0)$,
$(3,9)$. (a) Area of the first. (b) Area of the second. (c) Difference.}

\noindent\textbf{Solution (a).} Base $10$ along the $x$-axis, height $6$:
$\tfrac{1}{2}(10)(6)$, so \ans{30 \text{ square units}}

\noindent\textbf{Solution (b).} Base $8$, height $9$: $\tfrac{1}{2}(8)(9)$, so
\ans{36 \text{ square units}}

\noindent\textbf{Solution (c).} The second is larger: $36 - 30$, so
\ans{6 \text{ square units}}

\vspace{0.15cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Pentagon $A(0,0)$, $B(12,0)$, $C(12,5)$, $D(6,9)$, $E(0,5)$, one metre
per unit. (a) Area. (b) $CD$. (c) Why is the shoelace area independent of the
starting vertex?}

\noindent\textbf{Solution (a).} Shoelace:
\[ 0 + 60 + 78 + 30 + 0 = 168 , \]
and half of $168$ gives \ans{84 \text{ m}^2}

\noindent\textbf{Solution (b).} $CD = \sqrt{(6-12)^2 + (9-5)^2}
= \sqrt{36+16} = \sqrt{52} = 7.2111\ldots$, so \ans{CD = 7.21 \text{ m}}

\noindent\textbf{Solution (c) --- instructor-judged.} A complete response says
the shoelace sum contains exactly one term per edge, and starting at a different
vertex only rotates the order in which those same edge terms are added; addition
does not depend on order, so the total --- and therefore the area --- is
unchanged. A response that merely asserts the answer is the same, or that tries
one other starting vertex without explaining why, earns partial credit.
