\wstitleblock{Sequences, Series and the Binomial Theorem --- Diagnosing Index, Common-Ratio and Convergence Errors}{Algebra 2}{}

\noindent\small Three errors account for most wrong answers in this unit. Each
problem below contains one of them.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Index off by one:} $a_n = a_1 + (n-1)d$ and $a_n = a_1 r^{\,n-1}$
        both step $n-1$ times, because the first term is already there before any
        step is taken.
  \item \textbf{Ratio read as a difference:} $r$ is found by \emph{dividing}
        consecutive terms, and one pair is never enough --- check a second pair.
  \item \textbf{Convergence ignored:} $S_\infty = \dfrac{a_1}{1-r}$ is valid only
        when $|r| < 1$. Used outside that range it returns a confident number for
        a sum that does not exist.
  \item In each problem: name the error in words, then show corrected work.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{\textbf{Find and fix.} An arithmetic sequence has first term
$a = 7$ and common difference $d = 4$. Dev wants the $12$th term and writes
$a_{12} = 7 + 12(4) = 55$.
\par\smallskip
Say in one sentence why Dev counts one step too many, then compute the correct
$12$th term.}

\problem[5.5cm]{\textbf{Find and fix.} A geometric sequence begins
$5$, $10$, $20$, \ldots\ Kai finds the common ratio by subtracting consecutive
terms and reports $r = 5$.
\par\smallskip
\textbf{(a)} Find $r$ correctly from the first pair of terms.
\par
\textbf{(b)} Confirm it on the second pair. Say why checking one pair is not
enough to identify a sequence as geometric.}

\problem[6.5cm]{\textbf{Find and fix.} Consider the arithmetic series whose first
term is $3$ and whose common difference is $5$, summed over its first $10$ terms.
\par\smallskip
Ines writes $\;S_{10} = \dfrac{10}{2}\bigl(3 + [\,3 + 5(10)\,]\bigr) = 280$.
\par\smallskip
Her formula is right and her last term is wrong. Name the slip, then compute the
correct sum.}

\problem[6.5cm]{\textbf{Binomial expansion.} Expand $(x+2)^4$ completely, writing
the terms in descending powers of $x$.
\par\smallskip
Then check your work with the index rule: in the expansion of $(x+p)^n$, the
exponents on $x$ and on $p$ must add to $n$ in every single term.}

\problem[6.5cm]{\textbf{Find and fix.} An infinite geometric series has first term
$6$ and common ratio $r = \frac{3}{4}$. Nia writes $\;S = 6 \div \frac{3}{4} = 8$.
\par\smallskip
\textbf{(a)} State whether this series converges, and why.
\par
\textbf{(b)} Name the part of the formula Nia mis-copied, then compute the correct
sum.}

\problem[6cm]{\textbf{Find and fix.} An arithmetic sequence starts at $7$ and
increases by $4$ each step. A student wants to know which term equals $91$, sets
up $7 + 4n = 91$, and reports the answer.
\par\smallskip
The equation is missing one adjustment. Write the correct equation, solve it, and
state which term of the sequence equals $91$.}

\problem[7cm]{\textbf{Find and fix.} In the expansion of $(x+3)^5$, Vic wants the
coefficient of the $x^2$ term. He reasons: ``the power on $x$ is $2$, so
$k = 2$,'' and computes $\binom{5}{2}3^2 = 90$.
\par\smallskip
\textbf{(a)} Use the rule that the exponents on $x$ and on $3$ must add to $5$ to
find the correct value of $k$.
\par
\textbf{(b)} Compute the correct coefficient of the $x^2$ term.}

\problem[7cm]{\textbf{Explain.}
\par\smallskip
\textbf{(a)} Explain why an infinite geometric series has a finite sum only when
$|r| < 1$. Refer to what happens to $r^{\,n}$ as $n$ grows.
\par
\textbf{(b)} A classmate applies $S = \dfrac{a_1}{1-r}$ to a series of positive
terms whose ratio is $2$ and reports a \emph{negative} sum. Explain, without
computing anything, why that answer is impossible and what it reveals about the
formula's condition.
\par
\textbf{(c)} Write two checking habits: one that catches an off-by-one index, one
that catches a ratio read as a difference.
