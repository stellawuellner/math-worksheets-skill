\wstitleblock{Composite Solids, Density and Design}{High School Geometry}{}

\begin{center}
\small\textit{Composite solid volumes, composite surface areas, density and mass, and design constraints}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Useful formulas: cylinder $V = \pi r^2 h$, curved surface $= 2\pi r h$; cone $V = \frac{1}{3}\pi r^2 h$, curved surface $= \pi r \ell$; sphere $V = \frac{4}{3}\pi r^3$, surface $= 4\pi r^2$; density $=$ mass $\div$ volume.
  \item After the first few warm-ups the methods are mixed --- decide what the situation asks for before you reach for a formula.
  \item Use the $\pi$ key and round only at the final step, to the nearest hundredth.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem{A grain silo is a plain cylinder of radius $4$ m and height $7$ m. Find its volume.
\par\vspace*{3.8cm}\answerline{m$^3$}}

\problem{A concrete plinth is a rectangular block $1.2$ m long, $0.8$ m wide and $0.5$ m tall, with a cylindrical hole of radius $0.2$ m drilled straight through from top to bottom. Find the volume of concrete in the plinth.
\par\vspace*{4.2cm}\answerline{m$^3$}}

\problem{An open-topped cylindrical water tank has radius $2$ m and height $6$ m. Find the area of sheet metal needed for its curved wall and its base.
\par\vspace*{4.2cm}\answerline{m$^2$}}

\problem{A block of oak has volume $2400$ cm$^3$ and density $0.75$ g/cm$^3$. Find its mass.
\par\vspace*{3.8cm}\answerline{g}}

\problem{A farm silo is a cylinder of radius $3$ m and height $8$ m, topped by a hemisphere of the same radius. Find the total volume of the silo.
\par\vspace*{4.4cm}\answerline{m$^3$}}

\problem{A bronze statuette has mass $5.4$ kg, which is $5400$ g, and bronze has density $8.8$ g/cm$^3$. Find the volume of the statuette.
\par\vspace*{4.2cm}\answerline{cm$^3$}}

\problem{A cylindrical water tank must hold exactly $15$ m$^3$. The site allows a radius of only $1.5$ m. Find the height the tank must be built to.
\par\vspace*{4.2cm}\answerline{m}}

\problem{A grain hopper is a cone of radius $5$ cm and slant height $13$ cm sitting on a cylinder of the same radius and height $9$ cm. The cylinder's bottom is closed; the cone and cylinder meet with no exposed disc between them. Find the total outside surface area of the hopper.
\par\vspace*{4.6cm}\answerline{cm$^2$}}

\problem{A glass paperweight is a cube of edge $6$ cm with a hemispherical hollow of radius $2$ cm scooped out of one face. The glass has density $2.5$ g/cm$^3$. Find the mass of the paperweight.
\par\vspace*{4.6cm}\answerline{g}}

\problem{A cylindrical grain bin of radius $2.5$ m must hold $60$ m$^3$. It is built from steel weighing $18$ kg for each square metre, and only the curved wall and the flat base use steel.
\par\vspace{0.15cm}
(a) Find the height the bin must be built to.
\par\vspace*{2.4cm}\answerline{m}
\par\vspace{0.25cm}
(b) Find the mass of steel the bin needs.
\par\vspace*{2.6cm}\answerline{kg}
\par\vspace{0.25cm}
(c) The capacity must stay at $60$ m$^3$. Explain why doubling the radius would not halve the mass of steel needed.
\par\vspace*{3.0cm}}
