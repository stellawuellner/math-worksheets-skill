\sstitleblock{Adding Two-Digit Numbers with Place Value}

\skillheading{Skill 1 --- Breaking both numbers into tens and ones}

\begin{formulabox}\textbf{Rule:}\ A two-digit number is made of tens and ones.
Split both numbers, add the tens to the tens and the ones to the ones, then
put the two answers back together.\par
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item If the ones make ten or more, trade ten ones for one ten.
\end{itemize}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $26 + 32$
\step{Both numbers are two-digit, so I split each one by place value instead
of counting on.}
\step{Tens: $20 + 30 = 50$. \quad Ones: $6 + 2 = 8$. \quad Put them together:
$50 + 8$.}
$\Rightarrow\quad 26 + 32 = \ans{58}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $41 + 35$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{76}$}
\end{tryitbox}

\skillheading{Skill 2 --- Jumping tens, then ones, on a number line}

\begin{formulabox}\textbf{Rule:}\ Keep the first number whole and break only
the second one. Jump one ten at a time, then jump the ones. The number you land
on is the total.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $27 + 35$
\ \begin{tikzpicture}[scale=0.42, baseline=(current bounding box.center)]
  \draw[thick,->] (0,0) -- (9,0);
  \node[below] at (0,-0.1) {27};
  \node[below] at (8,-0.1) {62};
\end{tikzpicture}
\step{Adding on a line means I only have to break the second number, so I split
35 into three tens and five ones.}
\step{Three ten-jumps: $27 \to 37 \to 47 \to 57$. Then a jump of five ones:
$57 \to 62$.}
$\Rightarrow\quad 27 + 35 = \ans{62}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $32 + 24$ (two ten-jumps, then four ones)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56}$}
\end{tryitbox}

\skillheading{Skill 3 --- Making a friendly ten, then adjusting}

\begin{formulabox}\textbf{Rule:}\ Push one number up to the nearest ten,
because tens are easy to add. Then undo exactly what you added:\par
\centerline{$a + c = (a + b) + c - b$ \qquad where $a + b$ is a ten.}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad $28 + 35$
\step{28 is only 2 away from 30, and adding to a ten is easy, so I lend it 2
and pay the 2 back at the end.}
\step{$28 + 2 = 30$, then $30 + 35 = 65$, then give the 2 back: $65 - 2 = 63$.}
$\Rightarrow\quad 28 + 35 = \ans{63}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad $49 + 26$ (lend 1, then pay it back)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{75}$}
\end{tryitbox}

\skillheading{Skill 4 --- Rounding to ten to check a sum}

\begin{formulabox}\textbf{Rule:}\ Look at the ones digit. If it is 5 or more,
round up to the next ten; if it is 4 or less, round down. Add the rounded
numbers to get an \emph{estimate} --- the real answer should land near it.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad About how much is $43 + 29$?
\step{The question says ``about how much'', so I round each number to its
nearest ten before adding instead of adding exactly.}
\step{$43 \to 40$ (3 ones, round down) and $29 \to 30$ (9 ones, round up), so
$40 + 30$.}
$\Rightarrow\quad 43 + 29 \approx \ans{70}$
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad About how much is $62 + 18$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{80}$}
\end{tryitbox}
