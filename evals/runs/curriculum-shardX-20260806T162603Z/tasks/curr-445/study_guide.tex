\sstitleblock{Numerical, Graphical and Algebraic Ideas of a Limit}

\skillheading{Skill 1 --- Limits by algebra}

\begin{formulabox}
\textbf{Rule:} try substitution first. If it gives a number, that is the limit. If it gives
$\dfrac{0}{0}$, that is not an answer --- it is a signal that top and bottom share a factor.
Factor and cancel (or multiply by the conjugate), then substitute into what is left. The
cancelled function agrees with the original everywhere except at the point itself, and a limit
never looks at the point itself.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\lim_{x \to 2} \frac{x^2 - 4}{x - 2}$.
\step{Substitution gives $0/0$, so the factor $x - 2$ must be common to top and bottom --- factor and cancel it.}
\step{$\dfrac{(x-2)(x+2)}{x-2} = x + 2$ for $x \ne 2$, and substituting gives $2 + 2$.}

$\Rightarrow\quad$ the limit is $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\lim_{x \to 5} \frac{x^2 - 25}{x - 5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Limits from a table of values}

\begin{formulabox}
\textbf{Rule:} evaluate the function at inputs closing in on the target from both sides. If
both columns settle on the same number, that number is the limit --- but a table only
\emph{suggests} a limit. It shows finitely many inputs; the limit is a claim about all of them,
so confirm it algebraically whenever you can.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $f(x) = \dfrac{x^2 - 4}{x - 2}$ at $x = 1.99$.
\step{The function is undefined at $2$, so approach it: pick an input just below the target and read the output.}
\step{$\dfrac{1.99^2 - 4}{1.99 - 2} = \dfrac{-0.0399}{-0.01}$.}

$\Rightarrow\quad f(1.99) = \ans{3.99}$, already close to the value $4$ that algebra confirms.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate the same $f$ at $x = 2.01$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4.01}$}
\end{tryitbox}

\skillheading{Skill 3 --- One-sided limits from a graph}

\begin{formulabox}
\textbf{Rule:} $\displaystyle\lim_{x \to a} f(x)$ exists only when the left-hand and right-hand
limits agree. On a graph, trace the curve toward $a$ from the left and from the right and read
off the heights the curve heads for --- not the plotted value at $a$, which may be a separate
dot or missing entirely.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A function equals $3x - 1$ for $x < 1$. Find its left-hand limit at $x = 1$.
\step{Approaching from the left uses only the piece defined for $x < 1$, so substitute into that piece.}
\step{$3(1) - 1$.}

$\Rightarrow\quad \displaystyle\lim_{x \to 1^-} f(x) = \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The same function equals $x + 4$ for $x \ge 1$. Find its right-hand limit at $x = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 4 --- A series sum is a limit}

\begin{formulabox}
\textbf{Rule:} an infinite sum is not an addition you ever finish. It is defined as
$\displaystyle\lim_{N \to \infty} S_N$, the limit of the partial sums. For a geometric series
with $|r| < 1$, $S_N = \dfrac{a(1 - r^{N})}{1 - r}$, and since $r^{N} \to 0$ the limit is
$\dfrac{a}{1 - r}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find $\displaystyle\sum_{n=1}^{\infty}\left(\tfrac{1}{2}\right)^{n-1}$.
\step{The ratio $1/2$ has absolute value under $1$, so the partial sums converge and the limit formula applies.}
\step{First term $a = 1$, ratio $r = \tfrac{1}{2}$, so the sum is $\dfrac{1}{1 - 1/2}$.}

$\Rightarrow\quad$ the sum is $\ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find $\displaystyle\sum_{n=1}^{\infty}\left(\tfrac{1}{5}\right)^{n-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{4}}$}
\end{tryitbox}
