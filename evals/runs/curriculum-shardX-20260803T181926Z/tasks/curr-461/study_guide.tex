\sstitleblock{Differentiating Implicit Relations}

\skillheading{Skill 1 --- Differentiating a Relation Implicitly}

\begin{formulabox}
\textbf{Why it is legal:} on any arc where $y$ is a differentiable function of $x$,
both sides of a relation are the same function of $x$, so both sides may be
differentiated with respect to $x$.\\[3pt]
\textbf{The bookkeeping:} every $y$ that gets differentiated leaves behind a factor
$\dfrac{dy}{dx}$ (chain rule). Term by term:
$x^{3} \to 3x^{2}$; \ $y^{3} \to 3y^{2}\dfrac{dy}{dx}$; \
$xy \to y + x\dfrac{dy}{dx}$; \ $x^{2}y \to 2xy + x^{2}\dfrac{dy}{dx}$; \
a constant $\to 0$.\\[3pt]
\textbf{Then solve:} collect the $\dfrac{dy}{dx}$ terms on one side, factor,
divide. The answer usually contains both $x$ and $y$, and that is correct.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $\dfrac{dy}{dx}$ for $x^{2} + y^{2} = 9$.
\step{The relation cannot be solved for $y$ as a single function, but it can be differentiated as it stands, which is exactly what implicit differentiation is for.}
\step{Differentiating gives $2x + 2y\dfrac{dy}{dx} = 0$, so $2y\dfrac{dy}{dx} = -2x$ and $\dfrac{dy}{dx} = \ans{-\dfrac{x}{y} = \dfrac{-x}{\sqrt{9 - x^{2}}}}$ on the upper arc.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $\dfrac{dy}{dx}$ for $x^{2} + y^{2} = 16$, and write it
in terms of $x$ alone on the upper arc.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\dfrac{x}{y} = \dfrac{-x}{\sqrt{16 - x^{2}}}}$}
\end{tryitbox}

\skillheading{Skill 2 --- The Slope at a Point on the Curve}

\begin{formulabox}
\textbf{An implicit derivative needs both coordinates.} $\dfrac{dy}{dx}$ comes out
in terms of $x$ \emph{and} $y$, so a slope cannot be read from an $x$-value
alone.\\[3pt]
\textbf{Check the point first.} Substitute it into the original relation. If the
two sides do not agree, the point is not on the curve and any slope computed there
is meaningless.\\[3pt]
\textbf{Then substitute} both coordinates into the derivative and simplify.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad For $x^{2}y = 12$, find the slope at $(2, 3)$.
\step{Before differentiating anything, confirm the point is on the curve: $2^{2}\cdot 3 = 12$, so it is, and the slope there will mean something.}
\step{The product rule gives $2xy + x^{2}\dfrac{dy}{dx} = 0$, so $\dfrac{dy}{dx} = -\dfrac{2y}{x}$. At $(2, 3)$ that is $\ans{-3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad For $x^{2}y = 18$, check that $(3, 2)$ is on the curve and
find the slope there.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\dfrac{4}{3}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Tangent Lines and Special Points}

\begin{formulabox}
\textbf{Tangent line:} slope $m$ from the implicit derivative at the point, then
$y - y_1 = m(x - x_1)$ through that point.\\[3pt]
\textbf{Horizontal tangent:} the numerator of $\dfrac{dy}{dx}$ is zero and the
denominator is not. Set the numerator to 0, substitute the resulting relation back
into the original equation, and solve for the points.\\[3pt]
\textbf{Vertical tangent:} the denominator is zero and the numerator is not --- the
slope is undefined there, and $y$ is not a function of $x$ near that point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Write the tangent to $x^{2} + y^{2} = 10$ at $(1, 3)$.
\step{A tangent line needs a point and a slope, and the point is given, so the only missing piece is the implicit derivative evaluated there.}
\step{From $2x + 2y\dfrac{dy}{dx} = 0$, $\dfrac{dy}{dx} = -\dfrac{x}{y} = -\dfrac{1}{3}$ at $(1, 3)$, giving the line $\ans{y = -\tfrac{1}{3}(x - 1) + 3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Write the tangent to $x^{2} + y^{2} = 20$ at $(2, 4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = -\tfrac{1}{2}(x - 2) + 4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} differentiating $y^{2}$ as $2y$ is the classic slip --- the
chain-rule factor $\dfrac{dy}{dx}$ is not optional. And when a term holds both
variables, such as $xy$, the product rule comes first and the chain rule applies
inside it.
\end{watchoutbox}
