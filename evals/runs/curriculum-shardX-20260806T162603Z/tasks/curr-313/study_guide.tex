\sstitleblock{Corresponding Parts of Congruent Triangles}

\skillheading{Skill 1 --- Finding a matching side}

\begin{formulabox}
\textbf{Rule.} A congruence statement is read position by position. In
$\triangle ABC \cong \triangle DEF$ the vertices pair as $A \leftrightarrow D$,
$B \leftrightarrow E$, $C \leftrightarrow F$, so the sides pair as
$\overline{AB} \leftrightarrow \overline{DE}$,
$\overline{BC} \leftrightarrow \overline{EF}$,
$\overline{CA} \leftrightarrow \overline{FD}$.\par
Matching sides are congruent, so their two expressions are \emph{equal}: set
them equal and solve.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \cong \triangle DEF$ with $AB = (3x-5)$ cm and
$DE = 16$ cm. Find $x$.
\step{Want $x$; the statement pairs $\overline{AB}$ with $\overline{DE}$, so the
two lengths are equal --- that turns the geometry into one linear equation.}
\step{$3x - 5 = 16 \;\Rightarrow\; 3x = 21$}

$x = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle PQR \cong \triangle STU$ with $PQ = (4x+2)$ mm and
$ST = 26$ mm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} pair by \emph{position}, not by alphabet. In
$\triangle ABC \cong \triangle FED$, side $\overline{AB}$ matches
$\overline{FE}$ --- not $\overline{DE}$.
\end{watchoutbox}

\skillheading{Skill 2 --- Finding a matching angle}

\begin{formulabox}
\textbf{Rule.} Matching vertices carry matching angles. In
$\triangle ABC \cong \triangle XYZ$,
$\angle A \cong \angle X$, $\angle B \cong \angle Y$, $\angle C \cong \angle Z$.
Equal angles means equal measures, so an expression for one may be set equal to
the measure of the other and solved the same way.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \cong \triangle XYZ$ with
$m\angle B = (5x+8)^\circ$ and $m\angle Y = 53^\circ$. Find $x$.
\step{Want $x$; $B$ sits in the second position of both names, so $\angle B$ and
$\angle Y$ correspond and their measures are equal.}
\step{$5x + 8 = 53 \;\Rightarrow\; 5x = 45$}

$x = \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle ABC \cong \triangle XYZ$ with
$m\angle A = (7x-3)^\circ$ and $m\angle X = 46^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{7}$}
\end{tryitbox}

\skillheading{Skill 3 --- Using CPCTC for a missing part}

\begin{formulabox}
\textbf{Rule (CPCTC).} Once two triangles are known to be congruent,
\emph{every} pair of corresponding parts is congruent --- including parts no one
measured. Combine that with the angle sum of a triangle,
\[ m\angle 1 + m\angle 2 + m\angle 3 = 180^\circ, \]
to reach a missing measure in either triangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} $\triangle GHJ \cong \triangle KLM$ with $m\angle K = 42^\circ$
and $m\angle L = 61^\circ$. Find $m\angle J$, in degrees.
\step{Want $m\angle J$; $J$ corresponds to $M$, so by CPCTC $m\angle J = m\angle
M$ --- and $\angle M$ is the third angle of a triangle whose other two measures
are known.}
\step{$m\angle M = 180 - 42 - 61$}

$m\angle J = \ans{77}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle GHJ \cong \triangle KLM$ with $m\angle K = 35^\circ$
and $m\angle L = 80^\circ$. Find $m\angle J$, in degrees.\\[2pt]
\rotatebox{180}{\footnotesize check: $m\angle J = \ans{65}$ degrees}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} CPCTC is a conclusion, never a reason to start. First
establish the congruence (SSS, SAS, ASA, AAS or HL); only then may you claim a
pair of parts is congruent.
\end{watchoutbox}
