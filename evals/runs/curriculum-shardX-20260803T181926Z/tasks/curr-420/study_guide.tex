\sstitleblock{Modelling Periodic Behaviour}

\begin{formulabox}
\textbf{The model:} $y = A\sin\big(B(t - C)\big) + D$ \ (or $\cos$).\quad
$|A|$ amplitude, $D$ midline, $C$ horizontal shift,
period $= \frac{2\pi}{B}$.\\[3pt]
\textbf{Reading $A$ and $D$ off a graph} (max $20$, min $6$):
$A = \frac{20 - 6}{2} = 7$ \quad and \quad $D = \frac{20 + 6}{2} = 13$.\\[3pt]
\textbf{Always radians} on this topic. A calculator in degree mode returns a
plausible wrong number rather than an error, which is why it goes unnoticed.
\end{formulabox}

\skillheading{1. Reading the period off a graph}

\begin{formulabox}
A period is measured \textbf{peak to peak} (or trough to trough). A peak and
the \emph{next} trough are only \textbf{half} a period apart --- doubling that
gap is the step most often skipped. Once you have the period,
$B = \frac{2\pi}{\text{period}}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A curve has a maximum at $x = 3$ and its next minimum
at $x = 9$. Find the period.
\step{The two points given are a peak and the next trough, so the gap between
them is half a period, not a whole one.}
\step{The gap is $9 - 3 = 6$, so the period is $2 \times 6$.}

$\Rightarrow\quad \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A curve has a maximum at $x = 2$ and its next minimum at
$x = 5$. Find the period.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{2. Building a model from a situation}

\begin{formulabox}
Turn the words into the four parameters, one at a time: the swing gives $|A|$,
the centre gives $D$, ``one cycle every $P$ units'' gives
$B = \frac{2\pi}{P}$, and where the motion \emph{starts} decides between
$\sin$, $\cos$ and a leading minus sign. Starting at the minimum is
$-\cos$; starting at the midline going up is $+\sin$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A wheel turns once every $10$ minutes. What is $B$ in
its height model?
\step{Only the timing matters for $B$: one full turn is one period, so the
period is the turn time and nothing else in the story is needed.}
\step{$B = \frac{2\pi}{10}$, and the fraction reduces.}

$\Rightarrow\quad \ans{\frac{\pi}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A wheel turns once every $8$ minutes. What is $B$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{\pi}{4}}$}
\end{tryitbox}

\skillheading{3. Evaluating a model}

\begin{formulabox}
Substitute the time, simplify the angle to a familiar unit-circle value if you
can, then finish the arithmetic. Reduce the angle \emph{before} taking the
sine: $\frac{\pi(2)}{4}$ is $\frac{\pi}{2}$, and you know that one exactly.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $h(t) = 6\sin\!\left(\frac{\pi t}{4}\right) + 10$.
Find $h(2)$.
\step{The input is given, so this is substitution --- and the angle is in
radians, so no degree conversion belongs anywhere.}
\step{$\frac{\pi(2)}{4} = \frac{\pi}{2}$ and $\sin\frac{\pi}{2} = 1$, so
$h(2) = 6(1) + 10$.}

$\Rightarrow\quad \ans{16}$ \quad (the maximum: midline $10$ plus amplitude
$6$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $h$. Find $h(6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{4. Solving a periodic equation on one cycle}

\begin{formulabox}
Isolate the trig function first, then ask the unit circle for \textbf{every}
angle in one revolution with that value --- there are almost always two.
Convert each angle back to a value of $t$, then keep the ones inside the given
interval. One answer on a full cycle is a warning sign, not a result.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2\sin\!\left(\frac{\pi t}{6}\right) = 1$ for
$0 \le t < 12$.
\step{Isolate the sine before touching the unit circle, or the reference angle
comes out of the wrong equation.}
\step{$\sin\!\left(\frac{\pi t}{6}\right) = \frac{1}{2}$ at
$\frac{\pi t}{6} = \frac{\pi}{6}$ and at $\frac{\pi t}{6} = \frac{5\pi}{6}$,
giving $t = 1$ and $t = 5$.}

$\Rightarrow\quad \ans{t = 1 \text{ and } t = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $2\cos\!\left(\frac{\pi t}{6}\right) = 1$ for
$0 \le t < 12$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 2 \text{ and } t = 10}$}
\end{tryitbox}
