\wstitleblock{Radical Expressions and Rational Exponents --- Finding the Illegal Split}{Algebra 2}{}

\noindent\small A radical distributes over a \textbf{product}, never over a
\textbf{sum}: $\sqrt{ab}=\sqrt{a}\,\sqrt{b}$ is a rule, but $\sqrt{a+b}$ is
\emph{not} $\sqrt{a}+\sqrt{b}$.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Several problems below show somebody's work. Find the step where a
        radical (or a $1/2$ power) was distributed across a sum.
  \item Then do the problem correctly and put the correct value on the answer
        line. Show the testing or the algebra that settles it.
\end{itemize}
\normalsize

\vspace{0.2cm}

\problem[5cm]{\textbf{Test the claim with numbers you can check.} Work out
$\sqrt{9+16}$ by adding inside the radical first. Separately, work out
$\sqrt{9}+\sqrt{16}$, which comes to $7$. Write the correct value of
$\sqrt{9+16}$ on the answer line.}

\problem[5cm]{\textbf{Find and fix.} Priya writes
\[ \sqrt{36+64} \;=\; \sqrt{36}+\sqrt{64} \;=\; 14. \]
Name the step that is not legal, then compute $\sqrt{36+64}$ correctly.}

\problem[5cm]{Bo replaces $\sqrt{5+7}$ with $\sqrt{5}+\sqrt{7}$ and reports
$4.88$. That is the same illegal split, and this time neither number is a whole
number, so it is easy to miss. Compute $\sqrt{5+7}$ rounded to two decimal
places.}

\problem[6cm]{\textbf{Find and fix.} Dana writes
$\sqrt{50}+\sqrt{18}=\sqrt{68}\approx 8.25$, pulling two separate radicals into
one radical of the sum. Simplify $\sqrt{50}$ and $\sqrt{18}$ separately with the
product rule, combine the like radicals, and give the exact answer together with
its decimal value to two places.}

\problem[5cm]{The same claim written with rational exponents. Kofi starts from
$(25+144)^{1/2}$, gives each term its own one-half power, and gets
$25^{1/2}+144^{1/2}=17$. Compute $(25+144)^{1/2}$ correctly.}

\problem[7cm]{\textbf{Find and fix.} Solve $\sqrt{x+9}=5$. Jamal writes
$\sqrt{x}+\sqrt{9}=5$, so $\sqrt{x}=2$ and $x=4$. Substitute $x=4$ into the
original equation to show Jamal's answer fails, then solve the equation
correctly by squaring the whole left side at once.}

\problem{A rectangular patio measures $20$ feet by $21$ feet. Its diagonal
brace has length $\sqrt{20^2+21^2}$ feet. Sam writes the diagonal as
$20+21=41$ feet. Find the exact length of the diagonal brace, in feet.
\par\vspace*{6cm}\answerline{ft}}

\problem[8cm]{\textbf{Explain, then solve.}\par
\textbf{(a)} Square the expression $\sqrt{a}+\sqrt{b}$ and compare the result
with $a+b$. Use what you get to explain, in your own words, why
$\sqrt{a+b}=\sqrt{a}+\sqrt{b}$ fails for almost every pair of numbers.\par
\textbf{(b)} Exactly one value of $x$ with $x \ge 0$ makes
$\sqrt{x+25}=\sqrt{x}+5$ a true statement. Find that value, and write it on the
answer line.}
