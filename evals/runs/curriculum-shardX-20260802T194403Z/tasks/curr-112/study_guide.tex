\sstitleblock{Whole-Number Division: Up to Four Digits by One-Digit Divisors}

\skillheading{Skill 1 --- The long-division cycle}
\begin{formulabox}\textbf{The cycle: \ divide $\rightarrow$ multiply $\rightarrow$ subtract $\rightarrow$ bring down.}\\[3pt]
Start at the biggest place value the divisor will fit into, then turn the cycle
once more for every digit still waiting to come down.\\[2pt]
\textbf{Words to keep straight:} in $852 \div 4$, the \emph{dividend} is the
number being split, the \emph{divisor} is the size of each group, and the
\emph{quotient} is the answer.\\[2pt]
\textbf{Always check:} quotient $\times$ divisor $=$ dividend.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $852 \div 4$
\step{We are splitting 852 into groups of 4, and 4 does fit into the hundreds
digit, so run the cycle from the hundreds place downward.}
\step{Hundreds: $8 \div 4 = 2$, nothing left over. Tens: $5 \div 4 = 1$ with $1$
left over; bring the $2$ down beside that leftover to make $12$.}
\step{Ones: $12 \div 4 = 3$. Read the quotient across the top: $\ans{213}$, and
the check works: $213 \times 4 = 852$.}
\begin{center}$\begin{array}{r}
213 \\[-3pt]
4\,\overline{\smash{)}\,852}
\end{array}$\end{center}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $675 \div 5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{135}$}\end{tryitbox}

\skillheading{Skill 2 --- Four-digit dividends and zeros}
\begin{formulabox}\textbf{Rule:}\\[3pt]
A fourth digit just means one more turn of the same cycle. If the number you are
dividing at some step is \emph{smaller} than the divisor, that place gets a
$\mathbf{0}$ in the quotient --- write it, then bring down the next digit.
Every place you divide into earns a digit.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $3{,}612 \div 6$
\step{Six will not fit into 3 thousands, so the first place we can actually
divide is the hundreds --- which already tells us the quotient has three digits.}
\step{$36 \div 6 = 6$ hundreds, nothing left. The next digit down is $1$, which
is too small for $6$, so the tens digit of the quotient is $0$.}
\step{Bring down the $2$ to make $12$: $12 \div 6 = 2$ ones, giving $\ans{602}$.
Check: $602 \times 6 = 3{,}612$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $5{,}145 \div 5$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1029}$}\end{tryitbox}
\begin{watchoutbox}\textbf{(!) Watch out:} dropping that $0$ turns $602$ into
$62$ --- and $62 \times 6$ is only $372$, nowhere near $3{,}612$. The
multiplication check catches this mistake every time.
\end{watchoutbox}

\skillheading{Skill 3 --- Remainders, and what they mean}
\begin{formulabox}\textbf{Rule:}\\[3pt]
$\text{dividend} = \text{divisor} \times \text{quotient} + \text{remainder}$,
and the remainder is always \emph{smaller} than the divisor.\\[2pt]
Then ask what the leftover means in the story: \textbf{ignore} it (how many
\emph{full} boxes), \textbf{round up} (how many buses are needed), or
\textbf{report} it (how many are left over).
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $437 \div 8$
\step{We want how many 8s fit inside 437; 8 will not go in evenly, so divide as
far as possible and then name what is left.}
\step{$43 \div 8 = 5$ tens with $3$ left over, since $8 \times 5 = 40$. Bring
down the $7$ to make $37$: $37 \div 8 = 4$ ones with $5$ left over.}
\step{That $5$ is smaller than $8$, so it stays as the remainder:
$\ans{54}$ R $\ans{5}$. Check: $8 \times 54 + 5 = 437$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $293 \div 7$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{41}$ R $\ans{6}$}\end{tryitbox}

\skillheading{Skill 4 --- Estimate first, then check}
\begin{formulabox}\textbf{Rule:}\\[3pt]
Trade the dividend for a \emph{compatible number} --- a nearby number the
divisor goes into evenly --- and divide that instead. The estimate tells you how
big the real quotient should be, so a digit written in the wrong place shows up
immediately. Afterwards, multiply the quotient back by the divisor to check.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad About how much is $3{,}489 \div 6$?
\step{We only need a close answer here, so swap the dividend for a friendly
multiple of the divisor instead of dividing exactly.}
\step{$3{,}600$ sits close by and $6$ divides it evenly, so estimate with
$3{,}600 \div 6 = 600$.}
\step{The estimate is $\ans{600}$, so the exact quotient has to land near 600 ---
an answer like 58 or 5,800 would have to be a place-value slip.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad Estimate $2{,}918 \div 7$ with a compatible number.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{400}$}\end{tryitbox}
