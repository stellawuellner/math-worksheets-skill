\wstitleblock{Comparing Three-Digit Numbers}{Grade 2--3 Mathematics}{}

\begin{center}
\textit{Comparing three-digit numbers on a place-value chart, comparing numbers
shown with base-ten blocks, ordering three numbers, and answering questions
about real counts}
\end{center}

\noindent\small
\textbf{How to compare.} Write both numbers in a place-value chart, one digit
per box. Then start at the \textbf{left} and compare one place at a time.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Compare the hundreds first. The number with more hundreds is greater,
        no matter what the other digits are.
  \item Same hundreds? Compare the tens. Same tens? Compare the ones.
  \item A base-ten block picture counts the same way: $1$ flat is one hundred,
        $1$ rod is one ten, $1$ small cube is one.
  \item $>$ means ``is greater than'' and $<$ means ``is less than''. The point
        always aims at the smaller number.
\end{itemize}
\normalsize
