\wstitleblock{Is This Relation a Function?}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{One input, one output: tables, lists of pairs, graphs, and rules}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item A relation is a \emph{function} when every input is paired with
        exactly one output. The rule is about inputs only --- outputs are
        allowed to repeat.
  \item In a table or a list of pairs, hunt for an input that appears twice
        with different outputs. On a graph, look for two points sitting
        directly above each other.
  \item When a problem asks you to decide, give the reason as well as the
        decision.
\end{itemize}

\problem{A relation is shown in the table.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Input} & 1 & 2 & 3 & 4 \\ \hline
\textbf{Output} & 5 & 7 & 9 & 11 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What output is paired with the input 3?
        \par\vspace*{1.7cm}\ansline
  \item[(b)] Is this relation a function? Explain how the table shows it.
        \par\vspace*{2cm}\ansline
\end{itemize}}

\problem{A relation is the list of pairs $(5, 5)$, $(4, 9)$, $(5, 8)$,
$(6, 11)$.
\begin{itemize}
  \item[(a)] What output is paired with the input 6?
        \par\vspace*{1.6cm}\ansline
  \item[(b)] One input appears in two pairs. Compare its two outputs, in the
        order they are listed, using $>$, $<$, or $=$.
        \par\vspace*{1.6cm}\ansline
  \item[(c)] Is this relation a function? Explain your decision.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem{Look at graph P in the Graph Bank.
\begin{itemize}
  \item[(a)] Write the $y$-value of the plotted point whose $x$-value is 4.
        \par\vspace*{2cm}\ansline
  \item[(b)] Is the relation shown by graph P a function? Explain how the
        picture shows it.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem[3.6cm]{A relation is given by the rule $y = 4x - 3$, where $x$ is the
input and $y$ the output. Find the output when the input is 5.}

\problem{A relation is shown in the table. Notice that one input appears
twice.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Input} & 8 & 6 & 8 & 9 \\ \hline
\textbf{Output} & 12 & 5 & 8 & 5 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What output is paired with the input 9?
        \par\vspace*{1.6cm}\ansline
  \item[(b)] Compare the two outputs of the repeated input, in the order the
        table lists them, using $>$, $<$, or $=$.
        \par\vspace*{1.6cm}\ansline
  \item[(c)] Is this relation a function? Explain your decision.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem{A relation is given by the rule $y = 2x + 1$.
\begin{itemize}
  \item[(a)] Find the output when the input is 7.
        \par\vspace*{2.2cm}\ansline
  \item[(b)] Find the output when the input is 12.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{Look at graph Q in the Graph Bank.
\begin{itemize}
  \item[(a)] Two of its plotted points sit directly above each other. Compare
        their $y$-values, the lower point first, using $>$, $<$, or $=$.
        \par\vspace*{2cm}\ansline
  \item[(b)] Is the relation shown by graph Q a function? Explain how the
        picture shows it.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A relation is the list of pairs $(1, 4)$, $(2, 4)$, $(3, 4)$,
$(4, 4)$.
\begin{itemize}
  \item[(a)] What output is paired with the input 3?
        \par\vspace*{2cm}\ansline
  \item[(b)] The output 4 is used every time. Is this relation still a
        function? Explain your decision.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{Relation $R$ is the table below. Relation $S$ is the list of pairs
$(4, 3)$, $(6, 12)$, $(4, 11)$, $(8, 12)$.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{$R$ input} & 1 & 2 & 3 & 4 \\ \hline
\textbf{$R$ output} & 6 & 8 & 10 & 12 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What output does $R$ pair with the input 4?
        \par\vspace*{1.7cm}\ansline
  \item[(b)] In $S$, one input appears twice. Compare its two outputs, in the
        order they are listed, using $>$, $<$, or $=$.
        \par\vspace*{1.7cm}\ansline
  \item[(c)] Which of $R$ and $S$ is a function? Explain your decision for
        each one.
        \par\vspace*{2.1cm}\ansline
\end{itemize}}

\problem{A relation is given by the rule $y = 3x + 5$.
\begin{itemize}
  \item[(a)] Find the output when the input is 6.
        \par\vspace*{2cm}\ansline
  \item[(b)] Explain why a relation given by a rule is always a function,
        while a relation given as a list of pairs might not be.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}
