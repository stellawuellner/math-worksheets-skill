\sstitleblock{Take-Away and Comparing Stories}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Model a Take-From Story}

\begin{formulabox}
\textbf{Rule:} When things pop, fly off, or get eaten, the story is
subtraction: \ start $-$ taken away $=$ left. Draw the start, cross out what
goes, count what stays.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 7 ducks are on the pond. 2 swim away. How many ducks
are left?
\step{Swimming away is taking from, so we draw 7 ducks and cross out the 2 that leave.}
\step{Count the ducks with no cross: 5. So $7-2=5$. $\checkmark$}

$\Rightarrow\quad$ there are $\ans{5}$ ducks left
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 9 ants are on a log. 5 crawl away. Draw it and count
how many are left.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Write the Take-Away Equation}

\begin{formulabox}
\textbf{Rule:} The story numbers go into the equation in order:\par
start number $-$ number that goes away $=$ number left.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 6 frogs sit on a log. 2 hop off. Write the equation
and solve it.
\step{The start number is 6 and the go-away number is 2, so the equation is $6-2$.}
\step{Count back from 6 twice: 5, 4. $\checkmark$}

$\Rightarrow\quad 6 - 2 = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 8 birds are on a wire. 3 fly away. Write the equation
and solve it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Compare by Matching}

\begin{formulabox}
\textbf{Rule:} ``How many more?'' means match the two groups one to one.
The unmatched extras are the answer --- and subtraction counts them:
\ bigger $-$ smaller $=$ how many more.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Ria has 8 shells. Dev has 5 shells. How many more
does Ria have?
\step{Comparing means matching: each of Dev's 5 shells pairs with one of Ria's, and the extras answer the question.}
\step{$8-5=3$ shells have no partner. $\checkmark$}

$\Rightarrow\quad$ Ria has $\ans{3}$ more shells
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One team has 6 balls, the other has 2. How many more
does the first team have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} ``How many \emph{fewer}?'' and ``how many
\emph{more}?'' use the same subtraction! Both count the unmatched extras ---
so $9-6=3$ answers both questions about 9 and 6.
\end{watchoutbox}
