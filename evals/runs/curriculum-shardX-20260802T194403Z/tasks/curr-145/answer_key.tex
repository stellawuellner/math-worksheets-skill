\problem{\textbf{Jump rope: $4$ ft into inches.}\\
\textbf{Step 1.} Feet are the bigger unit, so the count of inches must be
larger --- multiply, do not divide: $1$ ft $=12$ in.\\
\textbf{Step 2.} $4 \cdot 12 = 48$. \ans{48\ \text{in}}}

\problem{\textbf{Bottle: $3$ L into millilitres.}\\
\textbf{Step 1.} Litres are the bigger unit, and $1$ L $=1000$ mL, so multiply
by $1000$.\\
\textbf{Step 2.} $3 \cdot 1000 = 3000$. \ans{3000\ \text{mL}}}

\problem{\textbf{Gift box: $8$ cm by $5$ cm by $3$ cm.}\\
\textbf{Step 1.} Volume of a rectangular prism $=$ length $\cdot$ width
$\cdot$ height; all three edges are already in centimetres, so no conversion
is needed.\\
\textbf{Step 2.} $8 \cdot 5 = 40$ (the bottom layer holds $40$ cubes), and
$40 \cdot 3 = 120$ for three layers. \ans{120\ \text{cm}^3}}

\problem{\textbf{$\angle ABD = 35^\circ$, $\angle DBC = 47^\circ$.}\\
\textbf{Step 1.} Ray $BD$ is between the other two rays, so the two small
angles add to the whole angle: $\angle ABC = \angle ABD + \angle DBC$.\\
\textbf{Step 2.} $35 + 47 = 82$. \ans{82^\circ}}

\problem{\textbf{Flour: $2$ kg $250$ g into grams.}\\
\textbf{Step 1.} Convert the kilogram part first, then add the leftover grams:
$1$ kg $=1000$ g.\\
\textbf{Step 2.} $2 \cdot 1000 = 2000$ g, and $2000 + 250 = 2250$.
\ans{2250\ \text{g}}}

\problem{\textbf{Bin: volume $240$ cm$^3$, base $8$ cm by $6$ cm.}\\
\textbf{Step 1.} Volume $=$ base area $\cdot$ height, and here the volume is
known while the height is missing, so divide instead of multiply.\\
\textbf{Step 2.} Base area $= 8 \cdot 6 = 48$ square cm.\\
\textbf{Step 3.} $240 \div 48 = 5$, and the check multiplies back:
$48 \cdot 5 = 240$. \ans{5\ \text{cm}}}

\problem{\textbf{Right angle $PQR$ split by ray $QS$; $\angle PQS = 34^\circ$.}\\
\textbf{Step 1.} The two parts make a \emph{right} angle, so they add to
$90^\circ$ --- not $180^\circ$.\\
\textbf{Step 2.} $90 - 34 = 56$, and the check adds back:
$34 + 56 = 90$. \ans{56^\circ}}

\problem{\textbf{Film: $2$ h $25$ min into minutes.}\\
\textbf{Step 1.} Convert the hours, then add the leftover minutes:
$1$ h $=60$ min.\\
\textbf{Step 2.} $2 \cdot 60 = 120$ min, and $120 + 25 = 145$.
\ans{145\ \text{min}}}

\problem{\textbf{Carton $12$ cm by $10$ cm by $4$ cm packed with $2$ cm cubes.}\\
\textbf{Step 1.} Both solids are boxes, so compare volumes: how many
cube-volumes fit inside the carton volume?\\
\textbf{Step 2.} Carton: $12 \cdot 10 \cdot 4 = 480$ cubic cm. Cube:
$2 \cdot 2 \cdot 2 = 8$ cubic cm.\\
\textbf{Step 3.} $480 \div 8 = 60$ cubes. Counting by rows agrees:
$6$ cubes along the length, $5$ across the width, $2$ layers high, and
$6 \cdot 5 \cdot 2 = 60$. \ans{60}}

\problem{\textbf{Rosa's protractor reading of $130^\circ$.}\\
\textbf{Step 1.} A protractor carries two scales running in opposite
directions, and the two numbers at any mark always add to $180$. Rosa lined her
ray up with the zero of one scale but read the number printed on the other.\\
\textbf{Step 2.} The benchmark check: an acute angle is smaller than a square
corner, so its measure must be under $90^\circ$. A reading of $130^\circ$ is
more than $90^\circ$ and contradicts the picture --- that is the signal to
re-read the other scale.\\
\textbf{Model explanation (accept any equivalent wording):} ``Rosa read the
wrong scale. The angle is acute, so the answer has to be under $90^\circ$;
I take the other scale's number instead. Before writing any measure down I
decide first whether the angle is acute or obtuse, then pick the scale that
agrees.''\\
\textbf{Step 3.} The matching number on the other scale is
$180 - 130 = 50$. \ans{50^\circ}}

\problem{\textbf{Tank $40$ cm by $25$ cm by $30$ cm, answer in litres.}\\
\textbf{Step 1.} The question asks for litres but the edges are in
centimetres, so find the volume in cubic centimetres first and convert at the
end.\\
\textbf{Step 2.} $40 \cdot 25 = 1000$, and $1000 \cdot 30 = 30000$ cubic cm.\\
\textbf{Step 3.} $30000 \div 1000 = 30$. \ans{30\ \text{L}}}

\problem{\textbf{Crate $2$ ft by $18$ in by $1$ ft, answer in cubic feet.}\\
\textbf{Step 1.} The three edges are not in the same unit, and a volume in
cubic feet needs every edge in feet --- convert the width before
multiplying.\\
\textbf{Step 2.} $18 \div 12 = 1.5$, so the width is $1.5$ ft.\\
\textbf{Step 3.} $2 \cdot 1.5 = 3$, and $3 \cdot 1 = 3$. Multiplying
$2 \cdot 18 \cdot 1$ instead mixes feet with inches and lands on $36$, a
number in no unit at all. \ans{3\ \text{ft}^3}}
