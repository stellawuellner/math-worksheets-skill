\sstitleblock{Getting the Centre, the Vector and the Factor Right}

\begin{formulabox}
\textbf{Three things a transformation problem can get wrong, and they fail differently.} A wrong \emph{vector} sends the figure the wrong way. A wrong \emph{centre} gives the right size in the wrong place. A wrong \emph{scale factor} gives the wrong size. Naming which one broke is half the diagnosis.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The translation vector, in the right order}

\begin{formulabox}
\textbf{Rule:} the vector from $P(x, y)$ to $P'(p, q)$ has components $p - x$ and $q - y$ --- \emph{image minus original}, in that order.\par
Reversing the subtraction negates both components, which points the translation exactly backwards. Check by applying your rule to the original point and seeing whether it lands on the image.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A translation maps $(6, 2)$ to $(-1, 5)$. Find its components.
\step{The vector runs from the original point to the image, so each component is the image coordinate minus the original one, not the other way round.}
\step{horizontal: $-1 - 6$ \quad and \quad vertical: $5 - 2$}

$\Rightarrow\quad ($ $\ans{-7}$ $,$ $\ans{3}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A translation maps $(9, 4)$ to $(2, 4)$. Give its horizontal component.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-7}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Dilating about the given centre}

\begin{formulabox}
\textbf{Rule:} about a centre $(h, j)$ with factor $k$, \ $(x, y) \to \bigl(h + k(x - h),\ \ j + k(y - j)\bigr)$.\par
Only when the centre is the origin does this collapse to $(kx,\ ky)$. Using $(kx, ky)$ with any other centre produces a figure of the correct size in the wrong position --- and the centre is the one point that must not move.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Dilate $(8, 4)$ about $(3, 1)$ by a factor of $2$.
\step{The centre is not the origin, so scale each coordinate's offset from the centre and add it back --- multiplying the raw coordinates would dilate about the origin instead.}
\step{$x' = 3 + 2(8 - 3) = 3 + 10$ \quad and \quad $y' = 1 + 2(4 - 1) = 1 + 6$}

$\Rightarrow\quad ($ $\ans{13}$ $,$ $\ans{7}$ $)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Dilate $(7, 2)$ about $(5, 2)$ by a factor of $4$ and give the $x$-coordinate.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Getting the scale factor from the right ratio}

\begin{formulabox}
\textbf{From lengths:} $k = \dfrac{\text{image length}}{\text{original length}}$ --- a quotient, never a difference, and a pure number with no unit.\par
\textbf{From areas:} the area ratio equals $k^2$, so $k = \sqrt{\text{area ratio}}$. Handing the area ratio straight over as $k$ is the commonest slip in this topic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two similar figures have areas $36$ and $100$. Find the linear scale factor.
\step{The ratio given is a ratio of areas, and areas scale by the square of the linear factor, so the ratio is already $k$ squared --- take a square root before using it.}
\step{$k^2 = 100/36$, so $k = \sqrt{100/36} = 10/6$}

$\Rightarrow\quad k \approx$ $\ans{1.67}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two similar figures have areas $20$ and $45$. Find the linear scale factor.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1.5}$}
\end{tryitbox}
