\sstitleblock{Writing Growth and Decay Models from Data}

\skillheading{Skill 1 --- The multiplier from a table}

\begin{formulabox}
\textbf{Model shape:} $y = a \cdot r^{\,t}$, where $a$ is the amount at $t = 0$ and $r$ is what you multiply by in each period.\par
\textbf{From a table over equal time steps:} $r$ is the ratio of consecutive values. Check it on a second pair --- if the ratio changes, the data is not exponential.\par
\textbf{Reading $r$:} $r > 1$ is growth, $0 < r < 1$ is decay, and $r$ is never negative.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Values $200, 240, 288$ at $t = 0, 1, 2$. Find $r$ and predict $t = 3$.
\step{Equal time steps means the ratio test is the right tool, and the value at $t = 0$ is already the starting amount $a$.}
\step{$240 \div 200 = 1.2$, and $288 \div 240 = 1.2$ too, so the model is $200 \cdot 1.2^{\,t}$.}
\step{At $t = 3$: $1.2^3 = 1.728$, so $200 \times 1.728$.}

$r = \ans{1.2}$ \quad and \quad value at $t = 3$ is $\ans{345.6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Values $500, 450, 405$ at $t = 0, 1, 2$. Find $r$ and the value at $t = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{0.9}$, value $\ans{364.5}$}
\end{tryitbox}

\skillheading{Skill 2 --- The multiplier from a percent}

\begin{formulabox}
\textbf{Growth of $p$ percent:} $r = 1 + \dfrac{p}{100}$ --- you keep the whole and add $p$ hundredths. So $6\%$ growth is $r = 1.06$.\par
\textbf{Decay of $p$ percent:} $r = 1 - \dfrac{p}{100}$ --- you keep what is left. So $20\%$ decay is $r = 0.8$.\par
\textbf{The commonest error:} taking the percent of the \emph{original} amount every period. Each period's change is a percent of the amount at the \emph{start of that period}, which is what raising $r$ to a power does for you.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad \$1500 grows $6\%$ a year. What is it worth after $5$ years?
\step{A fixed percent per period is exponential, so convert the percent to a multiplier and raise it to the number of periods.}
\step{$r = 1 + \frac{6}{100} = 1.06$, so the model is $1500 \cdot 1.06^{\,t}$.}
\step{$1.06^5 = 1.3382255776$, so $1500 \times 1.3382255776 = 2007.338\ldots$}

$r = \ans{1.06}$, \quad value $= \ans{2007.34}$ dollars
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A \$900 machine loses $20\%$ of its value a year. What is it worth after $4$ years?\\[2pt]
\rotatebox{180}{\footnotesize check: $r = \ans{0.8}$, value $\ans{368.64}$}
\end{tryitbox}

\skillheading{Skill 3 --- Solving with the model}

\begin{formulabox}
\textbf{Doubling and half-life models:} ``doubles every $k$ periods'' is $a \cdot 2^{\,t/k}$; ``halves every $k$ periods'' is $a \cdot \left(\frac{1}{2}\right)^{t/k}$. The $t/k$ counts how many doublings or halvings have happened.\par
\textbf{To solve for $t$:} divide by $a$ first, write the result as a power of the same base, then match exponents and solve the small equation in $t$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A count follows $80 \cdot 2^{\,t/4}$. When does it reach $640$?
\step{The unknown is in the exponent, so clear the coefficient and rewrite both sides as powers of two --- no logarithm needed.}
\step{Divide by $80$: $2^{\,t/4} = 8 = 2^3$, so $\frac{t}{4} = 3$.}
\step{Multiply by $4$. Check: $80 \cdot 2^3 = 640$.}

$t = \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A mass follows $96 \cdot \left(\frac{1}{2}\right)^{t/6}$. When does it reach $12$?\\[2pt]
\rotatebox{180}{\footnotesize check: $t = \ans{18}$}
\end{tryitbox}
