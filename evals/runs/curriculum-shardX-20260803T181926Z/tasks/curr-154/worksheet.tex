\wstitleblock{Reversed and Non-Equivalent Ratios}{Grades 6--7}{}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A ratio is written in the order the question asks for. ``$A$ to $B$'' means $A$ on top, $B$ on the bottom --- reversing it names a different comparison.
  \item Two ratios are \textbf{equivalent} when you can get one from the other by multiplying (or dividing) \emph{both} parts by the same number. Adding the same number to both parts does not work.
  \item To test equivalence, simplify both ratios or compare their unit rates.
  \item Show your work in the space, then write your answer on the line.
\end{itemize}
\normalsize

\problem[4cm]{A muffin recipe uses $3$ cups of flour for every $2$ cups of sugar.
Write the ratio of flour to sugar as a fraction in simplest form.}

\problem[4.5cm]{Are the ratios $4 : 6$ and $6 : 9$ equivalent?
Simplify both, then write $=$ or $\neq$ between them and say which it is.
\begin{center}\large $\dfrac{4}{6} \ \underline{\hspace{1.4cm}} \ \dfrac{6}{9}$\end{center}}

\problem[5.5cm]{\textbf{Find the mistake.} Ms. Ruiz's class has $12$ boys and $18$ girls.
Leo is asked for the ratio of \textbf{girls to boys} and writes $\dfrac{12}{18}$.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain in one sentence why the order changes what the ratio means.
  \item[(b)] Write the ratio of girls to boys correctly, in simplest form.
\end{itemize}}

\problem[5cm]{A store sells $3$ pencils for $2$ dollars. Complete the equivalent-ratio
table, then find the cost $x$ of $12$ pencils.\par
\begin{center}
\begin{tabular}{|c|c|c|}\hline
\textbf{pencils} & $3$ & $12$ \\\hline
\textbf{dollars} & $2$ & $x$ \\\hline
\end{tabular}
\end{center}}

\problem[6cm]{\textbf{Find the mistake.} A drink mix uses $4$ cups of water to
$6$ scoops of powder. Mia needs $6$ cups of water instead. She adds $2$ to each part and
writes $8$ scoops of powder.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain why adding the same number to both parts does not keep a ratio equivalent.
  \item[(b)] Find the number of scoops that really is equivalent to $4$ cups $:$ $6$ scoops.
\end{itemize}}

\problem[5cm]{One car travels $150$ miles on $5$ gallons of gas. Another travels
$210$ miles on $7$ gallons. Find each unit rate in miles per gallon, then write $=$ or
$\neq$ between the two rates to say whether they are equivalent.}

\problem[5.5cm]{\textbf{Find the mistake.} A printer prints $24$ pages in $3$ minutes.
Sam reports the rate as $0.125$ pages per minute.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Explain in one sentence how to tell which number belongs on top in a rate.
  \item[(b)] Find the correct rate in pages per minute.
\end{itemize}}

\problem[6cm]{\textbf{Challenge.} One row of this ratio table does not belong.\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}\hline
\textbf{blue paint} & $2$ & $4$ & $6$ & $8$ \\\hline
\textbf{white paint} & $5$ & $10$ & $15$ & $22$ \\\hline
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Compare the last column with $2 : 5$ by writing $<$, $=$, or $>$ between
    $\dfrac{8}{22}$ and $\dfrac{2}{5}$.
  \item[(b)] Replace the wrong entry with the number that makes the column equivalent
    to $2 : 5$.
\end{itemize}}
