\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item To round, find the place you are rounding to, then look at the \textbf{one digit to its right}. That digit decides: $5$ or more rounds up, less than $5$ stays.
  \item A rounded number keeps every place it started with. Rounding $5{,}048$ to the nearest hundred gives $5{,}000$ --- the zeros hold the tens and ones places open.
  \item Always round the \textbf{original} number. Never round twice in a row.
  \item Show your thinking in the space, then write your answer on the line.
\end{itemize}

\problem[5.5cm]{\textbf{Find the mistake.} Jo rounds $4{,}750$ to the nearest hundred
and writes $4{,}700$. She says, ``the next digit is a $5$, and $5$ is not more than
$5$, so the $7$ stays.''
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
  \item[(a)] Which digit decides this rounding, and what is the rule when that digit is
    exactly $5$? Answer in one sentence.
  \item[(b)] Round $4{,}750$ to the nearest hundred correctly.
\end{itemize}}

  \item[(a)] Explain in one sentence why the rounded number must keep its zeros.
  \item[(b)] Round $5{,}048$ to the nearest hundred correctly.
\end{itemize}}
