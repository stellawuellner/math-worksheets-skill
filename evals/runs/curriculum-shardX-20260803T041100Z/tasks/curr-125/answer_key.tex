\problem{\textbf{Rewrite $\tfrac{3}{4}$ with denominator $12$.}\\
\textbf{Step 1.} Ask what the old denominator was multiplied by to reach the
new one: $4 \times 3 = 12$, so the scale factor is $3$.\\
\textbf{Step 2.} An equivalent fraction keeps the same value only if the
numerator is scaled by the \emph{same} factor: $3 \times 3 = 9$.\\
\textbf{Step 3.} $\tfrac{3}{4} = \ans{\,\dfrac{9}{12}}$}

\problem{\textbf{Write $2\tfrac{1}{3}$ as a fraction greater than one.}\\
\textbf{Step 1.} The whole number counts complete groups of thirds: each whole
is $\tfrac{3}{3}$, so $2$ wholes are $\tfrac{6}{3}$.\\
\textbf{Step 2.} Add the fractional part: $\tfrac{6}{3} + \tfrac{1}{3}$, which
keeps the denominator and adds the numerators.\\
\textbf{Step 3.} $2\tfrac{1}{3} = \ans{\,\dfrac{7}{3}}$}

\problem{\textbf{Compare $\tfrac{3}{8}$ and $\tfrac{1}{2}$.}\\
\textbf{Step 1.} Use the half benchmark --- half of $8$ is $4$, so
$\tfrac{4}{8}$ is exactly one half.\\
\textbf{Step 2.} The numerator $3$ is less than $4$, so $\tfrac{3}{8}$ sits
below the half mark while $\tfrac{1}{2}$ is on it.\\
\textbf{Step 3.} $\ans{\,\tfrac{3}{8} < \tfrac{1}{2}}$}

\problem{\textbf{Order $\tfrac{1}{2}$, $\tfrac{1}{4}$, $\tfrac{3}{8}$ least to
greatest.}\\
\textbf{Step 1.} All three denominators divide $8$, so eighths is the common
form --- no benchmark needed once they match.\\
\textbf{Step 2.} $\tfrac{1}{2} = \tfrac{4}{8}$ and $\tfrac{1}{4} =
\tfrac{2}{8}$; $\tfrac{3}{8}$ is already there.\\
\textbf{Step 3.} Order the numerators $2 < 3 < 4$, then write the numbers back
in their original form:
$\ans{\,\tfrac{1}{4} < \tfrac{3}{8} < \tfrac{1}{2}}$}

\problem{\textbf{Compare $\tfrac{7}{8}$ and $\tfrac{5}{6}$.}\\
\textbf{Step 1.} Both are just below one, so the half benchmark is useless ---
compare the \emph{gap to one} instead.\\
\textbf{Step 2.} $\tfrac{7}{8}$ is $\tfrac{1}{8}$ short of one;
$\tfrac{5}{6}$ is $\tfrac{1}{6}$ short of one.\\
\textbf{Step 3.} With equal numerators, the larger denominator is the smaller
piece: $\tfrac{1}{8} < \tfrac{1}{6}$. The number with the smaller gap is
closer to one, hence larger: $\ans{\,\tfrac{7}{8} > \tfrac{5}{6}}$\\
\textit{Check with a common denominator ($24$):} $\tfrac{21}{24}$ against
$\tfrac{20}{24}$ --- same conclusion.}

\problem{\textbf{Write $\tfrac{11}{4}$ as a mixed number.}\\
\textbf{Step 1.} A fraction greater than one hides whole groups: ask how many
complete fourths-groups of $4$ fit inside $11$.\\
\textbf{Step 2.} $11 \div 4 = 2$ with remainder $3$, so there are $2$ wholes
and $3$ fourths left over.\\
\textbf{Step 3.} $\tfrac{11}{4} = \ans{\,2\,\tfrac{3}{4}}$}

\problem{\textbf{Order $2\tfrac{1}{2}$, $\tfrac{9}{4}$, $2\tfrac{3}{8}$
greatest to least.}\\
\textbf{Step 1.} Two are mixed numbers and one is not, so convert the odd one
out: $\tfrac{9}{4} = 2\tfrac{1}{4}$. Now every number has whole part $2$, so
only the fraction parts decide the order.\\
\textbf{Step 2.} Put the fraction parts over eighths:
$\tfrac{1}{2} = \tfrac{4}{8}$, $\tfrac{3}{8}$ stays, $\tfrac{1}{4} =
\tfrac{2}{8}$.\\
\textbf{Step 3.} Greatest to least by numerator, $4 > 3 > 2$, written back in
the given forms: $\ans{\,2\,\tfrac{1}{2} > 2\,\tfrac{3}{8} > \tfrac{9}{4}}$}

\problem{\textbf{Compare $\tfrac{5}{9}$ and $\tfrac{4}{7}$.}\\
\textbf{Step 1.} Both are just above one half ($\tfrac{4.5}{9}$ and
$\tfrac{3.5}{7}$ are the halves), so the benchmark ties and a common
denominator is the reliable route.\\
\textbf{Step 2.} $9 \times 7 = 63$ works as a common denominator:
$\tfrac{5}{9} = \tfrac{35}{63}$ and $\tfrac{4}{7} = \tfrac{36}{63}$.\\
\textbf{Step 3.} $35 < 36$, so $\ans{\,\tfrac{5}{9} < \tfrac{4}{7}}$}

\problem{\textbf{Write $\tfrac{5}{6}$ with denominator $30$.}\\
\textbf{Step 1.} $6 \times 5 = 30$, so the scale factor is $5$.\\
\textbf{Step 2.} Scale the numerator by the same factor: $5 \times 5 = 25$.\\
\textbf{Step 3.} $\tfrac{5}{6} = \ans{\,\dfrac{25}{30}}$}

\problem{\textbf{Order $\tfrac{3}{5}$, $\tfrac{1}{2}$, $\tfrac{7}{10}$,
$\tfrac{13}{20}$ least to greatest.}\\
\textbf{Step 1.} All four denominators divide $20$, so twentieths is the one
common form that needs no guessing.\\
\textbf{Step 2.} $\tfrac{3}{5} = \tfrac{12}{20}$, $\tfrac{1}{2} =
\tfrac{10}{20}$, $\tfrac{7}{10} = \tfrac{14}{20}$, and $\tfrac{13}{20}$ is
already there.\\
\textbf{Step 3.} The numerators order as $10 < 12 < 13 < 14$, so
$\ans{\,\tfrac{1}{2} < \tfrac{3}{5} < \tfrac{13}{20} < \tfrac{7}{10}}$}

\problem{\textbf{Compare $1\tfrac{3}{4}$ and $\tfrac{5}{3}$.}\\
\textbf{Step 1.} The forms differ, so put both in one form.
$\tfrac{5}{3} = 1\tfrac{2}{3}$, and now both have whole part $1$.\\
\textbf{Step 2.} Compare the fraction parts $\tfrac{3}{4}$ and $\tfrac{2}{3}$
over twelfths: $\tfrac{9}{12}$ against $\tfrac{8}{12}$.\\
\textbf{Step 3.} $9 > 8$, so $\ans{\,1\,\tfrac{3}{4} > \tfrac{5}{3}}$}

\problem{\textbf{Challenge: order $\tfrac{5}{3}$, $1\tfrac{1}{2}$,
$\tfrac{11}{8}$, $1\tfrac{5}{6}$ least to greatest.}\\
\textbf{Step 1.} Three forms are in play, so pick ONE common form. Every number
is between one and two, so mixed numbers with a common fraction denominator is
the shortest route; the denominators $3$, $2$, $8$, $6$ all divide $24$.\\
\textbf{Step 2.} Convert to mixed numbers first:
$\tfrac{5}{3} = 1\tfrac{2}{3}$ and $\tfrac{11}{8} = 1\tfrac{3}{8}$; the other
two are already mixed. Every whole part is $1$, so only the fraction parts
decide.\\
\textbf{Step 3.} Over twenty-fourths: $\tfrac{2}{3} = \tfrac{16}{24}$,
$\tfrac{1}{2} = \tfrac{12}{24}$, $\tfrac{3}{8} = \tfrac{9}{24}$, and
$\tfrac{5}{6} = \tfrac{20}{24}$.\\
\textbf{Step 4.} Ordering those numerators gives $9 < 12 < 16 < 20$, so in the
original forms:
$\ans{\,\tfrac{11}{8} < 1\,\tfrac{1}{2} < \tfrac{5}{3} < 1\,\tfrac{5}{6}}$}
