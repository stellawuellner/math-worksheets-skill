\sstitleblock{Composing Piecewise and Nonlinear Functions}

\skillheading{Skill 1 --- Writing the composed rule}

\begin{formulabox}
\textbf{Definition.} $(f \circ g)(x) = f\bigl(g(x)\bigr)$: replace \emph{every} $x$ in the rule for $f$ by the whole expression $g(x)$, then simplify.\par
Order matters: $(f \circ g)$ and $(g \circ f)$ are different functions.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2} - 2x$, $g(x) = x + 1$. Write $(f \circ g)(x)$.
\step{Both $x$'s in the rule for $f$ must be replaced --- substituting into one of them is the standard slip, so do it before any expanding.}
\step{$(x+1)^{2} - 2(x+1) = x^{2} + 2x + 1 - 2x - 2$.}

$\Rightarrow\quad \ans{(f \circ g)(x) = x^{2} - 1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2} + 4x$, $g(x) = x - 2$. Write $(f \circ g)(x)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(f \circ g)(x) = x^{2} - 4}$}
\end{tryitbox}

\skillheading{Skill 2 --- Evaluating through a piecewise rule}

\begin{formulabox}
\textbf{Two-step rule.} Compute the inner value first. Then compare \emph{that} number --- not the original input --- with the boundary to pick the branch.\par
In $(f \circ f)(a)$ the branch is chosen twice, and often differently each time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $g(x) = 2x - 3$ and $f(x) = x + 6$ for $x < 1$, $f(x) = x^{2}$ for $x \geq 1$. Find $(f \circ g)(5)$.
\step{The branch depends on the value entering $f$, so compute $g(5)$ before looking at the cases at all.}
\step{$g(5) = 7$, and $7 \geq 1$, so use $x^{2}$: $7^{2} = 49$.}

$\Rightarrow\quad \ans{(f \circ g)(5) = 49}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $f$ and $g$. Find $(f \circ g)(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(f \circ g)(1) = 5}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} testing the original input against the boundary is the most common piecewise-composition error. The boundary belongs to $f$, so only the number entering $f$ gets tested.
\end{watchoutbox}

\skillheading{Skill 3 --- Solving a composition equation}

\begin{formulabox}
\textbf{Method.} Write $(f \circ g)(x)$ as one expression, set it equal to the target, and solve. With a square root, isolate it and square both sides --- then check every candidate in the \emph{original} equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = \sqrt{x + 1}$, $g(x) = 3x$. Solve $(f \circ g)(x) = 4$.
\step{Compose first so there is a single radical to isolate, rather than two functions to juggle.}
\step{$\sqrt{3x + 1} = 4$, so $3x + 1 = 16$ and $3x = 15$; check: $\sqrt{16} = 4$.}

$\Rightarrow\quad \ans{x = 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = \sqrt{x - 2}$, $g(x) = 4x$. Solve $(f \circ g)(x) = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{11}{4}}$}
\end{tryitbox}
