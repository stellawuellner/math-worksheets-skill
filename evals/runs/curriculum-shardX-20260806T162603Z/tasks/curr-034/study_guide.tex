% =================== SKILL 1 ===================
\skillheading{Making a number from tens and ones}

\begin{formulabox}
\textbf{Rule:} 1 ten is worth 10. To make the number, start at 10 and count
on the ones. The tens digit is always written first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What number is 1 ten and 7 ones?
\step{We know the tens and the ones --- start at the value of the ten, then count on the ones.}
\step{1 ten is 10. Count on 7 ones: 10, 11, 12, 13, 14, 15, 16, 17.}

$\Rightarrow\quad$ 1 ten and 7 ones make $\ans{17}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What number is 1 ten and 5 ones?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{15}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Finding the tens and ones in a number}

\begin{formulabox}
\textbf{Rule:} The left digit counts the tens. The right digit counts the
ones. Do not swap them --- the places give the digits their meaning!
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many tens and ones are in 18?
\step{Read the number place by place --- left digit for tens, right digit for ones.}
\step{18 is 10 and 8 more, so the 1 counts the tens and the 8 counts the ones.}

$\Rightarrow\quad$ 18 has $\ans{1}$ ten and $\ans{8}$ ones
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many tens and ones are in 14?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$ ten, $\ans{4}$ ones}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} 3 tens and 1 one is NOT the same as 1 ten and 3 ones.
Swapping the digits makes a different, much bigger or much smaller number.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Comparing numbers with swapped digits}

\begin{formulabox}
\textbf{Rule:} Compare the tens first. More tens means a bigger number. Only
when the tens are equal do you compare the ones.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which is more: 4 tens 1 one, or 1 ten 4 ones?
\step{Build both numbers first, then compare their tens.}
\step{4 tens 1 one is 41. \ 1 ten 4 ones is 14. \ 4 tens beats 1 ten.}

$\Rightarrow\quad \ans{41}$ is more than $\ans{14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which is more: 3 tens 1 one, or 1 ten 3 ones?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{31}$ is more}
\end{tryitbox}
