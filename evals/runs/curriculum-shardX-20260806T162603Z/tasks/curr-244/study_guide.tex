\sstitleblock{Legs, Hypotenuses, and the Pythagorean Theorem}

\skillheading{Skill 1 --- Finding the hypotenuse: add the squares}

\begin{formulabox}
\textbf{The theorem:} \quad $a^2 + b^2 = c^2$\\[3pt]
$c$ is the \textbf{hypotenuse}: the side opposite the right angle, and always the longest side of
the triangle. $a$ and $b$ are the two \textbf{legs}, the sides that meet at the right angle.\\[3pt]
When both legs are known and the hypotenuse is missing: $c = \sqrt{a^2 + b^2}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A right triangle has legs $6$ and $8$. Find the hypotenuse.
\step{Both known sides meet at the right angle, so they are legs and the missing side is the hypotenuse --- add.}
\step{$c = \sqrt{6^2 + 8^2} = \sqrt{36 + 64} = \sqrt{100}$}
$\Rightarrow\quad c = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A right triangle has legs $5$ and $12$. Find the hypotenuse.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{Skill 2 --- Finding a leg: subtract the squares}

\begin{formulabox}
\textbf{Rearranged:} \quad $a^2 = c^2 - b^2$, \quad so \quad $a = \sqrt{c^2 - b^2}$\\[3pt]
Use this whenever the \textit{longest} side is one of the numbers you were given. The
hypotenuse's square always comes first, and the known leg's square is taken away from it. Getting
this backwards is the single most common Pythagorean mistake.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Hypotenuse $10$, one leg $6$. Find the other leg.
\step{The longest side is given, so the missing side is a leg --- subtract, do not add.}
\step{$a = \sqrt{10^2 - 6^2} = \sqrt{100 - 36} = \sqrt{64}$}
$\Rightarrow\quad a = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Hypotenuse $26$, one leg $24$. Find the other leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 3 --- Checking someone else's substitution}

\begin{formulabox}
\textbf{Two questions, in this order:}
\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Which given side is the \textit{longest}? That one is the hypotenuse, and it is the only side allowed on its own in $a^2 + b^2 = c^2$.
  \item Is the answer smaller than the hypotenuse? A leg must be. An answer bigger than the hypotenuse is wrong before you check the arithmetic.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Hypotenuse $20$, leg $16$. A student wrote $\sqrt{20^2 + 16^2} = 25.61$.
Correct it.
\step{The student's answer is bigger than the hypotenuse $20$, which is impossible --- so a leg was substituted as if it were the hypotenuse.}
\step{The missing side is a leg: $\sqrt{20^2 - 16^2} = \sqrt{400 - 256} = \sqrt{144}$}
$\Rightarrow\quad \ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Hypotenuse $41$, leg $40$. A student wrote $\sqrt{41^2 + 40^2} = 57.28$.
Correct it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} on a coordinate grid the horizontal and vertical gaps are the legs and
the segment itself is the hypotenuse --- so a distance is always found by \textit{adding} the
squares.
\end{watchoutbox}
