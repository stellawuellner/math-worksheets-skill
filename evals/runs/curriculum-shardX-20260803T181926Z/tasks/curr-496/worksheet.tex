\wstitleblock{Convergence and Taylor Series --- Geometric Series, $p$-Series, and the $n$th-Term Test}{AP Calculus BC}{}

\begin{center}\small\textit{Classifying geometric and p-series by their own parameters,
and using the nth-term test to detect divergence}\end{center}

\noindent\small\textbf{Directions:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Name the \emph{family} first (geometric? $p$-series? neither?), then name the
        parameter that decides it ($r$ or $p$), then state the conclusion.
  \item A conclusion needs its reason attached. ``Converges by the geometric series test,
        since $|r| = \tfrac{2}{5} < 1$'' earns full credit; ``converges'' does not.
  \item Give exact values --- fractions, not decimals --- unless a problem says otherwise.
  \item Show the substitution into whichever formula you use.
\end{itemize}
\normalsize

\bigskip

\problem[6cm]{Consider $\displaystyle\sum_{n=0}^{\infty} 3\left(\tfrac{2}{5}\right)^{n}$,
the series behind Model A.

\smallskip
(a) Write down its first term $a$ and its common ratio $r$.

\smallskip
(b) State which test settles it, and why.

\smallskip
(c) Find the exact sum.}

\problem[6cm]{Let $a_n = \dfrac{n}{2n + 5}$.

\smallskip
(a) Find $\displaystyle\lim_{n \to \infty} a_n$.

\smallskip
(b) State what the $n$th-term test therefore concludes about
$\displaystyle\sum_{n=1}^{\infty} a_n$, and say in one sentence why that conclusion
follows.}

\problem[6cm]{Consider $\displaystyle\sum_{n=1}^{\infty} \frac{1}{n^{3}}$.

\smallskip
(a) This is a $p$-series. Write down $p$.

\smallskip
(b) Fill in the comparison that decides it: $p \ \underline{\hspace{1.2cm}}\ 1$ (write
$>$, $<$, or $=$).

\smallskip
(c) State the conclusion and the test that gives it.}

\problem[7cm]{Consider
$\displaystyle\sum_{n=1}^{\infty} 6\left(-\tfrac{1}{3}\right)^{n-1}$.

\smallskip
(a) Identify $a$ and $r$. Note the sign of $r$ --- what does a negative ratio do to the
terms?

\smallskip
(b) Decide whether the series converges, and find the exact sum if it does.}

\problem[7cm]{Let $a_n = \dfrac{3n^{2} + 1}{5n^{2} - n}$.

\smallskip
(a) Find $\displaystyle\lim_{n \to \infty} a_n$ exactly.

\smallskip
(b) Apply the $n$th-term test to
$\displaystyle\sum_{n=1}^{\infty} \frac{3n^{2} + 1}{5n^{2} - n}$ and state the conclusion.

\smallskip
(c) A classmate says the terms of any rational expression shrink to nothing, so the test
tells you nothing here. Say in one sentence why that is wrong for this $a_n$.}

\problem[8cm]{Consider $\displaystyle\sum_{n=1}^{\infty} \frac{1}{\sqrt{n}}$. This one
needs both ideas on the sheet.

\smallskip
(a) Write it as $\sum 1/n^{p}$ and fill in the comparison $p \ \underline{\hspace{1.2cm}}\ 1$.

\smallskip
(b) Find $\displaystyle\lim_{n \to \infty} \frac{1}{\sqrt{n}}$.

\smallskip
(c) Your answer to (b) means the $n$th-term test is \emph{inconclusive} here. Explain what
inconclusive means, then give the conclusion the $p$-series test does reach.}

\problem{\textbf{A geometric model you can see.} A ball is dropped from a height of
$12$ feet. After every bounce it returns to $\tfrac{3}{4}$ of the height it fell from, and
it keeps bouncing forever.

\smallskip
(a) After the first drop, the ball travels up and back down on every bounce, so the bounce
distances form the series $\displaystyle\sum_{n=1}^{\infty} 24\left(\tfrac{3}{4}\right)^{n}$.
Explain in one sentence where the $24$ comes from, then find that sum exactly.

\smallskip
(b) Add the original drop to get the total vertical distance the ball travels, in feet.
\par\vspace*{9cm}\answerline{ft}}

\problem[9cm]{\textbf{Challenge.} A classmate writes:

\smallskip
\emph{``The terms of $\sum_{n=1}^{\infty} \tfrac{1}{n}$ shrink to zero. Since the terms go
to zero, the series converges.''}

\smallskip
Model B on the front page is that series. Write a short paragraph that (i) names exactly
what the $n$th-term test can and cannot prove, (ii) gives the correct classification of
$\sum 1/n$ with its reason, and (iii) contrasts it with a geometric series whose terms
\emph{also} go to zero but which does converge. Name that geometric series.}
