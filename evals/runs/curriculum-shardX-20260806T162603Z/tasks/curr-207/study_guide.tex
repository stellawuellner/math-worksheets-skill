% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The zero exponent}

\begin{formulabox}
\textbf{Rule:} $x^0 = 1$ for every base except zero.
\par\smallskip
\textbf{Why:} dividing a power by itself gives one, and the quotient law says the same
division gives an exponent of zero. Both answers describe the same quotient, so the zero
power must be one. Whatever sits inside a bracket, if the whole bracket carries the zero
exponent the value is one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $12^0$.
\step{The exponent is zero, so this is the zero-exponent rule and nothing needs multiplying out.}
\step{Dividing that base's first power by itself gives one, and the quotient law writes the same division with an exponent of zero.}

$\Rightarrow\quad 12^0 = \ans{1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(5t)^0$, where $t$ is not zero.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Turning a negative exponent into a fraction}

\begin{formulabox}
\textbf{Rule:} \quad $x^{-n} = \dfrac{1}{x^n}$ \quad and \quad
$\dfrac{1}{x^{-n}} = x^n$ \quad (for $x \neq 0$).
\par\smallskip
A negative exponent moves the base across the fraction bar --- it never puts a minus sign
on the answer. Each step down in the exponent divides by the base once more, which is what
carries the pattern below one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $4^{-3}$ as an exact fraction.
\step{The exponent is negative, so the job is a reciprocal, not a subtraction --- rewrite before evaluating anything.}
\step{Flipping gives one over the third power of the base, and that power multiplies out to sixty-four.}

$\Rightarrow\quad 4^{-3} = \ans{\tfrac{1}{64}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $2^{-5}$ as an exact fraction.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{32}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a negative exponent never makes a value negative. Powers of a
positive base stay positive, however far the exponent drops --- they just get small.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The laws when exponents go negative}

\begin{formulabox}
\textbf{Method:} apply the product or quotient law exactly as usual, letting the exponents
add or subtract with their signs. Multiply or divide the number factors separately. Then
clean up: rewrite any negative exponent as a fraction so the final answer has positive
exponents only.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(2x^{-4})(3x^6)$.
\step{Same base and a multiplication, so the product law still applies --- the negative exponent changes the arithmetic, not the rule.}
\step{The number factors multiply, and the exponents add with their signs, leaving a positive exponent so no clean-up is needed.}

$\Rightarrow\quad (2x^{-4})(3x^6) = \ans{6x^2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(5y^{-3})(4y^7)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20y^4}$}
\end{tryitbox}
