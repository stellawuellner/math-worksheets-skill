\sstitleblock{Radical Equations, Domains, and Approximate Solutions}

\begin{formulabox}
\textbf{Four facts everything here rests on.}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=1pt]
\item $a^{m/n} = \sqrt[n]{a^m} = \left(\sqrt[n]{a}\right)^m$ --- denominator is
the root, numerator is the power.
\item $\sqrt{ab} = \sqrt{a}\,\sqrt{b}$, but $\sqrt{a+b} \neq \sqrt{a}+\sqrt{b}$.
\item $\sqrt{\phantom{x}}\,$ returns the \textbf{non-negative} root.
\item Squaring is not reversible, so it can create solutions the original
equation never had.
\end{itemize}
\end{formulabox}

\skillheading{1. Radicals and rational exponents}

\begin{formulabox}
Read the fraction as an instruction: denominator $=$ which root, numerator $=$
which power. Take the \textbf{root first} to keep the numbers small.\\[2pt]
A negative exponent means \textbf{reciprocal}, never a negative value:
$a^{-p} = \dfrac{1}{a^{p}}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Evaluate $25^{3/2}$ without a calculator.
\step{The denominator names the root and the numerator the power, so take the
square root first and the arithmetic stays in two digits.}
\step{$\sqrt{25} = 5$, and then the numerator cubes it: $5^3$.}
\[ 25^{3/2} = \ans{125} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Evaluate $64^{2/3}$ without a calculator.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{16}$}
\end{tryitbox}

\skillheading{2. Simplifying and combining radicals}

\begin{formulabox}
Simplify by pulling out the largest \textbf{perfect-square factor}:
$\sqrt{45} = \sqrt{9}\sqrt{5} = 3\sqrt{5}$.\\[2pt]
Two radical terms combine only when the radicands match \emph{after} simplifying
--- then add the coefficients and keep the radical unchanged. Radicands
themselves are never added.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Combine $\sqrt{18} + \sqrt{8}$ into one term.
\step{The radicands differ, so nothing can be decided until both radicals are in
simplest form.}
\step{$\sqrt{18} = 3\sqrt{2}$ and $\sqrt{8} = 2\sqrt{2}$, so they are like terms
and the coefficients add.}
\[ \sqrt{18} + \sqrt{8} = \ans{5\sqrt{2}} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Combine $\sqrt{45} + \sqrt{20}$ into one term.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5\sqrt{5}}$}
\end{tryitbox}

\skillheading{3. Solving radical equations and checking}

\begin{formulabox}
\textbf{1.} Isolate the radical. \quad \textbf{2.} Note the domain (radicand
$\geq 0$) \emph{and} the sign condition (the other side must be $\geq 0$).
\quad \textbf{3.} Square. \quad \textbf{4.} Solve. \quad \textbf{5.} Substitute
every candidate into the \textbf{original} equation.\\[2pt]
Step 5 is not optional: squaring merges $A = B$ with $A = -B$, so the squared
equation answers a different question than the one asked.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Solve $\sqrt{x+6} = x$.
\step{The left side cannot be negative, so any solution needs $x \geq 0$ --- note
that before squaring, because squaring erases the sign.}
\step{Squaring gives $x + 6 = x^2$, so $x^2 - x - 6 = 0$ and $(x-3)(x+2) = 0$;
testing shows $x = -2$ fails the sign condition while $x = 3$ checks out.}
\[ x = \ans{3} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Solve $\sqrt{x+8} = x + 2$, and say which candidate is
extraneous.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

\skillheading{4. Approximating irrational values}

\begin{formulabox}
Simplify \textbf{first}, approximate \textbf{last}. Rounding an intermediate
radical and then multiplying spreads the error into the final digit.\\[2pt]
Useful anchors: $\sqrt{2} \approx 1.414$, $\sqrt{3} \approx 1.732$,
$\sqrt{5} \approx 2.236$. Sanity-check by bracketing between perfect squares.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Approximate $\sqrt{11}$ to two decimal places.
\step{There is no perfect-square factor to remove, so bracket the value first to
know what a reasonable answer looks like.}
\step{$\sqrt{9} = 3$ and $\sqrt{16} = 4$, so the value sits just above 3; the
decimal is 3.3166, which rounds to two places.}
\[ \sqrt{11} \approx \ans{3.32} \]
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Approximate $2\sqrt{7}$ to two decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.29}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{a}+\sqrt{b}$ is never $\sqrt{a+b}$. A negative
exponent reciprocates rather than negating. And a candidate that lies inside the
radical's domain can still be extraneous --- the sign of the \emph{other} side is
the second condition.
\end{watchoutbox}
