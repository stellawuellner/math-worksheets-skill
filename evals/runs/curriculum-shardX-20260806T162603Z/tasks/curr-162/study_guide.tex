\sstitleblock{Finding the Whole or the Percent Rate}

% =================== SKILL 1 ===================
\skillheading{Finding the Rate from a Part and a Whole}

\begin{formulabox}
\textbf{Rule:} the rate is the part measured against the whole, then read as
hundredths.
\[
  \text{rate} = \frac{\text{part}}{\text{whole}} \times 100 \ \text{percent}
\]
Divide first, multiply by 100 second. A rate above 100 percent means the part
came out bigger than the whole --- go back and check which number is which.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A team won 21 of its 28 matches. What percent did it
win?
\step{The part (21 wins) and the whole (28 matches) are both given, so the
missing number is the rate --- divide part by whole.}
\step{$21 \div 28 = $ $\ans{0.75}$, and $0.75 \times 100 = $ $\ans{75}$
percent.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A choir of 40 singers has 26 sopranos. What percent are
sopranos?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{65}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Finding the Whole from a Part and its Rate}

\begin{formulabox}
\textbf{Rule:} when the rate and the part are given and the whole is missing,
divide by the rate written as a decimal.
\[
  \text{whole} = \frac{\text{part}}{\text{rate} \div 100}
\]
The answer must come out \emph{larger} than the part whenever the rate is under
100 percent. That is the fastest sanity check there is.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 36 pages is 45 percent of a book. How many pages is the
whole book?
\step{The missing number is the whole, so turn the rate into a decimal and
divide the part by it.}
\step{$45 \div 100 = $ $\ans{0.45}$, and $36 \div 0.45 = $ $\ans{80}$ pages ---
larger than 36, as it must be.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 51 tickets is 30 percent of the tickets printed. How many
were printed?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{170}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Writing and Solving the Percent Equation}

\begin{formulabox}
\textbf{Rule:} one equation covers every case. Write
$\dfrac{p}{100} \times \text{whole} = \text{part}$, put the letter $x$ wherever
the missing number is, and solve the equation you get. Undo a multiplication by
dividing both sides by the same factor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 19 members is 25 percent of a club. Write and solve the
percent equation for the club's size $x$.
\step{The rate and the part are known, so the letter goes on the whole:
$\dfrac{25}{100}\,x = 19$.}
\step{$\dfrac{25}{100} = 0.25$, so $0.25\,x = 19$ and dividing both sides by
$0.25$ gives $x = $ $\ans{76}$ members.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{40}{100}\,x = 34$ for $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{85}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the rate is always part $\div$ whole, never whole
$\div$ part. Dividing the wrong way round on ``17 of 60 birds'' gives about 353
percent instead of 28.3 percent.
\end{watchoutbox}
