\sstitleblock{Right Triangles: Choosing the Method}

\begin{formulabox}
\textbf{Read the givens, then pick the tool.} Two \emph{sides} and no angle
$\rightarrow$ Pythagoras. An angle of $45^\circ$, $30^\circ$ or $60^\circ$
$\rightarrow$ the exact special-triangle ratios. Any other angle, or an unknown
angle $\rightarrow$ sine, cosine or tangent.
\end{formulabox}

\skillheading{Skill 1 --- The Pythagorean theorem}

\begin{formulabox}
\textbf{Rule.} With legs $a$, $b$ and hypotenuse $c$: \ $a^2 + b^2 = c^2$.
Solving for the hypotenuse you \emph{add} the squares; solving for a leg you
\emph{subtract} the known leg's square from the hypotenuse's.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has legs $6$ and $8$. Find the hypotenuse.
\step{Two sides are known and no angle is given, so trigonometry has nothing to
work with --- Pythagoras is the tool, and the unknown is the hypotenuse, so add.}
\step{$c = \sqrt{6^2 + 8^2} = \sqrt{36 + 64}$}

$c = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has hypotenuse $25$ and one leg $7$. Find the
other leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

\skillheading{Skill 2 --- Special right triangles}

\begin{formulabox}
\textbf{Two exact shapes worth memorising.} In a $45$--$45$--$90$ triangle the
legs are equal and the hypotenuse is a leg times $\sqrt{2}$. In a
$30$--$60$--$90$ triangle the side opposite $30^\circ$ is half the hypotenuse,
and the side opposite $60^\circ$ is that short leg times $\sqrt{3}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} An isosceles right triangle has legs of $5$. Find the
hypotenuse.
\step{Equal legs means the acute angles are both $45^\circ$, so the exact
$45$--$45$--$90$ ratio applies and no calculator work is needed to set it up.}
\step{$c = 5\sqrt{2} = 7.0710\ldots$}

$c \approx \ans{7.07}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} An isosceles right triangle has legs of $9$. Find the
hypotenuse.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12.73}$}
\end{tryitbox}

\skillheading{Skill 3 --- A side from an angle}

\begin{formulabox}
\textbf{Rule.} Label the sides relative to the known angle, then choose:
$\sin = \text{opp}/\text{hyp}$, $\cos = \text{adj}/\text{hyp}$,
$\tan = \text{opp}/\text{adj}$. If the unknown is on top, multiply; if it is
underneath, divide.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has hypotenuse $20$ and one acute angle
$28^\circ$. Find the side adjacent to that angle.
\step{The angle is not one of the special ones, so trigonometry it is; adjacent
with hypotenuse is cosine, and the unknown is on top.}
\step{$\text{adj} = 20\cos 28^\circ = 17.6589\ldots$}

$\text{adj} \approx \ans{17.66}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has a side of $14$ adjacent to a $61^\circ$
angle. Find the side opposite that angle.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{25.26}$}
\end{tryitbox}

\skillheading{Skill 4 --- An angle from two sides}

\begin{formulabox}
\textbf{Rule.} Form the ratio the two known sides make, then undo it with the
matching inverse: opp and hyp $\rightarrow \sin^{-1}$; adj and hyp
$\rightarrow \cos^{-1}$; opp and adj $\rightarrow \tan^{-1}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} A right triangle has a side of $9$ opposite $\angle A$ and a
hypotenuse of $20$. Find $m\angle A$.
\step{An angle is unknown and two sides are known --- opposite with hypotenuse
is the sine pairing, so the inverse sine recovers the angle.}
\step{$A = \sin^{-1}(9/20) = 26.7436\ldots^\circ$}

$m\angle A \approx \ans{26.74}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} A right triangle has legs $5$ and $12$. Find the acute angle
opposite the leg of length $5$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{22.62}$ degrees}
\end{tryitbox}
