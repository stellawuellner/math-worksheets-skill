\wstitleblock{Explaining Legal Moves in Equations and Inequalities}{Algebra 1}{}

\begin{center}
\textit{Balance models, the properties of equality, and why one move flips an inequality sign}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item A move is \emph{legal} when it keeps the equation true. Doing the same thing to both sides
        is legal; rewriting one side into an equal expression is legal. Anything else is not.
  \item When a problem asks you to name a move, say \emph{what was done} and \emph{that it was done
        to both sides} --- for example, ``subtracted $2x$ from both sides''.
  \item Blanks inside the balance model are working space. The blank marked \textbf{Answer} is the
        graded one.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[5.5cm]{A balance holds $x + 4$ grams on the left pan and $11$ grams on the right pan, and
it is level. Taking $4$ grams off \emph{both} pans keeps it level. Fill in the model to show the
balance after that move, then give the value of $x$.
\par\smallskip
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
 & \textbf{Left pan} & \textbf{Right pan}\\
\hline
Now            & $x + 4$ & $11$ \\[0.15cm]\hline
After removing $4$ from both pans & \scratchblank & \scratchblank \\[0.4cm]\hline
\end{tabular}
\end{center}
\par\smallskip
\ansline}

\problem[4.5cm]{Here is a complete solution, one line to the next:
\par\smallskip
\hspace*{1.2cm} $3x - 7 = 14 \qquad\longrightarrow\qquad 3x = 21 \qquad\longrightarrow\qquad x = 7$
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Name the move that turns the first line into the second. \ansline
  \item[(b)] Name the move that turns the second line into the third. \ansline
  \item[(c)] Use the same two kinds of move to solve $5x + 2 = 27$. \ansline
\end{itemize}}

\problem[5cm]{Solve $4(x - 3) = 20$. Two different first moves are legal here: dividing both sides
by $4$, or rewriting the left side using the distributive property. Either one works.}

\problem[5cm]{Solve the inequality $-3x < 12$. Write your answer as an inequality in $x$.}

\problem[4.5cm]{A student solved an equation like this:
\par\smallskip
\hspace*{1.2cm} $2x + 5 = 3x - 1 \qquad\longrightarrow\qquad 5 = x - 1 \qquad\longrightarrow\qquad 6 = x$
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Name the move used at each of the two steps. \ansline
  \item[(b)] Solve the equation yourself and give the value of $x$. \ansline
\end{itemize}}

\problem[5cm]{Solve $\dfrac{x}{5} - 3 = 4$.}

\problem[4.5cm]{Consider the inequality $5 - 2x \ge 11$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Solve it. Write your answer as an inequality in $x$. \ansline
  \item[(b)] Say whether the inequality sign changes direction anywhere in your work, and name the
             exact step where that happens or explain why it never does. \ansline
\end{itemize}}

\problem[5cm]{One of these steps is not legal:
\par\smallskip
\hspace*{1.2cm} $3(x + 2) = 18 \ \longrightarrow\ 3x + 2 = 18 \ \longrightarrow\ 3x = 16 \ \longrightarrow\ x = \dfrac{16}{3}$
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Name the step that is not legal and explain what rule it breaks. \ansline
  \item[(b)] Solve the equation correctly. \ansline
\end{itemize}}

\problem[5.5cm]{Solve the inequality $-\dfrac{x}{4} + 1 > 3$. Write your answer as an inequality
in $x$.}

\problem[6cm]{Consider the equation $\dfrac{2x - 1}{3} + 4 = x$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=\subsep]
  \item[(a)] Solve it. \ansline
  \item[(b)] Explain which property justifies each move you made, and mark clearly which of your
             moves were done to \emph{both} sides and which only rewrote \emph{one} side. \ansline
\end{itemize}}
