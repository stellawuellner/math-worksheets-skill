\sstitleblock{Slope, Parallel Lines and Perpendicular Lines}

\skillheading{Skill 1 --- Slope from two points}

\begin{formulabox}
\textbf{Rule.} $m = \dfrac{y_2-y_1}{x_2-x_1}$ --- rise over run. Take both
differences in the \emph{same} order; swapping one of them flips the sign. A run
of zero (two points with the same $x$) means a vertical line, whose slope is
undefined.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the slope of the line through $(1,2)$ and $(5,14)$.
\step{Want a slope from two points --- subtract the $y$-values for the rise and
the $x$-values for the run, keeping the same point first in both.}
\step{$m = \dfrac{14-2}{5-1} = \dfrac{12}{4}$}

$m = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the slope of the line through $(-2,6)$ and $(4,-3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m = \ans{-\tfrac{3}{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- The parallel test}

\begin{formulabox}
\textbf{Rule.} Two lines are parallel exactly when their slopes are
\emph{equal}. So a line parallel to a given one has the same slope --- find the
given line's slope and you are done; the points it passes through do not change
it.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the slope of any line parallel to the line through
$(0,1)$ and $(4,7)$.
\step{Parallel means equal slopes, so the whole job is the slope of the line I
was given.}
\step{$m = \dfrac{7-1}{4-0} = \dfrac{6}{4}$}

$m = \ans{\tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the slope of any line parallel to the line through
$(-1,5)$ and $(3,-3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m = \ans{-2}$}
\end{tryitbox}

\skillheading{Skill 3 --- The perpendicular test}

\begin{formulabox}
\textbf{Rule.} Two lines are perpendicular exactly when
$m_1 \times m_2 = -1$, that is, when each slope is the \emph{negative
reciprocal} of the other: flip the fraction and change the sign. Read straight
off two points, the perpendicular slope is
\[ -\frac{x_2-x_1}{y_2-y_1}. \]
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the slope of any line perpendicular to the line through
$(2,1)$ and $(6,9)$.
\step{Perpendicular means negative reciprocal, so I can swap rise and run and
change the sign in one step instead of computing the slope first.}
\step{$m = -\dfrac{6-2}{9-1} = -\dfrac{4}{8}$}

$m = \ans{-\tfrac{1}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the slope of any line perpendicular to the line through
$(-4,3)$ and $(2,-3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m = \ans{1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the negative reciprocal needs \emph{both} moves. Flipping
$\tfrac{2}{3}$ to $\tfrac{3}{2}$ without the minus sign, or negating to
$-\tfrac{2}{3}$ without flipping, gives a line that is not perpendicular.
\end{watchoutbox}

\skillheading{Skill 4 --- Classifying a figure by slopes}

\begin{formulabox}
\textbf{How the tests classify a quadrilateral.} Compute the slope of each side,
then read off: both pairs of opposite sides with equal slopes means a
parallelogram; exactly one pair means a trapezoid; a parallelogram whose
adjacent slopes multiply to $-1$ is a rectangle.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} In quadrilateral $ABCD$ the slope of $\overline{DC}$ is
$\tfrac{1}{3}$, and $A$ is at $(0,1)$ with $B$ at $(6,3)$. Find the slope of
$\overline{AB}$, and say whether $\overline{AB} \parallel \overline{DC}$.
\step{$\overline{AB}$ and $\overline{DC}$ are opposite sides, so the parallel
test applies: compute the slope of $\overline{AB}$ and compare.}
\step{$m_{AB} = \dfrac{3-1}{6-0} = \dfrac{2}{6}$, which equals the slope of
$\overline{DC}$, so the two sides are parallel.}

$m_{AB} = \ans{\tfrac{1}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} In quadrilateral $ABCD$ the slope of $\overline{DC}$ is
$\tfrac{1}{2}$, and $A$ is at $(-2,0)$ with $B$ at $(4,3)$. Find the slope of
$\overline{AB}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m_{AB} = \ans{\tfrac{1}{2}}$}
\end{tryitbox}
