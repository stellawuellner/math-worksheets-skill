\wstitleblock{The Derivative as a Limit and an Instantaneous Rate}{AP Calculus AB / BC}{}

\vspace{0.1cm}
\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt]
  \item The definition in use throughout:
        $\displaystyle f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}$. The
        quotient before the limit is an \emph{average} rate over a window of
        width $h$; the limit is the \emph{instantaneous} rate at $a$.
  \item Where a table appears, fill it in first. The table is the numerical
        picture of the limit --- it shows the average rates closing in on a
        value; the algebra then proves what that value is.
  \item Simplify the difference quotient \emph{before} letting $h \to 0$;
        substituting $h = 0$ first gives $0/0$, which says nothing.
\end{itemize}

\problem[4.5cm]{Let $f(x) = x^2$. Complete the table of difference quotients at
$a = 4$, then find $f'(4)$ from the definition by simplifying the quotient and
taking the limit.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$h$ & $1$ & $0.5$ & $0.1$ & $0.01$ \\ \hline
$\dfrac{f(4+h)-f(4)}{h}$ & & & & \\[0.8cm]\hline
\end{tabular}
\end{center}}

\problem{A cart's position is $s(t) = t^2 + 2t$ metres at time $t$
seconds. Complete the average-velocity table on the windows starting at
$t = 2$, then compute the instantaneous velocity at $t = 2$ from the limit
definition.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
window width $h$ & $1$ & $0.5$ & $0.1$ & $0.01$ \\ \hline
average velocity & & & & \\[0.8cm]\hline
\end{tabular}
\end{center}
\par\vspace*{4.5cm}\par\vspace*{0pt}\answerline{meters per second}}

\problem[4.5cm]{A tank holds $V(t) = t^3 - 6t^2 + 9t$ gallons of water $t$
minutes after a valve opens. Find $V'(t)$, and write one sentence saying what
$V'(t)$ measures about the tank --- not what it equals, but what it tells you.}

\problem{Using your $V'(t)$ from problem 3, find the instantaneous rate at
which the tank's volume is changing at $t = 5$ minutes. State whether the tank
is filling or draining at that instant, and how you know from the sign.
\par\vspace*{5cm}\par\vspace*{0pt}\answerline{gallons per minute}}

\problem[4.5cm]{The expression below is a derivative in disguise:
\[
\lim_{h \to 0} \frac{(5+h)^3 - 125}{h}
\]
Name the function $f$ and the point $a$ for which this is $f'(a)$, explain in
one sentence how you recognised them, then evaluate the limit.}

\problem[5cm]{Let $f(x) = \dfrac{1}{x}$. Use the limit definition to find
$f'(2)$. Combine the two fractions in the numerator over a common denominator
before you cancel the $h$; write the value as a decimal.}

\problem{A runner's distance from the start line is measured at three
times, with no formula available.
\par
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$t$ (seconds) & $2$ & $3$ & $4$ \\ \hline
distance (metres) & $29$ & $35$ & $41$ \\ \hline
\end{tabular}
\end{center}
\par
Estimate the runner's instantaneous velocity at $t = 3$ by using the window
that is centred on $t = 3$, and say in one sentence why that centred window is
a better estimate than either one-sided window.
\par\vspace*{4.5cm}\par\vspace*{0pt}\answerline{meters per second}}

\problem[6cm]{Take any linear function $f(x) = mx + k$. Using only the limit
definition, explain why the instantaneous rate of change is the same at every
point, and why no table of average rates for a linear function ever needs a
limit at all. Then state the units of $V'(t)$ from problem 3 and explain where
those units come from in the difference quotient. Write in complete sentences.
\noansline}
