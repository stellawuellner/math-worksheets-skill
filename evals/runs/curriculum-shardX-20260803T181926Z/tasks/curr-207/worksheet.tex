\wstitleblock{Zero and Negative Exponents}{Grade 8 / Pre-Algebra \quad Expressions and Equations}{}

\begin{center}
\textit{Evaluating zero and negative powers, rewriting expressions with
positive exponents, applying the exponent laws, and comparing powers}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Two definitions do all the work: \ $a^0 = 1$ for every $a \neq 0$, \ and
\ $a^{-n} = \dfrac{1}{a^{n}}$.
\item A negative exponent never makes a value negative. It means
\emph{reciprocal}.
\item Give numeric answers as exact fractions in lowest terms, and give
algebraic answers with \textbf{positive exponents only}.
\item Show the law you used (product, quotient, or power of a power) before
you simplify.
\end{itemize}

\problem[4cm]{Evaluate. \qquad $7^0$}

\problem[4cm]{Evaluate. Write your answer as a fraction. \qquad $5^{-2}$}

\problem[5cm]{Evaluate. Write your answer as a fraction.
\qquad $\left(\dfrac{2}{3}\right)^{-2}$}

\problem[5cm]{Rewrite with positive exponents only. \qquad $3x^{-2}$}

\problem[5cm]{Simplify using the product law, and write the result with a
positive exponent. \qquad $x^{5} \cdot x^{-8}$}

\problem[5.5cm]{Which is larger? Write $>$ or $<$ between the two powers so
the sentence is true. Show the value of each one.
\[ 2^{-3} \underline{\hspace{1.2cm}} 3^{-2} \]}

\problem[6cm]{A grain of sand has a mass of about $4.5 \times 10^{-3}$ grams.
Write that mass as an ordinary decimal number.}

\problem[6cm]{Simplify, and write the result with a positive exponent.
\qquad $\left(x^{-2}\right)^{3} \cdot x^{4}$}

\problem[6cm]{Rewrite with positive exponents only, then simplify.
\[ \frac{2x^{-3}}{y^{-2}} \]}

\problem[6cm]{Order these three powers from \textbf{least to greatest}. Write
the value of each one first.
\[ 5^{0}, \qquad 5^{-1}, \qquad 5^{1} \]}

\problem[7cm]{\textbf{Find and fix the mistake.} Dana was asked to evaluate
$4^{-2}$ and wrote $-16$. Explain in one sentence what Dana did wrong, then
write the correct value as a fraction.}

\problem[8cm]{Simplify completely. Your answer must have positive exponents
only.
\[ \frac{3x^{-2}y^{4}}{6x^{3}y^{-1}} \]}
