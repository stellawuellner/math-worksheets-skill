\sstitleblock{Reading Line Plots Marked in Fractions}

\skillheading{Skill 1 --- Reading a plot marked in fractions}

\begin{formulabox}
\textbf{Two kinds of number live on one plot.} Every \textsf{X} stands for one thing that was
measured. The fraction printed underneath says how big that one thing was.\par
\smallskip
So \emph{how many} questions are answered by counting \textsf{X}'s, and the answer is a whole
number. \emph{How much} questions are answered by the fractions, and the answer is a
length or an amount.\par
\smallskip
Check the scale before you read a single stack: one plot may count by eighths and the next by
quarters, even though the marks look identical.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many ribbons were $\frac{1}{2}$ inch long?
\begin{center}
\begin{tikzpicture}[x=1.3cm, y=0.34cm]
  \draw[thick] (0.4,0) -- (3.6,0);
  \foreach \v in {1,...,3} {\draw (\v,0.45) -- (\v,-0.45);}
  \node[below, font=\small] at (1,-0.45) {$\frac{1}{4}$};
  \node[below, font=\small] at (2,-0.45) {$\frac{1}{2}$};
  \node[below, font=\small] at (3,-0.45) {$\frac{3}{4}$};
  \foreach \v/\n in {1/1, 2/4, 3/1} {%
     \foreach \i in {1,...,\n} {\node[font=\small] at (\v,\i) {\textsf{X}};}}
  \node[font=\small] at (2,-5.4) {Ribbon length (inches)};
\end{tikzpicture}
\end{center}
\step{The question asks how many \emph{ribbons}, so the answer is the height of a stack ---
find the $\frac{1}{2}$ mark and count up, do not read the fraction.}
\step{Above the middle mark: \textsf{X}, \textsf{X}, \textsf{X}, \textsf{X}.}

$\Rightarrow\quad \ans{4}$ ribbons were half an inch long.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same plot, how many ribbons were $\frac{3}{4}$ inch long?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$ ribbon}
\end{tryitbox}

\skillheading{Skill 2 --- Comparing what the plot shows}

\begin{formulabox}
\textbf{Decide first what kind of answer the question wants.}\par
\smallskip
\emph{How many more of one than the other?} $\rightarrow$ subtract the two stack heights, and
the answer is a count.\par
\smallskip
\emph{How much longer / how much more?} $\rightarrow$ subtract the two fractions from the
bottom of the line, and the answer is a length or an amount. The longest is the rightmost
\textsf{X}; the shortest is the leftmost.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad On the ribbon plot, how much longer is the longest ribbon than the
shortest one?
\step{``How much longer'' asks about size, so this one uses the fractions along the bottom,
not the stacks.}
\step{Rightmost \textsf{X} is on $\frac{3}{4}$, leftmost is on $\frac{1}{4}$, and
$\frac{3}{4} - \frac{1}{4} = \frac{2}{4}$.}

$\Rightarrow\quad \ans{\tfrac{1}{2}}$ of an inch longer.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad On the same plot, how many \emph{more} ribbons were $\frac{1}{2}$ inch
long than were $\frac{1}{4}$ inch long?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ more ribbons}
\end{tryitbox}

\skillheading{Skill 3 --- Totalling and sharing equally}

\begin{formulabox}
\textbf{To total:} write down one fraction for every \textsf{X} --- a stack of four
$\frac{1}{2}$'s means four halves, not one --- rewrite them all over the same denominator, and
add.\par
\smallskip
\textbf{To share equally:} divide that total by the number of \textsf{X}'s (not by the number
of marks on the line). The share is what every item \emph{would} be if they were all levelled
out, so it need not be a size anything actually was.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad On the ribbon plot, how much ribbon is there altogether, and how long
would each ribbon be if they were all made the same?
\step{Six \textsf{X}'s means six lengths to add, so rewrite everything in quarters --- the
smallest unit on this line.}
\step{$\frac{1}{4} + \frac{2}{4} + \frac{2}{4} + \frac{2}{4} + \frac{2}{4} + \frac{3}{4}
= \frac{12}{4}$ inches, and sharing that among six ribbons gives $12 \div 6 = 2$ quarters
each.}

$\Rightarrow\quad \ans{3}$ inches in total, or $\ans{\tfrac{1}{2}}$ of an inch each.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Four ribbons measure $\frac{1}{2}$, $\frac{3}{4}$, $\frac{3}{4}$ and
$1$ inch. Made all the same, how long would each be?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{4}}$ of an inch each}
\end{tryitbox}
