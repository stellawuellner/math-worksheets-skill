\sstitleblock{Distance and Midpoint on the Coordinate Plane}

\skillheading{Skill 1 --- The distance formula}

\begin{formulabox}
\textbf{Where it comes from.} Draw the horizontal and vertical gaps between the
two points: they are the legs of a right triangle whose hypotenuse is the
segment. Pythagoras then gives
\[ d = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}. \]
Squaring erases the signs, so it never matters which point you call first.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the distance between $(1,2)$ and $(7,10)$.
\step{Want a straight-line length from two points --- so build the right
triangle: the legs are the gaps in $x$ and in $y$, and the distance is its
hypotenuse.}
\step{gaps $7-1 = 6$ and $10-2 = 8$, so
$d = \sqrt{6^2 + 8^2} = \sqrt{36+64}$}

$d = \ans{10}$ units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the distance between $(-2,1)$ and $(3,13)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $d = \ans{13}$ units}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} adding the two gaps gives how far you would \emph{walk}
along the grid, not the straight-line distance. The square root is what turns
two legs into one hypotenuse.
\end{watchoutbox}

\skillheading{Skill 2 --- The midpoint formula}

\begin{formulabox}
\textbf{Rule.} The midpoint is the pair of averages:
\[ M = \left( \frac{x_1+x_2}{2},\; \frac{y_1+y_2}{2} \right). \]
Half the horizontal run pairs with half the vertical rise, so the averaged point
lands on the segment, not merely between the endpoints. The answer is an
\emph{ordered pair}, and it may have a half in it.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} Find the midpoint of $(-3,8)$ and $(9,-2)$.
\step{Want the point half way along --- averaging is what "half way" means, and
each coordinate is averaged on its own.}
\step{$\dfrac{-3+9}{2} = 3$ \quad and \quad $\dfrac{8+(-2)}{2} = 3$}

$M = \ans{(3,\, 3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Find the midpoint of $(5,-7)$ and $(-1,3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $M = \ans{(2,\, -2)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Working back to a missing endpoint}

\begin{formulabox}
\textbf{Rule.} If $M$ is the midpoint of $\overline{AB}$ and you know $A$, then
each coordinate of $B$ is
\[ 2 \times (\text{coordinate of } M) - (\text{coordinate of } A), \]
because the midpoint is the average of the two. Do the $x$-coordinates and the
$y$-coordinates separately, then check by averaging your answer back.
\end{formulabox}

\begin{examplebox}
\textbf{Example.} The midpoint of $\overline{AB}$ has $x$-coordinate $4$, and
endpoint $A$ has $x$-coordinate $-2$. Find the $x$-coordinate of $B$.
\step{The midpoint's coordinate is the average of the two endpoints', so
doubling it and removing the endpoint I know leaves the one I want.}
\step{$x_B = 2(4) - (-2) = 8 + 2$}

$x_B = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} The midpoint of $\overline{AB}$ has $x$-coordinate $1$, and
endpoint $A$ has $x$-coordinate $6$. Find the $x$-coordinate of $B$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x_B = \ans{-4}$}
\end{tryitbox}
