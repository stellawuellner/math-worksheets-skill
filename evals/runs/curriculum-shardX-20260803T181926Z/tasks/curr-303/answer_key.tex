\problem{\textbf{Given $3x + 7 = 25$; prove $x = 6$.}
\par
Each line changes the equation, so each line names the property that permits
the change.
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $3x + 7 = 25$ & 1.\ Given \\ \hline
2.\ $3x = 18$ & 2.\ Subtraction Property of Equality \\ \hline
3.\ $x = 6$ & 3.\ Division Property of Equality \\ \hline
\end{tabular}}
\par\smallskip
$\ans{x = 6}$
\par
\emph{Manual review:} the Reasons column is written work --- accept
``subtracted $7$ from both sides'' only when the property is named.}

\problem{\textbf{$B$ between $A$ and $C$: $AB = 2x+5$, $BC = x+3$, $AC = 26$ cm.}
\par
\textbf{(a)} The \textbf{Segment Addition Postulate}. Betweenness is what
licenses adding the two parts; it is assumed, not proved.
\par
\textbf{(b)} Substitute into $AB + BC = AC$ and collect:
\[ (2x+5) + (x+3) = 26 \;\Longrightarrow\; 3x + 8 = 26 \;\Longrightarrow\; 3x = 18 \]
$\ans{x = 6}$
\par
\textbf{(c)} $AB = 2(6) + 5 = 12 + 5$.
\par
$\ans{AB = 17\ \text{cm}}$
\par
\emph{Check:} $BC = 6 + 3 = 9$ cm, and $17 + 9 = 26$ cm $\checkmark$
\par
\emph{Manual review:} part (a) is a named justification.}

\problem{\textbf{Given $5(x-2) = 3x + 8$; prove $x = 9$.}
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $5(x - 2) = 3x + 8$ & 1.\ Given \\ \hline
2.\ $5x - 10 = 3x + 8$ & 2.\ Distributive Property \\ \hline
3.\ $2x - 10 = 8$ & 3.\ Subtraction Property of Equality \\ \hline
4.\ $2x = 18$ & 4.\ Addition Property of Equality \\ \hline
5.\ $x = 9$ & 5.\ Division Property of Equality \\ \hline
\end{tabular}}
\par\smallskip
$\ans{x = 9}$
\par
\emph{Grading note:} line 3 must remove $3x$ from \emph{both} sides. A student
who writes $2x - 10 = 8$ with the reason ``combined like terms'' has the right
statement and the wrong reason --- the terms are on opposite sides of the equals
sign, so nothing is being combined; a quantity is being subtracted from both
sides. \emph{Manual review: read the Reasons column.}}

\problem{\textbf{$M$ is the midpoint of $\overline{AB}$: $AM = 4x-1$,
$MB = 2x+7$ cm.}
\par
\textbf{(a)} The proof runs from the definition of midpoint to a numerical
equation, then solves it:
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $M$ is the midpoint of $\overline{AB}$ & 1.\ Given \\ \hline
2.\ $\overline{AM} \cong \overline{MB}$ & 2.\ Definition of midpoint \\ \hline
3.\ $AM = MB$ & 3.\ Definition of congruent segments \\ \hline
4.\ $4x - 1 = 2x + 7$ & 4.\ Substitution \\ \hline
5.\ $2x - 1 = 7$ & 5.\ Subtraction Property of Equality \\ \hline
6.\ $2x = 8$ & 6.\ Addition Property of Equality \\ \hline
7.\ $x = 4$ & 7.\ Division Property of Equality \\ \hline
\end{tabular}}
\par\smallskip
\textbf{(b)} $\ans{x = 4}$, so $AM = 4(4) - 1 = 15$.
\par
$\ans{AM = 15\ \text{cm}}$
\par
\emph{Check:} $MB = 2(4) + 7 = 15$ cm, equal as the midpoint requires
$\checkmark$
\par
\emph{Manual review:} lines 2 and 3 are the two separate facts students most
often merge --- midpoint gives congruence, and congruence gives equal
\emph{measures}.}

\problem{\textbf{$A$, $B$, $C$, $D$ collinear in order: $AB = 3x$,
$BC = x+2$, $CD = 2x-1$, $AD = 37$ cm.}
\par
\textbf{(a)} The \textbf{Segment Addition Postulate}, applied twice, gives
$AB + BC + CD = AD$:
\[ 3x + (x+2) + (2x-1) = 37 \;\Longrightarrow\; 6x + 1 = 37
   \;\Longrightarrow\; 6x = 36 \]
$\ans{x = 6}$
\par
\textbf{(b)} $BC = 6 + 2$.
\par
$\ans{BC = 8\ \text{cm}}$
\par
\emph{Check:} $AB = 18$, $BC = 8$, $CD = 11$ cm, and $18 + 8 + 11 = 37$ cm
$\checkmark$}

\problem{\textbf{$\overline{AB} \cong \overline{CD}$: $AB = 3x+4$,
$CD = 5x-12$ cm.}
\par
\textbf{(a)}
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $\overline{AB} \cong \overline{CD}$ & 1.\ Given \\ \hline
2.\ $3x + 4 = 5x - 12$ & 2.\ Definition of congruent segments \\ \hline
3.\ $4 = 2x - 12$ & 3.\ Subtraction Property of Equality \\ \hline
4.\ $16 = 2x$ & 4.\ Addition Property of Equality \\ \hline
5.\ $x = 8$ & 5.\ Division Property of Equality \\ \hline
\end{tabular}}
\par\smallskip
$\ans{x = 8}$
\par
\textbf{(b)} $AB = 3(8) + 4 = 24 + 4$.
\par
$\ans{AB = 28\ \text{cm}}$
\par
\emph{Check:} $CD = 5(8) - 12 = 28$ cm, so the two segments really are
congruent $\checkmark$ \emph{Manual review: read the Reasons column.}}

\problem{\textbf{Find Jordan's mistake in $2(x+3) = 4x - 10$.}
\par
\textbf{Line 3 is the first wrong line.} Jordan subtracted $2x$ from the left
side only. The Subtraction Property of Equality permits subtracting the same
quantity from \emph{both} sides; taking it from one side changes the equation
into a different one, so every line after it is about the wrong equation.
\par
The corrected proof:
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $2(x + 3) = 4x - 10$ & 1.\ Given \\ \hline
2.\ $2x + 6 = 4x - 10$ & 2.\ Distributive Property \\ \hline
3.\ $6 = 2x - 10$ & 3.\ Subtraction Property of Equality \\ \hline
4.\ $16 = 2x$ & 4.\ Addition Property of Equality \\ \hline
5.\ $x = 8$ & 5.\ Division Property of Equality \\ \hline
\end{tabular}}
\par\smallskip
$\ans{x = 8}$
\par
\emph{Check:} $2(8+3) = 22$ and $4(8) - 10 = 22$ $\checkmark$. Jordan's $x = 4$
fails this check: $2(7) = 14$ but $4(4) - 10 = 6$.
\emph{Manual review: the student must name the line and the property.}}

\problem{\textbf{Synthesis: $\overline{AB} \cong \overline{CD}$ forces
$\overline{AC} \cong \overline{BD}$.}
\par
\textbf{(a)} The two long segments share $\overline{BC}$, so add it to each of
the congruent short segments:
\par
{\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{0.40\textwidth}|p{0.45\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{Statements}} & \multicolumn{1}{c|}{\textbf{Reasons}} \\ \hline
1.\ $A$, $B$, $C$, $D$ collinear in order; $\overline{AB} \cong \overline{CD}$ & 1.\ Given \\ \hline
2.\ $AB = CD$ & 2.\ Definition of congruent segments \\ \hline
3.\ $AB + BC = CD + BC$ & 3.\ Addition Property of Equality \\ \hline
4.\ $AB + BC = AC$; \ $BC + CD = BD$ & 4.\ Segment Addition Postulate \\ \hline
5.\ $AC = BD$ & 5.\ Substitution \\ \hline
6.\ $\overline{AC} \cong \overline{BD}$ & 6.\ Definition of congruent segments \\ \hline
\end{tabular}}
\par\smallskip
\textbf{(b)} $2x - 3 = x + 5$, so subtracting $x$ from both sides and adding
$3$ to both sides gives
\par
$\ans{x = 8}$
\par
Then $AB = 2(8) - 3 = 16 - 3$, which is
$\ans{AB = 13\ \text{cm}}$, and $CD = 8 + 5 = 13$ cm as the congruence
requires.
\par
Finally $AC = AB + BC = 13 + 6$.
\par
$\ans{AC = 19\ \text{cm}}$
\par
\emph{Check:} $BD = BC + CD = 6 + 13 = 19$ cm, equal to $AC$ exactly as part
(a) proved $\checkmark$
\emph{Manual review: read the student's two-column proof in part (a).}}
