\sstitleblock{Statistical Questions and Describing Variability}

% =================== SKILL 1 ===================
\skillheading{Telling a Statistical Question Apart}

\begin{formulabox}
\textbf{Rule:} a question is \textbf{statistical} when answering it means
collecting data whose values \emph{vary}. One named thing at one named moment
has one answer, so it is not statistical.\par
\textbf{Test:} could two honest answers differ? If yes, it is statistical, and
the range of the collected values is one way to show how much they vary.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad ``How old is the oldest tree in the park?'' or ``How old
are the trees in the park?'' --- which is statistical? Five trees were measured:
8, 11, 9, 14, 12 years.
\step{The first question names one tree and has a single answer; the second is
answered by collecting an age from every tree, and those ages differ, so the
second is the statistical one.}
\step{The variability is real and can be measured: the ages run from 8 to 14, a
range of $14 - 8 = $ $\ans{6}$ years.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Which is statistical --- ``how many pages did Sam read
today?'' or ``how many pages do the students read each day?'' --- and what is
the range of 15, 22, 18, 27, 20 pages?\\[2pt]
\rotatebox{180}{\footnotesize check: second question; range $\ans{12}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Measuring Spread: Range and Interquartile Range}

\begin{formulabox}
\textbf{Rule:} \textbf{range} $=$ largest $-$ smallest. It uses only the two end
values, so one unusual value changes it completely.\par
\textbf{Interquartile range} $=$ upper quartile $-$ lower quartile. Put the
values in order, split them into a lower half and an upper half, and take the
median of each half. It measures the width of the middle half of the data, so
end values do not move it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Twelve values in order: 1, 3, 4, 6, 8, 9, 11, 13, 14,
16, 18, 20. Find the quartiles and the interquartile range.
\step{Twelve values split evenly into a lower six and an upper six, so take the
median of each half.}
\step{Lower half 1, 3, 4, 6, 8, 9 gives $\ans{5}$; upper half 11, 13, 14, 16,
18, 20 gives $\ans{15}$; the interquartile range is $15 - 5 = \ans{10}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the interquartile range of 2, 4, 6, 8, 12, 14, 18,
20.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{11}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Describing a Data Set: Centre and Spread Together}

\begin{formulabox}
\textbf{Rule:} a description needs \textbf{a typical value} and \textbf{how
spread out} the values are. Two sets can share a mean and look nothing alike.\par
\textbf{Choosing the centre:} the mean is pulled towards an unusually large or
small value; the median is not. When one value sits far from the rest, the
median describes a typical member better.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Nine deliveries took 2, 3, 3, 4, 4, 5, 6, 9, 36 minutes.
Which measure describes a typical delivery?
\step{One value, 36 minutes, sits far above the rest, so compute both the mean
and the median and see how far apart they land.}
\step{The times total 72, so the mean is $72 \div 9 = \ans{8}$ minutes, while
the median (the fifth value) is $\ans{4}$ minutes --- the median, because seven
of the nine deliveries took under 8 minutes.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the median of 1, 2, 2, 3, 3, 4, 5, 8, 26, and say
whether it or the mean better describes a typical value.\\[2pt]
\rotatebox{180}{\footnotesize check: median $\ans{3}$, and it describes it
better}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an equal mean does not mean equal data. Two teams
averaging 22 points can have ranges of 4 and 24 --- the spread is the part that
tells them apart.
\end{watchoutbox}
