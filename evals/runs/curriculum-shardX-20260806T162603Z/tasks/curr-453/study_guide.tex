\sstitleblock{Making a Piecewise Model Continuous}

\begin{formulabox}
\textbf{Continuity at a junction $x = c$ means three things agree:} the
left-hand limit, the right-hand limit, and the value $f(c)$. Every problem on
this topic is the same move --- write the two one-sided limits at $c$, set them
equal, and solve for the unknown constant.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Matching the Two Sides at a Junction}

\begin{formulabox}
\textbf{Method:} substitute the junction input into \emph{each} piece
separately. The piece for $x < c$ gives the left-hand limit, the piece for
$x \ge c$ gives the right-hand limit. Set the two expressions equal and solve.
The unknown appears in exactly one of them, so the equation is linear.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = 2x + a$ for $x < 4$ and $f(x) = 3x - 5$ for
$x \ge 4$. Find $a$.
\step{Continuity at $4$ means the two pieces must hit the same height there, so evaluate each piece at $4$ and set them equal.}
\step{Left: $2(4) + a = 8 + a$. Right: $3(4) - 5 = 7$. So $8 + a = 7$, giving $a = -1$.}

$\Rightarrow\quad$ $\ans{a = -1}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 4x + a$ for $x < 3$ and $f(x) = 2x + 7$ for
$x \ge 3$. Find $a$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{a = 1}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Filling a Hole With the Limit Value}

\begin{formulabox}
\textbf{Method:} when a piece is a quotient that gives $\frac{0}{0}$ at the
junction, the discontinuity is a hole. The only value that removes it is the
limit, so compute the limit --- factor and cancel, or multiply by the conjugate
--- and set the constant equal to it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $p(x) = \dfrac{x^2-4}{x-2}$ for $x \neq 2$ and
$p(2) = c$. Find $c$.
\step{The formula is undefined at $2$, so $c$ has to be whatever the limit approaches --- compute the limit first, then read off $c$.}
\step{$\dfrac{x^2-4}{x-2} = \dfrac{(x-2)(x+2)}{x-2} = x+2$ for $x \neq 2$, and at $x = 2$ that is $4$.}

$\Rightarrow\quad$ $\ans{c = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $q(x) = \dfrac{x^2-36}{x-6}$ for $x \neq 6$ and
$q(6) = c$. Find $c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c = 12}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Two Junctions, Two Unknowns}

\begin{formulabox}
\textbf{Method:} a three-piece model has two junctions, so it gives two
equations in the two unknowns. Write both, then eliminate: solve one for $b$,
substitute into the other, and back-substitute.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^2$ for $x < 1$, \ $ax + b$ for
$1 \le x < 3$, \ $2x + 5$ for $x \ge 3$. Find $a$ and $b$.
\step{Each junction gives one equation, and two unknowns need two equations --- so write both junctions before solving either.}
\step{At $1$: $a + b = 1$, so $b = 1 - a$. At $3$: $3a + b = 11$, so $3a + (1-a) = 11$, giving $2a = 10$ and then $b = 1 - 5$.}

$\Rightarrow\quad$ $\ans{a = 5}$ \quad and \quad $\ans{b = -4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = 3x$ for $x < 2$, \ $ax + b$ for $2 \le x < 4$, \
$x^2 + 2$ for $x \ge 4$. Find $a$ and $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{a = 6}$, $\ans{b = -6}$}
\end{tryitbox}
