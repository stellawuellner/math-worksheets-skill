\sstitleblock{Rules, Tables, Graphs, and Sensible Domains}

\skillheading{Skill 1 --- From a rule to a table of values}

\begin{formulabox}
\textbf{Rule:} the input goes in, the output comes out. To fill a table, substitute each $x$ into the rule --- in parentheses, so signs and squares survive --- and record the result under it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Build a table for $f(x) = 4x - 1$ at $x = -1$ and $x = 2$.
\step{Each table entry is an evaluation: the top row holds inputs, the bottom row holds the outputs the rule produces.}
\step{$4(-1) - 1 = -4 - 1$ and $4(2) - 1 = 8 - 1$.}

$f(-1) = \ans{-5}$ \quad and \quad $f(2) = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = 5x - 2$, find $g(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $g(3) = \ans{13}$}
\end{tryitbox}

\skillheading{Skill 2 --- Solving when the output is given}

\begin{formulabox}
\textbf{Two different questions.} ``Find $f(5)$'' hands you the \emph{input}: substitute. ``Solve $f(x) = 5$'' hands you the \emph{output}: set the rule equal to $5$ and undo it one operation at a time.\par
On a table, the first reads across from a value in the top row; the second hunts backwards from a value in the bottom row.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 7x + 4$, solve $f(x) = 39$.
\step{The output is given and the input is unknown, so this is an equation to solve, not a substitution.}
\step{$7x + 4 = 39$; subtract $4$ to get $7x = 35$, then divide by $7$.}

$x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 9 - 2x$, solve $f(x) = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- From a rule to a graph}

\begin{formulabox}
\textbf{Rule:} every row of the table is one point $(x, f(x))$ on the graph. Plot two or three of them and join them --- for a linear rule the line through them is the whole graph.\par
\textbf{Reading it back:} the height where the graph meets the vertical axis is the output at input $0$; the place it crosses the horizontal axis is the input whose output is $0$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Graph $f(x) = 2x - 6$ and find where it crosses the horizontal axis.
\step{Two points fix a line, so evaluate at a convenient input first, then find the crossing by solving for the input that gives output zero.}
\step{$f(0) = 2(0) - 6$, which places the point $(0, -6)$ on the graph.}
\step{Crossing: $2x - 6 = 0$, so $2x = 6$.}

$f(0) = \ans{-6}$ \quad and the graph crosses at \quad $x = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $f(x) = 3x - 12$, find $f(0)$ and where the graph crosses the horizontal axis.\\[2pt]
\rotatebox{180}{\footnotesize check: $f(0) = \ans{-12}$, crossing at $x = \ans{4}$}
\end{tryitbox}

\skillheading{Skill 4 --- Which inputs the situation allows}

\begin{formulabox}
\textbf{Rule:} an algebraic rule accepts every number, but a model does not. Ask what the input measures: counts must be whole numbers, lengths and times cannot be negative, and a physical limit (seats, capacity, the moment a tank empties) caps the top end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A pool holds $300$ litres and drains $25$ litres a minute, so $W(t) = 300 - 25t$ litres after $t$ minutes. Find $W(4)$, and the moment the pool is empty.
\step{Time cannot run backwards and the pool cannot hold less than nothing, so the model only lives from the start until the pool empties --- find both ends.}
\step{$W(4) = 300 - 25(4) = 300 - 100$.}
\step{Empty means output zero: $300 - 25t = 0$, so $25t = 300$. The sensible inputs are the times from $0$ to that moment.}

$W(4) = \ans{200}$ litres, empty at \quad $t = \ans{12}$ minutes
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $V(t) = 150 - 15t$ litres after $t$ minutes. Find $V(4)$ and the time the tank is empty.\\[2pt]
\rotatebox{180}{\footnotesize check: $V(4) = \ans{90}$, empty at $t = \ans{10}$}
\end{tryitbox}
