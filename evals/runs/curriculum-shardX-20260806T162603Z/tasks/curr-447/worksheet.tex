\wstitleblock{Using Normal-Model z-Scores and Percentiles}{Precalculus / Statistics}{}

\begin{center}
\small Standardizing, unstandardizing, the empirical rule, and comparing across distributions
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Standardize with $z = \dfrac{x - \mu}{\sigma}$; go back with $x = \mu + z\sigma$.
  \item Every distribution on this sheet is assumed to be approximately normal.
  \item Percentages come from the empirical rule: about $68\%$ of values lie within $1$ standard deviation of the mean, about $95\%$ within $2$, and about $99.7\%$ within $3$.
  \item Give $z$-scores to two decimal places where they are not exact.
\end{itemize}

\vspace{0.15cm}

\problem[4cm]{An exam has mean $\mu = 500$ and standard deviation $\sigma = 100$. Find the
$z$-score of a score of $620$.}

\problem[4cm]{On the same exam, find the $z$-score of a score of $455$.}

\problem[4cm]{On the same exam, what percentage of scores fall below $600$? (Use the empirical
rule; give a whole percentage.)}

\problem[4cm]{On the same exam, a student's $z$-score is $1.5$. Find their exam score.}

\problem[0.4cm]{A student scores $88$ on a history test with $\mu = 80$ and $\sigma = 5$, and
$76$ on a biology test with $\mu = 70$ and $\sigma = 4$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the $z$-score on the history test. \quad \ansblank
  \item[(b)] Find the $z$-score on the biology test. \quad \ansblank
  \item[(c)] Explain which performance was stronger relative to its class, and why the raw scores alone cannot settle it.
\end{itemize}
\vspace*{3cm}}

\problem[0.4cm]{Adult heights in a population are approximately normal with $\mu = 170$ cm and
$\sigma = 8$ cm. About $95\%$ of heights lie between two values.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt, topsep=4pt]
  \item[(a)] Find the lower of those two heights. \answerline{cm}
  \item[(b)] Find the upper of those two heights. \answerline{cm}
\end{itemize}
\vspace*{2.8cm}}

\problem[4cm]{For the same heights ($\mu = 170$ cm, $\sigma = 8$ cm), find the $z$-score of a
height of $182$ cm.}

\problem[4.5cm]{For the same heights, what percentage of the population is shorter than
$162$ cm? (Use the empirical rule; give a whole percentage.)}

\problem[0.4cm]{A battery's life is approximately normal with $\mu = 40$ hours and
$\sigma = 3$ hours. Find the $97.5$th percentile of battery life --- the value with
$97.5\%$ of batteries below it.
\par\vspace*{3.2cm}\answerline{hours}}

\problem[4.5cm]{For the same batteries, one is chosen at random. Using the empirical rule,
$25$ batteries in every $1000$ last longer than the $97.5$th percentile. Give that probability
as a fraction in lowest terms.}

\problem[0.4cm]{On a test with $\mu = 72$ and $\sigma = 6$, a student's $z$-score is $-1.5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the student's raw score. \quad \ansblank
  \item[(b)] Explain what the negative sign tells you, and estimate the student's percentile from the empirical rule.
\end{itemize}
\vspace*{3.2cm}}

\problem[0.4cm]{\textbf{Synthesis.} Line~1 produces batteries with $\mu = 40$ hours and
$\sigma = 3$ hours; Line~2 produces them with $\mu = 45$ hours and $\sigma = 5$ hours. One
battery from Line~1 lasted $46$ hours; one from Line~2 lasted $52$ hours.
\begin{itemize}[leftmargin=1.2cm, itemsep=8pt, topsep=4pt]
  \item[(a)] Find the $z$-score of the Line~1 battery. \quad \ansblank
  \item[(b)] Find the $z$-score of the Line~2 battery. \quad \ansblank
  \item[(c)] Explain which result is more unusual for its own production line, even though one battery lasted longer than the other.
\end{itemize}
