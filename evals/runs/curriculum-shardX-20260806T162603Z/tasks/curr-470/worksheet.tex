\problem[5cm]{A circular oil slick spreads so that its radius grows at
$\dfrac{dr}{dt} = 0.5$ m/s. Find $\dfrac{dA}{dt}$ at the instant when
$r = 20$ m, to two decimal places (square metres per second).}

\problem{A rectangular garden is fenced on all four sides with $48$ m of fence.
Let $x$ be one side, so the area is $A(x) = x(24 - x)$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $A'(x)$.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Find the value of $x$ that maximizes the area.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{Let $f(x) = x^3 - 6x^2 + 9x + 2$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $f'(x)$.
    \par\vspace*{2.0cm}\ansline
  \item[(b)] Find the $x$-values of the critical points.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Find the $x$-value of the inflection point.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{An open-top box with a square base of edge $x$ centimetres and height
$h$ must hold $32000$ cm$^3$. Its surface area is $S = x^2 + 4xh$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Use the volume constraint to write $S$ as a function of $x$ alone.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] Find the base edge $x$ that minimizes the surface area.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{Let $f(x) = x^4 - 6x^2 + 5$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $f''(x)$.
    \par\vspace*{2.0cm}\ansline
  \item[(b)] Find the $x$-values of the inflection points.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Justify whether $f$ has a local maximum, a local minimum, or
    neither at $x = 0$.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem{Water runs into an inverted cone at $2$ m$^3$/min. The cone's radius
is always half its water depth $h$, so $r = \dfrac{h}{2}$ and
$V = \dfrac{1}{3}\pi r^2 h$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Write $V$ in terms of $h$ alone.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Find $\dfrac{dh}{dt}$ when $h = 4$ m, to two decimal places.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}

\problem{A differentiable function $g$ has $g'(x) = (x - 1)^2(x - 4)$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the $x$-values of the critical points of $g$.
    \par\vspace*{2.2cm}\ansline
  \item[(b)] Justify, from the sign of $g'$, what happens to $g$ at each
    critical point.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}

\problem[5.5cm]{Two cars leave the same intersection at the same moment: one
travels north at $60$ km/h, the other east at $80$ km/h. After $2$ hours they
are $120$ km and $160$ km from the intersection, so the distance between them
is $200$ km. How fast is that distance changing at this moment, in kilometres
per hour?}

\problem{A rectangle sits with its base on the $x$-axis and its two upper
corners on the parabola $y = 12 - x^2$. Its area is $A(x) = 24x - 2x^3$ for
$x > 0$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $A'(x)$.
    \par\vspace*{2.0cm}\ansline
  \item[(b)] Find every solution of $A'(x) = 0$.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Find the maximum area.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A spherical balloon is inflated at $\dfrac{dV}{dt} = 100$ cm$^3$/s,
where $V = \frac{4}{3}\pi r^3$ and the surface area is $S = 4\pi r^2$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $\dfrac{dr}{dt}$ when $r = 10$ cm, to two decimal places.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] Find $\dfrac{dS}{dt}$ at that same instant, to two decimal places.
    \par\vspace*{2.6cm}\ansline
  \item[(c)] Justify whether the surface area grows faster at $r = 10$ cm or at
    $r = 5$ cm.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}
