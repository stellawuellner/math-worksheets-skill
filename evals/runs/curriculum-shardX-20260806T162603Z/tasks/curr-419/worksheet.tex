\wstitleblock{Find and Fix: Degree--Radian and Quadrant--Sign Errors}{Precalculus}{}

\vspace{0.15cm}
\noindent\textbf{Directions.} Every problem below shows work a student actually handed in. In each one:
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=3pt]
  \item first produce the \emph{correct} answer yourself, in the form the question asks for;
  \item then say what went wrong, in a sentence or two. Name the step, not just the wrong number.
  \item Exact values (fractions and radicals) are expected unless a decimal is requested.
\end{itemize}
\vspace{0.15cm}

\problem{A student is asked to convert $\dfrac{5\pi}{6}$ radians to degrees, and writes
\[ \frac{5\pi}{6}\cdot\frac{\pi}{180} = 0.0457. \]
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct measure. \answerline{degrees}
  \vspace*{2.2cm}
  \item[(b)] Name the error in one sentence. \ansline
  \vspace*{2.2cm}
\end{itemize}}

\problem{A student evaluates $\cos 210^\circ$ this way: ``the reference angle is $30^\circ$, and $\cos 30^\circ = \dfrac{\sqrt{3}}{2}$, so $\cos 210^\circ = \dfrac{\sqrt{3}}{2}$.''
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct exact value. \ansline
  \vspace*{2.2cm}
  \item[(b)] Explain which step is wrong and why. \ansline
  \vspace*{2.2cm}
\end{itemize}}

\problem{A student solves $2\sin\theta - 1 = 0$ on the interval $0^\circ \le \theta < 360^\circ$ and reports $\theta = 0.5236$ and $\theta = 2.6180$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct solutions. \answerline{degrees}
  \vspace*{2.4cm}
  \item[(b)] Name the error in one sentence. \ansline
  \vspace*{2.4cm}
\end{itemize}}

\problem{A student solves $\tan\theta = -1$ on the interval $0^\circ \le \theta < 360^\circ$. They press $\tan^{-1}(-1)$, read $-45$, and report that $\theta = -45^\circ$ is the only solution.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give every solution in the interval. \answerline{degrees}
  \vspace*{2.6cm}
  \item[(b)] Two separate things went wrong here. Name both. \ansline
  \vspace*{2.6cm}
\end{itemize}}

\problem{A student evaluates $\sin\dfrac{7\pi}{6}$ and writes $-\dfrac{\sqrt{3}}{2}$, reasoning that ``the reference angle is $\dfrac{\pi}{6}$, and the third quadrant makes sine negative.''
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct exact value. \ansline
  \vspace*{2.2cm}
  \item[(b)] The sign reasoning is correct. Explain what is not. \ansline
  \vspace*{2.2cm}
\end{itemize}}

\problem{A student needs $\sin(2)$, where the $2$ is an angle in radians. Their calculator returns $0.0349$, and they write that down.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct value of $\sin(2)$ to four decimal places. \ansline
  \vspace*{2.4cm}
  \item[(b)] Explain what the calculator actually computed, and why the two results are so far apart. \ansline
  \vspace*{2.4cm}
\end{itemize}}

\problem{A student is told that $\sin\theta = -\dfrac{3}{5}$ and that $\cos\theta > 0$. They write: ``sine is negative, so $\theta$ is in the third quadrant, and there cosine is negative too, so $\cos\theta = -\dfrac{4}{5}$.''
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the correct exact value of $\cos\theta$. \ansline
  \vspace*{2.6cm}
  \item[(b)] State the correct quadrant, and explain what the student overlooked. \ansline
  \vspace*{2.6cm}
\end{itemize}}

\problem{A student solves $\cos\theta = -\dfrac{1}{2}$ on the interval $0 \le \theta < 2\pi$ and reports $\theta = 120$ and $\theta = 240$.
\begin{itemize}[leftmargin=1.1cm, topsep=6pt, itemsep=0pt]
  \item[(a)] Give the exact solutions in radians. \ansline
  \vspace*{2.8cm}
  \item[(b)] Name both errors, and state the check that would have caught them. \ansline
  \vspace*{2.8cm}
\end{itemize}}
