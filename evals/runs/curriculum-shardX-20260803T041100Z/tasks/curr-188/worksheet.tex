\wstitleblock{Graphing Proportional Relationships}{Grades 6--7 \quad Ratios and Proportional Relationships}{}

\begin{center}
\textit{Plotting a proportional relationship from a table, finding the unit
rate from a graph, and writing the equation of a line through the origin}
\end{center}

\noindent\textbf{Before you start.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A relationship is \textbf{proportional} when its graph is a straight line
\emph{through the origin} $(0,0)$.
\item The \textbf{unit rate} (constant of proportionality) $k$ is the rise over
the run measured from the origin: $k = \dfrac{y}{x}$ for any point on the line
except $(0,0)$.
\item The equation is always $y = kx$. State the units of $k$ whenever the
problem gives them.
\end{itemize}

\problem[2.8cm]{Dev is paid the same amount for every hour he walks dogs.

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Hours worked} & $0$ & $3$ & $5$ \\
\hline
\textbf{Dollars earned} & $0$ & $24$ & $40$ \\
\hline
\end{tabular}
\hspace{0.4cm}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=6, ymin=0, ymax=48,
  xtick={0,1,2,3,4,5,6}, ytick={0,8,16,24,32,40,48},

  tick label style={font=\scriptsize},
  xlabel={\scriptsize hours}, ylabel={\scriptsize dollars},
  width=6.4cm, height=5.4cm, enlargelimits=false]
\end{axis}
\end{tikzpicture}
\end{center}
Plot the three points and draw the straight line through them. Then find the
unit rate, in dollars per hour.}

\problem[2.8cm]{Not every straight line is proportional. This table and its
graph belong together.

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{$x$} & $1$ & $2$ & $4$ \\
\hline
\textbf{$y$} & $5$ & $8$ & $14$ \\
\hline
\end{tabular}
\hspace{0.3cm}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=5, ymin=0, ymax=16,
  xtick={0,1,2,3,4,5}, ytick={0,4,8,12,16},
  grid=both, grid style={line width=0.15pt, draw=gray!35},
