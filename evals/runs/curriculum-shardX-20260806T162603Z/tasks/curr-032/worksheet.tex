\wstitleblock{Teen Numerals and Number Words}{Kindergarten--Grade 1}{}

\begin{center}
\small reading teen number words, writing teen numerals, one ten and some ones, and comparing teen numbers
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item A teen numeral always starts with the digit for one ten. The second digit is the ones.
  \item Word bank: \emph{eleven, twelve, thirteen, fourteen, fifteen, sixteen, seventeen, eighteen, nineteen}.
  \item Say the number out loud before you write it. Best spelling counts --- sound it out.
\end{itemize}
