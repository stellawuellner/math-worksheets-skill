\sstitleblock{Where Two Lines Meet}

\skillheading{Skill 1 --- Testing a pair in both equations}

\begin{formulabox}
\textbf{Rule:} a pair solves a system only if it makes \emph{every} equation
true. Satisfying one equation and failing the other means the point is on one
line but not the other, so it is not the meeting point.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x + y = 7$ and $x - y = 1$.
\step{Both equations must hold at once, so combine them rather than solving
either one on its own --- adding them cancels $y$.}
\step{$2x = 8$, so $x = 4$; then $4 + y = 7$ gives $y = 3$. Check the second:
$4 - 3 = 1$.}

Both true, so this is the solution. \quad $\ans{(4,\, 3)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x + y = 9$ and $x - y = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(6,\, 3)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Reading the intersection off a graph}

\begin{formulabox}
\textbf{Rule:} the crossing point lies on both lines, so its coordinates
satisfy both equations. Read the point, then substitute to confirm it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the solution of $y = x + 2$ and $y = 6 - x$ from
the graph.
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=6, ymin=0, ymax=6,
  xtick={0,1,2,3,4,5,6}, ytick={0,1,2,3,4,5,6},
  grid=both, grid style={line width=0.25pt, draw=gray!55},
  major grid style={line width=0.4pt, draw=gray!70},
  tick label style={font=\tiny}, xlabel={$x$}, ylabel={$y$},
  width=5.6cm, height=4.4cm, enlargelimits=false]
  \addplot[thick, blue, domain=0:4, samples=2] {x + 2};
  \addplot[thick, red, domain=0:6, samples=2] {6 - x};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The lines cross at a grid corner, so read the coordinates there and then
confirm them in both equations.}
\step{The corner is 2 across and 4 up: $4 = 2 + 2$ and $4 = 6 - 2$, both
true.}

On both lines at once. \quad $\ans{(2,\, 4)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $y = x + 3$ and $y = 9 - x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\, 6)}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the shared row in tables}

\begin{formulabox}
\textbf{Rule:} when two tables share an input with the same output, that row
is the solution. If no row is shared, set the two rules equal and solve.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plan A is in the table; plan B follows $y = h + 12$.
When do they cost the same?
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$h$ & 1 & 2 & 3 \\ \hline
\textbf{Plan A} & 9 & 12 & 15 \\ \hline
\end{tabular}
\end{center}
\step{Plan A's table rises 3 for each hour and starts from 6 at zero hours, so
its rule is $y = 3h + 6$ and the two rules can be set equal.}
\step{$3h + 6 = h + 12$ gives $2h = 6$, so $h = 3$; the table's cost there is
$\ans{15}$.}

The shared row is the solution. \quad $\ans{(3,\, 15)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $h = 1, 2, 3$ with costs $8, 13, 18$
(rule $y = 5h + 3$); the other plan follows $y = h + 15$. Find the shared
pair.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\, 18)}$}
\end{tryitbox}

\skillheading{Skill 4 --- What the intersection means in a story}

\begin{formulabox}
\textbf{Rule:} the intersection is the moment the two options are equal. On
one side of it the first option is cheaper; on the other side the second one
is.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Club A charges \$10 to join plus \$4 a visit
($y = 4v + 10$); club B charges \$6 a visit ($y = 6v$). When are they equal?
\step{Equal cost means the two expressions take the same value, which is the
intersection of their graphs --- set them equal.}
\step{$4v + 10 = 6v$ gives $2v = 10$, so $v = 5$, and the cost there is
$6(5) = 30$ dollars.}

Below 5 visits club B is cheaper; above it, club A. \quad $\ans{(5,\, 30)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One club charges $y = 3v + 12$ and another charges
$y = 5v$. Find the (visits, cost) pair where they are equal.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(6,\, 30)}$}
\end{tryitbox}
