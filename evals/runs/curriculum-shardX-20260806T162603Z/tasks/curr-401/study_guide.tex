\sstitleblock{Analyzing Transformations with Multiple Parameters}

\skillheading{Skill 1 --- Reading the four parameters off a rule}

\begin{formulabox}
\textbf{Master form:} \quad $g(x) = a\,f\bigl(b(x - h)\bigr) + k$
\begin{itemize}[leftmargin=0.9cm, itemsep=0pt, topsep=2pt]
  \item $a$: vertical stretch (reflect in the $x$-axis if $a < 0$) \quad $k$: shift up
  \item $b$: horizontal stretch by $1/b$ (reflect in the $y$-axis if $b < 0$) \quad $h$: shift right
  \item Inside changes ($b$, $h$) act \emph{opposite} to how they look; outside changes ($a$, $k$) act as they look.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2}$ and $g(x) = f(x - 5) + 2$. Find $g(7)$.
\step{Only $h$ and $k$ are present, so the graph shifts right $5$ and up $2$ --- feed the shifted input to $f$ first.}
\step{$g(7) = f(7 - 5) + 2 = f(2) + 2 = 4 + 2$.}

$\Rightarrow\quad \ans{g(7) = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2}$ and $g(x) = f(x + 3) - 1$. Find $g(1)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g(1) = 15}$}
\end{tryitbox}

\skillheading{Skill 2 --- Evaluating a transformed function}

\begin{formulabox}
\textbf{Order of operations for $g(x) = a\,f(bx + c) + k$:}\par
transform the input, apply $f$, then transform the output. Nothing outside $f$ can ever change which value of $f$ you look up.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2} + 1$ and $g(x) = f(3x) - 4$. Find $g(2)$.
\step{The $3x$ is inside, so it changes the input before $f$ acts; the $-4$ is outside and only moves the answer afterwards.}
\step{$g(2) = f(6) - 4 = (36 + 1) - 4$.}

$\Rightarrow\quad \ans{g(2) = 33}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2} + 1$ and $h(x) = f(2x) + 5$. Find $h(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h(3) = 42}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $f(2x)$ is \emph{not} $2f(x)$. One doubles the input, the other doubles the output --- on a parabola they give different numbers at every point except the vertex.
\end{watchoutbox}

\skillheading{Skill 3 --- Expanding the transformed rule}

\begin{formulabox}
\textbf{To expand $g(x) = a\,f(x - h) + k$:} substitute $x - h$ for \emph{every} $x$ in the rule for $f$, expand, then multiply by $a$ and add $k$.\par
This is how you check a transformation answer: the expanded form must agree with the original at every test value.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $f(x) = x^{2}$ and $g(x) = 3f(x - 2) - 4$. Expand.
\step{Substitution comes first --- expanding before substituting is where the middle term goes missing.}
\step{$3(x - 2)^{2} - 4 = 3\left(x^{2} - 4x + 4\right) - 4 = 3x^{2} - 12x + 12 - 4$.}

$\Rightarrow\quad \ans{g(x) = 3x^{2} - 12x + 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $f(x) = x^{2}$ and $g(x) = 2f(x + 1) + 5$. Expand.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g(x) = 2x^{2} + 4x + 7}$}
\end{tryitbox}
