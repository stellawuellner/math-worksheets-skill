\wstitleblock{Choosing a Model from the Evidence}{Algebra 1}{}

\begin{center}\small\textit{linear models, quadratic models, and piecewise models}\end{center}

\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Before you compute, decide which family the situation belongs to. Constant \emph{first} differences point to a linear rule; constant \emph{second} differences point to a quadratic rule; a rule that changes at a break point is piecewise.
  \item Show the differences, the substitution, or the equation you solved. Write the final answer on the answer line.
\end{itemize}

\vspace{0.2cm}\noindent\rule{\linewidth}{0.4pt}

\problem{A shop prints team shirts. The table shows the total cost of an order.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}\hline
Shirts ordered $n$ & 0 & 1 & 2 & 3 \\ \hline
Total cost (dollars) & 25 & 32 & 39 & 46 \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.1cm}
(a) Is the total cost linear, quadratic, or piecewise? Explain which pattern in the table tells you.
\par\vspace*{2.6cm}\ansline
\par\vspace{0.15cm}
(b) Find the total cost of an order of 10 shirts, in dollars.
\par\vspace*{2.9cm}\ansline}

\problem{A stunt ramp launches a cart. Its height above the ramp, in feet, is recorded each second.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}\hline
Time $t$ (seconds) & 0 & 1 & 2 & 3 \\ \hline
Height (feet) & 0 & 44 & 56 & 36 \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.1cm}
(a) Is the height linear, quadratic, or piecewise? Explain which pattern in the table tells you.
\par\vspace*{2.6cm}\ansline
\par\vspace{0.15cm}
(b) The height fits the rule $h(t) = -16t^2 + 60t$. Find the height at $t = 2.5$ seconds.
\par\vspace*{2.9cm}\answerline{ft}}

\problem{A garage charges $\$3$ for the first hour of parking and $\$2$ for every hour after that.
\par\vspace{0.1cm}
(a) Is the charge linear, quadratic, or piecewise? Explain what in the pricing tells you.
\par\vspace*{2.3cm}\ansline
\par\vspace{0.15cm}
(b) Find the charge for parking 4 hours, in dollars.
\par\vspace*{2.6cm}\ansline}

\problem{A phone plan costs $\$12$ per month plus $\$0.05$ for each minute of talk time. One month the bill was $\$27.50$. Write and solve an equation for $m$, the number of minutes used.
\par\vspace*{5.2cm}\answerline{min}}

\problem{A model rocket is fired straight up from the ground. Its height in feet after $t$ seconds is $h(t) = -16t^2 + 64t$. Solve $h(t) = 0$ to find the two times at which the rocket is at ground level.
\par\vspace*{5.4cm}\answerline{s}}

\problem{A tank fills at 5 litres per minute for the first 8 minutes, then at 2 litres per minute. Find the volume in the tank 15 minutes after filling starts.
\par\vspace*{5.2cm}\answerline{L}}

\problem{Gym A charges a $\$40$ joining fee plus $\$15$ per month. Gym B charges a $\$25$ joining fee plus $\$20$ per month. Find the number of months $m$ and the total cost $y$ at which the two gyms cost the same. Give both values.
\par\vspace*{5.6cm}\ansline}

\problem{A rectangular dog run is fenced with 40 feet of fencing. If the width is $w$ feet, the length is $20 - w$ feet, so the area is $A(w) = w(20 - w)$.
\par\vspace{0.1cm}
(a) Is the area linear, quadratic, or piecewise as a function of $w$? Justify your answer from the rule.
\par\vspace*{2.4cm}\ansline
\par\vspace{0.15cm}
(b) Find the area when the width is 6 feet.
\par\vspace*{2.7cm}\answerline{ft$^2$}}

\problem{A courier charges $\$6$ for a parcel of up to 5 pounds, then $\$1.20$ for each pound above 5. A parcel cost $\$12$ to send. Write and solve an equation for its weight $w$.
\par\vspace*{5.4cm}\answerline{lb}}

\problem{A student looks at the table below and says ``the rule must be linear, because $y$ goes up every time.''
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}\hline
$x$ & 1 & 2 & 3 & 4 \\ \hline
$y$ & 3 & 7 & 13 & 21 \\ \hline
\end{tabular}
\end{center}
\par\vspace{0.1cm}
(a) Explain why that reason is not enough, and name the family the table really belongs to.
\par\vspace*{2.5cm}\ansline
\par\vspace{0.15cm}
(b) The table fits $y = x^2 + x + 1$. Predict $y$ when $x = 5$.
\par\vspace*{2.7cm}\ansline}

\problem{An electricity supplier charges $\$0.11$ per kilowatt-hour for the first 400 kilowatt-hours used in a month, and $\$0.16$ per kilowatt-hour after that. Find the bill, in dollars, for a month in which $k = 650$ kilowatt-hours were used.
\par\vspace*{5.4cm}\ansline}

\problem{A farm sells crates of apples. When $x$ hundred crates are sold, revenue is $R(x) = x(40 - 2x)$ hundred dollars and cost is $C(x) = 12x + 48$ hundred dollars.
\par\vspace{0.1cm}
(a) Name the family of each rule and justify each choice from the way the rule is built.
\par\vspace*{2.5cm}\ansline
\par\vspace{0.15cm}
(b) Solve $R(x) = C(x)$ to find both break-even values of $x$.
\par\vspace*{3.2cm}\ansline}
