\begin{center}
\small Vector motion, polar curve intersections, and accumulated quantities
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--3 are a warm-up, one on each idea. After that they are mixed: decide what is being asked before you pick a formula.
  \item Angles are in radians. Give exact answers where the problem says so, and decimals to the stated precision otherwise.
  \item Where an answer carries a unit, the answer line names it.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{A particle moves in the plane with position $x(t) = t^2 + 1$ and $y(t) = 3t$. Find its speed at $t = 2$.}

\problem[5.5cm]{Find every angle $\theta$ in $[0, 2\pi)$ at which the polar curves $r = 2$ and $r = 4\cos\theta$ meet. Give exact answers.}

\problem{Water enters a reservoir at the rate $r(t) = 5 + 2\cos t$ litres per hour. How much water enters between $t = 0$ and $t = \pi$ hours? Give an exact answer and a decimal to three places.
\par\vspace*{5cm}\answerline{L}}

\problem[6cm]{A curve is given by $x = \sin(2t)$, $y = t^2$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find $\dfrac{dx}{dt}$. \ansline
  \item[(b)] Find $\dfrac{dy}{dx}$ at $t = \dfrac{\pi}{6}$, to three decimal places. \ansline
\end{itemize}}

\problem[6.5cm]{Find the area of the region that lies inside the polar curve $r = 4\cos\theta$ and outside the circle $r = 2$. Give an exact answer and a decimal to three places.}

\problem{A particle travels along the curve $x(t) = 3t$, $y(t) = 2t^{3/2}$ for $0 \le t \le 4$, with lengths in metres. Find the distance it travels. Give an exact answer and a decimal to three places.
\par\vspace*{5.5cm}\answerline{m}}

\problem[6cm]{A particle in the plane has velocity components $x'(t) = 4\cos t$ and $y'(t) = 4\sin t$, and at $t = 0$ it is at the point $(2, 0)$. Find its position at $t = \dfrac{\pi}{2}$.}

\problem[7cm]{Consider the polar curves $r = 1 + \cos\theta$ and $r = 3\cos\theta$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find every $\theta$ in $[0, 2\pi)$ satisfying both equations. \ansline
  \item[(b)] Find the value of $r$ at those angles. \ansline
  \item[(c)] The two curves also both pass through the pole. Explain why solving the two equations simultaneously does not find that point.
\end{itemize}}

\problem[6.5cm]{Water flows through a pipe at the rate $r(t) = 12e^{-0.2t}$ litres per minute, where $t$ is in minutes.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] How much water flows through during the first $5$ minutes? Give an exact answer and a decimal to two places. \answerline{L}
  \item[(b)] At what time has the flow rate fallen to half its initial value? Give a decimal to three places. \answerline{min}
\end{itemize}}

\problem[6.5cm]{A particle starts at the origin and moves with velocity vector $\langle 2t,\ t^2 - 1\rangle$ metres per second, for $0 \le t \le 3$ seconds.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the total distance the particle travels on $[0, 3]$. \answerline{m}
  \item[(b)] Find the particle's position at $t = 3$. \ansline
  \item[(c)] Explain why the distance in part (a) is greater than the straight-line distance from the origin to the position in part (b).
\end{itemize}}
