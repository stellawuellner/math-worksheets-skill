\wstitleblock{Two-Step Measurement Stories}{Grade 3}{}

\begin{center}
\textit{Two-step stories about time, money, mass, and liquid volume}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}
  \item Every story here takes \emph{two} steps. Find the hidden first step,
        then answer the question.
  \item Watch the units: minutes, cents, dollars, grams (g), kilograms (kg),
        and liters (L).
  \item Show both steps in the work space, and write the unit with every answer.
\end{itemize}

% ---------------------------------------------------------------- Problem 1
\problem{Sam's recipe should take 40 minutes in all. He stirred the batter for
5 minutes and baked it for 30 minutes. How many minutes are left for cooling?
\par\vspace*{3cm}\answerline{minutes}}

% ---------------------------------------------------------------- Problem 2
\problem{Rosa buys a pencil for 35 cents and an eraser for 25 cents. She pays
with a one-dollar bill, which is worth 100 cents. How many cents does she get
back?
\par\vspace*{3cm}}

% ---------------------------------------------------------------- Problem 3
\problem{Mrs.\ Diaz buys 12 kg of apples and 8 kg of pears. Her cart can hold
30 kg in all. How many more kilograms of food can she add to the cart?
\par\vspace*{3cm}\answerline{kg}}

% ---------------------------------------------------------------- Problem 4
\problem{A soup pot holds 6 L. The chef pours in 2 L of broth and 3 L of water.
How many more liters can the pot hold?
\par\vspace*{3cm}\answerline{L}}

% ---------------------------------------------------------------- Problem 5
\problem{Gym class is 45 minutes long. The class plays 3 rounds of a game, and
each round takes 10 minutes. How many minutes of class are \emph{not} spent on
the game?
\par\vspace*{3cm}\answerline{minutes}}

% ---------------------------------------------------------------- Problem 6
\problem{Sticker packs cost 4 dollars each. Jon buys 3 packs and pays with a
20-dollar bill. How many dollars does he get back?
\par\vspace*{3cm}}

% ---------------------------------------------------------------- Problem 7
\problem{One can of beans has a mass of 400 g. Talia buys 2 cans of beans and
one 350 g jar of sauce. What is the total mass of her groceries, in grams?
\par\vspace*{3cm}\answerline{g}}

% ---------------------------------------------------------------- Problem 8
\problem{An empty aquarium holds 24 L when full. It is filled using 4 equal
buckets. After 3 of those buckets have been poured in, how many liters of
water are still needed to fill the aquarium?
\par\vspace*{3.4cm}\answerline{L}}

% ---------------------------------------------------------------- Problem 9
\problem{The ride on Train A takes 48 minutes. The trip on Train B takes
29 minutes, plus a 12-minute wait to change trains. How many minutes longer is
Train A's ride than Train B's whole trip?
\par\vspace*{3.2cm}\answerline{minutes}}

% ---------------------------------------------------------------- Problem 10
\problem{Kane saves 5 dollars each week for 4 weeks. The game he wants costs
32 dollars. How many more dollars does he still need?
\par\vspace*{3.2cm}}

% ---------------------------------------------------------------- Problem 11
\problem{One brick has a mass of 3 kg, and one wooden board has a mass of 7 kg.
Mr.\ Lee's wagon carries 4 bricks and one board. What is the total mass of the
load, in kilograms?
\par\vspace*{3.2cm}\answerline{kg}}

% ---------------------------------------------------------------- Problem 12
\problem{\textbf{Challenge.} Papa makes 18 L of lemonade for the school fair.
He fills 5 jugs that hold 2 L each, and then 4 bottles that hold 1 L each.
How many liters of lemonade are left over?
\par\vspace*{3.6cm}\answerline{L}}
