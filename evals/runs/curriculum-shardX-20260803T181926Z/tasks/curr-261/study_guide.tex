\sstitleblock{Interpreting the Solution of a Linear System on a Graph}

\skillheading{Skill 1 --- Reading the Solution Point Off a Graph}

\begin{formulabox}
\textbf{The model:} a line is the set of \emph{all} points whose coordinates make
its equation true. Two lines drawn on one grid therefore show two such sets, and
the point where they cross belongs to both --- that point, written $(x, y)$, is
the solution of the system.\\[3pt]
\textbf{Reading it:} find the crossing, count across for $x$, count up for $y$.
Order matters: (across, up).\\[3pt]
\textbf{Always check:} substitute the pair into \emph{both} original equations. A
point read by eye can sit a square off, and a point that satisfies only one
equation is on only one line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A graph shows $y = x + 3$ and $y = -2x + 9$ crossing two right and five up.
\step{A crossing point is the one place both equations can be true at once, so the reading gives a candidate and substitution is what settles it.}
\step{The candidate is $(2, 5)$. Check: $5 = 2 + 3$ and $5 = -2(2) + 9 = 5$. Both hold, so the solution is $\ans{(2,\ 5)}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A graph shows $y = x - 1$ and $y = -x + 7$ crossing four
right and three up. Write the solution and check it in both equations.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(4,\ 3)}$}
\end{tryitbox}

\skillheading{Skill 2 --- Solving a System by Graphing}

\begin{formulabox}
\textbf{Method:} write each equation as $y = mx + b$; plot $b$ on the $y$-axis;
step off the slope ($m$ = rise over run) to a second point; draw the line. Do it
for both, then read the crossing.\\[3pt]
\textbf{Standard form is fine too:} for $2x + y = 6$, the intercepts are quick ---
set $x = 0$ to get $(0, 6)$, set $y = 0$ to get $(3, 0)$.\\[3pt]
\textbf{Limits of graphing:} it is exact only when the crossing lands on a lattice
point. A crossing between grid lines gives an estimate, and you finish with
substitution or elimination.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $y = -x + 6$ and $y = 3x - 2$ by graphing.
\step{Both are already in slope-intercept form, so each line is fixed by its intercept plus one slope step --- no rearranging is needed before plotting.}
\step{Plot $(0, 6)$ and step down 1, right 1; plot $(0, -2)$ and step up 3, right 1. The lines meet at $(2, 4)$. Check: $-2 + 6 = 4$ and $3(2) - 2 = 4$, so the solution is $\ans{(2,\ 4)}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $y = 3x - 4$ and $y = x + 2$ by graphing, then check
your point in both equations.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(3,\ 5)}$}
\end{tryitbox}

\skillheading{Skill 3 --- One Solution, None, or Infinitely Many}

\begin{formulabox}
\textbf{Compare the slope-intercept forms before you draw:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item different slopes $\rightarrow$ the lines cross once $\rightarrow$ \textbf{one solution};
  \item same slope, different intercept $\rightarrow$ parallel, never touching $\rightarrow$ \textbf{no solution};
  \item same slope \emph{and} same intercept $\rightarrow$ one line drawn twice $\rightarrow$ \textbf{infinitely many solutions}, every point on the line.
\end{itemize}
\textbf{The algebra matches the picture:} set the two right sides equal. If the
$x$ terms cancel and leave a false sentence such as $-2 = 4$, the lines are
parallel. If they cancel and leave a true one such as $6 = 6$, the equations
describe the same line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Classify $y = 3x - 2$ and $y = 3x + 4$.
\step{Both slopes are 3 and the intercepts differ, so the lines run parallel and no crossing point can exist --- worth knowing before spending time on a graph.}
\step{Setting the right sides equal gives $3x - 2 = 3x + 4$; the $3x$ cancels and leaves $-2 = 4$, which is false for every $x$, so $\ans{\text{no solution}}$. Had the leftover sentence been true, the answer would instead be every point on the line.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Classify $y = 5x + 1$ and $y = 5x - 6$. How many solutions,
and what does the graph look like?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\text{none --- parallel lines}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a point that works in one equation is not a solution ---
it has to work in both. And ``no solution'' is not the same as ``the solution is
zero'': $(0, 0)$ is a perfectly good solution when both lines pass through the
origin, while parallel lines have no solution point at all.
\end{watchoutbox}
