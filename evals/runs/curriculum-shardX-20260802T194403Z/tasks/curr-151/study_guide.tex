\sstitleblock{Ratios and Equivalent Ratio Tables}

\skillheading{Skill 1 --- Reading a ratio off a model (ratio-from-model)}

\begin{formulabox}
\textbf{Three ways to write the same ratio:}\\[4pt]
$a$ to $b$ \qquad $a:b$ \qquad $\dfrac{a}{b}$\par
\smallskip
\textbf{Order is part of the meaning.} The quantity named \emph{first} in the
question is written first. Count the model, then simplify by dividing both
numbers by a common factor --- the model does not change, only the notation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A strip shows 6 blue counters and 9 white counters.
Write blue to white in simplest form.
\step{The question names blue first, so blue is the first number: this is a
part-to-part ratio read straight off the model.}
\step{Blue to white is 6 to 9, or $6:9$, or $\frac{6}{9}$; dividing both by 3
gives $\ans{\frac{2}{3}}$ --- for every 2 blue there are 3 white.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A strip shows 10 red counters and 4 green counters. Write
red to green in simplest form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{5}{2}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Part-to-part vs.\ part-to-whole (part-to-whole)}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
If the ratio of two parts is $a:b$, the whole is $a + b$ shares.\par
\smallskip
part to part $= \dfrac{a}{b}$ \qquad part to whole $= \dfrac{a}{a+b}$\par
\smallskip
Only the \textbf{part-to-whole} form may be read as ``that fraction of the group.''
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A fruit bowl has apples and oranges in the ratio 3 to 5.
What fraction of the fruit is apples?
\step{The question asks for a fraction \emph{of the whole bowl}, so build the
whole first instead of using the 5.}
\step{The whole is $3 + 5 = 8$ shares and apples are 3 of them, so the fraction
is $\ans{\frac{3}{8}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A team has forwards and defenders in the ratio 4 to 6.
What fraction of the team is forwards?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{2}{5}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Equivalent ratio tables (equivalent-ratio-table)}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
Multiply (or divide) \textbf{both} quantities by the same number to get an
equivalent ratio. Every column of a ratio table names the same relationship.\par
\smallskip
Setting two columns equal gives a proportion:
$\dfrac{x}{\text{new}} = \dfrac{\text{old part}}{\text{old whole step}}$,
then cross-multiply.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 4 tickets cost 10 dollars. What do 6 tickets cost?
\step{Cost per ticket is steady, so the two columns are equivalent ratios ---
write a proportion rather than adding.}
\step{$\frac{x}{6} = \frac{10}{4}$, so $4x = 60$ and the cost is $\ans{15}$
dollars. (Check: 2 tickets cost 5 dollars, and $3 \times 5 = 15$.)}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 3 books cost 12 dollars. What do 7 books cost?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$}
\end{tryitbox}

\skillheading{Skill 4 --- The value of a ratio (value-of-ratio)}

\begin{formulabox}
\textbf{Rule:}\\[4pt]
The \textbf{value} of the ratio $a:b$ is the quotient $a \div b$ --- one number,
the unit rate ``$a$-units per 1 $b$-unit.''\par
\smallskip
Two ratios are equivalent exactly when their values are equal, so comparing
values is how you decide which of two ratios is larger.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A car travels 18 miles in 3 hours. Write the ratio of
miles to hours and find its value.
\step{``Per hour'' means the hours become 1, so divide the first quantity by the
second.}
\step{The ratio is $\frac{18}{3}$, and its value is $18 \div 3 = \ans{6}$ miles
per hour.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A typist types 45 words in 5 minutes. Find the value of the
ratio of words to minutes.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $3:5$ and $\frac{3}{5}$ look alike but answer different
questions when the parts are \emph{both} parts: $3:5$ says 3 reds for every
5 blues, while ``$\frac{3}{5}$ of the group'' would need the whole, which is 8.
Ask ``compared with the other part, or with the whole?'' before writing.
\end{watchoutbox}
