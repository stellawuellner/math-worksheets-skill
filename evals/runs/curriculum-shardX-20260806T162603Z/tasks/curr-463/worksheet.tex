\wstitleblock{Second and Higher Derivatives: Motion, Decay, and Implicit Curves}{AP Calculus AB/BC}{}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Every model below states its units. Keep them attached to your work.
  \item Write derivatives in the notation the problem uses ($s''(t)$, $y^{(4)}$, $\dfrac{d^2y}{dx^2}$).
  \item Leave exact values (fractions, radicals, $e$) unless a decimal is asked for.
\end{itemize}

\vspace{0.2cm}

\problem[5.5cm]{A freight elevator's height above the ground floor is modelled by
$h(t) = 2t^4 - 5t^3 + 7t$, where $h$ is in metres and $t$ is in seconds.
Find the elevator's acceleration function $h''(t)$.}

\problem{A test cart runs along a straight track with position
$s(t) = t^3 - 9t^2 + 24t$, where $s$ is in metres and $t$ is in seconds.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the acceleration function $s''(t)$.
    \par\vspace*{3.4cm}\ansline
  \item[(b)] Find the cart's acceleration at $t = 5$ seconds.
    \par\vspace*{2.6cm}\answerline{m/s$^2$}
\end{itemize}}

\problem[5.5cm]{A buoy bobs on a swell so that its height above mean sea level is
$y(t) = 3\sin(2t)$ metres, with $t$ in seconds.
Find the fourth derivative $y^{(4)}(t)$.}

\problem[5.5cm]{A cooling tank loses water so that the volume remaining is
$V(t) = 500e^{-t/5}$ litres, with $t$ in minutes.
Find the third derivative $V'''(t)$.}

\problem{A survey drone's altitude is $A(t) = t^4 - 8t^3 + 18t^2 + 30$ metres,
with $t$ in seconds and $0 \le t \le 6$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the vertical acceleration $A''(t)$.
    \par\vspace*{3.2cm}\ansline
  \item[(b)] Find every time $t$ in the interval at which the vertical
    acceleration is zero.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}

\problem{A hanging cable is modelled near its lowest point by the curve
$y^2 = 4x$ with $y > 0$, where $x$ and $y$ are in metres.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Differentiate implicitly to find $\dfrac{dy}{dx}$ as a function of $x$ alone.
    \par\vspace*{3.4cm}\ansline
  \item[(b)] Find $\dfrac{d^2y}{dx^2}$ as a function of $x$ alone.
    \par\vspace*{3.4cm}\ansline
\end{itemize}}

\problem{A circular fairground ride has rim $x^2 + y^2 = 25$, with $x$ and $y$
in metres. A camera tracks the upper half of the rim, where $y > 0$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $\dfrac{d^2y}{dx^2}$ for the upper half, as a function of $x$ alone.
    \par\vspace*{3.6cm}\ansline
  \item[(b)] Evaluate $\dfrac{d^2y}{dx^2}$ at the point $(3, 4)$. Give an exact fraction.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}

\problem{A bacterial culture holds $P(t) = t^3 - 12t^2 + 36t + 50$ thousand cells
after $t$ hours, for $0 \le t \le 8$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $P''(t)$.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] The growth rate is $P'(t)$. Find the hour at which the growth rate
    is smallest.
    \par\vspace*{2.6cm}\ansline
  \item[(c)] Explain how the sign of $P''$ shows that the growth rate really is
    smallest there, and not largest.
    \par\vspace*{3.0cm}\ansline
\end{itemize}}
