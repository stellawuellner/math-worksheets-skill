\wstitleblock{Solving Trigonometric Equations on Restricted Intervals}{Precalculus \quad Trigonometric Identities and Equations}{}

\begin{center}\small\textit{Linear trigonometric equations, identity-first rewrites,
quadratics in a trigonometric function, and periodic models --- all on restricted
intervals}\end{center}

\noindent\textbf{Why the interval is part of the question.} \small A trigonometric
equation has infinitely many solutions, spaced one period apart. Restricting the
interval is what turns it into a question with a finite, listable answer. The
graph below is the whole idea: $y = \sin x$ meets the line $y = \tfrac{1}{2}$
twice on $[0, 2\pi)$ --- so ``solve $2\sin x - 1 = 0$ on $[0, 2\pi)$'' has exactly
two answers, and reporting one of them is only half an answer.\normalsize

\noindent\textbf{Directions.} Give \emph{every} solution in the stated interval,
in exact form (radians unless the problem says degrees). Show the rewrite or
factoring that reduced the equation, and state the units where the problem has
them.

\problem[4.5cm]{Solve on $[0, 2\pi)$: \quad $2\sin x - 1 = 0$

\smallskip
\small Use the graph above to predict how many solutions you should find before
you compute them.\normalsize}

\problem[4.5cm]{Solve on $[0, 2\pi)$: \quad $2\cos x + \sqrt{3} = 0$}

\problem[5cm]{Solve on $[-\pi, \pi)$: \quad $\tan x + 1 = 0$

\smallskip
\small The period of tangent is $\pi$, not $2\pi$, and the interval is not the
usual one --- both facts change the count.\normalsize}

\problem[5cm]{Verify the double-angle identity $\cos 2x = 1 - 2\sin^2 x$ by
starting from $\cos 2x = \cos^2 x - \sin^2 x$ and using $\cos^2 x + \sin^2 x = 1$.

\smallskip
\small Write the identity you end with --- you will use it in problem 5.\normalsize}

\problem[6cm]{Solve on $[0, 2\pi)$: \quad $\cos 2x + 3\sin x = 2$

\smallskip
\small Two different functions of two different angles cannot be combined. Use
problem 4's identity to write everything in terms of $\sin x$ first.\normalsize}

\problem[6cm]{Solve on $[0, 2\pi)$: \quad $2\cos^2 x - \cos x - 1 = 0$

\smallskip
\small This is a quadratic in $\cos x$. Factor it, then solve each factor
separately --- and check whether each value of $\cos x$ is attainable.\normalsize}

\problem[4.5cm]{A Ferris wheel turns at a constant rate. A rider's height above
the ground, in metres, $t$ seconds after boarding, is
\[ h(t) = 9 - 8\cos\!\left(\frac{\pi t}{15}\right). \]
Find the rider's height $10$ seconds after boarding.
\answerline{m}}

\problem[6cm]{For the same Ferris wheel, $h(t) = 9 - 8\cos\!\left(\dfrac{\pi t}{15}\right)$
metres. Find every time $t$ in the interval $[0, 30)$ seconds at which the rider
is exactly $13$ metres above the ground.

\smallskip
\small The interval is one full turn of the wheel. Give both times.
\answerline{s}}

\problem[6cm]{Solve on $[0, 2\pi)$: \quad $\sin 2x = \cos x$

\smallskip
\small Rewrite $\sin 2x$, move everything to one side and factor. Dividing both
sides by $\cos x$ would destroy solutions --- do not do it.\normalsize}

\problem[6cm]{Solve on $[0^\circ, 360^\circ)$: \quad
$2\sin^2\theta - \sin\theta - 1 = 0$

\smallskip
\small Degrees this time. Factor the quadratic in $\sin\theta$, solve each
factor, and give all solutions in degrees.\normalsize}
