\sstitleblock{Diagnosing Causation Claims and Biased Study Designs}

\skillheading{Skill 1 --- Association is not causation}

\begin{formulabox}
\textbf{Rule:} a correlation says the two variables move together, nothing more. Before
accepting a causal claim, look for a \emph{common cause} (a third variable driving both) and
for \emph{reverse causation} (the effect driving the supposed cause). $r^2$ measures how much
of the variation the linear fit accounts for --- it does not upgrade an association into a
cause.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two variables have $r = 0.80$. Find $r^2$.
\step{Squaring $r$ turns the strength of the association into the share of variation the fit accounts for --- still a description, not a cause.}
\step{$r^2 = 0.80^2 = 0.64$, so $64\%$ of the variation is accounted for.}

$\Rightarrow\quad r^2 = \ans{0.64}$, and $36\%$ is still unexplained --- room for other causes.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two variables have $r = 0.60$. Find $r^2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $r^2 = \ans{0.36}$}
\end{tryitbox}

\skillheading{Skill 2 --- Who was actually asked}

\begin{formulabox}
\textbf{Rule:} a proportion describes the group it was computed from. Ask where the sample came
from: \emph{voluntary response} (people chose to reply) over-represents strong opinions;
\emph{convenience} sampling (whoever was nearby) over-represents whoever gathers there;
\emph{undercoverage} leaves part of the population unreachable. Only a random sample of the
whole population supports a claim about the whole population.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Of $200$ people leaving a concert, $180$ said they enjoy live music. Find the sample proportion.
\step{The figure asked for describes the people actually asked, so the denominator is the sample size, not the town.}
\step{$P = \dfrac{180}{200}$, which reduces by $20$.}

$\Rightarrow\quad \ans{\tfrac{9}{10}}$ of \emph{those asked} --- a convenience sample of concertgoers, so it says nothing about the town.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Of $150$ people leaving a library, $120$ said they read daily. Find the sample proportion.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{4}{5}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Observational study or experiment?}

\begin{formulabox}
\textbf{Rule:} if the subjects chose their own group, the study is \emph{observational} and a
group difference may come from a \emph{confounding} variable rather than the treatment. Only an
\emph{experiment} --- the researcher assigns treatments at \emph{random} --- balances the
unknown variables and supports a causal conclusion. Blocking first, then randomizing inside
each block, handles a variable you already know matters.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Five volunteers who took a training scored $61, 63, 65, 67, 69$. Find the group's mean.
\step{Comparing groups starts with a number for each, so compute this group's mean before asking what the comparison is worth.}
\step{$\dfrac{61+63+65+67+69}{5} = \dfrac{325}{5}$.}

$\Rightarrow\quad$ mean $= \ans{65}$ --- but the volunteers self-selected, so the gap is not yet evidence.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The five who did not train scored $54, 56, 58, 60, 62$. Find their mean.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{58}$}
\end{tryitbox}

\skillheading{Skill 4 --- Relative change versus the base rate}

\begin{formulabox}
\textbf{Rule:} ``risk cut by half'' is a \emph{relative} change; the number of people affected
is the \emph{absolute} change, $\dfrac{a}{n} - \dfrac{b}{n}$. A large relative change on a tiny
base rate is a small absolute change. Always ask ``half of what?'' --- a percentage without its
base is not information.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Risk falls from $6$ cases per $500$ to $3$ per $500$. Find the absolute change in risk.
\step{The headline would say fifty percent; the absolute change is what tells you how many people are affected, so subtract the two rates.}
\step{$\dfrac{6}{500} - \dfrac{3}{500} = \dfrac{3}{500}$.}

$\Rightarrow\quad \ans{\tfrac{3}{500}}$, that is $3$ fewer cases per $500$ people.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Risk falls from $8$ per $400$ to $2$ per $400$. Find the absolute change.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{200}}$}
\end{tryitbox}
