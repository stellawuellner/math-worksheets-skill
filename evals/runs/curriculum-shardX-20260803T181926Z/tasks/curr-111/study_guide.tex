\sstitleblock{Quotients and Remainders in Whole-Number Division}

\skillheading{Skill 1 --- Equal groups and arrays}

\begin{formulabox}
\textbf{Model:} dividing means making equal groups. The \textbf{quotient} counts the
full groups; the \textbf{remainder} is what will not fill another group.
Written as $\text{dividend} = \text{divisor} \times \text{quotient} + \text{remainder}$.
The remainder is always \emph{smaller} than the divisor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $23$ counters in groups of $4$
\step{I want full groups of four, so I ring four counters at a time and keep ringing until fewer than four are left.}
\step{Five rings hold $5 \times 4 = 20$ counters and $23 - 20 = 3$ are outside, and $3 < 4$, so quotient $\ans{5}$, remainder $\ans{3}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $30$ counters in groups of $4$\\[2pt]
\rotatebox{180}{\footnotesize check: quotient $\ans{7}$, remainder $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Check by multiplying back}

\begin{formulabox}
\textbf{Rule:} $\text{divisor} \times \text{quotient} + \text{remainder} = \text{dividend}$.
Take the remainder off the total first, then divide what is left --- that also finds
a missing number of groups.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $6 \times \square + 2 = 32$
\step{The two left over were never put in a group, so I take them off the total before dividing.}
\step{$32 - 2 = 30$ went into groups of six, and $30 \div 6 = \ans{5}$ groups; check $6 \times 5 + 2 = 32$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $7 \times \square + 4 = 46$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimate the quotient first}

\begin{formulabox}
\textbf{Rule:} round the dividend to a nearby friendly number that the divisor goes
into evenly, then divide. The estimate tells you roughly how big the real quotient
should be, so a wildly wrong answer shows up at once.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Estimate $418 \div 8$
\step{I only need about how many, so I round the dividend to a hundred that eight divides evenly and use that instead.}
\step{$418$ rounds to $\ans{400}$, and $400 \div 8 = \ans{50}$, so the real quotient is close to fifty.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Estimate $287 \div 5$\\[2pt]
\rotatebox{180}{\footnotesize check: round to $\ans{300}$, so about $\ans{60}$}
\end{tryitbox}

\skillheading{Skill 4 --- Read what the remainder means}

\begin{formulabox}
\textbf{Three ways a story uses the remainder:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Round up} --- the leftovers still need a place (vans, tents, boxes).
  \item \textbf{Drop it} --- the leftovers cannot make another whole thing (bows, teams).
  \item \textbf{The remainder is the answer} --- ``how many are left over?''
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $27$ campers, each tent sleeps $6$. How many tents?
\step{Every camper needs a bed and a tent cannot be split, so the leftovers will still need a tent of their own.}
\step{$27 \div 6$ gives quotient $\ans{4}$ and remainder $\ans{3}$; those three still need a tent, so book $\ans{5}$ tents.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $34$ people, each boat holds $6$. How many boats?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$ R $\ans{4}$, so $\ans{6}$ boats}
\end{tryitbox}
