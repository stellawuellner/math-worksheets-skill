\sstitleblock{Diagnosing Causal Claims and Biased Study Designs}

\skillheading{Skill 1 --- Coverage and response rate}

\begin{formulabox}
\textbf{Two different fractions, two different faults.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Coverage} $=$ (units the frame can reach) $\div$ (units in the
        population). Low coverage is \textbf{undercoverage}: some units have
        probability zero of ever being selected.
  \item \textbf{Response rate} $=$ (units that answered) $\div$ (units contacted).
        Low response is \textbf{nonresponse bias} when the silent units differ
        from the ones who answered.
  \item \textbf{Voluntary response} is the worst case of both: respondents
        select themselves, so strong opinions are over-represented by design. A
        larger sample fixes none of this --- it shrinks the interval around a
        biased estimate.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A survey is mailed to $300$ households and $45$ of them
return it. Give the response rate as an exact fraction.

\step{The question is about who answered among those contacted, so the
denominator is the number contacted, not the number in the town.}

\step{Form $45$ over $300$ and reduce by the common factor $15$.}

The response rate is $\ans{\tfrac{3}{20}}$, so most of the contacted households
are unrepresented in the result.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A survey is mailed to $300$ households and $105$ return it.
Give the response rate as an exact fraction.

\rotatebox{180}{\footnotesize check: $\ans{\tfrac{7}{20}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Comparing two groups honestly}

\begin{formulabox}
\textbf{Before comparing two group means, ask who is in each mean.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Attrition:} average everyone who \emph{started}, not everyone who
        \emph{finished}. Dropping the leavers biases the treatment mean upward
        when the leavers were doing worst.
  \item \textbf{Assignment:} if subjects chose their own group, the study is
        observational and the gap is association.
  \item \textbf{Confounding:} name a variable that differs between the groups and
        moves the outcome on its own (motivation, baseline level, income, age).
        \textbf{The remedy} is random assignment from a single pool, with every
        randomized subject kept in the arithmetic.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Volunteers for a program scored $8$, $2$, $5$; people who
did not volunteer scored $4$, $1$, $1$. Find both means and judge the claim ``the
program works''.

\step{Average each group over its own members only --- the comparison is
meaningless if one mean quietly excludes the people who did badly.}

\step{Add each group's three values and divide by three.}

Volunteers: $\ans{5}$. Non-volunteers: $\ans{2}$. The gap is real, but the
subjects volunteered, so it is association --- motivation alone could produce it.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad One group scored $10$, $6$, $8$; the other scored $5$, $4$,
$3$. Find both means.

\rotatebox{180}{\footnotesize check: $\ans{8}$ and $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Is the sample representative?}

\begin{formulabox}
\textbf{Compare the sample's composition with the population's, on a variable you
already know.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Over-representation in \textbf{percentage points} is a
        \textbf{subtraction}: sample share $-$ population share.
  \item Dividing the two shares gives a ratio. Reporting a ratio as ``points''
        understates a large skew and overstates a small one.
  \item A visible skew on one known variable is evidence that unknown variables
        are skewed too. Fix it by scoping the conclusion to who was actually
        sampled, or by drawing a probability sample from a frame that covers the
        target population.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A mall-intercept sample is $48\%$ students, while the town
is $18\%$ students. By how many percentage points are students over-represented?

\step{Percentage points compare two shares by subtraction, so take the sample
share minus the population share rather than their ratio.}

\step{Subtract: $48 - 18$.}

Students are over-represented by $\ans{30}$ percentage points.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A sample is $62\%$ homeowners while the town is $25\%$
homeowners. By how many percentage points are homeowners over-represented?

\rotatebox{180}{\footnotesize check: $\ans{37}$}
\end{tryitbox}
