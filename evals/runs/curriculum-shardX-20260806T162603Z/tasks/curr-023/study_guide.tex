\sstitleblock{Adding Story Problems}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Act Out the Story}

\begin{formulabox}
\textbf{Rule:} An add-to story starts with some things, and then more come.
Show it! Draw a shape for each thing in the start, then a shape for each
thing that comes, and count them all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 2 birds sit in a tree. 3 more birds fly in. How many
birds now?
\step{The story adds more birds, so we draw the start, draw what comes, and count everything.}
\step{Draw 2 circles, then 3 more circles. Count: 1, 2, 3, 4, 5 --- so $2 + 3 = 5$. $\checkmark$}

$\Rightarrow\quad$ there are $\ans{5}$ birds
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 4 cats nap on a rug. 2 more cats join them. Draw it
and count. How many cats?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Write the Equation}

\begin{formulabox}
\textbf{Rule:} The two story numbers go into the equation, and the answer is
the total: \ start $+$ more $=$ total.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad 3 fish swim in a tank. 4 more fish are added. Write
the equation and solve it.
\step{The start number is 3 and the more number is 4, so those two numbers go into the equation.}
\step{$3 + 4 = {}$? Count on from 4: 5, 6, 7. So $3 + 4 = 7$. $\checkmark$}

$\Rightarrow\quad 3 + 4 = \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad 5 ducks are in the pond, and 4 more waddle in. Write
the equation and solve it.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Count On to Solve}

\begin{formulabox}
\textbf{Rule:} Start at one number and count on the other. Start with the
\textbf{bigger} number --- fewer counts means fewer chances to slip!
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $6 + 2$
\step{Counting on is fastest from the bigger number, so we start at 6 and count on 2.}
\step{Say 6, then count: 7, 8. So $6 + 2 = 8$. $\checkmark$}

$\Rightarrow\quad 6 + 2 = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $2 + 7$ \ (start from the bigger number!)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When you count on, do not say the start number as a
count. For $6+2$, say ``6\,'' --- then count ``7, 8\,''. Counting ``6, 7\,''
lands one short!
\end{watchoutbox}
