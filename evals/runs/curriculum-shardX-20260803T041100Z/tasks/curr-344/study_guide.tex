\sstitleblock{Scale Factor Confusion: Linear, Square, Cubic}

\skillheading{Skill 1 --- Scaling an area: multiply by the square of $k$}

\begin{formulabox}
\textbf{Rule:} for similar figures with linear scale factor $k$,
\[ \frac{\text{new length}}{\text{old length}} = k, \qquad
   \frac{\text{new area}}{\text{old area}} = k^2, \qquad
   \frac{\text{new volume}}{\text{old volume}} = k^3 \]
\textbf{Why:} an area is built from two perpendicular lengths and a dilation stretches
both, so the factor is used twice. Surface area follows the area rule, not the volume
rule.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A figure of area 20 is dilated by a scale factor of 4. Find the
new area.
\step{Area is asked for, and a dilation stretches both dimensions, so the factor is used
twice --- the area rule.}
\step{New area $= 20 \cdot 4^2 = 320$. Using the factor only once would give 80, which is
the length rule applied to an area.}
$\Rightarrow\quad$ new area $= \ans{320}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A figure of area 7 is dilated by a scale factor of 5. Find the new
area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{175}$}
\end{tryitbox}

\skillheading{Skill 2 --- Scaling a volume: multiply by the cube of $k$}

\begin{formulabox}
\textbf{Rule:} a volume is built from three perpendicular lengths, so a dilation uses the
factor three times: $V' = k^3 V$.\par
\smallskip
\textbf{The classic slip:} using $k^2$ on a volume (or $k^3$ on a surface area). Match the
exponent to what is being measured --- paint on the outside is an area, sand on the inside
is a volume, and they scale by \emph{different} amounts for the same $k$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two similar solids have linear scale factor 3. The smaller has
volume 15. Find the volume of the larger.
\step{A volume is asked for, and all three dimensions stretch, so the factor is used three
times --- the volume rule.}
\step{Larger volume $= 15 \cdot 3^3 = 405$. Squaring instead would give 135, the area rule
misapplied to a solid.}
$\Rightarrow\quad$ larger volume $= \ans{405}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two similar solids have linear scale factor 4. The smaller has
volume 6. Find the volume of the larger.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{384}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding $k$ from an area or volume ratio}

\begin{formulabox}
\textbf{Rule (running the rule backwards):}
\[ k = \sqrt{\frac{\text{area ratio}}{1}} \qquad\text{and}\qquad
   k = \sqrt[3]{\frac{\text{volume ratio}}{1}} \]
A ratio of areas or volumes is \textbf{never} the scale factor itself. Take the root that
matches the measurement, then confirm by scaling forwards.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two similar polygons have areas in the ratio 81 to 16. Find the
linear scale factor.
\step{The given ratio compares areas, and an area ratio is the square of the linear
factor, so undo the squaring with a square root.}
\step{$k = \sqrt{81/16} = 2.25$. Reporting the ratio itself would give about 5.06, which
is a ratio of areas, not of lengths.}
$\Rightarrow\quad k = \ans{2.25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two similar solids have volumes in the ratio 216 to 27. Find the
linear scale factor. (Take the cube root.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the exponent counts \emph{dimensions being scaled}, not the shape
of the figure. Perimeter and slant height scale by $k$; area, surface area and any
cross-section scale by $k^2$; volume and capacity scale by $k^3$. Always finish by
scaling forwards to check.
\end{watchoutbox}
