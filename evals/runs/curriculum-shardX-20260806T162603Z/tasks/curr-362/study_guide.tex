\sstitleblock{The Rational Root Theorem}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Which Numbers Are Even Possible}

\begin{formulabox}
\textbf{Theorem:} every rational zero of a polynomial with integer coefficients
has the form $\pm \frac{p}{q}$, where $p$ divides the \textit{constant term} and
$q$ divides the \textit{leading coefficient}.\\[3pt]
Build the list as: divisors of the constant, over divisors of the leading
coefficient, each with both signs. When the leading coefficient is $1$ the list
is just the divisors of the constant.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = 2x^{3} + 3x^{2} - 8x + 3$, list the
candidates and test the simplest one.
\step{The theorem narrows an infinite search to a short list, so write the list first and only then start substituting.}
\step{Divisors of $3$ over divisors of $2$ gives $\pm 1, \pm 3, \pm \frac{1}{2}, \pm \frac{3}{2}$. Testing $x = 1$: $2 + 3 - 8 + 3$.}

$\Rightarrow\quad \ans{f(1) = 0}$, so $1$ is a zero.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = 3x^{3} - 4x^{2} - 5x + 2$ the theorem allows
$\pm 1, \pm 2, \pm \frac{1}{3}, \pm \frac{2}{3}$. Find $g(2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{g(2) = 0}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Testing a Candidate Quickly}

\begin{formulabox}
\textbf{Two equivalent tests:} substitute the candidate, or run synthetic
division with it. Either way, a result of $0$ means the candidate is a zero and
$x - c$ is a factor; anything else means move on.\\[3pt]
Synthetic division is worth the extra care: when it works, it hands you the
quotient at the same time.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Test $x = -1$ in $f(x) = x^{3} - 2x^{2} - 5x + 6$.
\step{Testing means evaluating; the only question is whether the result is zero, so compute $f$ at the candidate and read the sign of the answer off the end.}
\step{$(-1) - 2 + 5 + 6$. The result is not $0$, so $-1$ is not a zero and $x + 1$ is not a factor.}

$\Rightarrow\quad \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Test $x = 2$ in the same $f(x) = x^{3} - 2x^{2} - 5x + 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-4}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Divide It Out and Finish}

\begin{formulabox}
\textbf{Once a zero $c$ is found:} divide by $x - c$. The quotient has degree one
lower, and every remaining zero is a zero of that quotient.\\[3pt]
A cubic drops to a quadratic, which you can factor or run through the quadratic
formula --- so you never need more than one lucky candidate.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Factor $x^{3} - 2x^{2} - 5x + 6$ completely.
\step{The candidate list is the divisors of $6$; $x = 1$ gives $0$, so $x - 1$ is a factor and the job becomes a quadratic.}
\step{Dividing by $x - 1$ leaves $x^{2} - x - 6$, and that factors as $(x - 3)(x + 2)$.}

$\Rightarrow\quad \ans{(x-1)(x-3)(x+2)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Factor $x^{3} + 3x^{2} - 4x - 12$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x-2)(x+2)(x+3)}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Reporting Every Zero}

\begin{formulabox}
\textbf{Count them:} a degree-$n$ polynomial has $n$ zeros counted with
multiplicity. If you have fewer, you are not finished.\\[3pt]
\textbf{And expect leftovers:} the theorem finds only \textit{rational} zeros.
Irrational or non-real zeros never appear on the candidate list --- they come
out of the leftover quadratic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find all zeros of $x^{3} - 7x + 6$.
\step{Degree three means three zeros, so plan to find one from the candidate list and get the other two from the quadratic that is left.}
\step{Divisors of $6$: $x = 1$ gives $1 - 7 + 6 = 0$. Dividing by $x - 1$ leaves $x^{2} + x - 6 = (x + 3)(x - 2)$.}

$\Rightarrow\quad \ans{x = 1,\ 2,\ -3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find all zeros of $2x^{3} - 3x^{2} - 11x + 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3,\ \tfrac{1}{2},\ -2}$}
\end{tryitbox}
