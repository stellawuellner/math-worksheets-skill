\sstitleblock{Reading the Evidence for Triangle Congruence}

\skillheading{Skill 1 --- Naming the criterion from the marks}

\begin{formulabox}
\begin{itemize}[leftmargin=1.1cm, itemsep=0pt, topsep=1pt]
  \item \textbf{SSS} --- three pairs of sides.
  \item \textbf{SAS} --- two sides and the angle \emph{between} them.
  \item \textbf{ASA} --- two angles and the side \emph{between} them.
  \item \textbf{AAS} --- two angles and a side \emph{not} between them.
  \item \textbf{HL} --- right triangles only: hypotenuse and one leg.
\end{itemize}
\textbf{Not a criterion: SSA.} Two sides and a non-included angle leave the second side free to swing to two closing points. HL is the one case rescued, because the right angle removes the second position.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two right triangles have hypotenuse $c = 13$ cm and one leg $a = 5$ cm, so they are congruent by HL. Find the remaining leg.
\step{HL names the two known parts as the hypotenuse and a leg, so the third side comes from the Pythagorean theorem rather than from any congruence rule.}
\step{Remaining leg $= \sqrt{c^2 - a^2} = \sqrt{169 - 25} = \sqrt{144}$.}

$\Rightarrow\quad \ans{12}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two right triangles have hypotenuse $25$ cm and one leg $7$ cm. Find the remaining leg.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{24}$}
\end{tryitbox}

\skillheading{Skill 2 --- From a criterion to an equation}

\begin{formulabox}
Once two triangles are congruent, \textbf{every} pair of corresponding parts is congruent. So a labelled side matches the side in the same position of the other triangle, and a labelled angle matches the angle in the same position.\par
\textbf{Read the correspondence off the congruence statement:} in $\triangle ABC \cong \triangle DEF$, $A$ matches $D$, $B$ matches $E$, $C$ matches $F$ --- so $\overline{AB}$ matches $\overline{DE}$ and $\angle B$ matches $\angle E$. Set the two expressions equal and solve.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\triangle ABC \cong \triangle DEF$ with $AB = 4x + 3$ cm and $DE = 2x + 15$ cm. Find $x$.
\step{The congruence statement pairs $A$ with $D$ and $B$ with $E$, so $\overline{AB}$ and $\overline{DE}$ are corresponding sides and their lengths must be equal.}
\step{$4x + 3 = 2x + 15$, so $2x = 12$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $\triangle ABC \cong \triangle DEF$ with $BC = 5x - 2$ cm and $EF = 3x + 14$ cm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 8}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the evidence nobody marked}

\begin{formulabox}
Three pieces of evidence carry no tick marks and have to be read off the figure:\par
\begin{itemize}[leftmargin=1.1cm, itemsep=0pt, topsep=1pt]
  \item a \textbf{shared side or angle} --- congruent to itself by the \emph{reflexive property};
  \item \textbf{vertical angles} at a crossing --- congruent by the Vertical Angles Theorem;
  \item a \textbf{third angle} --- the angle sum of a triangle is $180^\circ$, so two matching angles force the third, which converts an AAS setup into ASA.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two angles of a triangle measure $a = 52^\circ$ and $b = 61^\circ$. Find the third angle.
\step{Two pairs of angles already match, so the third pair is free evidence --- the angle sum fixes it without any extra given.}
\step{Third angle $= 180 - a - b = 180 - 52 - 61$.}

$\Rightarrow\quad \ans{67}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two angles of a triangle measure $48^\circ$ and $74^\circ$. Find the third.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{58}$}
\end{tryitbox}
