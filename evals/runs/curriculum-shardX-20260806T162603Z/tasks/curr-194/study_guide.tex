% =================== SKILL 1 ===================
\skillheading{Circumference and Area of a Circle}

\begin{formulabox}
\textbf{Formulas:} circumference $C = \pi d = 2\pi r$ \quad and \quad area
$A = \pi r^2$.\par
The \textbf{diameter is twice the radius}: $d = 2r$. Before using any formula,
ask: \emph{was I given $r$ or $d$?} The area formula only ever takes $r$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A circle has radius $5$ cm. Find its circumference.
\step{We were given the \emph{radius} and want the distance around --- use
$C = 2\pi r$.}
\step{$C = 2\pi \cdot 5 = 31.415\ldots$}

$\Rightarrow\quad \ans{31.42}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A circle has \emph{diameter} $18$ cm. Find its
circumference.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{56.55}$ cm}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Perimeter or Area? Ask What the Question Buys}

\begin{formulabox}
\textbf{Rule for rectangles:} perimeter $P = 2(\ell + w)$ measures the distance
\emph{around} --- fences, frames, trim. Area $A = \ell \cdot w$ measures the
surface \emph{covered} --- paint, carpet, sod. Perimeter answers come in m or
ft; area answers come in m$^2$ or ft$^2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A yard is $8$ m by $3$ m. How much fencing surrounds
it, and how much sod covers it?
\step{Fencing goes \emph{around} (perimeter); sod \emph{covers} (area) --- two
different formulas on the same numbers.}
\step{$P = 2(8 + 3) = 22$ and $A = 8 \cdot 3 = 24$.}

Fencing: $\ans{22}$ m \quad Sod: $\ans{24}$ m$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A frame surrounds a $6$ in by $4$ in photo. How many
inches of frame?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{20}$ in}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Catching Radius-vs-Diameter Mistakes}

\begin{formulabox}
\textbf{Checklist:} (1) If the area came out about \emph{four times} too big,
the diameter probably went in where the radius belonged. (2) If a
circumference came out exactly double or half what a classmate got, one of you
mixed up $C = \pi d$ and $C = 2\pi r$. (3) An "area" measured in plain meters
or feet is a perimeter in disguise.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Kim says a circle with diameter $20$ m has area
$\pi \cdot 20^2 = 1256.64$ square meters. Fix her work.
\step{The area formula takes the \emph{radius}, and the radius is half of
$20$, which is $10$.}
\step{$A = \pi \cdot 10^2 = 314.159\ldots$ --- one quarter of Kim's value.}

$\Rightarrow\quad \ans{314.16}$ m$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the area of a circle with \emph{diameter} $8$
in.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{50.27}$ in$^2$}
\end{tryitbox}
