\wstitleblock{Finding a Percent of a Quantity}{Grades 6--7}{}

\begin{center}\small
Benchmark percents on a hundred grid and a bar model, the decimal multiplier, the fraction shortcut, and percent applications
\end{center}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item ``Percent'' means \textbf{out of one hundred}. A percent of a quantity is a
        part of that quantity, so $30\%$ of $70$ means $\frac{30}{100}$ of $70$.
  \item Three routes give the same answer: split the whole on a \textbf{bar model},
        multiply by the \textbf{decimal}, or multiply by the \textbf{fraction}.
  \item Every problem here asks for money in dollars, a count, or a number ---
        no units are hidden in the arithmetic.
  \item Show the route you used, then write your answer on the line.
\end{itemize}
\normalsize

\problem[5cm]{\textbf{Bar model.} The whole bar below stands for $80$ books.
It is divided into ten equal parts, so each part is $10\%$ of the books.
\begin{center}
\begin{tikzpicture}
  \draw[thick, fill=blue!10] (0,0) rectangle (10,0.9);
  \foreach \i in {1,...,9}{\draw[thick] (\i,0) -- (\i,0.9);}
\end{tikzpicture}
\end{center}
Shade one part. How many books is $10\%$ of $80$?
Write the number inside the part you shaded.}

\problem[5cm]{\textbf{Bar model.} The bar below stands for $46$ marbles.
\begin{center}
\begin{tikzpicture}
  \draw[thick, fill=violet!10] (0,0) rectangle (10,0.9);
  \draw[thick] (5,0) -- (5,0.9);
\end{tikzpicture}
\end{center}
The bar is split into two equal parts. Explain what one part shows about
$50\%$, then find $50\%$ of $46$.}

\problem[5cm]{\textbf{Fraction shortcut.} A benchmark percent can be replaced by a
simple fraction: $25\%$ of an amount is the same as $\frac{1}{4}$ of it.
Use the fraction to find $25\%$ of a playlist of $64$ songs, and show the division
you used. Then check your answer by multiplying by the decimal instead.}

\problem[4.5cm]{\textbf{Find the mistake.} Dana wants $5\%$ of $120$.
She moves the decimal point one place to the left and writes $12$.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Which percent did Dana actually find? Explain how you know.
  \item Write the correct value of $5\%$ of $120$.
\end{itemize}}

\problem[5cm]{\textbf{Restaurant tip.} A meal costs $48$ dollars and the family
leaves a tip of $15\%$ of the bill.
How many dollars is the tip? Give your answer to the nearest cent.}

\problem{\textbf{Concert hall.} A hall has $250$ seats. On Tuesday $8\%$ of the seats
were empty. How many seats were empty?
\par\vspace*{5cm}\answerline{seats}}

\problem[5.5cm]{\textbf{Jacket on sale.} A jacket costs $35$ dollars and is marked
$20\%$ off.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Let $d$ be the discount in dollars. Write an equation for $d$ and solve it.
  \item Use your discount to find the sale price in dollars.
\end{itemize}}

\problem[5cm]{\textbf{An awkward percent.} A recipe calls for $12.5\%$ of a
$64$-gram block of chocolate.
$12.5\%$ is the benchmark fraction $\frac{1}{8}$.
Find the mass called for, and explain why the fraction route is easier here than
the decimal route.}

\problem[6cm]{\textbf{Two percents in a row.} A backpack costs $60$ dollars.
It is marked $25\%$ off, and then $8\%$ sales tax is added \emph{to the sale price}.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Let $s$ be the sale price in dollars. Write an equation for $s$ and solve it.
  \item Find the final price in dollars, to the nearest cent.
  \item Say why the tax is not taken on the original price.
\end{itemize}}
