\begin{center}
\textit{Slope from two points, rate of change read from a table, rate of change
read from an equation, and comparing two rates given in different forms}
\end{center}

\noindent\textbf{Before you start.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Slope is a \emph{rate}: how much $y$ changes for each $1$ that $x$
changes. \quad $m = \dfrac{y_2 - y_1}{x_2 - x_1}$
\item From a \textbf{graph}, pick two points the line passes exactly through.
From a \textbf{table}, use any two rows. From an \textbf{equation} in the form
$y = mx + b$, the rate is $m$.
\item To compare two rates given in different forms, get both into the same
units first, then compare the numbers.
\item A negative rate means the quantity is going \emph{down}. When you compare
rates that include negatives, order them on a number line, not by size.
\end{itemize}

  \item What does a $3$ mile ride cost, in dollars? \ansline

\problem{Water runs into a tub at a steady rate. The table shows how many
litres are in the tub after $x$ minutes.

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ (minutes) & $2$ & $5$ & $8$ \\ \hline
litres in tub & $14$ & $32$ & $50$ \\ \hline
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]
  \item How many litres run in between minute $5$ and minute $8$? \ansline
  \item What is the rate, in litres per minute? \ansline
\end{enumerate}}

\problem{Two streaming plans charge by the hour.

\smallskip

\par\smallskip
\begin{center}
\begin{tabular}{|c|c|c|}
\hline

\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=6pt]

  \item Write $<$ or $>$ so the sentence is true: \\[0.3cm]
