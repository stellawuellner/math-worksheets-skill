\sstitleblock{Signs and Denominators in Conic Equations}

\skillheading{Skill 1 --- The sign flip in the centre}

\begin{formulabox}
\textbf{Standard form subtracts:} $(x-h)^2 + (y-k)^2 = r^2$. So a plus inside a bracket means
a \emph{negative} coordinate --- $(x+5)^2$ gives $h = -5$.\par
\textbf{From the expanded form} $x^2 + y^2 + px + qy + f = 0$, each centre coordinate is half
its linear coefficient with the sign reversed: $h = -p/2$, $k = -q/2$. Two moves, halve and
reverse, and a quoted centre that matches the linear coefficients has skipped both.\par
\textbf{And the radius is a square root:} $r^2 = (p/2)^2 + (q/2)^2 - f$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the centre and radius of $x^2 + y^2 + 10x - 4y + 13 = 0$.
\step{Nothing can be read off until both squares are completed, so halve each linear coefficient, put it inside a bracket, and repay what that adds.}
\step{Half of $10$ is $5$ (adding $25$); half of $-4$ is $-2$ (adding $4$): \ $(x+5)^2 + (y-2)^2 = -13 + 25 + 4$.}
\step{Reading against $(x-h)^2 + (y-k)^2 = r^2$: the brackets flip the printed signs, and the right-hand side is $16$.}

$\Rightarrow\quad$ centre $\bigl(\ans{-5},\, \ans{2}\bigr)$, \quad radius $\ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the centre and radius of $x^2 + y^2 - 8x + 2y - 8 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: centre $\bigl(\ans{4}, \ans{-1}\bigr)$, radius $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Which denominator is which}

\begin{formulabox}
\textbf{Compare, do not assume.} In $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ the \emph{larger}
denominator carries the major axis --- it is not always the one under $x^2$. If the larger sits
under $y^2$, the major axis is vertical.\par
\textbf{Every denominator is a square,} so take a square root before reporting a length. The
same goes for the right-hand side of a circle equation: it is $r^2$, never $r$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find both semi-axes of $\dfrac{x^2}{9} + \dfrac{y^2}{25} = 1$.
\step{The first question is which denominator is larger, because that decides which axis is the major one --- reading left to right would answer it wrongly here.}
\step{$25 > 9$, so the major axis is vertical with half-length $\sqrt{25}$, and the minor half-length is $\sqrt{9}$.}

$\Rightarrow\quad$ semi-major $\ans{5}$ (vertical), \quad semi-minor $\ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find both semi-axes of $\dfrac{x^2}{49} + \dfrac{y^2}{16} = 1$, and say which way the major axis runs.\\[2pt]
\rotatebox{180}{\footnotesize check: semi-major $\ans{7}$ (horizontal), semi-minor $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Add or subtract for the foci}

\begin{formulabox}
\textbf{Ellipse subtracts:} $c^2 = a^2 - b^2$, and the foci end up \emph{inside} the curve, so
$c$ must come out smaller than $a$.\par
\textbf{Hyperbola adds:} $c^2 = a^2 + b^2$, and the foci end up \emph{beyond} the vertices, so
$c$ must come out larger than $a$.\par
\textbf{Use that as your check.} If an ellipse's focus lands outside its vertex, or a
hyperbola's lands inside, the wrong rule was used --- no further arithmetic needed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the focal distance of $\dfrac{x^2}{25} + \dfrac{y^2}{9} = 1$, then of $\dfrac{x^2}{16} - \dfrac{y^2}{9} = 1$.
\step{The operation between the two terms is the only thing that decides add or subtract, so read the sign before writing anything down.}
\step{Plus sign, so an ellipse: $c = \sqrt{25 - 9} = \sqrt{16}$, which is less than the vertex at $5$.}
\step{Minus sign, so a hyperbola: $c = \sqrt{16 + 9} = \sqrt{25}$, which is more than the vertex at $4$.}

$\Rightarrow\quad$ ellipse $c = \ans{4}$, \quad hyperbola $c = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the focal distance of $\dfrac{x^2}{100} + \dfrac{y^2}{36} = 1$, then of $\dfrac{x^2}{64} - \dfrac{y^2}{36} = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$ and $\ans{10}$}
\end{tryitbox}
