\sstitleblock{Comparing Two Sets by Counting On}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Which Set Has More?}

\begin{formulabox}
\textbf{Rule:} Count each set. The count that comes \textbf{later} in the
counting sequence is the bigger one --- that set has \emph{more}. If the
counts match, the sets have the \emph{same} number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Set A counts to 5 and set B counts to 8. Which set has
more?
\step{Compare the two counts by asking which one is said later when we count up.}
\step{Counting goes ``\ldots 5, 6, 7, 8 \ldots'' --- 8 comes later, so B has more: $\ans{5 < 8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Set A counts to 9 and set B counts to 6. Write the
comparison with $<$, $>$ or $=$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9 > 6}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A spread-out row can \emph{look} bigger. Only the
count decides --- not how much space the dots take up.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- How Many More?}

\begin{formulabox}
\textbf{Rule:} Start at the \textbf{smaller} count and count on up to the
\textbf{bigger} count. The number of hops you make is \emph{how many more}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad One row has 4 dots and another has 7. How many more in
the bigger row?
\step{Count on from the smaller count to the bigger one and keep track of the hops.}
\step{From 4: ``5, 6, 7'' --- that is 3 hops, so the bigger row has $\ans{3}$ more dots.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad One frame shows 6 dots and a full frame shows 10. How
many more in the full frame?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Making Two Sets Equal}

\begin{formulabox}
\textbf{Rule:} To make the small set equal to the big one, count on from the
small count to the big count. The hops tell how many to \textbf{add}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A has 3 stickers and B has 8. How many more stickers
does A need to have the same as B?
\step{The hops from A's count up to B's count are exactly the stickers A is missing.}
\step{From 3: ``4, 5, 6, 7, 8'' --- 5 hops, so A needs $\ans{5}$ more stickers.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A has 2 blocks and B has 6. How many more blocks does A
need?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}
