\sstitleblock{Similar Triangles: Sides, Perimeter and Area}

\begin{formulabox}
\textbf{The one rule everything below comes from.} If two figures are similar
with scale factor $k$ (target length over starting length), then every
\emph{length} is multiplied by $k$, every \emph{perimeter} is multiplied by $k$,
and every \emph{area} is multiplied by $k^2$.
\end{formulabox}

\skillheading{Skill 1 --- Finding a missing side}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \sim \triangle DEF$ with $AB = 4$ cm,
$BC = 9$ cm and $DE = 6$ cm. Find $EF$.
\step{Want $EF$; I have the pair $AB$ with $DE$ to build the ratio and the side
$BC$ that matches $EF$ --- one proportion, first triangle on top on both sides.}
\step{$\dfrac{4}{6} = \dfrac{9}{EF} \;\Rightarrow\; EF = 9 \cdot \dfrac{6}{4}$}

$EF = \ans{13.5}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle ABC \sim \triangle DEF$ with $AB = 10$ cm,
$BC = 6$ cm and $DE = 25$ cm. Find $EF$.\\[2pt]
\rotatebox{180}{\footnotesize check: $EF = \ans{15}$ cm}
\end{tryitbox}

\skillheading{Skill 2 --- Finding the scale factor}

\begin{examplebox}
\textbf{Example.} Two similar triangles have matching sides of $4$ cm and
$18$ cm. Find the scale factor from the smaller to the larger.
\step{Want $k$ going small to large, and I have one matching pair --- so divide
the target length by the starting length.}
\step{$k = \dfrac{18}{4}$}

$k = \ans{4.5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Two similar triangles have matching sides of $7$ cm and
$28$ cm. Find the scale factor from the smaller to the larger.\\[2pt]
\rotatebox{180}{\footnotesize check: $k = \ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Perimeter scales like a side}

\begin{examplebox}
\textbf{Example.} $\triangle ABC \sim \triangle DEF$ with $AB = 5$ cm and
$DE = 8$ cm. The perimeter of $\triangle ABC$ is $20$ cm. Find the perimeter of
$\triangle DEF$.
\step{Want a perimeter, and a perimeter is just three lengths added --- so it
scales by $k$ itself, never by $k^2$.}
\step{$k = \dfrac{8}{5}$, so $P = 20 \cdot \dfrac{8}{5}$}

$P = \ans{32}$ cm
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} $\triangle ABC \sim \triangle DEF$ with $AB = 6$ cm,
$DE = 9$ cm, and the perimeter of $\triangle ABC$ equal to $22$ cm. Find the
perimeter of $\triangle DEF$.\\[2pt]
\rotatebox{180}{\footnotesize check: $P = \ans{33}$ cm}
\end{tryitbox}

\skillheading{Skill 4 --- Area scales like the square}

\begin{examplebox}
\textbf{Example.} Two similar triangles have matching sides $3$ cm and $7$ cm.
The smaller has area $18$ cm$^2$. Find the area of the larger.
\step{Want an area; a triangle stretches in two directions at once, so the
factor is applied twice --- multiply by $k^2$.}
\step{$k = \dfrac{7}{3}$, so $A = 18 \left( \dfrac{7}{3} \right)^2$}

$A = \ans{98}$ cm$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} Two similar triangles have matching sides $5$ cm and $8$ cm.
The smaller has area $25$ cm$^2$. Find the area of the larger.\\[2pt]
\rotatebox{180}{\footnotesize check: $A = \ans{64}$ cm$^2$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} going \emph{backwards} from areas, take the square root
first. If the areas are in the ratio $9:4$, the sides and perimeters are in the
ratio $3:2$, not $9:4$.
\end{watchoutbox}
