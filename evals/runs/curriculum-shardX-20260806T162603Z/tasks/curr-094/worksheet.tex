\wstitleblock{Area or Perimeter? Fix the Mix-Up}{Grade 3}{}

\noindent\textbf{Directions:}
\begin{itemize}
  \item \emph{Area} is the space a shape covers inside --- counted in
        \emph{square} units.
  \item \emph{Perimeter} is the distance around the edge --- counted in plain
        length units.
  \item Some students below mixed the two up. Find the mistakes, fix them, and
        write the right unit with every answer.
\end{itemize}

% ---------------------------------------------------------------- Problem 1
\problem{A rectangular sticker is 7 cm long and 4 cm wide. What is the
\emph{area} of the sticker?
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (7,4);
  \node[below, font=\small] at (3.5,0) {7 cm};
  \node[right, font=\small] at (7,2) {4 cm};
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{cm$^2$}}

% ---------------------------------------------------------------- Problem 2
\problem{A rectangular garden is 8 m long and 5 m wide. A fence goes all the
way around it. How many meters of fence is that --- the \emph{perimeter} of
the garden?
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (8,5);
  \node[below, font=\small] at (4,0) {8 m};
  \node[right, font=\small] at (8,2.5) {5 m};
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{m}}

% ---------------------------------------------------------------- Problem 3
\problem{A rug is 6 ft long and 3 ft wide.\par
(a) Circle the word that tells how much \emph{floor the rug covers}:
\quad \textbf{AREA} \qquad \textbf{PERIMETER}\par
(b) Now find that amount for this rug.
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (6,3);
  \node[below, font=\small] at (3,0) {6 ft};
  \node[right, font=\small] at (6,1.5) {3 ft};
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{ft$^2$}}

% ---------------------------------------------------------------- Problem 4
\problem{\textbf{Find and fix.} Mateo's poster is 9 cm long and 5 cm wide. He
writes: ``The area is 28 square centimeters, because $9+5+9+5 = 28$.''\par
(a) What is the real area of the poster? \par
(b) Name Mateo's error.
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (9,5);
  \node[below, font=\small] at (4.5,0) {9 cm};
  \node[right, font=\small] at (9,2.5) {5 cm};
\end{tikzpicture}
\end{center}
\par\vspace*{1.8cm}\answerline{cm$^2$}
\par\vspace*{1.8cm}}

% ---------------------------------------------------------------- Problem 5
\problem{Nia is gluing a ribbon all the way around the edge of a card that is

% ---------------------------------------------------------------- Problem 6
\problem{The rectangle below sits on a grid, and each little square is one
square unit. The rectangle is 5 units long and 3 units wide. What is its
\emph{area}?
\begin{center}
\begin{tikzpicture}[scale=0.55]
  \fill[blue!15] (0,0) rectangle (5,3);
  \draw[gray!55, thin] (1,0) -- (1,3);
  \draw[gray!55, thin] (2,0) -- (2,3);
  \draw[gray!55, thin] (3,0) -- (3,3);
  \draw[gray!55, thin] (4,0) -- (4,3);
  \draw[gray!55, thin] (0,1) -- (5,1);
  \draw[gray!55, thin] (0,2) -- (5,2);
  \draw[thick] (0,0) rectangle (5,3);
\end{tikzpicture}
\end{center}
\par\vspace*{2.6cm}\answerline{square units}}

% ---------------------------------------------------------------- Problem 7
\problem{\textbf{Find and fix.} Jaya's square garden has sides that are 6 m
long. To buy fencing for the whole edge, she works out $6 \times 6 = 36$ and
buys 36 m of fence.\par
(a) How many meters of fence does the garden actually need?\par
(b) Explain what Jaya mixed up.
\begin{center}
\begin{tikzpicture}[scale=0.45]
  \draw[thick] (0,0) rectangle (6,6);
  \node[below, font=\small] at (3,0) {6 m};
\end{tikzpicture}
\end{center}
\par\vspace*{1.8cm}\answerline{m}
\par\vspace*{1.8cm}}

% ---------------------------------------------------------------- Problem 8
\problem{\textbf{Challenge.} A play area is a rectangle 10 m long and 3 m wide.\par
(a) How many square meters of grass cover it? \par\vspace*{1.6cm}\answerline{m$^2$}\par
(b) How many meters of fence go around it? \par\vspace*{1.6cm}\answerline{m}\par
(c) Write $>$, $<$, or $=$ to compare the two numbers you found:
\quad grass number \underline{\hspace{0.9cm}} fence number \ansline
\begin{center}
\begin{tikzpicture}[scale=0.4]
  \draw[thick] (0,0) rectangle (10,3);
  \node[below, font=\small] at (5,0) {10 m};
  \node[right, font=\small] at (10,1.5) {3 m};
\end{tikzpicture}
\end{center}}
