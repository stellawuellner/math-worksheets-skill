\sstitleblock{Counting Detective: Missing and Misplaced Numbers}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Filling a Missing Number}

\begin{formulabox}
\textbf{Rule:} Use the neighbors! Read the number just \textbf{before} the
empty box and count on one more. Then check: your answer must come right
before the next number on the track.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A track shows 25, 26, \_\_, 28. What goes in the empty
box?
\step{Read the neighbor before the box (26) and count on one --- then check against the neighbor after.}
\step{One more than 26 is $\ans{27}$, and 27 comes right before 28. It fits both neighbors.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A track shows 61, 62, \_\_, 64. What goes in the empty
box?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{63}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Don't copy the neighbor! The empty box holds a
\emph{new} number --- one more than the number before it.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Spotting a Counting Mistake}

\begin{formulabox}
\textbf{Rule:} Whisper-count from the start of the row while pointing at each
number. The first number that does \textbf{not} match your whisper-count is
the mistake.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The count ``9, 10, 12'' skips a number. What is
missing?
\step{Whisper-count and match: ``9, 10, \ldots'' --- the count says a number belongs between 10 and 12.}
\step{One more than 10 is $\ans{11}$, so the count should read 9, 10, 11, 12.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The count ``28, 29, 31'' skips a number. What is
missing?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{30}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Putting Numbers Back in Order}

\begin{formulabox}
\textbf{Rule:} Counting order is least to greatest: the number you would say
\textbf{first} when counting is the \textbf{least}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Put the cards 14, 12, 13 in counting order.
\step{Whisper-count and grab each card as its number is said --- the card said first is least.}
\step{Counting ``12, 13, 14'' collects them in order: $\ans{12}$, $\ans{13}$, $\ans{14}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Put the cards 8, 6, 7 in counting order.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$, $\ans{7}$, $\ans{8}$}
\end{tryitbox}
