\wstitleblock{Modeling Loci and Intersections with Conic Sections}{Algebra 2}{}

\begin{center}
\small\textit{Building a circle from a distance condition, using the focus and directrix of a parabola, and solving conic intersections}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--3 are a warm-up, one on each idea. After that the three are mixed, so read each problem and decide which tool it wants before you start.
  \item For a locus problem, translate the sentence into a distance equation first, then simplify.
  \item For an intersection, substitute one equation into the other and report every point, not just the first.
\end{itemize}
\vspace{0.15cm}

\problem[5cm]{A point $P(x, y)$ moves so that its distance from $C(3, -2)$ is always $5$. Write the equation of its path in the expanded form $x^{2} + y^{2} + Dx + Ey + F = 0$.}

\problem[5cm]{A point is equidistant from the focus $(0, 3)$ and the directrix $y = -3$; its path is the parabola $x^{2} = 12y$. Find $y$ when $x = 6$.}

\problem[5cm]{Find the $x$-coordinates where the parabola $y = x^{2}$ meets the horizontal line $y = 4$.}

\problem[6cm]{A circle has equation $x^{2} + y^{2} - 6x + 4y - 12 = 0$. Complete the square in $x$ and in $y$, then state the radius.}

\problem[0pt]{A parabola has its vertex at the origin and its focus at $(0, 5)$, so its equation has the form $x^{2} = 4py$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find the value of $4p$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Find both $x$-values on the parabola when $y = 5$.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[7cm]{Find every intersection point of the parabola $y = x^{2} - 3$ and the line $y = 2x$. Give your answers as ordered pairs.}

\problem[0pt]{A point $P(x, y)$ moves so that its distance from $A(0, 0)$ is always twice its distance from $B(6, 0)$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Square both distances, set up $PA^{2} = 4\,PB^{2}$, and simplify to the form $x^{2} + y^{2} + Dx + Ey + F = 0$.
  \par\vspace*{4.4cm}\ansline
  \item[(b)] Complete the square to find the radius of the resulting circle.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[7.5cm]{Find every intersection point of the circle $x^{2} + y^{2} = 25$ and the line $y = x + 1$. Give your answers as ordered pairs.}

\problem[0pt]{A parabola is the set of points equidistant from the focus $(2, 0)$ and the directrix $x = -2$. Its equation is $y^{2} = 8x$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find both $y$-values on the parabola when $x = 2$.
  \par\vspace*{2.6cm}\ansline
  \item[(b)] Explain, using the equidistance definition rather than the equation, why $(2, 4)$ lies on this parabola.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[0pt]{A point $P(x, y)$ moves so that the sum of the squares of its distances from $(-3, 0)$ and $(3, 0)$ is always $50$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Set up $PA^{2} + PB^{2} = 50$ and simplify to the form $x^{2} + y^{2} + F = 0$.
  \par\vspace*{4.0cm}\ansline
  \item[(b)] Describe the locus: what shape is it, where is it centred, and how big is it? Justify each part from your equation.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[8.5cm]{Find every intersection point of the circle $x^{2} + y^{2} = 25$ and the parabola $y = x^{2} - 13$. Give your answers as ordered pairs. (There are four.)}

\problem[0pt]{The line $y = x + k$ meets the circle $x^{2} + y^{2} = 8$ at exactly one point.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Substitute the line into the circle, set the discriminant of the resulting quadratic to zero, and find both values of $k$.
  \par\vspace*{4.6cm}\ansline
  \item[(b)] Explain what ``exactly one point'' means about the quadratic's roots and what it means about the picture.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}
