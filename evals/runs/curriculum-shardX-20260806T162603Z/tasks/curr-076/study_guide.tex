\sstitleblock{Anchor Facts: Twos, Fives, and Tens}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Twos and doubling}

\begin{formulabox}
\textbf{Rule:}\quad A twos fact is a double: $2 \times n = n + n$.\par
Doubling a twos fact gives a fours fact: double $2 \times n$ to get $4 \times n$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $4 \times 7$ by doubling.
\step{Four groups is two groups doubled, so start from the twos anchor and double it.}
\step{$2 \times 7 = 14$, and doubling gives $14 + 14 = 28$.}

$\Rightarrow\quad\ans{28}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $4 \times 9$ by doubling $2 \times 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Fives and skip counting}

\begin{formulabox}
\textbf{Rule:}\quad Every fives fact lands on a skip-count number ending in 5 or 0.\par
For a six-of-something fact, take the fives fact and add \emph{one more group}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $5 \times 7$, then use it for $6 \times 7$.
\step{Skip count by fives seven times for the anchor, then one more 7 gives the sixth group.}
\step{$5 \times 7 = 35$, so $6 \times 7 = 35 + 7 = 42$.}

$\Rightarrow\quad\ans{35}$ and $\ans{42}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $5 \times 9$, then $6 \times 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{45}$ and $\ans{54}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Tens, plus or minus a group}

\begin{formulabox}
\textbf{Rule:}\quad $10 \times n$ is $n$ with a zero after it.\par
A nines fact is one group less than the tens fact: $10 \times n$ minus one $n$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $9 \times 6$ using the tens anchor.
\step{Nine groups is one group of 6 fewer than ten groups, so start from $10 \times 6$ and subtract one 6.}
\step{$10 \times 6 = 60$, and $60 - 6 = 54$.}

$\Rightarrow\quad\ans{54}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $9 \times 4$ using $10 \times 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{36}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} When you add or take away a group, the group is the \emph{other} number, not the anchor. For $9 \times 6$, take away one \textbf{6} (not one 9, and not 1).
\end{watchoutbox}
