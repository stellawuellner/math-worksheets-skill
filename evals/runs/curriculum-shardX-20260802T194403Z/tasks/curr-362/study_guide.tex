\sstitleblock{The Rational Root Theorem: Finding Every Zero}

\noindent\small The theorem does not hand you the zeros --- it hands you a
\emph{short list of suspects}. Testing, dividing, and finishing the leftover
quadratic is where the zeros actually come from.\normalsize

\skillheading{1. Build the list and test a candidate (verify-a-root)}
\begin{formulabox}
\textbf{Theorem:} if a polynomial with integer coefficients has a rational zero
$\frac{p}{q}$ in lowest terms, then $p$ divides the \textbf{constant term} and
$q$ divides the \textbf{leading coefficient}.\\[3pt]
Candidates $= \pm\dfrac{\text{divisors of the constant}}{\text{divisors of the
leading coefficient}}$. \quad A candidate is a zero exactly when substituting it
gives $0$ (equivalently, synthetic division leaves remainder $0$).
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Is $x = 3$ a zero of $f(x) = x^3 - 2x^2 - 5x + 6$?
\step{The constant is $6$ and the leading coefficient is $1$, so $3$ is on the
candidate list --- test it by substitution.}
\step{$f(3) = 27 - 18 - 15 + 6$}
$\ans{f(3) = 0}$, so $x = 3$ is a zero and $(x-3)$ is a factor.
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Is $x = -2$ a zero of $x^3 + x^2 - 4x - 4$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{f(-2) = 0}$ --- yes}
\end{tryitbox}

\skillheading{2. Find all rational zeros (all-rational-zeros)}
\begin{formulabox}
\textbf{Routine:} list candidates $\to$ test until the remainder is $0$ $\to$
divide that factor out $\to$ factor or solve what is left.\\[3pt]
A leading coefficient other than $1$ puts fractions on the list: for
$2x^3 + \cdots + 3$ the candidates are $\pm1, \pm3, \pm\frac{1}{2},
\pm\frac{3}{2}$. Try small integers first --- they are usually the ones that
work.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad All zeros of $2x^3 + 3x^2 - 8x + 3$.
\step{Leading coefficient $2$ and constant $3$, so test the short integers on
the list before the halves.}
\step{$x=1$ gives $2+3-8+3=0$; dividing leaves $2x^2+5x-3 = (2x-1)(x+3)$.}
$\ans{x = 1,\ x = \tfrac{1}{2},\ x = -3}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find all zeros of $3x^3 - 4x^2 - 5x + 2$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 2,\ x = \tfrac{1}{3},\ x = -1}$}
\end{tryitbox}

\skillheading{3. Factor completely (factor-completely)}
\begin{formulabox}
\textbf{Link:} a zero $x = c$ means the factor $(x - c)$; a zero
$x = \frac{p}{q}$ means the integer factor $(qx - p)$.\\[3pt]
Keep dividing until what remains is a quadratic you can factor or a quadratic
with no real zeros. \emph{Completely} means no factor can be broken down
further over the integers.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Factor $x^3 - 6x^2 + 11x - 6$.
\step{Constant $-6$ and leading coefficient $1$: test the divisors of $6$ and
divide out the first one that works.}
\step{$x=1$ is a zero; division leaves $x^2 - 5x + 6 = (x-2)(x-3)$.}
$\ans{(x-1)(x-2)(x-3)}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Factor $x^3 + 3x^2 - x - 3$ completely.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(x+3)(x-1)(x+1)}$}
\end{tryitbox}

\skillheading{4. When the leftovers are not rational (mixed-zeros)}
\begin{formulabox}
\textbf{Key idea:} the theorem finds only \emph{rational} zeros. After dividing
one out, finish the quadratic with the quadratic formula --- that is where
irrational zeros ($a \pm \sqrt{b}$) and non-real zeros ($a \pm bi$) come from.
\\[3pt]
A degree-$n$ polynomial has $n$ zeros counted with multiplicity, so keep going
until you have $n$ of them.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad All zeros of $x^3 - 5x^2 + x + 7$.
\step{Only $\pm1, \pm7$ are candidates, so at most one zero will come from the
list --- expect to finish with the quadratic formula.}
\step{$x=-1$ works; division leaves $x^2 - 6x + 7$, so
$x = \dfrac{6 \pm \sqrt{36-28}}{2} = 3 \pm \sqrt{2}$.}
$\ans{x = -1,\ x = 3 \pm \sqrt{2}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Find all zeros of $x^3 - 2x^2 + 9x - 18$, including
non-real ones.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 2,\ x = 3i,\ x = -3i}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} An empty candidate list result means ``no \emph{rational}
zero,'' never ``no zero.'' Put $p$ over the constant's divisors and $q$ over the
leading coefficient's --- reversing them is the most common slip. And a zero of
$\frac{1}{2}$ pairs with the factor $(2x-1)$.
\end{watchoutbox}
