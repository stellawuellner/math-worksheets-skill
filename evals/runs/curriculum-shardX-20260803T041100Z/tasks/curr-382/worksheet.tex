\wstitleblock{Finding and Verifying Inverse Functions}{Algebra 2 \quad Composition, Inverses and Transformations}{}

\begin{center}
\textit{Finding an inverse by solving for the variable, verifying a pair by
composition, and using inverse values and domains}
\end{center}

\noindent\textbf{Directions.} Show the work, not just the result.
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item \textbf{To find an inverse:} write $y = f(x)$, swap $x$ and $y$, solve
the new equation for $y$, and rename the result $f^{-1}(x)$.
\item \textbf{To verify a pair:} show \emph{both} $f(g(x)) = x$ and
$g(f(x)) = x$. One direction alone is not enough.
\item $f^{-1}$ means the inverse function, never $\dfrac{1}{f}$.
\end{itemize}

\problem[5cm]{Find the inverse of $f(x) = x + 7$, then check your answer by
substituting it into $f$.}

\problem[5cm]{Find the inverse of $f(x) = 5x$, then check it by composing
$f$ with the function you found.}

\problem[5cm]{Find $f^{-1}(x)$ for $f(x) = 3x - 8$. Show the swap-and-solve
steps, and verify that $f(f^{-1}(x)) = x$.}

\problem[5cm]{Let $f(x) = 2x + 9$ and $g(x) = \dfrac{x-9}{2}$. Verify that
these two are inverses by computing \emph{both} $f(g(x))$ and $g(f(x))$ and
simplifying each one.}

\problem[5cm]{For $f(x) = 4x - 1$, find $f^{-1}(11)$. Do it by first writing a
formula for $f^{-1}(x)$, then evaluating that formula at $11$.}

\problem[5cm]{Find the inverse of $f(x) = \dfrac{x-4}{5}$. Check your answer by
simplifying $f^{-1}(f(x))$.}

\problem[5cm]{Find the inverse of $f(x) = \sqrt{x-2}$, whose domain is
$x \ge 2$. What is the domain of the inverse, and why is it not all real
numbers?}

\problem[5cm]{Sofia claims $f(x) = \dfrac{x+6}{2}$ and $g(x) = 2x - 6$ are
inverses. Test the claim by simplifying both $f(g(x))$ and $g(f(x))$, and say
whether she is right.}

\problem[5cm]{Let $f(x) = x^2$ with the domain restricted to $x \ge 0$. Write
$f^{-1}(x)$ and use it to find $f^{-1}(49)$. Why does the restriction matter
here?}

\problem[6cm]{Find the inverse of the rational function
$f(x) = \dfrac{2x+1}{x-3}$. Swap the variables, then collect every $y$ term on
one side and factor before dividing.}

\problem[5cm]{Let $f(x) = 3x + 5$. Find $f^{-1}(-4)$ two different ways: once
from a formula for $f^{-1}(x)$, and once by solving $f(x) = -4$ directly. The
two answers must agree.}

\problem[6cm]{Let $f(x) = x^2 - 4$.

\textbf{(a)} Explain in a sentence why $f$ has no inverse function on all of
the real numbers. \quad \textbf{(b)} Restrict the domain to $x \ge 0$, write
$f^{-1}(x)$ for that restricted function, and use it to find $f^{-1}(5)$.}
