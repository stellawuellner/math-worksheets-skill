\wstitleblock{Percent Applications}{Grade 6--7 \quad Ratios and Proportional Relationships}{}

\begin{center}\small\textit{Finding the percent rate, finding the whole, and percent increase or decrease}\end{center}

\noindent\small\textbf{Directions:} Every problem below gives you two of the three numbers
in the same relationship and asks for the third. Set up the proportion first, then solve.
Show your work; round only where the problem says to.\normalsize

\vspace{0.25cm}
\noindent\fbox{\begin{minipage}{0.965\linewidth}
\small\textbf{The pattern to copy}\par\vspace{2pt}
\[ \frac{\text{part}}{\text{whole}} \;=\; \frac{\text{percent}}{100} \]
\textbf{Worked model (rate).} 27 of 45 tickets were sold. What percent were sold?\par
$\dfrac{27}{45} = \dfrac{p}{100} \Rightarrow 45p = 2700 \Rightarrow p = 60$, so 60\% were sold.\par\vspace{2pt}
\textbf{Worked model (whole).} 27 tickets is 45\% of the printed run. How many were printed?\par
$\dfrac{27}{n} = \dfrac{45}{100} \Rightarrow 45n = 2700 \Rightarrow n = 60$, so 60 were printed.\par\vspace{2pt}
\textit{Same three numbers, different unknown --- read carefully which one is missing.}
\end{minipage}}

\problem[4.5cm]{In one section of a theater, 18 of the 60 seats are filled.
What \textbf{percent} of the seats are filled?}

\problem[4.5cm]{Dev answered 21 of the 25 questions on a quiz correctly.
What \textbf{percent} did he answer correctly?}

\problem[5cm]{Exactly 40\% of the hiking club walked to the trailhead. That was
12 members. \textbf{How many members} are in the club?}

\problem[4.5cm]{A soccer player made 36 of her 45 penalty attempts.
What \textbf{percent} did she make?}

\problem[5cm]{A jacket was marked down by 18 dollars, and that discount was 15\% of the
original price. What was the \textbf{original price}, in dollars?}

\problem[5cm]{Of the 250 people surveyed, 45 chose blue as their favorite color.
What \textbf{percent} chose blue?}

\problem[5cm]{A quality inspector found 6 cracked tiles in a shipment, which was 3\% of the
shipment. \textbf{How many tiles} were in the shipment?}

\problem[5.5cm]{A bakery's weekend sales rose from 250 loaves to 300 loaves.
By what \textbf{percent} did sales increase?}

\problem[5.5cm]{A purchase of 80 dollars came with 5.40 dollars of sales tax.
What is the \textbf{tax rate}, as a percent?}

\problem[5cm]{In a school election, 155 students voted yes, and that was 62\% of everyone
who voted. \textbf{How many students} voted?}

\problem[5.5cm]{A bike that used to cost 96 dollars now costs 72 dollars.
By what \textbf{percent} did the price decrease?}

\problem[6.5cm]{A gym's membership grew by 20\% this year and now stands at 168 members.
\textbf{How many members} did the gym have before the increase?\\[2pt]
\small(Careful: 168 is not the part that was added --- it is the new whole, which is
120\% of last year's membership.)\normalsize}
