\wstitleblock{Shapes and Their Names --- Turning a Shape Does Not Change It}{Kindergarten / Grade 1}{}

\begin{center}\small\textit{Naming circles, triangles, squares, rectangles, and hexagons no
matter which way they are turned}\end{center}

\noindent\small\textbf{For the grown-up helping:}
\begin{itemize}[label={$\bullet$}, leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Read each question aloud. The child does the looking, counting and colouring.
  \item Every shape on this sheet is drawn \emph{turned}. That is the whole idea: a
        triangle standing on its point is still a triangle, and a square balanced on a
        corner is still a square.
  \item When the child is unsure, ask ``how many straight sides?'' and ``how many
        corners?'' and let them touch each one as they count.
  \item Coloured pencils help. So does turning the page around.
\end{itemize}
\normalsize
