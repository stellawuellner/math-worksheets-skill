\wstitleblock{Decimal Multiplication \& Division Workshop}{Grade 5}{}
\begin{center}
\textit{Powers of ten, multiplying decimals, and dividing decimals}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item Use place-value reasoning: think about \emph{tenths} and
        \emph{hundredths}, not just digits.
  \item Show your work in the space under each problem, and write units where
        the problem asks for them.
\end{itemize}

\vspace{0.2cm}

\problem[3cm]{Multiply: \quad $3.6 \times 100$}

\problem[3cm]{Divide: \quad $45.2 \div 10$}

\problem[4cm]{Multiply: \quad $0.4 \times 0.6$ \quad
(Hint: tenths times tenths make \emph{hundredths}.)}

\problem{A notebook costs $2.35$ dollars. Amara buys $4$ notebooks.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] \emph{Estimate} the total cost by rounding the price to the
        nearest whole dollar first.\par\vspace*{2cm}\ansline
  \item[(b)] Now find the \emph{exact} total cost in dollars.
        \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[4.5cm]{Divide: \quad $7.2 \div 0.9$}

\problem{A watering can holds $2.5$ liters. Jonah fills it completely $6$
times to water the class garden. How many liters of water does he use in all?
\par\vspace*{4.5cm}\answerline{L}}

\problem{A ribbon is $4.5$ meters long. It is cut into pieces that are each
$0.5$ meter long. How many pieces are there?
\par\vspace*{4.5cm}\ansline}

\problem{One box of copy paper weighs $2.4$ kilograms. A pallet holds $100$
boxes. How much does a full pallet of boxes weigh?
\par\vspace*{4.5cm}\answerline{kg}}

\problem{Three friends share a $7.35$-dollar pack of trading cards equally.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] How many dollars does each friend pay?\par\vspace*{2.6cm}\ansline
  \item[(b)] Write $<$ or $>$: \quad each friend's share
        $\;\square\;$ $2.50$ dollars\par\vspace*{1.5cm}\ansline
\end{itemize}}

\problem{\textbf{Challenge.} Milk costs $1.2$ dollars per liter. Priya buys
$2.5$ liters and pays with a $5$-dollar bill.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] How many dollars does the milk cost?\par\vspace*{2.6cm}\ansline
  \item[(b)] How much change does Priya get?\par\vspace*{2.2cm}\ansline
\end{itemize}}
