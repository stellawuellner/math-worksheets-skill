\sstitleblock{The Law of Cosines: SAS and SSS Triangles}

\noindent\small The law of sines needs a side and the angle \emph{opposite} it.
SAS and SSS never hand you that pair, which is exactly when the law of cosines
is the only way in.\normalsize

\skillheading{1. SAS: two sides and the angle between them}
\begin{formulabox}
\textbf{Formula:} $c^{2} = a^{2} + b^{2} - 2ab\cos C$, where $C$ is the angle
\emph{between} the two known sides and $c$ is the side opposite it. The letters
rotate: for $a$ use $a^{2} = b^{2} + c^{2} - 2bc\cos A$.\\[3pt]
\textbf{Sign check:} if $C$ is obtuse, $\cos C$ is negative and the formula
\emph{adds}, so the new side is longer than both givens. If $C = 90^\circ$ the
term vanishes and it is the Pythagorean theorem.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $a = 6$, $b = 10$, $C = 60^\circ$. Find $c$.
\step{The known angle sits between the two known sides, so there is no
side-opposite-angle pair and the law of cosines is the only starting point.}
\step{$c^{2} = 36 + 100 - 2(6)(10)\cos 60^\circ = 136 - 120(0.5) = 76$, and
$c = \sqrt{76}$.}

$\ans{c \approx 8.72}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $a = 8$, $b = 5$, $C = 75^\circ$. Find $c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c \approx 8.26}$}
\end{tryitbox}

\skillheading{2. SSS: three sides, find an angle}
\begin{formulabox}
\textbf{Formula, rearranged for the angle:}
$\cos C = \dfrac{a^{2} + b^{2} - c^{2}}{2ab}$. The side opposite the angle you
want is the one that gets \emph{subtracted}; the other two are added and their
product is doubled underneath.\\[3pt]
\textbf{Why this beats the law of sines:} $\cos^{-1}$ returns angles from
$0^\circ$ to $180^\circ$, so an obtuse angle comes out obtuse. A negative
cosine means obtuse; a positive one means acute. Start with the largest side to
find the largest angle and the rest cannot be obtuse.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad $a = 4$, $b = 6$, $c = 8$. Find $C$.
\step{Three sides and no angles, so no ratio of the law of sines can be
written; rearrange the law of cosines for the angle opposite $c$.}
\step{$\cos C = \dfrac{16 + 36 - 64}{2(4)(6)} = \dfrac{-12}{48} = -0.25$, and
the negative value says $C$ is obtuse.}

$\ans{C \approx 104.48^\circ}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad $a = 7$, $b = 9$, $c = 12$. Find $C$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{C \approx 96.38^\circ}$}
\end{tryitbox}

\skillheading{3. Bearings and navigation}
\begin{formulabox}
\textbf{Turning a story into SAS:} draw the two legs from the point they share.
The angle at that point is the angle between the courses --- for two bearings
measured from the same place it is their \emph{difference}, and for a turn along
a path it is the interior angle the problem states. Then the distance apart is
the third side.\\[3pt]
Keep the units of the two legs the same, and give the answer in those units.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Two hikers leave a hut, one walking $3$ km and the other
$5$ km, with $120^\circ$ between their paths. How far apart are they?
\step{The two paths and the line between the hikers form a triangle whose known
angle lies between the two known sides --- SAS.}
\step{$c^{2} = 9 + 25 - 2(3)(5)\cos 120^\circ = 34 - 30(-0.5) = 49$.}

$\ans{c = 7\ \text{km}}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Two boats leave a jetty, one going $6$ km and the other
$10$ km, with $100^\circ$ between their courses.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c \approx 12.52\ \text{km}}$}
\end{tryitbox}
