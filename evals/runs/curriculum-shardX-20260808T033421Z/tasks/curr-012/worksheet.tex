\wstitleblock{Comparing Numerals 1 to 10}{Kindergarten--Grade 1}{}

\begin{center}
\small greater than, less than and equal to; one more and one less; putting three numerals in order
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item $<$ means \emph{is less than}. \quad $>$ means \emph{is greater than}. \quad $=$ means \emph{is equal to}.
  \item The wide open end of the symbol always faces the bigger number.
  \item You may count out loud, draw dots, or use a number line to help you.
\end{itemize}
