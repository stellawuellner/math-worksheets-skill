\sstitleblock{Solving Exponential Equations with Common Bases}

\skillheading{Skill 1 --- Equations that already share a base}

\begin{formulabox}
\textbf{One-to-one property.} If $b > 0$ and $b \neq 1$, then
$b^{M} = b^{N}$ \ \ exactly when \ \ $M = N$.\par
So once both sides are powers of the same base, the exponents become an ordinary equation.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2^{\,3x-1} = 2^{\,x+7}$.
\step{Both sides are already powers of $2$, so the one-to-one property turns this into a linear equation in the exponents.}
\step{$3x - 1 = x + 7$, so $2x = 8$.}

$\Rightarrow\quad \ans{x = 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $5^{\,4x} = 5^{\,x+9}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$}
\end{tryitbox}

\skillheading{Skill 2 --- Rewriting both sides to one common base}

\begin{formulabox}
\textbf{Tools for matching bases:} \ $(b^{m})^{n} = b^{mn}$, \quad
$\dfrac{1}{b^{n}} = b^{-n}$, \quad $\sqrt[n]{b} = b^{1/n}$.\par
Pick the \emph{smallest} base that generates both sides ($4$, $8$, $16$, $32$ are all powers of $2$; $9$, $27$, $81$ are all powers of $3$).
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4^{\,x+1} = 32$.
\step{Neither side is a power of the other, but $4$ and $32$ are both powers of $2$ --- rewrite with base $2$ first.}
\step{$\left(2^{2}\right)^{x+1} = 2^{5}$, so $2x + 2 = 5$.}

$\Rightarrow\quad \ans{x = \tfrac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $9^{\,x} = 243$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{5}{2}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\left(\frac{1}{2}\right)^{x}$ is $2^{-x}$, not $2^{x}$. A reciprocal base flips the sign of the whole exponent, brackets and all.
\end{watchoutbox}

\skillheading{Skill 3 --- Reading an exponential model from a table}

\begin{formulabox}
\textbf{Constant ratio $\Rightarrow$ exponential.} If each step in $x$ multiplies $f(x)$ by the same number $r$, then
$f(x) = f(0) \cdot r^{\,x}$.\par
To solve $f(0)\cdot r^{\,x} = V$, divide by $f(0)$ first, then match bases.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table gives $f(0)=7$, $f(1)=14$, $f(2)=28$, $f(3)=56$. Solve $f(x) = 448$.
\step{Every step doubles the value, so the ratio is $2$ and the model is $f(x) = 7 \cdot 2^{\,x}$ --- divide by the starting value to isolate the power.}
\step{$2^{\,x} = \dfrac{448}{7} = 64 = 2^{6}$.}

$\Rightarrow\quad \ans{x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $g(0)=5$, $g(1)=15$, $g(2)=45$, $g(3)=135$. Solve $g(x) = 405$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}
