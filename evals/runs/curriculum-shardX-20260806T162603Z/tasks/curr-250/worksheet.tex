\wstitleblock{Making and Qualifying Predictions from Scatter Plots}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Reading a scatter plot, predicting with its trend line, and judging when a prediction can be trusted}
\end{center}

\vspace{0.05cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item The three displays below are used by every problem. Each problem names the display it needs.
  \item Where a trend line is given as an equation, use the equation, not a ruler.
  \item Every prediction has to be judged as well as calculated: say when a number the line produces
        is one you would actually rely on.
\end{itemize}

\problem[4.5cm]{\textbf{Display A.} Any prediction you make from this display rests on the students
who were actually measured. How many students are plotted?}

\problem[4.5cm]{\textbf{Display A.} One student studied for $35$ minutes. Read that student's actual
quiz score off the plot --- you will compare it with a prediction later.}

\problem[5cm]{\textbf{Display A.} Use the trend line $y = 0.6x + 48$ to predict the quiz score of a
student who studies for $45$ minutes.}

\problem[5cm]{\textbf{Display B.} Use the trend line $y = 9.5 - 0.5x$ to predict the sleep of a
student who watches $5.5$ hours of television a day.
\par\vspace*{0.1cm}\answerline{hours}}

\problem[0.4cm]{\textbf{Display A.} A parent asks what score $200$ minutes of study would produce.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] Work out what the trend line predicts for $x = 200$. \ansline
  \par\vspace*{1.6cm}
  \item[(b)] Give two reasons this prediction should not be trusted. \ansline
  \par\vspace*{1.8cm}
\end{itemize}}

\problem[0.4cm]{\textbf{Display B.}
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] How much sleep did the student who watched $6$ hours of television get?
             \par\vspace*{0.1cm}\answerline{hours}
  \par\vspace*{1.4cm}
  \item[(b)] Describe the direction of the association, and say what that direction means for any
             prediction you make from this display. \ansline
  \par\vspace*{1.8cm}
\end{itemize}}

\problem[0.4cm]{\textbf{Display A.} One student studied for $60$ minutes and scored $85$.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] What score does the trend line predict for $60$ minutes? \ansline
  \par\vspace*{1.6cm}
  \item[(b)] By how much does the actual score exceed the prediction? \ansline
  \par\vspace*{1.6cm}
\end{itemize}}

\problem[5cm]{\textbf{Display B.} A prediction is easier to judge if you know what a typical value
looks like. Find the mean sleep time of the eight students.
\par\vspace*{0.1cm}\answerline{hours}}

\problem[0.4cm]{\textbf{Display C.}
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] What is the lowest spelling score in the display? \ansline
  \par\vspace*{1.4cm}
  \item[(b)] A classmate wants to predict a student's spelling score from their shoe size. Explain
             whether this display supports such a prediction. \ansline
  \par\vspace*{1.8cm}
\end{itemize}}

\problem[6cm]{\textbf{Display A.} A student wants to score $96$ on the next quiz. Using the trend
line $y = 0.6x + 48$, how many minutes of study does the model call for?
\par\vspace*{0.1cm}\answerline{min}}

\problem[0.4cm]{\textbf{Display B.} Somebody claims to watch $20$ hours of television a day.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] What does the trend line $y = 9.5 - 0.5x$ predict for their sleep?
             \par\vspace*{0.1cm}\answerline{hours}
  \par\vspace*{1.4cm}
  \item[(b)] Explain why that number cannot be treated as a real prediction. \ansline
  \par\vspace*{1.8cm}
\end{itemize}}

\problem[0.4cm]{\textbf{Display A.} A school newsletter prints the headline ``Studying causes higher
quiz scores'' beside this display.
\par\smallskip
\begin{itemize}[leftmargin=\subind, itemsep=2pt, topsep=2pt]
  \item[(a)] Use the trend line to predict the score of a student who studies $25$ minutes. \ansline
  \par\vspace*{1.5cm}
  \item[(b)] Two of the plotted students studied $30$ and $35$ minutes and scored $65$ and $70$.
             Find the difference between those two scores. \ansline
  \par\vspace*{1.5cm}
  \item[(c)] Explain whether Display A justifies the headline, and give one other explanation that
             would fit the same points. \ansline
  \par\vspace*{2cm}
\end{itemize}}
