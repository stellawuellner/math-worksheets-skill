\wstitleblock{Complex Number Operations}{Algebra 2}{}

\begin{center}
{\small\textit{Adding, subtracting, and multiplying complex numbers}}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item Write every final answer in standard form $a + bi$, with $a$ and $b$ real numbers.
  \item Whenever a term $i^2$ appears, replace it with $-1$ before you combine like terms.
  \item If an answer is purely real or purely imaginary, say so by writing it with the missing part as $0$ or by writing the single term (for example $29$ or $6i$).
  \item Show the multiplying-out step, not just the final line.
\end{itemize}

\vspace{0.2cm}

\problem[4.5cm]{Simplify: \quad $(8 + 3i) + (2 + 9i)$}

\problem[4.5cm]{Simplify: \quad $(7 - 4i) - (2 + 6i)$}

\problem[5cm]{Simplify: \quad $(-3 + 5i) + (9 - 11i) - (4 + 2i)$}

\problem[5cm]{Multiply and write the result in standard form: \quad $3i(4 - 5i)$}

\problem[5.5cm]{Multiply and write the result in standard form: \quad $(2 + 3i)(4 - i)$}

\problem[5cm]{Simplify: \quad $(10 - 2i) - (-3 + 7i)$}

\problem{Multiply: \quad $(5 - 2i)(5 + 2i)$
  \par\vspace*{4.2cm}\ansline
  \par\vspace{0.15cm}
  The two factors are \emph{conjugates}. Describe what kind of number the product
  turned out to be, and why the $i$ terms cancelled.
  \par\vspace*{2.6cm}\ansline}

\problem[5.5cm]{Expand and simplify: \quad $(3 + 4i)^2$}

\problem[5.5cm]{Multiply and write the result in standard form: \quad $(-6 + i)(2 - 5i)$}

\problem[5.5cm]{Simplify: \quad $2(4 - 3i) - 3(1 + 5i)$}

\problem[6cm]{Simplify: \quad $(2 - 3i)^2 + (2 + 3i)^2$}

\problem{Let $z = 3 - 2i$ and $w = -1 + 5i$.
\begin{itemize}[leftmargin=1.1cm, itemsep=2pt, topsep=4pt]
  \item[(a)] Compute $zw - 2z$ and write the result in standard form.
    \par\vspace*{5.5cm}\ansline
  \item[(b)] In one or two sentences, explain why the product of two numbers that each
    carry an imaginary part can end up with a real part that neither factor had.
    Point to the step where $i^2$ is replaced by $-1$.
    \par\vspace*{3.2cm}\ansline
\end{itemize}}
