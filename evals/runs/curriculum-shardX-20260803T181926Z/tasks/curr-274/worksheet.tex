\wstitleblock{Arithmetic Sequences as Linear Functions --- Finding and Fixing Off-by-One Index Errors}{Algebra 1}{}

\noindent\small Almost every wrong answer in this unit comes from one habit:
counting \emph{terms} when the rule counts \emph{steps}.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The explicit rule is $a_n = a_1 + (n-1)d$. The first term is already
        there before you move, so reaching term $n$ takes $n-1$ steps.
  \item The same sequence written as a line is $a_n = dn + (a_1 - d)$. Its
        constant is the value at $n = 0$, one step \emph{before} the sequence
        starts --- it is not the first term.
  \item From term $p$ to term $q$ inclusive there are $q - p + 1$ terms, not
        $q - p$. Both ends are terms.
  \item \textbf{The $n=1$ test:} put $n = 1$ into any rule you write. A correct
        rule gives back the first term.
\end{itemize}
In each problem below: name the error in words first, then show corrected work.
\normalsize

\vspace{0.2cm}

\problem[5cm]{\textbf{Find and fix.} An arithmetic sequence has first term
$a_1 = 6$ and common difference $d = 5$. Rosa wants the $10$th term and writes
$a_{10} = 6 + 10(5) = 56$.
\par\smallskip
Say in one sentence why Rosa takes one step too many, then compute the correct
$10$th term.}

\problem[5cm]{\textbf{Find and fix.} A sequence is described by the linear rule
$a_n = 4n + 7$. Theo says ``the first term is $7$, because that is the constant.''
\par\smallskip
\textbf{(a)} Say which term of the sequence the constant $7$ really is.
\par
\textbf{(b)} Compute the true first term $a_1$.}

\problem[5cm]{\textbf{Find and fix.} A list runs from the $7$th term of a
sequence through the $23$rd term, with no terms skipped. Milo counts the terms
by subtracting and reports $16$ terms.
\par\smallskip
Explain what Milo's subtraction leaves out, then give the correct number of
terms.}

\problem[5.5cm]{\textbf{Find and fix.} An arithmetic sequence starts at $5$ and
increases by $6$ each step. Asking which term equals $89$, a student writes
$5 + 6n = 89$ and reports term number $14$.
\par\smallskip
\textbf{(a)} Write the equation the explicit rule actually gives.
\par
\textbf{(b)} Solve it, and state which term of the sequence equals $89$.}

\problem[6cm]{\textbf{Find and fix.} Consider the arithmetic sequence with first
term $4$ and common difference $5$, summed over its first $12$ terms.
\par\smallskip
Priya writes $\;S_{12} = \dfrac{12}{2}\bigl(4 + [\,4 + 5 \cdot 12\,]\bigr) = 408$.
\par\smallskip
Her sum formula is right and her last term is wrong. Name the slip, then compute
the correct sum.}

\problem[6cm]{\textbf{Find and fix.} The sequence $3$, $10$, $17$, $24$, \ldots\
is to be written in the linear form $a_n = dn + b$. Dana writes $a_n = 7n + 3$
and reports $a_{20} = 143$.
\par\smallskip
\textbf{(a)} Test Dana's rule at $n = 1$ and say what it returns.
\par
\textbf{(b)} Write the correct linear rule and compute the correct $20$th term.}

\problem[7cm]{\textbf{Find and fix.} A sequence has $a_n = 3 + 4(n-1)$. Find the
sum of the terms from $a_{5}$ through $a_{20}$, inclusive.
\par\smallskip
A classmate computes the two end terms correctly, then multiplies their average
by $20 - 5$ and gets $735$.
\par\smallskip
\textbf{(a)} State how many terms the sum really contains.
\par
\textbf{(b)} Compute the correct sum.}

\problem[7.5cm]{\textbf{Explain.}
\par\smallskip
\textbf{(a)} Explain why the explicit rule multiplies the common difference by
$n - 1$ and not by $n$. Refer to what the first term costs in steps.
\par
\textbf{(b)} The same sequence written as a linear function has a constant that
is \emph{not} the first term. Explain what that constant is the value of, and why
a line through sequence points behaves that way.
\par
\textbf{(c)} An off-by-one index error usually produces a clean whole number, so
nothing looks wrong. Describe two checks a student can run that would catch it
anyway.
