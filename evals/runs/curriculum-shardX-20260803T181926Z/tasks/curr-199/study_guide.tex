\skillheading{Skill 1 --- Choosing the right denominator}

\begin{formulabox}
\textbf{Rule:} \quad
$P(\text{event}) = \dfrac{\text{number of outcomes you want}}{\text{number of
outcomes altogether}}$
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The denominator counts \emph{everything that could happen}, including
        the outcomes you want. Never the leftovers on their own.
  \item If something is removed and not replaced, the total changes too --- so
        re-check both numbers, not just the one the story mentions.
  \item Check: all the separate probabilities for one draw must add to $1$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A drawer holds $7$ red socks and $13$ blue socks. One is
taken without looking. Find $P(\text{red})$.
\step{Settle ``out of what?'' first: one sock is taken from the whole drawer, so
the denominator counts every sock, red ones included.}
\step{Altogether there are $7 + 13 = 20$ socks, and $7$ of them are red, so
$P(\text{red}) = \ans{\dfrac{7}{20}}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A jar holds $9$ mints and $6$ toffees. One sweet is taken
without looking. Find $P(\text{toffee})$ in lowest terms.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{2}{5}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\dfrac{7}{13}$ compares red socks with blue socks. That
is a \emph{ratio}, not a probability, and it is always too big.
\end{watchoutbox}

\skillheading{Skill 2 --- Deciding whether a sample is fair}

\begin{formulabox}
\textbf{The test:} could \emph{every} member of the group you want to describe
have been picked?
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item If where or how you asked rules some people out, the sample is
        \textbf{biased} --- and a bigger biased sample is still biased.
  \item Say which way it is wrong: asking at the bike rack over-states cycling.
  \item Scaling up a \emph{fair} sample: \quad
        estimate $=$ population $\times \dfrac{\text{count in sample}}{\text{size
        of sample}}$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $25$ students are drawn at random from a school roll of
$400$, and $10$ of them walk to school. Estimate how many of the $400$ walk.
\step{Names were drawn at random from the roll, so every student could have been
picked --- the sample passes the fairness test, and only then is it worth scaling
up.}
\step{The sample fraction is $\dfrac{10}{25}$, so the estimate is
$400 \times \dfrac{10}{25} = \ans{160}$ students.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $40$ students are drawn at random from a roll of $600$,
and $12$ of them play an instrument. Estimate how many of the $600$ do.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{180}$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding a median from a sample}

\begin{formulabox}
\textbf{Rule:} sort the values smallest to largest \emph{first}, then take the
middle one.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item An odd number of values has one middle value.
  \item An even number of values has two: the median is halfway between them.
  \item The median describes only the group you sampled. A median from a biased
        sample is a correct number about the wrong group.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find the median of $9,\ 4,\ 12,\ 7,\ 5,\ 10$.
\step{A median is a position in the sorted order, so nothing can be read off the
list as written --- sort it before looking for the middle.}
\step{Sorted: $4,\ 5,\ 7,\ 9,\ 10,\ 12$. There are six values, so take the mean
of the third and fourth: $\dfrac{7+9}{2} = \ans{8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find the median of $6,\ 2,\ 9,\ 3,\ 8,\ 4,\ 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$}
\end{tryitbox}
