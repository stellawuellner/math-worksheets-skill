\wstitleblock{Analyzing Transformations with Multiple Parameters}{Precalculus}{}

\begin{center}
\small\textit{Reading the parameters, evaluating a transformed function, and expanding the transformed rule}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Every rule on this sheet has the shape $g(x) = a\,f\bigl(b(x - h)\bigr) + k$. Work from the inside out: $b$ and $h$ act on $x$, and $a$ and $k$ act on the output.
  \item Where a table is given, use it as a picture of the parent function --- but write the algebra too.
  \item Give exact values, not decimals.
\end{itemize}
\vspace{0.15cm}

\problem[5cm]{Let $f(x) = x^{2}$ and $g(x) = 2f(x - 3) + 1$. Find $g(5)$.}

\problem[0pt]{Let $f(x) = x^{2}$ and $g(x) = f(x + 4) - 7$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $g(-2)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Describe, in order, the transformations that take the graph of $y = f(x)$ to the graph of $y = g(x)$. Name the direction of each shift.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = x^{2}$ and $g(x) = -\dfrac{1}{2}f(x + 4) - 3$. Write $g(x)$ in expanded form $ax^{2} + bx + c$.}

\problem[0pt]{Let $f(x) = x^{2} + 3$. Its values are shown below.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $-2$ & $-1$ & $0$ & $1$ & $2$ \\ \hline
$f(x)$ & $7$ & $4$ & $3$ & $4$ & $7$ \\ \hline
\end{tabular}
\end{center}
Let $g(x) = f(2x) - 5$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $g(1)$ using the table.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Find $g(2)$. (The table stops before you need it --- use the rule for $f$.)
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[0pt]{Let $f(x) = \sqrt{x}$ and $g(x) = -3f(x) + 2$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $g(9)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Describe, in order, the transformations that take $y = f(x)$ to $y = g(x)$.
  \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = x^{2} - 4x$ and $g(x) = f(x - 2) + 5$. Write $g(x)$ in expanded form $ax^{2} + bx + c$.}

\problem[0pt]{Let $p(x) = x^{3}$ and $q(x) = -p(x - 1) + 2$. Complete the table.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & $0$ & $1$ & $3$ \\ \hline
$q(x)$ & & $2$ & \\ \hline
\end{tabular}
\end{center}
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $q(0)$.
  \par\vspace*{2.2cm}\ansline
  \item[(b)] Find $q(3)$.
  \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem[0pt]{Let $f(x) = x^{3}$ and $g(x) = f(6 - x)$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Find $g(2)$.
  \par\vspace*{2.4cm}\ansline
  \item[(b)] Rewrite the input in the form $-(x - h)$, then describe, in order, the two transformations that take $y = f(x)$ to $y = g(x)$.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem[6cm]{Let $f(x) = \sqrt{x}$ and $g(x) = 3f(2x - 8) + 1$. Rewrite $g(x)$ in the form $a\sqrt{b(x - h)} + k$, so that every parameter can be read off directly.}

\problem[0pt]{Let $g(x) = 4x^{2} + 24x + 41$.
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=4pt]
  \item[(a)] Complete the square to write $g(x)$ in the form $a(x - h)^{2} + k$.
  \par\vspace*{3.4cm}\ansline
  \item[(b)] Describe, in order, the transformations that take $y = x^{2}$ to $y = g(x)$, and say why the order of the last two matters.
  \par\vspace*{2.8cm}\ansline
\end{itemize}}
