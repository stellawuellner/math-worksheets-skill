\wstitleblock{Sorting Shapes by What Makes Them That Shape}{Kindergarten--Grade 1}{}

\begin{center}
\textit{Counting sides and corners, sorting and ordering by those attributes, and telling defining attributes apart from colour and size}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Read this first}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item A \textbf{side} is a straight edge. A \textbf{corner} is where two sides meet.
  \item Sides and corners tell you what a shape IS. Colour, size, and which way it is turned do NOT.
  \item Count out loud and touch each side as you count it.
\end{itemize}
