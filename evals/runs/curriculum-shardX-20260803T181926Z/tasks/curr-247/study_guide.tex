% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading a Point off a Scatter Plot}

\begin{formulabox}
\textbf{Every dot is one individual} measured twice: its horizontal position is the first measurement, its vertical position is the second.\par
\textbf{To read a value:} find the given number on the horizontal axis, go straight UP to the dot, then straight ACROSS to the vertical axis. Report the vertical number --- the horizontal one is what you were given.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The points are $(1, 4)$, $(2, 7)$, $(3, 9)$, $(4, 12)$, $(5, 14)$. What is the value when $x = 3$?
\step{The question gives a horizontal value and asks for the matching vertical one, so find the dot above that number and read its height.}
\step{Above $x = 3$ the dot sits at height $9$.}

$\Rightarrow\quad \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Points $(2, 30)$, $(4, 26)$, $(6, 21)$, $(8, 18)$. What is the value when $x = 6$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Describing the Association}

\begin{formulabox}
\textbf{Direction:} as $x$ increases, does $y$ tend to rise (POSITIVE), fall (NEGATIVE), or neither (NO ASSOCIATION)?\par
\textbf{Strength:} tightly packed along one path is STRONG; a wide fuzzy band is WEAK.\par
\textbf{Shape:} one straight path is LINEAR; a bend is NONLINEAR.\par
\textbf{Then look for the exceptions:} an OUTLIER sitting far from the rest, or CLUSTERS in separate groups. Always say direction, strength and shape --- three words, not one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Describe $(1, 3)$, $(2, 5)$, $(3, 8)$, $(4, 10)$, $(5, 14)$, and give the largest and smallest values.
\step{Read the vertical values in order of the horizontal ones: their trend gives the direction, and how tightly they follow it gives the strength.}
\step{They climb steadily with no bend and no stray point: positive, strong, linear. Largest and smallest heights:}

$\ans{14}$ \quad and \quad $\ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Describe $(1, 18)$, $(2, 15)$, $(3, 13)$, $(4, 9)$, $(5, 6)$, and give the largest and smallest values.\\[2pt]
\rotatebox{180}{\footnotesize check: negative, strong, linear; $\ans{18}$ and $\ans{6}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Drawing a Line of Fit and Writing It}

\begin{formulabox}
\textbf{Drawing it:} lay a straight edge through the middle of the cloud so that roughly as many points sit above the line as below, and the points that miss it miss it by as little as possible. It does NOT have to pass through any data point, and it is never a line joining the first and last dots.\par
\textbf{Writing it:} pick TWO points that lie ON YOUR LINE. Then $m=\dfrac{y_{2}-y_{1}}{x_{2}-x_{1}}$, and put one of those points into $y = mx + b$ to get $b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line of fit passes through $(2, 5)$ and $(8, 17)$. Write its equation.
\step{Two points on the line are enough to pin it down: the slope comes from the pair, and the intercept from putting one point back into the equation.}
\step{$m=\dfrac{17-5}{8-2}=2$; then $5 = 2(2) + b$ gives $b = 5 - 4$.}

$m = \ans{2}$, \quad $b = \ans{1}$, \quad so $y = 2x + 1$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line of fit passes through $(1, 20)$ and $(6, 10)$. Find $m$ and $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $m = \ans{-2}$, $b = \ans{22}$}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Skill 4 --- Predicting from the Line}

\begin{formulabox}
\textbf{To predict:} put the given $x$ into $y = mx + b$ and work out $y$. Write the substitution before the answer.\par
\textbf{Say which kind of prediction it is:} INTERPOLATION inside the range of the data (trustworthy) or EXTRAPOLATION outside it (risky, and worthless once it gives an impossible value).\par
\textbf{Reading the slope:} $m$ is how much $y$ changes for each $1$ that $x$ increases --- an association in this sample, not proof of a cause.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Use $y = 2x + 1$ to predict the value at $x = 12$.
\step{A prediction from a line of fit is just a substitution, so replace the letter with the given number and evaluate.}
\step{$y = 2(12) + 1 = 24 + 1$.}

$\Rightarrow\quad \ans{25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Use $y = -2x + 22$ to predict the value at $x = 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}
