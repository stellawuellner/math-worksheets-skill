% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Clearing the fractions first}

\begin{formulabox}
\textbf{Rule:} multiply \emph{both sides} of the equation by a common denominator. Every term gets
multiplied --- including any term that had no fraction in it. After that step the equation contains
no fractions at all and is an ordinary two-step equation.
The common denominator is the smallest number all the denominators divide into.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{2} + \dfrac{x}{3} = 10$.
\step{Two different denominators make the arithmetic messy, so I clear them both at once rather than combining the fractions.}
\step{Denominators $2$ and $3$ divide into $6$; multiplying every term by $6$ gives $3x + 2x = 60$.}
\step{Collect and divide: $5x = 60$, so $x = 12$. Check: $6 + 4 = 10$.}

$\Rightarrow\quad \ans{x = 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{3} + \dfrac{x}{6} = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- A fraction in front of a bracket}

\begin{formulabox}
\textbf{Two legal routes, same answer:}
undo the fraction first --- multiply both sides by the reciprocal, so
$\frac{a}{b}(\,\cdot\,) = c$ becomes $(\,\cdot\,) = \frac{bc}{a}$ --- or distribute the fraction
across both terms in the bracket and then solve. Pick the first when the right-hand side divides
cleanly; pick the second when it does not.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{2(x - 4)}{3} = 6$.
\step{The whole bracket is multiplied by a fraction, and the right-hand side divides cleanly, so undoing the fraction is quicker than distributing.}
\step{Multiply both sides by $3$: $2(x - 4) = 18$. Divide both sides by $2$: $x - 4 = 9$.}
\step{Add $4$ to both sides. Check: $2(13 - 4) \div 3 = 18 \div 3 = 6$.}

$\Rightarrow\quad \ans{x = 13}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{3(x + 2)}{4} = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- The variable on both sides}

\begin{formulabox}
\textbf{Order of work:} clear the fractions, then move every variable term to one side with a
both-sides addition or subtraction, then move the numbers to the other side, then divide.
If the variable terms cancel and leave a \emph{true} statement such as $5 = 5$, every number works;
if they leave a \emph{false} one such as $-2 = 4$, nothing works and there is no solution.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{3x}{5} - 2 = \dfrac{x}{10} + 3$.
\step{There is an $x$ on each side and two denominators, so I clear the fractions before trying to gather the variable.}
\step{Multiply every term by $10$: $6x - 20 = x + 30$.}
\step{Subtract $x$ from both sides: $5x - 20 = 30$. Add $20$: $5x = 50$. Divide by $5$.}

$\Rightarrow\quad \ans{x = 10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{2} + 3 = \dfrac{x}{6} + 7$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 12}$}
\end{tryitbox}
