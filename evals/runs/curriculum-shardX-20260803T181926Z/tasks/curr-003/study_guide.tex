\sstitleblock{Numerals, Number Words, and How Many}

\begin{formulabox}
\textbf{One number, three ways.} The \textbf{picture} shows the things, the
\textbf{numeral} is the sign you write, and the \textbf{number word} is what
you say. They all name the same amount.\\[3pt]
\textbf{The how-many rule:} touch each thing once, and the \emph{last number
you say} is how many there are.\\[3pt]
\textbf{Frames are shortcuts:} a full five-frame is $5$ and a full ten-frame is
$10$, so start there and count on instead of counting from one.
\end{formulabox}

\skillheading{1. Count a picture and write the numeral}

\begin{formulabox}
Count the full frame first, then count \emph{on} for the loose things --- do
not go back to one. Then write the last number you said as a numeral.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A full five-frame and $3$ loose dots. How many?
\step{The frame is full, so it already shows five --- start there instead of
counting it again.}
\step{Count on for the loose dots: ``six, seven, eight''. That is $5 + 3$.}

The numeral is $\ans{8}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A full five-frame and $2$ loose dots. Write the numeral.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{2. Match a number word to its numeral}

\begin{formulabox}
Teen words are ``ten and some more'': \emph{eleven} is ten and one,
\emph{twelve} is ten and two, \emph{thirteen} is ten and three.\\[3pt]
Write the ten first, then the extras --- so \emph{thirteen} is $13$, never
$31$. A full ten-frame beside the loose counters shows you which way round it
goes.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} The word card says \emph{eleven}. What numeral is that?
\step{Teen words name a ten and some more, so build eleven out of a full
ten-frame plus the extras.}
\step{A full ten-frame is $10$; count on for the one extra counter:
``eleven''. That is $10 + 1$.}

The numeral is $\ans{11}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The word card says \emph{thirteen}. Write the numeral.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{13}$}
\end{tryitbox}

\skillheading{3. Compare two amounts shown in different ways}

\begin{formulabox}
Turn \emph{both} cards into numbers first --- count a picture, read a word,
keep a numeral as it is. Only then compare.\\[3pt]
$>$ means \emph{is greater than} and $<$ means \emph{is less than}. The open
end of the sign always faces the bigger number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} One card shows $4$ dots and $2$ more. The other shows $4$
dots and $4$ more. Which is less?
\step{Nothing can be compared until each card is one number, so finish both
counts first.}
\step{First card: $4 + 2$, which is six. Second card: $4 + 4$, which is eight.
Six is less than eight.}

First card $\ans{<}$ second card.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} One card shows $5$ dots and $3$ more; the other shows $5$ dots
and $1$ more. Write $>$ or $<$ between them.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{>}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a set that takes up more space is not always more ---
only the count decides. And when someone asks ``how many?'' after you have
counted, say the last number again instead of counting the whole set over.
\end{watchoutbox}
