\sstitleblock{Solving Quadratics Without Losing a Root or a Sign}

\skillheading{Skill 1 --- Every square root gets a plus-or-minus}

\begin{formulabox}
\textbf{Rule:} if $u^2 = k$ and $k > 0$, then $u = \pm\sqrt{k}$ --- two numbers
share one square, so one square root step produces \emph{two} branches.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Never divide both sides by something containing $x$: that erases the root
        where it equals zero. Factor instead.
  \item How many roots to expect: discriminant positive $\rightarrow$ two,
        zero $\rightarrow$ one (a double root), negative $\rightarrow$ none real.
  \item Finish by substituting every root back into the original equation.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $x^2 = 36$.
\step{The variable appears only as a square, so this is a square-root problem ---
and the square-root step is exactly where a root goes missing.}
\step{$x = \pm\sqrt{36}$, which splits into the two branches $x = +6$ and
$x = -6$.}

$\ans{x = -6 \text{ or } x = 6}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $x^2 = 81$. Give the complete solution set.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -9 \text{ or } x = 9}$}
\end{tryitbox}

\skillheading{Skill 2 --- Substitute signs, not just digits}

\begin{formulabox}
\textbf{Formula:} $x = \dfrac{-b \pm \sqrt{b^2 - 4ac}}{2a}$, \quad
$\Delta = b^2 - 4ac$.\par
\smallskip
\textbf{Two sign traps.} $-b$ means \emph{the opposite of} $b$: if $b$ is
negative, $-b$ is positive. And $-4ac$ becomes an \emph{addition} whenever $a$
and $c$ have opposite signs. Write $a$, $b$, $c$ with their signs in a list
before substituting, and put every negative in brackets.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $x^2 - 6x + 5 = 0$ with the formula; give the larger
root.
\step{Nothing is squared alone here, so the formula is the general tool --- and
$b$ is negative, which is where the sign slip happens.}
\step{$a = 1$, $b = -6$, $c = 5$, so $-b = 6$ and $\sqrt{b^2 - 4ac} = 4$:
$\; x = \dfrac{6 + 4}{2}$.}

The larger root is $\ans{x = 5}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $x^2 - 2x - 8 = 0$ with the formula and give the
larger root.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Reading a factored equation correctly}

\begin{formulabox}
\textbf{Zero-product property:} if $PQ = 0$ then $P = 0$ or $Q = 0$. Set
\emph{each factor} equal to zero and solve it as its own little equation.
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item The root has the \emph{opposite} sign to the constant in its factor.
  \item A factor with a coefficient needs two moves, not one: subtract, then divide.
  \item Count your roots: one factor, one root.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $(3x - 2)(x + 5) = 0$.
\step{The product is already zero and already factored, so the zero-product
property finishes it --- no expanding.}
\step{$3x - 2 = 0$ needs two moves: $3x = 2$, then divide by $3$. And
$x + 5 = 0$ gives $x = -5$.}

$\ans{x = \tfrac{2}{3} \text{ or } x = -5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $(2x - 5)(x + 1) = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{5}{2} \text{ or } x = -1}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A solution set with one root is a claim, not a default.
Either the discriminant is zero, or you lost a root at a square root or a
division. Check which before you hand it in.
\end{watchoutbox}
