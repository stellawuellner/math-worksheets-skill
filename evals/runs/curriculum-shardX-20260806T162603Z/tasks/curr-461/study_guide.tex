\sstitleblock{Differentiating Implicit Relations}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- The Basic Move}

\begin{formulabox}
\textbf{Rule:} differentiate both sides with respect to $x$. A term in $x$
behaves normally; a term in $y$ picks up a chain-rule factor:
\[
  \frac{d}{dx}\left(y^n\right) = ny^{n-1}\frac{dy}{dx}.
\]
Then gather the $\frac{dy}{dx}$ terms on one side, factor it out, and divide.
The answer normally contains both $x$ and $y$, and that is correct --- a relation
can have two different slopes above the same $x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $x^2 + 2y^2 = 22$, find the slope at $(2,3)$.
\step{The relation cannot be solved for $y$ in one clean branch here, so differentiate as it stands and let the chain rule carry the $dy/dx$.}
\step{$2x + 4y\frac{dy}{dx} = 0$ gives $\frac{dy}{dx} = -\frac{x}{2y}$, and at $(2,3)$ that is $-\frac{2}{6}$.}

$\Rightarrow\quad$ $\ans{-\tfrac{1}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $4x^2 + y^2 = 20$, find the slope at $(1,4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Products and Powers of y}

\begin{formulabox}
\textbf{Rule:} a term like $xy$ or $x^2y^3$ is a \emph{product} of a function of
$x$ and a function of $x$ --- because $y$ is one too. So the product rule
applies, and the $y$ factor still contributes its chain-rule
$\frac{dy}{dx}$:
\[
  \frac{d}{dx}(xy) = y + x\frac{dy}{dx}.
\]
Missing the first term is the single most common error, and it usually leaves a
suspiciously simple answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $xy + y^2 = 3$, find the slope at $(2,1)$.
\step{The $xy$ term is a product, so it needs both pieces of the product rule before anything can be collected.}
\step{$y + x\frac{dy}{dx} + 2y\frac{dy}{dx} = 0$ gives $\frac{dy}{dx} = -\frac{y}{x+2y}$, and at $(2,1)$ that is $-\frac{1}{4}$.}

$\Rightarrow\quad$ $\ans{-\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $3xy + y^2 = 18$, find the slope at $(1,3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-1}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Checking Against the Explicit Branch}

\begin{formulabox}
\textbf{Why it works:} when a relation \emph{can} be solved for $y$ near a
point, the implicit answer and the explicit answer must agree there --- they
describe the same tangent line. Substituting $y$ back into the implicit form is
the fastest way to confirm you did not lose a factor, and it is a good habit on
any circle, parabola or ellipse.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad The circle $x^2 + y^2 = 169$ contains $(5,12)$. Compare
the two methods there.
\step{Near $(5,12)$ the upper branch is $y = \sqrt{169 - x^2}$, so both routes are available and should give the same number.}
\step{Explicitly, $\frac{dy}{dx} = \frac{-x}{\sqrt{169-x^2}}$; implicitly, $2x + 2y\frac{dy}{dx} = 0$ gives $\frac{dy}{dx} = -\frac{x}{y}$, and at $(5,12)$ both read $-\frac{5}{12}$.}

$\Rightarrow\quad$ $\ans{\dfrac{-x}{\sqrt{169-x^2}}}$ \quad and at the point \quad $\ans{-\tfrac{5}{12}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The circle $x^2+y^2=100$ contains $(6,8)$. Find the
explicit derivative of the upper branch, and the implicit slope at that point.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{-x}{\sqrt{100-x^2}}}$ \ and \ $\ans{-\tfrac{3}{4}}$}
\end{tryitbox}
