\wstitleblock{Separable Differential Equations with Initial Conditions}{AP Calculus AB/BC}{}

\begin{center}
\small Separating variables, finding the constant from an initial condition, and using the particular solution
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Separate the variables first: get every $y$ (with $dy$) on one side and every $x$ (with $dx$) on the other before integrating.
  \item One constant of integration is enough. Put $+\,C$ on the $x$ side only.
  \item Substitute the initial values as soon as you have integrated, then solve for $C$ before rearranging.
  \item Where a decimal is asked for, round only at the very last step.
\end{itemize}

\vspace{0.2cm}

\problem[5.5cm]{Separate the variables in $\dfrac{dy}{dx} = 5x^4 y$ and integrate both sides. Write your result in the form $\ln|y| = F(x) + C$.}

\problem[5.5cm]{Separate the variables in $\dfrac{dy}{dx} = \dfrac{4x}{y}$ and integrate both sides. Write your result in the form $\frac{1}{2}y^2 = G(x) + C$.}

\problem[6.5cm]{Solve the initial value problem $\dfrac{dy}{dx} = 2xy$, \; $y(0) = 5$. Write the particular solution $y = f(x)$, then evaluate $y(1)$ to three decimal places.}

\problem[6.5cm]{A quantity $y$ satisfies $\dfrac{dy}{dx} = \dfrac{y}{4}$ with $y(0) = 12$ and $y > 0$. Solve the initial value problem, then find the value of $x$ at which $y = 48$. Give an exact value and a decimal to three places.}

\problem[6.5cm]{Separate the variables in $\dfrac{dy}{dx} = \dfrac{\cos x}{2y}$ and integrate both sides to obtain $y^2 = \sin x + C$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Carry out the separation and the integration.
  \item[(b)] Explain why this result does not determine a single function $y = f(x)$ until an initial condition is given.
\end{itemize}}

\problem[6.5cm]{Solve the initial value problem $\dfrac{dy}{dx} = \dfrac{3x^2}{y}$, \; $y(1) = 4$, for the branch with $y > 0$. Write the particular solution $y = f(x)$, then evaluate $y(2)$ to three decimal places.}

\problem[6cm]{A colony of bacteria has mass $P$ (in thousands of cells) after $t$ days and satisfies $\dfrac{dP}{dt} = 0.08P$ with $P(0) = 25$. Solve the initial value problem and find $P(10)$ to two decimal places.}

\problem[6.5cm]{Solve the initial value problem $\dfrac{dy}{dx} = e^{\,x-y}$, \; $y(0) = \ln 3$. \; (Hint: $e^{\,x-y} = e^x e^{-y}$.) Write the particular solution $y = f(x)$, then evaluate $y(\ln 5)$ to three decimal places.}

\problem[7cm]{Water drains from a tank so that the depth $h$, measured in centimetres, satisfies $\dfrac{dh}{dt} = -\dfrac{3}{2\sqrt{h}}$, where $t$ is in minutes. The depth at $t = 0$ is $36$ centimetres. Find the number of minutes until the tank is empty ($h = 0$), to the nearest tenth of a minute.}

\problem[7cm]{Consider the initial value problem $\dfrac{dy}{dx} = y^2 \sin x$, \; $y(0) = \frac{1}{2}$.
\begin{itemize}[leftmargin=0.9cm, itemsep=2pt, topsep=3pt, label={}]
  \item[(a)] Find the particular solution $y = f(x)$, and confirm that it gives $y(0) = \frac{1}{2}$.
  \item[(b)] Explain why the largest interval containing $x = 0$ on which this solution is defined stops before $x = \pi$.
\end{itemize}}
