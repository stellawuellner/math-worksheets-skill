\wstitleblock{Make Ten Practice}{Kindergarten--Grade 1}{}

\begin{center}
\textit{Missing partners to ten, decomposing ten, and making ten to add}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Ten is a friendly number! Partners like 7 and 3 fit together to make 10.
  \item After the warm-up, each problem needs its own plan --- look before you count.
  \item Draw dots or use a ten-frame any time it helps.
\end{itemize}

\problem[2.5cm]{Warm-up. The ten-frame shows 7 dots. How many empty boxes are
left to make 10?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.55, line width=0.8pt]
  \draw (0,0) rectangle (5,2);
  \foreach \x in {1,...,4} \draw (\x,0) -- (\x,2);
  \draw (0,1) -- (5,1);
  \foreach \p in {(0.5,1.5),(1.5,1.5),(2.5,1.5),(3.5,1.5),(4.5,1.5),(0.5,0.5),(1.5,0.5)}
    \fill \p circle (0.17);
\end{tikzpicture}
\end{center}
$7\;+$~\ansblank~$=\;10$}

\problem[2.5cm]{Warm-up. The ten-frame shows 4 dots. How many more dots would
fill it?
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.55, line width=0.8pt]
  \draw (0,0) rectangle (5,2);
  \foreach \x in {1,...,4} \draw (\x,0) -- (\x,2);
  \draw (0,1) -- (5,1);
  \foreach \p in {(0.5,1.5),(1.5,1.5),(2.5,1.5),(3.5,1.5)}
    \fill \p circle (0.17);
\end{tikzpicture}
\end{center}
$4\;+$~\ansblank~$=\;10$}

\problem[2.5cm]{Warm-up. Take 6 out of 10. How many are left?
\par\vspace{4pt}
$10\;-\;6\;=$~\ansblank}

\problem[3cm]{Make ten first: $9+4$ is the same as $9+1+3$.
\par\vspace{4pt}
$9\;+\;4\;=$~\ansblank}

\problem[2.5cm]{Think of a ten-frame with 2 dots in it. How many boxes are
empty?
\par\vspace{4pt}
$2\;+$~\ansblank~$=\;10$}

\problem[2.5cm]{Think: 7 and how many more make 10?
\par\vspace{4pt}
$10\;-\;7\;=$~\ansblank}

\problem[3cm]{Make ten: fill 8 up to 10, then add what is left over.
\par\vspace{4pt}
$8\;+\;6\;=$~\ansblank}

\problem[3cm]{Zoe has 10 berries. She puts 5 of them on her plate. The rest
stay in the bowl. How many berries are in the bowl?
\par\vspace{4pt}
$10\;-\;5\;=$~\ansblank}

\problem[3cm]{Pick your plan, then add.
\par\vspace{4pt}
$7\;+\;5\;=$~\ansblank}

\problem[4cm]{Challenge! Mo adds three numbers: $6+4+2$.
First find the partner: \ $6\;+$~\ansblank~$=\;10$.
Then add all three: \ $6\;+\;4\;+\;2\;=$~\ansblank.
Tell how making ten helped you add.}
