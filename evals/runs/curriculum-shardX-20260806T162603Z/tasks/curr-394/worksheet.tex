\wstitleblock{Find the Mistake: Indexes, Ratios and Convergence}{Algebra 2}{}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Every problem quotes a value someone actually got. Your job is the correct value --- and, where asked, the move that produced the wrong one.
  \item Watch the starting index. A rule written for $n \ge 1$ means the first term is at $n = 1$, and a sum written from $k = 1$ has no $k = 0$ term in it.
  \item An infinite geometric series has a sum only when the common ratio is between $-1$ and $1$. Check that before reaching for the formula.
\end{itemize}

\problem{A sequence is given by $a_n = 4n + 3$ for $n \ge 1$. Asked for the fifth term, a student wrote $19$.
\par\smallskip
(a)~Find the correct fifth term.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~Explain the error that produced $19$.
\par\vspace*{2.7cm}\ansline}

\problem{Asked to evaluate $\displaystyle\sum_{k=1}^{4} (2k + 1)$, a student wrote $25$. Find the correct value of the sum.
\par\vspace*{5cm}\ansline}

\problem{A geometric sequence begins $5$, $15$, $45$. A student reports the common ratio as $10$, ``because $15 - 5 = 10$''.
\par\smallskip
(a)~Find the correct common ratio.
\par\vspace*{2.6cm}\ansline
\par\smallskip
(b)~Using the correct ratio, find the sixth term of the sequence.
\par\vspace*{2.9cm}\ansline}

\problem{Consider the expansion of $(x + 2)^4$, written in descending powers of $x$.
\par\smallskip
(a)~Expand $(x + 2)^4$ completely.
\par\vspace*{3.2cm}\ansline
\par\smallskip
(b)~A student reports the coefficient of the \emph{third} term as $32$. Find the correct coefficient of the third term.
\par\vspace*{2.9cm}\ansline}

\problem{A student applies the infinite geometric sum formula to $3 + 6 + 12 + 24 + \cdots$ and writes $\dfrac{3}{1 - 2} = -3$.
\par\smallskip
(a)~Explain why that value is meaningless for this series.
\par\vspace*{2.8cm}\ansline
\par\smallskip
(b)~The series $3 + \dfrac{3}{2} + \dfrac{3}{4} + \cdots$ does have a sum. Find it.
\par\vspace*{3.2cm}\ansline}

\problem{The arithmetic sequence $a_n = 5n - 2$ is defined for $n \ge 1$. Asked for the sum of its first $20$ terms, a student wrote $1008$, having added $21$ terms. Find the correct sum.
\par\vspace*{5.4cm}\ansline}

\problem{In the expansion of $(2x + 3)^5$, a student reports the coefficient of $x^2$ as $540$, having used $2$ where the term needs $2$ squared. Find the correct coefficient of $x^2$.
\par\vspace*{5.6cm}\ansline}

\problem{A geometric sequence has all terms positive. Its third term is $20$ and its fifth term is $80$.
\par\smallskip
(a)~Find the common ratio.
\par\vspace*{2.6cm}\ansline
\par\smallskip
(b)~Find the first term.
\par\vspace*{2.4cm}\ansline
\par\smallskip
(c)~Find the sum of the first six terms.
\par\vspace*{2.9cm}\ansline}
