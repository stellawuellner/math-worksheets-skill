\problem{Solve for $x$: \qquad $\log_{2} x = 5$}

\textbf{Solution:} The definition of a logarithm says $\log_{b} A = c$ means $A = b^{c}$. Here $x = 2^{5} = 32$, and $32 > 0$, so the domain allows it. \ans{x = 32}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{6}(4x) = \log_{6} 20$}

\textbf{Solution:} Same base on both sides and the log function is one-to-one, so the arguments are equal: $4x = 20$, giving $x = 5$. Check: $4(5) = 20 > 0$. \ans{x = 5}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{3} x + \log_{3} 4 = \log_{3} 24$}

\textbf{Solution:} Condense the left side with the product rule, then equate arguments.
\begin{align*}
  \log_{3}(4x) &= \log_{3} 24 \\
  4x &= 24
\end{align*}
So $x = 6$, and $6 > 0$ so both original logarithms are defined. \ans{x = 6}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{3}(x - 1) = 4$}

\textbf{Solution:} Convert to exponential form: $x - 1 = 3^{4} = 81$, so $x = 82$. Check the argument: $82 - 1 = 81 > 0$. \ans{x = 82}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{5}(2x - 3) = \log_{5}(x + 4)$}

\textbf{Solution:} Equate the arguments: $2x - 3 = x + 4$, so $x = 7$. Check both arguments at $x = 7$: $2(7) - 3 = 11 > 0$ and $7 + 4 = 11 > 0$. \ans{x = 7}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\log x + \log(x - 3) = 1$. \textbf{(a)} Domain. \textbf{(b)} Solve.}

\textbf{Solution (a):} $\log x$ needs $x > 0$; $\log(x - 3)$ needs $x - 3 > 0$, i.e. $x > 3$. Both hold exactly when \ans{x > 3}

\textbf{Solution (b):} Condense and convert:
\begin{align*}
  \log\bigl(x(x - 3)\bigr) &= 1 \\
  x^{2} - 3x &= 10 \\
  x^{2} - 3x - 10 &= 0 \\
  (x - 5)(x + 2) &= 0
\end{align*}
The algebra offers $x = 5$ and $x = -2$, but $-2$ fails the domain from part (a), so it is extraneous. \ans{x = 5}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{1/2} x = -3$}

\textbf{Solution:} Convert to exponential form with the fractional base:
\[ x = \left(\tfrac{1}{2}\right)^{-3} = 2^{3} = 8 \]
A negative exponent on a reciprocal base flips it back. Check: $8 > 0$. \ans{x = 8}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{6}\!\left(x^{2} + 5x\right) = \log_{6} 24$}

\textbf{Solution:} Equate arguments, then test each candidate --- do not discard the negative one on sight.
\begin{align*}
  x^{2} + 5x &= 24 \\
  x^{2} + 5x - 24 &= 0 \\
  (x + 8)(x - 3) &= 0
\end{align*}
At $x = 3$: $9 + 15 = 24 > 0$ $\checkmark$. \quad At $x = -8$: $64 - 40 = 24 > 0$ $\checkmark$.
The argument here is a quadratic, not a linear expression, so a negative $x$ can still make it positive. Both values are solutions.
\[ \boxed{x = -8 \quad \text{or} \quad x = 3} \]

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\log_{4}(x + 6) - \log_{4} x = 1$. \textbf{(a)} Domain. \textbf{(b)} Solve.}

\textbf{Solution (a):} The arguments need $x + 6 > 0$ (so $x > -6$) and $x > 0$. The stricter of the two governs, so the domain is \ans{x > 0}

\textbf{Solution (b):} Condense with the quotient rule, then convert:
\begin{align*}
  \log_{4}\!\left(\frac{x + 6}{x}\right) &= 1 \\
  \frac{x + 6}{x} &= 4 \\
  x + 6 &= 4x
\end{align*}
So $3x = 6$ and $x = 2$, which satisfies $x > 0$. \ans{x = 2}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{Solve for $x$: \qquad $\log_{9} x = \dfrac{3}{2}$}

\textbf{Solution:} Convert, then read the fractional exponent as a root followed by a power:
\[ x = 9^{3/2} = \left(\sqrt{9}\right)^{3} = 3^{3} = 27 \]
Check: $27 > 0$. \ans{x = 27}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\log_{3} x + \log_{3}(x - 8) = 2$. \textbf{(a)} Domain. \textbf{(b)} Solve.}

\textbf{Solution (a):} $x > 0$ and $x - 8 > 0$. The second is stricter, so the domain is \ans{x > 8}

\textbf{Solution (b):} Condense with the product rule and convert:
\begin{align*}
  \log_{3}\bigl(x(x - 8)\bigr) &= 2 \\
  x^{2} - 8x &= 3^{2} = 9 \\
  x^{2} - 8x - 9 &= 0 \\
  (x - 9)(x + 1) &= 0
\end{align*}
$x = -1$ is outside $x > 8$, so it is extraneous; $x = 9$ satisfies it. \ans{x = 9}

\vspace{0.1cm}\noindent\rule{\linewidth}{0.2pt}

\problem{$\ln(x + 3) + \ln(x - 3) = \ln 16$. \textbf{(a)} Domain. \textbf{(b)} Solve.}

\textbf{Solution (a):} $x + 3 > 0$ gives $x > -3$; $x - 3 > 0$ gives $x > 3$. The stricter one governs, so the domain is \ans{x > 3}

\textbf{Solution (b):} Condense the left side, then equate arguments:
\begin{align*}
  \ln\bigl((x + 3)(x - 3)\bigr) &= \ln 16 \\
  x^{2} - 9 &= 16 \\
  x^{2} &= 25
\end{align*}
The square root step yields $x = 5$ and $x = -5$. Only $x = 5$ lies in $x > 3$; $x = -5$ would make both original arguments negative. \ans{x = 5}
