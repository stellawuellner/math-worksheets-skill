\sstitleblock{Vertices, Foci and Asymptotes}

\skillheading{Skill 1 --- Ellipse axes and vertices}

\begin{formulabox}
\textbf{Read the denominators.} In $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ the denominators
are the \emph{squares} of the half-axes, so take a square root before reporting a length.\par
\textbf{Which axis is long?} The larger denominator names it: larger under $x^2$ means a
horizontal major axis. Full axis lengths are $2a$ and $2b$ --- twice the half-lengths, which
is what a physical width or height usually means.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\dfrac{x^2}{49} + \dfrac{y^2}{25} = 1$, find both axis lengths.
\step{The question asks for widths, not half-widths, so square-root each denominator and then double --- the doubling is where most marks are lost.}
\step{$\sqrt{49} = 7$ and $\sqrt{25} = 5$, so the half-axes are $7$ and $5$.}

$\Rightarrow\quad$ major axis $\ans{14}$, \quad minor axis $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\dfrac{x^2}{81} + \dfrac{y^2}{36} = 1$, find both axis lengths.\\[2pt]
\rotatebox{180}{\footnotesize check: major $\ans{18}$, minor $\ans{12}$}
\end{tryitbox}

\skillheading{Skill 2 --- Ellipse foci}

\begin{formulabox}
\textbf{Subtract:} $c^2 = a^2 - b^2$, where $a$ is the larger half-axis. The foci lie on the
long axis, at $\pm c$ from the centre, so they are always \emph{inside} the curve.\par
\textbf{The defining property:} the two distances from any point of the ellipse to the two
foci always add to $2a$. That is why a whispering gallery works, and it is often what an
application is really asking for.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\dfrac{x^2}{100} + \dfrac{y^2}{64} = 1$, find the focal distance and the separation of the foci.
\step{Foci sit inside an ellipse, so the rule is the subtracting one --- and the question's ``separation'' then needs the doubling as well.}
\step{$c = \sqrt{100 - 64} = \sqrt{36}$, and the two foci are $2c$ apart.}

$\Rightarrow\quad c = \ans{6}$, \quad separation $\ans{12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\dfrac{x^2}{169} + \dfrac{y^2}{25} = 1$, find the focal distance and the separation of the foci.\\[2pt]
\rotatebox{180}{\footnotesize check: $c = \ans{12}$, separation $\ans{24}$}
\end{tryitbox}

\skillheading{Skill 3 --- Hyperbola vertices and foci}

\begin{formulabox}
\textbf{The positive term decides everything.} If $x^2$ is positive the branches open left
and right; if $y^2$ is positive they open up and down. The vertices sit at $\pm a$ from the
centre, where $a^2$ is the denominator of the \emph{positive} term.\par
\textbf{Add:} $c^2 = a^2 + b^2$ --- the opposite of the ellipse rule, and the reason a
hyperbola's foci always lie beyond its vertices.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $\dfrac{x^2}{16} - \dfrac{y^2}{9} = 1$, find the vertex and focal distances.
\step{The $x^2$ term is the positive one, so it supplies $a$; after that the only choice is add-or-subtract, and hyperbolas add.}
\step{$a = \sqrt{16} = 4$, and $c = \sqrt{16 + 9} = \sqrt{25}$.}

$\Rightarrow\quad$ vertex at $\ans{4}$, \quad focus at $\ans{5}$ \quad (focus beyond vertex, as always)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $\dfrac{y^2}{36} - \dfrac{x^2}{64} = 1$, find the vertex and focal distances.\\[2pt]
\rotatebox{180}{\footnotesize check: vertex $\ans{6}$, focus $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 4 --- Hyperbola asymptotes}

\begin{formulabox}
\textbf{Slopes depend on orientation.} For $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ the
asymptotes are $y = \pm\dfrac{b}{a}x$; for $\dfrac{y^2}{a^2} - \dfrac{x^2}{b^2} = 1$ they are
$y = \pm\dfrac{a}{b}x$. In both cases the number over the \emph{positive} term goes on top.\par
\textbf{What they mean:} far from the centre the branches get arbitrarily close to these
lines and never touch them.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the asymptote slopes of $\dfrac{x^2}{25} - \dfrac{y^2}{9} = 1$.
\step{The positive term is the one in $x$, which puts the $y$-denominator on top of the slope fraction --- get the orientation right before touching the arithmetic.}
\step{$a = \sqrt{25} = 5$ and $b = \sqrt{9} = 3$, so the slopes are $\pm b/a$.}

$\Rightarrow\quad$ slopes $\pm\ans{\tfrac{3}{5}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the asymptote slopes of $\dfrac{y^2}{16} - \dfrac{x^2}{49} = 1$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\pm\ans{\tfrac{4}{7}}$}
\end{tryitbox}
