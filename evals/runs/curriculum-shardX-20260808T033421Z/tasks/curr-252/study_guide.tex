\sstitleblock{Adding, Subtracting and Multiplying Polynomials}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Adding polynomials}

\begin{formulabox}
\textbf{Rule:} addition changes nothing inside the brackets. Drop them, then collect like terms
--- terms with the \emph{same variable and the same exponent}.
Only the coefficients are added: $a x^n + b x^n = (a+b)x^n$. A power that appears in only one
polynomial is simply copied down.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(4x^2 + 3x - 5) + (x^2 - 7x + 2)$.
\step{The operation is addition, so no signs change; the only work is deciding which terms are alike.}
\step{Squares: $4x^2 + x^2 = 5x^2$. \quad Linear: $3x - 7x = -4x$. \quad Constants: $-5 + 2 = -3$.}

$\Rightarrow\quad \ans{5x^2 - 4x - 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(2x^2 + 6x) + (5x^2 - x + 4)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7x^2 + 5x + 4}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Subtracting polynomials}

\begin{formulabox}
\textbf{Rule:} the minus sign belongs to \emph{every} term of the second bracket, not just the first:
\[
  A - (b_2x^2 + b_1x + b_0) \;=\; A - b_2x^2 - b_1x - b_0 .
\]
Rewrite the subtraction as an addition of the opposite polynomial, then collect like terms exactly
as in Skill 1.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $(6x^2 - 2x + 1) - (2x^2 + 5x - 4)$.
\step{Subtraction of a whole bracket, so before collecting anything I flip every sign in the second bracket.}
\step{$-(2x^2 + 5x - 4)$ becomes $-2x^2 - 5x + 4$; note the $-4$ becoming $+4$.}
\step{Collect: $6x^2 - 2x^2 = 4x^2$, \ $-2x - 5x = -7x$, \ $1 + 4 = 5$.}

$\Rightarrow\quad \ans{4x^2 - 7x + 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $(3x^2 + x - 6) - (x^2 - 4x + 2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2x^2 + 5x - 8}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiplying polynomials}

\begin{formulabox}
\textbf{Rule:} every term of the first bracket multiplies every term of the second. Multiply the
coefficients and \emph{add} the exponents: $ax^m \cdot bx^n = ab\,x^{m+n}$. Two binomials give four
products; a binomial times a trinomial gives six. Collect like terms at the end.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $(x + 7)(2x - 3)$.
\step{Two brackets of two terms each, so I expect four products and then one pair of like terms in the middle.}
\step{$x\cdot 2x = 2x^2$, \ $x\cdot(-3) = -3x$, \ $7\cdot 2x = 14x$, \ $7\cdot(-3) = -21$.}
\step{Collect the middle pair: $-3x + 14x = 11x$.}

$\Rightarrow\quad \ans{2x^2 + 11x - 21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $(3x - 2)(x + 5)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3x^2 + 13x - 10}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $(2x+3)^2$ is \emph{not} $4x^2 + 9$. A square is a product of two identical
brackets, so it has a middle term: $(2x+3)^2 = 4x^2 + 12x + 9$.
\end{watchoutbox}
