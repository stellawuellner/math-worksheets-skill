\wstitleblock{End Behavior and Multiplicity: Sketching Polynomials}{Precalculus --- Polynomial Function Analysis}{}

\noindent\textbf{End behavior from degree and leading coefficient, zeros and multiplicity, sketching from the features}

\vspace{0.15cm}
\noindent\textbf{The two models this sheet uses.}

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Degree} & \textbf{Leading coefficient} & \textbf{Left end} & \textbf{Right end} \\ \hline
even & positive & up & up \\ \hline
even & negative & down & down \\ \hline
odd & positive & down & up \\ \hline
odd & negative & up & down \\ \hline
\end{tabular}
\end{center}

\noindent At a zero, the \emph{multiplicity} of the factor decides the local
shape: odd multiplicity means the curve \textbf{crosses} the axis there, even
multiplicity means it \textbf{touches and turns back}. Between consecutive
zeros the sign cannot change, so one test value settles a whole interval.

\vspace{0.1cm}
\noindent\textbf{Directions:} write end behavior in limit notation, e.g.
$f(x) \to +\infty$ as $x \to -\infty$. Name the degree and the leading
coefficient you used.

\problem[5cm]{Describe the end behavior of $f(x) = 2x^3 - 5x + 1$ in both
directions. State the degree and the sign of the leading coefficient first,
then the two limits.}

\problem[5cm]{Describe the end behavior of $g(x) = -x^4 + 3x^2$ in both
directions. Does this graph's left end match its right end? Say why in terms of
the degree.}

\problem[5cm]{The polynomial $f(x) = (x + 3)(x - 1)^2$ is already factored.
List its real zeros, and beside each one state whether the graph crosses the
axis or touches and turns there.}

\problem[5.5cm]{Still with $f(x) = (x + 3)(x - 1)^2$: find $f(0)$, the
$y$-intercept, and find $f(-4)$ to determine the sign of $f$ to the left of its
smallest zero. Report both values.}

\problem[6cm]{Factor $p(x) = x^3 - 4x^2 + 4x$ completely, then list its zeros.
Which zero has multiplicity two, and what does the graph do there?}

\problem[5cm]{Describe the end behavior of $h(x) = -3x^5 + 2x^2 - 7$ in both
directions. A degree-five polynomial has ends that disagree --- the leading
coefficient decides which end does what.}

\problem[6cm]{For $f(x) = (x + 2)(x - 1)(x - 4)$, evaluate $f(0)$ and $f(2)$.
Each value fixes the sign of $f$ on the interval containing that test point;
state the two intervals and their signs.}

\problem[6.5cm]{Factor $q(x) = x^4 - 5x^2 + 4$ completely (treat it as a
quadratic in $x^2$ first), then list all four zeros in increasing order. State
the multiplicity of each.}

\problem[10cm]{Sketch $f(x) = (x + 2)^2(x - 3)$ on the axes below. Before you
draw: find $f(0)$, decide the end behavior from the expanded degree and leading
coefficient, and mark which zero is a crossing and which is a touch. Label the
$y$-intercept on your sketch.
