\sstitleblock{Rational Graphs: Holes, Asymptotes, Intercepts}

\skillheading{1. Cancel first: holes hide in the factors}

\begin{formulabox}
Hole at $x = c$ when $(x-c)$ cancels top and bottom. Height of the hole:
substitute $c$ into the \emph{simplified} function --- the original is
undefined there, so you cannot substitute into it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the hole of $\dfrac{x^2-16}{x^2+x-12}$.
\step{The top and bottom are both quadratics, so factor before deciding
anything --- a shared factor is a hole, a leftover bottom factor is an
asymptote.}
\step{$\dfrac{(x-4)(x+4)}{(x+4)(x-3)} = \dfrac{x-4}{x-3}$, so $(x+4)$
cancelled: the hole sits at $x = -4$.}
\step{Height: $\dfrac{-4-4}{-4-3} = \dfrac{-8}{-7} = \dfrac{8}{7}$.}

The hole is at $\ans{\left(-4,\ \tfrac{8}{7}\right)}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the hole of $\dfrac{x^2-25}{x^2+2x-15}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\left(-5,\ \tfrac{5}{4}\right)}$}
\end{tryitbox}

\skillheading{2. Vertical asymptotes: what is left underneath}

\begin{formulabox}
After cancelling, set each remaining denominator factor equal to zero. Those
$x$-values are vertical asymptotes: near them the graph shoots up or down
without bound, and the graph never crosses them.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Vertical asymptotes of $\dfrac{x-1}{x^2-2x-8}$.
\step{Only the denominator can create an asymptote, so factor it and check
whether the top shares any of those factors.}
\step{$x^2-2x-8 = (x-4)(x+2)$, and $x-1$ matches neither, so nothing cancels.}
\step{Set each factor to zero: $x-4=0$ and $x+2=0$.}

$\ans{x = 4}$ \quad and \quad $\ans{x = -2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Vertical asymptotes of $\dfrac{x+2}{x^2-9}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 3}$ and $\ans{x = -3}$}
\end{tryitbox}

\skillheading{3. Intercepts: where the graph meets the axes}

\begin{formulabox}
$x$-intercepts: zeros of the simplified numerator (a cancelled zero is a hole,
not an intercept). $y$-intercept: evaluate at $x = 0$ --- one substitution, and
it exists unless $x = 0$ is itself a hole or an asymptote.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Intercepts of $\dfrac{x^2-4x-5}{x^2-4}$.
\step{Two different questions, two different halves of the fraction: the top
controls where $y = 0$, and $x = 0$ gives the $y$-intercept.}
\step{$x^2-4x-5 = (x-5)(x+1)$, which is zero at $\ans{x = 5}$ and
$\ans{x = -1}$; neither value zeroes $x^2-4$, so both are real intercepts.}
\step{At $x = 0$: $\dfrac{0-0-5}{0-4} = \dfrac{-5}{-4} = \dfrac{5}{4}$.}

The $y$-intercept is $\ans{\left(0,\ \tfrac{5}{4}\right)}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Intercepts of $\dfrac{x^2+x-6}{x^2-1}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 2}$, $\ans{x = -3}$, $y$-int
$\ans{6}$}
\end{tryitbox}

\skillheading{4. End behaviour: which degree wins}

\begin{formulabox}
Compare degrees, top vs bottom: \textbf{bottom bigger} $\Rightarrow$ $y = 0$;
\textbf{equal} $\Rightarrow$ $y = \frac{\text{leading top}}{\text{leading
bottom}}$; \textbf{top bigger} $\Rightarrow$ no horizontal asymptote.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Horizontal asymptote of $\dfrac{4x^2-x}{2x^2+5}$.
\step{Only the fastest-growing term in each part matters far from the origin,
so compare degrees before doing any arithmetic.}
\step{Both degrees are $2$, so take the leading coefficients:
$\dfrac{4}{2} = 2$.}

$\ans{y = 2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Horizontal asymptote of $\dfrac{3x+1}{x^2-4}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 0}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a zero of the denominator is \emph{not} automatically a
vertical asymptote --- cancel first, or you will call a hole an asymptote. And
a graph may cross its horizontal asymptote in the middle; that line only
describes the far left and far right.
\end{watchoutbox}
