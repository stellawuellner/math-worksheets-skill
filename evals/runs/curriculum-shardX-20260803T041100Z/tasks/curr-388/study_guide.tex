\sstitleblock{Basic Matrix Operations}

\begin{formulabox}
\textbf{Size first, always.} A matrix with $m$ rows and $n$ columns is
$m \times n$. Addition and subtraction need \emph{identical} sizes; a product
$AB$ needs the columns of $A$ to equal the rows of $B$, and then $AB$ is
(rows of $A$) $\times$ (columns of $B$).\\[3pt]
\textbf{Entry by entry:} $(A \pm B)_{ij} = a_{ij} \pm b_{ij}$ and
$(kA)_{ij} = k\,a_{ij}$.\\[3pt]
\textbf{Row meets column:} $(AB)_{ij}$ is row $i$ of $A$ paired term by term
with column $j$ of $B$, then added. Order matters --- $AB$ and $BA$ are usually
different, and one may not even exist.\\[3pt]
\textbf{Determinant} ($2 \times 2$): $ad - bc$ for
$\left[\begin{smallmatrix} a & b \\ c & d \end{smallmatrix}\right]$.
\textbf{Equal matrices} have equal corresponding entries.
\end{formulabox}

\skillheading{1. Adding, subtracting and scaling}

\begin{formulabox}
Check the sizes match, then work position by position. A scalar reaches
\emph{every} entry --- the most common slip is multiplying only the first row.
Nothing here mixes entries from different positions.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $A + B$ for
$A = \left[\begin{smallmatrix} 2 & -1 \\ 4 & 3 \end{smallmatrix}\right]$,
$B = \left[\begin{smallmatrix} 5 & 2 \\ -3 & 1 \end{smallmatrix}\right]$.
\step{Both matrices are $2 \times 2$, so the sum exists and every entry of the
answer comes from the matching pair of entries --- no row-column pairing is
involved.}
\step{$2 + 5 = 7$, \ $-1 + 2 = 1$, \ $4 + (-3) = 1$, \ $3 + 1 = 4$.}
$A + B = \begin{bmatrix} \ans{7} & \ans{1} \\ \ans{1} & \ans{4} \end{bmatrix}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $3A$ for
$A = \left[\begin{smallmatrix} 2 & -5 \\ 1 & 4 \end{smallmatrix}\right]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\begin{bmatrix} \ans{6} & \ans{-15} \\
\ans{3} & \ans{12} \end{bmatrix}$}
\end{tryitbox}

\skillheading{2. Multiplying: row meets column}

\begin{formulabox}
Slide row $i$ of the left matrix down column $j$ of the right one, multiplying
the pairs you meet and adding the results. That single number is the entry in
row $i$, column $j$. If the row and the column have different lengths, the
product does not exist.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find $AC$ for
$A = \left[\begin{smallmatrix} 1 & 2 \\ 3 & 1 \end{smallmatrix}\right]$ and the
column $C = \left[\begin{smallmatrix} 4 \\ 2 \end{smallmatrix}\right]$.
\step{$A$ is $2 \times 2$ and $C$ is $2 \times 1$: the inner dimensions agree,
so the answer is a $2 \times 1$ column --- one number per row of $A$.}
\step{Row 1: $(1)(4) + (2)(2) = 8$. \quad Row 2: $(3)(4) + (1)(2) = 14$.}
$AC = \begin{bmatrix} \ans{8} \\ \ans{14} \end{bmatrix}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $AC$ for
$A = \left[\begin{smallmatrix} 2 & 3 \\ 1 & 4 \end{smallmatrix}\right]$ and
$C = \left[\begin{smallmatrix} 1 \\ 3 \end{smallmatrix}\right]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\begin{bmatrix} \ans{11} \\ \ans{13}
\end{bmatrix}$}
\end{tryitbox}

\skillheading{3. Determinants and matrix equations}

\begin{formulabox}
The $2 \times 2$ determinant is main diagonal minus anti-diagonal: $ad - bc$.
Subtract, never add. For a matrix equation, equal matrices means equal
corresponding entries, so one matrix statement becomes one ordinary equation
per position.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Find the determinant of
$\left[\begin{smallmatrix} 6 & 4 \\ 3 & 5 \end{smallmatrix}\right]$.
\step{A single number is wanted from a $2 \times 2$ matrix, and the determinant
formula is the only operation that does that: main diagonal product minus the
other one.}
\step{Main diagonal $(6)(5) = 30$; anti-diagonal $(4)(3) = 12$; subtract.}
The determinant is $\ans{18}$, and being nonzero it says the matrix is
invertible.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Find $x$ if
$\left[\begin{smallmatrix} x - 3 & 4 \\ 2 & 7 \end{smallmatrix}\right] =
\left[\begin{smallmatrix} 8 & 4 \\ 2 & 7 \end{smallmatrix}\right]$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{11}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} matrix multiplication is not commutative --- compute
$AB$ and $BA$ separately, and expect different answers or a product that does
not exist at all. When scaling, reach every entry. When taking a determinant,
subtract the second product rather than adding it.
\end{watchoutbox}
