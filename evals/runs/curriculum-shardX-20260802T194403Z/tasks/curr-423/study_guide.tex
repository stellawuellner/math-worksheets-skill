\sstitleblock{Solving Trigonometric Equations on Restricted Intervals}

\skillheading{1. Linear trig equations}

\begin{formulabox}
\textbf{Method:} isolate the trig function, find the \emph{reference angle} from
the positive value, then place one solution in each quadrant where that function
has the required sign. Add the period, keep what lands in the interval.\\[2pt]
\textbf{Signs:} sine $+$ in QI, QII \quad cosine $+$ in QI, QIV \quad tangent $+$
in QI, QIII.\\
\textbf{Periods:} $\sin$, $\cos$: $2\pi$ \quad $\tan$: $\pi$ (so a $2\pi$-long
interval holds \emph{two} tangent solutions, not one).
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2\cos x + 1 = 0$ on $[0, 2\pi)$.
\step{One trig function, first power --- isolate it and read the answer off the
unit circle.}
\step{$\cos x = -\tfrac{1}{2}$: reference angle $\tfrac{\pi}{3}$, and cosine is
negative in QII and QIII.}
\step{QII: $\pi - \tfrac{\pi}{3}$; \ QIII: $\pi + \tfrac{\pi}{3}$
$\;\Rightarrow\;$ $\ans{x = \tfrac{2\pi}{3},\ \tfrac{4\pi}{3}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2\sin x + \sqrt{2} = 0$ on $[0, 2\pi)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{5\pi}{4},\ \tfrac{7\pi}{4}}$}
\end{tryitbox}

\skillheading{2. Rewrite with an identity, then factor}

\begin{formulabox}
\textbf{Rule: one angle, one function, then factor.} An equation mixing $2x$ with
$x$, or $\sin$ with $\cos$, cannot be solved until a substitution makes the terms
comparable.\\[2pt]
$\sin 2x = 2\sin x\cos x$ \qquad $\cos 2x = 1 - 2\sin^2 x = 2\cos^2 x - 1$
\qquad $\sin^2 x + \cos^2 x = 1$\\[2pt]
\textbf{Never divide} both sides by a factor containing the variable --- dividing
by $\cos x$ deletes every solution where $\cos x = 0$. Move everything to one
side and factor instead.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $\sin 2x = \sin x$ on $[0, 2\pi)$.
\step{The two sides use different angles, so rewrite $\sin 2x$ to put everything
at angle $x$, then factor rather than divide.}
\step{$2\sin x\cos x - \sin x = 0 \;\Rightarrow\; \sin x\,(2\cos x - 1) = 0$}
\step{$\sin x = 0$ gives $0, \pi$; \ $\cos x = \tfrac{1}{2}$ gives
$\tfrac{\pi}{3}, \tfrac{5\pi}{3}$ $\;\Rightarrow\;$
$\ans{x = 0,\ \tfrac{\pi}{3},\ \pi,\ \tfrac{5\pi}{3}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $\sin 2x = 2\cos x$ on $[0, 2\pi)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = \tfrac{\pi}{2},\ \tfrac{3\pi}{2}}$}
\end{tryitbox}

\skillheading{3. Quadratic in a trig function}

\begin{formulabox}
\textbf{Method:} if the equation has the shape $au^2 + bu + c = 0$ with
$u = \sin x$ (or $\cos x$), factor it exactly as you would an ordinary quadratic,
then solve each factor.\\[2pt]
\textbf{Then check each value of $u$:} $\sin x$ and $\cos x$ only take values in
$[-1, 1]$, so a factor giving $u = 2$ contributes \emph{no} solutions --- and
$u = \pm 1$ contributes only \emph{one} per revolution, not two.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Solve $2\cos^2 x + \cos x - 1 = 0$ on $[0, 2\pi)$.
\step{Only $\cos x$ appears, to the second and first power --- that is a
quadratic, so factor in $\cos x$.}
\step{$(2\cos x - 1)(\cos x + 1) = 0$}
\step{$\cos x = \tfrac{1}{2}$ gives $\tfrac{\pi}{3}, \tfrac{5\pi}{3}$;
$\cos x = -1$ gives the single angle $\pi$ $\;\Rightarrow\;$
$\ans{x = \tfrac{\pi}{3},\ \pi,\ \tfrac{5\pi}{3}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Solve $2\sin^2 x - \sin x = 0$ on $[0, 2\pi)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 0,\ \tfrac{\pi}{6},\ \tfrac{5\pi}{6},\ \pi}$}
\end{tryitbox}

\skillheading{4. Periodic models}

\begin{formulabox}
\textbf{Model:} $d(t) = \text{midline} + \text{amplitude}\cdot
\sin\!\left(\frac{2\pi t}{\text{period}}\right)$.\\[2pt]
\textbf{Method:} set $d(t)$ equal to the target, isolate the trig function, then
substitute $u$ for the whole inside angle. \textbf{Translate the interval too:}
if $t \in [0, T)$ and the period is $T$, then $u \in [0, 2\pi)$ --- one
revolution, so the usual unit-circle count applies. Solve for $u$, convert each
answer back to $t$, and state the units.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Water depth is $d(t) = 5 + 3\sin\!\left(\frac{\pi t}{6}\right)$
metres, $t$ hours after midnight. When is the depth $6.5$ m on $[0, 12)$ hours?
\step{The target depth is a horizontal line across one period of the model, so
isolate the sine and solve on the matching angle interval.}
\step{$3\sin\!\left(\frac{\pi t}{6}\right) = 1.5 \;\Rightarrow\;
\sin u = \tfrac{1}{2}$ where $u = \frac{\pi t}{6} \in [0, 2\pi)$.}
\step{$u = \tfrac{\pi}{6}, \tfrac{5\pi}{6}$, and $t = \tfrac{6u}{\pi}$
$\;\Rightarrow\;$ $\ans{t = 1,\ 5}$ hours after midnight.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Same model. When is the depth $3.5$ m on $[0, 12)$ hours?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{t = 7,\ 11}$ hours}
\end{tryitbox}
