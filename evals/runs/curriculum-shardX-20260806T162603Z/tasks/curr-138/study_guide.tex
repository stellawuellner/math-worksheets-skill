% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Powers of Ten Move the Point}

\begin{formulabox}
\textbf{Rule:} multiplying by $10$, $100$, $1000$ moves the decimal point
$1$, $2$, $3$ places to the \emph{right}; dividing moves it that many places
to the \emph{left}. Each digit becomes worth ten times more (or less) per
step.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $2.7 \times 100$.
\step{Multiplying by a power of ten only changes what each digit is
\emph{worth}, so slide the decimal point --- two zeros means two places
right.}
\step{$2.7 \to 27 \to 270$ \quad (fill the empty place with a zero)}

$\Rightarrow\quad 2.7 \times 100 = \ans{270}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $36.5 \div 100$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.365}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Multiply, Then Count Decimal Places}

\begin{formulabox}
\textbf{Rule:} multiply as if both numbers were whole numbers, then give the
product as many decimal places as the \emph{two factors have together}.
Why: tenths $\times$ tenths $=$ hundredths.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $0.3 \times 0.7$.
\step{Multiply the digits as whole numbers first, then place the decimal
point by counting both factors' decimal places.}
\step{$3 \times 7 = 21$; one place $+$ one place $=$ two places, so the
answer is $21$ hundredths.}

$\Rightarrow\quad 0.3 \times 0.7 = \ans{0.21}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $0.5 \times 0.9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{0.45}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Divide with a Whole-Number Divisor}

\begin{formulabox}
\textbf{Rule:} to divide by a decimal, multiply \emph{both} numbers by the
same power of ten until the divisor is a whole number. Both numbers shift
together, so the quotient does not change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Divide $3.6 \div 0.4$.
\step{Make the divisor whole by shifting both numbers one place --- the
quotient stays the same because both grew tenfold.}
\step{$3.6 \div 0.4 = 36 \div 4$}

$\Rightarrow\quad 3.6 \div 0.4 = \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Divide $5.6 \div 0.8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} an answer can look ten times too big or too small and
still have the right digits. \emph{Estimate first} (round each number), then
check that your exact answer is close to the estimate.
\end{watchoutbox}
