\wstitleblock{Solving Logarithmic Equations with Domain Checks}{Precalculus / Advanced Algebra}{}

\noindent\textbf{Focus:} Rewriting a logarithm in exponential form, equating like
logarithms, combining logarithms before solving, and approximating solutions.

\vspace{4pt}
\noindent\textbf{The four-step pattern.} Follow it on every problem; the domain
step is not optional, because the algebra that clears logarithms can create
solutions the original equation never had.

\noindent\fbox{\begin{minipage}{0.97\linewidth}
\small
\textbf{Worked pattern:} \quad Solve $\log_4(x + 1) = 3$.\par
\textbf{1. Domain.} Every logarithm needs a positive argument: $x + 1 > 0$, so
$x > -1$.\par
\textbf{2. Rewrite.} $\log_b A = c$ means $A = b^c$, so $x + 1 = 4^3 = 64$.\par
\textbf{3. Solve.} $x = 63$.\par
\textbf{4. Check.} $63 > -1$, so the candidate is in the domain, and
$\log_4(64) = 3$ $\checkmark$. \quad Answer: $x = 63$.
\end{minipage}}

\vspace{6pt}
\noindent For each problem below: \textbf{state the domain restriction first},
solve, and check every candidate against that domain. Give exact answers unless
the problem asks for a decimal.

\vspace{0.2cm}

\problem[3.8cm]{Solve: \quad $\log_2 x = 5$}

\problem[3.8cm]{Solve: \quad $\log_3(x - 4) = 2$}

\problem[4.2cm]{Solve: \quad $\log_5(2x + 1) = \log_5(x + 9)$ \par
(Two logarithms with the same base are equal exactly when their arguments are
equal --- but only where both arguments are positive.)}

\problem[4.2cm]{No solving here --- build the tool. Use the product rule to
rewrite $\log_2(8x)$ in the form $c + \log_2 x$ for a constant $c$, and state
the values of $x$ for which the rewriting is valid.}

\problem[4.6cm]{Solve: \quad $\log_2 x + \log_2 (x - 2) = 3$ \par
(Combine the left side into a single logarithm first.)}

\problem[4.6cm]{Clearing the logarithms in problem 5 gives the quadratic
$x^2 - 2x - 8 = 0$, whose roots are $x = 4$ and $x = -2$.
Evaluate the argument $x - 2$ at the candidate $x = -2$, then use that value to
explain in one sentence why $-2$ cannot be a solution of the original equation.
State the domain of that equation as well.}

\problem[4.2cm]{Solve exactly (leave $e$ in your answer): \quad $\ln(x - 1) = 2$}

\problem[5cm]{Solve: \quad $\log_2(x^2 + 4x) = \log_2 12$ \par
(Careful --- do not assume the negative candidate is extraneous. Test it against
the domain before you decide.)}

\problem[4.6cm]{Solve for $x$ and round to the nearest hundredth: \quad
$\ln(2x - 1) = 3$}

\problem[4.6cm]{Solve: \quad $\log(x + 3) - \log(x - 1) = 1$ \par
(Base-10 logarithms. Use the quotient rule first.)}

\problem[4.6cm]{Solve for $x$ and round to the nearest hundredth: \quad
$\log_2(x + 5) = 4.7$}

\problem[5.4cm]{\textbf{Synthesis.} Solve: \quad $2\log_5 x - \log_5(x + 6) = 0$
\par
(Power rule, then quotient rule, then remember what an exponent of $0$ produces.
State the domain before you start and say which candidate it eliminates.)}
