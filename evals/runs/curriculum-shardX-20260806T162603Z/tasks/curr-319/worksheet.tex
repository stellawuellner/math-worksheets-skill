\wstitleblock{Finding the Error: Correspondence and Scale Factor}{High School Geometry}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Each problem shows a piece of work that reaches a wrong answer. The
        arithmetic in it is always correct --- the mistake is in how the
        proportion or the scale factor was set up.
  \item First redo the problem correctly and write the corrected value, with
        its unit, on the answer line.
  \item Then name the error in a sentence: say what was matched with what, and
        what should have been matched instead.
  \item A similarity statement is read position by position: in
        $\triangle ABC \sim \triangle DEF$, $A$ matches $D$, $B$ matches $E$,
        $C$ matches $F$.
\end{itemize}

\vspace{0.1cm}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = 6$ cm, $BC = 8$ cm, and
$DE = 9$ cm. A student wrote
\[ \frac{6}{9} = \frac{9}{EF} \qquad \text{so} \qquad EF = 13.5 \text{ cm.} \]
\par
(a) Find the correct length of $\overline{EF}$, in centimetres.
\answerline{cm}
\par\vspace*{2.6cm}
(b) Name the error in the student's proportion.
\par\vspace*{2.6cm}}

\problem{$\triangle PQR \sim \triangle STU$ with $PQ = 10$ in, $QR = 15$ in, and
$ST = 4$ in. A student wrote
\[ \frac{10}{4} = \frac{TU}{15} \qquad \text{so} \qquad TU = 37.5 \text{ in.} \]
\par
(a) Find the correct length of $\overline{TU}$, in inches. \answerline{in}
\par\vspace*{2.6cm}
(b) Name the error in the student's proportion.
\par\vspace*{2.6cm}}

\problem{$\triangle ABC \sim \triangle XYZ$ with $AB = 12$ m, $BC = 20$ m, and
$XY = 18$ m. A student wrote
\[ \text{scale factor from } \triangle ABC \text{ to } \triangle XYZ = \frac{12}{18},
   \qquad YZ = 20 \cdot \frac{12}{18} = 13.33 \text{ m.} \]
\par
(a) Find the correct scale factor from $\triangle ABC$ to $\triangle XYZ$.
\ansline
\par\vspace*{2.4cm}
(b) Find the correct length of $\overline{YZ}$, in metres. \answerline{m}
\par\vspace*{2.4cm}
(c) Explain what went wrong with the student's scale factor.
\par\vspace*{2.6cm}}

\problem{$\triangle JKL \sim \triangle MNP$ with $m\angle J = 40^\circ$ and
$m\angle K = 75^\circ$. A student concluded that $m\angle P = 75^\circ$.
\par
(a) Find the correct value of $m\angle P$, in degrees. \ansline
\par\vspace*{2.6cm}
(b) Name the error in the student's reasoning.
\par\vspace*{2.6cm}}

\problem{A poster shop reduces a triangular decal. The original decal
$\triangle ABC$ has $AB = 30$ cm and $BC = 42$ cm; the reduced copy
$\triangle DEF$ has $DE = 7.5$ cm. A student wrote
\[ EF = 42 \cdot \frac{30}{7.5} = 168 \text{ cm.} \]
\par
(a) Find the correct length of $\overline{EF}$, in centimetres.
\answerline{cm}
\par\vspace*{2.8cm}
(b) Explain which way the scale factor has to be applied here, and how the
student's answer shows the direction was reversed.
\par\vspace*{2.8cm}}

\problem{$\triangle ABC \sim \triangle DEF$ with $AB = (x+4)$ cm, $DE = 12$ cm,
$BC = 10$ cm, and $EF = 15$ cm. A student wrote
\[ \frac{x+4}{12} = \frac{15}{10} \qquad \text{so} \qquad x = 14. \]
\par
(a) Solve correctly for $x$. \ansline
\par\vspace*{2.8cm}
(b) Name the error in the student's proportion.
\par\vspace*{2.8cm}}
