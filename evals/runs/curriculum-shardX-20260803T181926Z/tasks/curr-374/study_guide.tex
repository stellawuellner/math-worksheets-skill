\sstitleblock{Radicals and Rational Exponents --- The Illegal Split}

\skillheading{Skill 1 --- Roots of a sum: test before you split}

\begin{formulabox}
\textbf{Legal:} $\sqrt{ab}=\sqrt{a}\,\sqrt{b}$ and $\sqrt{a/b}=\sqrt{a}/\sqrt{b}$
--- a radical distributes over multiplication and division.\par
\smallskip
\textbf{Not legal:} $\sqrt{a+b} \ne \sqrt{a}+\sqrt{b}$, and in exponent form
$(a+b)^{1/2} \ne a^{1/2}+b^{1/2}$. The radical sign is a grouping symbol: finish
the addition underneath it, then take one root of the single number you get.\par
\smallskip
\textbf{Why:} squaring $\sqrt{a}+\sqrt{b}$ gives $a+b+2\sqrt{ab}$, not $a+b$.
The split silently drops the cross term $2\sqrt{ab}$, so it is wrong for every
pair of positive numbers.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compute $\sqrt{49+576}$, and test the split against it.
\step{The radicand is a sum, so the product rule does not apply --- add first,
then take one root, and use the split only to see how far off it lands.}
\step{$49+576=625$, and $\sqrt{625}=25$; the split would give
$\sqrt{49}+\sqrt{576}=7+24=31$.}
$\sqrt{49+576}=\ans{25}$, not $31$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Compute $\sqrt{9+40}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Simplify each radical, then combine}

\begin{formulabox}
\textbf{Rule:} pull perfect-square factors out one radical at a time:
$\sqrt{k^2 m}=k\sqrt{m}$. Two radicals may be ADDED only when they are
\emph{like radicals} --- same number under the root --- and then they combine
like terms: $5\sqrt{2}+3\sqrt{2}=8\sqrt{2}$.\par
\smallskip
\textbf{Not legal:} pushing two radicals back into one,
$\sqrt{a}+\sqrt{b} \ne \sqrt{a+b}$. That is the same illegal split run
backwards.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\sqrt{75}+\sqrt{12}$.
\step{Two unlike radicals cannot be added as they stand and must not be merged
into one root, so simplify each radicand by its largest square factor first.}
\step{$\sqrt{75}=\sqrt{25\cdot 3}=5\sqrt{3}$ and
$\sqrt{12}=\sqrt{4\cdot 3}=2\sqrt{3}$; both are now multiples of $\sqrt{3}$, so
$5\sqrt{3}+2\sqrt{3}$ combines like $5x+2x$.}
$\sqrt{75}+\sqrt{12}=\ans{7\sqrt{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\sqrt{32}+\sqrt{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6\sqrt{2}}$}
\end{tryitbox}

\skillheading{Skill 3 --- Radical equations: square a whole side}

\begin{formulabox}
\textbf{Method:} get the radical alone on one side, then square that
\emph{entire} side --- never the terms inside it one at a time. Squaring undoes
the root: $\bigl(\sqrt{u}\bigr)^2=u$.\par
\smallskip
\textbf{Then check.} Squaring can create an answer that fails the original
equation, so every solution goes back into the equation you started with.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\sqrt{x+7}=4$.
\step{The radical already stands alone, so squaring the whole side is the tool
that removes it --- splitting the radicand into $\sqrt{x}+\sqrt{7}$ would change
the equation before solving it.}
\step{Squaring both sides gives $x+7=16$, so $x=16-7$, and the check reads
$\sqrt{9+7}=\sqrt{16}=4$ $\checkmark$.}
$\ans{x = 9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sqrt{x+5}=6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 31}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} perfect squares make the split look harmless ---
$\sqrt{36+64}$ ``gives'' $6+8=14$ so smoothly that nobody checks. Test any
splitting step on numbers you can evaluate both ways; one counterexample
settles it.
\end{watchoutbox}
