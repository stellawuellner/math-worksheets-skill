\wstitleblock{Rounding and Ordering on a Number Line}{Grade 2--3 \quad Place Value Through 1,000}{}

\begin{center}
\textit{Rounding to the nearest ten, rounding to the nearest hundred, comparing
two numbers, and putting numbers in order on a number line}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item Each number line below has three long marks: a left end, a middle, and a
right end. Write a number on each short line under the long marks.
\item To round, put your number on the line first. Then ask: which end is it
closer to? That end is your answer.
\item If a number lands exactly on the middle mark, round to the \emph{bigger}
end. That is the rule everyone agrees to use.
\item ``Order'' means smallest first unless the problem says otherwise. On a
number line, numbers get bigger as you move right.
\end{itemize}

\vspace{0.2cm}

\problem[3cm]{Round 48 to the nearest ten.

Label the left end with the ten just below 48, the right end with the ten just
above it, and the middle mark with the halfway number. Then mark 48 and circle
the end it is closer to.
\nline}

\problem[3cm]{Round 83 to the nearest ten.

Label the two ends and the middle, mark 83, and circle the closer end.
\nline}

\problem[3cm]{Round 260 to the nearest hundred.

This time the ends are hundreds, so each short tick is worth ten. Label the
ends and the middle, mark 260, and circle the closer end.
\nline}

\problem[3cm]{Round 418 to the nearest hundred.

Label the ends and the middle, mark 418, and circle the closer end.
\nline}

\problem[3cm]{Write the correct symbol, $<$ or $>$, between 536 and 563.

Put both numbers on the number line first. The one further to the right is the
greater number.
\nline}

\problem[3cm]{Round 365 to the nearest ten.

Read the middle mark carefully before you answer.
\nline}

\problem[4cm]{Put these three numbers in order from least to greatest:

\begin{center}
307 \qquad 370 \qquad 273
\end{center}

Mark all three on the number line, then write them in order on the answer line.
\nline}

\problem[3cm]{Round 650 to the nearest hundred.

Label the ends and the middle, mark 650, and circle the closer end.
\nline}

\problem[4cm]{Mia wrote $409 > 490$. She said, ``409 is bigger, because its
last digit 9 is bigger than the last digit 0.''

Put 409 and 490 on the number line. Then write the correct symbol, $<$ or $>$,
between them and say in one sentence what Mia should have compared first.
\nline}

\problem[4cm]{The number 472 sits between two tens on the number line.

\smallskip
(a) How many ones is 472 past 470? ~\ansblank

\smallskip
(b) How many ones is 472 short of 480? ~\ansblank

\smallskip
(c) Which hop was shorter? Use that to round 472 to the nearest
ten. ~\ansblank
\nline}

\problem[5cm]{Put these four numbers in order from greatest to least:

\begin{center}
418 \qquad 481 \qquad 184 \qquad 148
\end{center}

All four use the same three digits, so compare the hundreds place first, then
the tens.}

\problem[6cm]{\textbf{Challenge.} Three classes collected cans for the food
drive: Room A collected 287 cans, Room B collected 312 cans, and Room C
collected 296 cans.

\smallskip
(a) Put the three counts in order from least to greatest. \ansline

\smallskip
(b) Round each count to the nearest hundred. \ansline

\smallskip
(c) All three rounded counts came out the same. Write one sentence explaining
why the rounded numbers cannot tell you which room collected the most.}
