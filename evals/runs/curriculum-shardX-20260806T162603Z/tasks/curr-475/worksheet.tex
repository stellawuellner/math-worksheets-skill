\wstitleblock{From Riemann Sums to the Fundamental Theorem}{AP Calculus AB/BC}{}

\begin{center}
\small Riemann sums, accumulation, and conclusions from the Fundamental Theorem
\end{center}

\begin{itemize}[leftmargin=1.3cm, itemsep=1pt, topsep=2pt]
  \item Problems 1--3 warm up one idea each: a Riemann sum, an accumulated
        total, a derivative of an accumulation function. After that they are
        shuffled --- decide which idea a problem is about before you start.
  \item An estimate and an exact value are different answers. Read carefully
        which one is wanted.
  \item Give exact values (fractions, radicals) unless a decimal is requested.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Estimate $\displaystyle\int_0^2 x^2\,dx$ with a \emph{right}
Riemann sum using $4$ equal subintervals. Give the exact value of the sum.}

\problem{Water enters a tank at $r(t) = 4t + 3$ litres per minute. Find the
total volume that enters during the first $6$ minutes.
\par\vspace*{4.6cm}\answerline{L}}

\problem[4.6cm]{Let $\displaystyle G(x) = \int_2^x \left(t^3 - 1\right)dt$.
Find $G'(3)$.}

\problem{A cart's velocity is $v(t) = 3t^2$ metres per second. Find its
displacement between $t = 1$ and $t = 4$ seconds.
\par\vspace*{4.6cm}\answerline{m}}

\problem{Consider $\displaystyle\int_1^3 \frac{1}{x}\,dx$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Estimate it with a \emph{left} Riemann sum using $4$ equal
    subintervals. Give the exact value of the sum.
    \par\vspace*{3.4cm}\ansline
  \item[(b)] Explain whether that sum is an overestimate or an underestimate.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[5.4cm]{Let $\displaystyle H(x) = \int_1^{x^3}\left(2t + 1\right)dt$.
Find $H'(2)$.}

\problem{A cart's speed is recorded every $2$ seconds:
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$t$ (s) & $0$ & $2$ & $4$ & $6$ & $8$ \\ \hline
$v(t)$ (m/s) & $0$ & $5$ & $9$ & $12$ & $14$ \\ \hline
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=3pt]
  \item[(a)] Use a trapezoidal sum over the four subintervals to estimate
    $\displaystyle\int_0^8 v(t)\,dt$.
    \par\vspace*{2.8cm}\ansline
  \item[(b)] Say what units your estimate carries and what it measures.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{A reservoir changes at the net rate $r(t) = 10 - 2t$ thousand litres
per hour for $0 \le t \le 8$, and holds $40$ thousand litres at $t = 0$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the net change over the eight hours.
    \par\vspace*{2.4cm}\ansline
  \item[(b)] Find the amount held at $t = 8$.
    \par\vspace*{2.0cm}\ansline
  \item[(c)] Find the time at which the reservoir holds the most.
    \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{Let $\displaystyle F(x) = \int_0^x f(t)\,dt$, where
$f(t) = t^2 - 6t + 8$, on the interval $0 \le x \le 6$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find the $x$-values at which $F$ has a local extremum.
    \par\vspace*{2.6cm}\ansline
  \item[(b)] Justify which of them gives a local maximum of $F$.
    \par\vspace*{2.8cm}\ansline
\end{itemize}}

\problem{A tank fills at $r(t) = 6\sqrt{t}$ litres per minute for
$0 \le t \le 9$, and $\displaystyle A(x) = \int_0^x 6\sqrt{t}\,dt$.
\begin{itemize}[leftmargin=1.3cm, itemsep=0pt, topsep=5pt]
  \item[(a)] Find $A(9)$.
    \par\vspace*{2.6cm}\answerline{L}
  \item[(b)] Find $A'(4)$.
    \par\vspace*{2.2cm}\ansline
  \item[(c)] Explain how $A(9)$ and $A'(4)$ answer different questions about
    the tank.
    \par\vspace*{2.6cm}\ansline
\end{itemize}}
