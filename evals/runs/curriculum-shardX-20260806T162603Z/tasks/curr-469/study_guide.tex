\sstitleblock{Related Rates and Optimization: Getting the Setup Right}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Setting Up a Related-Rates Problem}

\begin{formulabox}
\textbf{Order matters.} Write the geometric relation first, differentiate
\emph{with respect to $t$}, and substitute the instant's numbers only at the
end. Every variable that changes picks up its own rate:
$\dfrac{d}{dt}\big(r^n\big) = nr^{n-1}\dfrac{dr}{dt}$.\par
Substituting $r = 5$ before differentiating turns a variable into a constant
and destroys the term you need.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A circular ripple's radius grows at $2$ cm/s. Find
$\dfrac{dA}{dt}$ when $r = 6$ cm.
\step{Area and radius are linked by $A = \pi r^2$, and both change with time, so differentiate that relation with respect to $t$ before any number goes in.}
\step{$\dfrac{dA}{dt} = 2\pi r\dfrac{dr}{dt} = 2\pi(6)(2) = 24\pi$}

$\Rightarrow\quad \ans{\frac{dA}{dt} \approx 75.40}$ square centimetres per second.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad The same ripple grows at $3$ cm/s. Find $\dfrac{dA}{dt}$
when $r = 10$ cm, to two decimals.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{188.50}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Building an Optimization Model}

\begin{formulabox}
\textbf{Two equations, then one.} Write the \emph{constraint} (what is fixed)
and the \emph{objective} (what is optimized), use the constraint to remove one
variable, then differentiate the objective and set the \emph{derivative} --- not
the objective --- to zero.\par
Check the geometry of the objective before differentiating: a wrong area or
volume formula cannot be rescued by correct calculus.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $100$ m of fence encloses a rectangle on all four sides.
Maximize the area.
\step{The perimeter is the constraint and the area is the objective, so solve the constraint for one side and put the area in a single variable.}
\step{$2x + 2y = 100$ gives $y = 50 - x$, so $A(x) = x(50 - x)$ and $A'(x) = 50 - 2x$.}

$\Rightarrow\quad \ans{A'(x) = 50 - 2x}$, \quad and $A'(x) = 0$ at $\ans{x = 25}$ metres.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same rectangle, $80$ m of fence. Find $A'(x)$ and the
value of $x$ that maximizes the area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{A'(x) = 40 - 2x}$, $\ans{x = 20}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Justifying That You Found the Extremum}

\begin{formulabox}
\textbf{A critical value is not yet an answer.} Discard any solution outside the
stated domain, then justify the survivor: $f'' > 0$ means a minimum, $f'' < 0$ a
maximum, or use a sign chart for $f'$. On a closed interval, compare the
critical values with both endpoints.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Minimize $f(x) = x + \dfrac{16}{x}$ on $x > 0$.
\step{Solving gives two critical values, so the real work is deciding which one the domain and the concavity allow.}
\step{$f'(x) = 1 - \dfrac{16}{x^2} = 0$ gives $x = \pm 4$; only $x = 4$ is in the domain, and $f''(x) = \dfrac{32}{x^3} > 0$ there, so it is a minimum with $f(4) = 4 + 4$.}

$\Rightarrow\quad \ans{x = -4 \text{ or } 4}$, \quad minimum value $\ans{f(4) = 8}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Minimize $f(x) = x + \dfrac{25}{x}$ on $x > 0$: list both
critical values, then give the minimum value.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = -5 \text{ or } 5}$, $\ans{f(5) = 10}$}
\end{tryitbox}
