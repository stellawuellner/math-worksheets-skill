\wstitleblock{Mixed Decimal Review --- Estimate It, Then Nail It}{Grade 5}{}
\begin{center}
\textit{Estimating, adding and subtracting, multiplying, and dividing
decimals}
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item The problems are mixed on purpose --- decide \emph{which} operation
        each one needs before you start computing.
  \item For estimates, round each number to the nearest whole number first.
  \item Show your work in the space under each problem.
\end{itemize}

\vspace{0.2cm}

\problem[3cm]{\emph{Estimate} by rounding each factor to the nearest whole
number: \quad $6.1 \times 4.8 \approx$}

\problem[3cm]{\emph{Estimate} by rounding each number to the nearest whole
number: \quad $19.6 \div 4.1 \approx$}

\problem[3.5cm]{Add: \quad $3.75 + 2.4$}

\problem[3.5cm]{Subtract: \quad $8 - 2.35$}

\problem[3.5cm]{Multiply: \quad $0.7 \times 0.3$}

\problem[3.5cm]{Divide: \quad $9.6 \div 0.8$}

\problem{A pack of markers costs $3.15$ dollars. Ms.\ Okafor buys $4$ packs
for her class.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] \emph{Estimate} the total cost by rounding the price to the
        nearest whole dollar.\par\vspace*{1.8cm}\ansline
  \item[(b)] Find the \emph{exact} total cost in dollars.
        \par\vspace*{2.4cm}\ansline
\end{itemize}}

\problem{Dev buys a sandwich for $2.35$ dollars and juice for $3.4$ dollars,
paying with a $10$-dollar bill. How many dollars of change does he get?
\par\vspace*{4.5cm}\ansline}

\problem{A $5.4$-meter plank is sawed into shelves that are each $0.9$ meter
long. How many shelves does that make?
\par\vspace*{4.5cm}\ansline}

\problem{Strawberries cost $2.4$ dollars per kilogram. How many dollars do
$1.5$ kilograms cost?
\par\vspace*{4.5cm}\ansline}

\problem{Ella estimates $3.2 \times 4.1$ by computing $3 \times 4 = 12$.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] Without working out the exact product, write $<$ or $>$: \quad
        $3.2 \times 4.1 \;\square\; 12$\par\vspace*{1.5cm}\ansline
  \item[(b)] Explain how you can be sure, using the way Ella rounded.
        \par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem{\textbf{Challenge.} Apples cost $2.1$ dollars per kilogram. Rosa
bags $3.9$ kilograms and pays with a $10$-dollar bill.
\begin{itemize}[leftmargin=1.4cm]
  \item[(a)] \emph{Estimate} the cost by rounding both numbers to the nearest
        whole number.\par\vspace*{1.8cm}\ansline
  \item[(b)] Find the \emph{exact} cost in dollars.\par\vspace*{2.2cm}\ansline
  \item[(c)] How many dollars of change does Rosa get? Check that your answer
        fits your estimate.\par\vspace*{2cm}\ansline
\end{itemize}}
