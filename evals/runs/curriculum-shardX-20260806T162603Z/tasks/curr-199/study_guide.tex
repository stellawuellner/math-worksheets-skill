\sstitleblock{Biased Samples and Mismatched Denominators}

% =================== SKILL 1 ===================
\skillheading{Getting the Denominator Right}

\begin{formulabox}
\textbf{Rule:} the bottom of a probability counts \emph{every} equally likely
outcome the draw could produce --- not one of the groups inside it.
\[
  P(\text{event}) = \frac{\text{favourable outcomes}}{\text{all outcomes}}
\]
\textbf{Tell:} ``5 red and 7 blue'' gives a denominator of 12, never 7. If your
fraction is bigger than 1, the denominator is certainly wrong.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A tin holds 6 red and 9 blue pegs. One is drawn at
random. Find the probability of red.
\step{The draw chooses from every peg in the tin, so the denominator is the
total $6 + 9 = 15$, not the 9 blue ones.}
\step{Six of the fifteen pegs are red, so the probability is $\frac{6}{15}$,
which reduces to $\ans{2/5}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A drawer holds 7 black socks and 13 white socks. Find the
probability of drawing a black sock.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7/20}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Spotting a Sample that Was Never Random}

\begin{formulabox}
\textbf{Rule:} a sample is random only when every member of the population had
the same chance of being chosen. If not, correct arithmetic still produces a
useless estimate.\par
\textbf{Tells:} the group was gathered \emph{because} of the thing being
measured (cinema-goers asked about cinemas), or people put themselves in by
replying, volunteering or turning up.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad At a football match, 24 of the 30 people asked said
football is their favourite sport. A student scales this to the town's 6000
residents. What does the arithmetic give, and why is it wrong?
\step{Scaling a sample means multiplying the population by the sample fraction,
so do that first and then ask where the sample came from.}
\step{$\frac{24}{30} \times 6000 = \ans{4800}$ residents --- correct arithmetic,
worthless estimate, because everyone asked was already at a football match.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Outside a library, 18 of 20 people asked said they read
daily. What would this sample estimate for a town of 5000, and why should nobody
believe it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4500}$, and the sample is readers}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{When Something Is Removed, Recount Both Numbers}

\begin{formulabox}
\textbf{Rule:} once an item is taken away and not replaced, the probability is
computed from what is \emph{left}. If the item removed was one of the favourable
ones, the top drops by one \textbf{and} the bottom drops by one. If it was not
favourable, only the bottom drops.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A jar holds 5 red and 7 green beads. One red bead is
taken out and kept. Find the probability that the next bead drawn is red.
\step{The bead removed was red, so it comes off both counts --- recount before
writing the fraction.}
\step{Four red beads are left out of 11 beads in total, so the probability is
$\ans{4/11}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A box holds 8 blue and 12 white tiles. One blue tile is
removed and kept. Find the probability that the next tile drawn is blue.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7/19}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a probability above 1 is impossible, and it almost
always means the denominator counted the wrong group. Check the bottom of the
fraction before you check anything else.
\end{watchoutbox}
