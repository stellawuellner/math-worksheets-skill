\sstitleblock{Signed Rational Numbers: Choosing the Right Move}

\skillheading{1. Adding and subtracting signed numbers}

\begin{formulabox}
\textbf{Rewrite every subtraction as an addition:} $p - q = p + (-q)$, so
$-6 - (-11)$ becomes $-6 + 11$.\par
\textbf{Then look at the signs.} Same signs: add the distances from zero and
keep the sign. Different signs: subtract the smaller distance from the larger
and keep the sign of the number further from zero.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $-5 - (-9)$.
\step{Subtraction with a negative behind it is the shape that causes sign
slips, so rewrite it as an addition before deciding anything.}
\step{$-5 + 9$: the signs differ, $9$ is further from zero, so the answer is
positive with size $9 - 5$.}

$-5 - (-9) = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $-7 - (-2)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$}
\end{tryitbox}

\skillheading{2. Multiplying and dividing signed numbers}

\begin{formulabox}
Work out the sizes first, then fix the sign by \textbf{counting the negative
factors}: an even count gives a positive answer, an odd count a negative one.
The same rule covers division, because dividing is multiplying by the
reciprocal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $(-6)(-7)$.
\step{Both numbers are negative, so the size and the sign can be settled
separately --- multiply the sizes, then count the minus signs.}
\step{$6 \times 7 = 42$, and there are two negative factors, an even count, so
the product is positive.}

$(-6)(-7) = \ans{42}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $-45 \div 9$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-5}$}
\end{tryitbox}

\skillheading{3. Comparing rationals in different forms}

\begin{formulabox}
Put both numbers in the \textbf{same form} first --- dividing the fraction out
to a decimal is usually quickest. Then use the number line: further left is
smaller, so among negatives, the one \emph{further from zero} is the smaller
number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $<$ or $>$ between $-\frac{2}{5}$ and $-0.5$.
\step{A fraction and a decimal cannot be compared by eye, so convert one of
them and compare like with like.}
\step{$-\dfrac{2}{5} = -0.4$, and $0.4$ is closer to zero than $0.5$, so
$-0.4$ sits to the right of $-0.5$.}

$-\dfrac{2}{5} \ans{>} -0.5$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$ or $>$ between $-\frac{7}{8}$ and
$-0.85$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{<}$}
\end{tryitbox}

\skillheading{4. Mixed expressions: the order that decides}

\begin{formulabox}
Brackets, then exponents, then multiplication and division \textbf{left to
right}, then addition and subtraction left to right. Equal-rank operations are
never swapped: $-24 \div (-6) \cdot (-2)$ divides first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $-2 - 3(-4)$.
\step{Two operations of different rank are competing, so the multiplication has
to be finished before the subtraction can happen.}
\step{$3(-4) = -12$, so the expression is $-2 - (-12) = -2 + 12$.}

$-2 - 3(-4) = \ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $-5 + \frac{6^2}{-9}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-9}$}
\end{tryitbox}
