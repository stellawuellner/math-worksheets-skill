\sstitleblock{Multiplying Fractions and Mixed Numbers}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Fraction times a whole number}

\begin{formulabox}
\textbf{Rule:} multiply the whole number by the numerator and keep the
denominator: the whole number counts how many copies of the fraction you
take. If the answer is bigger than 1, rewrite it as a mixed number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $\tfrac{3}{4} \times 8$.
\step{Take 8 copies of three fourths --- multiply 8 by the numerator and keep fourths.}
\step{$\tfrac{3 \times 8}{4} = \tfrac{24}{4} = \ans{6}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $\tfrac{2}{3} \times 12$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Fraction times a fraction}

\begin{formulabox}
\textbf{Rule:} multiply the numerators together and the denominators
together, then simplify:
$\tfrac{a}{b} \times \tfrac{c}{d} = \tfrac{a \times c}{b \times d}$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $\tfrac{2}{3} \times \tfrac{3}{5}$.
\step{Tops together and bottoms together, then look for a common factor to simplify.}
\step{$\tfrac{2 \times 3}{3 \times 5} = \tfrac{6}{15}$, and dividing top and bottom by 3 gives $\ans{\tfrac{2}{5}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $\tfrac{3}{4} \times \tfrac{2}{5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{3}{10}}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Multiplying mixed numbers}

\begin{formulabox}
\textbf{Plan:} rewrite every mixed number as an improper fraction
\emph{first}, multiply as in Skill 2, then change the answer back to a mixed
number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $1\tfrac{1}{2} \times 2\tfrac{1}{3}$.
\step{Rewrite both as improper fractions before multiplying anything.}
\step{$1\tfrac{1}{2} = \tfrac{3}{2}$ and $2\tfrac{1}{3} = \tfrac{7}{3}$, so $\tfrac{3}{2} \times \tfrac{7}{3} = \tfrac{21}{6} = \tfrac{7}{2} = \ans{3\tfrac{1}{2}}$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $2\tfrac{1}{2} \times 1\tfrac{1}{5}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} never multiply the whole parts and fraction parts
separately. $1\tfrac{1}{2} \times 2\tfrac{1}{3}$ is \emph{not}
$2\tfrac{1}{6}$ --- rewrite as improper fractions first.
\end{watchoutbox}
