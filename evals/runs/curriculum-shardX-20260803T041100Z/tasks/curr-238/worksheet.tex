\wstitleblock{Using Dilations to Reason About Similarity}{Grade 8 / Pre-Algebra --- Geometry}{}

\noindent\small Image coordinates, scale factor from lengths, slope under a dilation, proportions in similar figures, and how area scales

\medskip
\noindent\textbf{Notation and rules used on this sheet:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item A \textbf{dilation} with centre $C$ and scale factor $k$ sends each point $P$ to $P'$ so that $P'$ lies on ray $CP$ with $CP' = k \cdot CP$.
  \item Centre at the origin: $(x, y) \longmapsto (kx, ky)$. \quad Centre at $(h, m)$: $(x, y) \longmapsto \bigl(h + k(x-h),\ m + k(y-m)\bigr)$.
  \item A dilation multiplies every length by $k$ and leaves every angle measure unchanged, so a figure and its image are \textbf{similar}.
  \item Show the ratio you divided, and label every answer with its unit.
\end{itemize}
\normalsize

\problem[5cm]{A logo is drawn on a coordinate grid. The corner point is $P(3, 5)$,
in grid units. The design is dilated about the origin with scale factor $k = 2$.
Find the coordinates of the image point $P'$. Give the $x$-coordinate and the
$y$-coordinate separately, and say in one sentence how you got each one.}

\problem[5cm]{The same logo has a second corner at $Q(8, -6)$, in grid units.
This time it is dilated about the origin with scale factor $k = 0.5$.
Find the coordinates of the image point $Q'$, and state whether the image is an
enlargement or a reduction of the original.}

\problem{On the grid, segment $\overline{AB}$ runs from $A(0, 0)$ to $B(6, 8)$,
in grid units. After a dilation about the origin, its image $\overline{A'B'}$ runs
from $A'(0, 0)$ to $B'(9, 12)$.
\begin{enumerate}[label=(\alph*), itemsep=2.8cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the length $AB$.\par\vspace*{1cm} \answerline{units}
  \item Find the length $A'B'$. \answerline{units}
  \item Use the two lengths to find the scale factor $k$ of the dilation. \ansline
\end{enumerate}}

\problem[6cm]{A ramp is drawn as the segment from $(2, 3)$ to $(6, 9)$ on a grid.
It is dilated about the origin with scale factor $k = 3$, giving the image segment
from $(6, 9)$ to $(18, 27)$.
Find the slope of the original segment and the slope of the image segment.
Then state what your two answers show about the two segments.}

\problem[6.5cm]{Triangle $ABC$ is dilated about the origin with scale factor $k = 3$
to give triangle $A'B'C'$.
Explain, in two or three complete sentences, why $\triangle A'B'C'$ must be
\emph{similar} to $\triangle ABC$. Your explanation must say what the dilation does
to the three side lengths and what it does to the three angle measures.
\noansline}

\problem{A photograph measures $4$ centimetres by $6$ centimetres.
It is enlarged for a display so that the $6$ centimetre side becomes $15$ centimetres.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the scale factor of the enlargement. \ansline
  \item Find the length of the enlarged short side, in centimetres.\par\vspace*{1cm} \answerline{cm}
\end{enumerate}}

\problem[6cm]{A dilation has centre $C(1, 1)$ --- \emph{not} the origin --- and scale
factor $k = 3$. The point $M(3, 2)$ is on the design.
Use $(x, y) \longmapsto \bigl(h + k(x-h),\ m + k(y-m)\bigr)$ with $h = 1$ and $m = 1$
to find the image point $M'$. Give the $x$-coordinate and the $y$-coordinate separately.}

\problem{A triangular sail has sides of $6$ centimetres and $8$ centimetres on a
scale model. A larger sail is similar to the model, and the side matching the
$6$ centimetre side measures $9$ centimetres.
\begin{enumerate}[label=(\alph*), itemsep=3cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the scale factor from the model to the larger sail. \ansline
  \item Find the length of the side matching the $8$ centimetre side, in centimetres.\par\vspace*{1cm} \answerline{cm}
\end{enumerate}}

\problem{A rectangular tile is drawn with vertices $(0,0)$, $(4,0)$, $(4,3)$ and
$(0,3)$, in grid units. It is dilated about the origin with scale factor $k = 2$,
giving image vertices $(0,0)$, $(8,0)$, $(8,6)$ and $(0,6)$.
\begin{enumerate}[label=(\alph*), itemsep=2.6cm, leftmargin=1.5cm, topsep=4pt]
  \item Find the area of the original tile.\par\vspace*{1cm} \answerline{square units}
  \item Find the area of the image tile. \answerline{square units}
  \item Divide the image area by the original area. Explain how that number is related to $k$. \ansline
\end{enumerate}}

\problem{On a scale drawing of a school yard, a flagpole is $8$ centimetres tall
and the wall beside it is $6$ centimetres wide. The real wall is $21$ metres wide.
The drawing and the real yard are similar figures.
Find the scale factor from the drawing to the real yard, then find the real height of
the flagpole in metres. Show the ratio you used.\par\vspace*{7cm}
\answerline{m}}
