\sstitleblock{Dilations: Making and Measuring Similar Figures}

\noindent\small A dilation with centre $C$ and scale factor $k$ slides every
point along the ray from $C$ through it, until it stands $k$ times as far from
$C$. Angles never change and every length is multiplied by $k$, so the image is
\emph{similar} to the original --- same shape, resized.\normalsize

\skillheading{Dilating a point on the coordinate plane}
\begin{formulabox}
\textbf{Rule:} Centred at the origin: $(x, y) \mapsto (kx, ky)$.\par
Centred at $(h, j)$: $(x, y) \mapsto \big(h + k(x - h),\ j + k(y - j)\big)$ ---
measure from the centre, multiply by $k$, then step back out.\par
$k > 1$ enlarges, $0 < k < 1$ shrinks, and the centre itself never moves.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Dilate $A(4, -6)$ about the origin with $k = 0.5$.
\step{The centre is the origin, so no shifting is needed --- just multiply both
coordinates by the scale factor.}
\step{$0.5 \times 4 = 2$ and $0.5 \times (-6) = -3$.}

$A' = \ans{(2,\, -3)}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad Dilate $B(-3, 10)$ about the origin with $k = 3$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(-9,\, 30)}$}
\end{tryitbox}

\skillheading{Finding the scale factor from two lengths}
\begin{formulabox}
\textbf{Rule:} $k = \dfrac{\text{image length}}{\text{matching pre-image
length}}$ --- image on top, always.\par
Any pair of corresponding sides gives the same $k$; if two pairs disagree, the
figures are not similar and no dilation maps one to the other.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A $12$ cm segment is dilated onto an $18$ cm segment.
Find $k$.
\step{Scale factor compares image with pre-image, so divide --- image on top,
because the question asks what the original was multiplied by.}
\step{$k = \dfrac{18}{12}$, and dividing both parts by $6$ reduces it.}

$k = \ans{\frac{3}{2}}$, an enlargement
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A $20$ mm segment is dilated onto an $8$ mm segment. Find
$k$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{2}{5}}$}
\end{tryitbox}

\skillheading{Using the scale factor on a similar figure}
\begin{formulabox}
\textbf{Rule:} Under a dilation with factor $k$: every length (side, perimeter,
diagonal, height) is multiplied by $k$; every angle is unchanged; every area is
multiplied by $k^2$.\par
Perimeter is a length, so it takes $k$. Area is a product of two lengths, so it
takes $k$ twice.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A triangle has sides $5$, $6$ and $9$ cm and area $12$
square cm. Dilate it with $k = 3$. Find the new perimeter and area.
\step{Perimeter is built from lengths, so it scales by $k$; area is built from
two lengths, so it scales by $k^2$ --- decide which before computing.}
\step{Perimeter: $3 \times (5 + 6 + 9) = 3 \times 20$. Area:
$12 \times 3^2 = 12 \times 9$.}

Perimeter $\ans{60}$ cm, area $\ans{108}$ square cm
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A rectangle has perimeter $24$ cm and area $32$ square
cm. Dilate it with $k = 0.5$.\\[2pt]
\rotatebox{180}{\footnotesize check: perimeter $\ans{12}$, area $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Do not scale an area by $k$. A model at
$k = \frac{1}{12}$ has $\frac{1}{144}$ of the original area, not
$\frac{1}{12}$. And when the centre is not the origin, do not just multiply the
coordinates --- subtract the centre first, or the image lands in the wrong
place.
\end{watchoutbox}
