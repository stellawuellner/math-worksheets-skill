\sstitleblock{Find the Error: Rounding and Place Value}

\skillheading{Skill 1 --- Rounding Across a Nine}

\begin{formulabox}
\textbf{Rule:} when the digit you are rounding is a 9 (or the block is 49, 99,
199 \ldots) and the look-right digit says ``round up,'' the round-up
\textbf{carries} into the next place. Expect more than one digit to change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $5{,}971$ to the nearest hundred.
\step{The hundreds digit is 9 and the tens digit is $7 \geq 5$, so the 59
hundreds must round up to 60 hundreds --- a carry into the thousands.}
\step{60 hundreds $= 6{,}000$, and the rest zero out.}
$\Rightarrow\quad \ans{6{,}000}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $2{,}996$ to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3{,}000}$}
\end{tryitbox}

\skillheading{Skill 2 --- Zeros That Hold a Place}

\begin{formulabox}
\textbf{Rule:} a zero is not ``nothing'' --- it holds an empty place so every
other digit keeps its value. Dropping a placeholder zero shifts digits into the
wrong places. In products like $30 \times 40$, count \textbf{every} factor of
ten.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write $70{,}000 + 400 + 2$ in standard form.
\step{Fill all five places, putting a zero in each empty one: 7 ten-thousands,
0 thousands, 4 hundreds, 0 tens, 2 ones.}
\step{$70{,}000 + 400 + 2 = 70{,}402$}
$\Rightarrow\quad \ans{70{,}402}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $60 \times 50 = $ ? \ (Count the factors of ten!)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3{,}000}$}
\end{tryitbox}

\skillheading{Skill 3 --- What Rounding Does (and Does Not) Claim}

\begin{formulabox}
\textbf{Rule:} rounding replaces a number with the \textbf{nearest} number in
the named place. Many different numbers share one rounded value --- so equal
rounded values never prove the original numbers are equal.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round 618 and 624 to the nearest ten.
\step{Check each ones digit: 618 has $8 \geq 5$ (round up); 624 has $4 < 5$
(round down).}
\step{618 rounds up to 620, and 624 rounds down to 620 --- different numbers,
same rounded value.}
$\Rightarrow\quad \ans{620}$ and $\ans{620}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round 87 and 92 to the nearest ten.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{90}$ and $\ans{90}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the look-right digit decides \emph{up or down}; the
carry decides \emph{how many digits change}. Check both before you write your
answer.
\end{watchoutbox}
