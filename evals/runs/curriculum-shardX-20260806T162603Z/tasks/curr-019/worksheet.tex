\wstitleblock{Number Bond Fix-It Practice}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item A number bond shows a \textbf{whole} on top and two \textbf{parts} below.
  \item The two parts always add up to the whole.
  \item Some bonds and fact families on this page are broken. Find what is
        missing or wrong, and fix it!
\end{itemize}

\problem[3cm]{Look at the number bond. The whole is 8. One part is 3.
Write the missing part.
\par\vspace{6pt}
\begin{center}
\begin{tikzpicture}[scale=0.9, line width=0.8pt]
  \draw (0,1.0) circle (0.5);  \node at (0,1.0) {\large 8};
  \draw (-1.0,-0.45) circle (0.5); \node at (-1.0,-0.45) {\large 3};
  \draw (1.0,-0.45) circle (0.5);  \node at (1.0,-0.45) {\large ?};
  \draw (-0.30,0.60) -- (-0.70,-0.05);
  \draw (0.30,0.60) -- (0.70,-0.05);
\end{tikzpicture}
\end{center}
$3\;+$~\ansblank~$=\;8$}

\problem[3.5cm]{Sam wrote the fact family for the bond below:
\ $3+7=10$, \ $7+3=10$, \ $10-3=7$, \ $10-3=7$.
