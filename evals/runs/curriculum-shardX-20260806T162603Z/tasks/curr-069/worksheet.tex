\wstitleblock{Multiplication: Rows, Arrays, and the Total}{Grade 3}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item An array has \textbf{rows}, and every row holds the \textbf{same number} of things.
  \item The \textbf{total} counts everything in the array --- multiply the rows by the number in each row.
  \item Watch out! Some problems show work that mixes up the rows and the total. Your job is to find the mistake and fix it.
\end{itemize}

\problem[3cm]{This array has 3 rows with 5 dots in each row.
\begin{center}
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[2pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[2pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$
\end{center}
How many dots are there in all?}

\problem[3.5cm]{Nadia packs 4 bags for a picnic. She puts 6 apples in each bag. How many apples does she pack in all?}

\problem[3.5cm]{The chairs in the music room are set up in an array: 5 rows with 7 chairs in each row. How many chairs are in the music room in all?}

\problem{Maya's class made a sticker array with 4 rows and 8 stickers in each row. Maya wrote $4 + 8 = 12$ and says the array has 12 stickers.
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] What did Maya do wrong?\par\vspace{6pt}\noindent\underline{\hspace{12.5cm}}\par\vspace{14pt}\noindent\underline{\hspace{12.5cm}}
  \item[(b)] Find the correct number of stickers.\par\vspace*{2.6cm}\ansline
\end{itemize}}

\problem[3.5cm]{A muffin tray is set out in rows. Each row holds 6 muffins, and there are 24 muffins in all. How many rows are on the tray? You can use the equation $n \times 6 = 24$.}

\problem{Kai looks at this array of 5 rows with 5 dots in each row.
\begin{center}
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[2pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[2pt]

$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$
\end{center}
Kai says: ``$5 + 5 = 10$, so there are 10 dots.''
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] Say why Kai's answer does not count the whole array.\par\vspace{6pt}\noindent\underline{\hspace{12cm}}\par\vspace{14pt}\noindent\underline{\hspace{12cm}}

\problem{Rosa sets up chairs for a play: 7 rows with 8 chairs in each row. She counts the 7 rows, counts the 8 chairs in one row, and writes $7 + 8 = 15$ chairs.

  \item[(a)] Explain what Rosa's 15 really counted, and why it is not the total.\par\vspace{6pt}\noindent\underline{\hspace{11.5cm}}\par\vspace{14pt}\noindent\underline{\hspace{11.5cm}}
