% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Writing a Construction Plan}

\begin{formulabox}
\textbf{Legal moves:} set the compass to a span; strike an arc from a marked
point; draw a line, ray or segment through two marked points; mark a crossing.
Measuring with a ruler or protractor is not a move.\\[3pt]
\textbf{A plan says three things per step:} where the compass point sits, what
the span is, and whether the span changes. "Keep the same setting" is
information, not decoration --- it is what the justification will lean on.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plan a copy of $\overline{AB}$, where $A$ is at $(0, 0)$
and $B$ is at $(8, 6)$, onto a ray starting at $C$.
\step{The copy has to be congruent, not approximately equal, so the span itself must be transferred --- that rules out reading a length and re-drawing it.}
\step{Draw the ray from $C$; open the compass from $A$ to $B$ and do not change it; put the compass point on $C$ and strike an arc across the ray; mark the crossing. Its distance from $C$ is $\sqrt{64 + 36}$.}

$\Rightarrow\quad$ the copy must measure $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad You copy $\overline{PQ}$ with $P$ at $(1, 1)$ and $Q$ at
$(4, 5)$. What length must the copy have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Justifying with Equal Radii}

\begin{formulabox}
\textbf{The whole argument in one line:} arcs of equal radius create congruent
segments; congruent segments create congruent triangles; congruent triangles
hand you the angle or the perpendicular you claimed.\\[3pt]
\textbf{Template:} name the equal radii, name the shared side, cite side-side-side,
then name the corresponding part you wanted. A justification that only re-tells
the drawing has skipped every one of those.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Arcs of one radius centred at $A(0, 0)$ and at $B(8, 0)$
cross at $P(4, 3)$. Justify that $P$ is on the perpendicular bisector of
$\overline{AB}$.
\step{The two arcs were struck with one compass setting, so the claim to argue from is that $P$ is the same distance from each endpoint --- not that the picture looks symmetric.}
\step{From $A$ the run is $4$ and the rise is $3$; from $B$ the run is $4$ the other way and the rise is $3$. Both give $\sqrt{16 + 9}$, so the two distances are equal and $P$ is equidistant from the endpoints.}

$\Rightarrow\quad PA = PB = \ans{5}$, so $P$ lies on the perpendicular bisector.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Arcs of one radius centred at $(0, 0)$ and at $(10, 0)$
cross at $(5, 12)$. Find the distance to each endpoint.\\[2pt]
\rotatebox{180}{\footnotesize check: both are $\ans{13}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Checking a Construction on Coordinates}

\begin{formulabox}
\textbf{Distance formula:} the run and the rise are the legs of a right
triangle, so the distance between two points is $\sqrt{(\text{run})^{2} +
(\text{rise})^{2}}$.\\[3pt]
\textbf{Use it to audit a claim:} a midpoint claim means two equal halves; an
equidistant claim means two equal distances; a dilation claim means the two
distances from the centre are in the ratio of the scale factor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A construction claims $N(3, 5)$ is the midpoint of
$\overline{AB}$, where $A$ is at $(-1, 2)$ and $B$ is at $(7, 8)$. Check it.
\step{A midpoint claim is two claims at once --- equal halves and lying on the segment --- so compare the two halves rather than trusting the picture.}
\step{From $A$ to $N$ the run is $4$ and the rise is $3$; from $N$ to $B$ the run is $4$ and the rise is $3$ again. Both halves are $\sqrt{16 + 9}$, and the matching run and rise also show $N$ is on the segment.}

$\Rightarrow\quad AN = NB = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Is $(3, 5)$ the midpoint of the segment from $(-3, 1)$ to
$(9, 9)$? Give each half to two decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: both halves are $\ans{7.21}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} equal distances alone prove a point is on the
perpendicular bisector, but they do \emph{not} prove a point is a midpoint ---
for that the point must also lie on the segment. And a plan is not finished
until it says which spans stay unchanged.
\end{watchoutbox}
