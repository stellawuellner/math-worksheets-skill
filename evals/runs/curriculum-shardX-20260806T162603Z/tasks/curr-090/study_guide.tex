% =================== SKILL 1: time-intervals ===================
\skillheading{Time: Find the Used Minutes First}

\begin{formulabox}
\textbf{Rule:} Step 1 --- find the minutes already used (add them, or multiply
equal rounds). Step 2 --- subtract from the total time to answer the question.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Art class is 50 minutes. Painting took 15 minutes, then
cleanup took 10 minutes. How many minutes are left?
\step{The question asks for time left, but first I need the hidden number:
how many minutes are already used.}
\step{Used: $15 + 10 = 25$ minutes. Left: $50 - 25 = 25$.}

Minutes left: $\ans{25\ \text{minutes}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Music class is 40 minutes. Singing took 12 minutes and drums
took 18 minutes. How many minutes are left?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10\ \text{minutes}}$}
\end{tryitbox}

% =================== SKILL 2: money ===================
\skillheading{Money: Find the Total Cost First}

\begin{formulabox}
\textbf{Rule:} Step 1 --- find the total cost (add the prices, or multiply
price $\times$ how many). Step 2 --- subtract from the money paid to find the
change.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Toy cars cost 3 dollars each. Ali buys 2 cars and pays with
a 10-dollar bill. How many dollars does he get back?
\step{Change is money paid minus cost, so the hidden first step is the total
cost of the cars.}
\step{Cost: $2 \times 3 = 6$ dollars. Change: $10 - 6 = 4$.}

Ali gets back $\ans{4\ \text{dollars}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Notebooks cost 4 dollars each. Min buys 2 notebooks and pays
with a 10-dollar bill. How many dollars does she get back?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2\ \text{dollars}}$}
\end{tryitbox}

% =================== SKILL 3: mass ===================
\skillheading{Mass: Build the Load Piece by Piece}

\begin{formulabox}
\textbf{Rule:} Step 1 --- find the mass of the equal group (how many
$\times$ mass of one). Step 2 --- add or subtract the other mass. Keep every
mass in the \emph{same unit}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} One bag of flour is 300 g. A basket holds 2 bags of flour
and one 150 g box of sugar. What is the total mass?
\step{Two same-size bags make an equal group, so multiply first, then add the
sugar box.}
\step{Bags: $2 \times 300 = 600$ g. Total: $600 + 150 = 750$.}

Total mass: $\ans{750\ \text{g}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} One can is 200 g. A shelf holds 3 cans and one 250 g box.
What is the total mass?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{850\ \text{g}}$}
\end{tryitbox}

% =================== SKILL 4: liquid-volume ===================
\skillheading{Liquid Volume: How Much Is In, How Much Is Left}

\begin{formulabox}
\textbf{Rule:} Step 1 --- find how much liquid went in (how many
$\times$ size of each). Step 2 --- subtract from the full amount to find what
is still missing or left over.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A tank holds 15 L when full. Ray pours in 3 buckets of 4 L
each. How many liters are still needed?
\step{First find the hidden number --- how many liters the equal buckets put
in --- then compare with the full tank.}
\step{Poured in: $3 \times 4 = 12$ L. Still needed: $15 - 12 = 3$.}

Still needed: $\ans{3\ \text{L}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A pool tub holds 20 L when full. Jen pours in 4 buckets of
3 L each. How many liters are still needed?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8\ \text{L}}$}
\end{tryitbox}
