\sstitleblock{Classifying Quadrilaterals from Their Properties}

\begin{formulabox}
\textbf{The defining properties.} Each name adds a condition to the one above it.
\begin{itemize}[leftmargin=0.6cm, itemsep=0pt, topsep=1pt]
  \item \textbf{trapezoid:} exactly one pair of parallel sides
        (\emph{isosceles} if the legs are congruent, equivalently if each pair of
        base angles is congruent)
  \item \textbf{parallelogram:} both pairs of opposite sides parallel ---
        equivalently opposite sides congruent, or opposite angles congruent, or
        diagonals bisecting each other
  \item \textbf{rectangle:} a parallelogram with a right angle --- equivalently
        with congruent diagonals
  \item \textbf{rhombus:} a parallelogram with four congruent sides ---
        equivalently with perpendicular diagonals
  \item \textbf{square:} both a rectangle and a rhombus
  \item \textbf{kite:} two pairs of \emph{adjacent} congruent sides
\end{itemize}
\end{formulabox}

\skillheading{Skill 1 --- The angle sum of a quadrilateral}

\begin{examplebox}
\textbf{Example.} The interior angles of a quadrilateral measure $(2x)^\circ$,
$(4x)^\circ$, $(5x)^\circ$ and $(7x)^\circ$. Find $x$.
\step{Every quadrilateral splits into two triangles, so its angles add to
$360^\circ$ --- that one fact turns the four expressions into a single
equation.}
\step{$2x + 4x + 5x + 7x = 360 \;\Rightarrow\; 18x = 360$}

$x = \ans{20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} The interior angles of a quadrilateral measure $x^\circ$,
$(2x)^\circ$, $(3x)^\circ$ and $(6x)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{30}$}
\end{tryitbox}

\skillheading{Skill 2 --- Reading a defining property}

\begin{examplebox}
\textbf{Example.} In a parallelogram, one side measures $(2x+7)$ cm and the side
opposite it measures $23$ cm. Find $x$.
\step{The property that opposite sides of a parallelogram are congruent is what
lets the two expressions be set equal --- without it there is no equation.}
\step{$2x + 7 = 23 \;\Rightarrow\; 2x = 16$}

$x = \ans{8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} In a parallelogram, one side measures $(5x-6)$ cm and the side
opposite it measures $29$ cm. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} opposite sides are congruent, but \emph{consecutive}
angles are supplementary, not congruent. Setting the wrong pair equal is the
commonest slip on these.
\end{watchoutbox}

\skillheading{Skill 3 --- Naming the figure}

\begin{examplebox}
\textbf{Example.} In a quadrilateral both pairs of opposite angles are
congruent, and one pair measures $115^\circ$ each. Find the measure of each
angle in the other pair, and name the figure.
\step{Congruent opposite angles is a parallelogram test, so the figure is a
parallelogram; the remaining two angles are equal and take up whatever is left
of the $360^\circ$.}
\step{$\dfrac{360 - 2(115)}{2} = \dfrac{130}{2}$, and with no side or diagonal
information the most specific name stays \emph{parallelogram}.}

each remaining angle $= \ans{65}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it.} In a quadrilateral both pairs of opposite angles are congruent,
and one pair measures $100^\circ$ each. Find the measure of each angle in the
other pair.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{80}$ degrees}
\end{tryitbox}
