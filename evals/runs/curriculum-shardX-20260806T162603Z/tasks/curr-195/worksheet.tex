\problem{\textbf{Warm-up.} A closed box is $4$ cm long, $3$ cm wide, and
$2$ cm tall. Find its total surface area.
\par\vspace*{4.2cm}\answerline{cm$^2$}}

\problem{A cube-shaped tissue box has edges $5$ inches long. Find its
total surface area.
\par\vspace*{4.2cm}\answerline{in$^2$}}

\problem{A blueprint uses the scale $1$ cm $: 4$ m. A wall measures
$3.5$ cm on the blueprint. How long is the real wall?
\par\vspace*{4.2cm}\answerline{m}}

\problem{On a meter grid, an L-shaped patio has corners at $(0,0)$,
$(8,0)$, $(8,4)$, $(5,4)$, $(5,7)$, and $(0,7)$. Split it into rectangles and
find its area.
\par\vspace*{4.5cm}\answerline{m$^2$}}

\problem{A floor plan uses the scale $1$ in $: 5$ ft. A bedroom measures
$2$ in by $3$ in on the plan. Find the area of the \emph{real} bedroom.
\emph{(Careful: scale each side first.)}
\par\vspace*{4.5cm}\answerline{ft$^2$}}

\problem{A gift box is $6$ in long, $4$ in wide, and $3$ in tall.
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
  \item[(a)] How much wrapping paper is needed to cover all six faces exactly?
  \item[(b)] Paper is sold in sheets of $60$ square inches. How many whole
        sheets must you buy? \ansblank
\end{itemize}
\par\vspace*{3.6cm}\answerline{in$^2$}}

\problem{On a meter grid, a T-shaped stage has corners at $(0,0)$,
$(10,0)$, $(10,3)$, $(7,3)$, $(7,8)$, $(3,8)$, $(3,3)$, and $(0,3)$. Find its
area.
\par\vspace*{4.5cm}\answerline{m$^2$}}

\problem{A model bridge is drawn at the scale $1$ cm $: 8$ m. The real
bridge is $120$ m long. How long is the bridge on the drawing?
\par\vspace*{4.5cm}\answerline{cm}}

\problem{A logo triangle has area $7$ square centimeters. For a poster it
is enlarged by scale factor $3$ (every length becomes $3$ times as long). Find
the area of the enlarged logo.
\par\vspace*{4.5cm}\answerline{cm$^2$}}

\problem{An open-top aquarium is $60$ cm long, $30$ cm wide, and $40$ cm
tall. Glass is needed for the bottom and the four sides --- but \emph{not} the
top. How much glass is needed?
\par\vspace*{5cm}\answerline{cm$^2$}}

\problem{On a meter grid, a house-shaped sign has corners at $(0,0)$,
$(8,0)$, $(8,5)$, $(4,8)$, and $(0,5)$ --- a rectangle with a triangular top.
Find the sign's area.
\par\vspace*{5cm}\answerline{m$^2$}}

\problem{\textbf{Challenge.} An architect builds a $1:20$ scale model of a room
that is $4$ m long and $3$ m wide. (Remember: $1$ m $= 100$ cm.)
\begin{itemize}[leftmargin=1.2cm, itemsep=4pt]
  \item[(a)] Model floor length: \ansblank \quad Model floor width: \ansblank
  \item[(b)] Model floor area, in cm$^2$: \ansblank
  \item[(c)] Explain why the model's floor area is \emph{not} the real floor
        area divided by $20$. \par\vspace*{1.6cm}
\end{itemize}
