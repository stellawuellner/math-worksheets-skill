\sstitleblock{One Line, Four Ways to Show It}

\skillheading{Skill 1 --- Table to equation}

\begin{formulabox}
\textbf{Rule:} the row where the input is 0 gives $b$; the change in output
divided by the change in input gives $m$. Then write $y = mx + b$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write the equation for the table $x = 0, 2, 4$ with
$y = 5, 13, 21$.
\step{The table already contains the input 0, so $b$ can be read off and only
$m$ needs computing.}
\step{At $x = 0$ the output is 5. Across the table, $\dfrac{21 - 5}{4 - 0}
= \dfrac{16}{4}$.}

Slope in front, intercept behind. \quad $\ans{y = 4x + 5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A table gives $x = 0, 3, 6$ with $y = 7, 16, 25$. Write
its equation.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 3x + 7}$}
\end{tryitbox}

\skillheading{Skill 2 --- Context to equation}

\begin{formulabox}
\textbf{Rule:} the amount already there before anything happens is $b$; the
amount added (or lost) per unit is $m$. Words like ``per'', ``each'' and
``every'' point at the slope.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A club has 25 members and gains 9 more each month.
How many after 5 months?
\step{One number is a starting amount and the other is a rate, so sort them
before writing anything: 25 happens once, 9 happens every month.}
\step{$y = 9x + 25$, and at $x = 5$ this gives $45 + 25$.}

The rate multiplies the months. \quad $\ans{70}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A tank holds 40 litres and gains 6 litres a minute. How
many litres after 8 minutes?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{88}$}
\end{tryitbox}

\skillheading{Skill 3 --- Graph to equation}

\begin{formulabox}
\textbf{Rule:} read where the line crosses the vertical axis for $b$, then
count rise over run between two grid corners for $m$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Write the equation of the line drawn below, then find
$y$ at $x = 5$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, xmin=0, xmax=4, ymin=0, ymax=12,
  xtick={0,1,2,3,4}, ytick={0,2,4,6,8,10,12},
  grid=both, grid style={line width=0.25pt, draw=gray!55},
  major grid style={line width=0.4pt, draw=gray!70},
  tick label style={font=\tiny}, xlabel={$x$}, ylabel={$y$},
  width=6cm, height=4cm, enlargelimits=false]
  \addplot[thick, blue, domain=0:3, samples=2] {2*x + 4};
  \fill[blue] (axis cs:0,4) circle (1.7pt);
  \fill[blue] (axis cs:3,10) circle (1.7pt);
\end{axis}
\end{tikzpicture}
\end{center}
\step{The crossing point is marked on the axis, so take $b$ from there and get
$m$ from the two marked corners.}
\step{Crossing at 4; rise over run is $\dfrac{10 - 4}{3 - 0} = \dfrac{6}{3}$,
so the equation is $\ans{y = 2x + 4}$. At $x = 5$: $10 + 4$.}

Reading past the edge of the grid needs the equation. \quad $\ans{y = 14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line is marked at $(0, 1)$ and $(2, 7)$. Find its
slope.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{m = 3}$}
\end{tryitbox}

\skillheading{Skill 4 --- Deciding whether two forms match}

\begin{formulabox}
\textbf{Rule:} two representations match only when \emph{both} numbers agree.
One shared value proves nothing --- two different lines can cross.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Does the table $x = 0, 2, 4$ with $y = 8, 14, 20$
match $y = 3x + 8$?
\step{Comparing one value could be a coincidence, so test the pair of numbers
that defines a line: the starting value and the rate.}
\step{Table at $x = 4$ gives 20; the equation gives $12 + 8$, the same. Both
start at 8 and both climb 6 over 2 steps.}

Agreeing on both, so yes. \quad $\ans{20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Does the table $x = 0, 3, 6$ with $y = 10, 19, 28$
match $y = 3x + 10$? Check the value at $x = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{28}$, and yes}
\end{tryitbox}
