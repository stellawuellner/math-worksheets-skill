\sstitleblock{Limits by Factoring and Rationalizing}

\begin{formulabox}
\textbf{The whole method in one line:} if substitution gives $\frac{0}{0}$, the
numerator and denominator share a factor that is zero at the target. Cancel it,
then substitute. The cancelled function differs from the original at exactly one
point --- and a limit never looks at that point.
\end{formulabox}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Factor and Cancel a Common Factor}

\begin{formulabox}
\textbf{Patterns worth recognising:} \ $a^2 - b^2 = (a-b)(a+b)$, \
$a^3 - b^3 = (a-b)(a^2+ab+b^2)$, \ and $x^2+bx+c=(x+p)(x+q)$ with $p+q=b$,
$pq=c$. If $x \to k$ gives $\frac{0}{0}$, then $(x-k)$ is a factor of both.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to 5} \frac{x^2-25}{x-5}$
\step{Substituting $5$ gives $\frac{0}{0}$, so $(x-5)$ divides both parts --- factor and cancel rather than substitute.}
\step{$\dfrac{x^2-25}{x-5} = \dfrac{(x-5)(x+5)}{x-5} = x+5$ for $x \neq 5$, and $5+5 = 10$.}

$\Rightarrow\quad \lim_{x \to 5} \dfrac{x^2-25}{x-5} = $ $\ans{10}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to 6} \frac{x^2-36}{x-6}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Rationalize with the Conjugate}

\begin{formulabox}
\textbf{Rule:} when a square root blocks the cancelling, multiply top and bottom
by the conjugate of the part that contains it, using
$\left(\sqrt{A}-B\right)\left(\sqrt{A}+B\right) = A - B^2$. The radical
disappears from that part and the blocking factor appears.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\lim_{x \to 0} \frac{\sqrt{x+4}-2}{x}$
\step{Substituting $0$ gives $\frac{0}{0}$ and nothing factors, so clear the radical with the conjugate $\sqrt{x+4}+2$.}
\step{$\dfrac{\sqrt{x+4}-2}{x}\cdot\dfrac{\sqrt{x+4}+2}{\sqrt{x+4}+2} = \dfrac{(x+4)-4}{x\left(\sqrt{x+4}+2\right)} = \dfrac{1}{\sqrt{x+4}+2}$, and at $x=0$ that is $\frac{1}{4}$.}

$\Rightarrow\quad \lim_{x \to 0} \dfrac{\sqrt{x+4}-2}{x} = $ $\ans{\tfrac{1}{4}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to 0} \frac{\sqrt{x+25}-5}{x}$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{1}{10}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $\sqrt{x+4} \neq \sqrt{x}+2$. Splitting a root over a sum
is the single most common way this method goes wrong. Multiply by the conjugate
instead --- that is the only legal move.
\end{watchoutbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Reading a Table Before the Algebra}

\begin{formulabox}
\textbf{What a table can and cannot do:} inputs approaching the target from both
sides give a strong prediction of the limit, and they catch a sign or arithmetic
slip fast. They are evidence only --- a finite list of samples cannot prove the
value. Algebra proves it; the table tells you what to expect.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A table for $h(x)=\frac{x^2-1}{x-1}$ reads $1.99$ at
$x=0.99$ and $2.01$ at $x=1.01$. Find $\displaystyle\lim_{x \to 1} h(x)$ exactly.
\step{The two-sided table points at $2$, so I expect $2$ --- now prove it, because a table is never the proof.}
\step{$\dfrac{x^2-1}{x-1} = \dfrac{(x-1)(x+1)}{x-1} = x+1$ for $x \neq 1$, and $1+1 = 2$, matching the prediction.}

$\Rightarrow\quad \lim_{x \to 1} h(x) = $ $\ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\lim_{x \to 1} \frac{x^3-1}{x-1}$
exactly, then predict what a table near $x=1$ would show.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$}
\end{tryitbox}
