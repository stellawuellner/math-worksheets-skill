\sstitleblock{Picking a Method, and Reading the Roots}

\skillheading{Skill 1 --- When the squared part is already alone}

\begin{formulabox}
\textbf{Use square roots when} the equation looks like $(\text{something})^2 = k$. Get the square alone, root both sides, write $\pm$ straight away, then undo whatever is inside.\par\smallskip
Do \emph{not} expand $(x-2)^2$ first --- that turns two steps into five.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $(x-2)^2 = 9$.
\step{The squared bracket is already by itself and there is no other $x$ in the equation, so square roots are the shortest method.}
\step{$x - 2 = \pm 3$, so $x = 2 + 3$ or $x = 2 - 3$.}
\par
$x = \ans{-1}$ \ or \ $x = \ans{5}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $(x+1)^2 = 4$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-3}$ or $x = \ans{1}$}
\end{tryitbox}

\skillheading{Skill 2 --- Factor, then use the zero product}

\begin{formulabox}
\textbf{Use factoring when} one side is $0$ and the other factors quickly. Then $A \cdot B = 0$ forces $A = 0$ or $B = 0$.\par\smallskip
If there is no constant term, the common factor is $x$ --- factor it out, never divide by it, or the root $x = 0$ is lost.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 - x - 12 = 0$.
\step{One side is already zero and the leading coefficient is $1$, so a quick product-and-sum search is faster than the formula.}
\step{Product $-12$, sum $-1$: the pair is $-4$ and $3$, so $(x-4)(x+3) = 0$.}
\par
$x = \ans{-3}$ \ or \ $x = \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^2 + 6x + 8 = 0$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{-4}$ or $x = \ans{-2}$}
\end{tryitbox}

\skillheading{Skill 3 --- When nothing factors: the formula}

\begin{formulabox}
\textbf{Formula:} \quad $x = \dfrac{-b \pm \sqrt{b^2-4ac}}{2a}$\par\smallskip
It always works, so use it when a factor pair does not appear within a few seconds. Write $a$, $b$, $c$ with their signs first; the denominator is $2a$, not $2$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $x^2 - 2x - 4 = 0$; give the larger root to two decimal places.
\step{No pair of whole numbers multiplies to $-4$ and adds to $-2$, so factoring is a dead end and the formula is the method.}
\step{$a = 1$, $b = -2$, $c = -4$, so $-b = 2$ and the discriminant is $4 + 16 = 20$: $x = \dfrac{2 + \sqrt{20}}{2}$ for the larger root.}
\par
$x \approx \ans{3.24}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $x^2 + x - 3 = 0$; give the larger root to two decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $x \approx \ans{1.30}$}
\end{tryitbox}

\skillheading{Skill 4 --- Reading the roots back into the story}

\begin{formulabox}
\textbf{Two questions, in order:} first solve the equation and keep \emph{every} root; then ask which roots make sense for the quantity in the story. A root that fails the story is still a correct solution of the equation --- it is rejected for meaning, not for arithmetic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A stone is thrown up from the ground; after $t$ seconds its height in feet is $-16t^2 + 64t$. When does it land?
\step{Landing means the height is zero, so solve the height equation for $t$ and only then decide which root answers the question.}
\step{$-16t(t-4) = 0$, so $t = 0$ or $t = 4$. Both are real times; $t = 0$ is the throw itself.}
\par
$t = \ans{0}$ \ or \ $t = \ans{4}$, and the stone lands after $4$ seconds.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same throw, but the height is $-16t^2 + 48t$. Solve for $t$.\\[2pt]
\rotatebox{180}{\footnotesize check: $t = \ans{0}$ or $t = \ans{3}$}
\end{tryitbox}
