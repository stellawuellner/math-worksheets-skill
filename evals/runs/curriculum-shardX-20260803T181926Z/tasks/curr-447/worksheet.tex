\wstitleblock{Normal-Model $z$-Scores and Percentiles}{Precalculus / Advanced Statistics}{}

\begin{center}
\textit{Computing z-scores, reading percentiles from a standard normal table, reversing the process to find a value, and comparing scores from different distributions}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Standardise with $z = \dfrac{x - \mu}{\sigma}$, and go back with $x = \mu + z\sigma$.
  \item Every proportion in the table below is the area to the LEFT of that $z$. For an area to the right, subtract from $1$.
  \item Round $z$-scores to two decimal places and proportions to four, and state each answer as a proportion or a percentile as asked.
\end{itemize}

\problem[4cm]{Scores on a national test follow a normal model with mean $\mu = 500$ and standard deviation $\sigma = 100$. Find the $z$-score of a score of $640$, and say in one sentence what that number means.}

\problem[4cm]{For the same test, find the $z$-score of a score of $430$. Say whether the score is above or below the mean, and by how many standard deviations.}

\problem[4cm]{Use the table above. What proportion of a normal distribution lies below $z = 1.00$? Write that proportion, then state the matching percentile.}

\problem[4.5cm]{Use the table. What proportion of a normal distribution lies below $z = 1.50$, and what proportion lies ABOVE it? Show the subtraction you used for the second answer.}

\problem[4.5cm]{Adult heights in a population are modelled as normal with $\mu = 170$ cm and $\sigma = 8$ cm. Find the $z$-score of a height of $182$ cm, then use the table to find the proportion of adults shorter than $182$ cm.}

\problem[5cm]{Using the same height model, find the proportion of adults whose height is between $162$ cm and $182$ cm. Standardise both endpoints first, then combine the two table entries.}

\problem[5cm]{Still using $\mu = 170$ cm and $\sigma = 8$ cm: which height sits at the $90$th percentile? Use the table to find the closest $z$-score to that percentile, then convert it back to a height.}

\problem[5cm]{On the national test ($\mu = 500$, $\sigma = 100$), a scholarship goes to the top $2.28\%$ of candidates. Use the table to find the $z$-score that cuts off that top group, then find the minimum score a candidate needs.}

\problem[5.5cm]{Ana scored $640$ on the national test ($\mu = 500$, $\sigma = 100$) and $28$ on a different exam whose scores are normal with $\mu = 22$ and $\sigma = 4$. Compute both $z$-scores, compare them, and say which result is the relatively stronger performance and why.}

\problem[4.5cm]{Of every $10{,}000$ candidates who sit the national test, about $228$ score above $700$. If one candidate is chosen at random, what is the probability that they scored above $700$? Give the answer as a fraction in lowest terms.}

\problem[6cm]{A coach records the fitness-test scores of a squad of seven athletes: $12$, $14$, $16$, $18$, $20$, $22$, $24$. Treat these seven as the whole population.
\par\vspace{0.1cm}
(a) Find the mean $\mu$. \quad (b) Find the standard deviation $\sigma$. \quad (c) Find the $z$-score of the athlete who scored $24$.}

\problem[7cm]{A machine fills bottles. The fill volume is modelled as normal with $\mu = 500$ mL and $\sigma = 2.5$ mL, and the label promises at least $495$ mL.
\par\vspace{0.1cm}
(a) Find the $z$-score of $495$ mL.
\par\vspace{0.1cm}
(b) Use the table to find the proportion of bottles that are underfilled.
\par\vspace{0.1cm}
(c) The plant will accept an underfilled proportion of at most $0.1003$. Use the table to find the $z$-score with that area to its left, then find the mean $\mu$ the machine should be set to so that $495$ mL sits exactly at that $z$, keeping $\sigma = 2.5$ mL.
\par\vspace{0.1cm}
(d) State the one assumption every percentage above depends on, and what would go wrong without it.}
