\wstitleblock{Finding the Mistake in Sums of Radicals}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Testing whether two roots may be added, combining like radicals, and simplifying
before combining}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Two radical terms may be added only when the part \emph{under} the radical is
        already the same. Then you add the counts in front and leave the radical alone.
  \item A radical sign never distributes across a plus sign. When you suspect it has been
        treated as if it does, test the two sides with numbers.
  \item Leave answers in radical form unless a problem asks for a decimal.
\end{itemize}

\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem{Test the two sides of a tempting shortcut.
\begin{itemize}
  \item[(a)] $\sqrt{9} + \sqrt{16} =$~\ansblank
  \item[(b)] $\sqrt{9 + 16} =$~\ansblank
  \item[(c)] Write $<$, $>$, or $=$ between your two answers:
             (a)~\ansblank~(b)
\end{itemize}
\par\vspace*{4.5cm}}

\problem{Priya writes $\sqrt{36} + \sqrt{64} = \sqrt{36 + 64} = 10$.
\begin{itemize}
  \item[(a)] Work out the correct value of $\sqrt{36} + \sqrt{64}$. \quad
             Answer:~\ansblank
  \item[(b)] Explain what Priya did wrong, and state the rule she broke.
\end{itemize}
\par\vspace*{5.5cm}}

\problem[4.5cm]{Combine into a single radical term: \quad $3\sqrt{2} + 5\sqrt{2}$.}

\problem[4.5cm]{Combine into a single radical term: \quad $7\sqrt{5} - 2\sqrt{5}$.}

\problem{These roots are not exact, so test the shortcut with rounded decimals.
\begin{itemize}
  \item[(a)] To the nearest hundredth, $\sqrt{2} + \sqrt{3} \approx$~\ansblank
  \item[(b)] To the nearest hundredth, $\sqrt{5} \approx$~\ansblank
  \item[(c)] Write $<$, $>$, or $=$ between your two answers:
             (a)~\ansblank~(b)
\end{itemize}
\par\vspace*{4.5cm}}

\problem[5cm]{Simplify $\sqrt{8} + \sqrt{18}$ into a single radical term. Rewrite each
root first, using its largest perfect-square factor.}

\problem{Marco writes $4\sqrt{3} + 2\sqrt{3} = 6\sqrt{6}$.
\begin{itemize}
  \item[(a)] Work out the correct value of $4\sqrt{3} + 2\sqrt{3}$. \quad
             Answer:~\ansblank
  \item[(b)] Explain what Marco did wrong. Which part of the term is allowed to change,
             and which part is not?
\end{itemize}
\par\vspace*{5.5cm}}

\problem{\textbf{Challenge.} Consider $\sqrt{50} + \sqrt{32} - \sqrt{8}$.
\begin{itemize}
  \item[(a)] Simplify it into a single radical term. \quad Answer:~\ansblank
  \item[(b)] Justify why the three terms could be combined at all, even though the three
             numbers under the radicals started out different.
\end{itemize}
