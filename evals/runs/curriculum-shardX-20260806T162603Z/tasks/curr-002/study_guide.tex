% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Counting a Row of Objects}

\begin{formulabox}
\textbf{Rule:} Touch each object \textbf{one time} and say \textbf{one number}
for it. The last number you say tells \emph{how many} there are in all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Count the row:
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$
\step{Count one-to-one --- touch each dot once so no dot is skipped or counted twice.}
\step{``One, two, three, four, five, six.'' The count ends at six, so there are $\ans{6}$ dots.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Count the row:
$\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{8}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Saying numbers faster than you touch breaks the
one-number-per-object match --- that is how dots get skipped or counted twice.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Counting an Array}

\begin{formulabox}
\textbf{Rule:} In an array, count \textbf{one row}, then \textbf{count on}
through the next row. Every row of the same size adds the same number.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An array has two rows with four dots in each row. How
many dots?
\step{Count the first row, then keep the count going into the second row instead of starting over.}
\step{First row: four. Count on: ``five, six, seven, eight'' --- $4 + 4 = \ans{8}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad An array has two rows with five dots in each row. How
many dots?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Counting on a Ten-Frame}

\begin{formulabox}
\textbf{Rule:} A ten-frame has two rows of five boxes. A \textbf{full top row
is 5} --- you do not need to recount it. Count on from five for the bottom row.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A ten-frame has a full top row and two dots in the
bottom row. How many dots?
\step{The full top row is five without recounting, so start the count at five and count on.}
\step{``Six, seven'' --- $5 + 2 = \ans{7}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A ten-frame has a full top row and four dots in the
bottom row. How many dots?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{9}$}
\end{tryitbox}
