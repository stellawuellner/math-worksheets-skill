\wstitleblock{Factoring Cubes and Higher-Degree Expressions}{Algebra 2}{}

\begin{center}\small
Sum and difference of cubes, factoring by grouping, quartics in quadratic form, and reading zeros off the factors
\end{center}

\noindent\small\textbf{How to work:}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item \textbf{Always pull the GCF first.} What is left is usually a pattern you know.
  \item \textbf{Cubes:} $a^3 - b^3 = (a-b)(a^2+ab+b^2)$ and $a^3 + b^3 = (a+b)(a^2-ab+b^2)$.
        The binomial keeps the sign of the original; the trinomial's middle term takes
        the opposite sign, and its last term is never negative.
  \item \textbf{Grouping:} four terms, two at a time --- pull a GCF out of each pair and
        hope the leftover binomials match.
  \item \textbf{Quadratic form:} if the exponents are $2n$ and $n$, treat $x^n$ as a
        single letter, factor, then substitute back and keep factoring.
  \item Factor completely over the integers, and box your final factorisation.
\end{itemize}

\problem[5cm]{\textbf{Difference of cubes.} Factor completely: \quad $x^3 - 8$.\par
Name $a$ and $b$ first, then write the two factors.}

\problem[5cm]{\textbf{Sum of cubes with coefficients.} Factor completely:
\quad $27x^3 + 64$.\par
Both terms are perfect cubes even though neither is a bare power of $x$.}

\problem[5cm]{\textbf{Common factor first.} Factor completely: \quad $2x^3 - 54$.\par
Neither term is a perfect cube as written. Pull the greatest common factor out
before you look for the pattern.}

\problem[6cm]{\textbf{Factor by grouping.} Factor completely:
\quad $x^3 + 3x^2 - 4x - 12$.\par
Group the first two terms and the last two, pull a common factor out of each pair,
then factor what remains --- the leftover quadratic still factors.}

\problem[6cm]{\textbf{Quadratic in disguise.} Factor completely:
\quad $x^4 - 13x^2 + 36$.\par
Let $u = x^2$. Factor the quadratic in $u$, substitute $x^2$ back, and finish
factoring each piece.}

\problem[8cm]{\textbf{From factors to zeros.} The polynomial
$P(x) = x^3 + 2x^2 - 9x - 18$ factors by grouping.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Factor $P(x)$ completely.
  \item List all real zeros of $P$ in increasing order.
  \item Complete the sign table below by writing $+$ or $-$ for $P(x)$ on each interval.
\end{itemize}

\hline
\rule{0pt}{0.8cm} $x < -3$ & $-3 < x < -2$ & $-2 < x < 3$ & $x > 3$ \\ \hline
\rule{0pt}{1.1cm} & & & \\ \hline
\end{tabular}
\end{center}}

\problem[6cm]{\textbf{Solve a quartic.} Solve for all real $x$:
\quad $x^4 - 5x^2 + 4 = 0$.\par
Factor into four linear factors, then apply the zero-product property.
Write the solutions in increasing order.}

\problem[6cm]{\textbf{Find the mistake.} Sam factors $8x^6 - 27$ and writes
\begin{center}
$8x^6 - 27 = (2x^2 - 3)(4x^4 - 6x^2 + 9)$
\end{center}
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Explain which part of the difference-of-cubes pattern Sam broke.
  \item Write the correct complete factorisation.
\end{itemize}}

\problem[7cm]{\textbf{Everything at once.} Let $R(x) = 3x^5 - 24x^2$.
\begin{itemize}[leftmargin=1cm, itemsep=1pt, topsep=2pt]
  \item Factor $R(x)$ completely over the integers. (GCF first, then a cube pattern.)
  \item List the real zeros of $R$, and say which factor produces each one.
  \item One quadratic factor contributes no real zero. Which one, and how can you
        tell without solving it?
\end{itemize}}
