\sstitleblock{Working with Line Plots}

\skillheading{Skill 1 --- Reading a Line Plot}

\begin{formulabox}
\textbf{What the parts mean:} the numbers along the bottom are the \emph{values} measured. Each
$\times$ above a number is \emph{one thing} that had that value.\par
Height of a column $=$ how many things had that value. \quad
All the marks together $=$ how many things were measured.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many pets owners in this plot have $2$ pets?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}
  \draw[thick] (-0.6,0) -- (3.0,0);
  \foreach \i/\v in {0/1,1/2,2/3}{
    \draw[thick] (\i*1.2,-0.11) -- (\i*1.2,0.11);
    \node[below=2pt, font=\small] at (\i*1.2,-0.11) {$\v$};}
  \foreach \k in {1,2}{\node[font=\small] at (0,0.1+0.3*\k) {$\times$};}
  \foreach \k in {1,2,3,4}{\node[font=\small] at (1.2,0.1+0.3*\k) {$\times$};}
  \foreach \k in {1,2,3}{\node[font=\small] at (2.4,0.1+0.3*\k) {$\times$};}
\end{tikzpicture}
\end{center}
\step{The question names one value and asks how many, so I read a single column --- no adding across the plot is needed.}
\step{Find $2$ on the scale and count the marks stacked above it: there are $4$.}

number of owners with $2$ pets $= \ans{4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot has $3$ marks above $1$, $5$ marks above $2$, and $2$ marks above $3$.
How many things had the value $3$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2}$}
\end{tryitbox}

\skillheading{Skill 2 --- Building a Line Plot from Data}

\begin{formulabox}
\textbf{Three steps:} (1) draw a scale that covers every value in the list, with the tick marks
\emph{evenly spaced}; (2) put one $\times$ above the matching number for each value, stacking them in
a column; (3) count your marks --- there must be exactly as many marks as numbers in the list.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Plot the list $2, 3, 3, 1$.
\step{Every value in the list has to get its own mark, so I first make a scale wide enough for the smallest and largest values and then work through the list in order.}
\step{The values run from $1$ to $3$, so the scale is $1, 2, 3$. Marking each value in turn puts one $\times$ above $1$, one above $2$, and two above $3$.}

marks on the finished plot $= \ans{4}$, which matches the number of values in the list
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Plot the list $4, 2, 4, 5, 4$. How many marks should your finished plot have?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing and Summarizing the Data}

\begin{formulabox}
\textbf{Most common} $=$ the tallest column.\par
\textbf{How many more} $=$ subtract one column height from another.\par
\textbf{Difference between greatest and least} $=$ (rightmost value with a mark) $-$ (leftmost value
with a mark). Read the \emph{scale} for this one, not the column heights.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For the list $2, 3, 3, 1$, what is the difference between the greatest and the
least value?
\step{The question asks about the values themselves, not about how often they happened, so I look along the scale rather than at the heights of the columns.}
\step{The leftmost mark sits above $1$ and the rightmost above $3$, so the difference is $3 - 1$.}

difference $= \ans{2}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A plot shows the values $5, 7, 7, 8, 9$. What is the difference between the
greatest and the least value?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} counting the \emph{tick marks on the scale} is not the same as counting the
$\times$ marks. The scale shows which values are possible; the marks show how many things were
measured. Mixing them up is the most common line-plot mistake.
\end{watchoutbox}
