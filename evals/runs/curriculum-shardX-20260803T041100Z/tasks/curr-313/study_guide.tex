\sstitleblock{Using Congruence to Solve for Corresponding Parts}

\noindent\small Every problem of this type is the same three moves:
\textbf{(1)} read the congruence statement to find which parts match,
\textbf{(2)} set those two measures equal (CPCTC),
\textbf{(3)} solve the equation --- and substitute back if the question asks for the
measure rather than the variable.\normalsize

\skillheading{Skill 1 --- Corresponding sides}
\begin{formulabox}\textbf{Correspondence:} in $\triangle ABC \cong \triangle DEF$ the letters
match by position: $A \leftrightarrow D$, $B \leftrightarrow E$, $C \leftrightarrow F$.
A side is named by two letters, so $\overline{AB} \leftrightarrow \overline{DE}$,
$\overline{BC} \leftrightarrow \overline{EF}$, $\overline{AC} \leftrightarrow \overline{DF}$.\\[3pt]
\textbf{CPCTC:} corresponding parts of congruent triangles are congruent, so
$AB = DE$ --- an equation you can solve.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\triangle ABC \cong \triangle DEF$ with $AB = 3x - 1$ and $DE = 14$. Find $x$.
\step{$\overline{AB}$ and $\overline{DE}$ are named by matching positions, so CPCTC makes their measures equal --- write that equation.}
\step{$3x - 1 = 14$, so $3x = 15$ and $\ans{x = 5}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\triangle GHI \cong \triangle JKL$ with $GH = 4x + 3$ and $JK = 27$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}\end{tryitbox}

\skillheading{Skill 2 --- Corresponding angles}
\begin{formulabox}\textbf{Same rule, single letters:} in $\triangle PQR \cong \triangle STU$,
$\angle P \leftrightarrow \angle S$, $\angle Q \leftrightarrow \angle T$,
$\angle R \leftrightarrow \angle U$, and corresponding angles have equal measures.\\[3pt]
\textbf{When one measure is missing:} the three angles of either triangle add to $180$ degrees,
so find the missing angle in the triangle you know, then carry it across by CPCTC.
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\triangle PQR \cong \triangle STU$ with $m\angle Q = 6x - 8$ and $m\angle T = 40$ degrees. Find $x$.
\step{$Q$ and $T$ sit in the same position in the statement, so their measures are equal by CPCTC.}
\step{$6x - 8 = 40$, so $6x = 48$ and $\ans{x = 8}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\triangle PQR \cong \triangle STU$ with $m\angle R = 3y + 5$ and $m\angle U = 35$ degrees. Find $y$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{y = 10}$}\end{tryitbox}

\skillheading{Skill 3 --- Reading the correspondence correctly}
\begin{formulabox}\textbf{The order of the letters is the whole message.}
$\triangle ABC \cong \triangle DFE$ is \emph{not} the same statement as
$\triangle ABC \cong \triangle DEF$.
\begin{itemize}[itemsep=1pt, topsep=2pt]
  \item Write the two names one above the other and pair off the columns.
  \item Pair the letters in the order given --- never re-alphabetise the second triangle.
  \item Then a side pairs with the side named by the matching two columns.
\end{itemize}
\end{formulabox}
\begin{examplebox}\textbf{Example:}\quad $\triangle ABC \cong \triangle DFE$ with $AB = 2x + 1$ and $DF = 19$. Find $x$.
\step{Pair the columns: $A \leftrightarrow D$, $B \leftrightarrow F$, $C \leftrightarrow E$, so $\overline{AB}$ matches $\overline{DF}$ (not $\overline{DE}$).}
\step{$2x + 1 = 19$, so $2x = 18$ and $\ans{x = 9}$.}
\end{examplebox}
\begin{tryitbox}\textbf{Try it:}\quad $\triangle MNP \cong \triangle RTS$ with $MN = 5x - 2$ and $RT = 23$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 5}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} solving for $x$ is usually not the end. If the
question asks for a \emph{measure}, substitute your $x$ back into the expression --- and
check it against the other expression, which must give the same value.
\end{watchoutbox}
