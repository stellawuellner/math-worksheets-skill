\wstitleblock{Rewriting Circles and Parabolas in Standard Form}{Algebra 2}{}

\begin{center}
\small\textit{Completing the square on parabolas and circles, reading features off standard form, and building an equation from given features}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Standard forms: a parabola is $y = a(x-h)^2 + k$; a circle is $(x-h)^2 + (y-k)^2 = r^2$.
  \item To complete the square on $x^2 + px$, add $\left(\dfrac{p}{2}\right)^2$ --- and keep the equation balanced by adding it to the other side too (or subtracting it back on the same side).
  \item Factor out any leading coefficient before completing the square, and put it back afterwards.
\end{itemize}
