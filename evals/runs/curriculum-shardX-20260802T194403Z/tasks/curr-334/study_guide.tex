\sstitleblock{Quadrilaterals: Which Way Does the Implication Go?}

\noindent\small\textbf{The family tree runs one way.} square $\Rightarrow$ rhombus $\Rightarrow$
parallelogram $\Rightarrow$ quadrilateral, and square $\Rightarrow$ rectangle $\Rightarrow$
parallelogram. Reading \emph{down} the tree is always safe; reading \emph{up} it needs an extra
property, and that extra property is what every problem on the worksheet is testing.\normalsize

\skillheading{1. What a property is enough for}

\begin{formulabox}
\textbf{Upgrades that work} (each is a genuine theorem):\\[3pt]
parallelogram $+$ one right angle $\Rightarrow$ rectangle \qquad
parallelogram $+$ congruent diagonals $\Rightarrow$ rectangle\\[2pt]
parallelogram $+$ two congruent adjacent sides $\Rightarrow$ rhombus \qquad
parallelogram $+$ perpendicular diagonals $\Rightarrow$ rhombus\\[2pt]
rhombus $+$ one right angle $\Rightarrow$ square \qquad rectangle $+$ congruent adjacent sides
$\Rightarrow$ square\\[3pt]
\textbf{Not enough:} congruent sides alone never gives right angles, and congruent diagonals alone
never gives congruent sides.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A rhombus has consecutive angles $(3x + 20)^\circ$ and $(2x + 10)^\circ$. Find
$x$, then decide whether it must be a square.
\step{A rhombus is a parallelogram, so consecutive angles are supplementary --- that is the
equation to write, and the resulting angles are what decide ``square''.}
\step{$(3x + 20) + (2x + 10) = 180$, so $5x + 30 = 180$.}
$\Rightarrow\quad \ans{x = 30}$, giving angles $110^\circ$ and $70^\circ$: not right angles, so a
rhombus and \emph{not} a square.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} A rhombus has consecutive angles $(4x)^\circ$ and $(5x)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 20}$}
\end{tryitbox}

\skillheading{2. Testing a claim with side lengths}

\begin{formulabox}
\textbf{Distance formula:} $d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}$.\\[3pt]
Two \emph{adjacent} sides of unequal length kill ``rhombus'' and ``square'' instantly --- one
counterexample side pair is all it takes. Perpendicular diagonals give a rhombus only when they
also bisect each other (equal midpoints); without that you have a kite.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A(0,5)$, $B(2,0)$, $C(0,-3)$, $D(-2,0)$: the diagonals lie on the axes, so they
are perpendicular. Must $ABCD$ be a rhombus?
\step{Perpendicular diagonals are not sufficient on their own, so test two adjacent sides rather
than trusting the claim.}
\step{$AB = \sqrt{2^2 + 5^2} = \sqrt{29}$, while $BC = \sqrt{2^2 + 3^2} = \sqrt{13} \approx 3.61$.}
$\Rightarrow\quad AB \approx \ans{5.39} \neq BC$, so not a rhombus --- it is a kite.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $A(0,7)$, $B(2,0)$, $C(0,-4)$, $D(-2,0)$. Find $AB$ to two decimal places (and see
that $BC$ differs).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5.39}$ becomes $\ans{7.28}$}
\end{tryitbox}

\skillheading{3. The diagonal test for the most specific name}

\begin{formulabox}
\textbf{Rule:} once all four sides are known congruent (a rhombus), compare the \emph{diagonals}:
\\[3pt]
diagonals congruent $\Rightarrow$ square \qquad diagonals not congruent $\Rightarrow$ rhombus only.
\\[2pt]
The same comparison run on a rectangle tells you nothing new --- a rectangle always has congruent
diagonals. Order matters: sides first, then diagonals.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} $A(0,0)$, $B(3,1)$, $C(4,-2)$, $D(1,-3)$ has four congruent sides. Square or
rhombus?
\step{All sides congruent already gives a rhombus, so the deciding test is whether the diagonals
are congruent.}
\step{$AC = \sqrt{4^2 + 2^2} = \sqrt{20}$ and $BD = \sqrt{2^2 + 4^2} = \sqrt{20}$ --- equal.}
$\Rightarrow\quad AC = \ans{4.47} = BD$, so $ABCD$ is a square.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} The rhombus $E(0,0)$, $F(4,2)$, $G(8,0)$, $H(4,-2)$. Find diagonal $EG$, compare
it with $FH = 4$, and name the figure.\\[2pt]
\rotatebox{180}{\footnotesize check: $EG = \ans{8}$, so rhombus only}
\end{tryitbox}

\skillheading{4. Killing a converse with a counterexample}

\begin{formulabox}
\textbf{Rule:} to show an implication does \emph{not} run backwards, produce one figure with the
property but without the name. The classic pairs:\\[3pt]
perpendicular diagonals without a rhombus $\rightarrow$ \textbf{kite} \qquad congruent diagonals
without a rectangle $\rightarrow$ \textbf{isosceles trapezoid}\\[2pt]
The missing hypothesis in both is ``and it is a parallelogram''.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Is ``congruent diagonals $\Rightarrow$ rectangle'' always true? Test
$T(0,0)$, $U(6,0)$, $V(5,3)$, $W(1,3)$.
\step{To refute a converse I only need one figure with congruent diagonals that is not a rectangle,
so compute both diagonals of this trapezoid.}
\step{$TV = \sqrt{5^2 + 3^2} = \sqrt{34}$ and $UW = \sqrt{5^2 + 3^2} = \sqrt{34}$: congruent, yet
$TU$ and $WV$ are the only parallel pair and no angle is right.}
$\Rightarrow\quad TV = \ans{5.83} = UW$ in a shape that is \emph{not} a rectangle.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $T(0,0)$, $U(8,0)$, $V(6,4)$, $W(2,4)$. Find $TV$ (then check $UW$ matches).
\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7.21}$}
\end{tryitbox}
