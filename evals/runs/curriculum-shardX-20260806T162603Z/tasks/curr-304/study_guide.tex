\sstitleblock{Finding the Fault in a Proof}

\skillheading{Skill 1 --- Spotting circular reasoning}

\begin{formulabox}
\textbf{Circular:} the reason for a step is the step itself, or something that depends on it. ``$P$ because $Q$, and $Q$ because $P$'' is consistent with both being false, so it establishes nothing.\par
\textbf{The test:} follow each reason backwards. Every chain must end at a \emph{given}, a definition, a postulate, or a line already proved. If a chain loops, the proof is circular.\par
\textbf{The repair:} keep one direction and start it outside the loop --- from the given, or from a theorem that does not use the conclusion.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A proof says $\angle 1 \cong \angle 2$ ``because they are congruent''. Two lines cross, so replace the reason and use it: $m\angle 1 = 2x + 10$ and $m\angle 2 = 4x - 30$. Find $x$.
\step{The circular reason is replaced by the vertical angles theorem, which follows from the given intersecting lines --- and that theorem is what entitles me to set the two measures equal.}
\step{$2x + 10 = 4x - 30$, so $40 = 2x$.}

$\Rightarrow\quad \ans{x = 20}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Vertical angles measure $(3x + 5)^\circ$ and $(5x - 25)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 15}$}
\end{tryitbox}

\skillheading{Skill 2 --- Finding and fixing a broken step}

\begin{formulabox}
Read a planted proof \textbf{line by line}, checking two things each time: does the statement follow from the line above, and does the reason name the property that was actually used?\par
\textbf{The common break:} an operation done on one side only, or the wrong sign --- adding when the term had to be subtracted. The reason column still reads ``property of equality'', which is why the work looks correct.\par
\textbf{Report it precisely:} name the line, name the rule broken, then redo the arithmetic.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A student writes $7x - 4 = 3x + 16$, then $4x = 12$, then $x = 3$. Find and fix the fault.
\step{Comparing the two lines, the $-4$ was subtracted from both sides when it needed to be added to clear it, so the right-hand constant is wrong; the fix is to redo that one step properly.}
\step{$7x - 4 = 3x + 16$ gives $4x - 4 = 16$, so $4x = 20$.}

$\Rightarrow\quad \ans{x = 5}$, not $3$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A student writes $6x + 2 = 2x + 26$, then $4x = 28$, then $x = 7$. Redo it correctly.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{Skill 3 --- Catching an unsupported statement}

\begin{formulabox}
\textbf{Unsupported:} the statement may even be true, but its reason is not a given, a definition, a postulate, or an earlier line.\par
\textbf{Two frequent forms:} citing the picture (``the halves look equal''), and citing a real theorem whose hypothesis is not met --- calling an \emph{adjacent} pair vertical, for instance.\par
\textbf{The repair:} either add the missing given, or replace the reason with the theorem that does apply. Adjacent angles on a line are a linear pair, so they are \emph{supplementary}, not congruent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A linear pair is wrongly called vertical. Its angles measure $(4x - 10)^\circ$ and $(2x + 40)^\circ$. Use the right relationship to find $x$.
\step{The two angles are adjacent with their outer sides on one line, so the linear pair postulate applies and the measures are supplementary rather than equal.}
\step{$(4x - 10) + (2x + 40) = 180$, so $6x + 30 = 180$ and $6x = 150$.}

$\Rightarrow\quad \ans{x = 25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A linear pair measures $(5x + 6)^\circ$ and $(3x - 2)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 22}$}
\end{tryitbox}
