\sstitleblock{Building Bigger Shapes from Smaller Ones}

\skillheading{Skill 1 --- Counting the small squares in a built shape}

\begin{formulabox}
\textbf{Rule:} Small shapes build a bigger one when they leave \textbf{no gaps} and \textbf{no overlaps}.\par
To find how much a built shape covers, count its small squares --- touch each one once, and count a whole row at a time when the rows are straight.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many small squares cover this shape?
\begin{center}
\begin{tikzpicture}[scale=0.42]
  \foreach \x in {0,1,2} { \draw[gray!65, thin] (\x,0) rectangle (\x+1,1); }
  \draw[gray!65, thin] (0,1) rectangle (1,2);
  \draw[line width=1.2pt] (0,0) -- (3,0) -- (3,1) -- (1,1) -- (1,2) -- (0,2) -- cycle;
\end{tikzpicture}
\end{center}
\step{The shape is not a plain rectangle, so I cannot count one row and double it; instead I break it into straight rows and count each row on its own.}
\step{The bottom row holds $3$ small squares and there is $1$ more sitting above the left-hand end, and nothing is counted twice.}

$\ans{4}$ small squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many small squares cover a rectangle with $3$ in each row and $2$ rows?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6}$ small squares}
\end{tryitbox}

\skillheading{Skill 2 --- Naming the shape the pieces make}

\begin{formulabox}
\textbf{Rule:} The built shape gets its own name, and it is usually not the name of the pieces.\par
Look at the \emph{outside} edge only: four straight sides with square corners is a rectangle, and a rectangle with all four sides the same length is a square.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Two matching triangles are joined along their long edges. What shape do they make, and how much does it cover?
\begin{center}
\begin{tikzpicture}[scale=0.42]
  \foreach \x in {0,1} { \foreach \y in {0,1} { \draw[gray!65, thin] (\x,\y) rectangle (\x+1,\y+1); } }
  \draw[line width=1.2pt] (0,0) -- (2,0) -- (2,2) -- (0,2) -- cycle;
  \draw[line width=1.2pt] (0,0) -- (2,2);
\end{tikzpicture}
\end{center}
\step{To name a built shape I have to ignore the join line and trace only the outside edge, because the pieces inside are not what is being named.}
\step{The outside edge has four straight sides, square corners, and all four sides the same length, so it is a square; under it lie $2$ small squares in each of $2$ rows.}

a square, covering $\ans{4}$ small squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Two small squares are placed side by side. Name the new shape, and say how many small squares it covers.\\[2pt]
\rotatebox{180}{\footnotesize check: a rectangle, covering $\ans{2}$ small squares}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing two built shapes}

\begin{formulabox}
\textbf{Rule:} A longer outline does not mean more squares. Count both shapes, then compare the two counts with $<$, $>$, or $=$.\par
Two shapes that look nothing alike can cover exactly the same amount, because moving squares around adds none and takes none away.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Shape A covers $6$ small squares and shape B covers $4$. Which is greater?
\step{Comparing pictures by eye is what misleads here, so I set the two counts side by side and let the numbers decide.}
\step{Counting up, $4$ is said before $6$, so shape B covers less and shape A covers more; the wide end of the symbol faces the bigger count.}

$\ans{6 > 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Shape A covers $5$ small squares and so does shape B, although they look different. Which symbol goes between them?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5 = 5}$}
\end{tryitbox}

\skillheading{Skill 4 --- Finishing a shape that is not done yet}

\begin{formulabox}
\textbf{Rule:} To find how many more pieces a build still needs, do not add them one at a time.\par
Count what the finished outline would hold, count what is already there, and take one from the other: \quad still needed $=$ whole $-$ already placed.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A rectangle will hold $6$ small squares. So far $2$ are in place. How many more are needed?
\step{Placing squares one by one is slow and easy to miscount, so I work with the two totals instead and take the smaller from the bigger.}
\step{The finished rectangle holds $6$ and $2$ are already there, so what is missing is $6 - 2$.}

$\ans{4}$ more small squares
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A square will hold $9$ small squares and $6$ are in place. How many more are needed?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{3}$ more small squares}
\end{tryitbox}
