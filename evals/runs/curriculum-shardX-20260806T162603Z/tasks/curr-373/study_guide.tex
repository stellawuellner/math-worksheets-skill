\sstitleblock{Radical Equations and Extraneous Roots}

\skillheading{Skill 1 --- Isolate the radical, then undo it}

\begin{formulabox}
\textbf{Method:} get the radical alone on one side, then square both sides.
\[ \sqrt{A} = B \ \Longrightarrow \ A = B^2 \quad (\text{only usable when } B \ge 0) \]
Squaring before isolating leaves a cross term and makes the equation worse, so
the isolating step always comes first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $3\sqrt{x+2} - 5 = 7$.
\step{The radical is buried under a multiplication and a subtraction, so undo
those first --- squaring is only useful once the root stands alone.}
\step{$3\sqrt{x+2} = 12$, so $\sqrt{x+2} = 4$, so $x + 2 = 16$. Check:
$3\sqrt{16} - 5 = 12 - 5 = 7$.}

$\Rightarrow\quad \ans{x = 14}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $4\sqrt{x-3} + 1 = 13$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 12}$}
\end{tryitbox}

\skillheading{Skill 2 --- Test every candidate in the original}

\begin{formulabox}
\textbf{Why checking is not optional:} squaring loses the sign, because $3^2$ and
$(-3)^2$ are the same number. So the squared equation can have solutions the
original never had --- \textbf{extraneous roots}. Substitute every candidate back
into the \emph{original} equation and keep only the ones that work. A candidate
that makes the isolated radical equal a negative quantity always fails.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\sqrt{x+6} = x$.
\step{The right side is a variable, so it could be negative for some candidates
--- that is exactly the situation where squaring invents an extra root and the
check has to catch it.}
\step{$x + 6 = x^2$ gives $x^2 - x - 6 = 0$, so $(x-3)(x+2) = 0$ and the
candidates are $3$ and $-2$. Testing: $\sqrt{9} = 3$ works, but $\sqrt{4} = 2$
is not $-2$.}

$\Rightarrow\quad \ans{x = 3}$ \ (the candidate $-2$ is extraneous)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\sqrt{3x+4} = x$, keeping only what survives the
check.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$ (the candidate $-1$ is extraneous)}
\end{tryitbox}

\skillheading{Skill 3 --- Solving a radical model for its input}

\begin{formulabox}
\textbf{Method:} a model such as $v = k\sqrt{h}$ is solved the same way ---
divide by the constant, square, and state the answer in the model's unit. Only
the non-negative solution can be a physical measurement, so the check also
throws out negative candidates.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A shallow-water wave travels at $v = 3\sqrt{h}$ metres
per second, where $h$ is the depth in metres. Find the depth giving
$v = 12$ metres per second.
\step{The unknown sits under the root, so isolate the root by dividing by the
constant and then square to release it.}
\step{$3\sqrt{h} = 12$ gives $\sqrt{h} = 4$, and squaring gives $h = 16$. Check:
$3\sqrt{16} = 12$.}

$\Rightarrow\quad \ans{h = 16}$ metres
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $v = 5\sqrt{h}$ metres per second, find the depth
giving $v = 35$ metres per second.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{h = 49}$ metres}
\end{tryitbox}
