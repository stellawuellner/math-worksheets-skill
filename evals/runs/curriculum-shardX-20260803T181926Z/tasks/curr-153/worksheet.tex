\wstitleblock{Tape Diagrams and Double Number Lines for Ratios}{Grade 6--7 Mathematics}{}

\begin{center}
\textit{Reading and completing tape diagrams, filling in double number lines,
finding a unit rate from the line, and comparing two ratios with the diagrams}
\end{center}

\noindent\small
\textbf{Two pictures, one idea.} A \textbf{tape diagram} shows a ratio as equal
parts: every box is worth the same amount, so finding the value of \emph{one
box} unlocks the whole problem. A \textbf{double number line} pairs two
quantities that grow together: the two lines are marked so that matching points
sit above each other.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item On a tape diagram: value of one box $=$ known amount $\div$ number of
        boxes it covers.
  \item On a double number line: multiply or divide \emph{both} lines by the
        same number to move to a new pair.
  \item The \textbf{unit rate} is the number that sits above $1$ on the other
        line. It makes every later question one multiplication.
  \item Label your diagram before you compute, and write the units in your
        answer.
\end{itemize}
\normalsize

\problem{A punch recipe mixes juice and soda in the ratio $2 : 3$. The
tape diagram is drawn with $10$ cups of juice. How many cups of soda are
needed?
\par\smallskip
\begin{center}
\begin{tikzpicture}
  \node[left, font=\small] at (-0.1,0.3) {juice};
  \foreach \i in {0,1} {\draw[thick] (\i*1.2,0) rectangle (\i*1.2+1.2,0.6);}
  \node[right, font=\small] at (2.5,0.3) {$10$ cups in all};
  \node[left, font=\small] at (-0.1,-0.9) {soda};
  \foreach \i in {0,1,2} {\draw[thick] (\i*1.2,-1.2) rectangle (\i*1.2+1.2,-0.6);}
  \node[right, font=\small] at (3.7,-0.9) {$?$};
\end{tikzpicture}
\end{center}
\par\smallskip
First write the value of one box, then the amount of soda.
\par\vspace*{4.5cm}\answerline{cups}}

\problem{\textbf{Synthesis challenge.} A camp cook knows that $15$
kilograms of rice feeds $24$ people. The double number line below shows a
second point: $x$ kilograms feeds $8$ people.
\par\smallskip
\begin{center}
\begin{tikzpicture}[x=1.3cm]
  \draw[thick,->] (-0.3,0) -- (3.4,0);
  \draw[thick,->] (-0.3,-1.3) -- (3.4,-1.3);
  \foreach \x/\lab in {0/0, 1/x, 3/15} {
    \draw (\x,-0.12) -- (\x,0.12);
    \node[above=1pt, font=\small] at (\x,0.12) {$\lab$};
    \draw (\x,-1.42) -- (\x,-1.18);}
  \foreach \x/\lab in {0/0, 1/8, 3/24} {
    \node[below=1pt, font=\small] at (\x,-1.42) {$\lab$};}
  \node[right, font=\small] at (3.4,0) {kilograms};
  \node[right, font=\small] at (3.4,-1.3) {people};
\end{tikzpicture}
\end{center}
\par\smallskip
\textbf{(a)} Write a proportion from the two pairs and solve it for $x$.
\textbf{(b)} Then give the unit rate in kilograms per person.
