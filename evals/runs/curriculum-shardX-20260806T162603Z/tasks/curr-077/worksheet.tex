\wstitleblock{Fact Strategies: Turn It Around, Break It Apart}{Grade 3}{}
\begin{center}
\textit{turn-around facts, break-apart strategies, and choosing between them}
\end{center}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item \textbf{Turn-around facts:} the order of the factors never changes the product.
  \item \textbf{Break-apart facts:} split one factor into easy pieces, multiply each piece, and add the parts.
  \item Problems 3 and 4 show the pattern worked out --- after that, the strategy is yours to run.
\end{itemize}

\problem[2.8cm]{Kim knows $3 \times 7 = 21$. Use the turn-around fact: \quad $7 \times 3 = {}$~\ansblank}

\problem[2.8cm]{You know $6 \times 4 = 24$. What is $4 \times 6$? \quad \ansblank}

\problem[2.8cm]{Finish the worked pattern: \quad $7 \times 6 = 7 \times 5 + 7 \times 1 = 35 + 7 = {}$~\ansblank}

\problem[3.2cm]{Break the 7 apart, then fill in all three blanks:\par\vspace{4pt}
\begin{center}
$8 \times 7 \;=\; 8 \times 5 + 8 \times 2 \;=\;$ \ansblank ${}+{}$ \ansblank ${}={}$ \ansblank
\end{center}}

\problem[0pt]{Write the turn-around fact for $9 \times 2$: \ansline\par\vspace{8pt}
Both facts have the same product: \quad \ansblank\par\vspace*{2.4cm}}

\problem[2.8cm]{Finish the break-apart with a take-away: \quad $6 \times 9 = 6 \times 10 - 6 \times 1 = {}$~\ansblank}

\problem[0pt]{Trina wants to skip count to find the 8-and-5 fact. Circle the order that is easier to skip count:\par\vspace{4pt}
\begin{center}\textbf{$8 \times 5$ (count by 5s, eight times)} \qquad \textbf{$5 \times 8$ (count by 8s, five times)}\end{center}
Then find the product: \quad \ansblank\par\vspace*{2.6cm}}

\problem[2.8cm]{Break the 12 apart to solve: \quad $4 \times 12 = 4 \times 10 + 4 \times 2 = {}$~\ansblank}

\problem[0pt]{Solve $6 \times 7$ two different ways.
\begin{itemize}[leftmargin=1.3cm, itemsep=4pt]
  \item[(a)] Break the 7 into $5 + 2$: \quad $6 \times 7 = {}$~\ansblank
  \item[(b)] Break the 6 into $5 + 1$: \quad $7 \times 6 = {}$~\ansblank
  \item[(c)] Say why both ways must give the same product.\par\vspace{6pt}\noindent\underline{\hspace{12cm}}\par\vspace{14pt}\noindent\underline{\hspace{12cm}}\par\vspace*{1.9cm}
\end{itemize}}
