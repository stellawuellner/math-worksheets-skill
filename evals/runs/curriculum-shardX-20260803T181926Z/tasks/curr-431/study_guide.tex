\sstitleblock{Converting Between Rectangular and Polar Coordinates}

\begin{formulabox}
\textbf{The four conversion facts.} \quad
$x = r\cos\theta$ \quad $y = r\sin\theta$ \quad
$r^2 = x^2 + y^2$ \quad $\tan\theta = \dfrac{y}{x}$
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
\item Going \textbf{polar $\rightarrow$ rectangular} is pure substitution ---
there is nothing to decide.
\item Going \textbf{rectangular $\rightarrow$ polar}, $r$ is a distance and is
never negative, but $\theta$ always needs a decision: two angles on
$[0^\circ, 360^\circ)$ share the same tangent, half a turn apart.
\item Sketch the point before writing $\theta$. The quadrant is what picks
between the two candidates, and no formula supplies it.
\end{itemize}
\end{formulabox}

\skillheading{1. Polar to rectangular}

\begin{formulabox}
Substitute into $x = r\cos\theta$ and $y = r\sin\theta$ and simplify. Keep
exact values ($\frac{\sqrt{3}}{2}$, not $0.866$) unless a decimal is asked for.
Confirm with $x^2 + y^2 = r^2$, and check the signs against the quadrant of
$\theta$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Convert the polar point $\left(10, \frac{\pi}{6}\right)$ to
rectangular form.
\step{Nothing needs deciding here --- the angle is given, so both coordinates
come straight out of the two substitution formulas.}
\step{$x = 10\cos\frac{\pi}{6} = 10 \cdot \frac{\sqrt{3}}{2}$ and
$y = 10\sin\frac{\pi}{6} = 10 \cdot \frac{1}{2}$.}
\begin{center}
\begin{tikzpicture}[scale=0.28]
  \foreach \rr in {2,4,6,8,10} { \draw[gray!30, thin] (0,0) circle (\rr); }
  \foreach \aa in {0,30,60,90,120,150,180,210,240,270,300,330} { \draw[gray!30, thin] (0,0) -- (\aa:10); }
  \draw[->, gray!70] (-10.6,0) -- (11.4,0) node[right, font=\tiny] {$x$};
  \draw[->, gray!70] (0,-10.6) -- (0,11.4) node[above, font=\tiny] {$y$};
  \draw[very thick, blue!65!black] (0,0) -- (30:10);
  \fill[red] (30:10) circle (0.35);
\end{tikzpicture}
\end{center}
So the rectangular form is $\ans{(5\sqrt{3},\ 5)}$, in quadrant I as the
picture shows.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Convert $\left(8, \frac{2\pi}{3}\right)$ to rectangular
form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(-4,\ 4\sqrt{3})}$}
\end{tryitbox}

\skillheading{2. Rectangular to polar}

\begin{formulabox}
First $r = \sqrt{x^2+y^2}$, always positive. Then take the reference angle
$\arctan\left|\frac{y}{x}\right|$ and place it in the point's own quadrant: add
nothing in QI, subtract from $180^\circ$ in QII, add $180^\circ$ in QIII,
subtract from $360^\circ$ in QIV.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} Convert the rectangular point $(-8, 6)$ to polar form, with
the angle in degrees to two decimal places.
\step{The signs put the point in quadrant II, so the arctangent key cannot give
the final angle --- get the reference angle first and place it afterwards.}
\step{$r = \sqrt{64 + 36} = 10$, and the key returns
$\arctan\frac{6}{-8} = -36.87^\circ$; adding a half turn moves it into
quadrant II.}
So the polar form is $\ans{(10,\ 143.13^\circ)}$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} Convert $(-8, -6)$ to polar form, angle in degrees to two
decimal places.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(10,\ 216.87^\circ)}$}
\end{tryitbox}

\skillheading{3. Choosing the angle that fits the quadrant}

\begin{formulabox}
$\tan$ repeats every $180^\circ$, so $\tan\theta = k$ always has \emph{two}
solutions on one turn, and they sit in diagonally opposite quadrants. Solve for
both, then let the signs of $x$ and $y$ --- or the sketch --- keep one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} A point has $\tan\theta = -\sqrt{3}$ and lies in quadrant IV.
Find $\theta$ on $[0^\circ, 360^\circ)$.
\step{A negative tangent means $x$ and $y$ have opposite signs, which happens in
two quadrants --- so solve for both candidates before choosing.}
\step{The reference angle for $\sqrt{3}$ is $60^\circ$, and the two solutions
sit half a turn apart.}
The candidates are $\ans{\theta = 120^\circ \text{ or } 300^\circ}$, and only
$300^\circ$ lies in quadrant IV, so that is the polar angle.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} $\tan\theta = 1$ and the point is in quadrant III. Give both
candidates on $[0^\circ, 360^\circ)$, then pick the right one.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\theta = 45^\circ \text{ or }
225^\circ}$; quadrant III gives $225^\circ$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the arctangent key only ever answers between
$-90^\circ$ and $90^\circ$, so it can never return a quadrant II or III angle
--- those always need a half turn added. And on the $y$-axis, where $x = 0$,
$\frac{y}{x}$ does not exist at all: the angle is $90^\circ$ or $270^\circ$,
read straight off the picture.
\end{watchoutbox}
