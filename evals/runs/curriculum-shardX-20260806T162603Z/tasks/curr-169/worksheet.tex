\wstitleblock{Signed Rational Numbers: Subtraction \& Sign Errors}{Grade 6--7 Mathematics}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Show your work in the space under each problem.
  \item Watch the signs! Rewrite each subtraction as adding the opposite before you compute.
  \item Two problems show another student's work that contains a mistake --- find it and fix it.
\end{itemize}

\problem[3cm]{Evaluate: \quad $-5 - 8$}

\problem[3cm]{Evaluate: \quad $-7 - (-12)$}

\problem[3cm]{Evaluate: \quad $-(-8) - 15$}

\problem{Jorge computes $-4 - (-6)$ and writes
\[ -4 - (-6) = -10. \]
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] What error did Jorge make? Answer in one sentence.\par\vspace*{1.6cm}
  \item[(b)] Find the correct value of $-4 - (-6)$.\par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem[2.5cm]{Write $<$ or $>$ in the blank to make a true statement:
\[ -3 - 5 \;\; \raisebox{-2pt}{\ansblank} \;\; -3 - (-5) \]}

\problem[3.5cm]{Evaluate: \quad $3 - (2 - 9)$}

\problem{Maya computes $-\tfrac{1}{2} - \tfrac{3}{4}$ and writes
\[ -\tfrac{1}{2} - \tfrac{3}{4} = \tfrac{1}{4}. \]
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt]
  \item[(a)] Explain Maya's error in one sentence.\par\vspace*{1.6cm}
  \item[(b)] Find the correct value of $-\tfrac{1}{2} - \tfrac{3}{4}$.\par\vspace*{2.2cm}\ansline
\end{itemize}}

\problem[4.5cm]{\textbf{Challenge.} Evaluate: \quad $-\tfrac{3}{4} - \left(-\tfrac{5}{6}\right) + \left(-\tfrac{1}{3}\right)$}
