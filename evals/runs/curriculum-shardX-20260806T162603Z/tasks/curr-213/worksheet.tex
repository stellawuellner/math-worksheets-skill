\wstitleblock{Solving and Classifying Linear Equations}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{One solution, no solution, or infinitely many solutions}
\end{center}

\vspace{0.15cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Simplify each side of the equation first, then collect the variable
        terms on one side.
  \item The step where the variable terms combine or cancel is what tells you
        the classification --- show it.
  \item Write each final answer as a value of the variable, or as
        \textit{no solution}, or as \textit{infinitely many solutions}.
\end{itemize}

\vspace{0.15cm}

\problem[4.4cm]{Bike Barn rents bicycles for a \$9 flat fee plus \$4 per hour.
Wheel World charges a \$14 flat fee plus \$3 per hour. The equation
$9 + 4h = 14 + 3h$ compares the two rental costs in dollars, where $h$ is the
number of hours. Solve it, then state how many solutions it has.}

\problem[4.4cm]{One phone plan costs \$12 per month plus \$6 for each gigabyte
of data used. A second plan costs \$20 per month plus \$4 for each gigabyte.
Solve $12 + 6g = 20 + 4g$, where $g$ is the number of gigabytes used in a
month, then state how many solutions it has.}

\problem[4.4cm]{A gardener has $r$ rows of 5 plants and then adds 3 more rows
of 5 plants. Counting by groups gives $5(r + 3)$ plants; counting row by row
gives $5r + 15$ plants. Solve $5(r + 3) = 5r + 15$, then state how many
solutions it has.}

\problem[4.4cm]{Savings Plan A starts with \$25 and adds \$5 every week.
Savings Plan B starts with \$18 and adds \$5 every week. The equation
$25 + 5w = 18 + 5w$ asks when the two balances in dollars are equal, where $w$
is the number of weeks. Solve it, then state how many solutions it has.}

\problem[5cm]{A club spends $8(t - 2)$ dollars on snacks and $6t + 4$ dollars
on supplies, where $t$ is the number of members. Solve $8(t - 2) = 6t + 4$ to
find the membership at which the two amounts are equal, then state how many
solutions the equation has.}

\problem[4.4cm]{A rectangle is $x$ cm long and 4 cm wide. Mia writes its
perimeter as $2(x + 4)$ cm; Ben writes it as $2x + 8$ cm. Solve
$2(x + 4) = 2x + 8$, then state how many solutions it has.}

\problem[4.4cm]{On a track, runner A starts 30 m from the line and runs
6 m/s, so her distance is $30 + 6t$ metres after $t$ seconds. Runner B starts
12 m from the line and also runs 6 m/s: $12 + 6t$ metres. Solve
$30 + 6t = 12 + 6t$, then state how many solutions it has.}

\problem[5cm]{A pool is draining and holds $120 - 8m$ litres after $m$ minutes.
A second pool is filling and holds $40 + 12m$ litres after $m$ minutes. Solve
$120 - 8m = 40 + 12m$ to find when the two pools hold the same amount, then
state how many solutions the equation has.}

\problem{A print shop charges $4(2x + 5)$ dollars for $x$ posters. A rival
shop charges $8x + k$ dollars, where $k$ is a whole number of dollars.
\begin{itemize}
  \item[(a)] Find the value of $k$ that makes $4(2x + 5) = 8x + k$ true for
        every number of posters.
        \par\vspace*{2.7cm}\ansline
  \item[(b)] Explain how the simplified equation shows that every other value
        of $k$ gives an equation with no solution.
        \par\vspace*{2.7cm}\ansline
\end{itemize}}

\problem{Tank A holds $60 + 9t$ litres of water after $t$ minutes. Tank B
holds $3(3t + 12)$ litres after $t$ minutes.
\begin{itemize}
  \item[(a)] Simplify both sides of $60 + 9t = 3(3t + 12)$ and state how many
        solutions the equation has.
        \par\vspace*{2.7cm}\ansline
  \item[(b)] Explain what the simplified equation tells you about the two
        water amounts as the minutes go by.
        \par\vspace*{2.7cm}\ansline
\end{itemize}}
