\sstitleblock{Halves and Fourths --- Making Equal Shares}

\skillheading{Skill 1 --- Checking whether two parts are really halves}

\begin{formulabox}
\textbf{The rule:}\\[4pt]
\textbf{Two parts are halves only when the two parts hold the SAME number of
squares.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Cutting a shape into two pieces does \emph{not} make halves by itself.
  \item Count the squares in each part, then compare the two counts.
  \item Same count $\rightarrow$ halves. Different count $\rightarrow$ not halves.
\end{itemize}
To know what a fair half should be, share the whole count between the two parts.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A strip of $8$ squares is cut into $2$ parts. How many
squares belongs in each half?

\step{A half must hold the same count as the other half, so share the whole
strip equally between the two parts.}

\step{Share $8$ squares between $2$ parts: $4$ for one part and $4$ for the
other.}

Each half holds $\ans{4}$ squares. A part with any other count is not a half.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A strip of $10$ squares is cut into $2$ equal parts. How
many squares are in each half?

\rotatebox{180}{\footnotesize check: $\ans{5}$}
\end{tryitbox}

\skillheading{Skill 2 --- Fixing a cut that is not halves}

\begin{formulabox}
\textbf{How to fix an unequal cut:}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Put the two counts together to get the whole.
  \item Share the whole between the two parts. That is the \textbf{fair share}.
  \item Count up from the small part to the fair share. That is how many squares
        move across.
\end{itemize}
\textbf{Watch the size of the move:} moving the whole gap makes the small part
the big one. Only \emph{half} the gap moves.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad One part holds $3$ squares and the other holds $9$. How
many squares move so both parts are halves?

\step{Halves must match, so find the fair share first, then see how far the
small part is from it.}

\step{The two parts together hold $12$ squares, and shared between two parts the
fair share is $6$ each. Count up from $3$ to $6$.}

The move is $\ans{3}$ squares. Check: $3 + 3 = 6$ and $9 - 3 = 6$ $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad One part holds $2$ squares and the other holds $10$. How
many squares move so both parts are halves?

\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Fourths: all four parts must match}

\begin{formulabox}
\textbf{The rule:}\quad \textbf{Fourths are four parts that all hold the SAME
number of squares.}
\begin{itemize}[leftmargin=0.8cm, itemsep=1pt, topsep=2pt]
  \item Four pieces of any size are just four pieces, not fourths.
  \item Fold once and you have halves. Fold the halves again and you have fourths.
  \item Share the whole count into four equal groups to find one fourth.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad A rectangle of $12$ squares is cut into fourths. How many
squares are in one fourth?

\step{Fourths are four matching parts, so share the whole count into $4$ equal
groups.}

\step{Share $12$ squares into $4$ groups: $3$, $3$, $3$, $3$.}

One fourth holds $\ans{3}$ squares.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad A rectangle of $16$ squares is cut into fourths. How many
squares are in one fourth?

\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} The name tells you how many \emph{equal} parts, not just
how many cuts. Two parts that do not match are not halves, and four parts that
do not match are not fourths --- so always count both (or all four) parts before
you use the name.
\end{watchoutbox}
