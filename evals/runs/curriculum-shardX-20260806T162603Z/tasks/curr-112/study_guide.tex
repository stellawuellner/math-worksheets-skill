\sstitleblock{Dividing by a One-Digit Number}

\skillheading{Skill 1 --- Long Division, Digit by Digit}

\begin{formulabox}
\textbf{Four moves, repeated:} divide, multiply, subtract, bring down.\par
Work left to right through the dividend. Each answer digit goes \emph{directly above} the digit you
just used, so the places stay lined up.\par
If the divisor does not go into the digit you have, write $0$ above it and bring the next digit down.
Stop when there is nothing left to bring down; whatever is left below is the remainder, and it is
always smaller than the divisor.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $738 \div 5$
\step{The divisor is bigger than the first digit is comfortable with, so I work through the dividend one digit at a time and keep each answer digit above the digit it came from.}
\step{$5$ into $7$ goes once, leaving $2$; bring down $3$ to make $23$, and $5$ goes $4$ times leaving $3$; bring down $8$ to make $38$, and $5$ goes $7$ times leaving $3$.}

$738 \div 5 = \ans{147}$ remainder $3$ \quad (check: $147 \times 5 + 3 = 738$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $926 \div 4$. What is the quotient?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{231}$ remainder $2$}
\end{tryitbox}

\skillheading{Skill 2 --- Partial Quotients}

\begin{formulabox}
\textbf{Take out easy chunks of groups, then add the chunks.}\par
Pick a friendly multiple of the divisor --- $10$, $50$, $100$, $600$ groups --- subtract it, and repeat
on what is left.\par
Any chunks work; smaller chunks just take more rounds. The chunk sizes you took out are the partial
quotients, and their sum is the quotient.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $336 \div 3$ with partial quotients
\step{I would rather subtract friendly chunks than juggle carries, so I take out a round number of groups first and finish with what is left.}
\step{$3 \times 100 = 300$, and $336 - 300 = 36$; then $36 \div 3 = 12$, so the chunks are $100$ and $12$.}

$336 \div 3 = 100 + 12 = \ans{112}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $856 \div 8$ with partial quotients (start with $100$ groups of $8$).\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{107}$}
\end{tryitbox}

\skillheading{Skill 3 --- Estimate First, Check After}

\begin{formulabox}
\textbf{Before:} round the dividend to a friendly number and divide that. The estimate tells you how
many digits the quotient should have and roughly how big it is.\par
\textbf{After:} multiply back. \ (quotient) $\times$ (divisor) $+$ (remainder) must rebuild the
dividend exactly.\par
Rounding the dividend \emph{up} makes the estimate a little large; rounding \emph{down} makes it a
little small.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Round $2{,}874$ to the nearest thousand, ready to estimate $2{,}874 \div 6$.
\step{I round only the dividend, because a friendly dividend is what makes the estimate easy while the divisor stays exact.}
\step{Rounding to the thousand means looking at the hundreds digit, $8$; that is $5$ or more, so the number rounds up.}

$2{,}874$ to the nearest thousand is $\ans{3000}$, so the estimate is $3000 \div 6 = 500$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Round $5{,}390$ to the nearest thousand, ready to estimate a division.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{5000}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a $0$ in the quotient is a digit, not a gap. In $618 \div 6 = 103$ the zero
holds the tens place; leaving it out gives $13$, which the multiply-back check catches at once.
\end{watchoutbox}
