\sstitleblock{Integrating Rational Functions with Partial Fractions}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Distinct Linear Factors}

\begin{formulabox}
\textbf{Setup.} One constant per factor:
\[ \frac{1}{(x - a)(x - b)} = \frac{A}{x - a} + \frac{B}{x - b}. \]
Clear denominators, then substitute $x = a$ and $x = b$ in turn --- each choice
kills one unknown. Each piece integrates to a logarithm of an absolute value.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int \frac{1}{(x - 3)(x + 1)}\,dx$.
\step{The denominator is already two distinct linear factors, so one constant over each is the whole decomposition.}
\step{$1 = A(x+1) + B(x-3)$ gives $A = \tfrac{1}{4}$ at $x = 3$ and $B = -\tfrac{1}{4}$ at $x = -1$.}

$\Rightarrow\quad \ans{\frac{\ln|x-3|}{4} - \frac{\ln|x+1|}{4} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int \frac{1}{(x - 5)(x + 1)}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\frac{\ln|x-5|}{6} - \frac{\ln|x+1|}{6} + C}$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Repeated and Quadratic Factors}

\begin{formulabox}
\textbf{Repeated factor:} give it every power.
$\dfrac{A}{x - a} + \dfrac{B}{(x - a)^2}$.\par
\textbf{Irreducible quadratic:} its numerator is \emph{linear}.
$\dfrac{Bx + C}{x^2 + k}$, which integrates to a logarithm plus an arctangent.
Counting unknowns is the check: they must equal the degree of the denominator.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int \frac{x + 3}{(x + 1)^2}\,dx$.
\step{The factor is repeated, so the decomposition needs both the first and the second power rather than a single term.}
\step{$x + 3 = A(x+1) + B$ gives $A = 1$ and $B = 2$, so the integral splits into $\int\tfrac{1}{x+1}dx + \int\tfrac{2}{(x+1)^2}dx$.}

$\Rightarrow\quad \ans{\ln|x+1| - \frac{2}{x+1} + C}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int \frac{x + 5}{(x + 1)^2}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\ln|x+1| - \frac{4}{x+1} + C}$}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Totals from a Rational Rate}

\begin{formulabox}
\textbf{Reading the application.} If a rate is a rational function of time, the
total accumulated is a definite integral --- decompose, integrate to logarithms,
then subtract at the limits.\par
Two logarithms combine: $\ln|f| - \ln|g| = \ln\left|\dfrac{f}{g}\right|$, which
usually makes the arithmetic at the limits much shorter.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Evaluate $\displaystyle\int_0^1 \frac{4}{(x + 1)(x + 3)}\,dx$.
\step{This is a total over an interval, so decompose first and only then substitute the limits into the combined logarithm.}
\step{$\tfrac{2}{x+1} - \tfrac{2}{x+3}$ integrates to $2\ln\tfrac{x+1}{x+3}$, and at the limits that is $2\ln\tfrac{2}{4} - 2\ln\tfrac{1}{3}$.}

$\Rightarrow\quad \ans{2\ln\frac{3}{2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Evaluate $\displaystyle\int_0^3 \frac{4}{(x + 1)(x + 3)}\,dx$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2\ln 2}$}
\end{tryitbox}
