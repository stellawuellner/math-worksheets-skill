\sstitleblock{Solving and Graphing One-Step Inequalities}

% =================== SKILL 1 ===================
\skillheading{Undoing an Addition or a Subtraction}

\begin{formulabox}
\textbf{Rule:} add or subtract the same number on both sides. This \emph{never}
changes the direction of the sign.\par
\textbf{Graphing:} put the circle on the boundary number --- \textbf{open} for
$<$ and $>$, \textbf{filled} for $\le$ and $\ge$ --- then shade the side the
finished sign points to when the letter is written first.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and graph $b + 7 \le 10$.
\step{The 7 is added to $b$, so undo it by subtracting 7 from both sides; the
sign stays as it is.}
\step{$b \le 10 - 7$, that is $\ans{b \leq 3}$. Graph: filled circle at 3
(because of the bar under the sign), shaded to the left.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $c - 5 > 2$ and say which way the graph
shades.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c > 7}$, open circle, shaded right}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Dividing or Multiplying by a Positive Number}

\begin{formulabox}
\textbf{Rule:} divide (or multiply) both sides by the same \emph{positive}
number. The direction of the sign stays exactly as it was --- even when the
answer turns out negative. A negative answer is not a reason to turn the sign
around.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and graph $6h < 42$.
\step{The letter is multiplied by 6, so divide both sides by 6; six is positive,
so the sign keeps its direction.}
\step{$h < 42 \div 6$, that is $\ans{h < 7}$. Graph: open circle at 7, shaded to
the left.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{u}{5} \ge 3$ and say which way the graph
shades.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{u \geq 15}$, filled circle, shaded
right}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Dividing or Multiplying by a Negative Number}

\begin{formulabox}
\textbf{Rule:} dividing or multiplying both sides by a \emph{negative} number
reverses the order of the two sides, so the sign must \textbf{turn around}:
$<$ becomes $>$, $\le$ becomes $\ge$, and back again.\par
\textbf{Why:} $2 < 5$ is true, but multiplying both sides by $-1$ gives $-2$ and
$-5$, and $-2$ is the \emph{larger} of those two.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve and graph $-4z > 12$.
\step{The letter is multiplied by $-4$, so divide both sides by $-4$ --- and
because that divisor is negative, turn the sign around.}
\step{$z < 12 \div (-4)$, that is $\ans{z < -3}$. Graph: open circle at $-3$,
shaded to the left. Test $z = -4$: $-4 \times (-4) = 16 > 12$ $\checkmark$}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $-\dfrac{w}{2} \le 5$ and say which way the graph
shades.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{w \geq -10}$, filled circle, shaded
right}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} turn the sign around only when the number you divide or
multiply \emph{by} is negative. In $7w < -21$ the divisor is $+7$, so the sign
stays and the answer is $w < -3$.
\end{watchoutbox}
