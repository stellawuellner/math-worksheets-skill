\sstitleblock{Linear Equations with Distribution and Fractions}

\noindent\small\textbf{The order that always works:} \textbf{(1)} clear fractions by multiplying
\emph{every} term on \emph{both} sides by the LCD, \textbf{(2)} distribute across any brackets,
\textbf{(3)} collect variable terms on one side and numbers on the other, \textbf{(4)} divide to
isolate $x$ --- then substitute back into the original equation to check.\normalsize

\skillheading{1. Distribute first, then undo (distribute-then-solve)}

\begin{formulabox}
\textbf{Distributive property:} $a(b + c) = ab + ac$, and $a(b - c) = ab - ac$.\\[3pt]
The factor outside multiplies \textbf{every} term inside --- including the second one, with its
sign. A negative factor changes both signs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $4(x - 3) = 20$
\step{The variable is trapped inside brackets, so distribute the $4$ over both terms before undoing
anything.}
\step{$4x - 12 = 20$, so $4x = 32$ and $x = 32 \div 4$.}
$\Rightarrow\quad \ans{x = 8}$ \quad Check: $4(8 - 3) = 20$ $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $5(x + 2) = 45$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 7}$}
\end{tryitbox}

\skillheading{2. Clear the fractions with the LCD (clear-fractions)}

\begin{formulabox}
\textbf{Rule:} Find the LCD of every denominator, then multiply \textbf{every term on both sides}
by it --- whole-number terms included.\\[3pt]
$\dfrac{x}{a}$ multiplied by the LCD $L$ becomes $\dfrac{L}{a}\,x$. If a fraction bar covers a
whole numerator, keep that numerator in brackets: $L \cdot \dfrac{x + 3}{2} = \dfrac{L}{2}(x + 3)$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $\dfrac{x}{2} + \dfrac{x}{6} = 4$
\step{Two different denominators make this hard to collect, so multiply everything by the LCD of
$2$ and $6$, which is $6$.}
\step{$3x + x = 24$, so $4x = 24$.}
$\Rightarrow\quad \ans{x = 6}$ \quad Check: $3 + 1 = 4$ $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $\dfrac{x}{3} + \dfrac{x}{4} = 7$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 12}$}
\end{tryitbox}

\skillheading{3. Variables on both sides (variables-both-sides)}

\begin{formulabox}
\textbf{Rule:} Distribute on each side first, then move the \emph{smaller} variable term across so
the coefficient you are left with is positive.\\[3pt]
Whatever you subtract from one side, subtract from the other --- that is what keeps the equation
balanced.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Solve $2(x + 5) = 4x - 6$
\step{Both sides carry an $x$, so distribute first and then gather the variable on the side where
it stays positive.}
\step{$2x + 10 = 4x - 6$; subtract $2x$: $10 = 2x - 6$; add $6$: $16 = 2x$.}
$\Rightarrow\quad \ans{x = 8}$ \quad Check: $2(13) = 26$ and $4(8) - 6 = 26$ $\checkmark$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Solve $3(x - 1) = x + 9$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 6}$}
\end{tryitbox}

\skillheading{4. Rewriting into an equivalent form (rewrite-equivalent)}

\begin{formulabox}
\textbf{Rule:} Multiplying an expression by the LCD produces an \textbf{equivalent} expression
only if every term is multiplied.\\[3pt]
$L\left(\dfrac{x}{a} + \dfrac{b}{c}\right) = \dfrac{L}{a}x + \dfrac{Lb}{c}$. Doing this to one side
of an equation means doing it to the other side too.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Multiply $\dfrac{x}{3} - \dfrac{1}{2}$ by the LCD $6$
\step{The denominators are $3$ and $2$; their LCD is $6$, and multiplying by it clears both at
once --- no solving is involved here, only rewriting.}
\step{$6 \cdot \dfrac{x}{3} = 2x$ \quad and \quad $6 \cdot \dfrac{1}{2} = 3$.}
$\Rightarrow\quad \ans{2x - 3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Multiply $\dfrac{x}{5} + \dfrac{3}{4}$ by the LCD $20$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x + 15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} A minus sign in front of brackets or in front of a fraction bar reaches
\emph{every} term behind it: $-(x - 5) = -x + 5$. And when you clear fractions, the whole-number
terms get multiplied too --- forgetting them is the single most common lost mark.
\end{watchoutbox}
