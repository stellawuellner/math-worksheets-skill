\wstitleblock{One Line, Four Ways to Show It}{Grade 8 / Pre-Algebra}{}

\begin{center}
\textit{Contexts, tables, graphs, and equations that describe the same line}
\end{center}

\vspace{0.1cm}
\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item Whatever form a relationship arrives in, look for the same two
        numbers: the starting value (when the input is 0) and the change per
        unit of input.
  \item Two representations match only when \emph{both} numbers agree ---
        one shared value is not enough.
  \item Write your answers as numbers with the units the story uses.
\end{itemize}

\problem{A table of values is shown below.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 \\ \hline
$y$ & 6 & 10 & 14 & 18 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What is the $y$-intercept of the line this table describes?
        \par\vspace*{1.9cm}\ansline
  \item[(b)] What is its slope?
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem[3.6cm]{A taxi charges \$3 to start the trip plus \$2 for each mile.
Write the cost equation and use it to find the cost of a 12-mile ride, in
dollars.}

\problem{Line $P$ in the Graph Bank passes through the marked points $(0, 2)$
and $(4, 14)$.
\begin{itemize}
  \item[(a)] Find the slope of line $P$.
        \par\vspace*{1.9cm}\ansline
  \item[(b)] Write its equation and use it to find $y$ when $x = 7$.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem{The table below and the equation $y = 5x + 4$ are supposed to
describe the same line.
\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & 0 & 2 & 4 \\ \hline
$y$ & 4 & 14 & 24 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] What $y$-value does the table give at $x = 4$?
        \par\vspace*{1.7cm}\ansline
  \item[(b)] What $y$-value does the equation give at $x = 4$?
        \par\vspace*{1.7cm}\ansline
  \item[(c)] Do the table and the equation describe the same line? Explain
        how you decided.
        \par\vspace*{1.9cm}\ansline
\end{itemize}}

\problem{A student claims that line $R$ in the Graph Bank, which passes
through the marked points $(0, 1)$ and $(6, 13)$, is described by the equation

\begin{itemize}
  \item[(a)] Find the slope of line $R$ from its two marked points.

        \par\vspace*{1.8cm}\ansline

        \par\vspace*{2cm}\ansline

\problem{A gym charges a \$50 joining fee plus \$18 for each month of

        \par\vspace*{1.9cm}\ansline

        \par\vspace*{1.9cm}\ansline

\problem{A table of values is shown below.

\problem{A pool already holds 30 litres and gains 12 litres every minute. The

\par\vspace{0.15cm}
\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Minutes} & 0 & 2 & 4 \\ \hline
\textbf{Litres} & 30 & 54 & 78 \\ \hline
\end{tabular}
\end{center}
\begin{itemize}
  \item[(a)] How many litres does the table give at 4 minutes?
        \par\vspace*{1.7cm}\ansline
  \item[(b)] How many litres does the equation give at 4 minutes?
        \par\vspace*{1.7cm}\ansline

        \par\vspace*{1.9cm}\ansline
\end{itemize}}
