\begin{center}
\small\textit{Naming a conic from its equation, and reading centres, radii, vertices, foci and asymptotes}
\end{center}

\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=2pt]
  \item The four shapes below are the four conics. No values are shown: each problem supplies its own numbers.
  \item The sign test: with both squared terms on one side, let $A$ and $C$ be the coefficients of $x^2$ and $y^2$. If $A\cdot C > 0$ the curve closes up (circle when $A = C$, otherwise an ellipse); if $A \cdot C < 0$ it opens in two branches (hyperbola); if one squared term is missing, it is a parabola.
  \item Keep exact values --- radii and focal distances stay as whole numbers or surds, never decimals.
\end{itemize}

\vspace{0.1cm}
\begin{tabular}{@{}l@{\quad}l@{}}
\textbf{Circle:} $(x-h)^2 + (y-k)^2 = r^2$ &
\textbf{Parabola:} $y = a(x-h)^2 + k$ \\[2pt]
\textbf{Ellipse:} $\dfrac{(x-h)^2}{a^2} + \dfrac{(y-k)^2}{b^2} = 1$ &
\textbf{Hyperbola:} $\dfrac{(y-k)^2}{a^2} - \dfrac{(x-h)^2}{b^2} = 1$ \\
\end{tabular}
