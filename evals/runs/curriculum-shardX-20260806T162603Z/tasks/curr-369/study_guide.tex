\sstitleblock{Holes and Vertical Asymptotes}

\skillheading{Skill 1 --- Factor top and bottom, then cancel}

\begin{formulabox}
\textbf{Rule:} a rational expression can only be simplified by cancelling a
\emph{whole factor} that appears in both the numerator and the denominator ---
never a single term out of a sum. Factor completely first, cancel the shared
factors, and keep the original denominator's zeros in mind: cancelling changes
the formula, not the domain.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $\dfrac{x^2-16}{x^2-2x-8}$.
\step{Nothing cancels while the parts are sums, so factor the top and the bottom
completely before looking for anything to divide out.}
\step{$\dfrac{x^2-16}{x^2-2x-8} = \dfrac{(x-4)(x+4)}{(x-4)(x+2)}$, and the factor
$(x-4)$ is shared.}

$\Rightarrow\quad \ans{\dfrac{x+4}{x+2}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $\dfrac{x^2-25}{x^2+x-20}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x-5}{x-4}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Hole or vertical asymptote?}

\begin{formulabox}
\textbf{Test, applied to each zero of the ORIGINAL denominator:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item its factor \textbf{cancels} $\Rightarrow$ \textbf{hole}: one point missing from an otherwise ordinary curve;
  \item its factor \textbf{survives} $\Rightarrow$ \textbf{vertical asymptote}: the values run off without bound.
\end{itemize}
Both kinds stay out of the domain. If every denominator factor cancels, the graph
has no vertical asymptote at all.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad For $f(x) = \dfrac{x^2+3x-4}{x^2-6x+5}$, which denominator
zero gives a hole and which gives a vertical asymptote?
\step{The decision is made by the factored form, so factor both parts and see
which factor divides out and which one is left downstairs.}
\step{$\dfrac{(x-1)(x+4)}{(x-1)(x-5)}$: the factor $(x-1)$ cancels, so $x=1$ is a
hole, while $(x-5)$ survives, so $x=5$ is a vertical asymptote.}

$\Rightarrow\quad f(x) = \ans{\dfrac{x+4}{x-5}}$ \ (hole at $x=1$, asymptote at $x=5$)
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad For $g(x) = \dfrac{x^2-2x-8}{x^2-16}$, simplify, then name
the hole and the asymptote.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\dfrac{x+2}{x+4}}$, hole at $x=4$, asymptote at $x=-4$}
\end{tryitbox}

\skillheading{Skill 3 --- Finding the height of a hole}

\begin{formulabox}
\textbf{Rule:} the hole sits at the $x$-value of the cancelled factor, and its
height is the value of the \emph{simplified} expression there. Substituting into
the original gives $\frac{0}{0}$, which is not $0$ and not an answer.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the coordinates of the hole in
$f(x) = \dfrac{x^2-4}{x^2-5x+6}$.
\step{Both parts vanish at the same $x$, which is the signature of a hole, so
cancel the shared factor and read the height off what is left.}
\step{$\dfrac{(x-2)(x+2)}{(x-2)(x-3)} = \dfrac{x+2}{x-3}$; the cancelled factor
puts the hole at $x=2$, and there $\dfrac{2+2}{2-3} = \dfrac{4}{-1}$.}

$\Rightarrow\quad$ hole at $\bigl(2,\ \ans{-4}\bigr)$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the coordinates of the hole in
$g(x) = \dfrac{x^2-1}{x^2+2x-3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: hole at $\bigl(1, \ans{\tfrac{1}{2}}\bigr)$}
\end{tryitbox}
