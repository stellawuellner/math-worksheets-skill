\sstitleblock{Distribution \& Like-Term Errors}

\skillheading{Skill 1 --- Distribute to every term}

\begin{formulabox}
\textbf{Rule:} \quad $a(b + c) = ab + ac$ \quad --- the outside factor multiplies
\emph{every} term inside the parentheses, never just the first one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Expand $4(2x + 3)$
\step{The 4 must reach both terms inside the parentheses, so multiply each term by 4 separately before adding.}
\step{$4 \cdot 2x = 8x$ and $4 \cdot 3 = 12$.}

$\Rightarrow\quad 4(2x + 3) = \ans{8x + 12}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Expand $3(4x + 5)$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{12x + 15}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the classic error is multiplying only the first term.
If the outside factor is negative, every inside term also flips its sign.
\end{watchoutbox}

\skillheading{Skill 2 --- Combine only like terms}

\begin{formulabox}
\textbf{Rule:} like terms have the \emph{same variable part}. Add or subtract the
coefficients; the variable part stays. The variable never disappears:
$ax - bx = (a - b)x$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Simplify $7x - 2x + 4$
\step{Only the two $x$-terms are alike, so combine their coefficients and leave the constant term alone.}
\step{$7x - 2x = 5x$; the $+4$ has no like partner, so it stays.}

$\Rightarrow\quad 7x - 2x + 4 = \ans{5x + 4}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Simplify $9x - 5x + 2$\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4x + 2}$}
\end{tryitbox}

\skillheading{Skill 3 --- Test equivalence with a number}

\begin{formulabox}
\textbf{Rule:} equivalent expressions give the \emph{same value for every} $x$.
To expose a suspected error, substitute one number into both expressions:
different values prove they are NOT equivalent.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Is $2(x + 3)$ the same as $2x + 3$?
\step{A single test value is enough to disprove equivalence, so substitute an easy number into both expressions and compare.}
\step{At $x = 4$: \ $2(4 + 3) = \ans{14}$, \ but \ $2 \cdot 4 + 3 = \ans{11}$. Different values --- not equivalent.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Test $x = 3$ to decide whether $5(x + 1)$ and $5x + 5$ are
equivalent.\\[2pt]
\rotatebox{180}{\footnotesize check: both give $\ans{20}$ --- equivalent}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} one matching value \emph{suggests} equivalence but does not
prove it; one \emph{different} value proves non-equivalence for certain.
\end{watchoutbox}
