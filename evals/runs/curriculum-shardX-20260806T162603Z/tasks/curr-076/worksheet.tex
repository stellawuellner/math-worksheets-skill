\wstitleblock{Anchor Facts: Twos, Fives, and Tens}{Grade 3}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item An \textbf{anchor fact} is a fact you already know well --- your twos, fives, and tens.
  \item Harder facts sit right next to an anchor: one more group, one less group, or a double.
  \item Each model below shows how an anchor connects to the fact --- fill in every blank.
\end{itemize}

\problem[2.8cm]{Count by fives. Fill in the three missing numbers:\par\vspace{4pt}
\begin{center}
$5,\quad 10,\quad 15,\quad$ \ansblank$,\quad$ \ansblank$,\quad$ \ansblank
\end{center}
The 6th number you say is $5 \times 6$.}

\problem[0pt]{This array has 2 rows with 8 dots in each row.
\begin{center}
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$\\[2pt]
$\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet\;\;\bullet$
\end{center}
It shows a twos fact and a double at the same time:\par\vspace{4pt}
$2 \times 8 = {}$~\ansblank \qquad and \qquad $8 + 8 = {}$~\ansblank\par\vspace*{2.7cm}}

\problem[2.8cm]{Tens are the easiest anchors of all --- just say the number and add a zero as you skip count:\par\vspace{4pt}
$10 \times 4 = {}$~\ansblank \qquad\qquad $10 \times 8 = {}$~\ansblank}

\problem[2.8cm]{Use doubling. First the twos anchor, then double it:\par\vspace{4pt}
$2 \times 6 = {}$~\ansblank \qquad so \qquad $4 \times 6 = \text{double that} = {}$~\ansblank}

\problem[2.8cm]{Use the fives anchor, then add one more group of 8:\par\vspace{4pt}
$5 \times 8 = {}$~\ansblank \qquad so \qquad $6 \times 8 = 5 \times 8 + 8 = {}$~\ansblank}

\problem[2.8cm]{Use the tens anchor, then take one group of 6 away:\par\vspace{4pt}
$10 \times 6 = {}$~\ansblank \qquad so \qquad $9 \times 6 = 10 \times 6 - 6 = {}$~\ansblank}

\problem{Which anchor family helps most for $9 \times 8$? Circle one:\par\vspace{4pt}
\begin{center}\textbf{twos} \qquad\qquad \textbf{fives} \qquad\qquad \textbf{tens}\end{center}
Now use the anchor you circled to find it: \quad $9 \times 8 = {}$~\ansblank\par\vspace*{2.6cm}}

\problem{\textbf{Challenge.} Use any anchor fact to find $7 \times 6$.\par\vspace{4pt}
The anchor fact I used: \ansline\par\vspace{8pt}
$7 \times 6 = {}$~\ansblank\par\vspace{8pt}
Explain in one sentence how your anchor helped.\par\vspace{6pt}\noindent\underline{\hspace{12.5cm}}\par\vspace*{1.8cm}}
