\wstitleblock{Adding, Subtracting, and Multiplying Polynomials}{Algebra 1}{}

\noindent\small
Adding and subtracting polynomials, distributing a monomial, and multiplying
binomials. Write every answer in standard form --- terms in order of
decreasing degree, like terms combined.
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item Only \textbf{like terms} combine: $3x^2$ and $x^2$ do, $3x^2$ and $3x$
        do not.
  \item To subtract, distribute the minus sign to \emph{every} term of the
        second polynomial first, then add.
  \item Multiplying uses the distributive property: every term of the first
        factor times every term of the second. Exponents \emph{add}:
        $x^2 \cdot x^3 = x^5$.
\end{itemize}
\normalsize

\vspace{0.15cm}

\problem[4.5cm]{Add: \quad $(3x^2 + 5x - 2) + (x^2 - 8x + 7)$}

\problem[4.5cm]{Subtract: \quad $(7x^2 - 4x + 1) - (2x^2 + 6x - 9)$}

\problem[4.5cm]{Multiply: \quad $-3x^2(4x^3 - 2x + 5)$}

\problem[4.5cm]{Multiply: \quad $(x + 6)(x - 4)$}

\problem[5cm]{Add, and watch for the missing degrees: \quad
$(2x^3 - x + 4) + (5x^2 + x - 9)$}

\problem[5.5cm]{Subtract: \quad $(4x^3 + x^2 - 6) - (4x^3 - 3x^2 + 2x - 6)$
\par\smallskip
Two of the four terms disappear. Say in one sentence which ones and why.}

\problem[5.5cm]{Multiply: \quad $(2x - 5)(3x + 4)$}

\problem[5.5cm]{Add these two-variable polynomials: \quad
$(3x^2y - 5xy + 2y^2) + (x^2y + 5xy - 9y^2)$}

\problem[5.5cm]{Multiply: \quad $4xy^2(2x - 3y)$}

\problem[6cm]{A student wrote $(x - 3)^2 = x^2 - 9$. Explain in one sentence
why squaring a binomial is not the same as squaring each term, then write the
correct expansion of $(x - 3)^2$.}

\problem[6cm]{Multiply $(3x - 2)(x + 7)$ and write the product in standard
form. Then evaluate your product at $x = 2$. A classmate reports the value
$-2$; find the value yourself and say in one sentence what the classmate left
out.}

\problem[7cm]{\textbf{Synthesis challenge.} Simplify completely and write the
result in standard form:
\[ (x + 2)(x^2 - 3x + 5) - (x - 1)(x + 4) \]
Multiply both products first, then subtract the second from the first.}
