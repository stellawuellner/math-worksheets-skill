\wstitleblock{Making a Piecewise Model Continuous}{AP Calculus AB/BC}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Each model below is built in pieces. A model is continuous at a junction $x = c$ when the left-hand limit, the right-hand limit, and the value $f(c)$ are all the same number.
  \item So set the two one-sided limits equal at the junction and solve for the unknown constant. Write the equation before you solve it.
  \item Units are given in every stem. Carry them through, and check that your constant has a sensible size for the quantity it belongs to.
\end{itemize}

\problem{An oven's temperature, in degrees Celsius, $t$ minutes after it is
switched on is modelled by
\[
  T(t) = \begin{cases} 20t + a, & 0 \le t < 5 \\[2pt] 150 - 2t, & t \ge 5 \end{cases}
\]
Find the constant $a$ that makes $T$ continuous at $t = 5$ minutes.
\par\vspace*{4.6cm}\answerline{degrees}}

\problem[4.6cm]{A tank holds $V(x)$ litres of water $x$ minutes after filling
begins:
\[
  V(x) = \begin{cases} 4x + 6, & 0 \le x < 6 \\[2pt] kx, & x \ge 6 \end{cases}
\]
Find the constant $k$ that makes $V$ continuous at $x = 6$ minutes.}

\problem[4.6cm]{A sensor reports $r(x) = \dfrac{x^2 - 25}{x - 5}$ for every
pressure $x \neq 5$ bar, and its reading at $x = 5$ is a stored constant $c$.
Choose $c$ so that $r$ is continuous at $x = 5$ bar.}

\problem{A drone's height above the launch pad, in metres, $t$ seconds after
launch is
\[
  h(t) = \begin{cases} t^2 + a, & 0 \le t < 3 \\[2pt] 4t + 3, & t \ge 3 \end{cases}
\]
Find the constant $a$ that makes $h$ continuous at $t = 3$ seconds.
\par\vspace*{4.8cm}\answerline{m}}

\problem[5.4cm]{A calibration curve returns
$E(x) = \dfrac{\sqrt{x + 7} - 3}{x - 2}$ for every input $x \neq 2$, and the
value stored at $x = 2$ is a constant $c$. Choose $c$ so that $E$ is continuous
at $x = 2$. Give the exact value, not a decimal.}

\problem[4.8cm]{A three-stage conveyor speed, in metres per second, at position
$x$ metres along the belt is
\[
  f(x) = \begin{cases} x^2, & x < 2 \\[2pt] ax + b, & 2 \le x < 5 \\[2pt] 3x + 1, & x \ge 5 \end{cases}
\]
Both junctions must be continuous. Write the equation each junction gives, then
solve the pair.
\par\vspace{0.2cm}
$a = $ \ansblank \qquad $b = $ \ansblank}

\problem[5.2cm]{A phone plan charges $C(g)$ dollars for $g$ gigabytes of data:
\[
  C(g) = \begin{cases} 30 + 12g, & 0 \le g < 4 \\[2pt] a + 7g, & g \ge 4 \end{cases}
\]
\textbf{(a)} Find the constant $a$ that makes $C$ continuous at $g = 4$
gigabytes: \quad \ansblank
\par\vspace{0.2cm}
\textbf{(b)} Suppose the company published $a = 45$ instead. Explain what that
would do to the bill of a customer whose usage is very close to 4 gigabytes.}

where $x$ is measured in seconds. Find the constants $a$ and $b$ that make $g$
continuous at $x = 4$ seconds. All three conditions of continuity must hold.
