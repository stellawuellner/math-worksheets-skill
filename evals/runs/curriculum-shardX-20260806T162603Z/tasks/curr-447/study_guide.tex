\sstitleblock{Using Normal-Model z-Scores and Percentiles}

\skillheading{Skill 1 --- Standardizing: from a value to $z$}

\begin{formulabox}
\textbf{Formula:}\quad $z = \dfrac{x - \mu}{\sigma}$ --- how many standard deviations the value
sits above (positive) or below (negative) the mean. Subtract first, divide second. A $z$-score
carries no units, which is exactly what makes it comparable across different distributions.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A distribution has $\mu = 65$ and $\sigma = 6$. Find the $z$-score of $74$.
\step{The question asks how far above average the value is in standard-deviation units, so standardize.}
\step{$z = \dfrac{74 - 65}{6} = \dfrac{9}{6}$.}

$\Rightarrow\quad z = \ans{\tfrac{3}{2}}$, that is, $1.5$ standard deviations above the mean.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $\mu = 65$, $\sigma = 6$. Find the $z$-score of $58$.\\[2pt]
\rotatebox{180}{\footnotesize check: $z = \ans{-\tfrac{7}{6}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Unstandardizing: from $z$ back to a value}

\begin{formulabox}
\textbf{Formula:}\quad $x = \mu + z\sigma$. Read it as ``start at the mean, then walk $z$
standard deviations''. A negative $z$ walks downward. This is the move that answers every
``find the score at the $k$th percentile'' question once you know the $z$ for that percentile.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $\mu = 200$, $\sigma = 25$. Find the value with $z = 1.2$.
\step{We are given the position in standard-deviation units and want the raw value, so run the formula backwards.}
\step{$x = 200 + 1.2(25) = 200 + 30$.}

$\Rightarrow\quad x = \ans{230}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Same $\mu = 200$, $\sigma = 25$. Find the value with $z = -0.8$.\\[2pt]
\rotatebox{180}{\footnotesize check: $x = \ans{180}$}
\end{tryitbox}

\skillheading{Skill 3 --- Percentiles from the empirical rule}

\begin{formulabox}
\textbf{Rule:} in a normal model about $68\%$ of values lie within $1\sigma$ of the mean,
$95\%$ within $2\sigma$, and $99.7\%$ within $3\sigma$. The curve is symmetric, so half of the
remainder sits in each tail. To get a percentile, add $50$ (everything below the mean) to half
of the central percentage --- or subtract, for a point below the mean.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What percentage of a normal distribution lies below $z = 2$?
\step{The rule gives a symmetric central band, so split it in half and add the half of the curve below the mean.}
\step{$50 + \dfrac{95}{2} = 50 + 47.5$.}

$\Rightarrow\quad \ans{97.5}$ percent, so $z = 2$ marks the $97.5$th percentile.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad What percentage lies below $z = -2$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{2.5}$ percent}
\end{tryitbox}

\skillheading{Skill 4 --- Comparing across different distributions}

\begin{formulabox}
\textbf{Rule:} two raw values from different distributions cannot be compared directly ---
they are measured against different means and different spreads. Standardize each against
\emph{its own} $\mu$ and $\sigma$, then compare the $z$-scores. The larger $z$ is the more
extreme result relative to its own group, whatever the raw numbers say.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A time of $33$ comes from a group with $\mu = 30$, $\sigma = 2$. Find its $z$-score.
\step{Before any comparison can be made, each value has to be measured against its own group, so standardize this one first.}
\step{$z = \dfrac{33 - 30}{2}$.}

$\Rightarrow\quad z = \ans{\tfrac{3}{2}}$, which can now be set beside any other group's $z$.
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A score of $62$ comes from a group with $\mu = 50$, $\sigma = 10$. Find its $z$-score.\\[2pt]
\rotatebox{180}{\footnotesize check: $z = \ans{\tfrac{6}{5}}$}
\end{tryitbox}
