\wstitleblock{Solving Systems by Substitution}{Algebra 1}{}

\begin{center}
\textit{Substituting an expression that is ready, isolating a variable first, and reading the two special cases}
\end{center}

\vspace{0.1cm}
\noindent\textbf{The pattern to copy}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item If one equation already says ``$y =$ something'' or ``$x =$ something'', put that something
        into the \emph{other} equation in place of the variable. Never back into the same equation
        it came from --- that only tells you $0 = 0$.
  \item If neither equation is ready, isolate whichever variable has a coefficient of $1$ or $-1$;
        that keeps the fractions away.
  \item Solve for one variable, substitute back to get the other, and write the answer as an
        ordered pair $(x, y)$. Check it in \emph{both} original equations.
\end{itemize}
