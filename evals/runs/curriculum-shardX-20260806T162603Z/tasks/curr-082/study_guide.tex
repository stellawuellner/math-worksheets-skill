\sstitleblock{Fractions on the Number Line}

\skillheading{Skill 1 --- Reading the Point on a Line}

\begin{formulabox}
\textbf{Two counts make a fraction.}\par
Bottom number: how many equal parts the whole from $0$ to $1$ is cut into.\par
Top number: how many of those parts you count off from $0$.\par
One part is one jump, and the jump is always the same size on one line.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad What fraction is the marked point?
\par\vspace{2pt}
\begin{center}
\begin{tikzpicture}[x=1.5cm]
  \draw[thick,->] (-0.25,0) -- (3.4,0);
  \foreach \i in {0,1,2,3}{\draw[thick] (\i,-0.13) -- (\i,0.13);}
  \node[below] at (0,-0.16) {$0$};
  \node[below] at (3,-0.16) {$1$};
  \fill[red!80!black] (2,0) circle (2.4pt);
\end{tikzpicture}
\end{center}
\step{The point sits on a tick, so I do not have to measure anything --- I only have to count parts in the whole and then count ticks from zero.}
\step{The whole splits into $3$ equal parts, so one jump is $\frac{1}{3}$; the dot is on the second tick, which is $2$ jumps.}

the marked point is $\ans{\tfrac{2}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A whole cut into $6$ equal parts, with a dot on the fifth tick after $0$.
What fraction is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{6}}$}
\end{tryitbox}

\skillheading{Skill 2 --- Placing a Fraction by Counting Jumps}

\begin{formulabox}
\textbf{To place $\frac{a}{b}$:} cut the whole into $b$ equal parts, then count $a$ jumps from $0$.\par
The number of jumps IS the top number. A fraction bigger than one whole is nothing special --- just
keep jumping past $1$.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Where does $\frac{7}{8}$ go, and how many jumps is it?
\step{The bottom number tells me the jump size and the top number tells me how many jumps, so I set the jump to one eighth and count from zero.}
\step{Counting $\frac{1}{8}$ at a time reaches $\frac{7}{8}$ after $7$ jumps --- one tick short of $1$, so the dot goes on the seventh tick.}

number of jumps $= \ans{7}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many jumps of $\frac{1}{3}$ from $0$ land on $\frac{4}{3}$? (Careful ---
this one goes past $1$.)\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4}$}
\end{tryitbox}

\skillheading{Skill 3 --- Comparing Two Fractions on a Line}

\begin{formulabox}
\textbf{Rule:} on a number line, further to the right means larger.\par
Same bottom number? The one with more jumps is larger.\par
Same top number, different bottom? More parts in the whole makes each jump smaller, so the bigger
bottom number gives the smaller fraction.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Compare $\frac{2}{6}$ and $\frac{2}{3}$.
\step{The two fractions count the same number of jumps, so what decides the comparison is the size of a jump, not the count --- I compare the bottom numbers.}
\step{Sixths are smaller jumps than thirds, so two sixths lands closer to $0$ than two thirds does.}

$\ans{\tfrac{2}{6} < \tfrac{2}{3}}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $>$, $<$ or $=$ between $\frac{5}{8}$ and $\frac{3}{8}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{\tfrac{5}{8} > \tfrac{3}{8}}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} count the \emph{spaces} between ticks, not the ticks themselves. A whole cut
into fourths shows five tick marks but only four equal parts, and counting the marks is the most
common way to end up naming the wrong fraction.
\end{watchoutbox}
