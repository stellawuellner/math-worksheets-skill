\sstitleblock{Diagnosing Direct-Substitution and One-Sided-Sign Errors}

\skillheading{Skill 1 --- Reading What Substitution Reports}

\begin{formulabox}
\textbf{Substitute first, then read the report.} A real number means the function is continuous there and the substitution is the limit.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item $0/0$ is \emph{indeterminate}, not undefined --- it says the substitution failed, and nothing about the answer.
  \item Repair routes: factor and cancel; multiply by the conjugate; combine fractions over a common denominator.
  \item Cancelling is legal because the two expressions agree everywhere except at the point, and a limit never looks at the point itself.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\lim_{x \to 5} \frac{x^2 - 25}{x - 5}$.
\step{Substitution gives $0/0$, so the only thing it has reported is that $x - 5$ divides both parts; factor before substituting again.}
\step{$\dfrac{(x-5)(x+5)}{x-5} = x + 5$ for $x \ne 5$, and substituting into that gives $\ans{10}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\lim_{x \to -2} \frac{x^2 + 5x + 6}{x + 2}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{1}$}
\end{tryitbox}

\skillheading{Skill 2 --- One-Sided Signs at an Infinite Limit}

\begin{formulabox}
\textbf{Nonzero over zero is infinite --- two signs decide which infinity.} Take the sign of the numerator at the point, then the sign of the denominator on the side you are approaching from, and divide.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Approaching $a$ from the right makes $x - a$ a small \emph{positive} number; from the left it is a small \emph{negative} one.
  \item ``From the right, so $+\infty$'' is wrong whenever the numerator is negative.
  \item If the two sides give opposite infinities, the two-sided limit does not exist. That is a different statement from $\infty$.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Find $\displaystyle\lim_{x \to 3^{+}} \frac{x + 1}{x - 3}$.
\step{Substitution gives $4/0$, which is infinite rather than indeterminate, so the work left to do is entirely about signs.}
\step{The numerator is near $+4$ and $x - 3$ is a small positive number from the right, so the quotient is large and positive: $\ans{+\infty}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Find $\displaystyle\lim_{x \to 3^{-}} \frac{x + 1}{x - 3}$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-\infty}$}
\end{tryitbox}

\skillheading{Skill 3 --- Making a Piecewise Function Continuous}

\begin{formulabox}
\textbf{Continuity at a joint $x = a$ needs three things to agree:} the left-hand limit, the right-hand limit, and the value $f(a)$.
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item Evaluate each branch at the joint, set the two results equal, and solve for the unknown constant.
  \item The unknown is not the branch value. Reading the constant straight off one branch leaves a jump.
  \item Check by substituting your constant back: both branches must return the same number.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad $f(x) = x^2 + c$ for $x < 2$ and $f(x) = 3x + 4$ for $x \ge 2$. Find $c$.
\step{Continuity at the joint means the two branch expressions must produce the same number there, so evaluate each at the joint and set them equal.}
\step{The branches give $4 + c$ and $10$, so $4 + c = 10$ and $\ans{c = 6}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad $f(x) = x^2 + c$ for $x < 4$ and $f(x) = 5x + 1$ for $x \ge 4$. Find $c$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{c = 5}$}
\end{tryitbox}
