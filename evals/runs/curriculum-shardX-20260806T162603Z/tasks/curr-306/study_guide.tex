\sstitleblock{Naming the Angle Pairs a Transversal Makes}

\skillheading{Skill 1 --- Reading the picture and naming the pair}

\begin{formulabox}
Two words locate every pair. \textbf{Interior} means between the two lines; \textbf{exterior} means outside them. \textbf{Alternate} means opposite sides of the transversal; \textbf{same-side} means the same side.\par
\begin{itemize}[leftmargin=1.2cm, itemsep=0pt, topsep=1pt]
  \item Corresponding --- one interior, one exterior, same side: \emph{congruent}
  \item Alternate interior, alternate exterior: \emph{congruent}
  \item Same-side interior: \emph{supplementary}
\end{itemize}
These three lines need the two lines to be \textbf{parallel}. Vertical angles and linear pairs do not --- they hold at any crossing.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$ and $m\angle 4 = 112^\circ$, where $\angle 4$ and $\angle 6$ are same-side interior angles. Find $m\angle 6$.
\step{Both angles sit between the lines and on the same side of the transversal, so the pair is same-side interior --- and that name says supplementary, so I subtract from $180$ instead of copying the measure.}
\step{$m\angle 6 = 180 - 112$.}

$\Rightarrow\quad \ans{68}$ degrees
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad $j \parallel k$ and one same-side interior angle measures $133^\circ$. Find the other.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{47}$}
\end{tryitbox}

\skillheading{Skill 2 --- Turning the name into an equation}

\begin{formulabox}
\textbf{Congruent pair} (corresponding, alternate interior, alternate exterior): set the two expressions \textbf{equal}.\par
\textbf{Supplementary pair} (same-side interior, and any linear pair): \textbf{add} the two expressions and set the sum to $180$.\par
Write the name down before writing the equation. Choosing ``equal'' for a same-side pair is the most common error on this topic, and the arithmetic afterwards hides it.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$, and corresponding angles measure $(2x + 35)^\circ$ and $(4x - 15)^\circ$. Find $x$.
\step{The pair is corresponding and the lines are parallel, so the two angles are congruent --- which makes the equation an equality of the two expressions, not a sum.}
\step{$2x + 35 = 4x - 15$, so $50 = 2x$.}

$\Rightarrow\quad \ans{x = 25}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Alternate interior angles measure $(3x + 12)^\circ$ and $(5x - 28)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 20}$}
\end{tryitbox}

\skillheading{Skill 3 --- Chaining two relationships}

\begin{formulabox}
When the two angles are not a named pair, travel through a third angle: use a \textbf{corresponding} or \textbf{vertical} angle to move to the other intersection, then a \textbf{linear pair} to move around that vertex.\par
Supplementary chain: if a same-side interior pair is $(a)^\circ$ and $(b)^\circ$, then $a + b = 180$; solve for the unknown, then substitute back to get each measure.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad $j \parallel k$, and a same-side interior pair measures $(5x + 10)^\circ$ and $(3x + 26)^\circ$. Find $x$.
\step{The pair is same-side interior, so the two measures add to $180$; that single equation carries both expressions and the one unknown.}
\step{$(5x + 10) + (3x + 26) = 180$, so $8x + 36 = 180$ and $8x = 144$.}

$\Rightarrow\quad \ans{x = 18}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A same-side interior pair measures $(7x + 4)^\circ$ and $(2x + 5)^\circ$. Find $x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 19}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} no parallel marks, no parallel rules. Without them only vertical angles and linear pairs are guaranteed.
\end{watchoutbox}
