% =================== SKILL 1 ===================
\skillheading{Surface Area of a Rectangular Prism}

\begin{formulabox}
\textbf{Formula:} $SA = 2(\ell w + \ell h + wh)$ --- three pairs of matching
faces (top/bottom, front/back, two ends). A cube shortcut: $SA = 6s^2$. For an
\emph{open} container, list the faces that exist and leave out the missing one.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the surface area of a closed $5 \times 2 \times 3$
cm box.
\step{List the three face pairs from length, width, and height, then double
their sum.}
\step{$SA = 2(5 \cdot 2 + 5 \cdot 3 + 2 \cdot 3) = 2(10 + 15 + 6) = 2 \cdot 31
= 62$.}

$\Rightarrow\quad \ans{62}$ cm$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the surface area of a cube with $4$-inch
edges.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{96}$ in$^2$}
\end{tryitbox}

% =================== SKILL 2 ===================
\skillheading{Area of a Composite Shape}

\begin{formulabox}
\textbf{Strategy:} cut the shape into rectangles (and triangles) along its
corners, find each piece's area, and add. Read each piece's side lengths off
the corner coordinates by subtracting.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad An L-shape has corners $(0,0)$, $(6,0)$, $(6,3)$,
$(2,3)$, $(2,5)$, $(0,5)$. Find its area.
\step{Cut along $y = 3$: a $6 \times 3$ bottom rectangle and a $2 \times 2$
top-left piece.}
\step{$6 \cdot 3 = 18$ and $2 \cdot 2 = 4$; total $18 + 4 = 22$.}

$\Rightarrow\quad \ans{22}$ square units
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find the area of the L-shape with corners $(0,0)$,
$(7,0)$, $(7,2)$, $(3,2)$, $(3,6)$, $(0,6)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{26}$ square units}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Scale Drawings: Lengths}

\begin{formulabox}
\textbf{Rule:} a scale like $1$ cm $: 6$ m tells you what one drawing unit is
worth. Drawing $\to$ real: \emph{multiply} by the scale. Real $\to$ drawing:
\emph{divide} by the scale.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Scale $1$ cm $: 6$ m. A hallway is $4.5$ cm on the
drawing. How long is the real hallway?
\step{Drawing to real --- multiply the drawing length by the scale value.}
\step{$4.5 \cdot 6 = 27$.}

$\Rightarrow\quad \ans{27}$ m
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Scale $1$ in $: 9$ ft. A porch is $2.5$ in on the plan.
Find the real length.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{22.5}$ ft}
\end{tryitbox}

% =================== SKILL 4 ===================
\skillheading{Scale Factors and Area}

\begin{formulabox}
\textbf{Rule:} when every length is multiplied by scale factor $k$, area is
multiplied by $k^2$ --- both the length \emph{and} the width stretch. Doubling
lengths quadruples area; a $1:20$ model has $\tfrac{1}{400}$ the floor area.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A sticker of area $5$ cm$^2$ is enlarged by scale
factor $4$. Find the new area.
\step{Area scales by the \emph{square} of the factor, so multiply by $4^2$,
never by $4$.}
\step{$5 \cdot 4^2 = 5 \cdot 16 = 80$.}

$\Rightarrow\quad \ans{80}$ cm$^2$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A photo of area $2.5$ in$^2$ is enlarged by scale factor
$10$. Find the new area.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{250}$ in$^2$}
\end{tryitbox}
