\sstitleblock{Reading Arrays: Rows, Groups, and the Total}

\skillheading{Skill 1 --- Read an Array as Multiplication}

\begin{formulabox}
\textbf{The model:} an array is equal groups. Each \emph{row} is one group.\\[3pt]
\textbf{rows} $\times$ \textbf{how many in each row} $=$ \textbf{total}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item ``how many rows'' $\rightarrow$ count down the side (the number of groups)
  \item ``how many in each row'' $\rightarrow$ count across one row (the size of a group)
  \item ``how many in all / altogether / total'' $\rightarrow$ multiply the two
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad An array has 4 rows with 3 counters in each row. How many counters in all?
\step{``In all'' asks for every counter, not for the number of rows, so this is the multiply question.}
\step{Skip-count by threes, once for each of the 4 rows: 3, 6, 9, 12. So $4 \times 3 = \ans{12}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad An array has 5 rows with 2 counters in each row. How many counters in all?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{10}$}
\end{tryitbox}

\skillheading{Skill 2 --- Fix a Rows-Versus-Total Mistake}

\begin{formulabox}
\textbf{Two checks that catch the mistake every time:}
\begin{itemize}[leftmargin=0.8cm, itemsep=0pt, topsep=1pt]
  \item The total counts \emph{objects}. The number of rows counts \emph{groups}. They are different things.
  \item Unless every row holds exactly one object, the total is bigger than the number of rows.
\end{itemize}
\textbf{Adding is not the fix either:} rows $+$ in each row answers nothing in the picture.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad Sara has 6 rows with 4 beads in each row. She says, ``6 beads in all.''
\step{Sara reported the number of rows; the question asked for beads, so she answered the wrong one of the two counts.}
\step{Keep her two numbers and multiply: skip-count by fours six times --- 4, 8, 12, 16, 20, 24. So $6 \times 4 = \ans{24}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad Ben has 3 rows with 7 marbles in each row. He says ``3 marbles.'' What is the correct total?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{21}$}
\end{tryitbox}

\skillheading{Skill 3 --- Find the Missing Rows or Group Size}

\begin{formulabox}
\textbf{When the total is given and one part is missing:}\\[3pt]
rows $\times$ in each row $=$ total \quad $\rightarrow$ \quad ``$\;\underline{\hspace{0.9cm}}$ groups of the known size make the total''\\[3pt]
Skip-count by the number you know and stop at the total. The number of counts you said is the missing number. Then check by multiplying back.
\end{formulabox}

\begin{examplebox}
\textbf{Example:}\quad 20 pencils are put into rows of 5. How many rows?
\step{The total (20) and the group size (5) are known, so the missing number is how many groups --- that is the rows.}
\step{Skip-count by fives to 20: 5, 10, 15, 20 is 4 counts. Check $5 \times 4 = 20$ $\checkmark$, so $\ans{n = 4}$.}
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:}\quad 42 stickers are put into 6 equal rows. How many stickers are in each row?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{n = 7}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} Before you write an answer, say out loud what it counts ---
``rows'' or ``objects.'' If the question said \emph{in all} and your answer is the
row count, you answered the other question.
\end{watchoutbox}
