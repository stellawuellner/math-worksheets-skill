\sstitleblock{Place Value with Base-Ten Models}

% =================== SKILL 1 ===================
\skillheading{Skill 1 --- Reading a base-ten model}

\begin{formulabox}
\textbf{Block values:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item a \textbf{flat} (big square) $=$ 1 hundred
  \item a \textbf{rod} (tall bar) $=$ 1 ten
  \item a \textbf{cube} (small square) $=$ 1 one
\end{itemize}
Count each kind of block, then add the values.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A model shows 2 flats, 3 rods, and 4 cubes. What number is it?
\step{Count each size of block separately: flats are hundreds, rods are tens, cubes are ones.}
\step{2 hundreds $+$ 3 tens $+$ 4 ones $= 100 \cdot 2 + 10 \cdot 3 + 4$}
$\Rightarrow\quad \ans{234}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A model shows 3 flats, 1 rod, and 6 cubes. What number is it?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{316}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} If a model has \emph{no} rods, the tens place still
needs a digit --- write a zero there. Three flats and eight cubes make
\emph{three hundred eight}, never thirty-eight.
\end{watchoutbox}

% =================== SKILL 2 ===================
\skillheading{Skill 2 --- Hundreds, tens, and ones in a number}

\begin{formulabox}
\textbf{Rule:} Each digit's place tells its value.
In a three-digit number, the first digit counts \textbf{hundreds}, the middle
digit counts \textbf{tens}, and the last digit counts \textbf{ones}.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad How many hundreds, tens, and ones are in 358?
\step{Read the digits from left to right and name each place.}
\step{The 358 splits by place: hundreds digit, tens digit, ones digit.}
$\Rightarrow\quad \ans{3}$ hundreds, $\ans{5}$ tens, $\ans{8}$ ones
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad How many hundreds, tens, and ones are in 726?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{7}$ hundreds, $\ans{2}$ tens, $\ans{6}$ ones}
\end{tryitbox}

% =================== SKILL 3 ===================
\skillheading{Skill 3 --- Comparing two models}

\begin{formulabox}
\textbf{Rule:} Compare place by place, biggest first.
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item First compare \textbf{hundreds}. More hundreds wins.
  \item If they tie, compare \textbf{tens}. If those tie too, compare \textbf{ones}.
\end{itemize}
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Which is greater, 341 or 314?
\step{Compare hundreds first: both numbers have three hundreds, so move to the tens.}
\step{341 has four tens and 314 has only one ten, so 341 is greater.}
$\Rightarrow\quad \ans{341 > 314}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Write $<$, $>$, or $=$: compare 562 and 526.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{562 > 526}$}
\end{tryitbox}
