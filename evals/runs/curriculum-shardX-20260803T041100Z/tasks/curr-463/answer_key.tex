\problem{$f(x) = x^5 - 3x^3 + 2x$; find $f''(x)$. \\
Power rule, one rung at a time:
\[ f'(x) = 5x^4 - 9x^2 + 2 \qquad\Longrightarrow\qquad f''(x) = 20x^3 - 18x \]
\ans{f''(x) = 20x^3 - 18x}
\par\noindent\emph{Check the pattern:} each rung drops the degree by one, so a
quintic has a cubic second derivative $\checkmark$}

\problem{$g(x) = 2x^4 - x^2 + 7$; find $g'''(x)$. \\
Three rungs. The constant $7$ dies on the first rung and never returns:
\[ g'(x) = 8x^3 - 2x \qquad g''(x) = 24x^2 - 2 \qquad g'''(x) = 48x \]
\ans{g'''(x) = 48x}
\par\noindent\emph{Note for the grader:} a fourth derivative would be the
constant $48$, and every derivative after that is $0$ --- the degree, $4$, is
exactly how many rungs it takes to reach a constant.}

\problem{$f(x) = \sin(3x)$; find $f''(x)$. \\
Chain rule on each rung; the inner derivative is $3$ both times:
\[ f'(x) = 3\cos(3x) \qquad\Longrightarrow\qquad
   f''(x) = 3 \cdot \bigl(-\sin(3x)\bigr) \cdot 3 = -9\sin(3x) \]
\ans{f''(x) = -9\sin(3x)}
\par\noindent\emph{Common error:} writing $-\sin(3x)$, which drops the inner
derivative on the second rung. The factor of $3$ appears once per
differentiation, so the second derivative carries $3^2$.}

\problem{$h(x) = x^2 e^x$; find $h''(x)$. \\
Product rule twice, simplifying between rungs:
\begin{align*}
  h'(x) &= 2x e^x + x^2 e^x = e^x(x^2 + 2x) \\
  h''(x) &= e^x(x^2 + 2x) + e^x(2x + 2) = e^x(x^2 + 4x + 2)
\end{align*}
\ans{h''(x) = (x^2 + 4x + 2)e^x}
\par\noindent\emph{Why factoring first pays:} differentiating the unsimplified
$2xe^x + x^2e^x$ needs the product rule on two terms instead of one.}

\problem{Drone: $s(t) = t^3 - 6t^2 + 9t$ metres. \\
\textbf{(a) Acceleration.} Velocity is the first rung, acceleration the second:
\[ v(t) = s'(t) = 3t^2 - 12t + 9 \qquad\Longrightarrow\qquad a(t) = s''(t) = 6t - 12 \]
in metres per second squared. \ans{a(t) = 6t - 12}
\par\noindent\textbf{(b) Zero acceleration.} Solve $6t - 12 = 0$, giving
$t = 2$ seconds --- inside the stated window $0 \le t \le 5$. \ans{t = 2}
\par\noindent\emph{Interpretation to look for:} before $t = 2$ the acceleration
is negative and the drone is slowing its rise; after $t = 2$ it is positive.
That instant is the inflection point of the height graph.}

\problem{$f(x) = \ln x + x^3$, $x > 0$. \\
\textbf{(a) Second derivative.} The derivative of $\ln x$ is $x^{-1}$, whose
derivative is $-x^{-2}$:
\[ f'(x) = \frac{1}{x} + 3x^2 \qquad\Longrightarrow\qquad
   f''(x) = -\frac{1}{x^2} + 6x \]
\ans{f''(x) = -1/x^2 + 6x}
\par\noindent\textbf{(b) Value and concavity.} At $x = 2$:
$f''(2) = -\tfrac{1}{4} + 12 = 11.75$. \ans{f''(2) = 11.75}
\par\noindent Because $f''(2) > 0$, the graph of $f$ is concave up at $x = 2$.
\emph{Common error:} differentiating $x^{-1}$ to $x^{-2}$ without the sign
change, which flips the concavity conclusion.}

\problem{Circle $x^2 + y^2 = 25$; find $\dfrac{d^2y}{dx^2}$ and its value at
$(3,4)$. \\
\textbf{(a) First rung.} Differentiate both sides with respect to $x$, treating
$y$ as a function of $x$:
\[ 2x + 2y\frac{dy}{dx} = 0 \qquad\Longrightarrow\qquad \frac{dy}{dx} = -\frac{x}{y} \]
\textbf{Second rung.} Quotient rule on $-x/y$, remembering that $y$ is a
function of $x$:
\[ \frac{d^2y}{dx^2} = -\frac{y - x\dfrac{dy}{dx}}{y^2}
  = -\frac{y - x\left(-\dfrac{x}{y}\right)}{y^2}
  = -\frac{y^2 + x^2}{y^3} \]
Now substitute the original equation, $x^2 + y^2 = 25$:
\ans{\dfrac{d^2y}{dx^2} = \dfrac{-25}{(25-x^2)^{3/2}}}
\par\noindent which is the same as $-25/y^3$ on the upper arc, where
$y = \sqrt{25 - x^2}$.
\par\noindent\textbf{(b) At $(3,4)$.} Substituting $y = 4$:
$-25/4^3 = -25/64$. \ans{-\dfrac{25}{64}}
\par\noindent\emph{Sanity check:} the point $(3,4)$ is on the upper arc, which
is concave down, and the value is negative $\checkmark$}

\problem{Particle: $s(t) = t^4 - 4t^3 + 12$ metres. \\
\textbf{(a) Acceleration.}
\[ v(t) = s'(t) = 4t^3 - 12t^2 \qquad\Longrightarrow\qquad
   a(t) = s''(t) = 12t^2 - 24t \]
in metres per second squared. \ans{a(t) = 12t^2 - 24t}
\par\noindent\textbf{(b) Zero acceleration.} Factor:
$12t^2 - 24t = 12t(t - 2)$, so $a(t) = 0$ at $t = 0$ and $t = 2$ seconds, both
in the domain $t \ge 0$. \ans{t = 0, \ t = 2}
\par\noindent\textbf{(c) Concavity (manual review).} The factored form
$12t(t-2)$ is negative exactly when the two factors have opposite signs, which
on $t \ge 0$ happens for $0 < t < 2$. So the position graph is concave down on
$0 < t < 2$ and concave up for $t > 2$; $t = 2$ is an inflection point.
\emph{Credit any correct reasoning that reaches the interval $0 < t < 2$ from
the sign of $a(t)$ --- a sign chart, the factored form, or test values.}}
