\wstitleblock{Finding Inverses with Domain Restrictions}{Precalculus \quad Advanced Function Transformations and Composition}{}

\begin{center}
\textit{Finding the inverse and stating its domain, restricting a quadratic to
one branch before inverting, and verifying an inverse pair by composition}
\end{center}

\noindent\textbf{Directions.}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt, topsep=2pt]
\item A function has an inverse function only if it is \textbf{one-to-one}. If
it is not, restrict the domain first and say which branch you kept.
\item \textbf{An inverse is not finished until its domain is stated.} The
domain of $f^{-1}$ is the range of $f$, and the range of $f^{-1}$ is the
(restricted) domain of $f$.
\item Method: write $y = f(x)$, swap $x$ and $y$, solve for $y$, then choose
the sign or branch that matches the restriction you kept.
\item Check every inverse by composing: $f(f^{-1}(x))$ must simplify to $x$, or
by testing one value both ways.
\end{itemize}

\vspace{0.2cm}

\problem[5cm]{Let $f(x) = 3x - 7$, with domain all real numbers.

(a) Find a formula for $f^{-1}(x)$. \quad (b) Compute $f^{-1}(8)$.}

\problem[5.5cm]{Let $f(x) = \sqrt{x - 4}$, whose domain is $x \geq 4$.

(a) Find a formula for $f^{-1}(x)$ and state its domain.

(b) Compute $f^{-1}(3)$, then check your formula by computing $f(13)$.}

\problem[6cm]{Let $f(x) = x^2 - 6x + 5$, restricted to $x \geq 3$ so that it is
one-to-one.

(a) Find a formula for $f^{-1}(x)$ and state its domain.

(b) Compute $f^{-1}(12)$, then check by computing $f(7)$.}

\problem[6cm]{Let $g(x) = \dfrac{2x + 1}{x - 3}$, with $x \neq 3$.

(a) Verify by composition that $g^{-1}(x) = \dfrac{3x + 1}{x - 2}$.

(b) State the value that must be excluded from the domain of $g^{-1}$, and say
which feature of $g$ it comes from.

(c) Compute $g^{-1}(3)$.}

\problem[6cm]{Let $f(x) = (x + 2)^2 - 5$, restricted to $x \leq -2$.

(a) Find a formula for $f^{-1}(x)$ and state its domain. Say clearly which
branch of the parabola you kept and how that fixed the sign.

(b) Compute $f^{-1}(11)$, then check by computing $f(-6)$.}

\problem[6cm]{Let $f(x) = 9 - x^2$, restricted to $x \leq 0$.

(a) Find a formula for $f^{-1}(x)$ and state both its domain and its range.

(b) Compute $f^{-1}(5)$, then check by computing $f(-2)$.}

\problem[6cm]{Let $f(x) = \sqrt{2x - 6} + 1$, whose domain is $x \geq 3$.

(a) Find a formula for $f^{-1}(x)$ and state its domain.

(b) Compute $f^{-1}(5)$, then check by computing $f(11)$.}

\problem[6cm]{Priya claims that the inverse of $f(x) = (x + 4)^2$, restricted
to $x \geq -4$, is given by the rule $x^2 - 4$. Testing her rule at $x = 9$
gives $77$.

(a) Compute the true value of $f^{-1}(9)$, and check it by computing $f(-1)$.

(b) Give the correct formula for $f^{-1}(x)$, state its domain, and say in one
sentence what Priya reversed.}

\problem[7cm]{Let $f(x) = x^2 + 8x + 12$.

(a) Find the largest interval of the form $x \geq c$ on which $f$ is
one-to-one, and give the value of $c$.

(b) On that restricted domain, find a formula for $f^{-1}(x)$ and state its
domain.

(c) Compute $f^{-1}(5)$, then check by computing $f(-1)$.}

\problem[7cm]{Explain, in complete sentences, why a function must be one-to-one
before it can have an inverse function. Use $f(x) = x^2 - 4$ on all real
numbers as your example: compute $f(3)$ and $f(-3)$, explain exactly what goes
wrong when you try to define $f^{-1}(5)$, and describe one domain restriction
that repairs the problem --- including what the inverse's domain and range
become once you have made it.}
