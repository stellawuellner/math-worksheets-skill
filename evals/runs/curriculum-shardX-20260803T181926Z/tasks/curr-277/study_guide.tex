\sstitleblock{Exponential Functions: Evaluating, Graphing, Growth and Decay}

\noindent\small Every function on your worksheet has the same shape,
$f(x) = a \cdot b^{x}$: a starting value $a$ multiplied by a base $b$ once for
each step of $x$. Read $a$ as ``where it starts'' and $b$ as ``what it does
each step''.\normalsize

\skillheading{1. Evaluating an exponential function}
\begin{formulabox}
\textbf{Rule:} In $f(x) = a \cdot b^{x}$, do the \emph{power first}, then
multiply by $a$. \quad $a \cdot b^{x}$ never means $(a \cdot b)^{x}$.\\[3pt]
Two facts that cover the awkward exponents: $b^{0} = 1$ for any nonzero base,
and $b^{-n} = \dfrac{1}{b^{n}}$ --- a negative exponent gives a small positive
answer, never a negative one.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad For $f(x) = 4 \cdot 3^{x}$, find $f(2)$.
\step{The input replaces $x$ everywhere, and the order of operations puts the
power ahead of the multiplication.}
\step{$3^{2} = 9$, and then $4 \cdot 9$ finishes it.}

$f(2) = \ans{36}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad For $g(x) = 5 \cdot 2^{x}$, find $g(3)$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{40}$}
\end{tryitbox}

\skillheading{2. Graphing an exponential function and reading it}
\begin{formulabox}
\textbf{How to graph:} build a short table (use $x = -1, 0, 1, 2, 3$), plot the
points, and join them with a \emph{smooth curve} --- never a straight line.\\[3pt]
\textbf{What the picture always shows:} the curve crosses the $y$-axis at $a$;
it rises to the right when $b > 1$ (growth) and falls to the right when
$0 < b < 1$ (decay); it approaches the $x$-axis but never touches it.\\[3pt]
\textbf{Reading backwards:} to find the $x$ that gives a chosen height, write
both sides as powers of the same base and match the exponents.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad Graph $f(x) = 2^{x}$ and find the $x$ where the curve
reaches height $8$.
\step{The table gives the points to plot, and the same table answers the
backwards question, so build it first.}
\step{$f(0) = 1$, $f(1) = 2$, $f(2) = 4$, $f(3) = 8$: the height $8$ is reached
at the fourth point, and $2^{x} = 8 = 2^{3}$ confirms the exponents match.}

$\ans{x = 3}$
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad On the graph of $f(x) = 3^{x}$, at what $x$ is the height
$81$?\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x = 4}$}
\end{tryitbox}

\skillheading{3. Growth and decay models from a percent}
\begin{formulabox}
\textbf{Rule:} A percent change becomes a multiplier, applied once per period.
Growing by $r$ percent gives base $b = 1 + \frac{r}{100}$; shrinking by $r$
percent gives $b = 1 - \frac{r}{100}$.\\[3pt]
So $6$ percent growth is $b = 1.06$, and $12$ percent decay is $b = 0.88$ ---
the amount \emph{kept}, not the amount lost.
\end{formulabox}
\begin{examplebox}
\textbf{Example:}\quad A car worth $18000$ dollars loses $12$ percent of its
value each year. What is it worth after $3$ years?
\step{Losing a fixed percent each year is repeated multiplication, so this is
exponential decay and the base is the fraction kept.}
\step{Kept each year: $1 - 0.12 = 0.88$, so the value is
$18000 \cdot (0.88)^{3} = 18000 \cdot 0.681472$.}

$\ans{12266.50}$ dollars
\end{examplebox}
\begin{tryitbox}
\textbf{Try it:}\quad A town of $5000$ people grows $4$ percent per year. Find
the population after $6$ years, to the nearest whole person.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{6327}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} $3 \cdot 2^{4}$ is $48$, not $6^{4}$ --- the coefficient
is not part of the base. And a decay percent is not the base: losing $15$
percent means multiplying by $0.85$ every year, not by $0.15$.
\end{watchoutbox}
