\sstitleblock{Area Models for Two-Digit Multiplication}

\skillheading{Skill 1 --- Building the area model}

\begin{formulabox}\textbf{Rule:}\ Split each factor into \emph{tens $+$ ones},
put one factor across the top and the other down the side, and multiply every
row label by every column label. Four regions, four partial products, one sum.
\begin{center}
\begin{tabular}{|r|c|c|}
\hline
& tens & ones \\ \hline
tens & T$\times$T & O$\times$T\\ \hline
ones & T$\times$O & O$\times$O\\ \hline
\end{tabular}
\end{center}
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Multiply $21 \times 13$.
\step{Split by place value --- $20 + 1$ and $10 + 3$ --- because the model needs
one column and one row for each place.}
\step{Regions: $20 \times 10 = 200$, $1 \times 10 = 10$, $20 \times 3 = 60$,
$1 \times 3 = 3$. Adding all four: $\ans{273}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Multiply $34 \times 12$ with an area model.
\ \rotatebox{180}{\footnotesize check: $\ans{408}$}\end{tryitbox}

\skillheading{Skill 2 --- Estimating the product first}

\begin{formulabox}\textbf{Rule:}\ Round \emph{each} factor to the nearest ten,
then multiply the rounded factors. The estimate tells you how big the rectangle
should be, so a wildly different total means a region was missed.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Estimate $27 \times 43$.
\step{Both factors are being rounded, so look at each ones digit on its own ---
that is what decides up or down.}
\step{$27 \to 30$ and $43 \to 40$, so the estimate is
$30 \times 40 = \ans{1200}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Estimate $52 \times 38$ by rounding each
factor to the nearest ten.
\ \rotatebox{180}{\footnotesize check: $\ans{2000}$}\end{tryitbox}

\skillheading{Skill 3 --- Finding a missing partial product}

\begin{formulabox}\textbf{Rule:}\ Every region is (row label) $\times$ (column
label) --- so an empty region can always be rebuilt from its own two labels.
Never total a model until all four regions are filled.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad One region of the model for
$43 \times 25$ is blank: columns $40$ and $3$, rows $20$ and $5$, with $800$,
$200$ and $60$ already written.
\step{The blank sits where row $5$ meets column $3$, so its own labels rebuild
it --- no other region is needed.}
\step{$3 \times 5 = 15$, and then $800 + 200 + 60 + 15 = \ans{1075}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad A model for $62 \times 34$ has columns
$60$ and $2$, rows $30$ and $4$. Fill all four regions and total them.
\ \rotatebox{180}{\footnotesize check: $\ans{2108}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Two regions, not four, is the most
common error: $24 \times 36$ is \emph{not} $20 \times 30$ plus $4 \times 6$.
The two cross regions (tens meeting ones) are usually the biggest part of the
answer.
\end{watchoutbox}
