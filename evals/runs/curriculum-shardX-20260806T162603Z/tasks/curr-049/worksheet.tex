\wstitleblock{Fixing Shares That Are Not Really Halves or Fourths}{Kindergarten--Grade 1}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=1.1cm, itemsep=1pt, topsep=2pt]
  \item \emph{Halves} are two parts of the \textbf{same size}. \emph{Fourths} are four parts of the same size.
  \item Cutting something into two pieces does not always make halves --- the pieces must match.
  \item Every strip on this page is made of equal squares. Count the squares in a part to find its size.
\end{itemize}
