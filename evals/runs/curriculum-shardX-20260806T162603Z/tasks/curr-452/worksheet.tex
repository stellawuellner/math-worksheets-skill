\wstitleblock{Limits at the Edges}{AP Calculus AB/BC}{}

\begin{center}
One-sided limits, holes and vertical asymptotes, and end behavior
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Decide first \emph{which question the notation is asking}: a value approached from one side, behavior at a zero of the denominator, or behavior as $x$ runs off the axis.
  \item Acceptable answers are an exact number, $+\infty$, $-\infty$, or DNE. Write $+\infty$ or $-\infty$ when the values grow without bound in one definite direction --- that is more informative than DNE.
  \item Show the rewriting step (factor, cancel, divide by the highest power, or split the absolute value) before you state the answer.
\end{itemize}

\problem[3.6cm]{Let $f(x) = \dfrac{|x - 3|}{x - 3}$. Split the absolute value into
its two cases and evaluate:
\par\vspace{0.15cm}
\textbf{(a)} $\displaystyle\lim_{x \to 3^-} f(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 3^+} f(x) = $ \ansblank}

\problem[4.4cm]{The function $g(x) = \dfrac{x - 4}{\sqrt{x - 4}}$ is defined only
for $x > 4$, so only one side of $x = 4$ exists. Evaluate
$\displaystyle\lim_{x \to 4^+} g(x)$.}

\problem[5cm]{Evaluate \quad $\displaystyle\lim_{x \to \infty} \frac{3x^2 - 5x + 2}{4x^2 + 7}$}

\problem[3.6cm]{Let $h(x) = \dfrac{x + 3}{x - 5}$. The denominator is zero at
$x = 5$ and the numerator is not, so $x = 5$ is a vertical asymptote. Track the
sign of $x - 5$ on each side and evaluate:
\par\vspace{0.15cm}
\textbf{(a)} $\displaystyle\lim_{x \to 5^-} h(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 5^+} h(x) = $ \ansblank}

\problem[4.4cm]{A shipping cost in dollars for a parcel of weight $x$ kilograms
is modelled by
\[
  f(x) = \begin{cases} 3x - 1, & x < 2 \\[2pt] x^2 + 4, & x \ge 2 \end{cases}
\]
\textbf{(a)} $\displaystyle\lim_{x \to 2^-} f(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 2^+} f(x) = $ \ansblank
\par\vspace{0.15cm}
\textbf{(c)} Explain whether $\displaystyle\lim_{x \to 2} f(x)$ exists, and say
what your two answers above have to do with it.}

\problem[5cm]{Evaluate \quad $\displaystyle\lim_{x \to -\infty} \frac{2x^3 + x}{5 - x^3}$}

\problem[4cm]{Let $k(x) = \dfrac{x^2 - 4}{x^2 - 5x + 6}$. The denominator is zero
at \emph{both} $x = 2$ and $x = 3$, but only one of them is a vertical asymptote.
Factor, cancel, and evaluate:
\par\vspace{0.15cm}
\textbf{(a)} $\displaystyle\lim_{x \to 2} k(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to 3^+} k(x) = $ \ansblank}

\problem[5cm]{Evaluate \quad $\displaystyle\lim_{x \to \infty} \frac{5x^2 - 3}{2x^3 + x}$}

\problem[5.4cm]{Evaluate \quad $\displaystyle\lim_{x \to 2^-} \frac{x^2 - 4}{|x - 2|}$
\par\vspace{0.1cm}
{\small (For $x < 2$ the quantity $x - 2$ is negative, so $|x-2|$ is not $x - 2$.)}}

\problem[4.2cm]{\textbf{Challenge.} Let $r(x) = \dfrac{\sqrt{9x^2 + 4}}{3x - 1}$.
Divide top and bottom by $|x|$ and keep track of the sign of $x$ in each case:
\par\vspace{0.15cm}
\textbf{(a)} $\displaystyle\lim_{x \to \infty} r(x) = $ \ansblank
\qquad
\textbf{(b)} $\displaystyle\lim_{x \to -\infty} r(x) = $ \ansblank}
