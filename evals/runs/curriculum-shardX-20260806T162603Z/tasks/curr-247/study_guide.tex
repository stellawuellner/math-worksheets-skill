\sstitleblock{Drawing and Using Informal Lines of Fit}

\skillheading{Skill 1 --- Drawing the line}

\begin{formulabox}
\textbf{What makes a good line of fit:} one straight ruler line through the middle of the cloud,
following the overall trend, with about as many points above it as below. It need not touch any
point, and it is \textit{not} the line joining the first point to the last.\\[3pt]
\textbf{Anchor tip:} a good line passes close to the middle of the data --- the middle input paired
with the \textit{mean} of the outputs.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find the mean output to anchor a line of fit for these five points.
\begin{center}
\begin{tikzpicture}
\begin{axis}[width=5.4cm, height=3.0cm, axis lines=left, xmin=0, xmax=6, ymin=0, ymax=17,
  tick label style={font=\tiny}, grid=both, grid style={line width=0.15pt, draw=gray!55},
  major grid style={line width=0.3pt, draw=gray!55}, enlargelimits=false]
\addplot[only marks, mark=*, mark size=1.2pt, blue] coordinates {(1,4)(2,6)(3,9)(4,11)(5,15)};
\end{axis}
\end{tikzpicture}
\end{center}
\step{The anchor point sits at the mean of the outputs, so average the five plotted values first.}
\step{$(4 + 6 + 9 + 11 + 15) \div 5 = 45 \div 5$, so the line should pass near height $9$ at the middle input $3$.}
$\Rightarrow\quad \ans{9}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Five points have outputs $12$, $15$, $17$, $22$, $24$. Find the height the
line of fit should pass through.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{18}$}
\end{tryitbox}

\skillheading{Skill 2 --- The slope of the line you drew}

\begin{formulabox}
\textbf{Slope:} \quad $m = \dfrac{y_2 - y_1}{x_2 - x_1}$\\[3pt]
Use two points that lie \textbf{on your line}, not two data points --- the whole purpose of the
line is to smooth out the scatter. Choose two places where the line crosses grid corners, so the
arithmetic stays clean. A falling line gives a negative slope.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line of fit passes through $(2, 7)$ and $(6, 19)$. Find its slope.
\step{Two points on the line are given, so this is the slope formula and nothing else.}
\step{$m = \dfrac{19 - 7}{6 - 2} = \dfrac{12}{4}$}
$\Rightarrow\quad m = \ans{3}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line of fit passes through $(1, 20)$ and $(5, 8)$. Find its slope.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$}
\end{tryitbox}

\skillheading{Skill 3 --- Predicting with the line}

\begin{formulabox}
\textbf{Rule:} \quad $y = mx + b$ --- substitute the input you care about for $x$.\\[3pt]
The prediction is what the \textit{line} says, not what any real point says, so it is an estimate.
Trust it only for inputs inside the range of the data that was collected; far outside that range
there is no evidence the pattern continues.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line of fit is $y = 4x + 7$. Predict the value at $x = 9$.
\step{The equation is already in slope-intercept form, so predicting is one substitution.}
\step{$y = 4(9) + 7 = 36 + 7$}
$\Rightarrow\quad \ans{43}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line of fit is $y = -2x + 50$. Predict the value at $x = 6$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{38}$}
\end{tryitbox}

\skillheading{Skill 4 --- Writing the equation from two points}

\begin{formulabox}
\textbf{Two steps:}\\[3pt]
1. Slope: $m = \dfrac{y_2 - y_1}{x_2 - x_1}$. \quad 2. Intercept: put one of the points back into
$y = mx + b$ and solve, which gives $b = y - mx$.\\[3pt]
If one of the points already sits on the vertical axis, its output \textit{is} $b$ and step 2 is
free.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A line of fit passes through $(2, 11)$ and $(6, 23)$. Write its equation.
\step{Neither point is on the vertical axis, so find the slope first and then work back to the intercept.}
\step{$m = \dfrac{23 - 11}{6 - 2} = 3$, then $b = 11 - 3(2) = 11 - 6$.}
$\Rightarrow\quad m = \ans{3},\ b = \ans{5}, \text{ so } y = 3x + 5$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A line of fit passes through $(1, 17)$ and $(4, 8)$. Find $m$ and $b$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{-3}$ and $\ans{20}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} a line of fit describes an association. It never proves that one quantity
\textit{causes} the other.
\end{watchoutbox}
