\wstitleblock{Where Is It? Shapes on the Toy Shelf}{Grade 1}{}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Every problem uses the toy-shelf picture below. Look before you
    answer!
  \item Position words tell WHERE: top, bottom, above, below, left, right,
    next to, between.
\end{itemize}

\begin{center}
\textbf{The Toy Shelf} --- use this picture for every problem.\\[6pt]
\begin{tikzpicture}[scale=0.75]
% top shelf
\draw[thick] (0,0) -- (9.4,0);
\draw[thick] (0.5,0.55) circle (0.55);
\draw[thick] (2.5,0) -- (4.1,0) -- (3.3,1.2) -- cycle;
\draw[thick] (5,0) rectangle (6.1,1.1);
\draw[thick] (7,0) rectangle (9,0.9);
% bottom shelf
\draw[thick] (0,-2.6) -- (9.4,-2.6);
\draw[thick] (0.05,-2.6) -- (1.05,-2.6) -- (1.4,-1.95) -- (1.05,-1.3) -- (0.05,-1.3) -- (-0.3,-1.95) -- cycle;
\draw[thick] (3.3,-2.25) circle (0.35);
\draw[thick] (5.55,-2.6) -- (6.25,-1.9) -- (5.55,-1.2) -- (4.85,-1.9) -- cycle;
\end{tikzpicture}\\[4pt]
Top shelf, left to right: circle, triangle, square, rectangle.\\
Bottom shelf, left to right: hexagon, small circle, tilted square.
\end{center}

\problem[2.5cm]{(a) How many shapes are on the TOP shelf? \ansblank\par
(b) Which shape is farthest to the LEFT on the top shelf? \ansblank}

\problem[2.5cm]{(a) How many shapes are on the BOTTOM shelf? \ansblank\par
(b) Name the shape directly BELOW the triangle: \ansblank}

\problem[2.5cm]{Look at the top shelf.\par
(a) How many shapes sit BETWEEN the circle and the rectangle? \ansblank\par
(b) Name the shape right NEXT TO the rectangle: \ansblank}

\problem[2.5cm]{The top shelf holds 4 shapes and the bottom shelf holds 3
shapes. Write $>$, $<$, or $=$ in the blank:\par
\begin{center} 4 \ansblank\ 3 \end{center}}

\problem[3cm]{Look at the small circle on the bottom shelf.\par
(a) How many shapes are NEXT TO it? \ansblank\par
(b) Name the shape to the RIGHT of the small circle: \ansblank}

\problem[3.5cm]{Imagine the square moves from the top shelf down to the
bottom shelf.\par
(a) How many shapes would the bottom shelf hold then? \ansblank\par
(b) How many shapes would stay on the top shelf? \ansblank}

\problem[3cm]{Look at the hexagon and the circle above it. Finish the
sentence with a position word:\par
\begin{center}The hexagon is \ansblank\ the circle.\end{center}}

\problem[4cm]{Challenge!\par
(a) How many shapes are on the toy shelf in all? \ansblank\par
(b) A friend cannot see the shelf. Tell your friend where the TILTED SQUARE
is. Use two position words.}
