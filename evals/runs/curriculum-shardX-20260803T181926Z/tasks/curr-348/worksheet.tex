\wstitleblock{Dilations: Building and Measuring Similar Figures}{High School Geometry}{}

\noindent\small
Dilating points on the coordinate plane, finding a scale factor from two
lengths, and using a scale factor on a similar figure. Units are stated in
every problem.
\begin{itemize}[leftmargin=0.7cm, itemsep=1pt, topsep=3pt]
  \item A dilation with centre $C$ and scale factor $k$ sends each point $P$ to
        the point $P'$ on ray $CP$ with $CP' = k \cdot CP$. Centred at the
        origin, $(x, y) \mapsto (kx, ky)$; centred at $(h, j)$,
        $(x, y) \mapsto (h + k(x-h),\ j + k(y-j))$.
  \item A dilation image is \emph{similar} to its pre-image: every length is
        multiplied by $k$, every angle is unchanged, and area is multiplied by
        $k^2$.
  \item Give lengths in the units of the problem and show the computation that
        produced each one.
\end{itemize}
\normalsize
\vspace{0.2cm}

\problem[5cm]{A slide projector maps the slide onto a screen by a dilation
centred at the origin with scale factor $3$; coordinates on both the slide and
the screen are measured in centimetres. The point $A(2, 1)$ is printed on the
slide. Give the coordinates of its image $A'$ on the screen.}

\problem[6cm]{A triangular sail is cut from a pattern with vertices $A(0,0)$,
$B(4,0)$ and $C(4,3)$, measured in metres. The full-size sail is the image of
that pattern under a dilation centred at $A(0,0)$ with scale factor $2.5$.
Find the length of the full-size sail's side $B'C'$, then the length of its
side $A'C'$.}

\problem[5cm]{On a blueprint a rectangular patio is drawn $6$ centimetres long
and $4$ centimetres wide. A print shop enlarges the blueprint by a dilation so
that the long side of the drawing measures $21$ centimetres. Find the scale
factor of the enlargement, then find the width of the enlarged drawing.}

\problem[6cm]{A digital map zooms about the pin at $P(1, 2)$ with scale factor
$3$; coordinates are in kilometres east and north of the town hall. A trailhead
sits at $Q(3, 5)$. Give the coordinates of $Q'$, the trailhead's position on
the zoomed map.}

\problem[6cm]{A photograph measures $10$ centimetres by $15$ centimetres. A
print shop enlarges it by a dilation with scale factor $2.4$. \textbf{(a)} Find
the two side lengths of the enlargement. \textbf{(b)} The enlargement's area is
how many times the original's area?}

\problem[6cm]{A display model of a car is built as a dilation of the real car
with scale factor $\frac{1}{12}$. The real car is $4.8$ metres long, and the
glass in its windshield has area $0.72$ square metres. Ben computes the model's
windshield glass area as $0.06$ square metres. Explain what Ben did wrong, then
give the correct area of the model's windshield glass in square metres, and the
model's length in metres.}

\problem{Triangle $JKL$ has vertices $J(1, 1)$, $K(3, 1)$ and $L(3, 4)$, with
coordinates in units. \textbf{(a)} Plot $JKL$ and its image $J'K'L'$ under the
dilation centred at the origin with scale factor $2$. \textbf{(b)} Find the
length $K'L'$. \textbf{(c)} Draw the line through $J$ and $J'$, the line
through $K$ and $K'$, and the line through $L$ and $L'$. Describe what those
three lines do, and explain why triangle $J'K'L'$ must be similar to triangle
