\problem{\textbf{$\frac{dy}{dx}$ at $t=2$ for $x = t^2+1$, $y = t^3-3t$.} The
parametric slope is the ratio of the two $t$-derivatives, not a derivative of
$y$ alone.
\[ \frac{dy}{dx} = \frac{dy/dt}{dx/dt} = \frac{3t^2-3}{2t},
   \qquad t = 2:\quad \frac{12-3}{4} \]
$\ans{\frac{dy}{dx}\Big|_{t=2} = \tfrac{9}{4}}$}

\problem{\textbf{Speed at $t=2$ for $\mathbf{r}(t) = \langle t^3-3t,\ t^2\rangle$.}
Speed is the magnitude of the velocity vector.
\[ \mathbf{v}(t) = \langle 3t^2-3,\ 2t\rangle, \qquad
   \mathbf{v}(2) = \langle 9,\ 4\rangle \]
\[ |\mathbf{v}(2)| = \sqrt{9^2+4^2} = \sqrt{97} = 9.8488\ldots \]
$\ans{\sqrt{97} \approx 9.849}$
\par\smallskip
\emph{Watch the vocabulary:} velocity is the vector, speed is its magnitude, and
$\frac{dy}{dx} = \frac{4}{9}$ at this instant is a third, different quantity.}

\problem{\textbf{Total distance for $\mathbf{r}(t) = \langle t^2, \frac{2}{3}t^3\rangle$
on $[0,2]$.} Distance travelled is the integral of speed --- always the
integral of $|\mathbf{v}|$, never of a component.
\[ \mathbf{v}(t) = \langle 2t,\ 2t^2 \rangle, \qquad
   |\mathbf{v}(t)| = \sqrt{4t^2+4t^4} = 2t\sqrt{1+t^2}\ \ (t \ge 0) \]
\[ \int_0^2 2t\sqrt{1+t^2}\,dt
   = \left[\tfrac{2}{3}\left(1+t^2\right)^{3/2}\right]_0^2
   = \tfrac{2}{3}\left(5\sqrt{5}-1\right) = 6.7869\ldots \]
$\ans{\tfrac{2}{3}(5\sqrt{5}-1) \approx 6.787}$
\par\smallskip
The substitution $u = 1+t^2$, $du = 2t\,dt$ makes the antiderivative immediate ---
worth spotting, since the same integrand returns in the study guide's try-it.}

\problem{\textbf{Intersections of $r = 2+2\cos\theta$ and $r = 3$.} Set the two
radius expressions equal, since a shared point needs the same $r$ at the same
$\theta$.
\[ 2 + 2\cos\theta = 3 \;\Longrightarrow\; \cos\theta = \tfrac{1}{2}
   \;\Longrightarrow\; \theta = \tfrac{\pi}{3},\ \tfrac{5\pi}{3}
   \ \text{on } [0, 2\pi) \]
$\ans{\theta = \tfrac{\pi}{3},\ \theta = \tfrac{5\pi}{3}}$
\par\smallskip
\emph{Caution worth teaching:} solving $r_1 = r_2$ finds intersections that
occur at the \emph{same} $\theta$. Two polar curves can also cross at a point
reached at different angles (the pole is the usual culprit), so a sketch is part
of a complete answer. Here $r = 0$ on the cardioid at $\theta = \pi$, where the
circle has $r = 3$, so the pole is not a shared point.}

\problem{\textbf{Horizontal tangent for $x = t^3-3t$, $y = t^2-4t$.} A tangent
is horizontal where $\frac{dy}{dt} = 0$ \emph{and} $\frac{dx}{dt} \neq 0$; if
both vanish the point needs separate analysis.
\[ \frac{dy}{dt} = 2t - 4 = 0 \;\Longrightarrow\; t = 2, \qquad
   \frac{dx}{dt}\Big|_{t=2} = 3(4) - 3 = 9 \neq 0 \]
$\ans{t = 2}$
\par\smallskip
The point itself is $(x, y) = (2, -4)$, and $\frac{dy}{dx} = \frac{0}{9} = 0$
there, confirming the horizontal tangent.}

\problem{\textbf{One petal of $r = 3\sin(2\theta)$.} Polar area is
$\frac{1}{2}\int r^2\,d\theta$, taken over the interval that traces the region
exactly once --- here $0$ to $\frac{\pi}{2}$.
\[ A = \frac{1}{2}\int_0^{\pi/2} 9\sin^2(2\theta)\,d\theta
     = \frac{9}{2}\int_0^{\pi/2} \frac{1-\cos(4\theta)}{2}\,d\theta
     = \frac{9}{4}\left[\theta - \frac{\sin(4\theta)}{4}\right]_0^{\pi/2} \]
$\ans{\tfrac{9\pi}{8} \approx 3.534}$
\par\smallskip
\emph{Grading note:} integrating from $0$ to $2\pi$ traces all four petals and
gives four times too much. The half-angle identity is what makes $\sin^2$
integrable by hand.}

\problem{\textbf{$v(t)$ for $x(t) = t^2e^{-t}$.} Product rule, then factor ---
the factored form is what makes the sign analysis possible later.
\[ v(t) = 2t\,e^{-t} + t^2\left(-e^{-t}\right) = \left(2t - t^2\right)e^{-t} \]
$\ans{v(t) = (2t - t^2)e^{-t}}$
\par\smallskip
Since $e^{-t} > 0$ always, the sign of $v$ is the sign of $t(2-t)$: the particle
moves right on $0 < t < 2$ and left after $t = 2$.}

\problem{\textbf{Inside $r = 3$, outside $r = 2+2\cos\theta$.} Between
$\theta = \frac{\pi}{3}$ and $\theta = \frac{5\pi}{3}$ the circle is the outer
curve, so subtract the squares in that order:
\[ A = \frac{1}{2}\int_{\pi/3}^{5\pi/3}
      \left[3^2 - \left(2+2\cos\theta\right)^2\right] d\theta
    = \frac{1}{2}\int_{\pi/3}^{5\pi/3}
      \left(5 - 8\cos\theta - 4\cos^2\theta\right) d\theta \]
Using $4\cos^2\theta = 2 + 2\cos(2\theta)$, an antiderivative is
$3\theta - 8\sin\theta - \sin(2\theta)$, and evaluating at the limits gives
$2\pi + \frac{9\sqrt{3}}{2} = 14.0774\ldots$
$\ans{2\pi + \tfrac{9\sqrt{3}}{2} \approx 14.077}$
\par\smallskip
\emph{Grading note:} the limits must be the interval on which the circle is
outside the cardioid. Running from $-\frac{\pi}{3}$ to $\frac{\pi}{3}$ instead
computes the region where the cardioid is outside, and returns a negative
integrand.}

\problem{\textbf{Arc length of $x=t^2$, $y=t^3$ on $[0,1]$.} Arc length of a
parametric curve is the same integral as distance travelled --- the speed
integral.
\[ L = \int_0^1 \sqrt{(2t)^2 + (3t^2)^2}\,dt = \int_0^1 t\sqrt{4+9t^2}\,dt
    = \left[\frac{(4+9t^2)^{3/2}}{27}\right]_0^1
    = \frac{13\sqrt{13}-8}{27} \]
$\ans{\tfrac{13\sqrt{13}-8}{27} \approx 1.440}$
\par\smallskip
Pulling $t$ out of $\sqrt{4t^2+9t^4}$ is legal here because $t \ge 0$ on the
interval; on an interval containing negatives it would need $|t|$.}

\problem{\textbf{Manual review --- expected reasoning}
($\mathbf{r}(t) = \langle t^3-3t,\ t^2\rangle$, $0 \le t \le 3$).
\begin{enumerate}[leftmargin=1.2cm, itemsep=3pt, topsep=3pt]
  \item \textbf{Distance vs.\ displacement.} Displacement records only where the
  particle ends up relative to where it started; total distance adds up every
  bit of path, including motion that doubles back. Here $x'(t) = 3t^2-3$ is
  negative on $0 < t < 1$, so the particle does reverse horizontally, and the
  two quantities must differ. Correct set-ups:
  \[ \text{distance} = \int_0^3 \sqrt{(3t^2-3)^2 + (2t)^2}\,dt, \qquad
     \text{displacement} = \left|\mathbf{r}(3) - \mathbf{r}(0)\right|
     = \sqrt{18^2 + 9^2} \]
  Accept the displacement written as a vector $\langle 18, 9\rangle$ or as its
  magnitude. Do \emph{not} accept $\int_0^3 \mathbf{v}(t)\,dt$ as the distance.
  \item \textbf{Speeding up at $t=1$?} Compare the directions of velocity and
  acceleration: $\mathbf{v}(t) = \langle 3t^2-3,\ 2t\rangle$ and
  $\mathbf{a}(t) = \langle 6t,\ 2\rangle$, so
  $\mathbf{v}(1) = \langle 0, 2\rangle$ and $\mathbf{a}(1) = \langle 6, 2\rangle$.
  Then $\mathbf{v}(1)\cdot\mathbf{a}(1) = 0(6) + 2(2) = 4 > 0$, so the speed is
  increasing: the particle is \textbf{speeding up} at $t=1$.
  Equivalent accepted argument: $|\mathbf{v}(t)| = \sqrt{(3t^2-3)^2+4t^2}$ has a
  positive derivative at $t=1$. Do not accept ``acceleration is positive, so it
  speeds up'' with no reference to direction --- that is the standard error this
  problem targets.
\end{enumerate}}
