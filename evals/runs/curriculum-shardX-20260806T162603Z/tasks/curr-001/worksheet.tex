\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=2pt]
  \item Read the starting number out loud.
  \item Count forward by ones and write each missing number.
  \item The number tracks and chart rows are there to help you see the counting order.
\end{itemize}
