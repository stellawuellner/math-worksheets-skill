\wstitleblock{Comparing Groups of Objects: More, Fewer, and the Same}{Kindergarten--Grade 1}{}

\noindent\textbf{How to work each problem:}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Touch each object as you count it. Say the last number you said --- that is how many.
  \item Match the two groups one to one. Draw a line from one object to one object.
  \item If a group runs out first, it has \textbf{fewer}. The group with leftovers has \textbf{more}.
  \item If nothing is left over in either group, the groups are \textbf{the same}.
  \item The symbol always opens its mouth toward the bigger number:
        $>$ is ``is more than'', $<$ is ``is fewer than'', $=$ is ``is the same as''.
\end{itemize}
\vspace{0.15cm}
\noindent\rule{\linewidth}{0.4pt}

\problem[3cm]{Count each row. Circle the row that has \textbf{more}.
\begin{center}\Large
Amy: \ $\star\ \star\ \star\ \star\ \star$\\[0.4cm]
Ben: \ $\bullet\ \bullet\ \bullet$
\end{center}
Amy has \textbf{5}. Ben has \textbf{3}. Write $>$, $<$, or $=$ in the blank:
\quad $5\ $~\ansblank$\ 3$}

\problem[3cm]{Count each row. Draw a line to match one apple with one basket.
\begin{center}\Large
Apples: \ $\bullet\ \bullet\ \bullet\ \bullet$\\[0.4cm]
Baskets: \ $\square\ \square\ \square\ \square$
\end{center}
Nothing is left over, so the groups are \textbf{the same}. Write $>$, $<$, or $=$:
\quad $4\ $~\ansblank$\ 4$}

\problem[3.5cm]{A pond has \textbf{7} frogs. The pond has \textbf{4} lily pads.
\begin{center}\Large
Frogs: \ $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[0.4cm]
Pads: \ $\square\ \square\ \square\ \square$
\end{center}
Match one frog to one lily pad. Then cross out every frog that has a pad.
\textbf{How many more} frogs than lily pads are there? Write the subtraction
sentence that shows it.}

\problem[3cm]{Count each row. Circle the row that has \textbf{fewer}.
\begin{center}\Large
Red cubes: \ $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[0.4cm]
Blue cubes: \ $\square\ \square\ \square\ \square\ \square\ \square\ \square\ \square$
\end{center}
There are \textbf{6} red cubes and \textbf{8} blue cubes. Write $>$, $<$, or $=$:
\quad $6\ $~\ansblank$\ 8$}

\problem[4cm]{\textbf{Story.} Nina picks \textbf{9} flowers. Omar picks \textbf{6} flowers.

\noindent Draw Nina's flowers as dots in one row and Omar's flowers as dots in a
row right underneath, so each of Omar's dots is under one of Nina's dots.

\noindent \textbf{How many fewer} flowers did Omar pick? Write the subtraction
sentence that shows it.}

\problem[4cm]{\textbf{Put three groups in order.} Three cups hold marbles.
\begin{center}\Large
Cup A: \ $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[0.35cm]
Cup B: \ $\bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet\ \bullet$\\[0.35cm]
Cup C: \ $\bullet\ \bullet\ \bullet\ \bullet$
\end{center}
Write how many marbles are in each cup. Then write the three numbers in order
from \textbf{fewest} to \textbf{most}.}

\problem[4cm]{\textbf{Make them the same.} Sam has \textbf{5} shells. Kim has
\textbf{9} shells.

\noindent Draw Sam's shells as dots. Under them, draw Kim's shells as dots,
one under each of Sam's.

\noindent How many \textbf{more} shells does Sam need so that both children have
the same number? \quad $5 + $~\ansblank$\ = 9$}

\problem[4.5cm]{\textbf{Challenge.} Two friends want the same number of stickers.
Lee has \textbf{4} stickers. Jo has \textbf{9} stickers.

\noindent Draw the two rows and match them one to one. How many stickers must Lee
get so that the two groups are the same? Write the sentence
$4 + \underline{\hspace{1.2cm}} = 9$ with your number in the blank, then explain
in one sentence how the picture shows your answer.}
