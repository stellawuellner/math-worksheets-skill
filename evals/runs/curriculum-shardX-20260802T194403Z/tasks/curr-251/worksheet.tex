\wstitleblock{Terms, Factors, Coefficients, and Degree}{Algebra 1}{}

\begin{center}\small\itshape
Reading area models as terms, naming factors, tracking coefficients and
degree, and testing equivalent forms
\end{center}

\noindent\textbf{The four words, in one sentence each.}
A \textbf{term} is a piece of a sum. A \textbf{factor} is a piece of a
product. A \textbf{coefficient} is the number multiplying a power of the
variable in a term. The \textbf{degree} of a term is its exponent on the
variable, and the degree of a polynomial is the largest degree among its
terms.

\smallskip
\noindent\small In an area model, each \emph{cell} is one term of the product
and each \emph{side length} is one factor. Fill in every cell, then combine
like cells.\normalsize

\problem[4cm]{A rectangle has width $x$ and length $x + 5$. Fill in the model,
then write the area as a sum of terms.
\begin{center}
\begin{tabular}{|c|c|c|}\hline
$\times$ & $x$ & $5$ \\\hline
$x$ & \gcell & \gcell \\\hline
\end{tabular}
\end{center}
How many terms does your answer have, and what is the coefficient of $x$?}

\problem[4.5cm]{Fill in the model for $(x + 3)(x + 5)$, then write the product
as a polynomial in standard form.
\begin{center}
\begin{tabular}{|c|c|c|}\hline
$\times$ & $x$ & $5$ \\\hline
$x$ & \gcell & \gcell \\\hline
$3$ & \gcell & \gcell \\\hline
\end{tabular}
\end{center}
The model has four cells but your answer has fewer terms. Explain why in one
sentence.}

\problem[5cm]{The table lists the terms of one polynomial, out of order. Fill
in the coefficient and the degree of each term, then write the polynomial in
standard form (highest degree first).
\begin{center}
\begin{tabular}{|c|c|c|}\hline
\textbf{Term} & \textbf{Coefficient} & \textbf{Degree} \\\hline
$-9$ & \tcell & \tcell \\\hline
$6x$ & \tcell & \tcell \\\hline
$4x^3$ & \tcell & \tcell \\\hline
$-x^2$ & \tcell & \tcell \\\hline
\end{tabular}
\end{center}
What is the degree of the whole polynomial, and what is its leading
coefficient?}

\problem[5cm]{Write $x^2 + 7x + 12$ as a product of two factors. Then state
the degree of each factor and the degree of the product.}

\problem[5cm]{Jo says $3(2x + 5)$ is a sum of two terms. Ann says it is a
single term made of two factors. Who is describing the expression as written?
Rewrite it in the other form, and name the factors and the terms of each
form.}

\problem[5cm]{Add: $(3x^2 + 5x - 1) + (-3x^2 + 2x + 4)$. Give the sum, then
state its degree. Each polynomial you added has degree $2$ --- explain what
happened to the degree.}

\problem[5.5cm]{Fill in the model for $(2x - 3)(x + 4)$, then write the
product in standard form. Circle the cell that produced the constant term.
\begin{center}
\begin{tabular}{|c|c|c|}\hline
$\times$ & $x$ & $4$ \\\hline
$2x$ & \gcell & \gcell \\\hline
$-3$ & \gcell & \gcell \\\hline
\end{tabular}
\end{center}
What is the coefficient of $x$ in your answer?}

\problem[5.5cm]{Factor $6x^3 - 15x^2$ completely by pulling out the greatest
common factor. List every factor of your answer, and give the degree of each.}

\problem[6cm]{Multiply $(2x^3 - x)(x^2 + 4)$ and write the result in standard
form. Then state the degree and the leading coefficient of the product, and
explain how you could have predicted the degree from the two factors.}

\problem[6cm]{Maya says: ``$x^2 + 8x + 12$ and $(x + 2)(x + 6)$ cannot be the
same expression --- one is a sum of three terms and the other is a product of
two factors.'' Decide whether the two expressions are equivalent and show the
algebra. Then explain, in a sentence or two, how a single expression can be
both a product of two factors and a sum of three terms.}
