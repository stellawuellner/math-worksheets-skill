\problem{\textbf{Silo = cylinder $+$ hemisphere.} The two pieces share the circle where they
meet, but for \emph{volume} nothing is lost --- volumes simply add.
\begin{align*}
V_{\text{cyl}} &= \pi r^2 h = \pi (3)^2 (10) = 90\pi\\
V_{\text{hemi}} &= \tfrac{2}{3}\pi r^3 = \tfrac{2}{3}\pi (3)^3 = 18\pi\\
V &= 90\pi + 18\pi = 108\pi = 339.2920\ldots
\end{align*}
\ans{V \approx 339.29\ \text{m}^3}}

\problem{\textbf{Surface area with a hole.} Start from the closed prism, then correct for the
hole: the drilled circles are removed from the top and bottom faces, and the cylindrical wall
inside the hole is added.
\begin{align*}
\text{top}+\text{bottom} &= 2(12 \times 8) = 192 \quad\text{minus two circles } 2\pi(2)^2 = 8\pi\\
\text{four sides} &= 2(12 + 8)(5) = 200\\
\text{inside wall} &= 2\pi r h = 2\pi (2)(5) = 20\pi\\
S &= 192 - 8\pi + 200 + 20\pi = 392 + 12\pi = 429.6991\ldots
\end{align*}
\ans{S \approx 429.70\ \text{cm}^2}

\noindent{\small Note the net effect of the hole: $+12\pi$. Drilling a hole through a thick
plate \emph{increases} surface area whenever the wall added is bigger than the two circles
removed.}}

\problem{\textbf{Density.} Mass $=$ density $\times$ volume, so find the volume first.
\[ V = \pi r^2 h = \pi (2)^2 (15) = 60\pi \ \text{cm}^3, \qquad
   m = 7.8 \times 60\pi = 468\pi = 1470.2653\ldots \]
\ans{m \approx 1470.27\ \text{g}}}

\problem{\textbf{Reverse design.} The volume formula is known and the unknown is a dimension,
so write the equation and solve it rather than computing forward.
\[ V = Bh \;\Longrightarrow\; (12)(5)h = 900 \;\Longrightarrow\; 60h = 900 \;\Longrightarrow\; h = \frac{900}{60} \]
\ans{h = 15\ \text{cm}}}

\problem{\textbf{Cone $+$ hemisphere.} Add the two volumes; the shared circle again costs
nothing for volume.
\begin{align*}
V_{\text{cone}} &= \tfrac{1}{3}\pi r^2 h = \tfrac{1}{3}\pi (4)^2 (9) = 48\pi\\
V_{\text{hemi}} &= \tfrac{2}{3}\pi r^3 = \tfrac{2}{3}\pi (4)^3 = \tfrac{128}{3}\pi\\
V &= 48\pi + \tfrac{128}{3}\pi = \tfrac{272}{3}\pi = 284.8377\ldots
\end{align*}
\ans{V \approx 284.84\ \text{cm}^3}}

\problem{\textbf{Sphere, then density.}
\[ V = \tfrac{4}{3}\pi r^3 = \tfrac{4}{3}\pi (3)^3 = 36\pi \ \text{cm}^3, \qquad
   m = 8.5 \times 36\pi = 306\pi = 961.3273\ldots \]
\ans{m \approx 961.33\ \text{g}}}

\problem{\textbf{Composite surface area --- count faces, don't add whole surfaces.} The
painted skin is the cone's slant surface, the cylinder's side, and the bottom disc. The
circle where the cone meets the cylinder is interior and is painted on neither piece.
\begin{align*}
\text{cone slant} &= \pi r \ell = \pi (3)(5) = 15\pi\\
\text{cylinder side} &= 2\pi r h = 2\pi (3)(8) = 48\pi\\
\text{bottom disc} &= \pi r^2 = \pi (3)^2 = 9\pi\\
S &= 15\pi + 48\pi + 9\pi = 72\pi = 226.1946\ldots
\end{align*}
\ans{S \approx 226.19\ \text{cm}^2}

\noindent{\small Adding the cone's \emph{total} surface ($\pi r \ell + \pi r^2$) would
double-count the hidden circle and overstate the answer by $9\pi \approx 28.27$.}}

\problem{\textbf{Reverse density design.} Mass and density are given, so work backwards to
volume, then backwards again to the radius.
\[ V = \frac{m}{\rho} = \frac{5000}{8.8} = 568.18\ldots \ \text{cm}^3, \qquad
   \tfrac{4}{3}\pi r^3 = \frac{5000}{8.8} \]
\[ r = \sqrt[3]{\frac{3 \times 5000}{4\pi (8.8)}} = \sqrt[3]{135.66\ldots} = 5.1380\ldots \]
\ans{r \approx 5.14\ \text{cm}}

\noindent{\small Sense check: a $5.14$ cm bronze sphere is about the size of a large orange
and weighs $5$ kg --- bronze is dense.}}

\problem{\textbf{Composite volume, then density.} Volume first, mass second.
\begin{align*}
V_{\text{cyl}} &= \pi (1.5)^2 (4) = 9\pi\\
V_{\text{cone}} &= \tfrac{1}{3}\pi (1.5)^2 (2) = 1.5\pi\\
V &= 10.5\pi = 32.9867\ldots \ \text{m}^3\\
m &= 750 \times 10.5\pi = 7875\pi = 24740.0421\ldots
\end{align*}
\ans{m \approx 24740.04\ \text{kg}}

\noindent{\small The cone holds only one sixth of the cylinder's volume even though it is
half its height --- a cone is one third of its enclosing cylinder.}}

\problem{\textbf{(a) Solve the design equation for $h$.} Total volume $=$ cylinder $+$
hemisphere, with $r = 0.25$ known and $h$ unknown:
\[ \pi r^2 h + \tfrac{2}{3}\pi r^3 = 0.30
   \;\Longrightarrow\; h = \frac{0.30 - \tfrac{2}{3}\pi (0.25)^3}{\pi (0.25)^2}
   = \frac{0.30 - 0.032725}{0.196350} = 1.3612\ldots \]
so the cylinder is about $1.36$ metres tall. \ans{h \approx 1.36}

\textbf{(b) Mass.} The volume is fixed by the design, so the mass follows directly:
$m = 2400 \times 0.30$, i.e. $720$ kilograms. \ans{720}

\textbf{(c) Justification --- open response, read the student's reasoning.} A complete answer
makes both moves:
\emph{(i) The masses must be identical.} Mass $=$ density $\times$ volume, and both bollards
are specified to hold exactly $0.30$ m$^3$ of the same concrete, so both weigh $720$ kg no
matter how the volume is shaped.
\emph{(ii) A practical reason for Option B.} Any of these earns credit: the wider bollard is
much shorter (its cylinder height works out near $0.86$ m against $1.36$ m), so it has a lower
centre of mass and is harder to knock over; it needs a shallower excavation; it presents a
larger, more visible target to a vehicle. Accept any reason that follows from the geometry,
not from the mass.

\noindent{\small Common error to watch for: students who say Option B is heavier ``because it
is wider'' have not used the fixed-volume condition --- width buys height back, not mass.}}
