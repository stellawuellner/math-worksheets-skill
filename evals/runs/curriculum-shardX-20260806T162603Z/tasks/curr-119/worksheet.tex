\wstitleblock{When a Number Rule Stops Working}{Grades 4--5 Mathematics}{}

\noindent\textbf{Directions}
\begin{itemize}[leftmargin=0.9cm, itemsep=1pt, topsep=2pt]
  \item Each problem shows someone who looked at a few numbers and decided on a rule.
  \item A rule that says \emph{every} has to survive \emph{every} case. One number that breaks it is enough.
  \item Write the number the question asks for, then answer the explanation part in full sentences.
\end{itemize}

\vspace{0.25cm}

\problem[3cm]{Ravi is writing a pattern that starts at 4 and adds 6 each time. He wrote:
\[ 4, \quad 10, \quad 16, \quad 24, \quad 30 \]
\begin{itemize}
  \item[(a)] Write the correct 4th term. \ansline
  \item[(b)] Write the correct 5th term. \ansline
  \item[(c)] Explain what Ravi did wrong.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[4cm]{Mia says: ``Every whole number that ends in the digit 7 is prime.'' Find the smallest number greater than 7 that ends in 7 and is \emph{not} prime, and write it as a product of two whole numbers, neither of them 1.}

\problem[3cm]{Ben writes a pattern that starts at 3 and adds 4 each time. He checks the first three terms, sees that each one is prime, and says: ``Every number in my pattern is prime.''
\begin{itemize}
  \item[(a)] Write the first term of Ben's pattern that is \emph{not} prime. \ansline
  \item[(b)] Explain why checking three terms was not enough.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3cm]{Omar says: ``Every multiple of 3 is odd. Look at 3, 9, 15 and 21.''
\begin{itemize}
  \item[(a)] Write the smallest multiple of 3 that is greater than 3 and is even. \ansline
  \item[(b)] Explain why Omar's examples led him to a rule that is false.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3cm]{A pattern begins $1, \ 4, \ 9, \ 16, \ \ldots$ Toby sees that the first step is $+3$ and decides the rule is ``add 3 every time''. He writes 13 as the 5th term.
\begin{itemize}
  \item[(a)] Write the correct 5th term. \ansline
  \item[(b)] Explain why Toby's rule does not fit this pattern.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3cm]{Priya says: ``A pattern that starts with a bigger number always has bigger terms later on.'' Pattern A starts at 5 and adds 3 each time. Pattern B starts at 2 and adds 6 each time.
\begin{itemize}
  \item[(a)] Write the 10th term of pattern A. \ansline
  \item[(b)] Write the 10th term of pattern B. \ansline
  \item[(c)] Write $<$ or $>$ between your two answers. \ansline
  \item[(d)] Explain why Priya's rule is wrong.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3cm]{A pattern starts at 1 and each term is 3 times the term before it. Dana says: ``Every term is odd, so the rule must be add an even number each time.''
\begin{itemize}
  \item[(a)] Write the 6th term of the pattern. \ansline
  \item[(b)] Explain why Dana's rule cannot be the rule for this pattern.
\end{itemize}
\underline{\hspace{\linewidth}}\\[0.6cm]
\underline{\hspace{\linewidth}}}

\problem[3cm]{Ines says: ``A number divisible by 2 and by 3 is always divisible by 6. So a number divisible by 2 and by 4 must always be divisible by 8.''
\begin{itemize}
  \item[(a)] Write the smallest whole number greater than 8 that is divisible by both 2 and 4 but is \emph{not} divisible by 8. \ansline
  \item[(b)] Explain why the first rule works but the second one does not.
\end{itemize}
