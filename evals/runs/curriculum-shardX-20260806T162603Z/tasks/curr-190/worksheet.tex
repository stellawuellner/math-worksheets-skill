\wstitleblock{Proportional Relationships: Four Representations}{Grades 6--7 Mathematics}{}

\begin{center}
\large Moving between graphs, tables, equations, and real-world contexts
\end{center}

\noindent\textbf{Directions:}
\begin{itemize}[leftmargin=1.2cm, itemsep=1pt]
  \item Every proportional relationship can appear four ways: a \textbf{table},
        a \textbf{graph} (straight line through the origin), an
        \textbf{equation} $y = kx$, and a \textbf{story}. Each problem asks you
        to move from one to another.
  \item After the warm-up the representations mix, so read each problem
        carefully before choosing a method. The last problem combines them all.
\end{itemize}
