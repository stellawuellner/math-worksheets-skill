\sstitleblock{Modeling Loci and Intersections with Conics}

\skillheading{Skill 1 --- A circle from a distance condition}

\begin{formulabox}
\textbf{Locus definition.} ``Every point at distance $r$ from $(h, k)$'' is the circle
\[ (x - h)^{2} + (y - k)^{2} = r^{2} \]
Expanding gives $x^{2} + y^{2} + Dx + Ey + F = 0$; completing the square takes you back. Square distances before you simplify --- radicals almost never need to appear.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A point stays $3$ units from $(2, -5)$. Write its equation in expanded form.
\step{The condition is a fixed distance from a fixed point, which is the definition of a circle, so start from the squared distance formula rather than plotting points.}
\step{$(x-2)^{2} + (y+5)^{2} = 9$ expands to $x^{2} - 4x + 4 + y^{2} + 10y + 25 = 9$.}

$\Rightarrow\quad \ans{x^{2} + y^{2} - 4x + 10y + 20 = 0}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A point stays $6$ units from $(-1, 4)$. Write its equation in expanded form.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{x^{2} + y^{2} + 2x - 8y - 19 = 0}$}
\end{tryitbox}

\skillheading{Skill 2 --- A parabola from focus and directrix}

\begin{formulabox}
\textbf{Locus definition.} A parabola is every point equally far from the focus and from the directrix.\par
With the vertex at the origin: $x^{2} = 4py$ opens up or down with focus $(0, p)$ and directrix $y = -p$; $y^{2} = 4px$ opens sideways with focus $(p, 0)$ and directrix $x = -p$. Here $p$ is the vertex-to-focus distance.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad A parabola has vertex $(0,0)$ and focus $(0, 2)$. Find $4p$.
\step{The focus is directly above the vertex, so the axis is vertical and $p$ is just the vertical gap from vertex to focus.}
\step{$p = 2$, so $4p = 8$ and the equation is $x^{2} = 8y$, with directrix $y = -2$.}

$\Rightarrow\quad \ans{4p = 8}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad A parabola has vertex $(0,0)$ and focus $(0, 7)$. Find $4p$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{4p = 28}$}
\end{tryitbox}

\begin{watchoutbox}
\textbf{(!) Watch out:} the directrix sits on the \emph{opposite} side of the vertex from the focus, the same distance away. Putting it on the same side gives a parabola opening the wrong way.
\end{watchoutbox}

\skillheading{Skill 3 --- Where two conics meet}

\begin{formulabox}
\textbf{Method.} Substitute one equation into the other to get a single quadratic, solve it, then feed \emph{every} root back to recover the partner coordinate.\par
The number of solutions of that quadratic is the number of crossings: two roots means two points, a repeated root means a tangent, no real root means the curves miss.
\end{formulabox}

\begin{examplebox}
\textbf{Example:} \quad Find where $y = x^{2} + 1$ meets $y = 3x - 1$.
\step{Both equations already give $y$, so setting the right-hand sides equal removes $y$ in one step.}
\step{$x^{2} + 1 = 3x - 1$, so $x^{2} - 3x + 2 = 0$ and $(x-1)(x-2) = 0$; then $y = 3x - 1$ gives $2$ and $5$.}

$\Rightarrow\quad \ans{(1,\ 2)}$ \ and \ $\ans{(2,\ 5)}$
\end{examplebox}

\begin{tryitbox}
\textbf{Try it:} \quad Find where $y = x^{2} - 2$ meets $y = x$.\\[2pt]
\rotatebox{180}{\footnotesize check: $\ans{(2,\ 2)}$ and $\ans{(-1,\ -1)}$}
\end{tryitbox}
