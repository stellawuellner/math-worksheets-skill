\sstitleblock{Integer Exponent Laws: Product and Quotient}

\begin{formulabox}\textbf{The one idea behind every rule below:} a power counts
equal factors, so $2^3 \cdot 2^4$ is three factors of $2$ set beside four more
--- seven in all. Every law here is bookkeeping on that count. Bases never
change; only the exponent moves.
\end{formulabox}

\skillheading{Skill 1 --- Product of powers: add the exponents}

\begin{formulabox}\textbf{Rule:}\quad $b^{\,m} \cdot b^{\,n} = b^{\,m+n}$
\quad (same base only)\par
Why: $m$ factors beside $n$ factors is $m+n$ factors.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Simplify $x^3 \cdot x^5$.
\step{The bases match and the powers are multiplied, so this is the product law
--- the factor counts add.}
\step{$x^{3+5} = \ans{x^8}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Simplify $m^5 \cdot m^4$.
\ \rotatebox{180}{\footnotesize check: $\ans{m^9}$}\end{tryitbox}

\skillheading{Skill 2 --- Quotient of powers: subtract the exponents}

\begin{formulabox}\textbf{Rule:}\quad $\dfrac{b^{\,m}}{b^{\,n}} = b^{\,m-n}$
\quad ($b \neq 0$)\par
Why: each factor below cancels one factor above, so the exponents subtract.
Coefficients divide separately: $\dfrac{12x^5}{4x^2} = 3x^3$.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Simplify $\dfrac{x^7}{x^2}$.
\step{Same base in a quotient, so cancelling pairs is what the subtraction
records --- top exponent minus bottom exponent.}
\step{$x^{7-2} = \ans{x^5}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Simplify $\dfrac{10^8}{10^5}$ as a power
of $10$.
\ \rotatebox{180}{\footnotesize check: $\ans{10^3}$}\end{tryitbox}

\skillheading{Skill 3 --- Zero and negative exponents}

\begin{formulabox}\textbf{Rules:}\quad $b^{\,0} = 1$ \quad and \quad
$b^{-n} = \dfrac{1}{b^{\,n}}$ \quad ($b \neq 0$)\par
Neither is arbitrary: subtracting equal exponents gives $b^{\,0}$ for an
expression that obviously equals $1$, and subtracting a larger exponent from a
smaller one leaves the extra factors in the denominator.
\end{formulabox}

\begin{examplebox}\textbf{Example:}\quad Simplify $\dfrac{5^3}{5^5}$.
\step{Same base, so subtract --- and keep the sign of the result instead of
flipping the subtraction to avoid it.}
\step{$5^{3-5} = \ans{5^{-2}}$, which is $\dfrac{1}{5^2} = \dfrac{1}{25}$}
\end{examplebox}

\begin{tryitbox}\textbf{Try it:}\quad Simplify $\dfrac{2^3}{2^6}$ as a power
of $2$ with a negative exponent.
\ \rotatebox{180}{\footnotesize check: $\ans{2^{-3}}$}\end{tryitbox}

\begin{watchoutbox}\textbf{(!) Watch out:} Adding exponents is not multiplying
them. $2^5 \cdot 2^3 = 2^8 = 256$, never $2^{15}$. And a negative exponent does
not make a negative number: $4^{-3}$ is a small positive fraction.
\end{watchoutbox}
